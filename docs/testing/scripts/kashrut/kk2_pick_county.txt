# Kashrut check 2: pick a test county in western Europe that produces cattle,
# held by an AI count of another faith, and remember it. Run once, then run
# kk3_measure.txt for its numbers.
# Console: run kk2_pick_county.txt   Output: logs/debug.log, lines starting KK2.
random_county = {
	limit = {
		title_province = {
			geographical_region = world_europe_west
			production_of_trade_good_type = { target = trade_good_type:trade_good_cattle value >= 1 }
		}
		holder ?= {
			is_ai = yes
			highest_held_title_tier = tier_county
			NOT = { faith = faith:rabbinic_faith }
		}
	}
	set_global_variable = { name = kk_test_county value = this }
	save_scope_as = kk_county
	holder = { save_scope_as = kk_holder }
	debug_log = "KK2 test county chosen (scopes below: kk_county, kk_holder). Open it in the county view."
	debug_log_scopes = yes
}
if = {
	limit = { NOT = { exists = global_var:kk_test_county } }
	debug_log = "KK2 no county found; widen the region in this file"
}
