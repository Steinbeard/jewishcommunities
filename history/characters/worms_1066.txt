# =====================================================================
# Isaac ben Eliezer ha-Levi — chief rabbi of Worms.
#
# Source of truth: docs/scenarios/worms-1066.md (this mod), §3 and §5.
# Scope note: this file defines ONLY the character. No title history,
# no government/title files, and no bookmark entry are created here —
# those are explicitly out of scope for this task (the "Kehillah of
# Worms" landless title is being designed separately and does not exist
# yet; nothing here references it).
#
# DOCUMENTED FACT (per the scenario doc's sourcing: Jewish Encyclopedia,
# Encyclopedia.com, corroborated by Yaakov ben Yakar's Wikipedia page):
#   - Became head of the Worms yeshiva and chief rabbi of Worms in 1064,
#     succeeding Yaakov ben Yakar (Rashi's teacher, who died that year).
#   - One of Rashi's own subsequent teachers at Worms.
#   - Died c. 1070 per Abraham Zacuto (Zunz gives a wider 1070-1096
#     range) — documented as active chief rabbi through this scenario's
#     1066.9.15 start date.
#   - Name variant: some sources transliterate the patronymic "Eleazar"
#     instead of "Eliezer" for the same person; "Eliezer" used here per
#     the scenario doc's choice.
#   - Faith: rabbinism (a faith of vanilla's judaism_religion). Culture:
#     ashkenazi (vanilla's heritage_israelite culture family). Dynasty:
#     HaLevi — his own documented Levite lineage marker, not invented.
#
# ESTIMATED FOR GAMEPLAY PURPOSES — NOT ATTESTED BY ANY SOURCE. Flagged
# explicitly so these are never mistaken for documented history later:
#   - Birth date, 1000.1.1: no source gives a birth year at all. Per
#     this mod's design discussion (recorded in the scenario doc),
#     an *older* estimated birth year was deliberately chosen (c. 1000,
#     rather than e.g. c. 1030) so that the real, documented ~1070
#     death reads in-game as the natural end of a long-lived elder
#     sage's career (mid-60s at the 1066.9.15 scenario start, ~70 at
#     death) rather than an abruptly short reign for the player. The
#     death date itself is unchanged either way — only the frame around
#     it differs.
#   - Death date, 1070.1.1: only the year 1070 is documented (and even
#     that is Zacuto's "c. 1070", with Zunz's wider 1070-1096 range
#     noted as an alternative). No source gives a day or month, so
#     1.1 is used as a placeholder. This mirrors vanilla's own practice
#     for real historical figures known only to the year — see e.g.
#     Bishop Siegfried of Worms himself (vanilla character ID 33134,
#     history/characters/franconian.txt), who vanilla also dates to a
#     bare "1070.1.1" death.
#   - Traits: not documented anywhere; chosen only to fit the profile of
#     a renowned rabbinic scholar and yeshiva head, using existing
#     vanilla trait vocabulary (see inline comments below for the
#     reasoning behind each one). Per the scenario doc's own
#     "consistency note" (§3), this is exactly how vanilla already
#     handles real historical religious figures whose personalities
#     aren't recorded (real Popes, real Caliphs, Siegfried himself) —
#     not a new or special risk particular to this character.
# =====================================================================
9000002 = {
	name = "Isaac"
	dynasty = 9000001
	religion = "rabbinism"
	culture = "ashkenazi"

	# Traits — all four are judgment calls (see header); none are
	# individually documented for Isaac, only chosen to fit his
	# documented role as chief rabbi / yeshiva head / teacher of Rashi:
	#  - education_learning_4: high-tier Learning education trait,
	#    fitting the head of the Worms yeshiva and a teacher whose own
	#    students included Rashi. (education_learning_5 was considered
	#    and rejected — it bundles +3 Intrigue, an unrelated schemer
	#    flavor that doesn't fit a rabbinic scholar.)
	#  - theologian: lifestyle trait for dedicated religious scholarship
	#    (Learning +3, increased piety gain) — a direct thematic match
	#    for "chief rabbi" / yeshiva head.
	#  - just, patient: these are two of Judaism's own three listed
	#    virtue traits (alongside "compassionate") in vanilla's
	#    common/religion/religions/00_judaism.txt. Picking from a faith's
	#    own canonical virtue list is a low-risk, thematically-grounded
	#    choice for a faith leader rather than inventing something
	#    unlisted.
	trait = education_learning_4
	trait = theologian
	trait = just
	trait = patient

	1000.1.1 = { # ESTIMATED — see header. Not an attested birth date.
		birth = "1000.1.1"
	}
	1070.1.1 = { # ESTIMATED day/month — year 1070 is documented (c.), see header.
		death = "1070.1.1"
	}
}
