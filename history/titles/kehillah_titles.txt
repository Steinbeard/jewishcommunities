# =====================================================================
# Title history for the Kehillah of Worms.
#
# Source of truth: docs/scenarios/worms-1066.md.
#
# The grant date is 1064.1.1, not the 1066.9.15 bookmark date, and that
# is deliberate. Isaac ben Eliezer ha-Levi became head of the Worms
# yeshiva and chief rabbi of Worms in 1064, succeeding Yaakov ben Yakar
# -- a documented fact, sourced in the scenario doc section 3 and
# already recorded in history/characters/worms_1066.txt. Starting his
# tenure on the real date means that by the time the player picks up the
# bookmark he has been leading the community for two and a half years,
# which is both accurate and better framing than a leader who took
# office the morning play begins.
#
# The character (9000002) and his dynasty (9000001) are defined in
# history/characters/worms_1066.txt and common/dynasties/ha_levi.txt.
# Nothing here restates their sourcing; see those files.
#
# No end date is set. On his documented c. 1070 death the title passes
# by kehillah_appointment_succession_law to whichever community figure
# scores highest, and the player continues as them -- that handoff is
# the point of spec section 5, so it is left to the mechanic rather than
# scripted to a named successor. (There is no documented successor to
# script anyway; the sources name Isaac's predecessor, not his heir.)
# =====================================================================

d_kehillah_worms = {
	1064.1.1 = {
		holder = 9000002

		# The government is set here rather than in the character file
		# because it belongs to the office, not the man. Whoever holds
		# this title leads a Kehillah.
		government = kehillah_government

		# kehillah_appointment_succession_law also has
		# should_start_with = government_has_flag = government_is_kehillah,
		# so this should already apply on its own. Stated explicitly
		# anyway: the succession rule is the single most important
		# thing about this title, and it should not depend on
		# should_start_with evaluating the way we expect during history
		# execution.
		succession_laws = {
			kehillah_appointment_succession_law
		}
	}
}
