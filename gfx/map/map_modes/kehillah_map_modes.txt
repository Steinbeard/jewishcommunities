# =====================================================================
# The Jewish Communities map mode.
#
# A SECOND file in gfx/map/map_modes/, alongside vanilla's
# map_modes.txt, on the CK3 wiki's word ("Creating Custom Map Modes")
# that the folder loads additively. Vanilla itself ships only the one
# file and no DLC adds another, so this is the first place in this
# repo that relies on the folder merging -- if the map mode key below
# is unknown to SetMapMode in a live game, this file not loading is the
# first suspect. Nothing vanilla is overridden either way.
#
# HOW IT WORKS. color_mode = baronies paints every barony in its own
# title's colour. That colour is rewritten at runtime by
# kehillah_paint_communities_map_effect (common/scripted_effects/
# kehillah_map_view_effects.txt) from the colour titles in common/
# landed_titles/kehillah_map_color_titles.txt: every barony to a
# neutral base, then each community's host county to the colour of
# that community's Standing band. The paint runs from the map view's
# own toggle button (gui/kehillah_community_map_view.gui) via a
# scripted_gui, immediately before it switches to this mode -- so the
# mode is only ever looked at freshly painted.
#
# The vanilla file's @-constants are file-scoped, so the gradient block
# below spells out the "data map" values by hand (map_modes.txt lines
# 97-107: flat fill, no edge, half after-lighting blend), matching the
# look of county_development / county_control / economy_buildings,
# which are the vanilla modes this most resembles. One zoom step, like
# those three -- the flat look reads the same at every distance.
#
# The hover description is evaluated in PROVINCE scope (vanilla's own
# court_languages block reads county.holder from it, map_modes.txt:
# 537-545); the trigger reads the province variable the paint sets
# (changed 2026-09-20 from a county-title variable to a province one, so
# the tooltip only fires on the ONE barony actually painted, matching
# the coloring -- see kehillah_paint_communities_map_effect's own header
# in common/scripted_effects/kehillah_map_view_effects.txt).
# =====================================================================

kehillah_communities_map = {
	color_mode = baronies
	small_map_names = counties
	large_map_names = counties
	selection = holding
	fill_in_impassable = no

	barony_description = {
		first_valid = {
			desc = {
				triggered_desc = {
					trigger = {
						has_variable = kehillah_map_community
					}
					desc = KEHILLAH_MAP_MODE_TOOLTIP_COMMUNITY
				}
				triggered_desc = {
					trigger = { always = yes }
					desc = KEHILLAH_MAP_MODE_TOOLTIP_NONE
				}
			}
		}
	}

	gradient_parameters = {
		zoom_step =					2

		gradient_alpha_inside = 	1
		gradient_alpha_outside = 	1
		gradient_width = 			0.4
		gradient_color_mult =		1
		edge_width = 				0
		edge_sharpness = 			0
		edge_alpha = 				1
		edge_color_mult = 			0.7
		before_lighting_blend =		1
		after_lighting_blend =		0.5
	}
}
