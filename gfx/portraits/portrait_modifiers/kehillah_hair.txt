# Applied after ordinary hair selection, before situational/religious hair.
# Vanilla's MENA straight pool contains the Western long side part and short
# cuts, but excludes male_hair_western_01 (the blunt bowl cut). Reuse it
# rather than replacing the game's whole hairstyle gene definition.
kehillah_jewish_hair = {
	usage = game
	selection_behavior = weighted_random
	priority = 1

	kehillah_jewish_western_straight = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = hairstyles
				template = mena_hairstyles_straight
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100
				exists = this
				is_female = no
				faith.religion = religion:judaism_religion
				portrait_western_clothing_trigger = yes
				OR = {
					has_gene = {
						category = gene_hair_type
						template = hair_straight
					}
					has_gene = {
						category = gene_hair_type
						template = hair_straight_thin_beard
					}
				}
			}
		}
	}
}
