# =====================================================================
# Kehillah Wave 3 manual QA harness. NOT content -- a fixed set of
# hidden, console-only events for driving pillar variables and
# inspecting otherwise-invisible state during live testing.
#
# Written 2026-09-07 alongside docs/testing/wave3-testing-runbook.md --
# read that file for the full test plan these events support.
#
# None of these are hooked to any on_action, pulse, or trigger, so they
# never fire on their own -- the only way to reach one is the debug
# console (requires -debug_mode), while playing the Kehillah leader:
#
#     event kehillah_debug.<number>
#
# All are `hidden = yes` (no popup, matches this mod's own
# jewish_holiday_events.txt precedent for hidden setup events) and all
# either write state via set_variable or report state via debug_log,
# which lands in debug.log (not error.log) in the game's logs folder --
# not to be confused with the debug.txt in vanilla's own
# events/test_events/, an unrelated file.
# =====================================================================

namespace = kehillah_debug

# Temporary/live-QA setup for the Found a Jewish Community decision. It
# gives an already-playable Kehillah leader a valid vanilla adventurer
# title, then swaps government; the actual test still clicks the real
# decision and runs kehillah_found_community_effect.
kehillah_debug.60 = {
	type = character_event
	hidden = yes
	immediate = {
		create_adventurer_title = {
			name = adventurer_name_001
			holder = root
		}
		add_realm_law = landless_adventurer_succession_law
		change_government = landless_adventurer_government
		debug_log = "kehillah_debug.60: prepared root as a vanilla landless adventurer for founding test"
	}
}

# Reports the title-tier and domicile invariants for either a game-start
# or newly founded Kehillah. Run again after a monthly validity tick.
kehillah_debug.61 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.61: community state"
		debug_log_scopes = yes
		if = { limit = { primary_title = { tier = tier_duchy } } debug_log = "kehillah_debug.61: primary title is duchy" }
		else = { debug_log = "kehillah_debug.61: WARNING primary title is NOT duchy" }
		if = { limit = { primary_title = { is_kehillah_title_trigger = yes } } debug_log = "kehillah_debug.61: primary title is registered community" }
		else = { debug_log = "kehillah_debug.61: WARNING primary title is NOT registered" }
		if = { limit = { has_government = kehillah_government } debug_log = "kehillah_debug.61: government is kehillah" }
		else = { debug_log = "kehillah_debug.61: WARNING government is NOT kehillah" }
		if = { limit = { exists = domicile } debug_log = "kehillah_debug.61: domicile exists" }
		else = { debug_log = "kehillah_debug.61: WARNING domicile is missing" }
		if = { limit = { has_realm_law = kehillah_appointment_succession_law } debug_log = "kehillah_debug.61: appointment succession law is active" }
		else = { debug_log = "kehillah_debug.61: WARNING appointment succession law is missing" }
	}
}

# ---------------------------------------------------------------------
# Inspection: reports current band membership, courtier capacity
# pressure, and whether the local county holder has an outstanding
# Kehillah loan -- none of which Take Stock shows directly. Check
# debug.log after firing, not the screen (nothing pops up, by design).
# ---------------------------------------------------------------------

kehillah_debug.30 = {
	type = character_event
	hidden = yes
	immediate = {
		primary_title = {
			# Bare `kehillah_var_prosperity` etc. isn't valid trigger syntax --
			# reading a scripted variable requires the `var:` prefix (see
			# kehillah_pillar_at_least_trigger in
			# common/scripted_triggers/kehillah_scripted_triggers.txt for the
			# same pattern already used correctly). Caught by ck3-tiger,
			# 2026-09-07 -- this debug event was never actually fired during
			# that session's live testing, so the bug had gone unexercised.
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Prosperity band = Legendary" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Prosperity band = Flourishing" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Prosperity band = Healthy" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Prosperity band = Strained" }
			else = { debug_log = "kehillah_debug.30: Prosperity band = Crisis" }

			if = { limit = { var:kehillah_var_stability >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Stability band = Legendary" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Stability band = Flourishing" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Stability band = Healthy" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Stability band = Strained" }
			else = { debug_log = "kehillah_debug.30: Stability band = Crisis" }

			if = { limit = { var:kehillah_var_greatness >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Greatness band = Legendary" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Greatness band = Flourishing" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Greatness band = Healthy" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Greatness band = Strained" }
			else = { debug_log = "kehillah_debug.30: Greatness band = Crisis" }
		}

		if = {
			limit = { kehillah_courtier_count_value > kehillah_courtier_capacity_value }
			debug_log = "kehillah_debug.30: courtier count EXCEEDS capacity"
		}
		else = {
			debug_log = "kehillah_debug.30: courtier count within capacity"
		}

		domicile.domicile_location.county.holder ?= {
			if = {
				limit = { exists = var:kehillah_loan_amount_owed }
				debug_log = "kehillah_debug.30: local county holder HAS an outstanding Kehillah loan"
			}
			else = {
				debug_log = "kehillah_debug.30: local county holder has NO outstanding Kehillah loan"
			}
		}
	}
}

# ---------------------------------------------------------------------
# Fast-forward an already-outstanding loan to one quarterly tick before
# its 6-year term (kehillah_loan_term_years), to reach the default path
# (kehillah_loan_default_prosperity_loss, in
# kehillah_scripted_effects.txt) without waiting 6 real years. Take the
# Extend a Loan decision first -- this only has anything to fast-forward
# if a loan already exists.
# ---------------------------------------------------------------------

kehillah_debug.40 = {
	type = character_event
	hidden = yes
	immediate = {
		domicile.domicile_location.county.holder ?= {
			if = {
				limit = { exists = var:kehillah_loan_amount_owed }
				set_variable = { name = kehillah_loan_years_elapsed value = 5.9 }
				debug_log = "kehillah_debug.40: loan fast-forwarded to 5.9 years elapsed (term is 6, ticks by 0.25/quarter -- next tick should default)"
			}
			else = {
				debug_log = "kehillah_debug.40: no outstanding loan to fast-forward"
			}
		}
	}
}

# V16 Host Charter -> pillar-baseline probe. This must be fired only
# AFTER a normal host charter already exists; a newly start_tributary'd

# spec/v18-norman-conquest-and-new-communities.md -- the real trigger
# (a change of English royal house) is a genuinely emergent, hard-to-
# force-honestly fact about a live game's own war/succession outcomes,
# so these three events call kehillah_norman_conquest_evaluate_effect
# (common/scripted_effects/kehillah_norman_conquest_effects.txt)
# DIRECTLY, on a fresh throwaway character created with the right
# dynasty/culture for each branch, rather than on the real title:
# k_england holder. This exercises the entire downstream mechanism
# (branch classification, weighted fire/decline roll, all four
# founding effects, pillar init, registry, network cache, library
# seed, map view, Jewish Settlement Policy, Host Charter, and the
# announcement event) end to end. It does NOT exercise the on_title_
# gain WRAPPER itself (common/on_action/kehillah_on_actions.txt's
# kehillah_norman_conquest_check_on_title_gain) -- that trigger is
# simple enough (three conditions, already ck3-tiger-validated) to
# read rather than force live, and this mod's own on_title_gain hook
# already fires correctly for every Kehillah succession elsewhere in
# this file's own precedent. The one thing genuinely NOT covered by
# either: a REAL cross-dynasty transfer of title:k_england during live
# play. See the runbook note at the end of this block for how to at
# least partially exercise the wrapper's own "still House Godwinson,
# don't consume the flag" guard without needing a full simulated
# Conquest.
#
# NO DUPLICATE-LOCATION GUARD, ON PURPOSE. Unlike the player-facing
# Found a Jewish Community decision (kehillah_found_community_
# requires_empty_location_tt), calling one of these twice in the same
# session founds a SECOND set of four Anglia communities at the same
# four counties -- the real production path only ever fires once per
# game (gated by global_var kehillah_norman_conquest_resolved), and
# this harness deliberately bypasses that gate for repeatability.

# without expecting duplicate titles.

# Reports every Norman-founded Anglia community currently registered,
# and each one's cached charter/policy state -- the fast way to confirm
# a .90/.91/.92 call actually finished cleanly, without hunting for four
# separate titles by hand in the UI. Uses var:kehillah_norman_founded (a
# persistent title variable set by kehillah_norman_conquest_finish_
# founding_effect), NOT a saved effect scope -- a scope like
# scope:kehillah_norman_london_title does not survive past the effect
# call that created it, so re-deriving it in a SEPARATE event (this one)
# would always read as unset; this was caught and fixed before ever
# reaching a live game.
kehillah_debug.93 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.93: Anglia founding report"
		set_variable = { name = kehillah_debug_norman_found_count value = 0 }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { has_variable = kehillah_norman_founded }
			root = {
				change_variable = { name = kehillah_debug_norman_found_count add = 1 }
			}
			debug_log = "kehillah_debug.93: found a Norman-founded Anglia title"
			debug_log_scopes = yes
			if = {
				limit = { has_variable = kehillah_charter_cache_construction }
				debug_log = "kehillah_debug.93: this title has a committed regional charter cache"
			}
			else = {
				debug_log = "kehillah_debug.93: WARNING this title has NO committed charter cache yet (may just need another day to settle)"
			}
			holder = {
				if = { limit = { is_tributary = yes } debug_log = "kehillah_debug.93: its leader is a tributary (Host Charter established)" }
				else = { debug_log = "kehillah_debug.93: WARNING its leader is NOT a tributary" }
				if = {
					limit = { exists = domicile.domicile_location.county.holder.top_liege.primary_title }
					domicile.domicile_location.county.holder.top_liege.primary_title = {
						if = { limit = { has_variable = kehillah_jewish_settlement_policy } debug_log = "kehillah_debug.93: its host realm has a Jewish Settlement Policy variable set" }
						else = { debug_log = "kehillah_debug.93: WARNING its host realm has NO Jewish Settlement Policy variable" }
					}
				}
			}
		}
		if = {
			limit = { var:kehillah_debug_norman_found_count = 0 }
			debug_log = "kehillah_debug.93: WARNING zero Norman-founded titles registered -- run .90/.91/.92 first"
		}
		else = {
			debug_log = "kehillah_debug.93: total Norman-founded titles found"
			debug_log_scopes = yes
		}
	}
}
