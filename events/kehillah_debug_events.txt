# =====================================================================
# Kehillah Wave 3 manual QA harness. NOT content -- a fixed set of
# hidden, console-only events for driving pillar variables and
# inspecting otherwise-invisible state during live testing.
#
# Written 2026-09-07 alongside docs/testing/wave3-testing-runbook.md --
# read that file for the full test plan these events support.
#
# None of these are hooked to any on_action, pulse, or trigger, so they
# never fire on their own -- the only way to reach one is the debug
# console (requires -debug_mode), while playing the Kehillah leader:
#
#     event kehillah_debug.<number>
#
# All are `hidden = yes` (no popup, matches this mod's own
# jewish_holiday_events.txt precedent for hidden setup events) and all
# either write state via set_variable or report state via debug_log,
# which lands in debug.log (not error.log) in the game's logs folder --
# not to be confused with the debug.txt in vanilla's own
# events/test_events/, an unrelated file.
# =====================================================================

namespace = kehillah_debug

# Temporary/live-QA setup for the Found a Jewish Community decision. It
# gives an already-playable Kehillah leader a valid vanilla adventurer
# title, then swaps government; the actual test still clicks the real
# decision and runs kehillah_found_community_effect.
kehillah_debug.60 = {
	type = character_event
	hidden = yes
	immediate = {
		create_adventurer_title = {
			name = adventurer_name_001
			holder = root
			save_scope_as = kehillah_debug_60_camp_title
		}
		add_realm_law = landless_adventurer_succession_law
		change_government = landless_adventurer_government
		debug_log = "kehillah_debug.60: prepared root as a vanilla landless adventurer for founding test"
	}
}

# Reports the title-tier and domicile invariants for either a game-start
# or newly founded Kehillah. Run again after a monthly validity tick.
kehillah_debug.61 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.61: community state"
		debug_log_scopes = yes
		if = { limit = { primary_title = { tier = tier_duchy } } debug_log = "kehillah_debug.61: primary title is duchy" }
		else = { debug_log = "kehillah_debug.61: WARNING primary title is NOT duchy" }
		if = { limit = { primary_title = { is_kehillah_title_trigger = yes } } debug_log = "kehillah_debug.61: primary title is registered community" }
		else = { debug_log = "kehillah_debug.61: WARNING primary title is NOT registered" }
		if = { limit = { has_government = kehillah_government } debug_log = "kehillah_debug.61: government is kehillah" }
		else = { debug_log = "kehillah_debug.61: WARNING government is NOT kehillah" }
		if = { limit = { exists = domicile } debug_log = "kehillah_debug.61: domicile exists" }
		else = { debug_log = "kehillah_debug.61: WARNING domicile is missing" }
		if = { limit = { has_realm_law = kehillah_appointment_succession_law } debug_log = "kehillah_debug.61: appointment succession law is active" }
		else = { debug_log = "kehillah_debug.61: WARNING appointment succession law is missing" }
	}
}

# ---------------------------------------------------------------------
# Inspection: reports current band membership, courtier capacity
# pressure, and whether the local county holder has an outstanding
# Kehillah loan -- none of which Take Stock shows directly. Check
# debug.log after firing, not the screen (nothing pops up, by design).
# ---------------------------------------------------------------------

kehillah_debug.30 = {
	type = character_event
	hidden = yes
	immediate = {
		primary_title = {
			# Bare `kehillah_var_prosperity` etc. isn't valid trigger syntax --
			# reading a scripted variable requires the `var:` prefix (see
			# kehillah_pillar_at_least_trigger in
			# common/scripted_triggers/kehillah_scripted_triggers.txt for the
			# same pattern already used correctly). Caught by ck3-tiger,
			# 2026-09-07 -- this debug event was never actually fired during
			# that session's live testing, so the bug had gone unexercised.
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Prosperity band = Legendary" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Prosperity band = Flourishing" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Prosperity band = Healthy" }
			else_if = { limit = { var:kehillah_var_prosperity >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Prosperity band = Strained" }
			else = { debug_log = "kehillah_debug.30: Prosperity band = Crisis" }

			if = { limit = { var:kehillah_var_stability >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Stability band = Legendary" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Stability band = Flourishing" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Stability band = Healthy" }
			else_if = { limit = { var:kehillah_var_stability >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Stability band = Strained" }
			else = { debug_log = "kehillah_debug.30: Stability band = Crisis" }

			if = { limit = { var:kehillah_var_greatness >= kehillah_band_legendary_threshold } debug_log = "kehillah_debug.30: Greatness band = Legendary" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_flourishing_threshold } debug_log = "kehillah_debug.30: Greatness band = Flourishing" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_healthy_threshold } debug_log = "kehillah_debug.30: Greatness band = Healthy" }
			else_if = { limit = { var:kehillah_var_greatness >= kehillah_band_strained_threshold } debug_log = "kehillah_debug.30: Greatness band = Strained" }
			else = { debug_log = "kehillah_debug.30: Greatness band = Crisis" }
		}

		if = {
			limit = { kehillah_courtier_count_value > kehillah_courtier_capacity_value }
			debug_log = "kehillah_debug.30: courtier count EXCEEDS capacity"
		}
		else = {
			debug_log = "kehillah_debug.30: courtier count within capacity"
		}

		domicile.domicile_location.county.holder ?= {
			if = {
				limit = { exists = var:kehillah_loan_amount_owed }
				debug_log = "kehillah_debug.30: local county holder HAS an outstanding Kehillah loan"
			}
			else = {
				debug_log = "kehillah_debug.30: local county holder has NO outstanding Kehillah loan"
			}
		}
	}
}

# ---------------------------------------------------------------------
# Fast-forward an already-outstanding loan to one quarterly tick before
# its 6-year term (kehillah_loan_term_years), to reach the default path
# (kehillah_loan_default_prosperity_loss, in
# kehillah_scripted_effects.txt) without waiting 6 real years. Take the
# Extend a Loan decision first -- this only has anything to fast-forward
# if a loan already exists.
# ---------------------------------------------------------------------

kehillah_debug.40 = {
	type = character_event
	hidden = yes
	immediate = {
		domicile.domicile_location.county.holder ?= {
			if = {
				limit = { exists = var:kehillah_loan_amount_owed }
				set_variable = { name = kehillah_loan_years_elapsed value = 5.9 }
				debug_log = "kehillah_debug.40: loan fast-forwarded to 5.9 years elapsed (term is 6, ticks by 0.25/quarter -- next tick should default)"
			}
			else = {
				debug_log = "kehillah_debug.40: no outstanding loan to fast-forward"
			}
		}
	}
}

# V16 Host Charter -> pillar-baseline probe. This must be fired only
# AFTER a normal host charter already exists; a newly start_tributary'd
# contract is not registered for tributary_contract_set_obligation_level

# the existing quarterly debug probe. See v16 §8 test 5.

	type = character_event

				type = kehillah_charter_moneylending_rights

				type = kehillah_charter_quarter_rights
				level = kehillah_quarter_walled

			debug_log = "kehillah_debug.71: no live kehillah_host_charter; nothing changed"
		}
	}
}

# Silversmiths Quarrel restitution clamp (BLOCKERS 2026-09-25). The bug
# was an unaffordable negative add_gold on an accused with less gold than
# medium_gold_value. The fix computes the transfer as
# min(max(gold,0), medium_gold_value) via save_scope_value_as and uses
# that one value for both the deduction and the accuser's credit.
#
# This probe reproduces the FIXED ARITHMETIC AND THE add_gold CALLS on a
# deliberately poor accused. It does NOT re-run
# kehillah_bet_din_silversmiths_resolution_effect itself: that effect
# reads involved_activity variables and is only reachable inside a real
# hosted docket. What was broken was the clamp and the add_gold pair, and
# that is exactly what is exercised here -- the activity-variable
# plumbing around it was never implicated and is unchanged.
#
# EXPECT: accused ends at 0 gold, accuser is credited exactly 3, and
# error.log gains NO "Trying to add add_gold with negative value" line.
kehillah_debug.97 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.97: restitution clamp probe -- START"
		create_character = {
			culture = root.culture
			faith = root.faith
			rite = root.rite
			age = 40
			gender_female_chance = 0
			location = root.location
			save_scope_as = kehillah_debug_probe_accuser
		}
		create_character = {
			culture = root.culture
			faith = root.faith
			rite = root.rite
			age = 40
			gender_female_chance = 0
			location = root.location
			save_scope_as = kehillah_debug_probe_accused
		}
		# Normalise both purses to a known state. add_gold with a negative
		# value exactly equal to the character own gold is always
		# affordable, so this cannot itself trip the bug under test.
		scope:kehillah_debug_probe_accuser = { add_gold = { value = gold multiply = -1 } }
		scope:kehillah_debug_probe_accused = {
			add_gold = { value = gold multiply = -1 }
			add_gold = 3
		}

			save_scope_value_as = {
				name = kehillah_debug_probe_restitution
				value = {

# .103 -- ISOLATING THE add_gold CLAMP FAILURE (added 2026-09-25, after
# the first Sukkot live run).
#
# WHAT .97 ACTUALLY FOUND, live: with the accused holding 3 gold, the
# clamp computed restitution = 3.00 exactly (logged, and the accuser was
# credited exactly 3), and yet `add_gold = { value = <that scope value>
# multiply = -1 }` STILL threw "Trying to add add_gold with negative
# value", and the accused did not end at 0. So the deduction was refused
# even though, by every number the probe could read, it was affordable.
#
# That is not what the 2026-09-25 source fix assumed. It means the
# engine's affordability check for a negative add_gold is NOT simply
# |value| <= gold, and the same shape is what currently ships inside
# kehillah_bet_din_silversmiths_resolution_effect -- so the "resolved in
# source" claim in BLOCKERS.md is in doubt, not confirmed.
#

#   A. what gold a freshly created character actually has at each step
#      (probe characters may simply not behave like the real litigants
#      the shipped effect operates on -- that alone would explain
#      everything and would mean the shipped fix is fine)
#   B. whether the SNAPSHOT shape (save_scope_value_as, then deduct that
#      value) fails where the SELF-REFERENTIAL shape (deduct `gold`
#      computed at deduction time) succeeds
#
# If B is the difference, the self-referential shape is the fix and it
# is provably always affordable. If A shows the probe characters are the
# anomaly, the shipped effect needs re-testing on real litigants in a
# real docket instead, and .97/.103 should be retired as misleading.
# =====================================================================
kehillah_debug.103 = {

# =====================================================================
# .108 -- THE RESTITUTION TRANSFER, VERIFYING THE 2026-09-26 FIX.
#
# HISTORY, so this is not re-litigated a fourth time. The Silversmiths'
# Quarrel resolution moved restitution from the accused to the accuser
# with a hand-rolled pair of add_gold calls, one of them negative. That
# threw a real error in the 2026-09-24 live docket. Three probes and two
# fixes chased it:
#   .97  reproduced the error with the "fixed" clamp still in place.
#   .103 ruled out both standing explanations -- the probe characters
#        held gold exactly as script read them, and the self-referential
#        shape failed identically to the snapshot shape.
#   .108 (the previous, now-replaced version) was written to test a
#        same-tick-commitment hypothesis, and answered the question by
#        accident instead: its own three LITERAL negative add_gold calls
#        were rejected at SCRIPT LOAD, before it ever ran. See commit
#        fe8ba31 for that version and the log evidence.
#
# THE ACTUAL ANSWER, from the engine's own generated logs/effects.log:
# add_gold "adds gold to a character" and is additive only. It refuses a
# literal negative at load validation and a computed negative at runtime.
# Gold is removed with remove_short_term_gold, and TRANSFERRED with
# pay_short_term_gold = { target = X gold = Y }, which effects.log
# describes as "the scope character pays gold to the target character."
# Vanilla uses `add_gold = -` exactly zero times.
#
# WHAT THIS PROBE NOW DOES: runs the shipped effect's exact new shape --
# one pay_short_term_gold under the same min/max clamp -- on two probe
# characters, in the one configuration the original bug needed, a LOW-GOLD
# accused who cannot cover the full medium_gold_value award. Both purses
# are reported before and after. PASS means the accuser gained precisely
# what the accused lost, the accused did not go negative, and error.log
# gains no add_gold or pay_short_term_gold line at all.
#
# This is a shape test, not a substitute for a real docket: it exercises
# the transfer primitive under the clamp, not kehillah_bet_din.0051's
# option-to-direction wiring. A genuine Silversmiths' case is still the
# only end-to-end confirmation.
# =====================================================================

kehillah_debug.108 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.108: restitution transfer (pay_short_term_gold) -- START"

		create_character = {
			culture = root.culture
			faith = root.faith
			rite = root.rite
			age = 40
			gender_female_chance = 0
			location = root.location
			save_scope_as = kehillah_debug_accuser
		}
		create_character = {
			culture = root.culture
			faith = root.faith
			rite = root.rite
			age = 40
			gender_female_chance = 0
			location = root.location
			save_scope_as = kehillah_debug_accused
		}

		# Normalise both purses, then give the accused DELIBERATELY TOO
		# LITTLE -- 7 gold against an award capped at medium_gold_value
		# (50 in vanilla's own script_values). This is the exact
		# configuration that errored in the 2026-09-24 live docket.
		scope:kehillah_debug_accuser = { remove_short_term_gold = gold }
		scope:kehillah_debug_accused = {
			remove_short_term_gold = gold
			add_gold = 7
		}

		scope:kehillah_debug_accuser = {
			save_scope_value_as = { name = kehillah_debug_accuser_before value = gold }
		}
		scope:kehillah_debug_accused = {
			save_scope_value_as = { name = kehillah_debug_accused_before value = gold }

			# The shipped effect's exact shape, clamp and all.
			save_scope_value_as = {
				name = kehillah_debug_restitution
				value = {
					value = gold
					min = 0
					max = medium_gold_value
				}
			}
			pay_short_term_gold = {
				target = scope:kehillah_debug_accuser
				gold = scope:kehillah_debug_restitution
			}
			save_scope_value_as = { name = kehillah_debug_accused_after value = gold }
		}
		scope:kehillah_debug_accuser = {
			save_scope_value_as = { name = kehillah_debug_accuser_after value = gold }
		}

		debug_log = "kehillah_debug.108: accused started with 7, award capped at the accused's purse. Scope values below: *_before, *_after, and kehillah_debug_restitution."
		debug_log_scopes = yes

		# The clamp should have set restitution to 7, not medium_gold_value.

#   # <CK3 user dir>\run\probe.txt

# then `run probe.txt` from the console. Same mechanism the shim guide's
# "run <file>.txt + debug_log" section describes.
#
# The load-bearing line is the CONTRADICTION check at the end: all four
# consumer sites now read kehillah_has_acting_chief_rabbi_trigger, so

kehillah_debug.115 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.115: Chief Rabbi state (read-only) -- START"

		save_scope_value_as = { name = kehillah_debug_115_learning value = learning }
		save_scope_value_as = { name = kehillah_debug_115_threshold value = kehillah_serve_as_rabbi_learning_threshold }

			limit = { government_has_flag = government_is_kehillah }

		}
		else = {
			debug_log = "kehillah_debug.115: this character is NOT a Kehillah leader -- the rest is reported anyway, but read it with that in mind"
		}

		if = { limit = { has_trait = kehillah_rabbi_trait } debug_log = "kehillah_debug.115: HAS the rabbi trait" }
		else = { debug_log = "kehillah_debug.115: does NOT have the rabbi trait" }

			limit = { kehillah_can_serve_as_own_rabbi_trigger = yes }
			debug_log = "kehillah_debug.115: COULD serve as own rabbi (trait, gender gate and Learning threshold all satisfied)"
		}
		else = {
			debug_log = "kehillah_debug.115: could NOT serve as own rabbi -- compare the learning and threshold figures in the scope dump"
		}

		if = { limit = { has_character_flag = kehillah_serving_as_own_rabbi } debug_log = "kehillah_debug.115: TOGGLE ON serving as the community's own rabbi" }
		else = { debug_log = "kehillah_debug.115: TOGGLE OFF not serving as own rabbi" }

		if = { limit = { kehillah_has_acting_chief_rabbi_trigger = yes } debug_log = "kehillah_debug.115: ACTING RABBI YES the shared trigger reports an acting chief rabbi" }

		if = {

				}
			}
			debug_log = "kehillah_debug.115: CONTRADICTION the shared trigger says no while the seat or the toggle says yes -- report this, it breaks all four consumer sites at once"
		}

		debug_log_scopes = yes
		debug_log = "kehillah_debug.115: Chief Rabbi state (read-only) -- END"
	}

# kehillah_debug.119 -- S4b: exercise the REPLACEMENT path.
#
# kehillah_serve_as_rabbi_decision can now take the office off a sitting
# Chief Rabbi (kehillah_replace_appointed_rabbi_effect). Reaching that in
# normal play needs an appointed rabbi AND a leader who qualifies to
# replace him, which is a fair amount of setup; this does the whole thing
# and asserts the result.
#
# NOT read-only. It appoints, then dismisses, then leaves the leader
# holding the office. Run it on a throwaway save.
#
# WHY IT APPOINTS ITS OWN RABBI FIRST rather than requiring one to be
# there: the assertion that matters is "the seat is empty and the toggle is

		if = {
			limit = { employs_court_position = chief_rabbi_kehillah_position }
			debug_log = "kehillah_debug.119: SETUP a chief rabbi is already employed -- using him"
		}

			}

			limit = { employs_court_position = chief_rabbi_kehillah_position }
			debug_log = "kehillah_debug.119: SETUP OK the seat is filled, running the replacement"

			# The decision own effect body, in order.

			add_character_flag = kehillah_serving_as_own_rabbi

			}
			else = {
				debug_log = "kehillah_debug.119: TOGGLE EFFECTIVE FAIL the flag is set but the trigger still says no -- the revocation did not land before the flag"
			}

			if = {
				limit = {
					exists = scope:kehillah_deposed_rabbi
					scope:kehillah_deposed_rabbi = {
						has_opinion_modifier = {
							modifier = revoked_court_position_opinion
							target = root
						}
					}
				}
				debug_log = "kehillah_debug.119: OPINION PASS the deposed rabbi resents the leader for it"
			}
			else = {
				debug_log = "kehillah_debug.119: OPINION FAIL no revoked_court_position_opinion on the deposed rabbi"
			}

			# S4c: the trait is ordination, not a job title. This path is
			# the one that used to strip it (the office's three teardown
			# hooks), so a deposed rabbi who still holds kehillah_rabbi_trait
			# here is S4c's whole claim, observed on the site that broke it.

				limit = {
					exists = scope:kehillah_deposed_rabbi
					scope:kehillah_deposed_rabbi = { has_trait = kehillah_rabbi_trait }
				}

	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.121: S4c Beit Midrash ungating -- START"

					remove_domicile_building = kehillah_beit_midrash_03

					limit = { has_domicile_building = kehillah_beit_midrash_02 }
					remove_domicile_building = kehillah_beit_midrash_02

					limit = { has_domicile_building = kehillah_beit_midrash_01 }
					remove_domicile_building = kehillah_beit_midrash_01
					debug_log = "kehillah_debug.121: removed kehillah_beit_midrash_01"
				}

				# Positive control on the demolition. Without this line a PASS
				# below would be satisfied by a probe that quietly removed
				# nothing, which is the failure mode that makes "no errors"
				# readings worthless.
				if = {
					limit = { has_domicile_parameter = kehillah_unlocks_chief_rabbi }
					debug_log = "kehillah_debug.121: STRIP FAIL the quarter still has kehillah_unlocks_chief_rabbi -- a Beit Midrash tier survived, so nothing below tests the ungating"
				}
				else = {
					debug_log = "kehillah_debug.121: STRIP PASS the quarter no longer has kehillah_unlocks_chief_rabbi"
				}
			}
		}
		else = {
			debug_log = "kehillah_debug.121: ABORT no domicile on this character -- fire this on a community leader"
		}

		# Clear the two states that would refuse an appointment for reasons
		# that have nothing to do with the building, so a FAIL below can only
		# mean the gate.
		if = {
			limit = { kehillah_has_taken_up_rabbinate_trigger = yes }

			limit = { employs_court_position = chief_rabbi_kehillah_position }
			debug_log = "kehillah_debug.121: SETUP the seat was already filled -- revoking first so the appointment below is a real one"
			kehillah_replace_appointed_rabbi_effect = yes
			kehillah_end_serving_as_own_rabbi_effect = yes
		}

		random_courtier = {
			limit = {
				is_adult = yes
				faith = root.faith
				kehillah_leadership_gender_eligible_trigger = yes
				NOT = { this = root }
			}
			save_scope_as = kehillah_debug_121_appointee
		}

		if = {
			limit = { exists = scope:kehillah_debug_121_appointee }
			appoint_court_position = {
				recipient = scope:kehillah_debug_121_appointee

			if = {
				limit = { employs_court_position = chief_rabbi_kehillah_position }
				debug_log = "kehillah_debug.121: UNGATED PASS a Chief Rabbi was seated in a quarter with no Beit Midrash"
			}
			else = {
				debug_log = "kehillah_debug.121: UNGATED FAIL the appointment did not take -- something still gates the office on the building, or on a state this probe did not clear"
			}

			if = {
				limit = { scope:kehillah_debug_121_appointee = { has_trait = kehillah_rabbi_trait } }
				debug_log = "kehillah_debug.121: SEMICHA GRANTED PASS the appointee received kehillah_rabbi_trait on taking the office"

			else = {
				debug_log = "kehillah_debug.121: SEMICHA GRANTED FAIL on_court_position_received did not grant the trait"
			}

			if = {
				limit = { kehillah_has_acting_chief_rabbi_trigger = yes }
				debug_log = "kehillah_debug.121: PRESIDE GATE PASS the community can convene a Bet Din with no study hall -- which is the reason S4c matters, not a side effect"
			}
			else = {

			}
		}
		else = {
			debug_log = "kehillah_debug.121: ABORT no eligible courtier to appoint -- nothing below was tested"
		}

		debug_log_scopes = yes
		debug_log = "kehillah_debug.121: S4c Beit Midrash ungating -- END (now let a month pass and re-run kehillah_debug.118)"
	}

# ---------------------------------------------------------------------
# kehillah_debug.124 -- S6: the COMPLETION path, end to end, with its own
# assertions.
#
# Sets the daughter-community goal, pushes the credit marker the founding
# effect would have pushed, and then calls the same
# kehillah_check_community_goal_effect the quarterly pulse calls -- so
# what is exercised is the real completion path and not a shortcut into
# the reward.
#
# THE DAUGHTER GOAL IS USED DELIBERATELY. It is the only one of the four
# whose condition can be satisfied by a single set_variable, so the test
# does not have to fake a building, a pillar or a charter rung and then
# argue about whether the fake was faithful. Its reward pillar is
# Greatness, the same as the yeshiva goal's.
#
# THE ARITHMETIC IS CHECKED, not just the fact of a change: Greatness must
# land on exactly the pre-value plus kehillah_goal_reward_pillar, and the
# legacy counter on exactly its pre-value plus one. A reward applied twice
# by some double-call would pass a "did it go up" test and fails this one.
#
# Fire it on the community LEADER, on a community with NO goal running.
# ---------------------------------------------------------------------
kehillah_debug.124 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.124: S6 completion path -- START"

		if = {
			limit = { kehillah_has_active_goal_trigger = yes }
			debug_log = "kehillah_debug.124: SETUP WARNING a goal was already running and is being overwritten -- the counter assertion still holds, the pre-existing goal is simply discarded"
			primary_title = { kehillah_clear_community_goal_effect = yes }
		}

		if = {
			limit = { primary_title = { NOT = { has_variable = kehillah_var_goals_completed } } }
			primary_title = { set_variable = { name = kehillah_var_goals_completed value = 0 } }
		}

		save_scope_value_as = { name = kehillah_debug_124_greatness_before value = primary_title.var:kehillah_var_greatness }
		save_scope_value_as = { name = kehillah_debug_124_completed_before value = primary_title.var:kehillah_var_goals_completed }
		save_scope_value_as = {
			name = kehillah_debug_124_greatness_expected
			value = {
				value = primary_title.var:kehillah_var_greatness
				add = kehillah_goal_reward_pillar
			}
		}
		save_scope_value_as = {
			name = kehillah_debug_124_completed_expected
			value = {
				value = primary_title.var:kehillah_var_goals_completed
				add = 1
			}
		}

		kehillah_set_community_goal_effect = { GOAL = flag:daughter }

		# What kehillah_goal_credit_daughter_community_effect would have
		# stamped from inside the founding effect. Set directly here so the
		# test does not need a second character to actually found anything.
		primary_title = { set_variable = { name = kehillah_var_goal_daughter_credit value = 1 } }

		if = {
			limit = { kehillah_goal_is_met_trigger = yes }
			debug_log = "kehillah_debug.124: PRE-CHECK PASS the dispatcher sees the daughter goal as met"
		}
		else = {
			debug_log = "kehillah_debug.124: PRE-CHECK FAIL the dispatcher does NOT see the daughter goal as met -- nothing below will complete"
		}

		# The real path.
		kehillah_check_community_goal_effect = yes

		if = {
			limit = { NOT = { kehillah_has_active_goal_trigger = yes } }
			debug_log = "kehillah_debug.124: CLEARED PASS the goal is no longer running"
		}
		else = {
			debug_log = "kehillah_debug.124: CLEARED FAIL the goal is still set after the check"
		}

		if = {
			limit = { primary_title = { var:kehillah_var_greatness = scope:kehillah_debug_124_greatness_expected } }
			debug_log = "kehillah_debug.124: REWARD PASS Greatness rose by exactly the goal reward"
		}
		else = {
			debug_log = "kehillah_debug.124: REWARD FAIL Greatness is not pre-value plus the goal reward -- compare the before/expected figures in the dump"
		}

		if = {
			limit = { primary_title = { var:kehillah_var_goals_completed = scope:kehillah_debug_124_completed_expected } }
			debug_log = "kehillah_debug.124: LEGACY PASS the completed-goals counter rose by exactly one"
		}
		else = {
			debug_log = "kehillah_debug.124: LEGACY FAIL the completed-goals counter did not rise by exactly one"
		}

		if = {
			limit = { primary_title = { NOT = { has_variable = kehillah_var_goal_daughter_credit } } }
			debug_log = "kehillah_debug.124: TEARDOWN PASS the daughter credit was cleared, so it cannot silently complete the NEXT goal too"
		}
		else = {
			debug_log = "kehillah_debug.124: TEARDOWN FAIL the daughter credit survived the completion"
		}

		if = {

# ---------------------------------------------------------------------
# kehillah_debug.125 -- S6: the FAILURE path, end to end.
#
# Sets the yeshiva goal, then removes kehillah_var_goal_timer to stand in
# for the decade running out, then calls the real quarterly check. Removing
# the timer is not a cheat around the mechanism: the engine's own timeout
# does exactly this and nothing else, which is why
# kehillah_goal_deadline_passed_trigger reads "goal set, timer absent".
#
# ASSERTS THE COUNTER DOES NOT MOVE as well as that Stability falls by
# exactly the penalty. A failure that quietly incremented the legacy
# counter would be a worse bug than one that applied the wrong number, and
# only an explicit assertion catches it.
#
# Fire it on the community LEADER.
# ---------------------------------------------------------------------
kehillah_debug.125 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_debug.125: S6 failure path -- START"

		if = {
			limit = { kehillah_has_active_goal_trigger = yes }
			debug_log = "kehillah_debug.125: SETUP WARNING a goal was already running and is being overwritten"
			primary_title = { kehillah_clear_community_goal_effect = yes }
		}

		if = {
			limit = { primary_title = { NOT = { has_variable = kehillah_var_goals_completed } } }
			primary_title = { set_variable = { name = kehillah_var_goals_completed value = 0 } }
		}

		save_scope_value_as = { name = kehillah_debug_125_stability_before value = primary_title.var:kehillah_var_stability }
		save_scope_value_as = { name = kehillah_debug_125_completed_before value = primary_title.var:kehillah_var_goals_completed }
		save_scope_value_as = {
			name = kehillah_debug_125_stability_expected
			value = {
				value = primary_title.var:kehillah_var_stability
				subtract = kehillah_goal_failure_stability_loss
			}
		}

		kehillah_set_community_goal_effect = { GOAL = flag:yeshiva }
		primary_title = { remove_variable = kehillah_var_goal_timer }

		if = {
			limit = { kehillah_goal_deadline_passed_trigger = yes }
			debug_log = "kehillah_debug.125: PRE-CHECK PASS the expiry trigger reads this as an expired decade"
		}
		else = {
			debug_log = "kehillah_debug.125: PRE-CHECK FAIL the expiry trigger does not read this as expired -- nothing below will fail the goal"
		}

		kehillah_check_community_goal_effect = yes

		if = {
			limit = { NOT = { kehillah_has_active_goal_trigger = yes } }
			debug_log = "kehillah_debug.125: CLEARED PASS the goal is no longer running"
		}
		else = {
			debug_log = "kehillah_debug.125: CLEARED FAIL the goal survived its own deadline"
		}

		if = {
			limit = { primary_title = { var:kehillah_var_stability = scope:kehillah_debug_125_stability_expected } }
			debug_log = "kehillah_debug.125: PENALTY PASS Stability fell by exactly the failure penalty"
		}
		else = {
			debug_log = "kehillah_debug.125: PENALTY FAIL Stability is not pre-value minus the failure penalty"
		}

		if = {
			limit = { primary_title = { var:kehillah_var_goals_completed = scope:kehillah_debug_125_completed_before } }
			debug_log = "kehillah_debug.125: LEGACY PASS a failed goal did not touch the completed-goals counter"
		}
		else = {
			debug_log = "kehillah_debug.125: LEGACY FAIL a failed goal moved the completed-goals counter"
		}

		save_scope_value_as = { name = kehillah_debug_125_stability_after value = primary_title.var:kehillah_var_stability }

		debug_log_scopes = yes
		debug_log = "kehillah_debug.125: S6 failure path -- END"
	}
}

	}
}
