# Console-only probes for V50. Fire on any character; they change no state.
# event kehillah_history_debug.1: authored Kairouan loss at this bookmark.
# event kehillah_history_debug.2: root's current community place record.
namespace = kehillah_history_debug

kehillah_history_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_history_debug.1: Kairouan place memory"
		title:c_kairwan = {
			if = { limit = { has_variable = kehillah_history_original_year } debug_log = "kehillah_history_debug.1: ORIGINAL PASS" }
			else = { debug_log = "kehillah_history_debug.1: ORIGINAL MISSING" }
			if = { limit = { var:kehillah_history_original_year = 670 } debug_log = "kehillah_history_debug.1: YEAR 670 PASS" }
			else = { debug_log = "kehillah_history_debug.1: YEAR 670 FAIL" }
			if = { limit = { has_variable = kehillah_history_last_loss_year } debug_log = "kehillah_history_debug.1: LOSS RECORDED" }
			else = { debug_log = "kehillah_history_debug.1: NO LOSS AT THIS BOOKMARK" }
			if = { limit = { has_variable = kehillah_history_last_loss_year var:kehillah_history_last_loss_year = 1057 } debug_log = "kehillah_history_debug.1: LOSS YEAR 1057 PASS" }
			debug_log_scopes = yes
		}
	}
}

kehillah_history_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { primary_title = { has_variable = kehillah_history_county } }
			primary_title.var:kehillah_history_county = {
				if = { limit = { has_variable = kehillah_history_original_year } debug_log = "kehillah_history_debug.2: ORIGINAL PASS" }
				else = { debug_log = "kehillah_history_debug.2: ORIGINAL MISSING" }
				if = { limit = { var:kehillah_history_original_year = 960 } debug_log = "kehillah_history_debug.2: COUNTY YEAR 960 PASS" }
				else = { debug_log = "kehillah_history_debug.2: COUNTY YEAR 960 FAIL" }
				if = { limit = { has_variable = kehillah_history_last_loss_year } debug_log = "kehillah_history_debug.2: LOSS RECORDED" }
				if = { limit = { has_variable = kehillah_history_last_refounded_year } debug_log = "kehillah_history_debug.2: REFOUNDING RECORDED" }
				debug_log_scopes = yes
			}
			primary_title = {
				if = { limit = { var:kehillah_founded_year = 960 } debug_log = "kehillah_history_debug.2: TITLE YEAR 960 PASS" }
				else = { debug_log = "kehillah_history_debug.2: TITLE YEAR 960 FAIL" }
			}
		}
		else = { debug_log = "kehillah_history_debug.2: current title has no place record" }
	}
}
