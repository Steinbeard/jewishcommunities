# Hidden: the founding itself, fired by .0002's options and by the AI's
# decision. Not in the option or decision effect, because the engine
# dry-runs those to build their tooltips -- and even inside hidden_effect
# that dry run reached the not-yet-created title (live 2026-10-13: ~70-
# 100k null landed_title errors and a stall when .0002 opened). A
# triggered event's immediate is not part of the tooltip. Saved scopes
# kehillah_settlers_leader and kehillah_founding_county carry over.
kehillah_settlers.0003 = {
	type = character_event
	hidden = yes
	trigger = {
		exists = scope:kehillah_settlers_leader
		exists = scope:kehillah_founding_county
	}
	immediate = {
		kehillah_settlers_go_effect = yes
	}
}

# Hidden: the AI's whole choice (destination and leader), then .0003.
kehillah_settlers.0004 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_settlers_pick_effect = yes
		scope:kehillah_settlers_dest_1 ?= { save_scope_as = kehillah_founding_county }
		if = {
			limit = { exists = scope:kehillah_settlers_lead_baal }
			scope:kehillah_settlers_lead_baal = { save_scope_as = kehillah_settlers_leader }
		}
		else_if = {
			limit = { exists = scope:kehillah_settlers_lead_rabbi }
			scope:kehillah_settlers_lead_rabbi = { save_scope_as = kehillah_settlers_leader }
		}
		else = {
			scope:kehillah_settlers_lead_kin ?= { save_scope_as = kehillah_settlers_leader }
		}
		if = {
			limit = {
				exists = scope:kehillah_founding_county
				exists = scope:kehillah_settlers_leader
			}
			kehillah_settlers_go_effect = yes
		}
	}
}
