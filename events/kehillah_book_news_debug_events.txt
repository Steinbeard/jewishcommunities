# Console-only preview of generated ordinary-work news. Run as a Kehillah
# leader with another registered community available:
#   event kehillah_book_news_debug.1
#   event kehillah_book_news_debug.2
# The event sends real kehillah_book.0007 news one day later. It does not
# create an artifact, change a library, or consume the writing decision.
namespace = kehillah_book_news_debug

kehillah_book_news_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		set_variable = { name = kehillah_book_work value = flag:parsh_ladder }
		save_scope_as = kehillah_book_author
		debug_log = "kehillah_book_news_debug.1: set A Ladder Set on Earth and sent preview news"
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { NOT = { this = root.primary_title } exists = holder }
			holder = { trigger_event = { id = kehillah_book.0007 days = 1 } }
		}
	}
}
