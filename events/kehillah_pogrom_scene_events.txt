namespace = kehillah_pogrom

# =====================================================================
# V41 step 1 -- the pogrom unfolding (docs/spec/v41-danger-incidents-and-
# the-pogrom-unfolding.md section 3), 2026-10-14. Players only: the AI's
# communities go straight to kehillah_pogrom_effect.
#
# Current dispatcher: one to four encounters by threat grade, from six context-gated families,
# a day apart. Street, fire, forced conversion, rescue, escape and castle
# scenes share a dispatcher; no family repeats within an attack and the
# next attack cannot open with its predecessor's opening family.
# Each choice leaves a flag on the community's title (kehillah_scene_*),
# read by kehillah_pogrom_effect: who dies, who is baptized, what is lost,
# and the leader's own fate -- killed, taken for ransom, or baptized.
# Then the report (kehillah_mob.0010).
#
# Saved scopes, set by kehillah_pogrom_scenes_start_effect and refreshed
# here: kehillah_mob_leader (the preacher), kehillah_pogrom_neighbor (a
# Christian neighbour), kehillah_mob_town.
# =====================================================================

scripted_effect kehillah_pogrom_scene_scopes_effect = {
	var:kehillah_mob_ref ?= { save_scope_as = kehillah_mob_leader }
	var:kehillah_pogrom_neighbor_ref ?= { save_scope_as = kehillah_pogrom_neighbor }
	domicile.domicile_location ?= { save_scope_as = kehillah_mob_town }
	cp:councillor_marshal ?= { save_scope_as = kehillah_pogrom_watchman }
	random_courtier = {
		limit = {
			is_alive = yes
			is_adult = yes
			is_jewish_character_trigger = yes
		}
		save_scope_as = kehillah_pogrom_someone
	}
}

scripted_trigger kehillah_pogrom_scene_valid_trigger = {
 OR = { primary_title = { NOT = { has_variable = kehillah_crisis_stage } } primary_title.var:kehillah_crisis_stage = 2 }
	exists = var:kehillah_mob_ref
	var:kehillah_mob_ref = { is_alive = yes }
}

# V43 slice 0 (2026-10-14): the silver did not come. A player's leader,
# held with the captives, chooses: refuse and die, yield, or run.
# ---------------------------------------------------------------------
kehillah_pogrom.0040 = {
	type = character_event
	title = kehillah_pogrom.0040.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_mob_is_muslim_trigger = yes }
				desc = kehillah_pogrom.0040.desc_islam
			}
			desc = kehillah_pogrom.0040.desc
		}
	}
	theme = dread
	left_portrait = {
		character = root
		animation = fear
	}
	right_portrait = {
		trigger = { exists = scope:kehillah_mob_leader }
		character = scope:kehillah_mob_leader
		animation = dismissal
	}
	trigger = {
		is_alive = yes
		exists = var:kehillah_pogrom_captor_ref
	}
	immediate = {
		var:kehillah_pogrom_captor_ref = { save_scope_as = kehillah_mob_leader }
	}

	# Refuse.
	option = {
		name = kehillah_pogrom.0040.a
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_pogrom_martyr_greatness }
		hidden_effect = { kehillah_pogrom_scene_result_toast_effect = { TYPE = event_toast_text_bad KEY = captive_refused } }
		death = {
			death_reason = death_murder
			killer = scope:kehillah_mob_leader
		}
		ai_chance = { base = 50 }
	}

	# Yield: the water, or the words.
	option = {
		name = kehillah_pogrom.0040.b
		hidden_effect = { kehillah_pogrom_scene_result_toast_effect = { TYPE = event_toast_text_bad KEY = captive_yielded } }
		kehillah_pogrom_leader_baptized_effect = yes
		ai_chance = { base = 50 }
	}

	# Wait for the dark, and run: your prowess against his.
	option = {
		name = kehillah_pogrom.0040.c
		trigger = {
			scope:kehillah_mob_leader = { is_alive = yes }
			OR = {
				prowess >= 10
				intrigue >= 12
				has_trait = brave
			}
		}
		duel = {
			skill = prowess
			target = scope:kehillah_mob_leader
			40 = {
				desc = kehillah_pogrom.0040.escaped
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				min = 5
				custom_tooltip = kehillah_pogrom.0040.escaped_tt
				kehillah_pogrom_scene_result_toast_effect = { TYPE = event_toast_text_good KEY = captive_escaped }
			}
			60 = {
				desc = kehillah_pogrom.0040.caught
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
				kehillah_pogrom_scene_result_toast_effect = { TYPE = event_toast_text_bad KEY = captive_caught }
				death = {
