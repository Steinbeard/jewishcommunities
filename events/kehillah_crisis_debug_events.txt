# Hidden probes for a DISPOSABLE campaign, on the community being tested.
# .1 opens a warning with 500 gold; .2 completes one preparation by one synthetic
# fourteen-day tick; .3 logs state; .4 simulates a successful defence ending.
kehillah_crisis_debug.1 = {
 type = character_event hidden = yes
 trigger = { government_has_flag = government_is_kehillah }
 immediate = {
  add_gold = 500
  if = { limit = { primary_title = { NOT = { has_variable = kehillah_mob_threat } } } kehillah_mob_local_warning_effect = yes }
  primary_title.var:kehillah_mob_threat ?= { set_variable = { name = kehillah_mob_preach_days value = 90 } }
  debug_log = "kehillah_crisis_debug.1: isolated warning; Military Community Safety should show no war"
 }
}

    limit = { exists = scope:kehillah_debug_struck_lord }
    set_variable = { name = kehillah_crisis_sl_men_lord value = scope:kehillah_debug_struck_lord }
    set_variable = { name = kehillah_crisis_sl_refuge_lord value = scope:kehillah_debug_struck_lord }
