namespace = kehillah_crisis_debug
# Hidden probes for a DISPOSABLE campaign, on the community being tested.
# .1 opens a warning with 500 gold; .2 completes one preparation by one synthetic
# fourteen-day tick; .3 logs state; .4 simulates a successful defence ending.
kehillah_crisis_debug.1 = {
 type = character_event hidden = yes
 trigger = { government_has_flag = government_is_kehillah }
 immediate = {
  add_gold = 500
  if = { limit = { primary_title = { NOT = { has_variable = kehillah_mob_threat } } } kehillah_mob_local_warning_effect = yes }
  primary_title.var:kehillah_mob_threat ?= { set_variable = { name = kehillah_mob_preach_days value = 90 } }
  debug_log = "kehillah_crisis_debug.1: isolated warning; Military Community Safety should show no war"
 }
}
kehillah_crisis_debug.2 = {
 type = character_event hidden = yes
 trigger = { kehillah_crisis_warning_trigger = yes }
 immediate = { kehillah_crisis_tick_effect = { DAYS = 14 } trigger_event = kehillah_crisis_debug.3 }
}
kehillah_crisis_debug.3 = {
 type = character_event hidden = yes
 trigger = { government_has_flag = government_is_kehillah }
 immediate = {
  debug_log = "CRISIS stage=[THIS.Char.GetPrimaryTitle.MakeScope.Var('kehillah_crisis_stage').GetValue] agitation=[THIS.Char.GetPrimaryTitle.MakeScope.Var('kehillah_crisis_agitation').GetValue]"
  primary_title.var:kehillah_crisis_situation ?= { kehillah_crisis_race_mirror_effect = yes }
  debug_log = "CRISIS race armed=[THIS.Char.GetPrimaryTitle.MakeScope.Var('kehillah_crisis_armed_points').GetValue] spared=[THIS.Char.GetPrimaryTitle.MakeScope.Var('kehillah_crisis_spared_points').GetValue] (empty outside the race)"
  debug_log_scopes = yes
 }
}
kehillah_crisis_debug.4 = {
 type = character_event hidden = yes
 trigger = { government_has_flag = government_is_kehillah exists = primary_title.var:kehillah_mob_threat }
 immediate = { primary_title.var:kehillah_mob_threat = { kehillah_mob_dispersed_effect = yes } }
}
kehillah_crisis_debug.5 = {
 type = character_event hidden = yes
 trigger = { government_has_flag = government_is_kehillah primary_title = { has_variable = kehillah_crisis_stage } }
 immediate = {
  primary_title = {
   kehillah_crisis_situation_sync_effect = yes
   var:kehillah_crisis_situation ?= {
    debug_log = "kehillah_crisis_debug.5: situation instance and scopes"
    debug_log_scopes = yes
   }
  }
 }
}

# The race (2026-11-01), console-only: .6 decides the running race against the community (the host
# rises in Agitation; the quarter is struck in the Armed race); .7 decides it for the community
# (Spared; Quarter held). Takeover follows a few in-game days later. Then log.
kehillah_crisis_debug.6 = {
 type = character_event hidden = yes

   kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_armed_catalyst }

  }

}
