# activities/activity_types/kehillah_bet_din_conference.txt) via
# kehillah_bet_din_travel_pulse_effect (common/scripted_effects/
# kehillah_bet_din_scripted_effects.txt), which also owns the chance
# gate and the pick between the two events below -- nothing here fires
# on its own.
#
# root = whichever character is currently travelling to the conference
# -- a co-judge, or (since the same 2026-09-09 pass widened the guest
# list) an accepted Jewish adventurer or high-learning scholar. Kept
# deliberately generic to whoever that is: no reference to being a
# community leader specifically, no litigant names, nothing that
# assumes the traveller is one of the three Sh'um leaders.
#
# PURE FLAVOR, no mechanical effect: this is Part 1 minimal, matching
# the rest of this activity (see kehillah_bet_din_conference.txt's own
# header on deliberately not building a full pulse_actions catalog) --
# two vignettes proving the journey is not silent, not a systemic
# reward for travelling.

namespace = kehillah_bet_din_travel

	title = kehillah_bet_din_travel.0001.t
	desc = kehillah_bet_din_travel.0001.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		always = yes
	}

	option = {
		name = kehillah_bet_din_travel.0001.a
	}
}

# ---------------------------------------------------------------------
# .0002 -- A Fellow Traveler. REWRITTEN 2026-11-25 (user review: the
# original was a one-line anecdote with no merchant on screen and a
# vague "mostly true" wink). A Jewish merchant on the same road has a
# real grievance -- a partner who ran off with his goods -- and asks
# whether a great Bet Din would hear someone like him. The three answers
# are the three real routes an ordinary Jew had to a hearing: the local
# court, ikkuv ha-tefillah (halting the synagogue service until the
# community hears you), or a notable who carries the case for you.
# Small personal effects only; still no effect on the conference.
# The merchant is a throwaway character (silently_disappear_me).
# ---------------------------------------------------------------------
kehillah_bet_din_travel.0002 = {
	type = character_event
	title = kehillah_bet_din_travel.0002.t
	desc = kehillah_bet_din_travel.0002.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_travel_merchant
		animation = worry
	}

	trigger = {
		exists = location
	}

	immediate = {
		create_character = {
			template = merchant_template
			location = root.location
			save_scope_as = kehillah_travel_merchant
		}
		scope:kehillah_travel_merchant = {
			add_character_flag = {
				flag = silently_disappear_me
				years = 1
			}
		}
	}

	# Send him to the court of his own town.
	option = {
		name = kehillah_bet_din_travel.0002.a
		flavor = kehillah_bet_din_travel.0002.a.flavor
		add_learning_lifestyle_xp = minor_lifestyle_xp
		ai_chance = {
			base = 10
		}
	}

	# Teach him ikkuv ha-tefillah.
	option = {
		name = kehillah_bet_din_travel.0002.b
		flavor = kehillah_bet_din_travel.0002.b.flavor
		add_piety = minor_piety_gain
		add_learning_lifestyle_xp = minor_lifestyle_xp
		ai_chance = {
			base = 10
			modifier = {
				add = 10
				has_trait = just
			}
		}
	}

	# Carry the case to the conference yourself.
	option = {
		name = kehillah_bet_din_travel.0002.c
		flavor = kehillah_bet_din_travel.0002.c.flavor
		add_prestige = minor_prestige_gain
		add_stress = minor_stress_gain
		ai_chance = {
			base = 10
			modifier = {
				add = 10
				has_trait = compassionate
			}
			modifier = {
				factor = 0
				has_trait = callous
			}
		}
	}
}
