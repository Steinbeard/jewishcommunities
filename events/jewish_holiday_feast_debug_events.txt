# Console-only probes for the vanilla Feast overlay. Run on the planned
# host before starting a Feast; no game pulse calls these events.
namespace = jewish_holiday_feast_debug

jewish_holiday_feast_debug.1 = {
    type = character_event
    hidden = yes
    orphan = yes
    immediate = {
        if = { limit = { is_jewish_character_trigger = yes } debug_log = "jewish_holiday_feast_debug: Jewish character" }
        else = { debug_log = "jewish_holiday_feast_debug: NOT Jewish" }
        if = { limit = { rite = { rite_has_tenet = tenet_ritual_celebrations } } debug_log = "jewish_holiday_feast_debug: Ritual Celebrations tenet present" }
        else = { debug_log = "jewish_holiday_feast_debug: Ritual Celebrations tenet absent" }
        debug_log_scopes = yes
    }
}

jewish_holiday_feast_debug.2 = {
    type = character_event
    hidden = yes
    orphan = yes
    immediate = {
        add_character_flag = { flag = jewish_holiday_recent_pesach days = 45 }
        debug_log = "jewish_holiday_feast_debug: Pesach theme armed for 45 days"
    }
}
