# =====================================================================
# Learn Torah, the scheme -- the events. ADDED 2026-09-19. See common/
# schemes/scheme_types/kehillah_study_torah_scheme.txt for the scheme
# itself and common/scripted_effects/kehillah_library_effects.txt's own
# "STUDY TORAH AS A SCHEME" header for the numeric-ID bridge
# (kehillah_study_current_work_id) both events below read from.
#
# The scheme starts with 0002, which describes the newly opened work.
# 0001 fires at phase completion after the scheme has already resolved the
# current work, so its description can distinguish mastery from a return
# visit. Rare mid-course interactions are 0003 (a related text) and 0004
# (help from a learned friend or courtier).
#
# The completion event deliberately does not present the next work as a
# choice: the picker has already selected it from the library, and the
# single continuation option simply accepts that automatic choice.
# =====================================================================

namespace = kehillah_study_torah

# ---------------------------------------------------------------------
# kehillah_study_torah.0001 -- A COURSE OF STUDY CONCLUDES
# ---------------------------------------------------------------------

kehillah_study_torah.0001 = {
	type = character_event
	title = kehillah_study_torah.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 1 }
				desc = kehillah_study_torah.0001.mastered_desc
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 0 }
				desc = kehillah_study_torah.0001.unmastered_desc
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 2 }
				desc = kehillah_study_torah.0001.repeat_mastered_desc
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 3 }
				desc = kehillah_study_torah.0001.repeat_unmastered_desc
			}
			desc = kehillah_study_torah.0001.desc
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	# BROADENED 2026-09-19, same-day follow-up: was government_has_flag =
	# government_is_kehillah -- the scheme itself is no longer Kehillah-
	# leader-only (kehillah_study_torah_has_accessible_library_trigger's
	# own header, common/scripted_triggers/kehillah_scripted_triggers.txt,
	# has the three populations), so this event's own trigger cannot stay
	# narrower than the scheme that fires it.
	trigger = {
		is_jewish_character_trigger = yes
	}

	option = {
		name = kehillah_study_torah.0001.a

		# Re-resolve which library applies NOW, not whichever one applied
		# when this course of study last picked a work -- a moved capital
		# or a relocated adventurer may mean a different community's shelves
		# entirely (kehillah_study_torah_resolve_library_effect's own
		# header, common/scripted_effects/kehillah_library_effects.txt).
		# Then pick what comes next -- this is automatic, not a second set
		# of player choices. The selected work is revealed by the same
		# "You take up X" tooltip lines used since 2026-09-15. Guarded on a
		# library actually existing AND still owning at least one work: no
		# community reachable at all, or one reachable but donated down to
		# nothing, both mean nothing left to reveal here -- the scheme's
		# own on_invalidated will end it the next time its own valid
		# trigger is checked, if this is not transient. CHARACTER = root
		# is safe here -- a character_event's own root never shifts, it
		# already is the character throughout.
		kehillah_study_torah_resolve_library_effect = { CHARACTER = root }
		if = {
			limit = {
				exists = scope:kehillah_study_torah_library
				kehillah_owns_any_torah_work_trigger = { LIBRARY = scope:kehillah_study_torah_library }
			}
			kehillah_study_torah_pick_next_work_effect = { LIBRARY = scope:kehillah_study_torah_library }
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_study_torah.0003 -- A RELATED PASSAGE RETURNS
# ---------------------------------------------------------------------

kehillah_study_torah.0003 = {
	type = character_event
	title = kehillah_study_torah.0003.t
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						var:kehillah_study_current_work_id <= 4
						AND = { var:kehillah_study_current_work_id >= 5 var:kehillah_study_current_work_id <= 16 }
					}
				}
				desc = kehillah_study_torah.0003.parshanut_or_talmudics_desc
			}
			desc = kehillah_study_torah.0003.hashkafa_desc
		}
	}

	trigger = {
		is_jewish_character_trigger = yes
		has_variable = kehillah_study_current_work_id
		kehillah_study_torah_has_same_track_reference_trigger = yes
	}

	option = {
		name = kehillah_study_torah.0003.a
		scope:scheme = { add_scheme_progress = 90 }
		set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 }
	}

	option = {
		name = kehillah_study_torah.0003.b
		trigger = {
			any_courtier = { learning >= 12 }
		}
		random_courtier = {
			limit = { learning >= 12 }
			save_scope_as = kehillah_study_torah_mentor
		}
		scope:scheme = { add_scheme_progress = 45 }
		set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 }
	}
}
