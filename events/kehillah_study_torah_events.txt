# =====================================================================
# Learn Torah, the scheme -- the events. ADDED 2026-09-19. See common/
# schemes/scheme_types/kehillah_study_torah_scheme.txt for the scheme
# itself and common/scripted_effects/kehillah_library_effects.txt's own
# "STUDY TORAH AS A SCHEME" header for the numeric-ID bridge
# (kehillah_study_current_work_id) both events below read from.
#
# The scheme starts with 0002, which describes the newly opened work.
# 0001 fires at phase completion after the scheme has already resolved the
# current work, so its description can distinguish mastery from a return
# visit. Rare mid-course interactions are 0003 (a related text) and 0004
# (help from a learned friend or courtier).
#
# 2026-11-11: the completion event now offers the next work as a choice, one
# unmastered book per genre, plus "choose later" (Daniel's review). It used to
# draw the next work at random behind a single "Take up the next book" option;
# that draw is now only the reread offered once every book is mastered.
# =====================================================================

namespace = kehillah_study_torah

# ---------------------------------------------------------------------
# kehillah_study_torah.0001 -- A COURSE OF STUDY CONCLUDES
# ---------------------------------------------------------------------

kehillah_study_torah.0001 = {
	type = character_event
	title = kehillah_study_torah.0001.t
	desc = {
		first_valid = {
			# 2026-11-11: a stretch of the course passed with no book chosen.
			triggered_desc = {
				trigger = { has_variable = kehillah_study_choose_later }
				desc = kehillah_study_torah.0001.waiting_desc
			}
			triggered_desc = {
				trigger = {
					var:kehillah_study_torah_last_result = 1
					kehillah_talmud_torah_follower_trigger = yes
				}
				desc = {
					random_valid = {
						desc = kehillah_study_torah.0001.mastered_talmud_torah_desc
						desc = kehillah_study_torah.0001.mastered_talmud_torah_desc_2
						desc = kehillah_study_torah.0001.mastered_talmud_torah_desc_3
					}
				}
			}
			triggered_desc = {
				trigger = {
					var:kehillah_study_torah_last_result = 2
					kehillah_talmud_torah_follower_trigger = yes
				}
				desc = {
					random_valid = {
						desc = kehillah_study_torah.0001.repeat_talmud_torah_desc
						desc = kehillah_study_torah.0001.repeat_talmud_torah_desc_2
						desc = kehillah_study_torah.0001.repeat_talmud_torah_desc_3
					}
				}
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 1 }
				desc = {
					random_valid = {
						desc = kehillah_study_torah.0001.mastered_desc
						desc = kehillah_study_torah.0001.mastered_desc_2
						desc = kehillah_study_torah.0001.mastered_desc_3
					}
				}
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 0 }
				desc = kehillah_study_torah.0001.unmastered_desc
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 2 }
				desc = {
					random_valid = {
						desc = kehillah_study_torah.0001.repeat_mastered_desc
						desc = kehillah_study_torah.0001.repeat_mastered_desc_2
					}
				}
			}
			triggered_desc = {
				trigger = { var:kehillah_study_torah_last_result = 3 }
				desc = kehillah_study_torah.0001.repeat_unmastered_desc
			}
			desc = kehillah_study_torah.0001.desc
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	# BROADENED 2026-09-19, same-day follow-up: was government_has_flag =
	# government_is_kehillah -- the scheme itself is no longer Kehillah-
	# leader-only (kehillah_study_torah_has_accessible_library_trigger's
	# own header, common/scripted_triggers/kehillah_scripted_triggers.txt,
	# has the three populations), so this event's own trigger cannot stay
	# narrower than the scheme that fires it.
	trigger = {
		is_jewish_character_trigger = yes
	}

	# 2026-11-11 (Daniel: "We ought to fix this 'Next book' random chance flow ...
	# offer one option of each genre that you haven't read, if you have one
	# available, and also a 'come back later to choose' option"). The next book
	# is no longer drawn at random behind one option. Re-resolve which library
	# applies NOW (a moved capital or a relocated adventurer may mean different
	# shelves; kehillah_study_torah_resolve_library_effect's header, common/
	# scripted_effects/kehillah_library_effects.txt), then offer, per genre, one
	# work it holds that the reader has not mastered (kehillah_study_torah_offer_
	# effect, generated by scripts/kehillah_library_catalog.py). CHARACTER = root
	# is safe: a character_event's root is the character throughout.
	immediate = {
		kehillah_study_torah_clear_offers_effect = yes
		kehillah_study_torah_resolve_library_effect = { CHARACTER = root }
		if = {
			limit = {
				exists = scope:kehillah_study_torah_library
				kehillah_owns_any_torah_work_trigger = { LIBRARY = scope:kehillah_study_torah_library }
			}
			kehillah_study_torah_offer_effect = { LIBRARY = scope:kehillah_study_torah_library }
		}
	}

	# One option per genre with an unmastered book on the shelf. Each names its
	# book; its tooltip is that book's own "You take up X" line.
	option = {
		name = kehillah_study_torah.0001.genre_1
		trigger = { has_variable = kehillah_study_offer_1 }
		remove_variable = kehillah_study_choose_later
		kehillah_study_torah_take_offer_effect = { GENRE = 1 }
		kehillah_study_torah_clear_offers_effect = yes
		ai_chance = { base = 10 }
	}
	option = {
		name = kehillah_study_torah.0001.genre_2
		trigger = { has_variable = kehillah_study_offer_2 }
		remove_variable = kehillah_study_choose_later
		kehillah_study_torah_take_offer_effect = { GENRE = 2 }
		kehillah_study_torah_clear_offers_effect = yes
		ai_chance = { base = 10 }
	}
	option = {
		name = kehillah_study_torah.0001.genre_3
		trigger = { has_variable = kehillah_study_offer_3 }
		remove_variable = kehillah_study_choose_later
		kehillah_study_torah_take_offer_effect = { GENRE = 3 }
		kehillah_study_torah_clear_offers_effect = yes
		ai_chance = { base = 10 }
	}

	# Every book on the shelf is mastered: read one again, drawn as before
	# (the old picker, Tanakh and Talmud twice as likely).
	option = {
		name = kehillah_study_torah.0001.a
		trigger = {
			NOT = { has_variable = kehillah_study_offer_1 }
			NOT = { has_variable = kehillah_study_offer_2 }
			NOT = { has_variable = kehillah_study_offer_3 }
			exists = scope:kehillah_study_torah_library
		}
		remove_variable = kehillah_study_choose_later
		kehillah_study_torah_pick_next_work_effect = { LIBRARY = scope:kehillah_study_torah_library }
		kehillah_study_torah_clear_offers_effect = yes
		ai_chance = { base = 10 }
	}

	# Choose later: the course goes on with no book. The Beit Midrash's Study
	# button names the next one, free, without starting a new course
	# (kehillah_lib_study_<work>, generated; kehillah_study_waiting_for_choice_
	# trigger). Until then this event returns at the end of each stretch.
	option = {
		name = kehillah_study_torah.0001.later
		remove_variable = kehillah_study_current_work_id
		set_variable = { name = kehillah_study_choose_later value = yes }
		kehillah_study_torah_clear_offers_effect = yes
		custom_tooltip = kehillah_study_torah.0001.later_tt
		ai_chance = { base = 0 }
	}
}

# ---------------------------------------------------------------------
# kehillah_study_torah.0003 -- A RELATED PASSAGE RETURNS
# ---------------------------------------------------------------------

kehillah_study_torah.0003 = {
	type = character_event
	title = kehillah_study_torah.0003.t
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	desc = {
		first_valid = {
			triggered_desc = {
				# 2026-11-08: by the work's genre, not a fixed ID range (stale once
				# the catalog grew past its first 23 works).
				trigger = { kehillah_study_current_work_is_hashkafa_trigger = no }
				desc = kehillah_study_torah.0003.parshanut_or_talmudics_desc
			}
			desc = kehillah_study_torah.0003.hashkafa_desc
		}
	}

	trigger = {
		is_jewish_character_trigger = yes
		has_variable = kehillah_study_current_work_id
		kehillah_study_torah_has_same_track_reference_trigger = yes
	}

	option = {
		name = kehillah_study_torah.0003.a
		scope:scheme = { add_scheme_progress = 90 }
		set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 }
	}

	option = {
		name = kehillah_study_torah.0003.b
		trigger = {
			any_courtier = { learning >= 12 }
		}
		random_courtier = {
			limit = { learning >= 12 }
			save_scope_as = kehillah_study_torah_mentor
		}
		scope:scheme = { add_scheme_progress = 45 }
		set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 }
	}
}
