# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
#   0100        the Synod's rite-doctrine proposal
#   0101-0103   the host argues, a delegate votes, the ruling
# A synod is a Sitting of the Bet Din held with the Synod or Sanhedrin
# intent. Two ordinary cases are heard first; the third phase opens this
# doctrine debate, and the conference's on_complete fires its ruling.
namespace = kehillah_synod

# ---------------------------------------------------------------------
# kehillah_synod.0003 -- LANDED GOVERNMENT RE-CHECK (hidden)
# A day after the Sanhedrin is restored or changes hands, or a community
# or land changes hands under a Kehillah leader: the engine settles governments
# after effects return, so the transition is checked again
# (kehillah_reconcile_landed_government_effect, V46 spec section 9).
# ---------------------------------------------------------------------
kehillah_synod.0003 = {
	type = character_event
	hidden = yes

	immediate = {
		kehillah_reconcile_landed_government_effect = yes
		if = {
			limit = { kehillah_is_sanhedrin_head_trigger = yes }
			kehillah_sanhedrin_refresh_title_name_effect = yes
		}
		if = {
			limit = { has_government = kehillah_landed_government }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, on the landed government"
		}
		else_if = {
			limit = { has_government = kehillah_government }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, on the Kehillah government"
		}
		else = { debug_log = "KEHILLAH_LANDED_GOV: day-after check, on ANOTHER government" }
		# EXPERIMENT (2026-10-19, not yet live-verified on a LANDED leader): a
		# community leader who lost his Quarter on gaining land (the engine
		# moves him onto the liege's government, which drops it). The
		# temporary-title scaffold of kehillah_succession.0010, proven for
		# landless successors, is the only known way to provision a domicile.
		if = {
			limit = {
				is_landed = yes
				NOT = { exists = domicile }
				primary_title ?= { is_kehillah_title_trigger = yes }
			}
			primary_title = { save_scope_as = kq_title }
			create_adventurer_title = {
				name = kehillah_successor_scaffold_title_name
				holder = root
				government = kehillah_landed_government
				article = kehillah_successor_scaffold_title_article
				save_scope_as = kehillah_landed_scaffold_title
			}
			if = {
				limit = { exists = scope:kehillah_landed_scaffold_title }
				hidden_effect = { set_primary_title_to = scope:kq_title }
				destroy_title = scope:kehillah_landed_scaffold_title
			}
			else = { debug_log = "KEHILLAH_LANDED_GOV: scaffold NOT created for a landed leader" }
			if = {
				limit = { exists = domicile }
				debug_log = "KEHILLAH_LANDED_GOV: Quarter provisioned by the scaffold"
				kehillah_restore_quarter_effect = yes
			}
			else = { debug_log = "KEHILLAH_LANDED_GOV: Quarter still MISSING after the scaffold" }
			kehillah_reconcile_landed_government_effect = yes
		}
		if = {
			limit = { exists = domicile }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, has a domicile"
		}
		else = { debug_log = "KEHILLAH_LANDED_GOV: day-after check, NO domicile" }
		debug_log_scopes = yes
	}
}

		animation = reading
	}

# ---------------------------------------------------------------------
# kehillah_synod.0004 -- THE GAON'S OFFICE PASSES (hidden, V47)
# A day after an established office's academy changes hands: its new
# rabbi-leader becomes Gaon, or the office lapses.
# ---------------------------------------------------------------------
kehillah_synod.0004 = {
	type = character_event
	hidden = yes

	# No trigger: an event whose trigger fails is dropped without a trace,
	# and the first live test (2026-10-19) left no log line at all. The
	# checks are logged instead.
	immediate = {
		debug_log = "KEHILLAH_GAONATE: kehillah_synod.0004 fired"
		debug_log_scopes = yes
		if = {
			limit = {
				exists = global_var:kehillah_gaonate_office_academy
				primary_title ?= { this = global_var:kehillah_gaonate_office_academy }
			}
			kehillah_gaonate_academy_succession_effect = yes
		}
		else = { debug_log = "KEHILLAH_GAONATE: kehillah_synod.0004, not the academy's leader or no office" }
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0100 -- THE RITE-DOCTRINE PROPOSAL (host)
# The proposal is stored as a doctrine scope on the activity, following
# vanilla's Ecumenical Council final-debate pattern. Each option remains
# visible only while that doctrine change is available to this host.
# ---------------------------------------------------------------------
kehillah_synod.0100 = {
	type = character_event
	title = kehillah_synod.0100.t
	desc = kehillah_synod.0100.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = { exists = involved_activity }

	# 2026-11-25 dynamic agenda (docs/spec/2026-11-25-dynamic-synod-agenda.md):
	# this page offers the KINDS of matter that are live for this host; each
	# opens its own page. Nothing is paid until a motion is put.
	option = {
		name = kehillah_synod_kind_marriage
		trigger = { kehillah_synod_has_marriage_motion_trigger = yes }
		trigger_event = kehillah_synod.0106
		ai_chance = {
			base = 50
			modifier = { add = 50 kehillah_synod_can_propose_herem_polygamy_trigger = yes }
		}
	}
	# 2026-11-08 Sefer redesign: a question of belief -- raise a tenet to a
	# core tenet of the rite, or permit, ban or free it in the faith. The
	# choice runs through kehillah_reform.0001-.0003 (generated by
	# scripts/kehillah_reform_build.py) and returns as a tenet_reform motion.
	option = {
		name = kehillah_synod_option_tenet_reform
		trigger = { kehillah_reform_any_motion_trigger = yes }
		custom_tooltip = kehillah_synod_option_tenet_reform_desc
		kehillah_reform_clear_choice_effect = yes
		trigger_event = kehillah_reform.0001
		ai_chance = {
			base = 15
			modifier = { add = 200 rite ?= { has_variable = kehillah_reform_pending_tenet } }
		}
	}
	option = {
		name = kehillah_synod_kind_law
		trigger = { kehillah_synod_has_law_motion_trigger = yes }
		trigger_event = kehillah_synod.0109
		ai_chance = { base = 5 }
	}
	option = {
		name = kehillah_synod_kind_ordinance
		trigger = { kehillah_synod_has_communal_motion_trigger = yes }
		trigger_event = kehillah_synod.0105
		ai_chance = { base = 30 }
	}
	option = {
		name = kehillah_synod_kind_headship
		trigger = { kehillah_synod_can_propose_head_trigger = yes }
		trigger_event = kehillah_synod.0108
		ai_chance = { base = 60 }
	}
	option = {
		name = kehillah_synod_kind_undertaking
		trigger = { kehillah_synod_has_undertaking_motion_trigger = yes }
		trigger_event = kehillah_synod.0107
		ai_chance = { base = 20 }
	}
	# 2026-11-11 cherem track (docs/spec/2026-11-11-reviews-and-cherem.md
	# section 3.3): a herem on a community leader in Niddui. The synod takes
	# up the one deepest in it.
	option = {
		name = kehillah_synod_option_herem_person
		trigger = { kehillah_synod_can_propose_herem_person_trigger = yes }
		custom_tooltip = kehillah_synod_option_herem_person_desc
		hidden_effect = {
			ordered_in_global_list = {
				variable = kehillah_cherem_tracked
				limit = { kehillah_synod_herem_target_trigger = { HOST = root } }
				order_by = "has_trait_xp(kehillah_cherem_trait)"
				save_scope_as = kehillah_synod_herem_target
			}
			involved_activity ?= { set_variable = { name = kehillah_synod_herem_target value = scope:kehillah_synod_herem_target } }
		}
		kehillah_synod_put_motion_effect = { MOTION = flag:herem_person }
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = zealous }
		}
	}
	# Nothing on the agenda is ready (a candidate stayed home, a motion lapsed
	# since the intent was chosen). The sitting rises without a ruling; no
	# prestige is spent, and kehillah_synod.0103 does not fire.
	option = {
		name = kehillah_synod_kind_none
		trigger = {
			NOT = { kehillah_synod_has_marriage_motion_trigger = yes }
			NOT = { kehillah_reform_any_motion_trigger = yes }
			NOT = { kehillah_synod_has_communal_motion_trigger = yes }
			NOT = { kehillah_synod_can_propose_head_trigger = yes }
			NOT = { kehillah_synod_has_undertaking_motion_trigger = yes }
			NOT = { kehillah_synod_has_law_motion_trigger = yes }
			NOT = { kehillah_synod_can_propose_herem_person_trigger = yes }
		}
		custom_tooltip = kehillah_synod_kind_none_tt
		ai_chance = { base = 1 }
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0109 -- A QUESTION OF LAW AND CUSTOM (2026-11-25)
# Doctrines of the host's rite that are not settled Torah or Talmudic law.
# A row is offered only when the rite does not hold its doctrine, and says
# what it would replace. Its controversy is scored in immediate from a STEP
# that depends on the rite's present doctrine (kehillah_synod_step_*_value)
# and the rite's divergence (kehillah_synod_score_doctrine_effect), shown as
# a band, and stored on the sitting when the motion is put.
# ---------------------------------------------------------------------
kehillah_synod.0109 = {
	type = character_event
	title = kehillah_synod.0109.t
	desc = kehillah_synod.0109.desc
	theme = faith
	left_portrait = { character = root animation = thinking }
	trigger = { exists = involved_activity }

	immediate = {
		if = { limit = { kehillah_synod_can_propose_divorce_approval_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_divorce_approval STEP = kehillah_synod_step_divorce_approval_value VAR = kehillah_synod_c_divorce_approval } }
		if = { limit = { kehillah_synod_can_propose_divorce_allowed_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_divorce_allowed STEP = kehillah_synod_step_divorce_allowed_value VAR = kehillah_synod_c_divorce_allowed } }
		if = { limit = { kehillah_synod_can_propose_proselytism_encouraged_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = kehillah_proselytism_encouraged STEP = kehillah_synod_step_proselytism_encouraged_value VAR = kehillah_synod_c_proselytism_encouraged } }
		if = { limit = { kehillah_synod_can_propose_proselytism_discouraged_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = kehillah_proselytism_discouraged STEP = kehillah_synod_step_proselytism_discouraged_value VAR = kehillah_synod_c_proselytism_discouraged } }
		if = { limit = { kehillah_synod_can_propose_clergy_either_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_clerical_gender_either STEP = kehillah_synod_step_clergy_either_value VAR = kehillah_synod_c_clergy_either } }
		if = { limit = { kehillah_synod_can_propose_clergy_male_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_clerical_gender_male_only STEP = kehillah_synod_step_clergy_male_value VAR = kehillah_synod_c_clergy_male } }
		if = { limit = { kehillah_synod_can_propose_funeral_bewailment_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_funeral_bewailment STEP = kehillah_synod_step_funeral_bewailment_value VAR = kehillah_synod_c_funeral_bewailment } }

		if = { limit = { kehillah_synod_can_propose_recognition_trigger = yes } kehillah_synod_score_doctrine_effect = { DOCTRINE = kehillah_recognition_of_sages STEP = kehillah_synod_step_recognition_value VAR = kehillah_synod_c_recognition } }
	}

	option = {
		name = kehillah_synod_option_law_divorce_approval
		trigger = { kehillah_synod_can_propose_divorce_approval_trigger = yes }
		kehillah_synod_law_now_divorce_tt_effect = yes
		if = {
			limit = { rite ?= { rite_has_doctrine = doctrine_divorce_disallowed } }

		kehillah_synod_put_doctrine_motion_effect = { DOCTRINE = doctrine_divorce_approval VAR = kehillah_synod_c_divorce_approval }
		ai_chance = {
			base = 3
			modifier = { factor = 0.5 var:kehillah_synod_c_divorce_approval >= kehillah_synod_controversy_major }
		}
	}

	option = {
		name = kehillah_synod_option_law_divorce_allowed
		trigger = { kehillah_synod_can_propose_divorce_allowed_trigger = yes }

				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:rite_doctrine
				OR = { has_trait = zealous has_trait = stubborn }
			}
			modifier = {
				add = 40
				involved_activity.var:kehillah_synod_takkanah ?= flag:head_of_rite
				scope:kehillah_synod_nominee ?= { has_relation_rival = root }
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:head_of_rite
				NOT = { scope:kehillah_synod_nominee ?= root }
				has_trait = ambitious
				kehillah_synod_head_candidate_trigger = { HOST = scope:kehillah_synod_host }
			}
			modifier = {
				add = 20
				OR = {
					involved_activity.var:kehillah_synod_takkanah ?= flag:captives
					involved_activity.var:kehillah_synod_takkanah ?= flag:levy
				}
				has_trait = greedy

		}

# Fired from the Bet Din's on_complete, so every synod reaches it and
# nothing chains past it. Everything that reads the activity is settled in
# immediate, while the activity still exists: the player may answer days
# later. The host's own vote is the 1 seeded into votes_for.
# ---------------------------------------------------------------------
kehillah_synod.0103 = {
	type = character_event
	title = kehillah_synod.0103.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:kehillah_synod_outcome = flag:no_quorum }
				desc = kehillah_synod.0103.desc.no_quorum
			}
			triggered_desc = {
				trigger = { scope:kehillah_synod_outcome = flag:passed }
				desc = kehillah_synod.0103.desc.passed
			}
			desc = kehillah_synod.0103.desc.failed
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		exists = involved_activity
		involved_activity = { has_variable = kehillah_synod_takkanah }
	}

	immediate = {
		save_scope_value_as = {
			name = kehillah_synod_takkanah
			value = involved_activity.var:kehillah_synod_takkanah
		}
		involved_activity.var:kehillah_synod_proposed_doctrine ?= {
			save_scope_as = kehillah_synod_proposed_doctrine
		}
		involved_activity.var:kehillah_synod_herem_target ?= { save_scope_as = kehillah_synod_herem_target }
		involved_activity.var:kehillah_synod_nominee ?= { save_scope_as = kehillah_synod_nominee }
		involved_activity.var:kehillah_synod_proposed_doctrine ?= { save_scope_as = kehillah_synod_proposed_doctrine }
		save_scope_value_as = {
			name = kehillah_synod_level
			value = involved_activity.var:kehillah_synod_level
		}
		# 2026-11-25 law and custom: the controversy, for the ruling's cost.
		if = {
			limit = { exists = involved_activity.var:kehillah_synod_controversy }
			save_scope_value_as = {
				name = kehillah_synod_controversy
				value = involved_activity.var:kehillah_synod_controversy
			}
		}
		if = {
			limit = { kehillah_synod_quorum_present_trigger = no }
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:no_quorum }
		}
		else_if = {
			limit = { kehillah_synod_motion_carried_trigger = yes }
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:passed }
		}
		else = {
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:failed }
		}
		remove_variable = kehillah_synod_in_session
	}

	option = {
		name = kehillah_synod.0103.a
		if = {
			limit = { scope:kehillah_synod_outcome = flag:passed }
			kehillah_synod_enact_takkanah_effect = yes
		}
		else_if = {
			limit = { scope:kehillah_synod_outcome = flag:failed }
			kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_failed_greatness }
		}
	}

	after = {
		if = {
			limit = { scope:kehillah_synod_outcome = flag:passed }
			kehillah_notify_rite_players_effect = { TYPE = msg_kehillah_community_news TITLE = kehillah_news_synod_t DESC = kehillah_news_synod_passed_d }
		}
	}
}
