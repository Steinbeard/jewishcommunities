# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
#   0100        the Synod's rite-doctrine proposal
#   0101-0103   the host argues, a delegate votes, the ruling
# A synod is a Sitting of the Bet Din held with the Synod or Sanhedrin
# intent. Two ordinary cases are heard first; the third phase opens this
# doctrine debate, and the conference's on_complete fires its ruling.
namespace = kehillah_synod

# ---------------------------------------------------------------------
# kehillah_synod.0003 -- LANDED GOVERNMENT RE-CHECK (hidden)
# A day after the Sanhedrin is restored or changes hands, or a community
# or land changes hands under a Kehillah leader: the engine settles governments
# after effects return, so the transition is checked again
# (kehillah_reconcile_landed_government_effect, V46 spec section 9).
# ---------------------------------------------------------------------
kehillah_synod.0003 = {
	type = character_event
	hidden = yes

	immediate = {
		kehillah_reconcile_landed_government_effect = yes
		if = {
			limit = { kehillah_is_sanhedrin_head_trigger = yes }
			kehillah_sanhedrin_refresh_title_name_effect = yes
		}
		if = {
			limit = { has_government = kehillah_landed_government }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, on the landed government"
		}
		else_if = {
			limit = { has_government = kehillah_government }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, on the Kehillah government"
		}
		else = { debug_log = "KEHILLAH_LANDED_GOV: day-after check, on ANOTHER government" }
		# EXPERIMENT (2026-10-19, not yet live-verified on a LANDED leader): a
		# community leader who lost his Quarter on gaining land (the engine
		# moves him onto the liege's government, which drops it). The
		# temporary-title scaffold of kehillah_succession.0010, proven for
		# landless successors, is the only known way to provision a domicile.
		if = {
			limit = {
				is_landed = yes
				NOT = { exists = domicile }
				primary_title ?= { is_kehillah_title_trigger = yes }
			}
			primary_title = { save_scope_as = kq_title }
			create_adventurer_title = {
				name = kehillah_successor_scaffold_title_name
				holder = root
				government = kehillah_landed_government
				article = kehillah_successor_scaffold_title_article
				save_scope_as = kehillah_landed_scaffold_title
			}
			if = {
				limit = { exists = scope:kehillah_landed_scaffold_title }
				hidden_effect = { set_primary_title_to = scope:kq_title }
				destroy_title = scope:kehillah_landed_scaffold_title
			}
			else = { debug_log = "KEHILLAH_LANDED_GOV: scaffold NOT created for a landed leader" }
			if = {
				limit = { exists = domicile }
				debug_log = "KEHILLAH_LANDED_GOV: Quarter provisioned by the scaffold"
				kehillah_restore_quarter_effect = yes
			}
			else = { debug_log = "KEHILLAH_LANDED_GOV: Quarter still MISSING after the scaffold" }
			kehillah_reconcile_landed_government_effect = yes
		}
		if = {
			limit = { exists = domicile }
			debug_log = "KEHILLAH_LANDED_GOV: day-after check, has a domicile"
		}
		else = { debug_log = "KEHILLAH_LANDED_GOV: day-after check, NO domicile" }
		debug_log_scopes = yes
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0005 -- THE VOICE OF THE GENERATION (V47, Recognition of
# the Sages). The yearly check found this rabbi the most eminent of a rite
# with no head: he is acclaimed (Daniel: an immediate acclamation, not a
# letter). He may decline it.
# ---------------------------------------------------------------------
kehillah_synod.0005 = {
	type = character_event
	title = kehillah_synod.0005.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_is_minhag_ashkenaz_trigger = yes }
				desc = kehillah_synod.0005.desc.ashkenaz
			}
			desc = kehillah_synod.0005.desc.rabbinic
		}
	}
	theme = learning

	left_portrait = {
		character = root
		animation = reading
	}

	trigger = {
		kehillah_head_of_rite_merit_trigger = yes
	}

	# Accept.
	option = {
		name = kehillah_synod.0005.a
		if = {
			limit = { kehillah_is_minhag_ashkenaz_trigger = yes }
			custom_tooltip = kehillah_seek_head_of_rite_effect_rabbenu_tt
		}
		else = {
			custom_tooltip = kehillah_seek_head_of_rite_effect_gaon_tt
		}
		hidden_effect = { kehillah_gaonate_grant_effect = yes }
		ai_chance = {
			base = 70
			modifier = { add = -50 has_trait = humble }
			modifier = { add = 30 has_trait = ambitious }
			modifier = { add = 20 has_trait = arrogant }
		}
	}

	# Decline in humility.
	option = {
		name = kehillah_synod.0005.b
		add_piety = major_piety_gain
		add_character_flag = {
			flag = kehillah_declined_head_of_rite
			years = kehillah_head_of_rite_declined_years
		}
		custom_tooltip = kehillah_synod.0005.b.tt
		ai_chance = {
			base = 30
			modifier = { add = 60 has_trait = humble }
			modifier = { add = 20 has_trait = content }
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0004 -- THE GAON'S OFFICE PASSES (hidden, V47)
# A day after an established office's academy changes hands: its new
# rabbi-leader becomes Gaon, or the office lapses.
# ---------------------------------------------------------------------
kehillah_synod.0004 = {
	type = character_event
	hidden = yes

	# No trigger: an event whose trigger fails is dropped without a trace,
	# and the first live test (2026-10-19) left no log line at all. The
	# checks are logged instead.
	immediate = {
		debug_log = "KEHILLAH_GAONATE: kehillah_synod.0004 fired"
		debug_log_scopes = yes
		if = {
			limit = {
				exists = global_var:kehillah_gaonate_office_academy
				primary_title ?= { this = global_var:kehillah_gaonate_office_academy }
			}
			kehillah_gaonate_academy_succession_effect = yes
		}
		else = { debug_log = "KEHILLAH_GAONATE: kehillah_synod.0004, not the academy's leader or no office" }
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0108 -- THE HEADSHIP OF THE RITE (2026-11-25 page)
# The rite has no head (V47). The host nominates himself or the most
# eminent qualified delegate; the delegates vote on the nominee.
# ---------------------------------------------------------------------
kehillah_synod.0108 = {
    type = character_event
    title = kehillah_synod.0108.t
    desc = kehillah_synod.0108.desc
    theme = learning
    trigger = { exists = involved_activity }
    immediate = {
        remove_variable = kehillah_synod_candidate_1
        remove_variable = kehillah_synod_candidate_2
        remove_variable = kehillah_synod_candidate_3
        involved_activity = {
            ordered_attending_character = {
                limit = { kehillah_synod_head_candidate_trigger = { HOST = root } }
                order_by = { value = 0 primary_title = { add = kehillah_head_of_rite_title_score_value } }
                max = 3
                check_range_bounds = no
                if = { limit = { NOT = { exists = root.var:kehillah_synod_candidate_1 } } root = { set_variable = { name = kehillah_synod_candidate_1 value = prev } } }
                else_if = { limit = { NOT = { exists = root.var:kehillah_synod_candidate_2 } } root = { set_variable = { name = kehillah_synod_candidate_2 value = prev } } }
                else = { root = { set_variable = { name = kehillah_synod_candidate_3 value = prev } } }
            }
        }
        var:kehillah_synod_candidate_1 ?= { save_scope_as = kehillah_synod_candidate_1 }
        var:kehillah_synod_candidate_2 ?= { save_scope_as = kehillah_synod_candidate_2 }
        var:kehillah_synod_candidate_3 ?= { save_scope_as = kehillah_synod_candidate_3 }
    }

    option = {
        name = kehillah_synod_nominate_1
        trigger = { exists = var:kehillah_synod_candidate_1 kehillah_synod_rite_headship_open_trigger = yes }
        custom_tooltip = kehillah_synod_nominate_tt
        involved_activity = { set_variable = { name = kehillah_synod_nominee value = root.var:kehillah_synod_candidate_1 } }
        kehillah_synod_put_motion_effect = { MOTION = flag:head_of_rite }
        ai_chance = { base = 50 }
    }

    option = {
        name = kehillah_synod_nominate_2
        trigger = { exists = var:kehillah_synod_candidate_2 kehillah_synod_rite_headship_open_trigger = yes }
        custom_tooltip = kehillah_synod_nominate_tt
        involved_activity = { set_variable = { name = kehillah_synod_nominee value = root.var:kehillah_synod_candidate_2 } }
        kehillah_synod_put_motion_effect = { MOTION = flag:head_of_rite }
        ai_chance = { base = 10 }
    }

    option = {
        name = kehillah_synod_nominate_3
        trigger = { exists = var:kehillah_synod_candidate_3 kehillah_synod_rite_headship_open_trigger = yes }
        custom_tooltip = kehillah_synod_nominate_tt
        involved_activity = { set_variable = { name = kehillah_synod_nominee value = root.var:kehillah_synod_candidate_3 } }
        kehillah_synod_put_motion_effect = { MOTION = flag:head_of_rite }
        ai_chance = { base = 10 }
    }
    option = { name = kehillah_synod_kind_back trigger_event = kehillah_synod.0110 ai_chance = { base = 0 } }
}

# ---------------------------------------------------------------------

# Doctrines of the host's rite that are not settled Torah or Talmudic law.
# A row is offered only when the rite does not hold its doctrine, and says
# what it would replace. Its controversy is scored in immediate from a STEP
# that depends on the rite's present doctrine (kehillah_synod_step_*_value)
# and the rite's divergence (kehillah_synod_score_doctrine_effect), shown as
# a band, and stored on the sitting when the motion is put.
# ---------------------------------------------------------------------
kehillah_synod.0109 = {
    type = character_event
    title = kehillah_synod.0109.t
    desc = kehillah_synod.0109.desc
    theme = faith
    trigger = { exists = involved_activity }
    immediate = {

        remove_variable = kehillah_synod_practice_1
        if = {
            limit = { rite = { rite_has_doctrine = doctrine_monogamy rite_can_choose_doctrine = doctrine_polygamy } }
            set_variable = { name = kehillah_synod_practice_1 value = doctrine:doctrine_polygamy }
            set_variable = { name = kehillah_synod_practice_old_1 value = doctrine:doctrine_monogamy }
            kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_polygamy STEP = 35 VAR = kehillah_synod_practice_cost_1 }
        }
        else_if = {
            limit = { rite = { rite_has_doctrine = doctrine_polygamy rite_can_choose_doctrine = doctrine_monogamy } }
            set_variable = { name = kehillah_synod_practice_1 value = doctrine:doctrine_monogamy }
            set_variable = { name = kehillah_synod_practice_old_1 value = doctrine:doctrine_polygamy }
            kehillah_synod_score_doctrine_effect = { DOCTRINE = doctrine_monogamy STEP = 10 VAR = kehillah_synod_practice_cost_1 }
        }
        var:kehillah_synod_practice_1 ?= { save_scope_as = kehillah_synod_practice_1 }
        var:kehillah_synod_practice_old_1 ?= { save_scope_as = kehillah_synod_practice_old_1 }

            limit = { rite = { rite_has_doctrine = kehillah_proselytism_discouraged rite_can_choose_doctrine = kehillah_proselytism_encouraged } }

            set_variable = { name = kehillah_synod_practice_3 value = doctrine:doctrine_clerical_gender_either }
