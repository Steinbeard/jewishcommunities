# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
#   0001        the 1066 Rhineland synod: Gershom's ordinances renewed
#   0100, 0105  a synod's motion (two pages: marriage, then communal order)
#   0101-0103   the host argues, a delegate votes, the ruling
# A synod is a Sitting of the Bet Din held with the Synod or Sanhedrin
# intent. kehillah_synod_open_session_effect starts it beside the docket;
# the conference's on_complete fires the ruling.
namespace = kehillah_synod

# ---------------------------------------------------------------------
# kehillah_synod.0001 -- THE RHINELAND SYNOD, 1066
# Fired from kehillah_synod_on_game_start, 30 days in, for the leader of
# Worms (Mainz if Worms is gone). Gershom died in 1028; this is the
# generation of his students' students ratifying his takkanot, which is
# what founds Minhag Ashkenaz as a rite of its own.
# ---------------------------------------------------------------------
kehillah_synod.0001 = {
	type = character_event
	title = kehillah_synod.0001.t
	desc = kehillah_synod.0001.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_synod_colleague
		animation = personality_rational
	}

	trigger = {
		faith = faith:rabbinic_faith
		NOR = {
			exists = global_var:kehillah_minhag_ashkenaz_founded
			exists = global_var:kehillah_gershom_deferred
		}
	}

	immediate = {
		save_scope_value_as = { name = kehillah_synod_level value = flag:regional }
		if = {
			limit = {
				exists = title:d_kehillah_mainz.holder
				NOT = { title:d_kehillah_mainz.holder = root }
			}
			title:d_kehillah_mainz.holder = { save_scope_as = kehillah_synod_colleague }
		}
		else_if = {
			limit = { exists = title:d_kehillah_speyer.holder }
			title:d_kehillah_speyer.holder = { save_scope_as = kehillah_synod_colleague }
		}
	}

	# RENEW BOTH: no second wife, and no divorce against her will.
	option = {
		name = kehillah_synod.0001.a
		custom_tooltip = kehillah_synod_found_ashkenaz_tt
		hidden_effect = { kehillah_found_minhag_ashkenaz_effect = yes }
		custom_tooltip = kehillah_synod_divorce_ordinance_tt
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_gershom_divorce }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_1066_both_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_1066_both_greatness }
		ai_chance = { base = 70 }
	}

	# RENEW THE BAN ON A SECOND WIFE ALONE.
	option = {
		name = kehillah_synod.0001.b
		custom_tooltip = kehillah_synod_found_ashkenaz_tt
		hidden_effect = { kehillah_found_minhag_ashkenaz_effect = yes }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_1066_ban_greatness }
		ai_chance = { base = 25 }
	}

	# LEAVE EACH COMMUNITY TO ITS CUSTOM.
	option = {
		name = kehillah_synod.0001.c
		custom_tooltip = kehillah_synod_defer_tt
		set_global_variable = { name = kehillah_gershom_deferred value = yes }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_1066_defer_greatness }
		ai_chance = { base = 5 }
	}

}

# ---------------------------------------------------------------------
# kehillah_synod.0003 -- NASI GOVERNMENT RE-CHECK (hidden)
# A day after the Sanhedrin is restored or changes hands, or a community
# changes hands under a Nasi government: the engine settles governments
# after effects return, so the transition is checked again
# (kehillah_reconcile_nasi_government_effect, V46 spec section 9).
# ---------------------------------------------------------------------
kehillah_synod.0003 = {
	type = character_event
	hidden = yes

	immediate = {
		kehillah_reconcile_nasi_government_effect = yes
		if = {
			limit = { has_government = kehillah_nasi_government }
			debug_log = "KEHILLAH_NASI_GOV: day-after check, on the Nasi government"
		}
		else_if = {
			limit = { has_government = kehillah_government }
			debug_log = "KEHILLAH_NASI_GOV: day-after check, on the Kehillah government"
		}
		else = { debug_log = "KEHILLAH_NASI_GOV: day-after check, on ANOTHER government" }
		debug_log_scopes = yes
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0002 -- FOUNDING MINHAG ASHKENAZ (hidden)
# Scope: the founder. See kehillah_found_minhag_ashkenaz_effect for why the
# founding runs here and not in the option that asks for it.
# ---------------------------------------------------------------------
kehillah_synod.0002 = {
	type = character_event
	hidden = yes

	trigger = {
		NOT = { exists = global_var:kehillah_minhag_ashkenaz_founded }
	}

	immediate = {
		kehillah_found_minhag_ashkenaz_now_effect = yes
		if = {
			limit = { exists = global_var:kehillah_minhag_ashkenaz_founded }
			kehillah_notify_region_players_effect = { TYPE = msg_kehillah_community_news TITLE = kehillah_news_synod_t DESC = kehillah_news_synod_ashkenaz_d }
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0101 -- THE HOST ARGUES THE MOTION
# A won argument marks the activity, and the delegates' votes lean on
# that mark.
# ---------------------------------------------------------------------
kehillah_synod.0101 = {
	type = character_event
	title = kehillah_synod.0101.t
	desc = {
		desc = kehillah_synod.0101.desc
		first_valid = {
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:herem_polygamy } desc = kehillah_synod_motion_herem_polygamy }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:herem_all_israel } desc = kehillah_synod_motion_herem_all_israel }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:ketubbah_clause } desc = kehillah_synod_motion_ketubbah_clause }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:gentile_courts } desc = kehillah_synod_motion_gentile_courts }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:informers } desc = kehillah_synod_motion_informers }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:hayishuv } desc = kehillah_synod_motion_hayishuv }
			desc = kehillah_synod_motion_letters
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		exists = involved_activity
		involved_activity = { has_variable = kehillah_synod_takkanah }
	}

	# From the sources: the sages before us.
	option = {
		name = kehillah_synod.0101.a
		duel = {
			skill = learning
			value = high_skill_rating
			50 = {
				desc = kehillah_synod_argument_won
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				modifier = {
					add = 10
					has_trait = theologian
				}
				min = 5
				involved_activity ?= { set_variable = kehillah_synod_persuaded }
			}
			50 = {
				desc = kehillah_synod_argument_lost
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 25 learning >= high_skill_rating }
		}
	}

	# From the community's need: what the people suffer without it.
	option = {
		name = kehillah_synod.0101.b
		duel = {
			skill = diplomacy
			value = high_skill_rating
			50 = {
				desc = kehillah_synod_argument_won
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				modifier = {
					add = 10
					has_trait = just
				}
				min = 5
				involved_activity ?= { set_variable = kehillah_synod_persuaded }
			}
			50 = {
				desc = kehillah_synod_argument_lost
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 25 diplomacy >= high_skill_rating }
		}
	}

	# Say nothing; let the delegates weigh it themselves.
	option = {
		name = kehillah_synod.0101.c
		ai_chance = { base = 5 }
	}

	after = {
		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				every_attending_character = {
					limit = { kehillah_synod_delegate_trigger = { HOST = root } }
					trigger_event = { id = kehillah_synod.0102 days = kehillah_synod_vote_delay_days }
				}
			}
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_synod.0102 -- A DELEGATE VOTES
# ---------------------------------------------------------------------
kehillah_synod.0102 = {
	type = character_event
	title = kehillah_synod.0102.t
	desc = {
		desc = kehillah_synod.0102.desc
		first_valid = {
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:herem_polygamy } desc = kehillah_synod_motion_herem_polygamy }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:herem_all_israel } desc = kehillah_synod_motion_herem_all_israel }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:ketubbah_clause } desc = kehillah_synod_motion_ketubbah_clause }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:gentile_courts } desc = kehillah_synod_motion_gentile_courts }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:informers } desc = kehillah_synod_motion_informers }
			triggered_desc = { trigger = { involved_activity.var:kehillah_synod_takkanah ?= flag:hayishuv } desc = kehillah_synod_motion_hayishuv }
			desc = kehillah_synod_motion_letters
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_synod_host
		animation = personality_bold
	}

	trigger = {
		exists = involved_activity
		involved_activity = { has_variable = kehillah_synod_takkanah }
	}

	immediate = {
		involved_activity.activity_host = { save_scope_as = kehillah_synod_host }
	}

	# FOR.
	option = {
		name = kehillah_synod.0102.a
		involved_activity ?= { change_variable = { name = kehillah_synod_votes_for add = 1 } }
		ai_chance = {
			base = 50
			opinion_modifier = {
				who = root
				opinion_target = scope:kehillah_synod_host
				multiplier = 0.5
			}
			modifier = {
				add = 25
				involved_activity ?= { has_variable = kehillah_synod_persuaded }
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:herem_polygamy
				OR = { has_trait = zealous has_trait = chaste }
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:gentile_courts
				has_trait = just
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:informers
				has_trait = paranoid
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:hayishuv
				has_trait = greedy
			}
			min = 1
		}
	}

	# AGAINST.
	option = {
		name = kehillah_synod.0102.b
		involved_activity ?= { change_variable = { name = kehillah_synod_votes_against add = 1 } }
		ai_chance = {
			base = 40
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:herem_polygamy
				has_trait = lustful
			}
			modifier = {
				add = 20
				involved_activity.var:kehillah_synod_takkanah ?= flag:hayishuv
				has_trait = generous
			}
			modifier = {
				add = 15
				has_trait = stubborn
			}
			min = 1
		}
	}
}


# ---------------------------------------------------------------------
# kehillah_synod.0103 -- THE RULING (host)
# Fired from the Bet Din's on_complete, so every synod reaches it and
# nothing chains past it. Everything that reads the activity is settled in
# immediate, while the activity still exists: the player may answer days
# later. The host's own vote is the 1 seeded into votes_for.
# ---------------------------------------------------------------------
kehillah_synod.0103 = {
	type = character_event
	title = kehillah_synod.0103.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:kehillah_synod_outcome = flag:no_quorum }
				desc = kehillah_synod.0103.desc.no_quorum
			}
			triggered_desc = {
				trigger = { scope:kehillah_synod_outcome = flag:passed }
				desc = kehillah_synod.0103.desc.passed
			}
			desc = kehillah_synod.0103.desc.failed
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		exists = involved_activity
		involved_activity = { has_variable = kehillah_synod_takkanah }
	}

	immediate = {
		save_scope_value_as = {
			name = kehillah_synod_takkanah
			value = involved_activity.var:kehillah_synod_takkanah
		}
		save_scope_value_as = {
			name = kehillah_synod_level
			value = involved_activity.var:kehillah_synod_level
		}
		if = {
			limit = { kehillah_synod_quorum_present_trigger = no }
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:no_quorum }
		}
		else_if = {
			limit = { kehillah_synod_motion_carried_trigger = yes }
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:passed }
		}
		else = {
			save_scope_value_as = { name = kehillah_synod_outcome value = flag:failed }
		}
		remove_variable = kehillah_synod_in_session
	}

	option = {
		name = kehillah_synod.0103.a
		if = {
			limit = { scope:kehillah_synod_outcome = flag:passed }
			kehillah_synod_enact_takkanah_effect = yes
		}
		else_if = {
			limit = { scope:kehillah_synod_outcome = flag:failed }
			kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_failed_greatness }
		}
	}

	after = {
		if = {
			limit = { scope:kehillah_synod_outcome = flag:passed }
			kehillah_notify_region_players_effect = { TYPE = msg_kehillah_community_news TITLE = kehillah_news_synod_t DESC = kehillah_news_synod_passed_d }
		}
	}
}
