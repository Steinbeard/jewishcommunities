	type = character_event

	theme = faith
	left_portrait = { character = root animation = thinking }

	immediate = {

	}
	option = {
		name = kehillah_din_review
		if = { limit = { OR = { kehillah_judicial_learning_value >= 10 has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 30 } } } involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } } }
		else = { involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } } }
		custom_tooltip = kehillah_din_review_tt
		ai_chance = { base = 35 }
	}
	option = {
		name = kehillah_din_source_10
		# 2026-11-23: shown greyed out when no judge knows a source.
		trigger = { custom_description = { text = kehillah_din_no_source_tt kehillah_din_can_cite_trigger = { CASE = 10 JUDGE = root } } }
		show_as_unavailable = { always = yes }
		custom_tooltip = kehillah_din_source_10_tt
		involved_activity = { kehillah_bet_din_default_direction_effect = { CASE = 10 VALUE = 1 } }
		kehillah_din_cite_source_effect = { CASE = 10 JUDGE = root }
		ai_chance = { base = 45 }
	}
	option = {
		name = kehillah_din_combo_10
		trigger = { root = { OR = { has_trait = compassionate has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 30 } } diplomacy >= 12 } OR = { involved_activity.var:kehillah_bet_din_av_beit_din = { NOT = { this = root } stewardship >= 12 } involved_activity.var:kehillah_bet_din_judge2 = { NOT = { this = root } stewardship >= 12 } involved_activity.var:kehillah_bet_din_judge3 = { NOT = { this = root } stewardship >= 12 } } }
		custom_tooltip = kehillah_din_combo_10_tt
		involved_activity = { kehillah_bet_din_default_direction_effect = { CASE = 10 VALUE = 2 } change_variable = { name = kehillah_bet_din_tally add = kehillah_din_combination_tally } }
		ai_chance = { base = 40 }
	}

	after = { if = { limit = { exists = involved_activity.var:kehillah_bet_din_convening_title.holder } involved_activity.var:kehillah_bet_din_convening_title.holder = { trigger_event = { id = kehillah_bet_din.0103 days = 1 } } } }
}
