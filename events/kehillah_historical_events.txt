namespace = kehillah_historical

# Hidden follow-ups for vanilla's spawned historical Jews, fired by the
# override in common/scripted_effects/kehillah_historical_spawn_effects.txt.

# A day after the spawn: the rabbis among them hold the Rabbi trait.
kehillah_historical.1 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_alive = yes
				is_rabbinic_authority_jewish_trigger = yes
				OR = {
					has_variable = rashi
					has_variable = maimonides
					has_variable = saadia
					has_variable = ezra
					has_variable = nachmanides
					has_variable = crescas
					has_variable = gersonides
				}
			}
			kehillah_grant_rabbi_trait_effect = yes
		}
		# Ten days in all, so a player who was offered the figure has answered.
		trigger_event = { id = kehillah_historical.2 days = 9 }
	}
}
