# Console-only probes, intended for a disposable save on a Kehillah leader.
# event kehillah_sermon_debug.1 reports eligibility and track tier.
# event kehillah_sermon_debug.2 raises Parshanut to 65 and opens the choice.
namespace = kehillah_sermon_debug

kehillah_sermon_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_sermon_debug.1: START"
		if = { limit = { government_has_flag = government_is_kehillah has_trait = kehillah_rabbi_trait exists = primary_title } debug_log = "kehillah_sermon_debug.1: eligible rabbinic leader" }
		else = { debug_log = "kehillah_sermon_debug.1: eligibility fails" }
		if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 } } debug_log = "kehillah_sermon_debug.1: master reward" }
		else_if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 65 } } debug_log = "kehillah_sermon_debug.1: adept reward" }
		else_if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 30 } } debug_log = "kehillah_sermon_debug.1: studied reward" }
		else = { debug_log = "kehillah_sermon_debug.1: novice reward" }
	}
}

kehillah_sermon_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { government_has_flag = government_is_kehillah has_trait = kehillah_rabbi_trait exists = primary_title }
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = 65 }
			trigger_event = { id = kehillah_sermon.0001 days = 0 }
			debug_log = "kehillah_sermon_debug.2: sermon opened; inspect title pillar before and after choice"
		}
		else = { debug_log = "kehillah_sermon_debug.2: not a rabbinic Kehillah leader" }
	}
}
