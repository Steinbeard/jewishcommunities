# Console-only probes, intended for a disposable save on a Kehillah leader
# (2026-11-02; rebuilt 2026-11-11 with the sermons,
# docs/spec/2026-11-11-rulings-sermons-and-gates.md section 3).
# event kehillah_sermon_debug.1 reports who may preach, the reward tier, and
#   which texts fit the weekly portion (read-only).
# event kehillah_sermon_debug.2 raises Parshanut to 65 and begins a sermon.
namespace = kehillah_sermon_debug

kehillah_sermon_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_sermon_debug.1: START"
		if = { limit = { kehillah_sermon_preacher_trigger = yes } debug_log = "kehillah_sermon_debug.1: you may preach" }
		else = { debug_log = "kehillah_sermon_debug.1: you may not preach" }
		if = { limit = { cp:councillor_court_chaplain ?= { kehillah_sermon_preacher_trigger = yes } } debug_log = "kehillah_sermon_debug.1: your Dayyan may preach" }
		else = { debug_log = "kehillah_sermon_debug.1: no Dayyan who may preach" }
		if = { limit = { kehillah_sermon_knows_torah_trigger = yes } debug_log = "kehillah_sermon_debug.1: knows the Torah" }
		if = { limit = { kehillah_sermon_time_of_trouble_trigger = yes } debug_log = "kehillah_sermon_debug.1: a time of trouble" }
		if = { limit = { kehillah_sermon_parshanut_gain_value >= kehillah_sermon_pillar_gain_master } debug_log = "kehillah_sermon_debug.1: master reward" }
		else_if = { limit = { kehillah_sermon_parshanut_gain_value >= kehillah_sermon_pillar_gain_adept } debug_log = "kehillah_sermon_debug.1: adept reward" }
		else_if = { limit = { kehillah_sermon_parshanut_gain_value >= kehillah_sermon_pillar_gain_studied } debug_log = "kehillah_sermon_debug.1: studied reward" }
		else = { debug_log = "kehillah_sermon_debug.1: novice reward" }
		save_scope_as = kehillah_sermon_holder
		kehillah_sermon_pick_portion_effect = yes
		if = { limit = { var:kehillah_sermon_cand_grounded ?= flag:work_tanakh_torah } debug_log = "kehillah_sermon_debug.1: PORTION GROUNDED the Torah" }
		else_if = { limit = { exists = var:kehillah_sermon_cand_grounded } debug_log = "kehillah_sermon_debug.1: PORTION GROUNDED another work" }
		else = { debug_log = "kehillah_sermon_debug.1: PORTION GROUNDED none" }
		if = { limit = { exists = var:kehillah_sermon_cand_newer } debug_log = "kehillah_sermon_debug.1: PORTION NEWER yes" }
		else = { debug_log = "kehillah_sermon_debug.1: PORTION NEWER none" }
		kehillah_sermon_clear_effect = yes
	}
}

kehillah_sermon_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { government_has_flag = government_is_kehillah kehillah_sermon_leader_can_preach_trigger = yes }
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = 65 }
			}
			kehillah_sermon_begin_effect = yes
			debug_log = "kehillah_sermon_debug.2: sermon begun; inspect the title's pillars before and after"
		}
		else = { debug_log = "kehillah_sermon_debug.2: no one here may preach" }
	}
}
