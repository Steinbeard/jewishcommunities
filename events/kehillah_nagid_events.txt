# =====================================================================
# THE NAGID OFFICE -- the Nagid spec, sections 7, 8 and 13 (docs/spec/
# 2026-10-30-the-nagid.md), 2026-10-30, slice 4.
#   .0001 the audience: the ruler confirms the seat's leader as Nagid and,
#         since 2026-11-01, names his place at court in the same choice
#         (kehillah_nagid_request_confirmation_effect, or founding the
#         office). Scopes: kehillah_nagid_seat_leader, kehillah_nagid_host.
#   .0002 a player ruler hears the seat's petition to found the office
#         (kehillah_ask_nagid_office_decision); founding it leads into
#         the audience. Scope: kehillah_nagid_seat_leader.
#   .0003 slice 5: a player ruler changes his Nagid's place at court (the
#         Change His Post interaction, a Vizier off the Dhimma, or a
#         founding with no audience). No scopes needed: the Nagid is read
#         from the office.
#   .0090 game start, 1066: Joseph ibn Naghrilla as Nagid of Badis's
#         realm, which starts Favored. Scope: Granada's leader.
#   .0091 game start: the office founded, vacant, in the realm of this
#         seat (Fustat in 1066 and 1178). Scope: the seat's leader.
# The office's name follows the ruler (KehillahNagidOfficeName).
# =====================================================================
namespace = kehillah_nagid

kehillah_nagid.0001 = {
	type = character_event
	title = kehillah_nagid.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:kehillah_nagid_seat_leader = { kehillah_is_nagid_trigger = yes } }
				desc = kehillah_nagid.0001.reconfirm_desc
			}
			desc = kehillah_nagid.0001.desc
		}
		triggered_desc = {
			trigger = { kehillah_nagid_vizier_open_trigger = yes }
			desc = kehillah_nagid.0003.desc_vizier
		}
	}
	theme = realm
	override_background = { reference = throne_room }
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_nagid_seat_leader
		animation = personality_honorable
	}

	trigger = {
		exists = scope:kehillah_nagid_seat_leader
		scope:kehillah_nagid_seat_leader = {
			is_alive = yes
			kehillah_nagid_can_hold_trigger = yes
		}
		kehillah_realm_has_nagid_office_trigger = yes
	}

	# The audience (2026-11-01): a player confirms the leader and names his
	# place at court in one choice. Each option pays the fee.
	option = {
		name = kehillah_nagid.0001.vizier
		trigger = { is_ai = no kehillah_nagid_vizier_open_trigger = yes }
		kehillah_nagid_confirm_as_effect = { SERVICE = vizier }
	}
	option = {
		name = kehillah_nagid.0001.treasurer
		trigger = { is_ai = no }
		kehillah_nagid_confirm_as_effect = { SERVICE = treasurer }
	}
	option = {
		name = kehillah_nagid.0001.physician
		trigger = { is_ai = no }
		kehillah_nagid_confirm_as_effect = { SERVICE = physician }
	}
	option = {
		name = kehillah_nagid.0001.envoy
		trigger = { is_ai = no }
		kehillah_nagid_confirm_as_effect = { SERVICE = envoy }
	}
	option = {
		name = kehillah_nagid.0001.a
		trigger = { is_ai = no }
		kehillah_nagid_confirm_effect = yes
	}

	# An AI ruler confirms and names the service by rule
	# (kehillah_nagid_service_ai_pick_effect), as before the audience.
	option = {
		name = kehillah_nagid.0001.a
		trigger = { is_ai = yes }
		kehillah_nagid_confirm_effect = yes
		hidden_effect = { kehillah_nagid_service_ai_pick_effect = yes }
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = greedy }
			modifier = { add = 20 stewardship >= 12 }
			modifier = { add = 20 kehillah_outlook_band_value >= 1 }
		}
	}

	# Withhold confirmation for now.
	option = {
		name = kehillah_nagid.0001.b
		primary_title = {
			set_variable = { name = kehillah_nagid_withheld value = yes years = kehillah_nagid_withheld_years }
		}
		scope:kehillah_nagid_seat_leader = {
			remove_variable = kehillah_nagid_asked
			hidden_effect = { kehillah_realm_news_add_effect = { KIND = nagid_withheld } }
			if = {
				limit = { kehillah_is_nagid_trigger = yes }
				kehillah_nagid_step_down_effect = yes
			}
			send_interface_message = {
				type = msg_kehillah_community_news_bad
				title = kehillah_nagid_withheld_msg_t
				desc = kehillah_nagid_withheld_msg_d
				left_icon = root
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = 30 kehillah_outlook_band_value <= -1 }
			modifier = { add = 20 has_trait = zealous }
			modifier = {
				add = 20
				OR = {
					kehillah_teaching_strict_trigger = yes
					kehillah_teaching_exclusionary_trigger = yes
				}
			}
		}
	}
}

kehillah_nagid.0002 = {
	type = character_event
	title = kehillah_nagid.0002.t
	desc = kehillah_nagid.0002.desc
	theme = realm
	left_portrait = {
		character = scope:kehillah_nagid_seat_leader
		animation = personality_rational
	}

	trigger = {
		exists = scope:kehillah_nagid_seat_leader
		scope:kehillah_nagid_seat_leader = { is_alive = yes }
		kehillah_nagid_can_found_trigger = yes
	}

	# Found the office, then receive its leader in audience (.0001).
	option = {
		name = kehillah_nagid.0002.a
		kehillah_nagid_found_office_effect = yes
		custom_tooltip = kehillah_nagid_audience_follows_tt
		trigger_event = kehillah_nagid.0001
	}

	option = {
		name = kehillah_nagid.0002.b
		scope:kehillah_nagid_seat_leader = {
			send_interface_message = {
				type = msg_kehillah_community_news_bad
				title = kehillah_nagid_petition_refused_msg_t
				desc = kehillah_nagid_petition_refused_msg_d
				left_icon = root
			}
		}
	}
}

		}
	}

	option = {
		name = kehillah_nagid.0003.treasurer
		trigger = {
			NOT = { kehillah_nagid_service_is_trigger = { SERVICE = treasurer } }
		}
		kehillah_nagid_set_service_effect = { SERVICE = treasurer }
		ai_chance = {
			base = 10
			modifier = {
				add = 30
				scope:kehillah_nagid_seat_leader = { stewardship >= kehillah_nagid_service_tier_3_skill }
			}
			modifier = { add = 20 has_trait = greedy }
		}
	}

	option = {
		name = kehillah_nagid.0003.physician
		trigger = {
			NOT = { kehillah_nagid_service_is_trigger = { SERVICE = physician } }
		}
		kehillah_nagid_set_service_effect = { SERVICE = physician }
		ai_chance = {
			base = 10
			modifier = {
				add = 30
				scope:kehillah_nagid_seat_leader = { learning >= kehillah_nagid_service_tier_3_skill }
			}
			modifier = { add = 20 has_trait = zealous }
		}
	}

	option = {
		name = kehillah_nagid.0003.envoy
		trigger = {
			NOT = { kehillah_nagid_service_is_trigger = { SERVICE = envoy } }
		}
		kehillah_nagid_set_service_effect = { SERVICE = envoy }
		ai_chance = {
			base = 10
			modifier = {
				add = 30
				scope:kehillah_nagid_seat_leader = { diplomacy >= kehillah_nagid_service_tier_3_skill }
			}
		}
	}

	# Keep the service already named.
	option = {
		name = kehillah_nagid.0003.keep
		trigger = { kehillah_nagid_in_service_trigger = yes }
		ai_chance = { base = 20 }
	}

	# No service at court: the office stands without one.
	option = {
		name = kehillah_nagid.0003.none
		kehillah_nagid_clear_service_effect = yes
		ai_chance = { base = 0 }
	}
}

kehillah_nagid.0090 = {
	type = character_event
	hidden = yes
	trigger = {
		government_has_flag = government_is_kehillah
		kehillah_is_realm_seat_trigger = yes
	}
	immediate = {
		save_scope_as = kehillah_nagid_seat_leader
		domicile.domicile_location.county.holder.top_liege = {
			kehillah_set_settlement_law_effect = { RUNG = encouraged VALUE = 3 }
			kehillah_favored_cost_refresh_effect = yes
			if = {
				limit = { NOT = { kehillah_realm_has_nagid_office_trigger = yes } }
				kehillah_nagid_found_office_effect = yes
			}
			kehillah_nagid_appoint_effect = yes
			# Joseph was Badis's vizier, as his father was Habbus's. A
			# player Badis can change it with the Change His Post
			# interaction.
			if = {
				limit = {
					kehillah_nagid_vizier_open_trigger = yes
					NOT = { kehillah_nagid_service_is_trigger = { SERVICE = vizier } }
				}
				kehillah_nagid_set_service_effect = { SERVICE = vizier }
			}
		}
		debug_log = "kehillah_nagid.0090: the 1066 Nagid of Granada seated"
	}
}

	}
