# event kehillah_sefer_v2_debug.9 -- make the player meet the general
# (non-Rashi) path of kehillah_sefer_ambition_rashi_torah_trigger, so the
# gold option shows in Write a Sefer (kehillah_book.0001): the Rabbi trait,
# Learning 14, Parshanut 60, and the Torah, Targum Onkelos and Genesis Rabbah
# mastered. Run before .5, which writes Rashi on the Torah. Changes the save.
kehillah_sefer_v2_debug.9 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { NOT = { has_trait = kehillah_rabbi_trait } }
			add_trait = kehillah_rabbi_trait
		}
		add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = 60 }
		while = {
			limit = { learning < 14 }
			add_learning_skill = 1
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_tanakh_torah } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_tanakh_torah }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_genesis_rabbah } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_genesis_rabbah }
		}
		if = { limit = { kehillah_sefer_can_write_trigger = yes } debug_log = "kehillah_sefer_v2_debug.9 CAN WRITE yes" }
		else = { debug_log = "kehillah_sefer_v2_debug.9 CAN WRITE no" }
		if = { limit = { kehillah_sefer_ambition_rashi_torah_trigger = yes } debug_log = "kehillah_sefer_v2_debug.9 AMBITION rashi_torah met" }
		else = { debug_log = "kehillah_sefer_v2_debug.9 AMBITION rashi_torah NOT met (already written? rite not Rabbinic?)" }
	}
}

# event kehillah_sefer_v2_debug.10 -- take Talmud Bavli: Moed off the
# player's library and refresh its cache, so the Greatness tooltip shows
# the "Bavli orders missing from the library" row (-15 per order) instead
# of "A Complete Shas". Changes the save.
kehillah_sefer_v2_debug.10 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_lib_refresh_own_library_effect = yes
		if = {
			limit = { exists = var:kehillah_own_library }
			var:kehillah_own_library = {
				if = {
					limit = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_moed } }
					remove_list_variable = { name = kehillah_library_works target = flag:work_bavli_moed }
				}
				kehillah_lib_refresh_cache_effect = yes
				if = { limit = { var:kehillah_lib_foundation_missing ?= 1 } debug_log = "kehillah_sefer_v2_debug.10 MOED removed: one order missing" }
				else = { debug_log = "kehillah_sefer_v2_debug.10 cache does not show exactly one missing order" }
			}
		}
		else = { debug_log = "kehillah_sefer_v2_debug.10 NO own library" }
	}
}

# event kehillah_sefer_v2_debug.11 -- the tenet mechanics outside the books
# (docs/spec/2026-11-08-sefer-tenet-mechanics.md): give the player Pesak and
# Avelei Tziyon as personal tenets (force_set_personal_tenet drops the worst
# one if the slots are full), then check that Pesak raised his judicial
# Learning by 2 and that Send Alms to Jerusalem is open to him. Changes the save.
kehillah_sefer_v2_debug.11 = {
	type = character_event
	hidden = yes
	immediate = {
		save_temporary_scope_value_as = { name = kehillah_judicial_before value = kehillah_judicial_learning_value }
		if = {
			limit = { NOT = { has_personal_tenet = tenet_kehillah_pesak } }
			force_set_personal_tenet = tenet:tenet_kehillah_pesak
		}
		if = {
			limit = { has_personal_tenet_flag = kehillah_tenet_pesak_bench_personal }
			debug_log = "kehillah_sefer_v2_debug.11 PESAK personal flag yes"
		}
		else = { debug_log = "kehillah_sefer_v2_debug.11 PESAK personal flag NO (no personal tenet slot?)" }
		if = {
			limit = { kehillah_judicial_learning_value >= { value = scope:kehillah_judicial_before add = 2 } }
			debug_log = "kehillah_sefer_v2_debug.11 PESAK judicial Learning +2 (or already counted)"
		}
		else = { debug_log = "kehillah_sefer_v2_debug.11 PESAK judicial Learning unchanged" }
		if = {
			limit = { NOT = { has_personal_tenet = tenet_kehillah_avelei_tziyon } }
			force_set_personal_tenet = tenet:tenet_kehillah_avelei_tziyon
		}
		if = {
			limit = { kehillah_tenet_mourns_zion_trigger = yes }
			debug_log = "kehillah_sefer_v2_debug.11 ZION alms decision open"
		}
		else = { debug_log = "kehillah_sefer_v2_debug.11 ZION alms decision closed" }
	}
}

# event kehillah_sefer_v2_debug.12 -- Karaites have no clergy (read-only).
# For the first Karaite community leader (else the player): is he Karaite,
# does he pass the rabbi-or-Karaite gates, is he still never a rabbi, never
# clergy, and who is his community's rabbi. A Karaite should log KARAITE yes,
# GATES yes yes, RABBI no no, CLERGY no (unless he sits as a Dayyan).
kehillah_sefer_v2_debug.12 = {
	type = character_event
	hidden = yes
	immediate = {
		random_in_global_list = {
			variable = kehillah_registered_communities
			limit = { holder ?= { kehillah_is_karaite_character_trigger = yes } }
			holder = { save_scope_as = kehillah_dbg_karaite }
		}
		if = {
			limit = { NOT = { exists = scope:kehillah_dbg_karaite } }
			save_scope_as = kehillah_dbg_karaite
			debug_log = "kehillah_sefer_v2_debug.12 no Karaite community: probing the player"
		}
		scope:kehillah_dbg_karaite = {
			debug_log = "kehillah_sefer_v2_debug.12 START"
			debug_log_scopes = yes
			if = { limit = { kehillah_is_karaite_character_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 KARAITE yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 KARAITE no" }
			if = { limit = { kehillah_rabbi_or_karaite_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 GATE rabbi_or_karaite yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 GATE rabbi_or_karaite no" }
			if = { limit = { kehillah_ordained_or_karaite_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 GATE ordained_or_karaite yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 GATE ordained_or_karaite no" }
			if = { limit = { has_trait = kehillah_rabbi_trait } debug_log = "kehillah_sefer_v2_debug.12 RABBI trait yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 RABBI trait no" }
			if = { limit = { kehillah_is_ordained_rabbi_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 RABBI ordained yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 RABBI ordained no" }
			if = { limit = { is_clergy = yes } debug_log = "kehillah_sefer_v2_debug.12 CLERGY yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 CLERGY no" }
			if = { limit = { kehillah_leader_is_rav_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 STYLED Rav (wrong for a Karaite)" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 STYLED not Rav" }
			if = { limit = { kehillah_torah_reader_eligible_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 STUDY open" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 STUDY closed" }
			# Who kehillah_karaite_scholar_xp_effect pays: Toraic and holding the trait.
			if = {
				limit = { kehillah_is_karaite_character_trigger = yes has_trait = lifestyle_scholar }
				debug_log = "kehillah_sefer_v2_debug.12 SCHOLAR XP yes: sermons, Bet Din rulings, study and siyum add Religious Scholar"
			}
			else_if = {
				limit = { kehillah_is_karaite_character_trigger = yes }
				debug_log = "kehillah_sefer_v2_debug.12 SCHOLAR XP no: lacks Religious Scholar (Examine Scripture earns it)"
			}
			else = { debug_log = "kehillah_sefer_v2_debug.12 SCHOLAR XP n/a: not Toraic" }
			if = { limit = { kehillah_journey_can_plan_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 YESHIVA VISIT open" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 YESHIVA VISIT closed (cooldown, prison or not a leader?)" }
			if = { limit = { kehillah_has_community_rabbi_trigger = yes } debug_log = "kehillah_sefer_v2_debug.12 COMMUNITY RABBI yes" }
			else = { debug_log = "kehillah_sefer_v2_debug.12 COMMUNITY RABBI no" }
			kehillah_save_community_rabbi_effect = yes
			if = {
