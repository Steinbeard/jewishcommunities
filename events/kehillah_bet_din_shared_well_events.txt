			desc = kehillah_bet_din.0414.fact_false

			triggered_desc = { trigger = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great } desc = kehillah_bet_din.0414.quality_great }

			desc = kehillah_bet_din.0414.quality_poor

	right_portrait = { character = scope:kbd_litigant_a animation = worry }
	lower_right_portrait = { character = scope:kbd_litigant_b animation = disapproval }
	trigger = { exists = involved_activity involved_activity.var:kehillah_din_case = 41 }
	immediate = {
		# The hidden fact changes the risk of the first and third rulings.
		# The investigation remains the dependable, costlier middle course.
		if = {
			limit = { involved_activity.var:kehillah_din_case41_shop_use_allowed = 1 involved_activity.var:kehillah_bet_din_case41_direction = 1 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 6 } }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_din_case41_shop_use_allowed = 0 involved_activity.var:kehillah_bet_din_case41_direction = 3 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 2 } }
		}
		if = {
			limit = { involved_activity.var:kehillah_bet_din_case41_direction = 3 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 3 } }
		}
		if = {
			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability add = 41 } change_variable = { name = kehillah_var_greatness add = 12 } }
			kehillah_bet_din_record_case_result_effect = { SCORE = 2 CASE = 41 TIER = great TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			kehillah_bet_din_reward_judges_for_ruling_effect = { XP = 3 }

			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_good }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability add = 7 } change_variable = { name = kehillah_var_greatness add = 4 } }

			kehillah_bet_din_reward_judges_for_ruling_effect = { XP = 1 }
		}
		else = {
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 10 } change_variable = { name = kehillah_var_greatness subtract = 4 } }
			kehillah_bet_din_record_case_result_effect = { SCORE = -1 CASE = 41 TIER = poor TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
		}
		if = {

			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 8 } }

			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 4 } }

		}

			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 2 } }
			custom_tooltip = kehillah_din_case41_three_tt
		}
		kehillah_din_case_party_response_effect = { DIRECTION = kehillah_bet_din_case41_direction }
	}
	option = { name = kehillah_din_record ai_chance = { base = 30 } }
	after = { kehillah_bet_din_advance_docket_effect = yes }
