# Console-only, read-only collection probe. Run as a Kehillah leader:
#   event kehillah_sefer_debug.1
# Inspect debug.log for tier, held work flags, and personally mastered
# work flags. This reads the same title and character lists as the panel.
namespace = kehillah_sefer_debug

kehillah_sefer_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_sefer_debug.1 START"
		if = {
			limit = { exists = primary_title }
			primary_title = {
				if = {
					limit = { holder = { domicile ?= { has_domicile_building_or_higher = kehillah_beit_midrash_03 } } }
					debug_log = "kehillah_sefer_debug.1 TIER 3"
				}
				else_if = {
					limit = { holder = { domicile ?= { has_domicile_building_or_higher = kehillah_beit_midrash_02 } } }
					debug_log = "kehillah_sefer_debug.1 TIER 2"
				}
				else = { debug_log = "kehillah_sefer_debug.1 TIER 1 OR NO HALL" }
				if = {
					limit = { has_variable_list = kehillah_library_works }
					every_in_list = {
						variable = kehillah_library_works
						debug_log = "kehillah_sefer_debug.1 HELD WORK"
						debug_log_scopes = yes
					}
				}
			}
		}
		if = {
			limit = { has_variable_list = kehillah_mastered_works }
			every_in_list = {
				variable = kehillah_mastered_works
				debug_log = "kehillah_sefer_debug.1 MASTERED WORK"
				debug_log_scopes = yes
			}
		}
		debug_log = "kehillah_sefer_debug.1 END"
	}
}

# In a fresh test save, make the personal-mastery foldout reachable.
# Console only: event kehillah_sefer_debug.2
kehillah_sefer_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_mishnah } } }
			add_to_variable_list = { name = kehillah_studied_works target = flag:work_mishnah }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }
		}
		debug_log = "kehillah_sefer_debug.2: gave this character Mishnah mastery for the library UI check"
	}
}

# Rebuild and inspect the acquisition catalog without starting a commission:
#   event kehillah_sefer_debug.4
kehillah_sefer_debug.4 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_sefer_debug.4 ACQUISITION CANDIDATES START"
		kehillah_library_refresh_acquisition_candidates_effect = yes
		primary_title = {
			every_in_list = {
				variable = kehillah_library_acquisition_candidates
				debug_log = "kehillah_sefer_debug.4 AVAILABLE MISSING WORK"
				debug_log_scopes = yes
			}
			if = {
				limit = {
					current_date < 1080.1.1
					is_target_in_variable_list = { name = kehillah_library_acquisition_candidates target = flag:work_chovot_halevavot }
				}
				debug_log = "kehillah_sefer_debug.4 FAIL CHOVOT SHOWN BEFORE CIRCULATION"
			}
			else = { debug_log = "kehillah_sefer_debug.4 PASS CHOVOT HIDDEN BEFORE CIRCULATION OR DATE REACHED" }
			if = {
				limit = {
					current_date < 1140.1.1
					is_target_in_variable_list = { name = kehillah_library_acquisition_candidates target = flag:work_kuzari }
				}
				debug_log = "kehillah_sefer_debug.4 FAIL KUZARI SHOWN BEFORE CIRCULATION"
			}
			else = { debug_log = "kehillah_sefer_debug.4 PASS KUZARI HIDDEN BEFORE CIRCULATION OR DATE REACHED" }
		}
		debug_log = "kehillah_sefer_debug.4 ACQUISITION CANDIDATES END"
	}
}

# Finish the community's current copy immediately, preserving the exact title
# event used by natural delayed delivery:
#   event kehillah_sefer_debug.5
kehillah_sefer_debug.5 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { primary_title = { has_variable_list = kehillah_library_commission_work } }
			primary_title = { trigger_event = kehillah_library.0003 }
			debug_log = "kehillah_sefer_debug.5 FORCED CURRENT TITLE COMMISSION COMPLETION"
		}
		else = { debug_log = "kehillah_sefer_debug.5 NO ACTIVE COMMISSION" }
	}
}

# V49 Yeshiva Visit from the Library. Reports the player's planned journey,
# whether the planner will offer only that community, the planning event's
# suggestion pools, and any pending carried-home copy:
#   event kehillah_sefer_debug.6
kehillah_sefer_debug.6 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_sefer_debug.6 JOURNEY PLAN BEGIN"
		if = {
			limit = { exists = var:kehillah_journey_plan_title }
			var:kehillah_journey_plan_title = {
				debug_log = "kehillah_sefer_debug.6 PLAN TITLE"
				debug_log_scopes = yes
			}
		}
		else = { debug_log = "kehillah_sefer_debug.6 NO PLAN TITLE" }
		if = {
			limit = { kehillah_journey_plan_valid_trigger = yes }
			debug_log = "kehillah_sefer_debug.6 PLAN VALID: PLANNER OFFERS ONLY THIS COMMUNITY"
		}
		else = { debug_log = "kehillah_sefer_debug.6 NO VALID PLAN: PLANNER OFFERS THE ORDINARY FIELD" }
		if = {
			limit = { kehillah_journey_can_plan_trigger = yes }
			debug_log = "kehillah_sefer_debug.6 MAY PLAN A VISIT"
		}
		else = { debug_log = "kehillah_sefer_debug.6 MAY NOT PLAN A VISIT (rabbi, adult, free, off cooldown)" }
		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				debug_log = "kehillah_sefer_debug.6 IN ACTIVITY"
				debug_log_scopes = yes
				if = {
					limit = { has_variable = kehillah_journey_work }
					debug_log = "kehillah_sefer_debug.6 ACTIVITY CARRIES A PLANNED WORK"
				}
			}
		}
		kehillah_journey_build_offers_effect = yes
		every_in_list = {
			variable = kehillah_journey_offer_priority
			debug_log = "kehillah_sefer_debug.6 SUGGESTION (HOME LACKS IT)"
			debug_log_scopes = yes
		}
		every_in_list = {
			variable = kehillah_journey_offer_works
			debug_log = "kehillah_sefer_debug.6 SUGGESTION POOL"
			debug_log_scopes = yes
		}
		clear_variable_list = kehillah_journey_offer_works
		clear_variable_list = kehillah_journey_offer_priority
		if = {
			limit = { primary_title = { has_variable = kehillah_library_commission_carried } }
			debug_log = "kehillah_sefer_debug.6 A CARRIED-HOME COPY IS PENDING (event kehillah_sefer_debug.5 completes it)"
		}
		debug_log = "kehillah_sefer_debug.6 JOURNEY PLAN END"
	}
}
