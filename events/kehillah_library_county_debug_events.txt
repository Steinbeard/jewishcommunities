# Console only, disposable save: event kehillah_library_county_debug.1
# Run as a community leader before/after a county grant. Repeats relocation
# without granting land or changing the collection; logs shelf count and seat.
namespace = kehillah_library_county_debug

kehillah_library_county_debug.1 = {
	type = character_event
	hidden = yes
	trigger = { primary_title ?= { is_kehillah_title_trigger = yes } }
	immediate = {
		primary_title = {
			set_variable = { name = kehillah_library_probe_count value = kehillah_lib_work_count_value }
			kehillah_lib_refresh_seat_effect = yes
			kehillah_lib_refresh_seat_effect = yes
			if = {
				limit = { var:kehillah_library_probe_count = kehillah_lib_work_count_value }
				debug_log = "KEHILLAH_LIBRARY_COUNTY: shelf count preserved after repeated refresh"
			}
			else = { debug_log = "KEHILLAH_LIBRARY_COUNTY: FAIL shelf count changed" }
			if = {
				limit = { exists = var:kehillah_library_temple_seat }
				debug_log = "KEHILLAH_LIBRARY_COUNTY: library seated at temple"
				var:kehillah_library_temple_seat = { debug_log_scopes = yes }
			}
			else = { debug_log = "KEHILLAH_LIBRARY_COUNTY: no temple seat; collection retained in Quarter" }
			debug_log_scopes = yes
			remove_variable = kehillah_library_probe_count
			kehillah_lib_link_card_county_effect = yes
		}
	}
}
