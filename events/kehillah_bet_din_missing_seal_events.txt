		}
		first_valid = {
			triggered_desc = { trigger = { involved_activity.var:kehillah_din_case84_seal_required = 1 } desc = kehillah_bet_din.0844.fact_true }
			desc = kehillah_bet_din.0844.fact_false
		}

			triggered_desc = { trigger = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great } desc = kehillah_bet_din.0844.quality_great }
			triggered_desc = { trigger = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_good } desc = kehillah_bet_din.0844.quality_good }
			desc = kehillah_bet_din.0844.quality_poor
		}
	}
	theme = faith
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:kbd_litigant_a animation = worry }
	lower_right_portrait = { character = scope:kbd_litigant_b animation = disapproval }
	trigger = { exists = involved_activity involved_activity.var:kehillah_din_case = 84 }
	immediate = {
		kehillah_din_count_argument_panel_effect = { CASE = 84 FACT = seal_required }
		# The hidden fact changes the risk of the first and third rulings.
		# The investigation remains the dependable, costlier middle course.
		if = {
			limit = { involved_activity.var:kehillah_din_case84_seal_required = 1 involved_activity.var:kehillah_bet_din_case84_direction = 1 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 6 } }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_din_case84_seal_required = 0 involved_activity.var:kehillah_bet_din_case84_direction = 3 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 2 } }
		}
		if = {
			limit = { involved_activity.var:kehillah_bet_din_case84_direction = 3 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = 3 } }
		}
		if = {
			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability add = 84 } change_variable = { name = kehillah_var_greatness add = 12 } }
			kehillah_bet_din_record_case_result_effect = { SCORE = 2 CASE = 84 TIER = great TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			kehillah_bet_din_reward_judges_for_ruling_effect = { XP = 3 }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_good }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability add = 7 } change_variable = { name = kehillah_var_greatness add = 4 } }
			kehillah_bet_din_record_case_result_effect = { SCORE = 1 CASE = 84 TIER = good TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			kehillah_bet_din_reward_judges_for_ruling_effect = { XP = 1 }
		}
		else = {
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 10 } change_variable = { name = kehillah_var_greatness subtract = 4 } }
			kehillah_bet_din_record_case_result_effect = { SCORE = -1 CASE = 84 TIER = poor TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
		}
		if = {
			limit = { involved_activity.var:kehillah_bet_din_case84_direction = 1 }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 8 } }
			custom_tooltip = kehillah_din_case84_one_tt
		}
		else_if = {
			limit = { involved_activity.var:kehillah_bet_din_case84_direction = 2 }
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 4 } }
			custom_tooltip = kehillah_din_case84_two_tt
		}
		else = {
			involved_activity.var:kehillah_bet_din_convening_title = { change_variable = { name = kehillah_var_stability subtract = 2 } }
			custom_tooltip = kehillah_din_case84_three_tt
		}
		kehillah_din_case_party_response_effect = { DIRECTION = kehillah_bet_din_case84_direction }
	}
	option = { name = kehillah_din_record ai_chance = { base = 30 } }
	after = { kehillah_din_clear_argument_case_effect = yes kehillah_bet_din_advance_docket_effect = yes }
