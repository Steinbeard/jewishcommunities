# =====================================================================
# The Norman Conquest resolving into an invitation to settle. ADDED
# 2026-09-25 (v18 spec). Fired two days after kehillah_norman_conquest_
# execute_effect (common/scripted_effects/kehillah_norman_conquest_
# effects.txt) has already done the real work -- founded all four Anglia
# Kehillot and set the host realm's Jewish Settlement Policy to
# Encouraged -- so this is purely a report, the same "report, not a
# negotiation" shape kehillah_settlement_conditions.0004 already
# establishes for an ordinary fresh charter. root = the new holder of
# title:k_england (player or AI; AI acknowledges immediately, same as
# every other notification-only event in this mod).
# =====================================================================

kehillah_norman_conquest.0001 = {
	type = character_event
	title = kehillah_norman_conquest.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { dynasty = dynasty:752 }
				desc = kehillah_norman_conquest.0001.william.desc
			}
			triggered_desc = {
				trigger = {
					OR = {
						dynasty = dynasty:499
						culture = culture:norwegian
					}
				}
				desc = kehillah_norman_conquest.0001.harald.desc
			}
			desc = kehillah_norman_conquest.0001.other.desc
		}
	}
	theme = realm

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = kehillah_norman_conquest.0001.a
		ai_chance = { base = 100 }
	}
}


# =====================================================================
# ROADMAP S8 (2026-09-29): the player's road to London. See
# kehillah_norman_conquest_execute_effect's header for the three-event
# shape. global_var:kehillah_norman_london_offer_pending holds the invitee
# while the offer is open; whichever event resolves it removes it first.
# =====================================================================

# The invitation. Scope: the invitee.
kehillah_norman_conquest.0010 = {
	type = character_event
	title = kehillah_norman_conquest.0010.t
	desc = kehillah_norman_conquest.0010.desc
	theme = realm

	trigger = {
		exists = global_var:kehillah_norman_london_offer_pending
		global_var:kehillah_norman_london_offer_pending = root
		kehillah_norman_london_invitation_eligible_trigger = yes
	}

	immediate = {
		global_var:kehillah_norman_inviting_king ?= { save_scope_as = kehillah_norman_king }
	}

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_norman_king
		animation = personality_bold
	}

	# Accept: cross now, found tomorrow (.0011). The crossing and the
	# founding are split across a day on purpose -- the founding effect
	# names and places the community from root.location, and a same-block
	# read straight after set_location is an assumption this mod has not
	# tested; one day later it is not.
	option = {
		name = kehillah_norman_conquest.0010.a
		remove_global_variable = kehillah_norman_london_offer_pending
		custom_tooltip = kehillah_norman_conquest.0010.a.tt
		hidden_effect = {
			if = {
				limit = { exists = domicile }
				domicile = { move_domicile = title:c_middlesex.title_province }
			}
			set_location = {
				location = title:c_middlesex.title_province
				stick_to_location = yes
			}
			trigger_event = { id = kehillah_norman_conquest.0011 days = 1 }
		}
	}

	# Decline: London is founded by others, exactly as it would have been.
	# NOT by calling kehillah_norman_found_london_effect here: CK3 walks an
	# option's effects -- hidden_effect included -- to build its tooltip
	# every frame the event is on screen, and in that dry run create_
	# character makes no one, so the founding's scope:..._london_leader
	# block errored ~15,600 times in one live test (2026-09-29 S8a log).
	# Hand off instead to the fallback, .0012, a day later: it still sees
	# the pending offer (left in place for it here) and founds the AI
	# London outside any tooltip. Fired on root, not the king: .0012 reads
	# nothing from its scope, and root is certainly alive to receive it.
	option = {
		name = kehillah_norman_conquest.0010.b
		custom_tooltip = kehillah_norman_conquest.0010.b.tt
		hidden_effect = {
			trigger_event = { id = kehillah_norman_conquest.0012 days = 1 }
		}
	}
}

# The founding, a day after the crossing. Scope: the (former) invitee.
kehillah_norman_conquest.0011 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				kehillah_norman_london_invitation_eligible_trigger = yes
				exists = location.county
				location.county = title:c_middlesex
			}
			# The king's welcome, BEFORE the charter is made, for the same
			# reason kehillah_norman_conquest_finish_founding_effect sets it
			# first: the fresh charter reads the policy at creation.
			if = {
				limit = { exists = title:c_middlesex.holder.top_liege.primary_title }
				title:c_middlesex.holder.top_liege.primary_title = {
					set_variable = { name = kehillah_jewish_settlement_policy value = 3 }
				}
			}
			kehillah_found_community_effect = yes
			if = {
				limit = { government_has_flag = government_is_kehillah }
				primary_title = {
					set_title_name = kehillah_norman_london_title_name
					set_variable = { name = kehillah_norman_founded value = yes }
					set_variable = { name = kehillah_norman_player_founded value = yes }
				}
				debug_log = "kehillah_norman_conquest.0011: PLAYER FOUNDED LONDON"
				# S8b: a year on, the crown comes asking (.0020).
				trigger_event = { id = kehillah_norman_conquest.0020 years = 1 }
			}
			else = {
				debug_log = "kehillah_norman_conquest.0011: WARNING founding effect ran but the player is not in the Kehillah government"
			}
		}
		else = {
			# Died, lost eligibility, or never arrived: London must still
			# exist, so found it for the AI rather than leave a hole.
			debug_log = "kehillah_norman_conquest.0011: WARNING invitee not in London or no longer eligible -- founding London for the AI instead"
			kehillah_norman_found_london_effect = yes
		}
	}
}

# The fallback. Scope: any character -- it reads only the global offer.
# Fired by a decline (on the decliner, next day) and, as the unanswered-
# offer backstop, on both the king and the invitee at sixty days.
kehillah_norman_conquest.0012 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = global_var:kehillah_norman_london_offer_pending }
			remove_global_variable = kehillah_norman_london_offer_pending
			debug_log = "kehillah_norman_conquest.0012: London offer lapsed unanswered -- founding it for the AI"
			kehillah_norman_found_london_effect = yes
		}
	}
}


# =====================================================================
# ROADMAP S8b (2026-09-29): the crown as a repeat borrower. Scope: the
# player who founded London (.0011 schedules the first of these a year
# after the founding). Every three years the community's host -- the
# King, in any ordinary game -- asks for a loan through the ordinary
# loan contract and its offer event (kehillah_task_contract.0003), at
# kehillah_loan_royal_principal_multiplier times the usual sum. Skipped,
# with a logged reason, while the crown still owes or the community
# cannot lend that much; the chain reschedules itself either way and
# ends when this character no longer leads a player-founded London.
# =====================================================================
kehillah_norman_conquest.0020 = {
	type = character_event
	hidden = yes

	trigger = { is_alive = yes }

	immediate = {
		if = {
			limit = {
				government_has_flag = government_is_kehillah
				exists = primary_title
				primary_title = { has_variable = kehillah_norman_player_founded }
				exists = domicile.domicile_location.county.holder.top_liege
			}
			domicile.domicile_location.county.holder.top_liege = { save_scope_as = employer }
			set_variable = {
				name = kehillah_loan_royal_request
				value = yes
				days = kehillah_loan_royal_request_days
			}
			if = {
				limit = {
					scope:employer = { NOT = { exists = var:kehillah_loan_amount_owed } }
					can_create_task_contract = {
						type_name = kehillah_loan_contract
						employer = scope:employer
					}
				}
				create_task_contract = {
					task_contract_type = kehillah_loan_contract
					task_contract_tier = root.task_contract_tier_value
					location = root.domicile.domicile_location
					task_contract_employer = scope:employer
					save_scope_as = new_contract
				}
				debug_log = "kehillah_norman_conquest.0020: ROYAL REQUEST OFFERED the crown asks London for a loan"
				trigger_event = kehillah_task_contract.0003
			}
			else = {
				remove_variable = kehillah_loan_royal_request
				debug_log = "kehillah_norman_conquest.0020: ROYAL REQUEST SKIPPED the crown still owes, or the community cannot lend this much (gold, Healthy Prosperity)"
			}
			trigger_event = { id = kehillah_norman_conquest.0020 years = 3 }
		}
		else = {
			debug_log = "kehillah_norman_conquest.0020: chain ends -- no longer the leader of a player-founded London"
		}
	}
}
