namespace = kehillah_din_argument_debug

# Console only: event kehillah_din_argument_debug.1 on a judge in a sitting.
kehillah_din_argument_debug.1 = {
	type = character_event
	hidden = yes
	trigger = { exists = involved_activity }
	immediate = {
		involved_activity = {
			debug_log = "kehillah_din_argument_debug: current case and argument state"
			if = { limit = { exists = var:kehillah_din_case } debug_log = "kehillah_din_argument_debug: case set" }
			if = { limit = { exists = var:kehillah_din_argument_seat } debug_log = "kehillah_din_argument_debug: seat set" }
			if = { limit = { exists = var:kehillah_din_argument_position } debug_log = "kehillah_din_argument_debug: position set" }
			if = { limit = { exists = var:kehillah_din_argument_chance } debug_log = "kehillah_din_argument_debug: uncited chance set" }
			if = { limit = { exists = var:kehillah_din_argument_final_chance } debug_log = "kehillah_din_argument_debug: final chance set" }
			if = { limit = { var:kehillah_din_argument_success = 1 } debug_log = "kehillah_din_argument_debug: argument held" }
			if = { limit = { exists = var:kehillah_din_panel_winner } debug_log = "kehillah_din_argument_debug: panel winner set" }
			if = { limit = { exists = var:kehillah_din_vote_1_position } debug_log = "kehillah_din_argument_debug: presiding vote set" }
			if = { limit = { exists = var:kehillah_din_vote_2_position } debug_log = "kehillah_din_argument_debug: second vote set" }
			if = { limit = { exists = var:kehillah_din_vote_3_position } debug_log = "kehillah_din_argument_debug: third vote set" }
		}
	}
}
