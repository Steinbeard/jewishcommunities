# First phase: choose how to assemble the evidence. A study partner can
# help both completion and quality, but never bypasses an entry gate.
kehillah_writing.0100 = {
	type = character_event
	title = kehillah_writing.0100.t
	desc = kehillah_writing.0100.desc
	theme = learning
	left_portrait = { character = root animation = thinking }
	immediate = {
		random_courtier = {
			limit = { is_adult = yes is_imprisoned = no learning >= 10 is_jewish_character_trigger = yes }
			weight = { base = 1 modifier = { add = 3 learning >= 16 } }
			save_scope_as = kehillah_writing_candidate
		}
	}
	option = {
		name = kehillah_writing.0100.a
		change_variable = { name = kehillah_book_tally add = 10 }
		custom_tooltip = kehillah_writing.0100.a.tt
	}
	option = {
		name = kehillah_writing.0100.b
		trigger = { exists = scope:kehillah_writing_candidate }
		set_variable = { name = kehillah_writing_partner value = scope:kehillah_writing_candidate }
		change_variable = { name = kehillah_book_tally add = 8 }
		custom_tooltip = kehillah_writing.0100.b.tt
	}
	option = {
		name = kehillah_writing.0100.c
		random_list = {
			50 = { change_variable = { name = kehillah_book_tally add = 20 } custom_tooltip = kehillah_writing.0100.c.success }
			50 = { change_variable = { name = kehillah_book_tally subtract = 10 } custom_tooltip = kehillah_writing.0100.c.failure }
		}
	}
}

# Second phase: objections become the draft's shape. The long revision
# costs another scheme phase but improves quality.
kehillah_writing.0200 = {
	type = character_event
	title = kehillah_writing.0200.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:responsa } desc = kehillah_writing.0200.desc.responsa }
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:poetry } desc = kehillah_writing.0200.desc.poetry }
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:dialogue } desc = kehillah_writing.0200.desc.dialogue }
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:systematic_dialogue } desc = kehillah_writing.0200.desc.systematic }
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:ethics } desc = kehillah_writing.0200.desc.ethics }
			triggered_desc = { trigger = { var:kehillah_writing_form ?= flag:devotion } desc = kehillah_writing.0200.desc.devotion }
			triggered_desc = { trigger = { var:kehillah_book_genre ?= flag:talmudics } desc = kehillah_writing.0200.desc.talmud }
			desc = kehillah_writing.0200.desc
		}
	}
	theme = learning
	left_portrait = { character = root animation = thinking }
	option = {
		name = kehillah_writing.0200.a
		if = { limit = { learning >= 16 } change_variable = { name = kehillah_book_tally add = 15 } custom_tooltip = kehillah_writing.0200.a.success }
		else = { change_variable = { name = kehillah_book_tally subtract = 5 } custom_tooltip = kehillah_writing.0200.a.failure }
	}
	option = {
		name = kehillah_writing.0200.b
		change_variable = { name = kehillah_book_tally add = 8 }
		custom_tooltip = kehillah_writing.0200.b.tt
	}
	option = {
		name = kehillah_writing.0200.c
		set_variable = { name = kehillah_writing_extra_time value = yes }
		change_variable = { name = kehillah_book_tally add = 15 }
		add_stress = minor_stress_gain
		custom_tooltip = kehillah_writing.0200.c.tt
	}
}
