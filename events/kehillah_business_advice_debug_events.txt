# Console-only diagnostic and disposable eligibility setup. Never called by gameplay.
namespace = kehillah_business_advice_debug
kehillah_business_advice_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		save_scope_value_as = { name = business_unassigned value = kehillah_business_unassigned_count_value }
		if = { limit = { kehillah_business_staffing_advice_trigger = yes } debug_log = "BUSINESS ADVICE staffing AVAILABLE" }
		else = { debug_log = "BUSINESS ADVICE staffing UNAVAILABLE" }
		if = { limit = { kehillah_business_expansion_advice_trigger = yes } debug_log = "BUSINESS ADVICE expansion AVAILABLE" }
		else = { debug_log = "BUSINESS ADVICE expansion UNAVAILABLE" }
		debug_log = "BUSINESS ADVICE unassigned count follows; this report never refreshes or pays income"
		debug_log_scopes = yes
	}
}
kehillah_business_advice_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		every_courtier = {
			limit = { kehillah_business_unassigned_head_trigger = yes is_courtier_of = prev house.var:kehillah_household_of = prev.primary_title }
			house = { set_variable = { name = kehillah_household_assignment_cooldown value = yes days = 2 } }
		}
		debug_log = "BUSINESS ADVICE disposable setup: unassigned families committed for TWO DAYS"
		trigger_event = kehillah_business_advice_debug.1
	}
}

# Force the native action through its real predicate, without waiting for the
# engine's periodic important-action scan. This does not prove scan timing.
kehillah_business_advice_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { kehillah_business_staffing_advice_trigger = yes }
			try_create_important_action = { important_action_type = action_kehillah_staff_business actor = root }
			debug_log = "BUSINESS ADVICE native staffing alert requested"
		}
		else = { debug_log = "BUSINESS ADVICE no staffing alert requested: no eligible vacancy" }
	}
}

# Probe the exact native interface helper used by alert clicks. This establishes
# view behavior only, not icon visibility or automatic important-action scans.
kehillah_business_advice_debug.4 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = domicile }
			kehillah_business_alert_open_quarter_effect = yes
			debug_log = "BUSINESS ADVICE native alert Quarter view helper requested"
		}
	}
}
