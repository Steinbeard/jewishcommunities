	type = character_event
	hidden = yes

		domicile ?= { move_domicile = root.location }

	}

# 2026-11-25. Whoever an AI lord's letter called decides, some months
# after the letter (so a player called with it can answer first). The
# first community founded there takes the site; a later answer finds the
# call closed. A community sends settlers, an adventurer takes his band
# there, a merchant sends a factor with families of the company.
kehillah_invitation.0020 = {
	type = character_event
	hidden = yes
	trigger = {
		kehillah_invitation_call_open_trigger = yes
	}
	immediate = {
		kehillah_invitation_letter_setup_effect = yes
		if = {
			limit = {
				government_has_flag = government_is_kehillah
				exists = primary_title
				kehillah_invitation_can_answer_trigger = yes
				OR = {
					exists = scope:kehillah_settlers_lead_baal
					exists = scope:kehillah_settlers_lead_rabbi
					exists = scope:kehillah_settlers_lead_kin
				}
			}
			random = {
				chance = kehillah_invitation_ai_answer_chance
				if = {
					limit = { exists = scope:kehillah_settlers_lead_baal }
					scope:kehillah_settlers_lead_baal = { save_scope_as = kehillah_settlers_leader }
				}
				else_if = {
					limit = { exists = scope:kehillah_settlers_lead_rabbi }
					scope:kehillah_settlers_lead_rabbi = { save_scope_as = kehillah_settlers_leader }
				}
				else = {
					scope:kehillah_settlers_lead_kin = { save_scope_as = kehillah_settlers_leader }
				}
				kehillah_invitation_send_settlers_effect = yes
				kehillah_settlers_go_effect = yes
				debug_log = "kehillah_invitation.0020: an AI community answered a lord's call"
			}
		}
		else_if = {
			limit = {
				has_government = landless_adventurer_government
				kehillah_invitation_band_can_answer_trigger = yes
			}
			random = {
				chance = kehillah_invitation_ai_band_chance
				kehillah_invitation_band_set_out_effect = yes
				debug_log = "kehillah_invitation.0020: an AI adventurer set out for a lord's call"
			}
		}
		else_if = {
			limit = {
				has_government = merchant_government
				exists = scope:kehillah_settlers_lead_merchant
				kehillah_invitation_merchant_can_answer_trigger = yes
			}
			random = {
				chance = kehillah_invitation_ai_merchant_chance
				scope:kehillah_settlers_lead_merchant = { save_scope_as = kehillah_settlers_leader }
				kehillah_invitation_merchant_send_effect = yes
				kehillah_invitation_merchant_go_effect = yes
				debug_log = "kehillah_invitation.0020: an AI merchant answered a lord's call"
			}
		}
		debug_log_scopes = yes
	}
}

# 2026-11-25. A player's band has reached the county a lord invited it to
