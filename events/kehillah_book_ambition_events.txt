# only place where a manuscript is chosen and the ambition is rechecked.

	title = kehillah_book_ambition.0001.t
	desc = kehillah_book_ambition.0001.desc

	option = { name = kehillah_book_ambition.0001.a trigger_event = kehillah_book_ambition.0002 }
	option = { name = kehillah_book_ambition.0001.b trigger_event = kehillah_book_ambition.0003 }
	option = { name = kehillah_book_ambition.0001.c trigger_event = kehillah_book_ambition.0004 }
	option = { name = kehillah_book_ambition.close }
}

kehillah_book_ambition.0002 = {
	type = character_event
	title = kehillah_book_ambition.0002.t
	desc = kehillah_book_ambition.category_desc
	theme = learning
	option = {
		name = kehillah_book_ambition.rashi_torah
		trigger = { NOT = { has_global_variable = kehillah_wk_work_rashi_torah_written } }
		custom_tooltip = kehillah_book_ambition.rashi_torah.tt
		if = { limit = { kehillah_sefer_ambition_rashi_torah_trigger = yes } custom_tooltip = kehillah_book_ambition.ready }
		else = { custom_tooltip = kehillah_book_ambition.not_ready }
	}
	option = { name = kehillah_book_ambition.close }
