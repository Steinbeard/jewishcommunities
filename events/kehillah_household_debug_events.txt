# Console-only probes for the ba'alei batim (V26 part 2); never called by
# gameplay. Play or select a Kehillah leader, then:
#   event kehillah_household_debug.1  -- register, refresh, report
#   event kehillah_household_debug.2  -- force an arrival
#   event kehillah_household_debug.3  -- force a departure attempt
#   event kehillah_household_debug.4  -- run the full yearly pass
#   event kehillah_household_debug.5  -- report every community's count
#   event kehillah_household_debug.6  -- why the succession standing term
#                                        does or doesn't apply, per councillor
#   event kehillah_household_debug.7  -- make root a lay leader and check
#                                        faith-organization membership
#   event kehillah_household_debug.8  -- READ-ONLY clerical-continuation
#                                        report: clergy, organization,
#                                        player heir, title heir
#   event kehillah_household_debug.9  -- strip the Rabbi trait (pure lay
#                                        leader), then run .8's report
#   event kehillah_household_debug.10 -- have an AI community's lowest
#                                        household move to YOUR community,
#                                        firing the .0003 arrival event
namespace = kehillah_household_debug

kehillah_household_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.1: households -- START"
		kehillah_household_bootstrap_effect = yes
		kehillah_household_refresh_effect = yes
		primary_title = {
			if = {
				limit = { kehillah_household_count_value >= 2 }
				debug_log = "kehillah_household_debug.1: FLOOR PASS at least two households"
			}
			else = { debug_log = "kehillah_household_debug.1: FLOOR FAIL fewer than two households" }
			if = {
				limit = { kehillah_household_count_value <= kehillah_household_cap_value }
				debug_log = "kehillah_household_debug.1: CAP PASS count within cap"
			}
			else = { debug_log = "kehillah_household_debug.1: CAP OVER count above cap (departures will trim it)" }
			if = {
				limit = { any_in_list = { variable = kehillah_households this = root.house } }
				debug_log = "kehillah_household_debug.1: LEADER-HOUSE FAIL leader's own house is registered"
			}
			else = { debug_log = "kehillah_household_debug.1: LEADER-HOUSE PASS leader's house not registered" }
			if = {
				limit = {
					any_in_list = {
						variable = kehillah_households
						NOT = { exists = var:kehillah_household_head }
					}
				}
				debug_log = "kehillah_household_debug.1: HEAD FAIL a household has no head"
			}
			else = { debug_log = "kehillah_household_debug.1: HEAD PASS every household has a head" }
			every_in_list = {
				variable = kehillah_households
				var:kehillah_household_head = {
					save_scope_as = kh_dbg_head
					debug_log = "kehillah_household_debug.1: HOUSEHOLD (head and standing in scopes dump)"
					debug_log_scopes = yes
				}
			}
		}
		debug_log = "kehillah_household_debug.1: households -- END"
	}
}

kehillah_household_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.2: forced arrival -- START"
		domicile.domicile_location = { save_scope_as = kh_here_location }
		kehillah_household_arrival_effect = yes
		if = {
			limit = { exists = scope:kh_arr_house }
			debug_log = "kehillah_household_debug.2: ORIGIN HOUSEHOLD an existing household is moving (event for players)"
		}
		else_if = {
			limit = { exists = scope:kh_arr_origin }
			debug_log = "kehillah_household_debug.2: NEW FAMILY from a named community (event for players)"
		}
		else = { debug_log = "kehillah_household_debug.2: NEW FAMILY with no origin" }
		debug_log = "kehillah_household_debug.2: forced arrival -- END"
	}
}

kehillah_household_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.3: forced departure -- START"
		domicile.domicile_location = { save_scope_as = kh_here_location }
		kehillah_household_refresh_effect = yes
		kehillah_household_departure_effect = yes
		if = {
			limit = { exists = scope:kh_dep_dest }
			debug_log = "kehillah_household_debug.3: DESTINATION found (event for players; moved at once for AI)"
			scope:kh_dep_dest = { debug_log_scopes = yes }
		}
		else = { debug_log = "kehillah_household_debug.3: NO DESTINATION with room (or no household)" }
		debug_log = "kehillah_household_debug.3: forced departure -- END"
	}
}

kehillah_household_debug.4 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.4: yearly pass -- START"
		kehillah_household_yearly_effect = yes
		debug_log = "kehillah_household_debug.4: yearly pass -- END"
	}
}

kehillah_household_debug.6 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.6: succession standing term -- START"
		primary_title = { save_scope_as = kh_dbg_title }
		every_councillor = {
			save_scope_as = kh_dbg_councillor
			debug_log = "kehillah_household_debug.6: COUNCILLOR (scopes dump follows)"
			debug_log_scopes = yes
			house ?= {
				save_scope_as = kh_dbg_house
				debug_log = "kehillah_household_debug.6: HOUSE resolves via house ?="
				if = {
					limit = { has_variable = kehillah_household_of }
					if = {
						limit = { var:kehillah_household_of = scope:kh_dbg_title }
						if = {
							limit = { has_variable = kehillah_household_standing }
							debug_log = "kehillah_household_debug.6: PASS registered here with standing (term should apply)"
						}
						else = { debug_log = "kehillah_household_debug.6: registered here but NO standing yet" }
					}
					else = { debug_log = "kehillah_household_debug.6: registered to ANOTHER community" }
				}
				else = { debug_log = "kehillah_household_debug.6: house is NOT a registered household" }
			}
			if = {
				limit = { exists = house }
				debug_log = "kehillah_household_debug.6: exists = house is TRUE"
			}
			else = { debug_log = "kehillah_household_debug.6: exists = house is FALSE" }
		}
		debug_log = "kehillah_household_debug.6: succession standing term -- END"
	}
}

# Lay-leader clerical succession check: strips the Rabbi trait from root,
# re-runs the leader setup (which should join the faith's organization by
# script), and reports membership. Kill root afterwards to see the death
# screen a Parnas gets.
kehillah_household_debug.7 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.7: lay leader organization -- START"
		if = {
			limit = { exists = organization }
			debug_log = "kehillah_household_debug.7: BEFORE member of an organization"
		}
		else = { debug_log = "kehillah_household_debug.7: BEFORE not a member" }
		if = {
			limit = { exists = faith.organization }
			debug_log = "kehillah_household_debug.7: the faith HAS an organization"
		}
		else = { debug_log = "kehillah_household_debug.7: the faith has NO organization" }
		if = {
			limit = { has_trait = kehillah_rabbi_trait }
			remove_trait = kehillah_rabbi_trait
			debug_log = "kehillah_household_debug.7: removed the Rabbi trait"
		}
		if = {
			limit = { exists = organization }
			debug_log = "kehillah_household_debug.7: AFTER REMOVAL still a member"
		}
		else = { debug_log = "kehillah_household_debug.7: AFTER REMOVAL no longer a member" }
		kehillah_update_leader_flavor_effect = yes
		if = {
			limit = { exists = organization }
			debug_log = "kehillah_household_debug.7: PASS lay leader is a member after the leader setup"
		}
		else = { debug_log = "kehillah_household_debug.7: FAIL lay leader is not a member after the leader setup" }
		debug_log = "kehillah_household_debug.7: lay leader organization -- END"
	}
}

kehillah_household_debug.5 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.5: all communities -- START"
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { exists = holder }
			if = {
				limit = { kehillah_household_count_value >= 2 }
				debug_log = "kehillah_household_debug.5: community with 2+ households"
			}
			else_if = {
				limit = { kehillah_household_count_value >= 1 }
				debug_log = "kehillah_household_debug.5: community with ONE household"
			}
			else = { debug_log = "kehillah_household_debug.5: community with NO households" }
		}
		debug_log = "kehillah_household_debug.5: all communities -- END"
	}
}


# ---------------------------------------------------------------------
# .8 / .9 -- the clerical-continuation probe (ROADMAP H1).
#
# .8 reports, and changes nothing, so it is safe to run before AND after
# anything else. Run it on the Kehillah leader immediately before a
# `kill <id>`: the single thing that decides Game Over vs. "Continue as
# <appointee>" is whether PLAYER HEIR exists at that instant, and the
# only reason this mod has one for a non-dynastic appointee is clergy
# status (see common/governments/kehillah_government.txt's
# mechanic_type = theocracy comment).
#
# .9 makes the leader a PURE LAY leader first -- no Rabbi trait, so no
# trait-sourced clergy -- and then reports. If mechanic_type = theocracy
# is doing its job, .9 must still report CLERGY YES, ORGANIZATION YES
# and PLAYER HEIR YES. Unlike .7 it does not call the leader-setup
# effect, so nothing can mask the government's own contribution.
# ---------------------------------------------------------------------

kehillah_household_debug.8 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.8: clerical continuation -- START"
		debug_log_scopes = yes

		if = {
			limit = { has_trait = kehillah_rabbi_trait }
			debug_log = "kehillah_household_debug.8: LEADER IS RABBINIC (has kehillah_rabbi_trait)"
		}
		else = { debug_log = "kehillah_household_debug.8: LEADER IS LAY (no kehillah_rabbi_trait)" }

		if = {
			limit = { government_has_mechanic = theocracy }
			debug_log = "kehillah_household_debug.8: MECHANIC YES government_has_mechanic = theocracy"
		}
		else = { debug_log = "kehillah_household_debug.8: MECHANIC NO government_has_mechanic = theocracy is false" }

		if = {
			limit = { is_clergy = yes }
			debug_log = "kehillah_household_debug.8: CLERGY YES is_clergy = yes"
		}
		else = { debug_log = "kehillah_household_debug.8: CLERGY NO is_clergy = no" }

		if = {
			limit = { exists = organization }
			debug_log = "kehillah_household_debug.8: ORGANIZATION YES member of an organization"
		}
		else_if = {
			limit = { exists = faith.organization }
			debug_log = "kehillah_household_debug.8: ORGANIZATION NO the faith has one, root is not in it"
		}
		else = { debug_log = "kehillah_household_debug.8: ORGANIZATION NO the faith has no organization at all" }

		# The decisive one.
		if = {
			limit = { exists = player_heir }
			debug_log = "kehillah_household_debug.8: PLAYER HEIR YES -- death should offer Continue as the heir"
			if = {
				limit = {
					exists = dynasty
					player_heir = { dynasty ?= root.dynasty }
				}
				debug_log = "kehillah_household_debug.8: player heir is of root's OWN dynasty (not the clerical path)"
			}
			else = { debug_log = "kehillah_household_debug.8: player heir is NON-DYNASTIC -- this is the clerical path" }
		}
		else = { debug_log = "kehillah_household_debug.8: PLAYER HEIR NO -- death would be Game Over" }

		# REGRESSION GUARD for the mechanic_type = theocracy change: the
		# Kehillah must still be on its own appointment law and not on
		# vanilla's clerical_appointment_law. See common/scripted_triggers/
		# kehillah_clerical_appointment_law_override.txt.
		#
		# THE REALM LAW IS THE VERDICT, not the title law. The Kehillah's
		# law arrives as a realm law on the holder -- history writes
		# `succession_laws = { ... }` and kehillah_found_community_effect
		# calls add_realm_law -- so has_title_law is false for it, as it
		# is for most of vanilla's appointment laws too. Vanilla's own
		# checks pair the two in an OR for exactly this reason
		# (common/character_interactions/pam_interactions.txt:443-444:
		# `holder ?= { has_realm_law = clerical_appointment_law }` OR
		# `has_title_law = clerical_appointment_law`). The title-law lines
		# below are INFORMATIONAL; a 2026-10-09 live run read
		# "no title-level law" with REALM LAW PASS at the same instant,
		# and that is the correct, healthy state.
		if = {
			limit = { has_realm_law = kehillah_appointment_succession_law }
			debug_log = "kehillah_household_debug.8: REALM LAW PASS kehillah_appointment_succession_law"
		}
		else_if = {
			limit = { has_realm_law = clerical_appointment_law }
			debug_log = "kehillah_household_debug.8: REALM LAW FAIL vanilla clerical_appointment_law took the slot"
		}
		else = { debug_log = "kehillah_household_debug.8: REALM LAW FAIL neither expected law (check the realm law list in scopes)" }

		if = {
			limit = { primary_title ?= { has_title_law = kehillah_appointment_succession_law } }
			debug_log = "kehillah_household_debug.8: (info) the title ALSO carries a title-level kehillah_appointment_succession_law"
		}
		else_if = {
			limit = { primary_title ?= { has_title_law = clerical_appointment_law } }
			debug_log = "kehillah_household_debug.8: LAW FAIL the title carries a title-level clerical_appointment_law"
		}
		else = { debug_log = "kehillah_household_debug.8: (info) no title-level succession law, which is expected here" }

		# Separate from player heir: who the title itself would pass to.
		if = {
			limit = {
				exists = primary_title
				primary_title = { any_title_heir = { always = yes } }
			}
			debug_log = "kehillah_household_debug.8: TITLE HEIR YES the primary title has at least one heir"
			primary_title = {
				every_title_heir = { debug_log_scopes = yes }
			}
		}
		else = { debug_log = "kehillah_household_debug.8: TITLE HEIR NO the primary title has no heir" }

		debug_log = "kehillah_household_debug.8: clerical continuation -- END"
	}
}

kehillah_household_debug.9 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.9: force a pure lay leader -- START"
		if = {
			limit = { has_trait = kehillah_rabbi_trait }

# ---------------------------------------------------------------------
# .10 -- the player-RECEIVES-a-family path (ROADMAP H2).
#
# kehillah_household.0003 only fires when an AI community loses its
# lowest-standing household TO the player's community, and the yearly
# pass picks its destination at random out of every community with room.
# Waiting for that to happen naturally is the reason this path has never
# been exercised. This probe forces exactly it, through the real
# effects -- kehillah_household_save_lowest_effect picks the same family
# the yearly pass would, and kehillah_household_leave_effect does the
# whole move -- with only the DESTINATION forced to root's community.
#
# Run it on the player's Kehillah leader. Needs at least one other
# registered community with a household; probe .5 reports which have
# any. It migrates real courtiers between real courts, so it is not
# read-only: use it on a throwaway run, not one you want to keep.
# ---------------------------------------------------------------------

kehillah_household_debug.10 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_household_debug.10: forced arrival from another community -- START"

		primary_title = { save_scope_as = kh_dep_dest }

		random_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				NOT = { this = root.primary_title }
				exists = holder
				holder = {
					government_has_flag = government_is_kehillah
					is_ai = yes
				}
				kehillah_household_count_value >= 1
			}
			holder = { save_scope_as = kh_dep_leader }
			kehillah_household_save_lowest_effect = yes
		}

		if = {
			limit = {
				exists = scope:kh_dep_leader
				exists = scope:kh_low_house
			}
			scope:kh_low_house = { save_scope_as = kh_dep_house }
			debug_log = "kehillah_household_debug.10: DONOR FOUND an AI community and its lowest household (scopes dump follows)"
			debug_log_scopes = yes

			# The real move. This is what tells a player recipient, via
			# kehillah_household.0003 -- so root should see the event.
			kehillah_household_leave_effect = yes

			if = {
				limit = {
					primary_title = {
						any_in_list = {
							variable = kehillah_households
							this = scope:kh_dep_house
						}
					}
				}
				debug_log = "kehillah_household_debug.10: PASS the family is now registered at root's community"
			}
			else = { debug_log = "kehillah_household_debug.10: FAIL the family did not end up registered at root's community" }

			if = {
				limit = {
					scope:kh_dep_leader.primary_title = {
						any_in_list = {
							variable = kehillah_households
							this = scope:kh_dep_house
						}
					}
				}
				debug_log = "kehillah_household_debug.10: FAIL the family is still registered at the donor community too"
			}
			else = { debug_log = "kehillah_household_debug.10: PASS the family is no longer registered at the donor community" }
		}
		else = { debug_log = "kehillah_household_debug.10: BLOCKED no other AI community with a household (run .5 to see counts)" }

		debug_log = "kehillah_household_debug.10: forced arrival from another community -- END"
	}
}
