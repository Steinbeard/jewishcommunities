# ---------------------------------------------------------------------
# kehillah_sermon.0001 -- a community's sermon: the topic.
# Root: the Kehillah leader; var:kehillah_sermon_preacher speaks.
# ---------------------------------------------------------------------
kehillah_sermon.0001 = {
	type = character_event
	title = kehillah_sermon.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { var:kehillah_sermon_preacher ?= root } }
				desc = kehillah_sermon.0001.desc_dayyan
			}
			desc = kehillah_sermon.0001.desc
		}
	}
	theme = learning
	left_portrait = {
		character = var:kehillah_sermon_preacher
		animation = thinking
	}
	trigger = { exists = var:kehillah_sermon_preacher }

	# The preview: what the sermon gives before any text is cited.
	immediate = {
		var:kehillah_sermon_preacher = {
			save_temporary_scope_value_as = { name = kehillah_sermon_base value = kehillah_sermon_parshanut_gain_value }
		}
		set_variable = { name = kehillah_sermon_base value = scope:kehillah_sermon_base }
		kehillah_sermon_save_host_effect = yes
	}

	option = {
		name = kehillah_sermon.0001.portion
		show_as_tooltip = { kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = root.var:kehillah_sermon_base } }
		hidden_effect = { kehillah_sermon_choose_topic_effect = { TOPIC = portion } }
		ai_chance = {
			base = 30
			modifier = { add = 30 NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_healthy_threshold } } }
		}
	}
	option = {
		name = kehillah_sermon.0001.neighbours
		show_as_tooltip = { kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = root.var:kehillah_sermon_base } }
		hidden_effect = { kehillah_sermon_choose_topic_effect = { TOPIC = neighbours } }
		ai_chance = {
			base = 30
			modifier = { add = 30 NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold } } }
		}
	}
	option = {
		name = kehillah_sermon.0001.tzedakah
		show_as_tooltip = { kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = root.var:kehillah_sermon_base } }
		hidden_effect = { kehillah_sermon_choose_topic_effect = { TOPIC = tzedakah } }
		ai_chance = {
			base = 30
			modifier = { add = 30 NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_healthy_threshold } } }
		}
	}
	option = {
		name = kehillah_sermon.0001.peace
		# 2026-11-11: the lord who hosts the community, not `liege` (which is
		# the leader himself when the community is under no lord).
		trigger = { kehillah_sermon_has_host_trigger = yes }
		show_as_tooltip = {
			scope:kehillah_sermon_host ?= { add_opinion = { target = root modifier = respect_opinion opinion = root.var:kehillah_sermon_base } }
		}
		hidden_effect = { kehillah_sermon_choose_topic_effect = { TOPIC = peace } }
		ai_chance = {
			base = 15
			modifier = { add = 25 scope:kehillah_sermon_host ?= { opinion = { target = root value < 0 } } }
		}
	}
	option = {
		name = kehillah_sermon.0001.trouble
		trigger = { kehillah_sermon_time_of_trouble_trigger = yes }
		show_as_tooltip = {
			kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = root.var:kehillah_sermon_base }
			add_stress = kehillah_sermon_trouble_stress
		}
		hidden_effect = { kehillah_sermon_choose_topic_effect = { TOPIC = trouble } }
		ai_chance = { base = 80 }
	}
}

# ---------------------------------------------------------------------
# kehillah_sermon.0002 -- a text for the sermon (twice: the first, then
# the second). Root: the holder, who keeps the candidates.
# ---------------------------------------------------------------------
kehillah_sermon.0002 = {
	type = character_event
	title = kehillah_sermon.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_sermon_step ?= 2 }
				desc = kehillah_sermon.0002.desc_second
			}
			desc = kehillah_sermon.0002.desc_first
		}
	}
	theme = learning
	left_portrait = {
		character = var:kehillah_sermon_preacher
		animation = thinking
	}
	trigger = { exists = var:kehillah_sermon_cand_grounded }

	# The most established text he has mastered on the topic.
	option = {
		name = kehillah_sermon.0002.grounded
		hidden_effect = { kehillah_sermon_cite_effect = { WHICH = grounded } }
		ai_chance = { base = 60 }
	}
	# A newer one, which the sermon helps toward its own standing.
	option = {
		name = kehillah_sermon.0002.newer
		trigger = {
			exists = var:kehillah_sermon_cand_newer
			NOT = { var:kehillah_sermon_cand_newer = var:kehillah_sermon_cand_grounded }
		}
		hidden_effect = { kehillah_sermon_cite_effect = { WHICH = newer } }
		ai_chance = { base = 30 }
	}
	option = {
		name = kehillah_sermon.0002.none
		hidden_effect = { kehillah_sermon_after_text_effect = yes }
		ai_chance = { base = 5 }
	}
}

# ---------------------------------------------------------------------
# kehillah_sermon.0003 -- who preaches (2026-11-11, Daniel). Root: a
# Kehillah leader who may preach himself, with a Dayyan who may too. A Dayyan
# who preaches chooses his own texts.
# ---------------------------------------------------------------------
kehillah_sermon.0003 = {
	type = character_event
	title = kehillah_sermon.0003.t
	desc = kehillah_sermon.0003.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = cp:councillor_court_chaplain
		animation = personality_honorable
	}
	trigger = { exists = cp:councillor_court_chaplain }

	immediate = {
		cp:councillor_court_chaplain = { save_scope_as = kehillah_sermon_dayyan }
	}

	option = {
		name = kehillah_sermon.0003.self
		custom_tooltip = kehillah_sermon.0003.self.tt
		hidden_effect = {
			set_variable = { name = kehillah_sermon_preacher value = root }
			trigger_event = kehillah_sermon.0001
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 diplomacy >= cp:councillor_court_chaplain.diplomacy }
			modifier = { add = -40 kehillah_sermon_preacher_banned_trigger = yes }
		}
	}
	option = {
		name = kehillah_sermon.0003.dayyan
		custom_tooltip = kehillah_sermon.0003.dayyan.tt
		hidden_effect = {
			set_variable = { name = kehillah_sermon_preacher value = cp:councillor_court_chaplain }
			trigger_event = kehillah_sermon.0001
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 cp:councillor_court_chaplain.diplomacy > diplomacy }
			modifier = { add = -40 cp:councillor_court_chaplain = { kehillah_sermon_preacher_banned_trigger = yes } }
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_sermon.0004 -- the delivery (2026-11-11, Daniel). Root: the
# holder. The option shows the roll: the preacher's Diplomacy, his texts,
# his manner and any ban on him (kehillah_sermon_roll_effect).
# ---------------------------------------------------------------------
kehillah_sermon.0004 = {
	type = character_event
	title = kehillah_sermon.0004.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_sermon_preacher ?= root }
				desc = kehillah_sermon.0004.desc_self
			}
			desc = kehillah_sermon.0004.desc_other
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = var:kehillah_sermon_cited_2 exists = var:kehillah_sermon_cited_1 }
				desc = kehillah_sermon.0004.desc_two
			}
			triggered_desc = {
				trigger = { exists = var:kehillah_sermon_cited_1 }
				desc = kehillah_sermon.0004.desc_one
			}
			triggered_desc = {
				trigger = { exists = var:kehillah_sermon_cited_2 }
				desc = kehillah_sermon.0004.desc_one_second
			}
			desc = kehillah_sermon.0004.desc_none
		}
	}
	theme = learning
	left_portrait = {
		character = var:kehillah_sermon_preacher
		animation = personality_bold
	}
	trigger = { exists = var:kehillah_sermon_preacher }

	immediate = {
		save_scope_as = kehillah_sermon_holder
		var:kehillah_sermon_preacher = { save_scope_as = kehillah_sermon_preacher }
		var:kehillah_sermon_title ?= { save_scope_as = kehillah_sermon_title }
		kehillah_sermon_save_host_effect = yes
	}

	option = {
		name = kehillah_sermon.0004.a
		kehillah_sermon_roll_effect = yes
		hidden_effect = { kehillah_sermon_clear_effect = yes }
	}
}
