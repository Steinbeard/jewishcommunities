# Console only: event kehillah_hostile_rite_name_debug.1 on a Kehillah leader.
# Creates a Christian preacher at the community town, gives him its county
# faith, and runs the same reuse-or-found path as a mob. Inspect the saved
# rite scope and the preacher's rite window. Rerun only on a fresh save when
# checking each of the four first-use names.
namespace = kehillah_hostile_rite_name_debug

kehillah_hostile_rite_name_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = domicile.domicile_location.county.faith
				domicile.domicile_location.county.faith.religion = religion:christianity_religion
			}
			create_character = {
				age = 40
				gender = male
				faith = domicile.domicile_location.county.faith
				culture = domicile.domicile_location.county.culture
				location = domicile.domicile_location
				dynasty = none
				save_scope_as = kehillah_hostile_rite_name_probe_preacher
			}
			scope:kehillah_hostile_rite_name_probe_preacher = {
				kehillah_mob_join_hostile_rite_effect = yes
				save_scope_as = kehillah_hostile_rite_name_probe_result
			}
			debug_log = "kehillah_hostile_rite_name_debug.1: preacher joined or founded a hostile Christian rite"
			debug_log_scopes = yes
		}
	}
}
