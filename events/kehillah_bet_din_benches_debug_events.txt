# Console-only probes for the bench and the benches (2026-11-25).
# kehillah_bet_din_benches_debug.1 -- run on the player during a sitting:
#   event kehillah_bet_din_benches_debug.1
# Writes to debug.log who sits where, who is lay, and the benches' weights.
namespace = kehillah_bet_din_benches_debug

kehillah_bet_din_benches_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { kehillah_bet_din_qualified_judge_trigger = yes }
			debug_log = "KBD_BENCHES player QUALIFIED"
		}
		else = { debug_log = "KBD_BENCHES player NOT qualified" }
		if = {
			limit = { kehillah_bet_din_has_presiding_authority_trigger = yes }
			debug_log = "KBD_BENCHES player community HAS presiding authority"
		}
		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				var:kehillah_bet_din_av_beit_din ?= {
					debug_log = "KBD_BENCHES AV (convener)"
					debug_log_scopes = yes
					if = { limit = { has_variable = kehillah_bet_din_lay_seat } debug_log = "KBD_BENCHES AV is LAY" }
				}
				var:kehillah_bet_din_judge2 ?= {
					debug_log = "KBD_BENCHES JUDGE2"
					debug_log_scopes = yes
					if = { limit = { has_variable = kehillah_bet_din_lay_seat } debug_log = "KBD_BENCHES JUDGE2 is LAY" }
				}
				var:kehillah_bet_din_judge3 ?= {
					debug_log = "KBD_BENCHES JUDGE3"
					debug_log_scopes = yes
					if = { limit = { has_variable = kehillah_bet_din_lay_seat } debug_log = "KBD_BENCHES JUDGE3 is LAY" }
				}
				if = {
					limit = { has_variable = kehillah_din_rows_open }
					debug_log = "KBD_BENCHES rows window OPEN"
					debug_log_scopes = yes
				}
				every_attending_character = {
					limit = { has_activity_intent = kehillah_bet_din_follow_docket_intent }
					debug_log = "KBD_BENCHES follower"
				}
				every_attending_character = {
					limit = { has_activity_intent = kehillah_bet_din_complaint_intent }
					debug_log = "KBD_BENCHES complainant"
					debug_log_scopes = yes
				}
			}
		}
	}
}
