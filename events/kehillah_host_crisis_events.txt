# =====================================================================
# V36 -- the host relationship, its levers and its crises (docs/spec/
# v36-host-relationship-levers-and-crises.md), 2026-10-13.
# =====================================================================
#   .0001 The Lord's Demands -- a Cold or Hostile host demands a levy, or
#         a worse charter (kehillah_host_demands_check_effect).
#   .0010 A rising strikes -- what the lord did (players; kehillah_
#         popular_rising_strike_effect).
# =====================================================================
namespace = kehillah_host_crisis

kehillah_host_crisis.0001 = {
	type = character_event
	title = kehillah_host_crisis.0001.t
	desc = kehillah_host_crisis.0001.desc
	theme = realm
	left_portrait = {
		character = scope:kehillah_demanding_host
		animation = anger
	}
	right_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		exists = domicile.domicile_location.county.holder.top_liege
	}

	immediate = {
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_demanding_host }
		save_scope_value_as = { name = kehillah_levy value = kehillah_host_levy_value }
		# For the desc: loc reads numbers through a character's variable
		# (Char.MakeScope.Var), not a saved scope value (live 2026-10-13).
		set_variable = { name = kehillah_host_levy value = scope:kehillah_levy }
		if = {
			limit = {
				exists = scope:kehillah_demanding_host.faith.religious_head
				NOT = { scope:kehillah_demanding_host.faith.religious_head = scope:kehillah_demanding_host }
				scope:kehillah_demanding_host.faith.religious_head = { kehillah_outlook_on_jews_value >= 15 }
			}
			scope:kehillah_demanding_host.faith.religious_head = { save_scope_as = kehillah_appeal_hof }
		}
		# The Shtadlan pleads; without one, the leader goes himself.
		if = {
			limit = { exists = cp:councillor_chancellor }
			cp:councillor_chancellor = { save_scope_as = kehillah_plea_envoy }
		}
		else = {
			save_scope_as = kehillah_plea_envoy
		}
	}

	# Pay the levy: certain, and dear.
	option = {
		name = kehillah_host_crisis.0001.a
		trigger = { gold >= scope:kehillah_levy }
		show_as_unavailable = { always = yes }
		pay_short_term_gold = {
			target = scope:kehillah_demanding_host
			gold = scope:kehillah_levy
		}
		scope:kehillah_demanding_host = {
			add_opinion = { target = root modifier = pleased_opinion opinion = 10 }
		}
		kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = kehillah_host_levy_prosperity }
		ai_chance = {
			base = 50
		}
	}

	# Plead with the lord (2026-10-18: the odds' factors shown).
	option = {
		name = {
			trigger = { NOT = { scope:kehillah_plea_envoy = root } }
			text = kehillah_host_crisis.0001.b
		}
		name = {
			trigger = { scope:kehillah_plea_envoy = root }
			text = kehillah_host_crisis.0001.b_self
		}
		random_list = {
			50 = {
				desc = kehillah_host_crisis_plea_success
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_demanding_host }
				compare_modifier = {
					target = scope:kehillah_plea_envoy
					value = diplomacy
					multiplier = 2
					offset = inverted_medium_skill_rating
				}
				opinion_modifier = {
					who = scope:kehillah_demanding_host
					opinion_target = root
					multiplier = 0.25
				}
				modifier = {
					add = -10
					scope:kehillah_demanding_host = { has_trait = greedy }
				}
				min = 5
				custom_tooltip = kehillah_host_demands_dropped_tt
				scope:kehillah_demanding_host = {
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = 3 } }
				}
				custom_tooltip = kehillah_host_demands_plea_outlook_tt
			}
			50 = {
				desc = kehillah_host_crisis_plea_failure
				min = 5
				kehillah_host_demands_refused_effect = yes
			}
		}
		ai_chance = {
			base = 30
		}
	}

	# Appeal to the head of the faith: the lord resents it either way.
	option = {
		name = kehillah_host_crisis.0001.c
		trigger = { exists = scope:kehillah_appeal_hof }
		random_list = {
			40 = {
				modifier = {
					add = 25
					custom_tooltip = {
						text = kehillah_factor_hof_protective
						scope:kehillah_appeal_hof = { kehillah_outlook_band_value >= 2 }
					}
				}
				modifier = {
					add = 10
					custom_tooltip = {
						text = kehillah_factor_hof_well_disposed
						scope:kehillah_appeal_hof = { kehillah_outlook_band_value = 1 }
					}
				}
				# A zealous lord fears the Church's rebuke.
				modifier = {
					add = 15
					scope:kehillah_demanding_host = { has_trait = zealous }
				}
				min = 5
				desc = kehillah_host_crisis_appeal_success
				custom_tooltip = kehillah_host_demands_dropped_tt
				scope:kehillah_demanding_host = {
					add_opinion = { target = root modifier = angry_opinion opinion = -15 }
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -3 } }
				}
				custom_tooltip = kehillah_host_demands_appeal_resent_tt
			}
			60 = {
				min = 5
				desc = kehillah_host_crisis_appeal_failure
				kehillah_host_demands_refused_effect = yes
				scope:kehillah_demanding_host = {
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -5 } }
				}
				custom_tooltip = kehillah_host_demands_appeal_angered_tt
			}
		}
		ai_chance = {
			base = 20
		}
	}

	# Refuse.
	option = {
		name = kehillah_host_crisis.0001.d
		kehillah_host_demands_refused_effect = yes
		scope:kehillah_demanding_host = {
			hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -3 } }
		}
		custom_tooltip = kehillah_host_demands_refuse_outlook_tt
		ai_chance = {
			base = 10
		}
	}
}

# V36b. The Lord's Favour -- a Protective or Well-disposed host grants a
# charter term a step better (kehillah_host_favour_check_effect).
kehillah_host_crisis.0002 = {
	type = character_event
	title = kehillah_host_crisis.0002.t
	desc = kehillah_host_crisis.0002.desc
	theme = realm
	left_portrait = {
		character = scope:kehillah_favouring_host
		animation = happiness
	}
	right_portrait = {
		character = root
		animation = personality_rational
	}

	trigger = {
		exists = domicile.domicile_location.county.holder.top_liege
	}

	immediate = {
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_favouring_host }
		kehillah_charter_improve_one_term_effect = yes
	}

	# Thank him with a gift.
	option = {
		name = kehillah_host_crisis.0002.a
		trigger = { gold >= medium_gold_value }
		remove_short_term_gold = medium_gold_value
		scope:kehillah_favouring_host = {
			add_opinion = { target = root modifier = pleased_opinion opinion = 15 }
			custom_tooltip = kehillah_host_favour_gift_outlook_tt
			hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = 3 } }
		}
		ai_chance = {
			base = 40
		}
	}

	# Thank him in the synagogue's prayers.
	option = {
		name = kehillah_host_crisis.0002.b
		ai_chance = {
			base = 60
		}
	}
}
