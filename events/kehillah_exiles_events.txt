# =====================================================================
# EXILES -- V34 section 6, 2026-10-13.
#   .0001 hidden: an AI exile band arrives where it set out for, and
#         founds there if the place still welcomes them.
#   .0002 debug: make root's community expelled exiles now (console).
# =====================================================================
namespace = kehillah_exiles

kehillah_exiles.0001 = {
	type = character_event
	hidden = yes
	trigger = { has_variable = kehillah_exiles_from }
	immediate = {
		if = {
			limit = {
				exists = location.county
				location.county = { kehillah_founding_site_trigger = yes }
			}
			kehillah_found_community_here_effect = yes
			debug_log = "kehillah_exiles.0001: exiles arrived and settled"
		}
		else = {
			debug_log = "kehillah_exiles.0001: exiles arrived but the place no longer welcomes them"
		}
	}
}

# Console: event kehillah_exiles.0002 on a Kehillah leader -- his
# community is driven out now (no seizure), and becomes exiles.
kehillah_exiles.0002 = {
	type = character_event
	hidden = yes
	trigger = { government_has_flag = government_is_kehillah }
	immediate = {
		if = {
			limit = { exists = domicile.domicile_location.county }
			set_variable = { name = kehillah_exiles_from value = domicile.domicile_location.county }
		}
		kehillah_dissolve_community_effect = yes
		debug_log = "kehillah_exiles.0002: community dissolved as exiles"
	}
}
