kehillah_crisis.0200 = {
 type = character_event
 title = kehillah_crisis.0200.t
 desc = kehillah_crisis.0200.desc
 theme = diplomacy
 left_portrait = { character = root animation = thinking }
 right_portrait = { character = scope:kehillah_crisis_title.holder animation = worry }
 trigger = {
  kehillah_crisis_relay_valid_trigger = yes
  scope:kehillah_crisis_title = { NOT = { has_variable = kehillah_crisis_judgement_sent } }
  scope:kehillah_crisis_title.holder.domicile.domicile_location.county.holder.top_liege = root
 }
 immediate = {
  scope:kehillah_crisis_title = {
   set_variable = { name = kehillah_crisis_judgement_sent value = yes }
   set_variable = { name = kehillah_crisis_ruler value = root }
  }
 }
 option = {
  name = kehillah_crisis.0200.a
  scope:kehillah_crisis_title = { set_variable = { name = kehillah_crisis_response value = 1 } set_variable = { name = kehillah_crisis_royal_protection value = yes years = 2 } }
  if = { limit = { gold >= 20 } pay_short_term_gold = { target = scope:kehillah_crisis_title.holder gold = 20 } }
  ai_chance = { base = 35 modifier = { add = 80 kehillah_outlook_on_jews_value >= 15 } modifier = { add = 40 has_trait = just } modifier = { add = 40 scope:kehillah_crisis_title.var:kehillah_crisis_had_protection > 0 } }
 }
 option = {
  name = kehillah_crisis.0200.b
  scope:kehillah_crisis_title = { set_variable = { name = kehillah_crisis_response value = 2 } }
  ai_chance = { base = 25 modifier = { add = 35 has_trait = greedy } }
 }
 option = {
  name = kehillah_crisis.0200.c
  scope:kehillah_crisis_title = { set_variable = { name = kehillah_crisis_response value = 3 } }
  ai_chance = { base = 20 modifier = { add = 25 has_trait = craven } }
 }
 option = {
  name = kehillah_crisis.0200.d
  trigger = { kehillah_outlook_on_jews_value < -15 scope:kehillah_crisis_title.var:kehillah_crisis_had_protection = 0 }
  scope:kehillah_crisis_title = { set_variable = { name = kehillah_crisis_response value = 4 } }
  ai_chance = { base = 5 modifier = { add = 35 kehillah_outlook_on_jews_value <= -35 } modifier = { add = 20 has_trait = zealous } }
 }
 after = { hidden_effect = { kehillah_crisis_deliver_judgement_effect = yes } }
}
