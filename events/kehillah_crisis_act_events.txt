namespace = kehillah_crisis_act

# =====================================================================
# The three standing actions' scenes, 2026-11-01
# (docs/spec/2026-11-01-three-standing-actions.md). Opened by
# kehillah_crisis_prepare_decision, _protect_decision and _case_decision.
# Each option is one sub-action. An option that does not belong to this
# phase or opening is hidden; one that belongs but is blocked shows
# greyed out with the reason. The checks and effects are the ones the
# old one-row-per-action decisions used.
# Pages (2026-11-01): a scene shows three actions, then More options; Not
# now only on the last page. Each scene's immediate lists its actions in
# order (kehillah_crisis_act_page_start_effect and _list_effect), and each
# option shows only on its page (kehillah_crisis_act_on_page_trigger).
#   .0001 Prepare the Quarter
#   .0002 Seek Protection
#   .0003 Make Our Case
# Scope: the community leader.
# =====================================================================

scripted_effect kehillah_crisis_act_scopes_effect = {
	primary_title.var:kehillah_mob_threat ?= { save_scope_as = kehillah_crisis_agitator }
	# An accusation's witness, while they live (2026-11-01).
	primary_title.var:kehillah_acc_witness ?= {
		if = {
			limit = { is_alive = yes }
			save_scope_as = kehillah_crisis_act_witness
		}
	}
	domicile.domicile_location.county.holder ?= { save_scope_as = kehillah_crisis_lord }
	# Granada: Joseph's wife and son, and the community of Lucena that could take them in.
	if = {
		limit = { kehillah_crisis_opening_is_trigger = { KEY = vizier } }
		character:9000234 ?= {
			if = {
				limit = { is_alive = yes is_courtier_of = root }
				save_scope_as = kehillah_granada_son
			}
		}
		character:9000233 ?= {
			if = {
				limit = { is_alive = yes is_courtier_of = root }
				save_scope_as = kehillah_granada_wife
			}
		}
		title:d_kehillah_lucena.holder ?= { save_scope_as = kehillah_granada_lucena }
	}
}
