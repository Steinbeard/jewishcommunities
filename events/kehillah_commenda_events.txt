namespace = kehillah_commenda
# =====================================================================
# The commenda returns (2026-11-20). common/decisions/kehillah_commenda_decisions.txt.
#   .0001 The voyage comes home -- or does not. The roll is in the option, so
#         its odds show on the tooltip.
# =====================================================================

kehillah_commenda.0001 = {
	type = character_event
	title = kehillah_commenda.0001.t
	desc = kehillah_commenda.0001.desc
	theme = stewardship
	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		primary_title ?= { has_variable = kehillah_commenda_stake }
	}

	immediate = {
		primary_title = {
			save_scope_value_as = { name = kehillah_commenda_stake value = var:kehillah_commenda_stake }
			remove_variable = kehillah_commenda_stake
		}
	}

	option = {
		name = kehillah_commenda.0001.a
		random_list = {
			45 = {
				desc = kehillah_commenda.0001.profit
				modifier = {
					add = 15
					domicile ?= { has_domicile_building_or_higher = kehillah_diaspora_network_01 }
				}
				modifier = {
					add = 10
					domicile ?= { has_domicile_building_or_higher = kehillah_funduq_03 }
				}
				modifier = {
					add = 10
					stewardship >= 12
				}
				add_gold = {
					value = scope:kehillah_commenda_stake
					multiply = 1.7
				}
				kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = kehillah_commenda_profit_prosperity }
				send_interface_toast = {
					type = event_toast_text_and_effect_good
					title = kehillah_toast_success
					desc = kehillah_commenda.0001.profit_desc
					left_icon = root
				}
			}
			30 = {
				desc = kehillah_commenda.0001.even
				add_gold = scope:kehillah_commenda_stake
				send_interface_toast = {
					type = event_toast_text_and_effect_neutral
					title = kehillah_commenda.0001.even
					desc = kehillah_commenda.0001.even_desc
					left_icon = root
				}
			}
			25 = {
				desc = kehillah_commenda.0001.loss
				modifier = {
					add = -10
					domicile ?= { has_domicile_building_or_higher = kehillah_diaspora_network_01 }
				}
				# 2026-11-25: the Rosh ha-Shmirah's Escort the Travellers.
				modifier = {
					add = kehillah_escort_commenda_loss_reduction
					cp:councillor_marshal ?= { is_performing_council_task = task_kehillah_escort_travellers }
				}
				add_gold = {
					value = scope:kehillah_commenda_stake
					multiply = 0.2
				}
				send_interface_toast = {
					type = event_toast_text_and_effect_bad
					title = kehillah_toast_failure
					desc = kehillah_commenda.0001.loss_desc
					left_icon = root
				}
			}
		}
	}
}
