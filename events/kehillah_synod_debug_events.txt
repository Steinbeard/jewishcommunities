# Console-only probes for V45 (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
#   .1  report: general rite's marriage doctrine, vanilla markers, Ashkenaz
#       founded or deferred, and the scoped character's rite
#   .2  fire the 1066 Rhineland synod on the scoped character
#   .3  found Minhag Ashkenaz directly (changes the save), then report
namespace = kehillah_synod_debug

kehillah_synod_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { rite:rabbinism = { rite_has_doctrine = doctrine_polygamy } }
			debug_log = "synod_debug: rabbinism polygamy"
		}
		else_if = {
			limit = { rite:rabbinism = { rite_has_doctrine = doctrine_monogamy } }
			debug_log = "synod_debug: rabbinism monogamy"
		}
		else = { debug_log = "synod_debug: rabbinism other marriage doctrine" }
		if = {
			limit = {
				rite:rabbinism = {
					has_variable = rabbinicism_accepted_monogamy
					has_variable = rabbinicism_rejected_monogamy
				}
			}
			debug_log = "synod_debug: vanilla yearly event suppressed"
		}
		else = { debug_log = "synod_debug: vanilla yearly event NOT suppressed" }
		if = {
			limit = { exists = global_var:kehillah_minhag_ashkenaz_founded }
			debug_log = "synod_debug: ashkenaz founded"
			if = {
				limit = { rite:minhag_ashkenaz = { rite_has_doctrine = doctrine_monogamy } }
				debug_log = "synod_debug: ashkenaz monogamy"
			}
			if = {
				limit = { rite:minhag_ashkenaz = { rite_has_culture = culture:ashkenazi } }
				debug_log = "synod_debug: ashkenaz has ashkenazi culture"
			}
		}
		else = { debug_log = "synod_debug: ashkenaz not founded" }
		if = {
			limit = { exists = global_var:kehillah_gershom_deferred }
			debug_log = "synod_debug: gershom deferred"
		}
		if = {
			limit = { exists = global_var:kehillah_minhag_ashkenaz_founded rite = rite:minhag_ashkenaz }
			debug_log = "synod_debug: scoped character is Minhag Ashkenaz"
		}
		else_if = {
			limit = { rite = rite:rabbinism }
			debug_log = "synod_debug: scoped character is Rabbinism"
		}
		else = { debug_log = "synod_debug: scoped character has another rite" }
		if = {
			limit = { kehillah_in_ashkenazi_region_trigger = yes }
			debug_log = "synod_debug: scoped character in Ashkenaz"
		}
		if = {
			limit = { kehillah_synod_has_open_takkanah_trigger = yes }
			debug_log = "synod_debug: open takkanah available"
		}
		if = {
			limit = { kehillah_synod_region_has_quorum_trigger = yes }
			debug_log = "synod_debug: region has quorum"
		}
		if = {
			limit = { kehillah_is_regional_greatness_leader_trigger = yes }
			debug_log = "synod_debug: regional greatness leader"
		}
		debug_log_scopes = yes
	}
}

kehillah_synod_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		trigger_event = kehillah_synod.0001
		debug_log = "synod_debug: fired kehillah_synod.0001"
	}
}

kehillah_synod_debug.3 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		kehillah_found_minhag_ashkenaz_effect = yes
		trigger_event = kehillah_synod_debug.1
	}
}
