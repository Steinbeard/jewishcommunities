# Console-only probes for V45 (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
#   .1  report: general rite's marriage doctrine, vanilla markers, Ashkenaz
#       founded or deferred, and the scoped character's rite
#   .2  fire the 1066 Rhineland synod on the scoped character
#   .3  found Minhag Ashkenaz directly (changes the save), then report
namespace = kehillah_synod_debug

kehillah_synod_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { rite:rabbinism = { rite_has_doctrine = doctrine_polygamy } }
			debug_log = "synod_debug: rabbinism polygamy"
		}
		else_if = {
			limit = { rite:rabbinism = { rite_has_doctrine = doctrine_monogamy } }
			debug_log = "synod_debug: rabbinism monogamy"
		}
		else = { debug_log = "synod_debug: rabbinism other marriage doctrine" }
		if = {
			limit = {
				rite:rabbinism = {
					has_variable = rabbinicism_accepted_monogamy
					has_variable = rabbinicism_rejected_monogamy
				}
			}
			debug_log = "synod_debug: vanilla yearly event suppressed"
		}
		else = { debug_log = "synod_debug: vanilla yearly event NOT suppressed" }
		if = {
			limit = { exists = global_var:kehillah_minhag_ashkenaz_founded }
			debug_log = "synod_debug: ashkenaz founded"
			if = {
				limit = { rite:minhag_ashkenaz = { rite_has_doctrine = doctrine_monogamy } }
				debug_log = "synod_debug: ashkenaz monogamy"
			}
			if = {
				limit = { rite:minhag_ashkenaz = { rite_has_culture = culture:ashkenazi } }
				debug_log = "synod_debug: ashkenaz has ashkenazi culture"
			}
		}
		else = { debug_log = "synod_debug: ashkenaz not founded" }
		if = {
			limit = { exists = global_var:kehillah_gershom_deferred }
			debug_log = "synod_debug: gershom deferred"
		}
		if = {
			limit = { exists = global_var:kehillah_minhag_ashkenaz_founded rite = rite:minhag_ashkenaz }
			debug_log = "synod_debug: scoped character is Minhag Ashkenaz"
		}
		else_if = {
			limit = { rite = rite:rabbinism }
			debug_log = "synod_debug: scoped character is Rabbinism"
		}
		else = { debug_log = "synod_debug: scoped character has another rite" }
		if = {
			limit = { kehillah_in_ashkenazi_region_trigger = yes }
			debug_log = "synod_debug: scoped character in Ashkenaz"
		}
		if = {
			limit = { kehillah_synod_has_open_takkanah_trigger = yes }
			debug_log = "synod_debug: open takkanah available"
		}
		if = {
			limit = { kehillah_synod_region_has_quorum_trigger = yes }
			debug_log = "synod_debug: region has quorum"
		}
		if = {
			limit = { kehillah_is_regional_greatness_leader_trigger = yes }
			debug_log = "synod_debug: regional greatness leader"
		}
		if = {
			limit = { kehillah_is_sanhedrin_head_trigger = yes }
			debug_log = "synod_debug: head of the Sanhedrin"
		}
		if = {
			limit = { kehillah_sanhedrin_has_quorum_trigger = yes }
			debug_log = "synod_debug: sanhedrin has quorum"
		}
		if = {
			limit = { has_variable = kehillah_synod_cooldown }
			debug_log = "synod_debug: synod cooldown active"
		}
		debug_log_scopes = yes
	}
}

kehillah_synod_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		trigger_event = kehillah_synod.0001
		debug_log = "synod_debug: fired kehillah_synod.0001"
	}
}

kehillah_synod_debug.3 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		kehillah_found_minhag_ashkenaz_effect = yes
		trigger_event = kehillah_synod_debug.1
	}
}

# V46 the Sanhedrin.
#   .4  make the scoped character eligible to restore it (changes the save):
#       the Rabbi trait, all three tracks mastered, and piety for level 5
#   .5  report: Sanhedrin holder, the faith's head, seats, the scoped
#       character's standing
kehillah_synod_debug.4 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { NOT = { has_trait = kehillah_rabbi_trait } }
			add_trait = kehillah_rabbi_trait
		}
		add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = 100 }
		add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 100 }
		add_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = 100 }
		add_piety = 50000
		trigger_event = kehillah_synod_debug.5
	}
}

kehillah_synod_debug.5 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { exists = title:d_kehillah_sanhedrin.holder }
			debug_log = "synod_debug: sanhedrin exists"
			title:d_kehillah_sanhedrin.holder = { debug_log_scopes = yes }
		}
		else = { debug_log = "synod_debug: sanhedrin does not exist" }
		if = {
			limit = { exists = faith:rabbinic_faith.religious_head }
			debug_log = "synod_debug: rabbinic faith has a head"
		}
		else = { debug_log = "synod_debug: rabbinic faith has no head" }
		if = {
			limit = { faith:rabbinic_faith = { any_clerical_elector_title = { has_variable = kehillah_sanhedrin_seat } } }
			debug_log = "synod_debug: sanhedrin seats exist"
		}
		else = { debug_log = "synod_debug: no sanhedrin seats" }
		if = {
			limit = { kehillah_masters_all_rabbinic_tracks_trigger = yes }
			debug_log = "synod_debug: scoped character masters all tracks"
		}
		if = {
			limit = { piety_level >= 5 }
			debug_log = "synod_debug: scoped character at top piety level"
		}
		if = {
			limit = { kehillah_sanhedrin_has_founding_members_trigger = yes }
			debug_log = "synod_debug: enough founding members"
		}
		if = {
			limit = { kehillah_holds_sanhedrin_seat_trigger = yes }
			debug_log = "synod_debug: scoped character holds a seat"
		}
		if = {
			limit = { primary_title ?= { is_kehillah_title_trigger = yes } }
			debug_log = "synod_debug: primary title is the community"
		}
		else = { debug_log = "synod_debug: PRIMARY TITLE IS NOT THE COMMUNITY" }
		debug_log_scopes = yes
	}
}
