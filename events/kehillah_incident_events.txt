# =====================================================================
# V41 step 4b -- new incidents for the director (docs/spec/v41-danger-
# incidents-and-the-pogrom-unfolding.md section 7), 2026-10-14. Each is an
# incident in the county's faith (+1); the odds of each answer come from
# the people involved (kehillah_incident_scene_values.txt).
#   .0001 The Talmud on Trial  (Paris, 1240: Nicholas Donin)
#   .0010 The Badge            (Lateran IV, 1215; the ghiyar under Islam)
#   .0020 Holy Week            (the blow at Toulouse; the stones of
#                               Beziers, ended for a payment in 1160)
#   .0030 The Bonds            (York, 1190: the debt records burned)
#   .0090 What came of it (players)
# Scope: a Kehillah leader. Fired by kehillah_incident_director_effect.
# =====================================================================

# ---------------------------------------------------------------------
# .0001 The Talmud on Trial. A Christian host.
# ---------------------------------------------------------------------
kehillah_incident.0001 = {
	type = character_event
	title = kehillah_incident.0001.t
	desc = kehillah_incident.0001.desc
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:kehillah_incident_accuser
		animation = scheme
	}
	trigger = {
		exists = primary_title
		exists = domicile.domicile_location.county.holder.top_liege
	}
	immediate = {
		kehillah_incident_scopes_effect = yes
		domicile.domicile_location.county.faith ?= { kehillah_faith_pressure_add_effect = { AMOUNT = 1 } }
		primary_title = { set_variable = { name = kehillah_talmud_cooldown value = yes years = kehillah_talmud_cooldown_years } }
		# A man who studied among us and was baptized (Donin).
		create_character = {
			age = { 28 50 }
			gender = male
			faith = scope:kehillah_incident_host.faith
			culture = root.culture
			location = scope:kehillah_incident_town
			dynasty = none
			trait = zealous
			learning = { 12 20 }
			intrigue = { 6 14 }
			save_scope_as = kehillah_incident_accuser
		}
		scope:kehillah_incident_host = { add_courtier = scope:kehillah_incident_accuser }
	}

	# We will answer him in the hall: your learning against his.
	option = {
		name = kehillah_incident.0001.a
		duel = {
			skill = learning
			target = scope:kehillah_incident_accuser
			50 = {
				desc = kehillah_incident_out_acquitted
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				modifier = {
					add = 10
					scope:kehillah_incident_host = { has_trait = just }
				}
				min = 5
				custom_tooltip = kehillah_incident_talmud_safe_tt
				kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_talmud_acquitted_greatness }
				kehillah_incident_report_effect = { OUTCOME = talmud_acquitted }
			}
			50 = {
				desc = kehillah_incident_out_condemned
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
				kehillah_talmud_burned_effect = yes
				kehillah_incident_report_effect = { OUTCOME = talmud_burned }
			}
		}
		ai_chance = {
			base = 25
			modifier = {
				add = 40
				learning > scope:kehillah_incident_accuser.learning
			}
		}
	}

	# The Dayyan answers him: a more learned voice than yours.
	option = {
		name = kehillah_incident.0001.f
		trigger = {
			exists = scope:kehillah_incident_dayyan
			scope:kehillah_incident_dayyan.learning > learning
		}
		# The Dayyan is the challenger; the community still bears the result.
		scope:kehillah_incident_dayyan = {
			duel = {
				skill = learning
				target = scope:kehillah_incident_accuser
				50 = {
					desc = kehillah_incident_out_acquitted
					compare_modifier = {
						value = scope:duel_value
						multiplier = 3.5
					}
					modifier = {
						add = 15
						learning >= 18
					}
					kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
					min = 5
					root = {
						custom_tooltip = kehillah_incident_talmud_safe_tt
						kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_talmud_acquitted_greatness }
						kehillah_incident_report_effect = { OUTCOME = talmud_acquitted }
					}
				}
				50 = {
					desc = kehillah_incident_out_condemned
					compare_modifier = {
						value = scope:duel_value
						multiplier = -3.5
					}
					min = 5
					root = {
						kehillah_talmud_burned_effect = yes
						kehillah_incident_report_effect = { OUTCOME = talmud_burned }
					}
				}
			}
		}
		ai_chance = { base = 35 }
	}

	# The Shtadlan goes to the castle before the day comes.
	option = {
		name = kehillah_incident.0001.e
		trigger = { exists = scope:kehillah_incident_shtadlan }
		random_list = {
			30 = {
				desc = kehillah_incident_out_dropped
				modifier = {
					add = 15
					scope:kehillah_incident_shtadlan = { diplomacy >= 12 }
				}
				modifier = {
					add = 15
					scope:kehillah_incident_shtadlan = { diplomacy >= 18 }
				}
				opinion_modifier = {
					who = scope:kehillah_incident_host
					opinion_target = scope:kehillah_incident_shtadlan
					multiplier = 0.25
				}
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				min = 5
				custom_tooltip = kehillah_incident_talmud_safe_tt
				kehillah_incident_report_effect = { OUTCOME = talmud_dropped }
			}
			70 = {
				desc = kehillah_incident_out_condemned
				kehillah_talmud_burned_effect = yes
				kehillah_incident_report_effect = { OUTCOME = talmud_burned }
			}
		}
		ai_chance = { base = 30 }
	}

	# Hide the books: your intrigue against his (he knows which houses).
	option = {
		name = kehillah_incident.0001.b
		duel = {
			skill = intrigue
			target = scope:kehillah_incident_accuser
			40 = {
				desc = kehillah_incident_out_hidden
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				modifier = {
					add = 10
					has_trait = deceitful
				}
				modifier = {
					add = 10
					has_trait = schemer
				}
				min = 5
				custom_tooltip = kehillah_incident_talmud_safe_tt
				kehillah_incident_report_effect = { OUTCOME = talmud_hidden }
			}
			60 = {
				desc = kehillah_incident_out_found
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
				kehillah_talmud_burned_effect = yes
				scope:kehillah_incident_host ?= {
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -5 } }
				}
				kehillah_incident_report_effect = { OUTCOME = talmud_found }
			}
		}
		ai_chance = {
			base = 15
			modifier = {
				add = 20
				OR = {
					has_trait = deceitful
					has_trait = schemer
				}
			}
		}
	}

	# Offer the lord silver to give them back.
	option = {
		name = kehillah_incident.0001.c
		trigger = { gold >= kehillah_incident_bribe_cost_value }
		show_as_unavailable = { always = yes }
		pay_short_term_gold = {
			target = scope:kehillah_incident_host
			gold = kehillah_incident_bribe_cost_value
		}
		random_list = {
			40 = {
				desc = kehillah_incident_out_redeemed
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				modifier = {
					add = 20
					scope:kehillah_incident_host = { has_trait = greedy }
				}
				min = 5
				custom_tooltip = kehillah_incident_talmud_safe_tt
				kehillah_incident_report_effect = { OUTCOME = talmud_redeemed }
			}
			60 = {
				desc = kehillah_incident_out_cheated
				modifier = {
					add = 15
					scope:kehillah_incident_host = { has_trait = just }
				}
				min = 5
				kehillah_talmud_burned_effect = yes
				kehillah_incident_report_effect = { OUTCOME = talmud_cheated }
			}
		}
		ai_chance = {
			base = 15
			modifier = {
				add = 20
				scope:kehillah_incident_host = { has_trait = greedy }
			}
		}
	}

	# Let them take the books.
	option = {
		name = kehillah_incident.0001.d
		kehillah_talmud_burned_effect = yes
		kehillah_incident_report_effect = { OUTCOME = talmud_burned }
		ai_chance = { base = 5 }
	}
}

		name = kehillah_incident.0010.e

# .0020 Holy Week. A Christian host, March or April.
# ---------------------------------------------------------------------
kehillah_incident.0020 = {
	type = character_event
	title = kehillah_incident.0020.t
	desc = {
		random_valid = {
			desc = kehillah_incident.0020.desc_blow
			desc = kehillah_incident.0020.desc_stones
		}
	}
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		trigger = { exists = scope:kehillah_incident_sent }
		character = scope:kehillah_incident_sent
		animation = fear
	}
	trigger = {
		exists = primary_title
		exists = domicile.domicile_location.county.holder.top_liege
	}
	immediate = {
		kehillah_incident_scopes_effect = yes
		domicile.domicile_location.county.faith ?= { kehillah_faith_pressure_add_effect = { AMOUNT = 1 } }
		primary_title = { set_variable = { name = kehillah_holy_week_cooldown value = yes years = kehillah_holy_week_cooldown_years } }
		random_courtier = {
			limit = {
				is_adult = yes
				is_male = yes
				is_jewish_character_trigger = yes
				NOT = { root.player_heir ?= this }
				NOT = { is_spouse_of = root }
			}
			save_scope_as = kehillah_incident_sent
		}
	}

	# I will go myself.
	option = {
		name = kehillah_incident.0020.a
		add_piety = medium_piety_gain
		add_stress = medium_stress_gain
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_holy_week_self_stability }
		random_list = {
			70 = {
				compare_modifier = {
					value = prowess
					multiplier = 2
				}
				desc = kehillah_incident_out_bruised
				kehillah_incident_report_effect = { OUTCOME = holy_week_bruised }
			}
			30 = {
				compare_modifier = {
					value = prowess
					multiplier = -2
				}
				min = 5
				desc = kehillah_incident_out_wounded
				increase_wounds_no_death_effect = { REASON = beaten }
				kehillah_incident_report_effect = { OUTCOME = holy_week_wounded }
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				add = 30
				OR = {
					has_trait = brave
					has_trait = humble
				}
			}
		}
	}

	# Send another man.
	option = {
		name = kehillah_incident.0020.b
		trigger = { exists = scope:kehillah_incident_sent }
		scope:kehillah_incident_sent = {
			add_stress = medium_stress_gain
			add_opinion = {
				target = root
				modifier = angry_opinion
				opinion = -25
			}
		}
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_holy_week_sent_stability }
		kehillah_incident_report_effect = { OUTCOME = holy_week_sent }
		ai_chance = {
			base = 40
			modifier = {
				add = 30
				OR = {
					has_trait = craven
					has_trait = arrogant
				}
			}
		}
	}

	# Pay the bishop to end the custom.
	option = {
		name = kehillah_incident.0020.c
		trigger = { gold >= kehillah_incident_bribe_cost_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = kehillah_incident_bribe_cost_value
		random_list = {
			35 = {
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				modifier = {
					add = 20
					scope:kehillah_incident_host = { has_trait = greedy }
				}
				min = 5
				desc = kehillah_incident_out_ended
				custom_tooltip = kehillah_incident_holy_week_ended_tt
				primary_title = { set_variable = { name = kehillah_holy_week_redeemed value = yes } }
				kehillah_incident_report_effect = { OUTCOME = holy_week_ended }
			}
			65 = {
				modifier = {
					add = 10
					scope:kehillah_incident_host = { has_trait = just }
				}
				min = 5
				desc = kehillah_incident_out_kept
				kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_holy_week_sent_stability }
				kehillah_incident_report_effect = { OUTCOME = holy_week_kept }
			}
		}
		ai_chance = { base = 15 }
	}

	# Nobody goes. (Brave or wrathful.)
	option = {
		name = kehillah_incident.0020.d
		trigger = {
			OR = {
				has_trait = brave
				has_trait = wrathful
			}
		}
		random_list = {
			20 = {
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				compare_modifier = {
					value = prestige_level
					multiplier = 5
				}
				min = 5
				desc = kehillah_incident_out_refused
				add_prestige = medium_prestige_gain
				kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_holy_week_refuse_greatness }
				kehillah_incident_report_effect = { OUTCOME = holy_week_refused }
			}
			80 = {
				desc = kehillah_incident_out_riot
				custom_tooltip = kehillah_incident_mob_tt
				hidden_effect = { trigger_event = { id = kehillah_mob.0030 days = 1 } }
				kehillah_incident_report_effect = { OUTCOME = holy_week_riot }
			}
		}
		ai_chance = { base = 15 }
	}
}

					has_trait = deceitful
					has_trait = schemer
				}
			}
		}
	}

	# Ask the lord to guard the house.
	option = {
		name = kehillah_incident.0030.c
		random_list = {
			40 = {
				kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
				modifier = {
					add = -25
					custom_tooltip = {
						text = kehillah_factor_lord_owes_us
						scope:kehillah_incident_host = { exists = var:kehillah_loan_amount_owed }
					}
				}
				modifier = {
					add = -10
					scope:kehillah_incident_host = { has_trait = greedy }
				}
				min = 5
				desc = kehillah_incident_out_guarded
				custom_tooltip = kehillah_incident_bonds_safe_tt
				kehillah_incident_report_effect = { OUTCOME = bonds_guarded }
			}
			60 = {
				desc = kehillah_incident_out_riot
				kehillah_bonds_burned_effect = yes
				custom_tooltip = kehillah_incident_mob_tt
				hidden_effect = { trigger_event = { id = kehillah_mob.0030 days = 1 } }
				kehillah_incident_report_effect = { OUTCOME = bonds_riot }
			}
		}
		ai_chance = { base = 30 }
	}

	# Forgive his debt before they come.
	option = {
		name = kehillah_incident.0030.d
		trigger = { exists = scope:kehillah_incident_debtor }

			scope:kehillah_incident_debtor = {

				opinion_modifier = {
					who = scope:kehillah_incident_debtor
					opinion_target = root
					multiplier = 0.25
				}
				min = 10
				desc = kehillah_incident_out_forgiven
				custom_tooltip = kehillah_incident_bonds_safe_tt
				scope:kehillah_incident_debtor = {
					add_opinion = {
						target = root
						modifier = grateful_opinion
						opinion = 20
