# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs

# =====================================================================
#   .0001 The letter reaches the posek.
#   .0002 The reply reaches the asking community.
# =====================================================================

	# Answer.
	option = {

		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sheelah_answer_greatness }

			add_opinion = { target = root modifier = respect_opinion opinion = 10 }

		ai_chance = { base = 60 }

	option = {
		name = kehillah_sheelah.0001.b
		trigger = { kehillah_sheelah_can_cite_trigger = yes }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sheelah_careful_greatness }
		add_learning_lifestyle_xp = 10
		add_stress = minor_stress_gain
		scope:kehillah_sheelah_asker = {
			add_opinion = { target = root modifier = respect_opinion opinion = 20 }
		}
		add_opinion = { target = scope:kehillah_sheelah_asker modifier = kehillah_din_consulted_opinion }
		kehillah_sheelah_reply_effect = { QUALITY = 2 }
		ai_chance = { base = 80 }
	}
	# Decline: a great community is expected to answer.
	option = {
		name = kehillah_sheelah.0001.c
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sheelah_decline_greatness }
		kehillah_sheelah_reply_effect = { QUALITY = 0 }
		ai_chance = { base = 5 }
	}
}
