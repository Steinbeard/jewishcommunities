# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs
# a gift to the answering community and is limited by a cooldown; answering
# is the reward of Greatness. Content: the Bet Din docket's 14 cases
# (kehillah_din_letter_N, kehillah_din_answer_N, kehillah_din_answer_careful_N).
# =====================================================================
#   .0001 The letter reaches the posek.
#   .0002 The reply reaches the asking community.
#   .0021 Slice 3: a strong tie offers the followed community's rite.
# =====================================================================

	right_portrait = { character = scope:kehillah_sheelah_asker animation = worry }

		}

		kehillah_sheelah_tie_posek_tooltip_effect = yes
		kehillah_sheelah_reply_effect = { QUALITY = 2 }
		ai_chance = { base = 80 }
	}
	# Decline: a great community is expected to answer.
	option = {
		name = kehillah_sheelah.0001.c
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sheelah_decline_greatness }
		kehillah_sheelah_reply_effect = { QUALITY = 0 }
		ai_chance = { base = 5 }
	}
}

	}

		var:kehillah_study_help_reply_from = { save_scope_as = kehillah_study_teacher }
	}

	# A full explanation: the current phase succeeds.
	option = {
		name = kehillah_sheelah.0012.a
		trigger = {
			var:kehillah_study_help_reply_quality = 2
			var:kehillah_study_help_work ?= var:kehillah_study_current_work_id
		}
		custom_tooltip = kehillah_sheelah.0012.mastery_tt
		hidden_effect = { set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 } }
		random_scheme = {
			type = kehillah_study_torah
			add_scheme_progress = kehillah_study_help_progress
		}
	}
	# An explanation: the current phase moves on.
	option = {
		name = kehillah_sheelah.0012.a
		trigger = {
			var:kehillah_study_help_reply_quality = 1
			var:kehillah_study_help_work ?= var:kehillah_study_current_work_id
		}
		random_scheme = {
			type = kehillah_study_torah
			add_scheme_progress = kehillah_study_help_progress
		}
	}
	# Declined, or the student has moved on to another work.
	option = {
		name = kehillah_sheelah.0012.b
		trigger = {
			OR = {
				var:kehillah_study_help_reply_quality = 0
				NOT = { var:kehillah_study_help_work ?= var:kehillah_study_current_work_id }
			}
		}
	}

	after = {
		remove_variable = kehillah_study_help_reply_from
		remove_variable = kehillah_study_help_reply_quality
		remove_variable = kehillah_study_help_work
	}

# =====================================================================
# SLICE 3: rite pull (2026-11-22). Spec sections 5 and 12.
#   .0021 A strong tie to a community of another rite of our faith: take up
#         its customs, or keep our own. Fired by kehillah_sheelah_tie_yearly_effect,
#         at most once in ten years. The AI weighs the tie's strength against
#         its own Stability.
# =====================================================================

kehillah_sheelah.0021 = {
	type = character_event
	title = kehillah_sheelah.0021.t
	desc = kehillah_sheelah.0021.desc
	theme = faith
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:kehillah_rite_pull_leader animation = personality_rational }

	trigger = {
		kehillah_sheelah_is_leader_trigger = yes
		primary_title = { kehillah_sheelah_tie_sound_trigger = yes }
		exists = primary_title.var:kehillah_sheelah_follows.holder.rite
		NOT = { rite = primary_title.var:kehillah_sheelah_follows.holder.rite }
	}

	immediate = {
		primary_title.var:kehillah_sheelah_follows = {
			save_scope_as = kehillah_rite_pull_title
			holder = { save_scope_as = kehillah_rite_pull_leader }
		}
	}

	# Take up their customs.
	option = {
		name = kehillah_sheelah.0021.a
		kehillah_sheelah_adopt_rite_effect = yes
		ai_chance = {
			base = 30
			modifier = {
				add = 20
				primary_title.var:kehillah_sheelah_follows_strength >= 8
			}
			modifier = {
				add = 20
				primary_title.var:kehillah_sheelah_follows_strength >= 10
			}
			# A settled community keeps its customs; a troubled one looks
			# to the authority it already follows.
			modifier = {
				add = -15
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_flourishing_threshold }
			}
			modifier = {
				add = -15
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_legendary_threshold }
			}
			modifier = {
				add = 15
				NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold } }
			}
			modifier = {
				add = 15
				NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_strained_threshold } }
			}
			modifier = { add = -20 has_trait = stubborn }
			modifier = { add = -20 has_trait = zealous }
			modifier = { add = 10 has_trait = humble }
			min = 0
		}
	}

	# Keep our own.
	option = {
		name = kehillah_sheelah.0021.b
		custom_tooltip = kehillah_sheelah.0021.b.tt
		hidden_effect = {
			primary_title = {
				change_variable = { name = kehillah_sheelah_follows_strength subtract = kehillah_sheelah_rite_pull_refusal_loss }
				if = {
					limit = { var:kehillah_sheelah_follows_strength <= 0 }
					kehillah_sheelah_tie_clear_effect = yes
				}
			}
		}
		ai_chance = { base = 50 }
	}
}
