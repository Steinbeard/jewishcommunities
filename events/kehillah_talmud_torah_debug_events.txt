# Console-only probes. Run in a disposable save as the intended reader.
# .2 resolves Mishnah as a successful study course and changes the save.
namespace = kehillah_talmud_torah_debug

kehillah_talmud_torah_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { rite = { rite_has_tenet = tenet_talmud_torah } }
			debug_log = "talmud_torah_debug: tenet present"
		}
		else = { debug_log = "talmud_torah_debug: tenet absent" }
		if = {
			limit = { kehillah_torah_reader_eligible_trigger = yes }
			debug_log = "talmud_torah_debug: reader eligible"
		}
		else = { debug_log = "talmud_torah_debug: reader ineligible" }
		debug_log_scopes = yes
	}
}

kehillah_talmud_torah_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		set_variable = { name = kehillah_study_torah_mastery_bonus value = 1 }
		kehillah_study_specific_work_effect = { WORK = flag:work_mishnah TRACK = talmudics }
		if = {
			limit = { var:kehillah_study_torah_last_result = 1 }
			debug_log = "talmud_torah_debug: first mastery"
		}
		else_if = {
			limit = { var:kehillah_study_torah_last_result = 2 }
			debug_log = "talmud_torah_debug: repeat mastery"
		}
		else = { debug_log = "talmud_torah_debug: unexpected result" }
		debug_log_scopes = yes
	}
}
