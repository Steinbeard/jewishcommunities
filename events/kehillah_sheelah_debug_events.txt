namespace = kehillah_sheelah_debug
# =====================================================================
# She'elot slices 2 and 3 (2026-11-22), console only. Fire on a Kehillah
# leader (event kehillah_sheelah_debug.N <character id>).
#   .1  Report: whom this community follows and how strongly, how many
#       follow it, and its Greatness term.
#   .2  Run the yearly tie step now (fade, cleanup, rite pull), ignoring
#       the rite-pull cooldown.
# =====================================================================

kehillah_sheelah_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { exists = primary_title.var:kehillah_sheelah_follows }
			primary_title.var:kehillah_sheelah_follows = { save_scope_as = kehillah_debug_followed }
			save_scope_value_as = { name = kehillah_debug_strength value = primary_title.var:kehillah_sheelah_follows_strength }
			debug_log = "kehillah_sheelah_debug.1: follows (scope kehillah_debug_followed) at strength kehillah_debug_strength"
		}
		else = {
			debug_log = "kehillah_sheelah_debug.1: follows no one"
		}
		save_scope_value_as = { name = kehillah_debug_followers value = kehillah_sheelah_follower_count_value }
		save_scope_value_as = { name = kehillah_debug_follower_greatness value = kehillah_bd_follower_greatness_value }
		debug_log = "kehillah_sheelah_debug.1: followers kehillah_debug_followers, Greatness term kehillah_debug_follower_greatness"
		debug_log_scopes = yes
	}
}

kehillah_sheelah_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		primary_title ?= { remove_variable = kehillah_sheelah_rite_pull_cooldown }
		kehillah_sheelah_tie_yearly_effect = yes
		debug_log = "kehillah_sheelah_debug.2: yearly tie step run"
		debug_log_scopes = yes
	}
}
