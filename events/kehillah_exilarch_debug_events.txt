# Console only: run as the claimant in a disposable campaign.
# Event 1 reports state. For event 2, set the character variable
# kehillah_claim_debug_target to any claimed community title; event 3
# grants the vacant Exilarchate to a qualifying Ben David leader.
namespace = kehillah_exilarch_debug

kehillah_exilarch_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "KEHILLAH_EXILARCH_PROBE: before"
		debug_log_scopes = yes
		title:d_kehillah_baghdad = { debug_log_scopes = yes }
		title:d_kehillah_exilarchate = { debug_log_scopes = yes }
		primary_title ?= { debug_log_scopes = yes }
	}
}

kehillah_exilarch_debug.2 = {
	type = character_event
	hidden = yes
	trigger = {
		is_jewish_character_trigger = yes
		has_government = landless_adventurer_government
		exists = var:kehillah_claim_debug_target
		var:kehillah_claim_debug_target = { is_kehillah_title_trigger = yes exists = holder }
		is_target_in_variable_list = {
			name = kehillah_community_claims
			target = var:kehillah_claim_debug_target
		}
	}
	immediate = {
		var:kehillah_claim_debug_target = { save_scope_as = kehillah_claim_seat }
		kehillah_press_community_claim_effect = yes
		debug_log = "KEHILLAH_SEAT_CLAIM_PROBE: after"
		debug_log_scopes = yes
	}
}

kehillah_exilarch_debug.3 = {
	type = character_event
	hidden = yes
	trigger = {
		is_jewish_character_trigger = yes
		has_trait = ben_david
		government_has_flag = government_is_kehillah
		primary_title = { is_kehillah_title_trigger = yes }
		NOT = { exists = title:d_kehillah_exilarchate.holder }
	}
	immediate = {
		kehillah_grant_exilarchate_effect = yes
		debug_log = "KEHILLAH_EXILARCH_PROBE: recognized"
		debug_log_scopes = yes
	}
}
