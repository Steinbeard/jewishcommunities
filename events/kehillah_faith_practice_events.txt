kehillah_faith_practice.0003 = {
    type = character_event
    trigger = { kehillah_has_sacred_liturgy_trigger = yes }
    title = kehillah_faith_practice.0003.t
    desc = kehillah_faith_practice.0003.desc
    theme = faith
    left_portrait = { character = root animation = reading }
    option = {
        name = kehillah_faith_practice.0003.compose
        random = {
            chance = kehillah_liturgy_composition_chance_value
            add_piety = medium_piety_gain
            add_prestige = minor_prestige_gain
            if = {
                limit = { NOT = { has_trait = lifestyle_poet } }
                if = { limit = { NOT = { has_variable = kehillah_successful_liturgies } } set_variable = { name = kehillah_successful_liturgies value = 0 } }
                change_variable = { name = kehillah_successful_liturgies add = 1 }
                if = { limit = { var:kehillah_successful_liturgies >= 3 } add_trait = lifestyle_poet }
            }
            if = { limit = { has_personal_tenet = tenet_kehillah_sacred_liturgy } add_piety = minor_piety_gain }
        }
        add_learning_lifestyle_xp = 50
        ai_chance = { base = 50 modifier = { factor = 2 has_trait = lifestyle_poet } }
    }
    option = {
        name = kehillah_faith_practice.0003.perform
        add_piety = minor_piety_gain
        add_stress = minor_stress_loss
        if = { limit = { has_personal_tenet = tenet_kehillah_sacred_liturgy } add_prestige = minor_prestige_gain }
        ai_chance = { base = 50 }
    }
}
