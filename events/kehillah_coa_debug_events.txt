# Hidden, console-only probe for the curated community arms pool. On a test
# save, play a Kehillah leader and run: event kehillah_coa_debug.1
# This deliberately rerolls the current title's arms so the template can be
# checked without founding a new community for every sample.
namespace = kehillah_coa_debug

kehillah_coa_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_government = kehillah_government
				exists = primary_title
			}
			primary_title = { kehillah_generate_community_coa_effect = yes }
			debug_log = "kehillah_coa_debug.1: regenerated community title arms"
		}
	}
}
