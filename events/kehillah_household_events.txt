# =====================================================================
# Ba'alei batim (V26 part 2): arrivals, departures and the register.
# Effects: common/scripted_effects/kehillah_household_effects.txt.
# Player-only events; AI communities resolve the same moves directly.
# =====================================================================

namespace = kehillah_household


# ---------------------------------------------------------------------
# kehillah_household.0001 -- A family asks to settle.
# From kehillah_household_arrival_effect. Saved scopes: kh_arr_origin
# (if any community exists to come from), kh_arr_house (if a real
# household is moving from there).
# ---------------------------------------------------------------------
kehillah_household.0001 = {
	type = character_event
	title = kehillah_household.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:kh_arr_house }
				desc = kehillah_household.0001.desc_move
			}
			triggered_desc = {
				trigger = { exists = scope:kh_arr_origin }
				desc = kehillah_household.0001.desc_new
			}
			desc = kehillah_household.0001.desc_unknown
		}
	}
	theme = family
	left_portrait = {
		character = scope:kh_arr_head
		animation = personality_honorable
	}

	immediate = {
		if = {
			limit = { exists = scope:kh_arr_house }
			scope:kh_arr_house.var:kehillah_household_head = { save_scope_as = kh_arr_head }
		}
		else = {
			kehillah_household_create_head_effect = { LEADER = root LOCATION = root.domicile.domicile_location }
			scope:kh_new_head = { save_scope_as = kh_arr_head }
			if = {
				limit = { exists = scope:kh_arr_origin }
				scope:kh_arr_head.house = { set_variable = { name = kehillah_household_origin value = scope:kh_arr_origin } }
			}
		}
	}

	# Welcome them.
	option = {
		name = kehillah_household.0001.a
		custom_tooltip = kehillah_household_joins_tt
		hidden_effect = { trigger_event = kehillah_household.0004 }
	}

	# Admit them for a settlement fee.
	option = {
		name = kehillah_household.0001.b
		add_gold = minor_gold_value
		custom_tooltip = kehillah_household_joins_tt
		hidden_effect = { trigger_event = kehillah_household.0004 }
		scope:kh_arr_head = {
			add_opinion = { target = root modifier = kehillah_household_fee_opinion }
		}
	}

	# Refuse them the right to settle.
	option = {
		name = kehillah_household.0001.c
		if = {
			limit = { exists = scope:kh_arr_house }
			custom_tooltip = kehillah_household_refused_stay_tt
		}
		else_if = {
			limit = { exists = scope:kh_arr_origin.holder }
			custom_tooltip = kehillah_household_refused_return_tt
			hidden_effect = { trigger_event = kehillah_household.0005 }
		}
		else = {
			custom_tooltip = kehillah_household_refused_tt
		}
		ai_chance = { base = 0 }
	}
}

# ---------------------------------------------------------------------
# kehillah_household.0004 / .0005 -- hidden follow-ups to .0001 that do
# the settling. Kept out of .0001's options because the option tooltip
# evaluates its effects before the wife and children exist, which threw
# thousands of script errors in the 2026-10-09 live test.
# ---------------------------------------------------------------------
kehillah_household.0004 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_household_0001_settle_effect = yes
	}
}

kehillah_household.0005 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = scope:kh_arr_origin.holder }
			kehillah_household_settle_new_effect = { HEAD = scope:kh_arr_head LEADER = scope:kh_arr_origin.holder }
		}
	}
}


# ---------------------------------------------------------------------
# kehillah_household.0002 -- A household means to leave.
# From kehillah_household_departure_effect. Saved scopes: kh_dep_house,
# kh_dep_leader (root), kh_dep_dest.
# ---------------------------------------------------------------------
kehillah_household.0002 = {
	type = character_event
	title = kehillah_household.0002.t
	desc = kehillah_household.0002.desc
	theme = family
	left_portrait = {
		character = scope:kh_dep_head
		animation = worry
	}

	immediate = {
		scope:kh_dep_house.var:kehillah_household_head = { save_scope_as = kh_dep_head }
	}

	# Let them go.
	option = {
		name = kehillah_household.0002.a
		custom_tooltip = kehillah_household_leaves_tt
		hidden_effect = { kehillah_household_leave_effect = yes }
	}

	# Cover what staying costs them.
	option = {
		name = kehillah_household.0002.b
		trigger = { gold >= kehillah_household_persuade_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = kehillah_household_persuade_gold_value
		custom_tooltip = kehillah_household_stays_tt
	}

	# Persuade them to stay.
	option = {
		name = kehillah_household.0002.c
		trigger = { diplomacy >= 12 }
		show_as_unavailable = { always = yes }
		custom_tooltip = kehillah_household_stays_tt
	}
}


# ---------------------------------------------------------------------
# kehillah_household.0003 -- A household from another community has
# moved here (the player received an AI community's departing family).
# Saved scopes: kh_mig_house, kh_dep_leader (the community they left).
# ---------------------------------------------------------------------
kehillah_household.0003 = {
	type = character_event
	title = kehillah_household.0003.t
	desc = kehillah_household.0003.desc
	theme = family
	left_portrait = {
		character = scope:kh_new_arrival
		animation = happiness
	}

	trigger = {
		exists = scope:kh_mig_house.var:kehillah_household_head
	}

	immediate = {
		scope:kh_mig_house.var:kehillah_household_head = { save_scope_as = kh_new_arrival }
		scope:kh_dep_leader.primary_title = { save_scope_as = kh_left_title }
	}

	option = {
		name = kehillah_household.0003.a
	}
}


# ---------------------------------------------------------------------
# kehillah_household.0010 -- The register of householders.
# From kehillah_review_households_decision.
# ---------------------------------------------------------------------
kehillah_household.0010 = {
	type = character_event
	title = kehillah_household.0010.t
	desc = {
		desc = kehillah_household.0010.desc
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = scope:kh_rev_1 } }
				desc = kehillah_household.0010.none
			}
			desc = kehillah_household.0010.line_1
		}
		triggered_desc = {
			trigger = { exists = scope:kh_rev_2 }
			desc = kehillah_household.0010.line_2
		}
		triggered_desc = {
			trigger = { exists = scope:kh_rev_3 }
			desc = kehillah_household.0010.line_3
		}
		triggered_desc = {
			trigger = { exists = scope:kh_rev_4 }
			desc = kehillah_household.0010.line_4
		}
		desc = kehillah_household.0010.room
	}
	theme = family
	left_portrait = {
		character = scope:kh_rev_1
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:kh_rev_2
		animation = personality_honorable
	}

	immediate = {
		kehillah_household_bootstrap_effect = yes
		kehillah_household_refresh_effect = yes
		primary_title = {
			ordered_in_list = {
				variable = kehillah_households
				order_by = var:kehillah_household_standing
				position = 0
				var:kehillah_household_head ?= { save_scope_as = kh_rev_1 }
			}
			if = {
				limit = { kehillah_household_count_value > 1 }
				ordered_in_list = {
					variable = kehillah_households
					order_by = var:kehillah_household_standing
					position = 1
					var:kehillah_household_head ?= { save_scope_as = kh_rev_2 }
				}
			}
			if = {
				limit = { kehillah_household_count_value > 2 }
				ordered_in_list = {
					variable = kehillah_households
					order_by = var:kehillah_household_standing
					position = 2
					var:kehillah_household_head ?= { save_scope_as = kh_rev_3 }
				}
			}
			if = {
				limit = { kehillah_household_count_value > 3 }
				ordered_in_list = {
					variable = kehillah_households
					order_by = var:kehillah_household_standing
					position = 3
					var:kehillah_household_head ?= { save_scope_as = kh_rev_4 }
				}
			}
		}
	}

	option = {
		name = kehillah_household.0010.a
	}
}
