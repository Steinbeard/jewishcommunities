# A decree against the Jews (docs/spec/2026-11-01-decree-situation.md), 2026-11-01. Console-only:
# .1 decrees the badge (the ghiyar under a Muslim lord), .2 the servants' decree, .3 logs the state.
namespace = kehillah_decree_debug

kehillah_decree_debug.1 = {
	type = character_event
	hidden = yes
	trigger = { government_has_flag = government_is_kehillah }
	immediate = {
		if = {
			limit = { domicile.domicile_location.county.holder.top_liege.faith.religion ?= religion:islam_religion }
			kehillah_decree_begin_effect = { KIND = ghiyar }
		}
		else = { kehillah_decree_begin_effect = { KIND = badge } }
		trigger_event = kehillah_decree_debug.3
	}
}
kehillah_decree_debug.2 = {
	type = character_event
	hidden = yes
	trigger = { government_has_flag = government_is_kehillah }
	immediate = {
		kehillah_decree_begin_effect = { KIND = servants }
		trigger_event = kehillah_decree_debug.3
	}
}
kehillah_decree_debug.3 = {
	type = character_event
	hidden = yes
	trigger = { government_has_flag = government_is_kehillah }
	immediate = {
		if = {
			limit = { exists = primary_title.var:kehillah_decree_situation }
			primary_title.var:kehillah_decree_situation = {
				debug_log = "kehillah_decree_debug.3: decree running"
				debug_log_scopes = yes
			}
		}
		else = { debug_log = "kehillah_decree_debug.3: no decree" }
	}
}
