# =====================================================================
# VISIT THE JEWISH COMMUNITY -- events. 2026-11-14 (Daniel).
# docs/spec/2026-11-14-adventurers-seats-and-invitations.md section 4.
#
#   .0001  The adventurer calls on the community in the county he stands in.
#          Fired by kehillah_visit_community_decision.
#          root = the adventurer, scope:kehillah_visit_title = the community.

namespace = kehillah_visit

# The small preface entered from the base game's Visit Local Settlement map
# action. The native main-square window has four option slots, so inserting a
# fifth choice there hides its normal Leave button. This preserves that menu
# intact while keeping the community call inside the same map action.
kehillah_visit.0002 = {
	type = character_event
	title = kehillah_visit.0002.t
	desc = kehillah_visit.0002.desc
	window = visit_settlement_window
	theme = realm
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_visit_leader
		animation = personality_honorable
	}

	immediate = {
		scope:kehillah_visit_title.holder = { save_scope_as = kehillah_visit_leader }
	}

	option = {
		name = kehillah_visit.0002.a
		trigger_event = kehillah_visit.0001
	}
	option = {
		name = kehillah_visit.0002.b
		trigger_event = ep3_laamp_decision_event.1000
	}
	option = { name = kehillah_visit.0002.c }
}

kehillah_visit.0001 = {
	type = character_event
	title = kehillah_visit.0001.t
	desc = {
		desc = kehillah_visit.0001.desc
		triggered_desc = {
			trigger = {
				scope:kehillah_visit_title = { has_variable = kehillah_visit_recruit_cooldown }
			}
			desc = kehillah_visit.0001.desc_spare
		}
	}
	theme = realm
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_visit_leader
		animation = personality_honorable
	}

	immediate = {
		scope:kehillah_visit_title.holder = { save_scope_as = kehillah_visit_leader }
	}

	# Pray with the congregation.
	option = {
		name = kehillah_visit.0001.a
		add_piety = minor_piety_gain
		set_variable = {
			name = kehillah_visit_community_cooldown
			value = yes
			months = 6
		}
		scope:kehillah_visit_leader = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = kehillah_visit_welcome_opinion_value
			}
		}
	}

	# Take a family into the company.
	option = {
		name = kehillah_visit.0001.b
		show_as_unavailable = { always = yes }
		trigger = {
			custom_description = {
				text = kehillah_visit_family_spare_tt
				scope:kehillah_visit_title = { NOT = { has_variable = kehillah_visit_recruit_cooldown } }
			}
			gold >= kehillah_visit_family_cost_value
		}
		add_gold = {
			value = kehillah_visit_family_cost_value
			multiply = -1
		}
		custom_tooltip = kehillah_visit.0001.b.tt
		set_variable = {
			name = kehillah_visit_community_cooldown
			value = yes
			months = 6
		}
		hidden_effect = {
			kehillah_household_create_head_effect = {
				LEADER = scope:kehillah_visit_leader
				LOCATION = root.location
			}
			kehillah_household_add_family_effect = {
				HEAD = scope:kh_new_head
				LEADER = root
			}
			scope:kehillah_visit_title = {
				set_variable = {
					name = kehillah_visit_recruit_cooldown
					value = yes
					years = kehillah_visit_recruit_cooldown_years_value
				}
			}
		}
	}

	# Speak with the leader about the seat.
	option = {
		name = kehillah_visit.0001.c
		trigger = {
			kehillah_cand_can_weigh_seat_trigger = {
				CAND = root
				SEAT = scope:kehillah_visit_title
			}
		}
		scope:kehillah_visit_title = { save_scope_as = cand_seat }
		set_variable = {
			name = kehillah_visit_community_cooldown
			value = yes
			months = 6
		}
		trigger_event = kehillah_candidacy.0001
	}

	# 2026-11-23: make this community home (docs/spec/2026-11-23-home-community.md).
	option = {
		name = kehillah_home_visit_join
		trigger = {
			kehillah_home_traveller_trigger = yes
			NOT = { kehillah_home_is_home_trigger = { TITLE = scope:kehillah_visit_title } }
			scope:kehillah_visit_title = { kehillah_home_title_valid_for_trigger = { MEMBER = root } }
		}
		kehillah_home_join_effect = { TITLE = scope:kehillah_visit_title }
	}

	# Take my leave.
	option = {
		name = kehillah_visit.0001.d
		set_variable = {
			name = kehillah_visit_community_cooldown
			value = yes
			months = 6
		}
	}

}
