# =====================================================================
# VISIT THE JEWISH COMMUNITY -- events. 2026-11-16 (Daniel).
# docs/spec/2026-11-14-adventurers-seats-and-invitations.md section 4.
#
#   .0001  A Jewish adventurer calls on the local community from the
#          Tavern row of the base-game Visit Local Settlement event.
#   .0002  The community leader makes introductions for a household
#          recruitment, with a chosen specialty.
# =====================================================================

namespace = kehillah_visit

kehillah_visit.0001 = {
	type = character_event
	title = kehillah_visit.0001.t
	desc = {
		desc = kehillah_visit.0001.desc
		triggered_desc = {
			trigger = { scope:kehillah_visit_title = { has_variable = kehillah_visit_recruit_cooldown } }
			desc = kehillah_visit.0001.desc_spare
		}
	}
	theme = realm
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_visit_leader
		animation = personality_honorable
	}

	immediate = {
		scope:kehillah_visit_title.holder = { save_scope_as = kehillah_visit_leader }
	}

	# A guest's welcome: fellowship, Piety, and the leader's regard.
	option = {
		name = kehillah_visit.0001.a
		add_piety = minor_piety_gain
		scope:kehillah_visit_leader = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = kehillah_visit_welcome_opinion_value
			}
		}
	}

	# The leader introduces a traveller to households who may choose the road.
	option = {
		name = kehillah_visit.0001.b
		trigger_event = kehillah_visit.0002
	}

	# Speak with the leader about the seat.
	option = {
		name = kehillah_visit.0001.c
		trigger = {
			kehillah_cand_can_weigh_seat_trigger = {
				CAND = root
				SEAT = scope:kehillah_visit_title
			}
		}
		scope:kehillah_visit_title = { save_scope_as = cand_seat }
		trigger_event = kehillah_candidacy.0001
	}

	# 2026-11-23: make this community home (docs/spec/2026-11-23-home-community.md).
	option = {
		name = kehillah_home_visit_join
		trigger = {
			kehillah_home_traveller_trigger = yes
			NOT = { kehillah_home_is_home_trigger = { TITLE = scope:kehillah_visit_title } }
			scope:kehillah_visit_title = { kehillah_home_title_valid_for_trigger = { MEMBER = root } }
		}
		kehillah_home_join_effect = { TITLE = scope:kehillah_visit_title }
	}

	option = {
		name = kehillah_visit.0001.d
	}

	# The settlement decision has already created its temporary visit state.
	# This route leaves before any page-specific variables exist, so only the
	# native menu's shared list, return marker, and visit lock need clearing.
	after = {
		clear_variable_list = list_of_options
		remove_variable ?= list_of_options
		remove_variable ?= return_from_location
		remove_variable ?= ep3_laamp_decision_1000_is_visiting
	}
}
