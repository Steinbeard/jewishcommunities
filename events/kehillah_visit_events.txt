namespace = kehillah_visit

kehillah_visit.0001 = {
	type = character_event
	title = kehillah_visit.0001.t
	desc = {
		desc = kehillah_visit.0001.desc
		triggered_desc = {
			trigger = {
				scope:kehillah_visit_title = { has_variable = kehillah_visit_recruit_cooldown }
			}
			desc = kehillah_visit.0001.desc_spare
		}
	}
	theme = realm
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_visit_leader
		animation = personality_honorable
	}

	immediate = {
		scope:kehillah_visit_title.holder = { save_scope_as = kehillah_visit_leader }
	}

	# Pray with the congregation.
	option = {
		name = kehillah_visit.0001.a
		add_piety = minor_piety_gain
		scope:kehillah_visit_leader = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = kehillah_visit_welcome_opinion_value
			}
		}
	}

	# Take a family into the company.
	option = {
		name = kehillah_visit.0001.b
		show_as_unavailable = { always = yes }
		trigger = {
			custom_description = {
				text = kehillah_visit_family_spare_tt
				scope:kehillah_visit_title = { NOT = { has_variable = kehillah_visit_recruit_cooldown } }
			}
			gold >= kehillah_visit_family_cost_value
		}
		add_gold = {
			value = kehillah_visit_family_cost_value
			multiply = -1
		}
		custom_tooltip = kehillah_visit.0001.b.tt
		hidden_effect = {
			kehillah_household_create_head_effect = {
				LEADER = scope:kehillah_visit_leader
				LOCATION = root.location
			}
			kehillah_household_add_family_effect = {
				HEAD = scope:kh_new_head
				LEADER = root
			}
			scope:kehillah_visit_title = {
				set_variable = {
					name = kehillah_visit_recruit_cooldown
					value = yes
					years = kehillah_visit_recruit_cooldown_years_value
				}
			}
		}
	}

	# Speak with the leader about the seat.
	option = {
		name = kehillah_visit.0001.c
		trigger = {
			kehillah_cand_can_weigh_seat_trigger = {
				CAND = root
				SEAT = scope:kehillah_visit_title
			}
		}
		scope:kehillah_visit_title = { save_scope_as = cand_seat }
		trigger_event = kehillah_candidacy.0001
	}

	# Take my leave.
	option = {
		name = kehillah_visit.0001.d
	}
}
