namespace = kehillah_home_debug
# =====================================================================
# The home community (2026-11-23), console only.
#   .1  Fire on a traveller: report home, stay count, tithe and reward.
#   .2  Fire on a traveller: run the seasonal step now.
#   .3  Fire on a traveller: run the yearly tithe now.
# =====================================================================

kehillah_home_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { kehillah_home_traveller_trigger = yes }
			debug_log = "kehillah_home_debug.1: eligible traveller"
		}
		else = {
			debug_log = "kehillah_home_debug.1: NOT an eligible traveller"
		}
		if = {
			limit = { exists = var:kehillah_home_community }
			var:kehillah_home_community = { save_scope_as = kehillah_debug_home }
			debug_log = "kehillah_home_debug.1: home is (scope kehillah_debug_home)"
		}
		else = {
			debug_log = "kehillah_home_debug.1: no home"
		}
		if = {
			limit = { exists = var:kehillah_home_stay_title }
			var:kehillah_home_stay_title = { save_scope_as = kehillah_debug_stay }
			save_scope_value_as = { name = kehillah_debug_seasons value = var:kehillah_home_stay_seasons }
			debug_log = "kehillah_home_debug.1: staying with (scope kehillah_debug_stay) for kehillah_debug_seasons seasons"
		}
		save_scope_value_as = { name = kehillah_debug_tithe value = kehillah_home_tithe_value }
		save_scope_value_as = { name = kehillah_debug_reward value = kehillah_home_tithe_reward_value }
		debug_log = "kehillah_home_debug.1: tithe kehillah_debug_tithe, reward kehillah_debug_reward"
		debug_log_scopes = yes
	}
}

kehillah_home_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		kehillah_home_seasonal_effect = yes
		debug_log = "kehillah_home_debug.2: seasonal step run"
	}
}

kehillah_home_debug.3 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		kehillah_home_yearly_effect = yes
		debug_log = "kehillah_home_debug.3: yearly step run"
	}
}
