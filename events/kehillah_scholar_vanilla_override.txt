# Jewish Communities override of the annual Religious Scholar event.
namespace = pam_generic_events

# Gain Religious Scholar
pam_generic_events.0001 = {
	id_override_priority = 1
	type = character_event
	title = pam_generic_events.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_is_jewish_rabbi_trigger = yes }
				desc = kehillah_scholar_study.desc
			}
			desc = pam_generic_events.0001.desc
		}
	}
	theme = christian

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:chaplain
		animation = reading
	}

	cooldown = { years = 20 }

	trigger = {
		is_available = yes
		age >= 10
		OR = {
			NOT = { has_trait = lifestyle_scholar }
			kehillah_is_jewish_rabbi_trigger = yes
		}
		trigger_if = {
			limit = { kehillah_is_jewish_rabbi_trigger = yes }
			exists = cp:councillor_court_chaplain
		}
		cp:councillor_court_chaplain ?= {
			opinion = {
				target = root
				value > 10
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			government_is_religious_trigger = yes
			add = 0.25
		}
	}

	immediate = {
		cp:councillor_court_chaplain = { save_scope_as = chaplain }
	}

	option = {
		name = {
			trigger = { kehillah_is_jewish_rabbi_trigger = yes }
			text = kehillah_scholar_study.a
		}
		name = pam_generic_events.0001.a
		if = {
			limit = { kehillah_is_jewish_rabbi_trigger = yes }
			add_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = 10 }
		}
		else = { add_trait = lifestyle_scholar }

		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_zeal = 0.5
			}
			modifier = {
				government_is_religious_trigger = yes
				factor = 2
			}
		}
	}

	option = { # I'm good ty
		name = pam_generic_events.0001.b

		ai_chance = {
			base = 100
			modifier = {
				has_trait = cynical
				factor = 0
			}
		}
	}
}
