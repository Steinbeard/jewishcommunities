# =====================================================================
# V38 slice 2 -- missionizing: the disputation and compelled sermons
# (docs/spec/v38-anti-jewish-pressure.md section 5), 2026-10-13. Daniel:
# "more events about characters trying to missionize the community".
# Paris 1240 (Nicholas Donin), Barcelona 1263 (Pablo Christiani and
# Nachmanides), Tortosa 1413; the friars' sermons in the synagogues.
# Reworked 2026-10-18 (v38 addendum): the odds are learning duels shown
# in the tooltip, pillar changes go through kehillah_change_pillar_effect,
# and before kehillah_friars_year (1216) the missionaries are monks and
# priests, not friars.
# =====================================================================
#   .0001 The Disputation   .0002 its outcome report (players)
#   .0010 Compelled Sermons
# =====================================================================
namespace = kehillah_mission

kehillah_mission.0001 = {
	type = character_event
	title = kehillah_mission.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { current_year >= kehillah_friars_year }
				desc = kehillah_mission.0001.desc_friar
			}
			desc = kehillah_mission.0001.desc_monk
		}
		desc = kehillah_mission.0001.desc_champion
	}
	theme = learning
	left_portrait = {
		character = scope:kehillah_friar
		animation = scheme
	}
	right_portrait = {
		character = scope:kehillah_disputant
		animation = worry
	}
	lower_center_portrait = scope:kehillah_mission_host

	trigger = { exists = domicile.domicile_location.county.holder.top_liege }

	immediate = {
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_mission_host }
		domicile.domicile_location = { save_scope_as = kehillah_mission_town }
		# The community's champion: its most learned adult Jew.
		ordered_courtier = {
			limit = {
				is_adult = yes
				is_jewish_character_trigger = yes
			}
			order_by = learning
			save_scope_as = kehillah_disputant
		}
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:kehillah_disputant }
					learning > scope:kehillah_disputant.learning
				}
			}
			save_scope_as = kehillah_disputant
		}
		# The challenger: a convert, now a friar (or, earlier, a monk).
		create_character = {
			faith = scope:kehillah_mission_host.faith
			culture = domicile.domicile_location.county.culture
			location = domicile.domicile_location
			age = { 30 50 }
			gender_female_chance = 0
			trait = zealous
			learning = { 10 18 }
			save_scope_as = kehillah_friar
		}
		set_variable = { name = kehillah_mission_host_ref value = scope:kehillah_mission_host }
	}

	# Dispute boldly: the champion's learning against his, and the lord
	# judges. Much to win, more to lose.
	option = {
		name = kehillah_mission.0001.a
		scope:kehillah_disputant = {
			duel = {
				skill = learning
				target = scope:kehillah_friar
				50 = {
					desc = kehillah_mission_disputation_won
					compare_modifier = {
						value = scope:duel_value
						multiplier = 3.5
					}
					kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_mission_host }
					min = 5
					root = {
						kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_disputation_won_greatness }
						custom_tooltip = kehillah_mission_bold_won_tt
						hidden_effect = {
							scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = 5 } }
							scope:kehillah_mission_host.faith = { kehillah_faith_pressure_add_effect = { AMOUNT = -1 } }
							set_variable = { name = kehillah_mission_outcome value = flag:won }
						}
					}
				}
				50 = {
					desc = kehillah_mission_disputation_lost
					compare_modifier = {
						value = scope:duel_value
						multiplier = -3.5
					}
					min = 5
					root = {
						kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_disputation_bold_lost_greatness }
						kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_disputation_bold_lost_stability }
						custom_tooltip = kehillah_mission_bold_lost_tt
						hidden_effect = {
							scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = -5 } }
							scope:kehillah_mission_host.faith = { kehillah_faith_pressure_add_effect = { AMOUNT = 2 } }
							kehillah_charter_censor_text_effect = yes
							set_variable = { name = kehillah_mission_outcome value = flag:lost }
						}
					}
				}
			}
		}
		hidden_effect = { trigger_event = { id = kehillah_mission.0002 days = 2 } }
		ai_chance = {
			base = 30
			modifier = {
				add = 30
				scope:kehillah_disputant.learning > scope:kehillah_friar.learning
			}
		}
	}

	# Dispute cautiously: better odds of conceding nothing, little glory.
	option = {
		name = kehillah_mission.0001.b
		scope:kehillah_disputant = {
			duel = {
				skill = learning
				target = scope:kehillah_friar
				65 = {
					desc = kehillah_mission_disputation_held
					compare_modifier = {
						value = scope:duel_value
						multiplier = 3
					}
					min = 5
					root = {
						kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_disputation_held_greatness }
						hidden_effect = { set_variable = { name = kehillah_mission_outcome value = flag:held } }
					}
				}
				35 = {
					desc = kehillah_mission_disputation_lost
					compare_modifier = {
						value = scope:duel_value
						multiplier = -3
					}
					min = 5
					root = {
						kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_disputation_cautious_lost_stability }
						custom_tooltip = kehillah_mission_cautious_lost_tt
						hidden_effect = {
							scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = -2 } }
							set_variable = { name = kehillah_mission_outcome value = flag:lost }
						}
					}
				}
			}
		}
		hidden_effect = { trigger_event = { id = kehillah_mission.0002 days = 2 } }
		ai_chance = { base = 50 }
	}

	# Pay to be excused: certain, and a little shameful.
	option = {
		name = kehillah_mission.0001.c
		trigger = { gold >= medium_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = medium_gold_value
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_disputation_excused_greatness }
		custom_tooltip = kehillah_mission_excused_tt
		hidden_effect = { scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = -2 } } }
		ai_chance = { base = 20 }
	}

	after = {
		hidden_effect = {
			scope:kehillah_friar = { death = { death_reason = death_vanished } }
		}
	}
}

# The disputation's outcome (players).
kehillah_mission.0002 = {
	type = character_event
	title = kehillah_mission.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_mission_outcome ?= flag:won }
				desc = kehillah_mission.0002.won
			}
			triggered_desc = {
				trigger = { var:kehillah_mission_outcome ?= flag:held }
				desc = kehillah_mission.0002.held
			}
			desc = kehillah_mission.0002.lost
		}
	}
	theme = learning
	trigger = { is_ai = no }
	left_portrait = {
		character = root
		animation = personality_rational
	}
	option = {
		name = kehillah_mission.0002.a
		hidden_effect = {
			if = {
				limit = { has_variable = kehillah_mission_outcome }
				remove_variable = kehillah_mission_outcome
			}
		}
	}
}

kehillah_mission.0010 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { current_year >= kehillah_friars_year }
				desc = kehillah_mission.0010.t
			}
			desc = kehillah_mission.0010.t_priests
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { current_year >= kehillah_friars_year }
				desc = kehillah_mission.0010.desc
			}
			desc = kehillah_mission.0010.desc_priests
		}
	}
	theme = faith
	left_portrait = {
		character = scope:kehillah_mission_host
		animation = idle
	}
	right_portrait = {
		character = root
		animation = worry
	}

	trigger = { exists = domicile.domicile_location.county.holder.top_liege }

	immediate = {
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_mission_host }
		domicile.domicile_location = { save_scope_as = kehillah_mission_town }
	}

	# Sit in silence: no offence given, but the sermons wear on the
	# congregation, and someone may listen too well.
	option = {
		name = kehillah_mission.0010.a
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_sermon_silence_stability }
		random_list = {
			75 = {
				desc = kehillah_mission_silence_held
			}
			25 = {
				desc = kehillah_mission_silence_convert
				custom_tooltip = kehillah_mission_convert_tt
				hidden_effect = { kehillah_mission_conversion_effect = yes }
			}
		}
		ai_chance = { base = 50 }
	}

	# Answer them from the Torah: your learning against trained preachers.
	option = {
		name = kehillah_mission.0010.b
		duel = {
			skill = learning
			value = decent_skill_rating
			50 = {
				desc = kehillah_mission_sermon_answered
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				min = 5
				kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sermon_answered_greatness }
			}
			50 = {
				desc = kehillah_mission_sermon_offended
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				min = 5
				kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_sermon_offended_stability }
				custom_tooltip = kehillah_mission_offended_tt
				hidden_effect = {
					scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = -5 } }
					scope:kehillah_mission_host.faith = { kehillah_faith_pressure_add_effect = { AMOUNT = 1 } }
				}
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				add = 20
				learning >= 15
			}
		}
	}

	# Pay them to preach elsewhere.
	option = {
		name = kehillah_mission.0010.c
		trigger = { gold >= minor_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = minor_gold_value
		custom_tooltip = kehillah_mission_bribed_tt
		hidden_effect = { scope:kehillah_mission_host = { kehillah_shift_outlook_effect = { AMOUNT = -1 } } }
		ai_chance = { base = 30 }
	}
}
