# V16 startup cache hand-off. Running the global registry scan from the
# on_game_start bootstrap can stall CK3 while title and domicile scopes are
# still being constructed. The Worms leader is a stable 1066 event target;
# one in-game day later all registered communities are complete and the
# cache can be written safely. Later foundations refresh immediately from
# their own effect, when the registry is already live.
kehillah_settlement_conditions.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		kehillah_refresh_all_settlement_networks_effect = yes
		debug_log = "kehillah_settlement_conditions.0001: refreshed startup nearby-community cache"
	}
}

# Fired by the fresh Christian/Muslim charter path one day after creation.
kehillah_settlement_conditions.0002 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				is_tributary = yes
				OR = {
					has_subject_contract_group = kehillah_christian_host_charter_prototype
					has_subject_contract_group = kehillah_muslim_host_charter_prototype
				}
			}
			kehillah_apply_fresh_charter_policy_effect = yes
			# The obligation write commits after this event. Read the live
			# tributary flags only on the next day, never during bootstrap or
			# the writer itself.
			trigger_event = { id = kehillah_settlement_conditions.0003 days = 1 }
			debug_log = "kehillah_settlement_conditions.0002: applied deferred fresh-charter policy"
		}
	}
}

# V16 cache hand-off. This executes one day after .0002's concrete
# obligation writes, when the tributary contract has committed. Its effect
# is intentionally the sole live subject-contract reader; every gameplay
# and UI consumer reads the title cache it writes. Quarterly reconciliation
# handles later voluntary contract edits (no generic CK3 on_action exists
# for subject-contract changes).
kehillah_settlement_conditions.0003 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				is_tributary = yes
				OR = {
					has_subject_contract_group = kehillah_christian_host_charter_prototype
					has_subject_contract_group = kehillah_muslim_host_charter_prototype
				}
			}
			kehillah_refresh_charter_state_cache_effect = yes
			kehillah_queue_charter_notification_effect = yes
			debug_log = "kehillah_settlement_conditions.0003: refreshed committed regional-charter cache"
		}
	}
}

# V20 historical-start snapshot, reconciled. 2026-10-18: the opening pillars
# are now written at game start from a predicted charter and network
# (kehillah_on_game_start), so nobody waits three days for them. This day-three
# pass, when the nearby-community cache is written and fresh regional charter
# obligations are committed and cached, rewrites them once from the real
# baseline (buildings, officers, skills, urban opportunity, network, and actual
# charter terms), correcting whatever the guess got wrong.
kehillah_settlement_conditions.0008 = {
	type = character_event
	hidden = yes

	immediate = {
		every_in_global_list = {
			variable = kehillah_registered_communities
			kehillah_reconcile_provisional_start_pillars_effect = yes
		}
		kehillah_refresh_map_view_for_players_effect = yes
		debug_log = "kehillah_settlement_conditions.0008: reconciled provisional historical pillars against committed baselines"
	}
}

# Delivered only after .0003 has observed the committed contract. This is a
# report, not a negotiation: the concrete Host Charter is now the source of
# the community's pillar terms, while the host policy remains its shared
# frontier stance.
kehillah_settlement_conditions.0004 = {
	# 2026-10-29: a letter from the host ruler rather than a two-portrait
	# scene -- a charter is a document handed down. The ruler's outlook band
	# (kehillah_outlook_band_value, frozen below as kehillah_charter_tone)
	# sets the letter's voice; the terms block and footer are unchanged
	# facts. The voice is flavour only: the terms come from realm policy, so
	# a hostile ruler can be bound by a generous law and a protective one by
	# a harsh law. Situations: fresh charter, renewal for a new community
	# leader (2026-10-15, 20-year soak), new host ruler.
	type = letter_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_charter_renewal_trigger = yes }
				desc = kehillah_settlement_conditions.0004.t_renewed
			}
			desc = kehillah_settlement_conditions.0004.t
		}
	}
	immediate = {
		save_scope_value_as = { name = kehillah_charter_tone value = 0 }
		# Tributary charters do not populate root.GetLiege. Resolve the ruler
		# of the community's physical host realm, as the notification queue does.
		domicile.domicile_location.county.holder.top_liege = {
			save_scope_as = kehillah_charter_host
			save_scope_value_as = {
				name = kehillah_charter_tone
				value = kehillah_outlook_band_value
			}
		}
	}
	sender = scope:kehillah_charter_host
	opening = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone >= 2 }
				desc = kehillah_settlement_conditions.0004.opening.muslim.protective
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = 1 }
				desc = kehillah_settlement_conditions.0004.opening.muslim.well_disposed
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = 0 }
				desc = kehillah_settlement_conditions.0004.opening.muslim.indifferent
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = -1 }
				desc = kehillah_settlement_conditions.0004.opening.muslim.cold
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0004.opening.muslim.hostile
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone >= 2 }
				desc = kehillah_settlement_conditions.0004.opening.christian.protective
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 1 }
				desc = kehillah_settlement_conditions.0004.opening.christian.well_disposed
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 0 }
				desc = kehillah_settlement_conditions.0004.opening.christian.indifferent
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = -1 }
				desc = kehillah_settlement_conditions.0004.opening.christian.cold
			}
			desc = kehillah_settlement_conditions.0004.opening.christian.hostile
		}
	}
	desc = {
		# Why the letter was written.
		first_valid = {
			triggered_desc = {
				trigger = {
					primary_title = { has_variable = kehillah_charter_suzerain_changed }
					kehillah_cached_muslim_charter_trigger = yes
				}
				desc = kehillah_settlement_conditions.0004.why.new_ruler.muslim
			}
			triggered_desc = {
				trigger = { primary_title = { has_variable = kehillah_charter_suzerain_changed } }
				desc = kehillah_settlement_conditions.0004.why.new_ruler.christian
			}
			triggered_desc = {
				trigger = {
					kehillah_charter_renewal_trigger = yes
					kehillah_cached_muslim_charter_trigger = yes
				}
				desc = kehillah_settlement_conditions.0004.why.renewed.muslim
			}
			triggered_desc = {
				trigger = { kehillah_charter_renewal_trigger = yes }
				desc = kehillah_settlement_conditions.0004.why.renewed.christian
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0004.why.fresh.muslim
			}
			desc = kehillah_settlement_conditions.0004.why.fresh.christian
		}
		# The ruler's voice.
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone >= 2 }
				desc = kehillah_settlement_conditions.0004.voice.muslim.protective
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = 1 }
				desc = kehillah_settlement_conditions.0004.voice.muslim.well_disposed
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = 0 }
				desc = kehillah_settlement_conditions.0004.voice.muslim.indifferent
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes scope:kehillah_charter_tone = -1 }
				desc = kehillah_settlement_conditions.0004.voice.muslim.cold
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0004.voice.muslim.hostile
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone >= 2 }
				desc = kehillah_settlement_conditions.0004.voice.christian.protective
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 1 }
				desc = kehillah_settlement_conditions.0004.voice.christian.well_disposed
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 0 }
				desc = kehillah_settlement_conditions.0004.voice.christian.indifferent
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = -1 }
				desc = kehillah_settlement_conditions.0004.voice.christian.cold
			}
			desc = kehillah_settlement_conditions.0004.voice.christian.hostile
		}
		# The clauses themselves.
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0004.terms.muslim
			}
			desc = kehillah_settlement_conditions.0004.terms.christian
		}
		# The sanction, in the same voice.
		first_valid = {
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone >= 2 }
				desc = kehillah_settlement_conditions.0004.close.protective
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 1 }
				desc = kehillah_settlement_conditions.0004.close.well_disposed
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = 0 }
				desc = kehillah_settlement_conditions.0004.close.indifferent
			}
			triggered_desc = {
				trigger = { scope:kehillah_charter_tone = -1 }
				desc = kehillah_settlement_conditions.0004.close.cold
			}
			desc = kehillah_settlement_conditions.0004.close.hostile
		}
		# Outside the letter: what the tone came from, and what to do next.
		first_valid = {
			triggered_desc = {
				trigger = { primary_title = { has_variable = kehillah_charter_suzerain_changed } }
				desc = kehillah_settlement_conditions.0004.footer.new_ruler
			}
			desc = kehillah_settlement_conditions.0004.footer
		}
	}

	option = {
		name = {
			trigger = { scope:kehillah_charter_tone >= 2 }
			text = kehillah_settlement_conditions.0004.a.protective
		}
		name = {
			trigger = { scope:kehillah_charter_tone = 1 }
			text = kehillah_settlement_conditions.0004.a.well_disposed
		}
		name = {
			trigger = { scope:kehillah_charter_tone = 0 }
			text = kehillah_settlement_conditions.0004.a.indifferent
		}
		name = {
			trigger = { scope:kehillah_charter_tone = -1 }
			text = kehillah_settlement_conditions.0004.a.cold
		}
		name = {
			trigger = { scope:kehillah_charter_tone <= -2 }
			text = kehillah_settlement_conditions.0004.a.hostile
		}
		primary_title = { remove_variable = kehillah_charter_suzerain_changed }
		# Who last received this community's charter (kehillah_charter_
		# renewal_trigger). A different leader next time is a renewal.
		primary_title = { set_variable = { name = kehillah_charter_received_by value = root } }
		ai_chance = { base = 100 }
	}
}

# A ruler receives this once per host per resolution day, regardless of how
# many communities resolved their fresh contracts. The counter is deliberately
# retained until acknowledgement so it is safe for same-day events to coalesce.
kehillah_settlement_conditions.0005 = {
	type = character_event
	title = kehillah_settlement_conditions.0005.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_charter_notice_count = 1 }
				desc = kehillah_settlement_conditions.0005.single.desc
			}
			desc = kehillah_settlement_conditions.0005.multiple.desc
		}
	}
	theme = realm

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = kehillah_settlement_conditions.0005.a
		remove_character_flag = kehillah_charter_notice_pending
		remove_variable = kehillah_charter_notice_count
		ai_chance = { base = 100 }
	}
}

# Community-side counterpart to .0006. The outgoing agreement's actual
# cached terms are restated, along with a deliberately non-mechanical first
# reading of the successor's personal posture toward Jewish subjects.
kehillah_settlement_conditions.0007 = {
	type = character_event
	title = kehillah_settlement_conditions.0007.t
	immediate = {
		# As in .0004: tributary charters do not populate root.GetLiege, so
		# the new steward's posture is read from the physical host realm.
		domicile.domicile_location.county.holder.top_liege ?= {
			save_scope_as = kehillah_charter_host
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_cached_christian_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0007.christian.desc
			}
			triggered_desc = {
				trigger = { kehillah_cached_muslim_charter_trigger = yes }
				desc = kehillah_settlement_conditions.0007.muslim.desc
			}
		}
	}
	theme = realm

	left_portrait = {
		character = root
		animation = worry
	}

	option = {
		name = kehillah_settlement_conditions.0007.a
		ai_chance = { base = 100 }
	}
}
