# =====================================================================
# A decree against the Jews as a situation (docs/spec/2026-11-01-decree-
# situation.md), 2026-11-01: the badge (Lateran IV, 1215; the ghiyar of
# the Pact of Umar) and the servants' decree (Gregory VII, 1078-1081).
#   .0001 The decree is proclaimed (the situation's first scene)
#   .0010 Answer the Decree (kehillah_decree_answer_decision)
#   .0060 The verdict (players)
# Scope: the community's leader. Each answer counts once per decree
# (kehillah_decree_not_done_trigger); its race points show in the native
# catalyst tooltip.
# =====================================================================

# Scopes for every decree text: the lord's people, and the rolled instigator and spouse.
scripted_effect kehillah_decree_scene_scopes_effect = {
	kehillah_incident_scopes_effect = yes
 primary_title = { save_scope_as = kehillah_decree_scene_data }
 primary_title.var:kehillah_decree_situation ?= { save_scope_as = kehillah_decree_scene_data }
	scope:kehillah_decree_scene_data.var:kehillah_decree_instigator_char ?= { save_scope_as = kehillah_decree_instigator }
	scope:kehillah_decree_scene_data.var:kehillah_decree_spouse ?= { save_scope_as = kehillah_decree_spouse }
}

# Buy an exemption for your own household (the mark only): if it is enforced, you do not wear it.

	kehillah_decree_mark_done_effect = { ONCE = exempt }
	pay_short_term_gold = {
		target = scope:kehillah_incident_host
		gold = kehillah_incident_small_bribe_cost_value

	primary_title = { set_variable = { name = kehillah_decree_household_exempt value = yes } }
}

# Petition the lord: the Shtadlan, or the leader without one.
scripted_effect kehillah_decree_petition_effect = {
	kehillah_decree_mark_done_effect = { ONCE = petition }
	if = {
		limit = { scope:kehillah_incident_host ?= { is_ai = no } }
		kehillah_decree_ask_lord_effect = { ASK = petition }
	}
	else = {
		random_list = {
			40 = {
				desc = kehillah_decree_out_persuaded
				kehillah_incident_petition_odds_modifier = yes
				modifier = {
					add = -10
					custom_tooltip = {
						text = kehillah_decree_factor_cleric
						kehillah_decree_detail_trigger = { VAR = instigator VALUE = cleric }
					}
				}
				modifier = {
					add = -15
					custom_tooltip = {
						text = kehillah_decree_factor_zeal
						kehillah_decree_detail_trigger = { VAR = instigator VALUE = zeal }
					}
				}
				min = 5
				primary_title = { kehillah_decree_add_effect = { CATALYST = kehillah_decree_petition_catalyst } }
			}
			60 = {
				desc = kehillah_decree_out_refused
				primary_title = { kehillah_decree_add_effect = { CATALYST = kehillah_decree_petition_failed_catalyst } }
			}
		}
	}
}

# Threaten to leave the realm (Castile, 1219): the lord weighs what he would lose.
scripted_effect kehillah_decree_leave_effect = {
	kehillah_decree_mark_done_effect = { ONCE = leave }
	if = {
		limit = { scope:kehillah_incident_host ?= { is_ai = no } }
		kehillah_decree_ask_lord_effect = { ASK = leave }
	}
	else = {
		random_list = {
			40 = {
				desc = kehillah_decree_out_believed
				modifier = {
					add = 15
					scope:kehillah_incident_host = { has_trait = greedy }
				}
				modifier = {
					add = 10
					custom_tooltip = {
						text = kehillah_decree_factor_prosperous
						primary_title ?= {
							has_variable = kehillah_var_prosperity
							var:kehillah_var_prosperity >= 700
						}
					}
				}
				compare_modifier = {
					value = stewardship
					multiplier = 1.5
					offset = inverted_medium_skill_rating
				}
				min = 5
				primary_title = { kehillah_decree_add_effect = { CATALYST = kehillah_decree_leave_catalyst } }
			}
			60 = {
				desc = kehillah_decree_out_bluff_called
				primary_title = { kehillah_decree_add_effect = { CATALYST = kehillah_decree_leave_failed_catalyst } }
				scope:kehillah_incident_host ?= {
					kehillah_shift_outlook_effect = { AMOUNT = kehillah_decree_leave_outlook }
				}
			}
		}
	}
}

	}
