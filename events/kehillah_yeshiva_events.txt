# ---------------------------------------------------------------------
# .0010 -- a scholar arrives. Root = the leader. Ordained rabbis come
# looking for a place at a new yeshiva: candidates for the Dayyan seat,
# judges for the Bet Din, and a supply of rabbis for the community.
# ---------------------------------------------------------------------
kehillah_yeshiva.0010 = {
	type = character_event
	title = kehillah_yeshiva.0010.t
	desc = kehillah_yeshiva.0010.desc
	theme = learning
	left_portrait = {
		character = scope:kehillah_yeshiva_scholar
		animation = personality_rational
	}

	trigger = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		kehillah_has_yeshiva_trigger = yes
	}

	immediate = {
		kehillah_generate_rabbi_candidate_effect = {
			SLOT = kehillah_yeshiva_scholar
			AGE_MIN = 28
			AGE_MAX = 55
			EDUCATION = education_learning_3
			LEARNING_MIN = 10
			LEARNING_MAX = 15
		}
		scope:kehillah_yeshiva_scholar = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 20 }
		}
	}

	option = {
		name = kehillah_yeshiva.0010.a
		add_courtier = scope:kehillah_yeshiva_scholar
	}
	option = {
		name = kehillah_yeshiva.0010.b
		# Turned away -- he goes to the pool, where other communities' Seek
		# a Rabbi can still find him.
		ai_chance = { base = 0 }
	}
}

# ---------------------------------------------------------------------
# .0101 -- a Jewish traveller at a yeshiva town (POI on_visit).
# Root = the traveller. scope:kehillah_poi_title = the community.
# ---------------------------------------------------------------------
kehillah_yeshiva.0101 = {
	type = character_event
	title = kehillah_yeshiva.0101.t
	desc = kehillah_yeshiva.0101.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
		trigger = { exists = scope:kehillah_rosh_yeshiva }
	}

	immediate = {
		scope:kehillah_poi_title.holder ?= { kehillah_save_rosh_yeshiva_effect = yes }
	}

	# Sit in on a lecture.
	option = {
		name = kehillah_yeshiva.0101.a
		add_learning_lifestyle_xp = medium_lifestyle_xp
		add_piety = minor_piety_gain
	}
	# Speak with the Rosh Yeshiva himself.
	option = {
		name = kehillah_yeshiva.0101.b
		trigger = {
			exists = scope:kehillah_rosh_yeshiva
			NOT = { scope:kehillah_rosh_yeshiva = root }
		}
		add_opinion = {
			target = scope:kehillah_rosh_yeshiva
			modifier = respect_opinion
			opinion = 10
		}
		scope:kehillah_rosh_yeshiva = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 10
			}
		}
		add_piety = minor_piety_gain
	}
}

# ---------------------------------------------------------------------
# .0102 -- a non-Jewish traveller at a yeshiva town (POI on_visit).
# Root = the traveller. Learning about the tenets of Judaism through a
# curious scholar's eyes; respectful, never a conversion pitch.
# ---------------------------------------------------------------------
kehillah_yeshiva.0102 = {
	type = character_event
	title = kehillah_yeshiva.0102.t
	desc = kehillah_yeshiva.0102.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
		trigger = { exists = scope:kehillah_rosh_yeshiva }
	}

	immediate = {
		scope:kehillah_poi_title.holder ?= { kehillah_save_rosh_yeshiva_effect = yes }
	}

	# Ask about the Hebrew text of scripture -- the Christian Hebraist's
	# question.
	option = {
		name = kehillah_yeshiva.0102.a
		add_learning_lifestyle_xp = medium_lifestyle_xp
		if = {
			limit = { exists = scope:kehillah_rosh_yeshiva }
			add_opinion = {
				target = scope:kehillah_rosh_yeshiva
				modifier = respect_opinion
				opinion = 5
			}
		}
	}
	# Debate a student.
	option = {
		name = kehillah_yeshiva.0102.b
		duel = {
			skill = learning
			value = 10
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				desc = kehillah_yeshiva.0102.b.win
				add_prestige = minor_prestige_gain
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				desc = kehillah_yeshiva.0102.b.lose
				add_learning_lifestyle_xp = minor_lifestyle_xp
			}
		}
	}
	# Pass on.
	option = {
		name = kehillah_yeshiva.0102.c
	}
}
