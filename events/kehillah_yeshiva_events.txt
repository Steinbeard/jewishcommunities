# looking for a place at a new yeshiva: candidates for the Dayyan seat,
# judges for the Bet Din, and a supply of rabbis for the community.

	type = character_event
	title = kehillah_yeshiva.0010.t
	desc = kehillah_yeshiva.0010.desc
	theme = learning

		character = scope:kehillah_yeshiva_scholar

	}

	trigger = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		kehillah_has_yeshiva_trigger = yes
	}

	immediate = {
		kehillah_generate_rabbi_candidate_effect = {
			SLOT = kehillah_yeshiva_scholar
			AGE_MIN = 28
			AGE_MAX = 55
			EDUCATION = education_learning_3
			LEARNING_MIN = 10
			LEARNING_MAX = 15
		}
		scope:kehillah_yeshiva_scholar = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 20 }
		}
	}

	option = {
		name = kehillah_yeshiva.0010.a
		add_courtier = scope:kehillah_yeshiva_scholar
	}
	option = {
		name = kehillah_yeshiva.0010.b
		# Turned away -- he goes to the pool, where other communities' Seek
		# a Rabbi can still find him.
		ai_chance = { base = 0 }
	}
}

			modifier = respect_opinion
			opinion = 10

				target = root
				modifier = respect_opinion
				opinion = 10
			}
		}
		add_piety = minor_piety_gain
	}
}

# ---------------------------------------------------------------------
# .0102 -- a non-Jewish traveller at a yeshiva town (POI on_visit).
# Root = the traveller. Learning about the tenets of Judaism through a
# curious scholar's eyes; respectful, never a conversion pitch.
# ---------------------------------------------------------------------
kehillah_yeshiva.0102 = {
	type = character_event
	title = kehillah_yeshiva.0102.t
	desc = kehillah_yeshiva.0102.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
		trigger = { exists = scope:kehillah_rosh_yeshiva }
	}

	immediate = {
		scope:kehillah_poi_title.holder ?= { kehillah_save_rosh_yeshiva_effect = yes }
	}

	# Ask about the Hebrew text of scripture -- the Christian Hebraist's
	# question.
	option = {
		name = kehillah_yeshiva.0102.a
		add_learning_lifestyle_xp = medium_lifestyle_xp
		if = {
			limit = { exists = scope:kehillah_rosh_yeshiva }
			add_opinion = {
				target = scope:kehillah_rosh_yeshiva
				modifier = respect_opinion
				opinion = 5
			}
		}
	}
	# Debate a student.
	option = {
		name = kehillah_yeshiva.0102.b
		duel = {
			skill = learning
			value = 10
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				desc = kehillah_yeshiva.0102.b.win
				add_prestige = minor_prestige_gain
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				desc = kehillah_yeshiva.0102.b.lose
				add_learning_lifestyle_xp = minor_lifestyle_xp
			}
		}
	}
	# Pass on.
	option = {
		name = kehillah_yeshiva.0102.c
	}
}

# ---------------------------------------------------------------------
# .0210 -- hidden. Root = the leader of the community an alumnus came
# home to; his title's kehillah_alumnus_incoming names him. Seats him as
# Dayyan if the seat is still free (see kehillah_place_alumnus_effect).
# ---------------------------------------------------------------------
kehillah_yeshiva.0210 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { primary_title ?= { has_variable = kehillah_alumnus_incoming } }
			primary_title.var:kehillah_alumnus_incoming = { save_scope_as = kehillah_arriving_alumnus }
			if = {
				limit = {
					NOT = { exists = cp:councillor_court_chaplain }
					scope:kehillah_arriving_alumnus = {
						is_alive = yes
						liege ?= root
					}
				}
				assign_councillor_type = {
					type = councillor_court_chaplain
					target = scope:kehillah_arriving_alumnus
				}
			}
			if = {
				limit = { cp:councillor_court_chaplain ?= scope:kehillah_arriving_alumnus }
				debug_log = "kehillah_yeshiva.0210: an alumnus took a Dayyan seat"
			}
			else = {
				debug_log = "kehillah_yeshiva.0210: the alumnus could NOT be seated (he stays as a rabbi at court)"
			}
			primary_title = { remove_variable = kehillah_alumnus_incoming }
		}
	}
}

# =====================================================================
# .0320 -- THE GRADUATING CLASS (V29 section 11.1, 2026-10-12). Once a
# year, after the yearly ordination (kehillah_yeshiva_graduation_effect),
# a player leader decides where the class's most promising graduate goes.
# Each path feeds a different pillar.
# =====================================================================
kehillah_yeshiva.0320 = {
	type = character_event
	title = kehillah_yeshiva.0320.t
	desc = kehillah_yeshiva.0320.desc
	theme = learning
	left_portrait = {
		character = scope:kehillah_valedictorian
		animation = happiness
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
		trigger = {
			exists = scope:kehillah_rosh_yeshiva
			NOT = { scope:kehillah_rosh_yeshiva = root }
		}
	}
	trigger = {
		exists = scope:kehillah_valedictorian
		scope:kehillah_valedictorian = { is_alive = yes }
	}
	immediate = {
		scope:kehillah_valedictorian = { save_scope_as = kehillah_graduate }
		domicile.domicile_location.county.holder ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = kehillah_local_lord
			}
		}
	}

	# Keep him: a rabbi for the community's own courts.
	option = {
		name = kehillah_yeshiva.0320.a
		scope:kehillah_valedictorian = {
			if = {
				limit = { has_character_flag = kehillah_visiting_student }
				remove_character_flag = kehillah_visiting_student
			}
		}
		primary_title = { change_variable = { name = kehillah_var_stability add = kehillah_yeshiva_keep_graduate_stability } }
		custom_tooltip = kehillah_yeshiva.0320.a.tt
		ai_chance = { base = 50 }
	}

	# Send him where a community has no rabbi.
	option = {
		name = kehillah_yeshiva.0320.b
		trigger = { kehillah_alumnus_destination_available_trigger = yes }
		kehillah_place_alumnus_effect = yes
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_yeshiva_send_graduate_fame }
		scope:kehillah_alumnus_destination.holder ?= {
			add_opinion = { target = root modifier = respect_opinion opinion = kehillah_yeshiva_send_graduate_opinion }
		}
		custom_tooltip = kehillah_yeshiva.0320.b.tt
		ai_chance = { base = 50 }
	}

	# Recommend him to the lord of the town.
	option = {
		name = kehillah_yeshiva.0320.c
		trigger = {
			exists = scope:kehillah_local_lord
			# One's own kin is not handed to a Christian lord's court.
			NOT = { scope:kehillah_valedictorian = { is_close_or_extended_family_of = root } }
		}
		scope:kehillah_local_lord = {
			add_courtier = scope:kehillah_valedictorian
			add_opinion = { target = root modifier = pleased_opinion opinion = kehillah_yeshiva_lord_graduate_opinion }
			# V34: a learned Jew in his service softens his outlook.
			custom_tooltip = kehillah_yeshiva_lord_outlook_tt
			hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = 10 } }
		}
		scope:kehillah_valedictorian = {
			if = {
				limit = { has_character_flag = kehillah_visiting_student }
				remove_character_flag = kehillah_visiting_student
			}
		}
		primary_title = { change_variable = { name = kehillah_var_prosperity add = kehillah_yeshiva_lord_graduate_prosperity } }
		custom_tooltip = kehillah_yeshiva.0320.c.tt
		ai_chance = { base = 0 }
	}
}
