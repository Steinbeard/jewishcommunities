# ---------------------------------------------------------------------
# .0102 -- a non-Jewish traveller at a yeshiva town (POI on_visit).
# Root = the traveller. Learning about the tenets of Judaism through a
# curious scholar's eyes; respectful, never a conversion pitch.
# ---------------------------------------------------------------------
kehillah_yeshiva.0102 = {
	type = character_event
	title = kehillah_yeshiva.0102.t
	desc = kehillah_yeshiva.0102.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
		trigger = { exists = scope:kehillah_rosh_yeshiva }
	}

	immediate = {
		scope:kehillah_poi_title.holder ?= { kehillah_save_rosh_yeshiva_effect = yes }
	}

	# Ask about the Hebrew text of scripture -- the Christian Hebraist's
	# question.
	option = {
		name = kehillah_yeshiva.0102.a
		add_learning_lifestyle_xp = medium_lifestyle_xp
		if = {
			limit = { exists = scope:kehillah_rosh_yeshiva }
			add_opinion = {
				target = scope:kehillah_rosh_yeshiva
				modifier = respect_opinion
				opinion = 5
			}
		}
	}
	# Debate a student.
	option = {
		name = kehillah_yeshiva.0102.b
		duel = {
			skill = learning
			value = 10
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				desc = kehillah_yeshiva.0102.b.win
				add_prestige = minor_prestige_gain
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				desc = kehillah_yeshiva.0102.b.lose
				add_learning_lifestyle_xp = minor_lifestyle_xp
			}
		}
	}
	# Pass on.
	option = {
		name = kehillah_yeshiva.0102.c
	}
}

# =====================================================================
# HOLD A ZMAN -- V29 section 9.1 (2026-10-11).
#   .0201  the zman begins: students gather, choose the term's focus
#   .0202  the zman ends: students advance, the ready are ordained, fame
# Root = the Kehillah leader.
# =====================================================================

kehillah_yeshiva.0201 = {
	type = character_event
	title = kehillah_yeshiva.0201.t
	desc = kehillah_yeshiva.0201.desc
	theme = learning
	left_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = personality_rational
	}
	right_portrait = {
		character = root
		animation = thinking
		trigger = { NOT = { scope:kehillah_rosh_yeshiva ?= root } }
	}

	immediate = {
		kehillah_save_rosh_yeshiva_effect = yes
		kehillah_gather_zman_students_effect = yes
		# Something happens during the term, most of the time (.0205).
		trigger_event = { id = kehillah_yeshiva.0205 days = { 15 30 } }
	}

	# Talmud: the head (and a rabbi leader) deepen in Talmudics.
	option = {
		name = kehillah_yeshiva.0201.a
		custom_tooltip = kehillah_yeshiva.0201.a.tt
		scope:kehillah_rosh_yeshiva ?= {
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = kehillah_zman_focus_head_xp }
		}
		if = {
			limit = {
				has_trait = kehillah_rabbi_trait
				NOT = { scope:kehillah_rosh_yeshiva ?= root }
			}
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = kehillah_zman_focus_leader_xp }
		}
		trigger_event = { id = kehillah_yeshiva.0202 days = { 45 63 } }
		ai_chance = { base = 40 }
	}
	# Scripture and commentary: Parshanut, and every student learns more.
	option = {
		name = kehillah_yeshiva.0201.b
		custom_tooltip = kehillah_yeshiva.0201.b.tt
		scope:kehillah_rosh_yeshiva ?= {
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_zman_focus_head_xp }
		}
		every_courtier = {
			limit = { has_character_flag = kehillah_zman_student }
			add_learning_skill = 1
		}
		trigger_event = { id = kehillah_yeshiva.0202 days = { 45 63 } }
		ai_chance = { base = 40 }
	}
	# Philosophy: Hashkafa for the head and a rabbi leader.
	option = {
		name = kehillah_yeshiva.0201.c
		custom_tooltip = kehillah_yeshiva.0201.c.tt
		scope:kehillah_rosh_yeshiva ?= {
			add_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = kehillah_zman_focus_head_xp }
		}
		if = {
			limit = {
				has_trait = kehillah_rabbi_trait
				NOT = { scope:kehillah_rosh_yeshiva ?= root }
			}
			add_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = kehillah_zman_focus_leader_xp }
		}
		trigger_event = { id = kehillah_yeshiva.0202 days = { 45 63 } }
		ai_chance = { base = 20 }
	}
}

# ---------------------------------------------------------------------
# .0210 -- hidden. Root = the leader of the community an alumnus came
# home to; his title's kehillah_alumnus_incoming names him. Seats him as
# Dayyan if the seat is still free (see kehillah_place_alumnus_effect).
# ---------------------------------------------------------------------
kehillah_yeshiva.0210 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { primary_title ?= { has_variable = kehillah_alumnus_incoming } }
			primary_title.var:kehillah_alumnus_incoming = { save_scope_as = kehillah_arriving_alumnus }
			if = {
				limit = {
					NOT = { exists = cp:councillor_court_chaplain }
					scope:kehillah_arriving_alumnus = {
						is_alive = yes
						liege ?= root
					}
				}
				assign_councillor_type = {
					type = councillor_court_chaplain
					target = scope:kehillah_arriving_alumnus
				}
			}
			if = {
				limit = { cp:councillor_court_chaplain ?= scope:kehillah_arriving_alumnus }
				debug_log = "kehillah_yeshiva.0210: an alumnus took a Dayyan seat"
			}
			else = {
				debug_log = "kehillah_yeshiva.0210: the alumnus could NOT be seated (he stays as a rabbi at court)"
			}
			primary_title = { remove_variable = kehillah_alumnus_incoming }
		}
	}
}

# ---------------------------------------------------------------------
# DURING THE TERM -- V29 section 9.1 (2026-10-11). Root = the leader;
# this term's students carry kehillah_zman_student.
#   .0205  hidden: pick what happens (or nothing)
#   .0206  a brilliant student
#   .0207  two students quarrel over a ruling
# ---------------------------------------------------------------------
kehillah_yeshiva.0205 = {
	type = character_event
	hidden = yes
	trigger = { has_character_flag = kehillah_zman_in_progress }
	immediate = {
		random_list = {
			45 = {
				trigger = { any_courtier = { has_character_flag = kehillah_zman_student } }
				trigger_event = kehillah_yeshiva.0206
			}
			35 = {
				trigger = {
					any_courtier = {
						count >= 2
						has_character_flag = kehillah_zman_student
					}
				}
				trigger_event = kehillah_yeshiva.0207
			}
			20 = {}
		}
	}
}

kehillah_yeshiva.0206 = {
	type = character_event
	title = kehillah_yeshiva.0206.t
	desc = kehillah_yeshiva.0206.desc
	theme = learning
	left_portrait = {
		character = scope:kehillah_bright_student
		animation = personality_rational
	}
	right_portrait = {
		character = scope:kehillah_rosh_yeshiva
		animation = happiness
		trigger = { exists = scope:kehillah_rosh_yeshiva }
	}

	immediate = {
		kehillah_save_rosh_yeshiva_effect = yes
		random_courtier = {
			limit = { has_character_flag = kehillah_zman_student }
			weight = {
				base = 1
				modifier = { factor = 3 learning >= 10 }
			}
			save_scope_as = kehillah_bright_student
		}
	}

	# Give him the head's own time: he learns fast.
	option = {
		name = kehillah_yeshiva.0206.a
		scope:kehillah_bright_student = {
			add_learning_skill = 3
			add_character_flag = kehillah_promising_student
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}
	}
	# Treat all alike: the whole class benefits a little.
	option = {
		name = kehillah_yeshiva.0206.b
		every_courtier = {
			limit = { has_character_flag = kehillah_zman_student }
			add_learning_lifestyle_xp = minor_lifestyle_xp
		}
		add_piety = minor_piety_gain
	}
}

kehillah_yeshiva.0207 = {
	type = character_event
	title = kehillah_yeshiva.0207.t
	desc = kehillah_yeshiva.0207.desc
	theme = learning
	left_portrait = {
		character = scope:kehillah_disputant_a
		animation = anger
	}
	right_portrait = {
		character = scope:kehillah_disputant_b
		animation = disapproval
	}

	immediate = {
		kehillah_save_rosh_yeshiva_effect = yes
		random_courtier = {
			limit = { has_character_flag = kehillah_zman_student }
			save_scope_as = kehillah_disputant_a
		}
		random_courtier = {
			limit = {
				has_character_flag = kehillah_zman_student
				NOT = { this = scope:kehillah_disputant_a }
			}
			save_scope_as = kehillah_disputant_b
		}
	}

	# Let them argue it out: that is how a yeshiva learns.
	option = {
		name = kehillah_yeshiva.0207.a
		scope:kehillah_disputant_a = { add_learning_skill = 1 }
		scope:kehillah_disputant_b = { add_learning_skill = 1 }
		random = {
			chance = 25
			scope:kehillah_disputant_a = {
				set_relation_rival = {
					target = scope:kehillah_disputant_b
					reason = kehillah_rival_yeshiva_quarrel
				}
			}
		}
	}
	# The Rosh Yeshiva rules. A good ruling raises the yeshiva's name.
	option = {
		name = kehillah_yeshiva.0207.b
		trigger = { exists = scope:kehillah_rosh_yeshiva }
		if = {
			limit = { scope:kehillah_rosh_yeshiva = { kehillah_judicial_learning_value >= kehillah_zman_quality_good } }
			custom_tooltip = kehillah_yeshiva.0207.b.good.tt
			kehillah_add_yeshiva_fame_effect = { AMOUNT = 2 }
		}
		else = {
			custom_tooltip = kehillah_yeshiva.0207.b.poor.tt
			scope:kehillah_disputant_a = {
				add_opinion = {
					target = scope:kehillah_rosh_yeshiva
					modifier = disappointed_opinion
					opinion = -10
				}
			}
		}
	}
}
