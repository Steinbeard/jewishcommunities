# Console-only probes for the crisis political-consequences pass (2026-11-12).
# Run as the leader of an affected Kehillah in a disposable save:
#   event kehillah_crisis_political_debug.1  report whether the host has an active crisis
#   event kehillah_crisis_political_debug.2  apply the Quarter Struck political backlash

namespace = kehillah_crisis_political_debug

kehillah_crisis_political_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = domicile.domicile_location.county.holder.top_liege }
			domicile.domicile_location.county.holder.top_liege = {
				if = { limit = { kehillah_ruler_has_active_community_crisis_trigger = yes } debug_log = "kehillah_crisis_political_debug.1 active crisis: invitation correctly blocked" }
				else = { debug_log = "kehillah_crisis_political_debug.1 no active crisis: invitation may be considered" }
			}
		}
	}
}

kehillah_crisis_political_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_crisis_faction_victory_effect = yes
		debug_log = "kehillah_crisis_political_debug.2 applied Quarter Struck political backlash"
	}
}
