namespace = jewish_holiday_feast

# A single flavor event inside an ordinary vanilla Feast. Jewish hosts get
# the scene; Ritual Celebrations adds rewards without a new intent or decision.
jewish_holiday_feast.0001 = {
    type = activity_event
    theme = feast_activity

    title = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_pesach }
                desc = jewish_holiday_feast.0001.t.pesach
            }
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_sukkot }
                desc = jewish_holiday_feast.0001.t.sukkot
            }
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_shavuot }
                desc = jewish_holiday_feast.0001.t.shavuot
            }
            desc = jewish_holiday_feast.0001.t.generic
        }
    }
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_pesach }
                desc = jewish_holiday_feast.0001.desc.pesach
            }
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_sukkot }
                desc = jewish_holiday_feast.0001.desc.sukkot
            }
            triggered_desc = {
                trigger = { has_character_flag = jewish_holiday_recent_shavuot }
                desc = jewish_holiday_feast.0001.desc.shavuot
            }
            desc = jewish_holiday_feast.0001.desc.generic
        }
    }

    left_portrait = {
        character = root
        animation = happiness
    }

    trigger = {
        this = scope:activity.activity_host
        is_jewish_character_trigger = yes
        scope:activity = {
            has_activity_option = {
                category = special_type
                option = feast_type_generic
            }
        }
    }

    option = {
        name = jewish_holiday_feast.0001.a
        if = {
            limit = { rite = { rite_has_tenet = tenet_ritual_celebrations } }
            add_piety = minor_piety_gain
            scope:activity = {
                every_attending_character = {
                    limit = {
                        this != root
                        is_close_family_of = root
                    }
                    add_opinion = {
                        target = root
                        modifier = helpful_opinion
                        opinion = 5
                    }
                }
            }
        }
        # 2026-11-08 tenet mechanics: Minhag Avoteinu -- the custom of the
        # fathers kept at the festival steadies the host's community.
        if = {
            limit = {
                government_has_flag = government_is_kehillah
                OR = {
                    rite ?= { rite_has_parameter = kehillah_tenet_minhag_feast }
                    has_personal_tenet_flag = kehillah_tenet_minhag_feast_personal
                }
            }
            kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_tenet_minhag_feast_stability }
        }
    }
}
