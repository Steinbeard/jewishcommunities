# vizier of the Zirid emir Badis ibn Habus, son of Samuel ha-Nagid. The

# =====================================================================

# The night. Hidden.
kehillah_granada.0003 = {
 type = character_event
 hidden = yes
 trigger = { kehillah_granada_chain_valid_trigger = yes NOT = { has_character_flag = kehillah_granada_night_done } }
 immediate = {
  if = {
   limit = {
    NOT = { has_character_flag = kehillah_granada_riders_came }
    primary_title = { kehillah_crisis_title_race_running_trigger = yes }
    primary_title.var:kehillah_mob_threat ?= { has_character_flag = kehillah_crisis_granada_agitator has_character_flag = kehillah_mob_preaching }
   }
   kehillah_granada_almeria_riders_effect = yes
  }
  else_if = {
   limit = { primary_title.var:kehillah_mob_threat ?= { has_character_flag = kehillah_crisis_granada_agitator has_character_flag = kehillah_mob_preaching } }
   add_character_flag = kehillah_granada_night_done
   primary_title.var:kehillah_mob_threat = { kehillah_mob_preaching_ends_effect = yes }
  }
  else_if = {
   limit = { NOT = { primary_title = { has_variable = kehillah_crisis_outcome } } }
   # Old saves without a warning get a short grace period, not an instant war.
   kehillah_granada_warning_effect = yes
   primary_title.var:kehillah_mob_threat ?= { set_variable = { name = kehillah_mob_preach_days value = 28 } }
  }
 }
}

# Joseph was said to have poisoned the emir's heir Buluggin (d. 1064) and
# to plot with al-Mu'tasim of Almeria. On 30 December 1066 the crowd
