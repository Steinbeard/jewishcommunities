# =====================================================================
# V39 slice 4 -- Granada, 30 December 1066 (docs/spec/v39-pogroms-and-
# the-popular-crusade.md section 9), 2026-10-14. Joseph ibn Naghrela,
# vizier of the Zirid emir Badis ibn Habus, son of Samuel ha-Nagid. The
# faqih and poet Abu Ishaq of Elvira's qasida against the Jewish vizier;
# the charge that Joseph plotted with al-Mu'tasim of Almeria; the night
# the Sanhaja and the city stormed the palace. Historically everything
# failed: Joseph was killed and the community destroyed. Here good play
# can save Joseph or part of the community, rarely both.
#   .0001 The Poem of Abu Ishaq        (about a month after the start)
#   .0002 Whispers at Court            (about two months later)
#   .0003 (hidden) the night: the mob rises in the city, or the year ends
#   .0004 The Year Ends Quietly        (the poet silenced, and luck)
#   .0005 The Night of the Sanhaja     (the mob rises)
# Scope: the leader of the Kehillah of Granada.
# =====================================================================

scripted_effect kehillah_granada_scopes_effect = {
	domicile.domicile_location.county.holder.top_liege ?= { save_scope_as = kehillah_granada_emir }
	title:c_almeria.holder ?= { save_scope_as = kehillah_granada_almeria }
}

kehillah_granada.0001 = {
	type = character_event
	title = kehillah_granada.0001.t
	desc = kehillah_granada.0001.desc
	theme = intrigue
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:kehillah_granada_emir
		animation = thinking
	}
	trigger = { kehillah_granada_chain_valid_trigger = yes }
	immediate = {
		kehillah_granada_scopes_effect = yes
	}

	# Ask the emir to silence the poet.
	option = {
		name = kehillah_granada.0001.a
		trigger = { exists = scope:kehillah_granada_emir }
		random_list = {
			0 = {
				modifier = { add = kehillah_granada_silence_chance_value }
				desc = kehillah_granada_silenced
				custom_tooltip = kehillah_granada_silenced_tt
				add_character_flag = kehillah_granada_silenced
			}
			0 = {
				modifier = { add = kehillah_granada_silence_fail_value }
				desc = kehillah_granada_not_silenced
				custom_tooltip = kehillah_granada_not_silenced_tt
			}
		}
		ai_chance = { base = 40 }
	}

	# Double your guard.
	option = {
		name = kehillah_granada.0001.b
		trigger = { gold >= minor_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = minor_gold_value
		random_list = {
			0 = {
				modifier = { add = kehillah_granada_guard_chance_value }
				desc = kehillah_granada_guarded
				custom_tooltip = kehillah_granada_guarded_tt
				add_character_flag = kehillah_granada_guarded
				kehillah_granada_adjust_severity_effect = { AMOUNT = -15 }
			}
			0 = {
				modifier = { add = kehillah_granada_guard_fail_value }
				desc = kehillah_granada_not_guarded
				custom_tooltip = kehillah_granada_not_guarded_tt
			}
		}
		ai_chance = { base = 25 }
	}

	# Seek a friend in Almeria.
	option = {
		name = kehillah_granada.0001.c
		trigger = { exists = scope:kehillah_granada_almeria }
		random_list = {
			0 = {
				modifier = { add = kehillah_granada_almeria_chance_value }
				desc = kehillah_granada_escape_route
				custom_tooltip = kehillah_granada_escape_route_tt
				add_character_flag = kehillah_granada_escape_route
			}
			0 = {
				modifier = { add = kehillah_granada_almeria_fail_value }
				desc = kehillah_granada_plot_discovered
				custom_tooltip = kehillah_granada_plot_discovered_tt
				add_character_flag = kehillah_granada_plot_discovered
				kehillah_granada_adjust_severity_effect = { AMOUNT = 15 }
				scope:kehillah_granada_emir ?= {
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -10 } }
				}
				# The night comes sooner.
				trigger_event = { id = kehillah_granada.0003 days = 45 }
			}
		}
		ai_chance = { base = 10 }
	}

	# He is a poet; poets rail.
	option = {
		name = kehillah_granada.0001.d
		add_stress = minor_stress_loss
		ai_chance = { base = 25 }
	}
}

# The night. Hidden.
kehillah_granada.0003 = {
	type = character_event
	hidden = yes
	trigger = {
		kehillah_granada_chain_valid_trigger = yes
		NOT = { has_character_flag = kehillah_granada_night_done }
	}
	immediate = {
		add_character_flag = kehillah_granada_night_done
		if = {
			limit = { has_character_flag = kehillah_granada_silenced }
			random_list = {
				50 = {
					trigger_event = kehillah_granada.0004
				}
				50 = {
					kehillah_granada_night_effect = yes
				}
			}
		}
		else = {
			kehillah_granada_night_effect = yes
		}
	}
}

# V43 slice 1 (docs/spec/v43-after-the-violence-and-granada.md), 2026-10-
# 14. Joseph ha-Nagid's family, and Granada seen from the rest of
# Sepharad. Daniel: "an event chain for Sephardi kehilla leaders
# explaining the context behind the Granada uprising, foreshadowing it."
#   .0006 Azariah                      (Joseph's own; about six weeks in)
#   .0007 The City Gate                (Granada's leader, after the night)
#   .0010 A Letter from Granada        (other Sephardi leaders; ~3 weeks)
#   .0011 The Poem Travels             (other Sephardi leaders; ~7 weeks)
#   .0012 What They Say in Granada     (other Sephardi leaders; ~12 weeks)
#   .0013 News from Granada            (other Sephardi leaders; January)
# History: Samuel ha-Nagid (d. 1056) served the emirs Habbus and Badis as
# vizier and general for thirty years; his son Joseph succeeded him. The
# Sanhaja Berber elite resented a Jewish vizier; the faqih Abu Ishaq of
# Elvira, Joseph's rival, wrote a qasida calling for the Jews' slaughter;
# Joseph was said to have poisoned the emir's heir Buluggin (d. 1064) and
# to plot with al-Mu'tasim of Almeria. On 30 December 1066 the crowd
# stormed the palace and killed him; his body was crucified on the city
# gate. His widow and their son Azariah escaped to Lucena.
# =====================================================================

kehillah_granada.0006 = {
	type = character_event
	title = kehillah_granada.0006.t
	desc = kehillah_granada.0006.desc
	theme = family
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:kehillah_granada_son
		animation = happiness
	}
	trigger = {
		kehillah_granada_chain_valid_trigger = yes
		NOT = { has_character_flag = kehillah_granada_night_done }
		character:9000234 ?= {
			is_alive = yes
			is_courtier_of = root
		}
	}
	immediate = {
		character:9000234 = { save_scope_as = kehillah_granada_son }
		character:9000233 ?= {
			if = {
				limit = {
					is_alive = yes
					is_courtier_of = root
				}
				save_scope_as = kehillah_granada_wife
			}
		}
		title:d_kehillah_lucena.holder ?= { save_scope_as = kehillah_granada_lucena }
	}

	# Send them to Lucena before the winter.
	option = {
		name = kehillah_granada.0006.a
		trigger = { exists = scope:kehillah_granada_lucena }
		add_character_flag = kehillah_granada_family_sent
		kehillah_granada_family_to_lucena_effect = yes
		add_stress = minor_stress_gain
		ai_chance = { base = 30 }
	}

	# They stay. A Naghrilla does not run.
	option = {
		name = kehillah_granada.0006.b
		add_prestige = minor_prestige_gain
		ai_chance = {
			base = 50
			modifier = {
				add = 30
				has_trait = arrogant
			}
		}
	}

	# Send the boy with his tutor; his mother stays with you.
	option = {
		name = kehillah_granada.0006.c
		trigger = {
			exists = scope:kehillah_granada_lucena
			exists = scope:kehillah_granada_wife
		}
		add_character_flag = kehillah_granada_family_sent
		scope:kehillah_granada_lucena = { add_courtier = scope:kehillah_granada_son }
		custom_tooltip = kehillah_granada_son_sent_tt
		ai_chance = { base = 20 }
	}
}
