# ---------------------------------------------------------------------
# kehillah_responsa.0002 -- the compiled volume is finished.
#
# Hidden, no options, fires at days = 0 from
# kehillah_compile_responsa_decision. It exists for one reason: to get
# kehillah_book_complete_effect OUT of that decision's effect block.
#
# A decision effect is evaluated to build its tooltip, and in that pass
# set_variable writes nothing -- so the book effect, called inline right
# after the two variables it needs were "set", read them as unset on every
# tooltip rebuild frame and grew error.log by ~250,000 lines in one second
# on 2026-09-28. Called from here instead, it runs exactly once, after the
# decision's writes have actually landed. That is also how the shipped
# Write a Book chain has always called it (events/kehillah_book_events.txt
# option effect, with the variables set days earlier), so this converges on
# the one call pattern proven in a running game rather than inventing a
# second one.
#
# Nothing else belongs in here. If it grows options or a description, the
# decision should open it properly instead of firing it hidden.
# ---------------------------------------------------------------------
kehillah_responsa.0002 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "kehillah_responsa.0002: compiling the responsa volume through the book system"
		kehillah_book_complete_effect = yes
	}
}
