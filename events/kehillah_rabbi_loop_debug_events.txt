# Console-only probes for the rabbi loop. Never called from game content.
# While hosting an active Bet Din, run `event kehillah_rabbi_loop_debug.1`
# to inspect bench/scoring; .2 forces one good-ruling XP grant and a positive
# closing event for reward QA. Use a disposable save for .2.
namespace = kehillah_rabbi_loop_debug

kehillah_rabbi_loop_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_rabbi_loop_debug.1: Bet Din reward state START"
		if = {
			limit = { exists = involved_activity }
			debug_log = "kehillah_rabbi_loop_debug.1: ACTIVITY PRESENT"
			if = {
				limit = { exists = involved_activity.var:kehillah_bet_din_convening_title }
				debug_log = "kehillah_rabbi_loop_debug.1: CONVENING TITLE PRESENT"
			}
			if = {
				limit = { exists = involved_activity.var:kehillah_bet_din_av_beit_din }
				debug_log = "kehillah_rabbi_loop_debug.1: AV BEIT DIN PRESENT"
				involved_activity.var:kehillah_bet_din_av_beit_din = {
					if = {
						limit = { has_trait = kehillah_rabbi_trait }
						debug_log = "kehillah_rabbi_loop_debug.1: AV HAS SEMICHA"
					}
					if = {
						limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 1 } }
						debug_log = "kehillah_rabbi_loop_debug.1: AV TALMUDICS XP AT LEAST 1"
					}
				}
			}
			if = {
				limit = { exists = involved_activity.var:kehillah_bet_din_judge2 }
				debug_log = "kehillah_rabbi_loop_debug.1: JUDGE 2 PRESENT"
			}
			if = {
				limit = { exists = involved_activity.var:kehillah_bet_din_judge3 }
				debug_log = "kehillah_rabbi_loop_debug.1: JUDGE 3 PRESENT"
			}
			if = {
				limit = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_adequate } }
				debug_log = "kehillah_rabbi_loop_debug.1: SESSION SCORE POSITIVE"
			}
		}
		else = {
			debug_log = "kehillah_rabbi_loop_debug.1: NO ACTIVITY"
		}
		debug_log_scopes = yes
		debug_log = "kehillah_rabbi_loop_debug.1: Bet Din reward state END"
	}
}

kehillah_rabbi_loop_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = involved_activity.var:kehillah_bet_din_convening_title
				exists = involved_activity.var:kehillah_bet_din_av_beit_din
				exists = involved_activity.var:kehillah_bet_din_judge2
				exists = involved_activity.var:kehillah_bet_din_judge3
			}
			debug_log = "kehillah_rabbi_loop_debug.2: forcing one good ruling and positive close"
			kehillah_bet_din_reward_judges_for_ruling_effect = { XP = kehillah_bet_din_case_rabbi_xp_good }
			involved_activity = {
				set_variable = { name = kehillah_bet_din_session_score value = kehillah_bet_din_session_threshold_adequate }
				set_variable = { name = kehillah_bet_din_cases_heard_count value = 1 }
			}
			trigger_event = kehillah_bet_din.0099
		}
		else = {
			debug_log = "kehillah_rabbi_loop_debug.2: BLOCKED no fully staffed Bet Din activity"
		}
	}
}

# V23 delayed-letter scope spike. Run .4 as a Kehillah leader. It stores
# the sender's TITLE on a different living community leader, delivers .5
# after two days, and sends .6 to whoever holds that title on receipt.
# No gameplay state is changed beyond temporary character variables.
kehillah_rabbi_loop_debug.4 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = primary_title }
			random_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					exists = holder
					NOT = { this = root.primary_title }
					holder = { is_alive = yes }
				}
				holder = {
					set_variable = { name = kehillah_debug_letter_asking_title value = root.primary_title }
					debug_log = "kehillah_rabbi_loop_debug.4: scheduled letter on recipient"
					trigger_event = { id = kehillah_rabbi_loop_debug.5 days = 2 }
				}
			}
		}
	}
}

kehillah_rabbi_loop_debug.5 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = var:kehillah_debug_letter_asking_title }
			debug_log = "kehillah_rabbi_loop_debug.5: sender title survived delayed cross-character event"
			var:kehillah_debug_letter_asking_title = {
				if = {
					limit = { exists = holder }
					holder = { trigger_event = { id = kehillah_rabbi_loop_debug.6 days = 2 } }
				}
			}
			remove_variable = kehillah_debug_letter_asking_title
		}
		else = { debug_log = "kehillah_rabbi_loop_debug.5: FAILED sender title lost" }
	}
}

}

# Disposable-save case 7 integration probe. Make the seated bench lack
# the two relevant mastered works, give the Baghdad leader semicha and
# the Mishnah as an outside source, then open the real host event. This
# exercises the no-local-source candidate and delayed-letter path.
kehillah_rabbi_loop_debug.7 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = involved_activity.var:kehillah_bet_din_convening_title
				exists = involved_activity.var:kehillah_bet_din_av_beit_din
				exists = involved_activity.var:kehillah_bet_din_judge2
				exists = involved_activity.var:kehillah_bet_din_judge3
			}
			involved_activity.var:kehillah_bet_din_av_beit_din = { clear_variable_list = kehillah_mastered_works }
			involved_activity.var:kehillah_bet_din_judge2 = { clear_variable_list = kehillah_mastered_works }
			involved_activity.var:kehillah_bet_din_judge3 = { clear_variable_list = kehillah_mastered_works }
			title:d_kehillah_baghdad.holder = {
				if = { limit = { NOT = { has_trait = kehillah_rabbi_trait } } add_trait = kehillah_rabbi_trait }
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah } } }
					add_to_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }
				}
			}
			debug_log = "kehillah_rabbi_loop_debug.7: no-source bench and Baghdad correspondent prepared"
			trigger_event = kehillah_bet_din.0061
		}
		else = { debug_log = "kehillah_rabbi_loop_debug.7: BLOCKED no fully staffed Bet Din activity" }
	}
}
