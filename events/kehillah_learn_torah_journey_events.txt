# =====================================================================
# Learn Torah -- the travel journey. ADDED 2026-09-15 at user request
# ("implement the travel to study a book"). REWRITTEN the same day, at
# further user request ("studying should happen when you reach the
# target court, not when you return home... make it another activity
# like bet din"): now fired from activity_kehillah_learn_torah_journey's
# own on_phase_active (common/activities/activity_types/kehillah_
# learn_torah_journey.txt) instead of a hand-rolled start_travel_plan's
# on_arrival_event, which kehillah_visit_library_interaction (common/
# character_interactions/kehillah_character_interactions.txt) used to
# own before being retired the same pass.
#
# WHY THIS VERSION NEEDS NO PERSISTED DESTINATION VARIABLE AT ALL, unlike
# the retired interaction's own kehillah_journey_destination_title
# character variable. That variable existed specifically because travel
# started in one effect chain (the interaction's on_accept) and arrival
# fired a SEPARATE one, potentially weeks later, past this mod's own
# proven same-session scope-persistence guarantees. With a real
# activity, on_phase_active fires ONLY once the host has actually
# arrived and the phase has gone active -- root is already physically AT
# the destination (root.location) by the time this event's own immediate
# runs, so the destination can simply be looked up fresh, in this same
# effect chain, every time: which registered community's own domicile
# sits at root's current location. No variable needs to survive a
# journey any more because nothing needs to be written before the
# journey starts at all.
# =====================================================================

namespace = kehillah_learn_torah_journey

# ---------------------------------------------------------------------
# kehillah_learn_torah_journey.0001 -- ARRIVAL: STUDY AT THEIR BEIT
# MIDRASH
# root: the traveller (activity_type.info: on_phase_active's own "root =
# character in this phase", carried into this event by the plain
# trigger_event call that fires it), now physically at the community
# whose domicile occupies root.location. Three track options, same
# shape as kehillah_learn_torah.0001 (events/kehillah_learn_torah_
# events.txt) and calling the SAME three kehillah_study_<track>_effect
# (common/scripted_effects/kehillah_library_effects.txt), just with
# LIBRARY = scope:kehillah_journey_destination_title instead of primary_
# title -- the entire reason those effects took a $LIBRARY$ parameter in
# the first place. A fourth option lets the traveller leave without
# studying, for player agency.
# ---------------------------------------------------------------------

kehillah_learn_torah_journey.0001 = {
	type = character_event
	title = kehillah_learn_torah_journey.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:kehillah_journey_host }
				desc = kehillah_learn_torah_journey.0001.desc
			}
			desc = kehillah_learn_torah_journey.0001.desc.fallback
		}
	}
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:kehillah_journey_host
		animation = interested
		trigger = { exists = scope:kehillah_journey_host }
	}

	trigger = {
		government_has_flag = government_is_kehillah
	}

	# Finds which registered community's own domicile sits exactly at
	# root's current physical location, and saves both the TITLE (what
	# the $LIBRARY$-parameterized study effects and triggers need) and
	# its current holder (for the portrait and loc text -- read fresh,
	# not the character who happened to hold it when any earlier step
	# ran, so a succession at the destination is handled the same
	# correctly-by-construction way this mod's retired interaction
	# already established for this same feature). every_in_global_list,
	# not any_/ordered_, because the match itself (not a boolean or a
	# ranked pick) is what is needed here -- in practice at most one
	# entry ever matches, since two communities cannot share a domicile
	# location.
	immediate = {
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { domicile.domicile_location = root.location }
			}
			save_scope_as = kehillah_journey_destination_title
		}
		if = {
			limit = { exists = scope:kehillah_journey_destination_title.holder }
			scope:kehillah_journey_destination_title.holder = { save_scope_as = kehillah_journey_host }
		}
	}

	# PARSHANUT.
	option = {
		name = kehillah_learn_torah_journey.0001.a
		trigger = { kehillah_owns_parshanut_work_trigger = { LIBRARY = scope:kehillah_journey_destination_title } }
		kehillah_study_parshanut_effect = { LIBRARY = scope:kehillah_journey_destination_title }
		ai_chance = { base = 30 }
	}

	# TALMUDICS.
	option = {
		name = kehillah_learn_torah_journey.0001.b
		trigger = { kehillah_owns_talmudics_work_trigger = { LIBRARY = scope:kehillah_journey_destination_title } }
		kehillah_study_talmudics_effect = { LIBRARY = scope:kehillah_journey_destination_title }
		ai_chance = { base = 30 }
	}

	# HASHKAFA.
	option = {
		name = kehillah_learn_torah_journey.0001.c
		trigger = { kehillah_owns_hashkafa_work_trigger = { LIBRARY = scope:kehillah_journey_destination_title } }
		kehillah_study_hashkafa_effect = { LIBRARY = scope:kehillah_journey_destination_title }
		ai_chance = { base = 30 }
	}

	# LEAVE WITHOUT STUDYING.
	option = {
		name = kehillah_learn_torah_journey.0001.d
		custom_tooltip = kehillah_learn_torah_journey.0001.d.tt
		ai_chance = { base = 10 }
	}

	# ENDS THE ACTIVITY'S OWN (SOLE) PHASE, regardless of which option
	# was chosen -- the engine's own activity-completion machinery
	# returns the host home from here, the same standard behavior any
	# travelled vanilla activity already relies on for its own host
	# (Pilgrimage, Hunt, Tour) -- see kehillah_learn_torah_journey.txt's
	# own header for why that specific claim, for a HOST rather than a
	# guest, is this build's one real unverified assumption. involved_
	# activity, not scope:activity -- this runs from inside a fired
	# EVENT, not from the activity_type's own native effect blocks, the
	# identical reasoning kehillah_bet_din_advance_docket_effect (common/
	# scripted_effects/kehillah_bet_din_scripted_effects.txt) already
	# documents for the same distinction.
	after = {
		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				progress_activity_phase_after = { days = 1 }
			}
		}
	}
}
