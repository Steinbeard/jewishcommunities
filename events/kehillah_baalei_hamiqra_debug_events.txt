# Console-only probes. .2 advances scholarship and changes the save.
namespace = kehillah_baalei_hamiqra_debug

kehillah_baalei_hamiqra_debug.1 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		if = {
			limit = { rite = { rite_has_tenet = tenet_baalei_hamiqra } }
			debug_log = "baalei_hamiqra_debug: rite has Baalei ha-Miqra"
		}
		else = { debug_log = "baalei_hamiqra_debug: rite lacks Baalei ha-Miqra" }
		if = {
			limit = { rite:rabbinism = { has_tenet_status = { tenet = tenet_baalei_hamiqra status = prohibited } } }
			debug_log = "baalei_hamiqra_debug: Rabbinism prohibits Baalei ha-Miqra"
		}
		else = { debug_log = "baalei_hamiqra_debug: Rabbinism prohibition missing" }
		if = {
			limit = { rite:karaism = { has_tenet_status = { tenet = tenet_religious_legal_pronouncements status = prohibited } } }
			debug_log = "baalei_hamiqra_debug: Karaism prohibits Rabbinic Halakha"
		}
		else = { debug_log = "baalei_hamiqra_debug: Karaism prohibition missing" }
		debug_log_scopes = yes
	}
}

kehillah_baalei_hamiqra_debug.2 = {
	type = character_event
	hidden = yes
	orphan = yes
	immediate = {
		kehillah_record_miqra_study_effect = { XP = 8 }
		if = {
			limit = { has_trait = lifestyle_scholar }
			debug_log = "baalei_hamiqra_debug: Religious Scholar present"
		}
		else = { debug_log = "baalei_hamiqra_debug: Religious Scholar not yet acquired" }
		debug_log_scopes = yes
	}
}
