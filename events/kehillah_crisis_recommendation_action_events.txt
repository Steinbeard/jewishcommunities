# One action, one scene: the council shortcut and the standing menu both arrive
# here. The confirm option rechecks the original action gate before spending.
# fast: the same confirmation scene from counsel or the standing action.
kehillah_crisis_recommend.0101 = {
 type = character_event
 title = kehillah_crisis_recommend.0101.t
 desc = kehillah_crisis_recommend.0101.desc
 theme = faith
 left_portrait = { character = root animation = thinking }
 right_portrait = {
  trigger = { exists = scope:kehillah_crisis_action_adviser }
  character = scope:kehillah_crisis_action_adviser
  animation = personality_rational
 }
 trigger = { kehillah_crisis_can_act_trigger = { ONCE = kehillah_crisis_fasted } }
 immediate = { cp:councillor_court_chaplain ?= { save_scope_as = kehillah_crisis_action_adviser } }
 option = {
  name = kehillah_crisis_recommend.0101.confirm
  trigger = { kehillah_crisis_can_act_trigger = { ONCE = kehillah_crisis_fasted } }
  hidden_effect = { kehillah_crisis_commit_action_effect = yes }
  kehillah_crisis_recommended_fast_effect = yes
 }
 option = {
  name = kehillah_crisis_recommend.0101.risk
  flavor = kehillah_crisis_recommend.0101.risk.tt
  hidden_effect = { kehillah_crisis_commit_action_effect = yes }
  kehillah_crisis_recommended_challenge_effect = { BONUS = kehillah_crisis_recommend_dayyan_bonus_value SUCCESS = kehillah_crisis_recommended_fast_challenge_success_effect FAILURE = kehillah_crisis_recommended_fast_challenge_failure_effect }
 }
 option = {
  name = { trigger = { primary_title = { has_variable = kehillah_crisis_pending_category } } text = kehillah_crisis_phase_actions.back }
  name = { trigger = { primary_title = { NOT = { has_variable = kehillah_crisis_pending_category } } } text = kehillah_crisis_recommend.action_decline }
  trigger = { is_ai = no }
  hidden_effect = { kehillah_crisis_return_to_options_effect = yes }
 }
}
