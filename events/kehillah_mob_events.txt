# =====================================================================
# V39 slice 1 -- the Mob's events (docs/spec/v39-pogroms-and-the-popular-
# crusade.md section 5), 2026-10-13. The target community's leader, player
# or AI, gets each; the AI answers by ai_chance.
#   .0001 A Preacher Raises a Host   (the host forms: early answers)
#   .0002 They Are Coming for Us     (days away: the last answers on the road)
#   .0003 At the Gates               (arrival: the last answers)
#   .0005 The Lord Gives Us Up       (refuge withdrawn; players)
#   .0010 After the Pogrom           (the report; the hostages' ransom)
#   .0020 The Host Is Broken         (spared; players)
# Every check is a random_list, so the option shows its odds.
# =====================================================================

# Shared immediate: the host's leader, his place, the lord.
scripted_effect kehillah_mob_event_scopes_effect = {
	var:kehillah_mob_ref ?= { save_scope_as = kehillah_mob_leader }
	var:kehillah_mob_location_ref ?= { save_scope_as = kehillah_mob_place }
	domicile.domicile_location ?= { save_scope_as = kehillah_mob_town }
	domicile.domicile_location.county.holder ?= { save_scope_as = kehillah_mob_lord }
	domicile.domicile_location.county.holder.top_liege ?= { save_scope_as = kehillah_mob_overlord }
	scope:kehillah_mob_leader.faith.religious_head ?= { save_scope_as = kehillah_mob_hof }
	# The host's men -- or, while he is still preaching, his following
	# (live 2026-10-14: "0 people" during the preaching).
	if = {
		limit = { scope:kehillah_mob_leader ?= { has_variable = kehillah_mob_size } }
		set_variable = { name = kehillah_mob_size_ref value = scope:kehillah_mob_leader.var:kehillah_mob_size }
	}
	else_if = {
		limit = { scope:kehillah_mob_leader ?= { has_variable = kehillah_mob_following } }
		set_variable = { name = kehillah_mob_size_ref value = scope:kehillah_mob_leader.var:kehillah_mob_following }
	}
}

# After the pogrom: the report, and the hostages.
kehillah_mob.0010 = {
	type = character_event
	title = kehillah_mob.0010.t
	desc = {
		desc = kehillah_mob.0010.desc
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_pogrom_martyrs > 0 }
				desc = kehillah_mob.0010.desc_martyrs
			}
			desc = kehillah_mob.0010.desc_no_martyrs
		}
		first_valid = {
			triggered_desc = {
				trigger = { var:kehillah_pogrom_hostages > 0 }
				desc = kehillah_mob.0010.desc_hostages
			}
			desc = kehillah_mob.0010.desc_no_hostages
		}
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = kehillah_pogrom_leader_fell }
				desc = kehillah_mob.0010.desc_leader_fell
			}
			triggered_desc = {
				trigger = { has_variable = kehillah_pogrom_leader_baptized }
				desc = kehillah_mob.0010.desc_leader_baptized
			}
			triggered_desc = {
				trigger = { has_variable = kehillah_pogrom_leader_taken }
				desc = kehillah_mob.0010.desc_leader_taken
			}
			desc = kehillah_mob.0010.desc_end
		}
	}
	theme = dread
	left_portrait = {
		character = root
		animation = grief
	}
	right_portrait = {
		character = scope:kehillah_mob_leader
		animation = rage
	}
	immediate = {
		kehillah_mob_event_scopes_effect = yes
		set_variable = { name = kehillah_pogrom_ransom_ref value = kehillah_pogrom_ransom_cost_value }
	}

	# Redeem the captives.
	option = {
		name = kehillah_mob.0010.a
		trigger = {
			var:kehillah_pogrom_hostages > 0
			gold >= kehillah_pogrom_ransom_cost_value
		}
		show_as_unavailable = { var:kehillah_pogrom_hostages > 0 }
		remove_short_term_gold = kehillah_pogrom_ransom_cost_value
		custom_tooltip = kehillah_mob_ransomed_tt
		hidden_effect = {
			scope:kehillah_mob_leader ?= {
				every_prisoner = { release_from_prison = yes }
			}
		}
		ai_chance = { base = 80 }
	}

	# We cannot pay.
	option = {
		name = kehillah_mob.0010.b
		trigger = { var:kehillah_pogrom_hostages > 0 }
		custom_tooltip = kehillah_mob_not_ransomed_tt
		hidden_effect = {
			scope:kehillah_mob_leader ?= {
				every_prisoner = {
					release_from_prison = yes
					# A player's own leader chooses, in a scene (V43 slice 0).
					if = {
						limit = {
							this = root
							root = { is_ai = no }
						}
						root = { set_variable = { name = kehillah_pogrom_leader_choice value = yes } }
					}
					else = {
					random_list = {
						50 = {
							root = { change_variable = { name = kehillah_pogrom_dead add = 1 } }
							# The leader himself dies in `after`, once the report is
							# done with him.
							if = {
								limit = { this = root }
								root = { set_variable = { name = kehillah_pogrom_leader_fell value = yes } }
							}
							else = {
								death = { death_reason = death_murder killer = scope:kehillah_mob_leader }
							}
						}
						50 = {
							root = { change_variable = { name = kehillah_pogrom_baptized add = 1 } }
							if = {
								limit = { this = root }
								root = { set_variable = { name = kehillah_pogrom_leader_baptized value = yes } }
							}
							else = {
								kehillah_anus_baptize_effect = yes
								set_variable = { name = kehillah_anus value = root.primary_title }
								root.primary_title = {
									add_to_variable_list = { name = kehillah_anusim target = prev }
								}
								move_to_pool_at = root.domicile.domicile_location
							}
						}
					}
					}
				}
			}
		}
		ai_chance = { base = 20 }
	}

	# We will remember them.
	option = {
		name = kehillah_mob.0010.c
		trigger = { var:kehillah_pogrom_hostages <= 0 }
		ai_chance = { base = 100 }
	}

	after = {
		hidden_effect = {
			# V41: the leader baptized at the font (before the host goes).
			if = {
				limit = { has_variable = kehillah_pogrom_leader_baptized }
				remove_variable = kehillah_pogrom_leader_baptized
				kehillah_pogrom_leader_baptized_effect = yes
			}
			if = {
				limit = { has_variable = kehillah_pogrom_leader_taken }
				remove_variable = kehillah_pogrom_leader_taken
			}
			# V43 slice 0: the unransomed leader's choice, the next day.
			if = {
				limit = { has_variable = kehillah_pogrom_leader_choice }
				remove_variable = kehillah_pogrom_leader_choice
				scope:kehillah_mob_leader ?= {
					root = { set_variable = { name = kehillah_pogrom_captor_ref value = prev } }
				}
				trigger_event = { id = kehillah_pogrom.0040 days = 1 }
			}
			scope:kehillah_mob_leader ?= {
				if = {
					limit = { is_alive = yes }
					kehillah_mob_after_strike_effect = yes
				}
			}
			if = {
				limit = { has_variable = kehillah_pogrom_leader_fell }
				remove_variable = kehillah_pogrom_leader_fell
				death = { death_reason = death_murder }
			}
		}
	}
}

# =====================================================================
# V42 (docs/spec/v42-the-crisis-as-a-place.md), 2026-10-14. The crisis as
# a place: the elders' hub shows where things stand, and branches -- at
# most four choices to an event, "something else..." for more -- to the
# people who can help. Each one asked answers in a scene of their own,
# to the leader or to the messenger sent. Players only: AI communities
# answer through kehillah_mob_ai_answers_effect, with the same odds.
#   .0100 The Elders Gather (the hub: the state of things)
#   .0101 Protection (the lord, the overlord, the Church)
#   .0102 The Church (the lord's chaplain, the head of the faith)
#   .0104 How to approach him (go yourself, or send a messenger)
#   .0105 The messenger's errand (refuge, or his men)
#   .0110 Before the lord (yourself)
#   .0112 / .0113 What we can offer
#   .0111 The lord's answer
#   .0130 Before the churchman (yourself)
#   .0131 The Church's answer
#   .0140 / .0142 In the preacher's camp
#   .0141 The preacher's answer
#   .0150 Our own defence
# State: on the community title (the mechanics, cleared by kehillah_mob_
# clear_title_effect); mirrored onto the leader for the text, which can
# only read a character's variables.
# =====================================================================

# The hub fires only while the host is still on the road.
scripted_trigger kehillah_mob_hub_open_trigger = {
	is_ai = no
	exists = var:kehillah_mob_ref
	var:kehillah_mob_ref = {
		is_alive = yes
		NOT = { has_character_flag = kehillah_mob_at_gates }
		NOT = { has_character_flag = kehillah_mob_done }
	}
}

# Someone the elders can still ask for protection.
# Scope: the leader; scope:kehillah_mob_patron. Something else to offer:
# what you know of him, an old friendship, the physician's service.
scripted_trigger kehillah_mob_more_offers_trigger = {
	OR = {
		has_usable_hook = scope:kehillah_mob_patron
		has_relation_friend = scope:kehillah_mob_patron
		has_trait = lifestyle_physician
	}
}

# Anything at all to offer him.
scripted_trigger kehillah_mob_any_offer_trigger = {
	OR = {
		kehillah_mob_more_offers_trigger = yes
		gold >= medium_gold_value
		scope:kehillah_mob_patron = { var:kehillah_loan_lender_title ?= root.primary_title }
		primary_title = {
			has_variable = kehillah_charter_cache_security
			var:kehillah_charter_cache_security >= 1
		}
	}
}

scripted_trigger kehillah_mob_protection_left_trigger = {
	OR = {
		AND = {
			exists = scope:kehillah_mob_lord
			NOT = { scope:kehillah_mob_lord = root }
			primary_title = { NOT = { is_target_in_variable_list = { name = kehillah_mob_asked_refuge target = scope:kehillah_mob_lord } } }
		}
		AND = {
			exists = scope:kehillah_mob_overlord
			NOT = { scope:kehillah_mob_overlord = scope:kehillah_mob_lord }
			primary_title = { NOT = { is_target_in_variable_list = { name = kehillah_mob_asked_refuge target = scope:kehillah_mob_overlord } } }
		}
		AND = {
			exists = scope:kehillah_mob_chaplain
			primary_title = { NOT = { is_target_in_variable_list = { name = kehillah_mob_asked_condemn target = scope:kehillah_mob_chaplain } } }
			NOT = { kehillah_mob_decided_trigger = { WHO = scope:kehillah_mob_chaplain LIST = kehillah_mob_condemned_by } }
			NOT = { kehillah_mob_decided_trigger = { WHO = scope:kehillah_mob_chaplain LIST = kehillah_mob_condemn_refused_list } }
		}
		AND = {
