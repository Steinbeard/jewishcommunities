 # The Dayyan answers him: a more learned voice than mine.
 option = {
  name = kehillah_crisis_race.0100.a
  hidden_effect = { kehillah_crisis_commit_action_effect = yes }
  trigger = {
   exists = scope:kehillah_crisis_dayyan
   scope:kehillah_crisis_dayyan.learning > learning
  }
  scope:kehillah_crisis_dayyan = {
   duel = {
    skill = learning
    target = scope:kehillah_crisis_agitator_scope
    50 = {
     desc = kehillah_crisis_race.0100.success
     compare_modifier = { value = scope:duel_value multiplier = 3.5 }
     min = 5
     send_interface_toast = {
      type = event_toast_text_and_effect_good
      title = kehillah_toast_success
      desc = kehillah_crisis_race.0100.success_desc
      left_icon = root
      root.primary_title = { hidden_effect = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_efforts_catalyst POINTS = kehillah_crisis_ev_medium } } custom_tooltip = kehillah_crisis_tt_spared_moderate }
     }
    }
    50 = {
     desc = kehillah_crisis_race.0100.failure
     compare_modifier = { value = scope:duel_value multiplier = -3.5 }
     min = 5
     send_interface_toast = {
      type = event_toast_text_and_effect_bad
      title = kehillah_toast_failure
      desc = kehillah_crisis_race.0100.failure_desc
      left_icon = root
      root.primary_title = { hidden_effect = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_misstep_catalyst POINTS = kehillah_crisis_ev_medium } } custom_tooltip = kehillah_crisis_tt_armed_moderate }
     }
    }
   }
  }
  ai_chance = { base = 60 }
