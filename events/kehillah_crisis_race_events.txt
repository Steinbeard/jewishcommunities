# The attack as a race (docs/spec/2026-11-01-the-attack-as-a-race.md): the scenes behind the
# crisis window's actions. Every outcome adds a named catalyst through
# kehillah_crisis_race_add_effect, which does nothing once the race has ended.

# Answer him in public (step 1c). The Dayyan, if more learned than the leader, or the leader.
kehillah_crisis_race.0100 = {
 type = character_event
 title = kehillah_crisis_race.0100.t
 desc = kehillah_crisis_race.0100.desc
 theme = learning
 left_portrait = { character = root animation = personality_rational }
 right_portrait = { character = scope:kehillah_crisis_agitator_scope animation = scheme }
 trigger = { kehillah_crisis_warning_trigger = yes }
 immediate = {
  primary_title.var:kehillah_mob_threat = { save_scope_as = kehillah_crisis_agitator_scope }
  cp:councillor_court_chaplain ?= { save_scope_as = kehillah_crisis_dayyan }
 }

 # The Dayyan answers him: a more learned voice than mine.
 option = {
  name = kehillah_crisis_race.0100.a
  trigger = {
   exists = scope:kehillah_crisis_dayyan
   scope:kehillah_crisis_dayyan.learning > learning
  }
  scope:kehillah_crisis_dayyan = {
   duel = {
    skill = learning
    target = scope:kehillah_crisis_agitator_scope
    50 = {
     desc = kehillah_crisis_race.0100.success
     compare_modifier = { value = scope:duel_value multiplier = 3.5 }
     min = 5
     root.primary_title = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_act_answered_catalyst } }
    }
    50 = {
     desc = kehillah_crisis_race.0100.failure
     compare_modifier = { value = scope:duel_value multiplier = -3.5 }
     min = 5
     root.primary_title = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_act_answer_failed_catalyst } }
    }
   }
  }
  ai_chance = { base = 60 }
 }
 # I will answer him myself.
 option = {
  name = kehillah_crisis_race.0100.b
  duel = {
   skill = learning
   target = scope:kehillah_crisis_agitator_scope
   50 = {
    desc = kehillah_crisis_race.0100.success
    compare_modifier = { value = scope:duel_value multiplier = 3.5 }
    min = 5
    primary_title = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_act_answered_catalyst } }
   }
   50 = {
    desc = kehillah_crisis_race.0100.failure
    compare_modifier = { value = scope:duel_value multiplier = -3.5 }
    min = 5
    primary_title = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_act_answer_failed_catalyst } }
   }
  }
  ai_chance = { base = 40 }
 }
}
