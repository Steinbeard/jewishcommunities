# Console-only probes; never called by gameplay. Select any character, then run
# `event kehillah_patronym_debug.1`; read debug.log, and read the name shown on the portrait.
namespace = kehillah_patronym_debug

kehillah_patronym_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_patronym_debug.1: probing selected character (name is in the scope dump below)"
		if = {
			limit = { exists = father }
			debug_log = "kehillah_patronym_debug.1: HAS FATHER (patronym expected)"
		}
		else = { debug_log = "kehillah_patronym_debug.1: NO FATHER (no patronym expected)" }
		if = { limit = { has_trait = kohen } debug_log = "kehillah_patronym_debug.1: KOHEN" }
		if = { limit = { has_trait = levite } debug_log = "kehillah_patronym_debug.1: LEVITE" }
		if = { limit = { has_trait = ben_david } debug_log = "kehillah_patronym_debug.1: BEN/BAT DAVID" }
		if = {
			limit = { faith.religion != religion:judaism_religion }
			debug_log = "kehillah_patronym_debug.1: NON-JEWISH"
		}
		debug_log_scopes = yes
	}
}
