# ---------------------------------------------------------------------
# Study in the host community's library. Hidden, so the study effect is
# never built into a tooltip (kehillah_journey_study_work_effect's header
# has the 2026-10-21 lesson). Reuses the Learn Torah journey's per-work
# study dispatch, so mastery, sparks, languages and the place of learning
# are recorded exactly as on a journey.
# ---------------------------------------------------------------------
kehillah_bet_din_benches.0020 = {
	type = character_event
	hidden = yes
	immediate = {
		involved_activity ?= {
			activity_host ?= {
				primary_title ?= { save_scope_as = kehillah_journey_destination_title }
			}
		}
		if = {
			limit = {
				exists = scope:kehillah_journey_destination_title
				scope:kehillah_journey_destination_title = { has_variable_list = kehillah_library_works }
			}
			scope:kehillah_journey_destination_title = {
				random_in_list = {
					variable = kehillah_library_works
					limit = { NOT = { root = { is_target_in_variable_list = { name = kehillah_mastered_works target = prev } } } }
					save_scope_as = kbd_study_work
				}
				if = {
					limit = { NOT = { exists = scope:kbd_study_work } }
					random_in_list = {
						variable = kehillah_library_works
						save_scope_as = kbd_study_work
					}
				}
			}
			set_variable = { name = kehillah_journey_studied_work value = scope:kbd_study_work }
			set_variable = { name = kehillah_journey_work_loc value = scope:kbd_study_work }
			kehillah_journey_resolve_study_effect = yes
			remove_variable = kehillah_journey_studied_work
			if = {
				limit = { is_ai = no }
				trigger_event = kehillah_bet_din_benches.0021
			}
			else = { remove_variable = kehillah_journey_work_loc }
		}
		else = {
			add_learning_lifestyle_xp = kehillah_bet_din_rows_lifestyle_xp
			if = {
				limit = { is_ai = no }
				trigger_event = kehillah_bet_din_benches.0021
			}
		}
	}
}
