namespace = kehillah_crisis_recommend

# The counsel goes to the selected action's scene, not a standing category menu.
kehillah_crisis_recommend.0001 = {
 type = character_event
 title = kehillah_crisis_recommend.0001.t
 desc = {
  first_valid = {
   triggered_desc = {
    trigger = { has_variable = kehillah_recommend_count var:kehillah_recommend_count = 0 }
    desc = kehillah_crisis_recommend.0001.desc_empty
   }
   triggered_desc = {
    trigger = { primary_title = { has_variable = kehillah_crisis_armed_points var:kehillah_crisis_armed_points >= 70 } }
    desc = kehillah_crisis_recommend.0001.desc_urgent
   }
   triggered_desc = {
    trigger = { exists = scope:kehillah_recommend_dayyan }
    desc = kehillah_crisis_recommend.0001.desc_dayyan
   }
   desc = kehillah_crisis_recommend.0001.desc
  }
 }
 theme = stewardship
 left_portrait = { character = root animation = thinking }
 right_portrait = {
  trigger = { exists = scope:kehillah_recommend_dayyan }
  character = scope:kehillah_recommend_dayyan
  animation = personality_rational
 }
 trigger = { kehillah_crisis_warning_trigger = yes }
 immediate = {
  cp:councillor_court_chaplain ?= { save_scope_as = kehillah_recommend_dayyan }
  domicile.domicile_location.county.holder ?= { save_scope_as = kehillah_recommend_lord }
  kehillah_crisis_recommend_clear_effect = yes
  set_variable = { name = kehillah_recommend_count value = 0 }
  if = {
   limit = { primary_title = { has_variable = kehillah_crisis_armed_points var:kehillah_crisis_armed_points >= 70 } }
   if = { limit = { kehillah_crisis_recommend_fast_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = fast } }
   if = { limit = { kehillah_crisis_recommend_powerful_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = powerful } }
   if = { limit = { kehillah_crisis_recommend_answer_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = answer } }
   if = { limit = { kehillah_crisis_recommend_town_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = town } }
  }
  else = {
   if = { limit = { kehillah_crisis_recommend_fast_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = fast } }
   if = { limit = { kehillah_crisis_recommend_powerful_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = powerful } }
   if = { limit = { kehillah_crisis_recommend_answer_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = answer } }
   if = { limit = { kehillah_crisis_recommend_advocacy_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = advocacy } }
   if = { limit = { kehillah_crisis_recommend_watch_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = watch } }
   if = { limit = { kehillah_crisis_recommend_town_trigger = yes } kehillah_crisis_recommend_list_effect = { KEY = town } }
  }
 }
 option = {
  name = kehillah_crisis_recommend.0001.fast
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = fast } kehillah_crisis_recommend_fast_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.fast.tt
  hidden_effect = { trigger_event = kehillah_crisis_recommend.0101 }
 }
 option = {
  name = kehillah_crisis_recommend.0001.powerful
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = powerful } kehillah_crisis_recommend_powerful_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.powerful.tt
  hidden_effect = { kehillah_crisis_approach_lord_effect = { PATRON = scope:kehillah_recommend_lord } }
 }
 option = {
  name = kehillah_crisis_recommend.0001.answer
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = answer } kehillah_crisis_recommend_answer_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.answer.tt
  hidden_effect = { kehillah_crisis_answer_public_effect = yes }
 }
 option = {
  name = kehillah_crisis_recommend.0001.town
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = town } kehillah_crisis_recommend_town_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.town.tt
  hidden_effect = { trigger_event = kehillah_crisis_recommend.0104 }
 }
 option = {
  name = kehillah_crisis_recommend.0001.advocacy
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = advocacy } kehillah_crisis_recommend_advocacy_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.advocacy.tt
  hidden_effect = { trigger_event = kehillah_crisis_recommend.0102 }
 }
 option = {
  name = kehillah_crisis_recommend.0001.watch
  trigger = { kehillah_crisis_recommend_slot_trigger = { KEY = watch } kehillah_crisis_recommend_watch_trigger = yes }
  custom_tooltip = kehillah_crisis_recommend.0001.watch.tt
  hidden_effect = { trigger_event = kehillah_crisis_recommend.0103 }
 }
 option = { name = kehillah_crisis_recommend.0001.later }
 after = { kehillah_crisis_recommend_clear_effect = yes remove_variable = kehillah_recommend_open }
}
