# =====================================================================
# THE COUNCIL CONSIDERS THE JEWS -- V34 section 3, 2026-10-13. Drawn
# into an ecumenical council's doctrine debate (pam_ecumenical_council_
# debate_doctrines_events). The council's host or the head of faith
# decides; the AI by its outlook on the Jews. Lateran III (1179) and IV
# (1215), the Paris disputation (1240) -- or a reaffirmation of Sicut
# Judaeis.
# =====================================================================
namespace = kehillah_council

kehillah_council.0001 = {
	type = activity_event
	title = kehillah_council.0001.t
	desc = kehillah_council.0001.desc
	theme = faith
	left_portrait = {
		character = root
		animation = thinking
	}
	trigger = {
		exists = involved_activity
		OR = {
			involved_activity.activity_host = root
			faith.religious_head ?= root
		}
		faith.main_rite = {
			kehillah_rite_has_teaching_trigger = yes
			NOT = { has_variable = kehillah_teaching_change_cooldown }
		}
	}
	immediate = {
		faith.main_rite = { save_scope_as = kehillah_council_rite }
	}

	# Separate them, and send preachers among them -- or, further, deny
	# their books and their place.
	option = {
		name = kehillah_council.0001.a
		trigger = {
			faith.main_rite = {
				NOR = {
					rite_has_doctrine = kehillah_teaching_exclusion
					rite_has_doctrine = kehillah_teaching_abolition
				}
			}
		}
		faith.main_rite = {
			kehillah_rite_teaching_stricter_effect = yes
			kehillah_rite_teaching_changed_effect = yes
		}
		ai_chance = {
			base = 10
			modifier = { add = 60 kehillah_outlook_convinced_trigger = yes }
			modifier = { add = 20 has_trait = zealous }
			modifier = { factor = 0.1 kehillah_outlook_protector_trigger = yes }
		}
	}

	# Reaffirm the old protection (Sicut Judaeis).
	option = {
		name = kehillah_council.0001.b
		trigger = {
			faith.main_rite = {
				NOR = {
					rite_has_doctrine = kehillah_teaching_witness
					rite_has_doctrine = kehillah_teaching_imperial_law
					rite_has_doctrine = kehillah_teaching_dhimma
				}
			}
		}
		faith.main_rite = {
			kehillah_rite_teaching_gentler_effect = yes
			kehillah_rite_teaching_changed_effect = yes
		}
		ai_chance = {
			base = 10
			modifier = { add = 60 kehillah_outlook_protector_trigger = yes }
			modifier = { factor = 0.1 kehillah_outlook_convinced_trigger = yes }
		}
	}

	# Leave the matter where it stands.
	option = {
		name = kehillah_council.0001.c
		ai_chance = {
			base = 40
			modifier = { add = 30 kehillah_outlook_pragmatist_trigger = yes }
		}
	}
}
