# =====================================================================
# Jewish holiday flavor events.
#
# Fires from random_yearly_playable_pulse (see
# common/on_action/jewish_holiday_on_actions.txt) for any Jewish
# character, not only the Kehillah leader. Faith gating per holiday is
# explained in common/scripted_triggers/jewish_holiday_scripted_triggers.txt.
#
# TONE NOTE, carried over from kehillah_protection_events.txt because
# it applies just as much here: these are warm, funny, occasionally
# bittersweet domestic moments -- not host-relations drama. Hanukkah's
# "Light in the Window" event in particular is written as a personal,
# private choice about pride and comfort, with no host reaction modeled
# and no risk roll attached. The moment a holiday flavor event starts
# asking whether the neighbours are watching, Phase 4 (the host-realm
# threat layer) is being built early and badly, inside a file that is
# supposed to be able to fire for characters who have no host-relations
# system attached to them at all -- most of this pool.
#
# Every event resolves in one step: no chains, no state carried to next
# year. A holiday is a mood for a season, not a plot.
# =====================================================================

namespace = jewish_holidays

# ---------------------------------------------------------------------
# OPTION DESIGN (2026-10-19 rework). Each option pays in a different
# currency -- the warm memory (a pleasant evening), piety, prestige,
# relief from stress, a person's good opinion, lifestyle experience,
# gold -- and stress_impact makes each one the natural choice for some
# temperament. No option may be strictly better than another in the
# same event: before the rework the warm memory was stacked onto the
# "good" option, and in ten of thirteen events another option could
# never be worth taking (Daniel, on The Last Hour: "the first is just
# strictly better than the last").
# ---------------------------------------------------------------------

# ---------------------------------------------------------------------
# jewish_holidays.0001 -- AN OMEN IN THE HONEY
# Rosh Hashanah. Universal: is_jewish_character_trigger.
# ---------------------------------------------------------------------

jewish_holidays.0001 = {
	type = character_event
	title = jewish_holidays.0001.t
	desc = jewish_holidays.0001.desc
	theme = faith

	left_portrait = {
		character = root
		animation = prayer
	}

	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
	}

	# READ AN OMEN -- a good year: the warm memory. Hard for the
	# suspicious to believe.
	option = {
		name = jewish_holidays.0001.a

		add_character_modifier = {
			modifier = jewish_holiday_warm_memory_modifier
			months = 6
		}
		custom_tooltip = jewish_holidays.0001.a.tt
		stress_impact = {
			paranoid = minor_stress_impact_gain
			cynical = minor_stress_impact_gain
		}

		ai_chance = {
			base = 50
			ai_value_modifier = { ai_rationality = -0.5 }
		}
	}

	# READ IT THE OTHER WAY -- who shall live and who shall die: the
	# Days of Awe taken to heart. Piety; a comfort to the fearful.
	option = {
		name = jewish_holidays.0001.b

		add_piety = minor_piety_gain
		custom_tooltip = jewish_holidays.0001.b.tt
		stress_impact = {
			base = minor_stress_impact_gain
			paranoid = minor_stress_impact_loss
			craven = minor_stress_impact_loss
			zealous = minor_stress_impact_loss
		}

		ai_chance = { base = 25 }
	}

	# DISMISS IT -- "you shall not practise divination": the sages had no
	# use for omens. Learning; a relief to the rational.
	option = {
		name = jewish_holidays.0001.c

		add_learning_lifestyle_xp = minor_lifestyle_xp
		custom_tooltip = jewish_holidays.0001.c.tt
		stress_impact = {
			cynical = minor_stress_impact_loss
			paranoid = minor_stress_impact_gain
		}

		ai_chance = {
			base = 25
			ai_value_modifier = { ai_rationality = 0.5 }
		}
	}
}

# ---------------------------------------------------------------------
# jewish_holidays.0003 -- THE FOUR QUESTIONS
# Passover. Universal, but needs an actual child in the household to
# ask Mah Nishtana -- gated on that rather than on faith alone.
# ---------------------------------------------------------------------

jewish_holidays.0003 = {
	type = character_event
	title = jewish_holidays.0003.t
	desc = jewish_holidays.0003.desc
	theme = family

	left_portrait = {
		character = root
		animation = happiness
	}

	right_portrait = {
		character = scope:holiday_child
		animation = admiration
	}

	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
		# `is_child` isn't a real CK3 trigger (caught by ck3-tiger,
		# 2026-09-07). is_adult = no is the vanilla pattern.
		any_child = {
			is_adult = no
		}
	}

	immediate = {
		add_character_flag = { flag = jewish_holiday_recent_pesach days = 8 }
		random_child = {
			limit = { is_adult = no }
			save_scope_as = holiday_child
		}
	}

	# ANSWER PATIENTLY -- the child's affection, and a warm evening.
	option = {
		name = jewish_holidays.0003.a

		scope:holiday_child = {
			add_opinion = {
				modifier = helpful_opinion
				target = root
				opinion = 15
			}
		}
		add_character_modifier = {
			modifier = jewish_holiday_warm_memory_modifier
			months = 6
		}
		custom_tooltip = jewish_holidays.0003.a.tt
		stress_impact = {
			impatient = minor_stress_impact_gain
			patient = minor_stress_impact_loss
		}

		ai_chance = { base = 55 }
	}

	# TURN IT INTO A LESSON -- thorough, a little long: piety and
	# learning, and the child is less charmed.
	option = {
		name = jewish_holidays.0003.b

		add_piety = minor_piety_gain
		add_learning_lifestyle_xp = minor_lifestyle_xp
		custom_tooltip = jewish_holidays.0003.b.tt
		stress_impact = {
			lazy = minor_stress_impact_gain
			diligent = minor_stress_impact_loss
		}

		ai_chance = { base = 25 }
	}

	# WAVE IT OFF -- the food is hot and the day was long: rest, at the
	# child's expense.
	option = {
		name = jewish_holidays.0003.c

		scope:holiday_child = {
			add_opinion = {
				modifier = hurt_opinion
				target = root
				opinion = -10
			}
		}
		add_stress = minor_stress_loss
		custom_tooltip = jewish_holidays.0003.c.tt
		stress_impact = {
			impatient = minor_stress_impact_loss
			gluttonous = minor_stress_impact_loss
			compassionate = minor_stress_impact_gain
		}

		ai_chance = { base = 20 }
	}
}
