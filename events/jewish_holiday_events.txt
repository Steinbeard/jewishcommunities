# =====================================================================
# Jewish holiday flavor events.
#
# Fires from random_yearly_playable_pulse (see
# common/on_action/jewish_holiday_on_actions.txt) for any Jewish
# character, not only the Kehillah leader. Faith gating per holiday is
# explained in common/scripted_triggers/jewish_holiday_scripted_triggers.txt.
#
# TONE NOTE, carried over from kehillah_protection_events.txt because
# it applies just as much here: these are warm, funny, occasionally
# bittersweet domestic moments -- not host-relations drama. Hanukkah's
# "Light in the Window" event in particular is written as a personal,
# private choice about pride and comfort, with no host reaction modeled
# and no risk roll attached. The moment a holiday flavor event starts
# asking whether the neighbours are watching, Phase 4 (the host-realm
# threat layer) is being built early and badly, inside a file that is
# supposed to be able to fire for characters who have no host-relations
# system attached to them at all -- most of this pool.
#
# Every event resolves in one step: no chains, no state carried to next
# year. A holiday is a mood for a season, not a plot.
# =====================================================================

namespace = jewish_holidays

# ---------------------------------------------------------------------
# jewish_holidays.0001 -- AN OMEN IN THE HONEY
# Rosh Hashanah. Universal: is_jewish_character_trigger.
# ---------------------------------------------------------------------

jewish_holidays.0001 = {
	type = character_event
	title = jewish_holidays.0001.t
	desc = jewish_holidays.0001.desc
	theme = faith

	left_portrait = {
		character = root
		animation = prayer
	}

	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
	}

	# READ AN OMEN -- take the honey, the stumbled blessing, the
	# guttering candle as a sign the year ahead will be good.
	option = {
		name = jewish_holidays.0001.a

		add_character_modifier = {
			modifier = jewish_holiday_warm_memory_modifier
			months = 6
		}
		custom_tooltip = jewish_holidays.0001.a.tt

		ai_chance = { base = 50 }
	}

	# READ IT THE OTHER WAY -- best to go carefully this year.
	option = {
		name = jewish_holidays.0001.b

		add_stress = minor_stress_gain
		custom_tooltip = jewish_holidays.0001.b.tt

		ai_chance = { base = 25 }
	}

	# DISMISS IT -- a stumble over bread is a stumble over bread.
	option = {
		name = jewish_holidays.0001.c

		custom_tooltip = jewish_holidays.0001.c.tt

		ai_chance = { base = 25 }
	}
}

# ---------------------------------------------------------------------
# jewish_holidays.0003 -- THE FOUR QUESTIONS
# Passover. Universal, but needs an actual child in the household to
# ask Mah Nishtana -- gated on that rather than on faith alone.
# ---------------------------------------------------------------------

jewish_holidays.0003 = {
	type = character_event
	title = jewish_holidays.0003.t
	desc = jewish_holidays.0003.desc
	theme = family

	left_portrait = {
		character = root
		animation = happiness
	}

	right_portrait = {
		character = scope:holiday_child
		animation = admiration
	}

	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
		# `is_child` isn't a real CK3 trigger (caught by ck3-tiger,
		# 2026-09-07; this was the known, previously-unfixed bug flagged
		# in the 2026-09-07 localization patch pass and wave3 runbook).
		# is_adult = no is the actual vanilla pattern for "not yet an
		# adult" -- see e.g. events/activities/pilgrimage_activity/
		# pilgrimage_events.txt for the same usage.
		any_child = {
			is_adult = no
		}
	}

	immediate = {
		random_child = {
			limit = { is_adult = no }
			save_scope_as = holiday_child
		}
	}

	# ANSWER PATIENTLY -- the whole point of asking.
	option = {
		name = jewish_holidays.0003.a

		scope:holiday_child = {
			add_opinion = {
				modifier = helpful_opinion
				target = root
				opinion = 15
			}
		}
		add_character_modifier = {
			modifier = jewish_holiday_warm_memory_modifier
			months = 6
		}
		custom_tooltip = jewish_holidays.0003.a.tt

		ai_chance = { base = 55 }
	}

	# TURN IT INTO A LESSON -- thorough, a little long, well meant.
	option = {
		name = jewish_holidays.0003.b

		add_piety = minor_piety_gain
		custom_tooltip = jewish_holidays.0003.b.tt

		ai_chance = { base = 25 }
	}

	# WAVE IT OFF -- there is food getting cold.
	option = {
		name = jewish_holidays.0003.c

		scope:holiday_child = {
			add_opinion = {
				modifier = hurt_opinion
				target = root
				opinion = -10
			}
		}
		add_character_modifier = {
			modifier = jewish_holiday_awkward_memory_modifier
			months = 6
		}
		custom_tooltip = jewish_holidays.0003.c.tt

		ai_chance = { base = 20 }
	}
}
