# Realm view of Jewish communities (2026-10-28). Console-only probes,
# never subscribed to an on_action. Run as a Kehillah leader (or with
# one as the event's root via ck3_script):
#   .1  records one piece of each news kind for this community, so the
#       feed can be read without waiting for incidents.
#   .2  builds the view onto this character as if they had pressed the
#       chip on their own window, and logs what was built.
#   .3  clears this realm's whole news feed.
# docs/spec/2026-10-28-realm-jewish-communities-view.md
namespace = kehillah_realm_view_debug

kehillah_realm_view_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_realm_news_add_effect = { KIND = talmud_trial }
		kehillah_realm_news_add_effect = { KIND = badge }
		kehillah_realm_news_add_effect = { KIND = holy_week }
		kehillah_realm_news_add_effect = { KIND = bonds }
		kehillah_realm_news_add_effect = { KIND = servants_decree }
		kehillah_realm_news_add_effect = { KIND = mob_preaching }
		kehillah_realm_news_add_effect = { KIND = pogrom }
		kehillah_realm_news_add_effect = { KIND = accusation_riot }
		kehillah_realm_news_add_effect = { KIND = anusim_discovered }
		kehillah_realm_news_add_effect = { KIND = popular_rising }
		debug_log = "kehillah_realm_view_debug.1: ten news entries recorded"
	}
}

kehillah_realm_view_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		save_scope_as = target
		kehillah_realm_view_build_effect = yes
		save_scope_value_as = { name = kehillah_dbg_rv_count value = var:kehillah_rv_count }
		save_scope_value_as = { name = kehillah_dbg_rv_danger value = var:kehillah_rv_danger_count }
		save_scope_value_as = { name = kehillah_dbg_rv_news value = var:kehillah_rv_news_count }
		var:kehillah_rv_lord = { save_scope_as = kehillah_dbg_rv_lord }
		var:kehillah_rv_top = { save_scope_as = kehillah_dbg_rv_top }
		var:kehillah_rv_worst ?= { save_scope_as = kehillah_dbg_rv_worst }
		debug_log_scopes = yes
		debug_log = "kehillah_realm_view_debug.2: realm view built"
		every_in_list = {
			variable = kehillah_rv_list
			debug_log_scopes = yes
			debug_log = "kehillah_realm_view_debug.2: row"
		}
	}
}

kehillah_realm_view_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		domicile.domicile_location.county.holder.top_liege.primary_title ?= {
			kehillah_rv_debug_clear_slot_effect = { N = 1 }
			kehillah_rv_debug_clear_slot_effect = { N = 2 }
			kehillah_rv_debug_clear_slot_effect = { N = 3 }
			kehillah_rv_debug_clear_slot_effect = { N = 4 }
			kehillah_rv_debug_clear_slot_effect = { N = 5 }
			kehillah_rv_debug_clear_slot_effect = { N = 6 }
			kehillah_rv_debug_clear_slot_effect = { N = 7 }
			kehillah_rv_debug_clear_slot_effect = { N = 8 }
			kehillah_rv_debug_clear_slot_effect = { N = 9 }
			kehillah_rv_debug_clear_slot_effect = { N = 10 }
			kehillah_rv_debug_clear_slot_effect = { N = 11 }
			kehillah_rv_debug_clear_slot_effect = { N = 12 }
			kehillah_rv_debug_clear_slot_effect = { N = 13 }
			kehillah_rv_debug_clear_slot_effect = { N = 14 }
			kehillah_rv_debug_clear_slot_effect = { N = 15 }
		}
		debug_log = "kehillah_realm_view_debug.3: realm news cleared"
	}
}
