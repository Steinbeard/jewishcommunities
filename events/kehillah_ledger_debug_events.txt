# Console-only ledger probe: event kehillah_ledger_debug.1
# Rebuilds the ledger and logs dated/ranked counts and the resulting flags.
namespace = kehillah_ledger_debug

kehillah_ledger_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_ledger_refresh_rites_effect = yes
		remove_variable = kehillah_ledger_by_name
		remove_variable = kehillah_ledger_by_rank
		kehillah_ledger_rebuild_effect = yes
		debug_log = "LEDGER dated count: [ROOT.Var('kehillah_lib_catalog_count').GetValue]"
		set_variable = kehillah_ledger_by_rank
		kehillah_ledger_rebuild_effect = yes
		set_variable = { name = kehillah_ledger_probe_rows value = 0 }
		every_in_list = {
			variable = kehillah_lib_catalog_works
			root = { change_variable = { name = kehillah_ledger_probe_rows add = 1 } }
			debug_log = "LEDGER ranked row"
			debug_log_scopes = yes
		}
		debug_log = "LEDGER ranked count: [ROOT.Var('kehillah_ledger_probe_rows').GetValue]"
		remove_variable = kehillah_ledger_probe_rows
		debug_log = "LEDGER rank probe END"
	}
}
