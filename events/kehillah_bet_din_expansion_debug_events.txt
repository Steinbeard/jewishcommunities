namespace = kehillah_bet_din_expansion_debug

# Console only: event kehillah_bet_din_expansion_debug.1
# Reports the active case and its result state without changing a hearing.
kehillah_bet_din_expansion_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { exists = involved_activity }
			debug_log = "kehillah_bet_din_expansion_debug: active activity found"
			if = { limit = { exists = involved_activity.var:kehillah_din_case involved_activity.var:kehillah_din_case = 10 } debug_log = "kehillah_bet_din_expansion_debug: schoolroom active" }
			if = { limit = { exists = involved_activity.var:kehillah_din_case involved_activity.var:kehillah_din_case = 11 } debug_log = "kehillah_bet_din_expansion_debug: butcher active" }
			if = { limit = { exists = involved_activity.var:kehillah_din_case involved_activity.var:kehillah_din_case = 12 } debug_log = "kehillah_bet_din_expansion_debug: Shabbat passage active" }
			if = { limit = { exists = involved_activity.var:kehillah_din_case involved_activity.var:kehillah_din_case = 13 } debug_log = "kehillah_bet_din_expansion_debug: marked purse active" }
			if = { limit = { exists = involved_activity.var:kehillah_bet_din_tally } debug_log = "kehillah_bet_din_expansion_debug: tally is set" }
			else = { debug_log = "kehillah_bet_din_expansion_debug: tally is not set" }
		}
		else = { debug_log = "kehillah_bet_din_expansion_debug: no active activity" }
	}
}
