# Console-only probes for natural founding (docs/spec/v34-outlook-
# teachings-and-founding.md section 1). Fire on anyone:
#   event kehillah_founding_debug.1  -- found a community now with a spawned founder
#   event kehillah_founding_debug.2  -- root sends out settlers now (skips the decision's gates)
namespace = kehillah_founding_debug

kehillah_founding_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_founding_debug.1: natural founding now"
		kehillah_natural_founding_effect = yes
	}
}

kehillah_founding_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_founding_debug.2: send out settlers now"
		trigger_event = kehillah_settlers.0004
	}
}

# Console: event kehillah_founding_debug.3 -- for every community, each
# gate of Send Out Settlers (kehillah_settlers_decisions.txt) and the AI's
# chance. Changes nothing. (2026-10-13: AI communities founded nothing in
# 14 observed years.)
kehillah_founding_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_founding_debug.3: settlers gates -- START"
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { exists = holder }
			holder = {
				save_scope_as = kehillah_mother_leader
				save_scope_value_as = { name = kehillah_dbg_courtiers value = kehillah_courtier_count_value }
				save_scope_value_as = { name = kehillah_dbg_capacity value = kehillah_courtier_capacity_value }
				save_scope_value_as = { name = kehillah_dbg_overflow value = kehillah_settlers_overflow_value }
				save_scope_value_as = { name = kehillah_dbg_gold value = gold }
				save_scope_value_as = { name = kehillah_dbg_cost value = medium_gold_value }
				debug_log_scopes = yes
				if = {
					limit = { is_ai = yes }
					debug_log = "kehillah_founding_debug.3: gate ai yes"
				}
				if = {
					limit = { NOT = { kehillah_quarter_has_room_trigger = yes } }
					debug_log = "kehillah_founding_debug.3: gate crowded yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate crowded NO"
				}
				if = {
					limit = { any_county = { kehillah_settlers_invited_destination_trigger = yes } }
					debug_log = "kehillah_founding_debug.3: gate invited destination yes"
				}
				if = {
					limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_strained_threshold } }
					debug_log = "kehillah_founding_debug.3: gate prosperity yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate prosperity NO"
				}
				if = {
					limit = { primary_title = { has_variable = kehillah_settlers_cooldown } }
					debug_log = "kehillah_founding_debug.3: gate cooldown RUNNING"
				}
				if = {
					limit = { any_courtier = { kehillah_settlers_leader_candidate_trigger = yes } }
					debug_log = "kehillah_founding_debug.3: gate leader yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate leader NO"
				}
				if = {
					limit = { any_county = { kehillah_settlers_destination_trigger = yes } }
					debug_log = "kehillah_founding_debug.3: gate destination yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate destination NO"
				}
				if = {
					limit = { gold >= medium_gold_value }
					debug_log = "kehillah_founding_debug.3: gate gold yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate gold NO"
				}
				if = {
					limit = { is_available_adult = yes }
					debug_log = "kehillah_founding_debug.3: gate available yes"
				}
				else = {
					debug_log = "kehillah_founding_debug.3: gate available NO"
				}
				debug_log = "kehillah_founding_debug.3: -- community end"
			}
		}
		debug_log = "kehillah_founding_debug.3: settlers gates -- END"
	}
}
