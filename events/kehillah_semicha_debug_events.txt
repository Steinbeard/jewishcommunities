# Console-only probes for ordination / semicha (V21 slice 3); never
# called by gameplay. Play or select a character, then:
#   event kehillah_semicha_debug.1  -- READ-ONLY: candidate, curriculum,
#                                      institution, readiness
#   event kehillah_semicha_debug.2  -- make root an unordained candidate:
#                                      strip the Rabbi trait, raise
#                                      Learning to the floor if needed
#   event kehillah_semicha_debug.3  -- grant the curriculum directly (two
#                                      works, two fields), then report
#   event kehillah_semicha_debug.4  -- build a Beit Midrash on root's own
#                                      quarter, then report
#
# The usual shortest path to testing Seek Semicha on the Worms start, as
# the Kehillah leader: .2, then .3, then .4 if the quarter has no study
# hall yet, then open Decisions. .1 after each to see what moved.
#
# .2/.3/.4 all CHANGE STATE (a trait, a skill, variable lists, a domicile
# building). Use them on a throwaway run.

namespace = kehillah_semicha_debug

kehillah_semicha_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.1: ordination state -- START"
		debug_log_scopes = yes

		if = {
			limit = { has_trait = kehillah_rabbi_trait }
			debug_log = "kehillah_semicha_debug.1: ALREADY ORDAINED has kehillah_rabbi_trait (Seek Semicha is hidden for him)"
		}
		else = { debug_log = "kehillah_semicha_debug.1: UNORDAINED no kehillah_rabbi_trait" }

		if = {
			limit = { is_rabbinic_authority_jewish_trigger = yes }
			debug_log = "kehillah_semicha_debug.1: FAITH PASS rabbinic-authority Jewish"
		}
		else = { debug_log = "kehillah_semicha_debug.1: FAITH FAIL not a rabbinic-authority Jewish faith" }

		if = {
			limit = { kehillah_leadership_gender_eligible_trigger = yes }
			debug_log = "kehillah_semicha_debug.1: GENDER PASS eligible under the current leadership rule"
		}
		else = { debug_log = "kehillah_semicha_debug.1: GENDER FAIL not eligible under the current leadership rule" }

		if = {
			limit = { learning >= kehillah_ordination_learning_floor_value }
			debug_log = "kehillah_semicha_debug.1: LEARNING PASS at or above the ordination floor"
		}
		else_if = {
			limit = { learning >= kehillah_ordination_foundation_learning_floor_value }
			debug_log = "kehillah_semicha_debug.1: LEARNING FOUNDATION BAND only (6-7), below the floor to ordain"
		}
		else = { debug_log = "kehillah_semicha_debug.1: LEARNING FAIL below the foundation band" }

		if = {
			limit = { kehillah_rabbi_ordination_candidate_trigger = yes }
			debug_log = "kehillah_semicha_debug.1: CANDIDATE PASS may enter the ordination path"
		}
		else = { debug_log = "kehillah_semicha_debug.1: CANDIDATE FAIL may not enter the ordination path" }

		if = {
			limit = { kehillah_rabbi_study_eligible_trigger = yes }
			debug_log = "kehillah_semicha_debug.1: STUDY PASS Learn Torah's widened gate admits him"
		}
		else = { debug_log = "kehillah_semicha_debug.1: STUDY FAIL Learn Torah's gate refuses him" }

		if = {
			limit = {
				has_variable_list = kehillah_mastered_works
				variable_list_size = {
					name = kehillah_mastered_works
					value >= kehillah_ordination_works_required_value
				}
			}
			debug_log = "kehillah_semicha_debug.1: WORKS PASS enough distinct mastered works"
		}
		else = { debug_log = "kehillah_semicha_debug.1: WORKS FAIL not enough distinct mastered works" }

		if = {
			limit = {
				has_variable_list = kehillah_apprentice_fields
				variable_list_size = {
					name = kehillah_apprentice_fields
					value >= kehillah_ordination_fields_required_value
				}
			}
			debug_log = "kehillah_semicha_debug.1: FIELDS PASS enough distinct study fields"
		}
		else = { debug_log = "kehillah_semicha_debug.1: FIELDS FAIL not enough distinct study fields" }

		if = {
			limit = { kehillah_semicha_institution_available_trigger = { CHARACTER = root } }
			debug_log = "kehillah_semicha_debug.1: INSTITUTION PASS a Beit Midrash within reach"
		}
		else = { debug_log = "kehillah_semicha_debug.1: INSTITUTION FAIL no Beit Midrash within reach" }

		# The one that decides whether Seek Semicha is selectable, given
		# the institution above.
		if = {
			limit = { kehillah_rabbi_ordination_ready_trigger = yes }
			debug_log = "kehillah_semicha_debug.1: READY PASS Seek Semicha should be available"
		}
		else = { debug_log = "kehillah_semicha_debug.1: READY FAIL Seek Semicha should be greyed out" }

		kehillah_semicha_resolve_institution_effect = { CHARACTER = root }
		if = {
			limit = { exists = scope:kehillah_semicha_community }
			debug_log = "kehillah_semicha_debug.1: resolved institution community (scopes dump follows)"
			scope:kehillah_semicha_community = { debug_log_scopes = yes }
		}
		else = { debug_log = "kehillah_semicha_debug.1: no institution community resolved" }

		debug_log = "kehillah_semicha_debug.1: ordination state -- END"
	}
}

kehillah_semicha_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.2: make root an unordained candidate -- START"
		if = {
			limit = { has_trait = kehillah_rabbi_trait }
			remove_trait = kehillah_rabbi_trait
			debug_log = "kehillah_semicha_debug.2: removed the Rabbi trait"
		}
		if = {
			limit = { learning < kehillah_ordination_learning_floor_value }
			add_learning_skill = 10
			debug_log = "kehillah_semicha_debug.2: added 10 Learning to clear the floor"
		}
		trigger_event = kehillah_semicha_debug.1
		debug_log = "kehillah_semicha_debug.2: make root an unordained candidate -- END"
	}
}

kehillah_semicha_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.3: grant the curriculum -- START"

		# Two real works from two different fields, matching what
		# kehillah_study_specific_work_effect would have recorded:
		# Targum Onkelos (parshanut) and the Mishnah (talmudics).
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah } } }
			add_to_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_apprentice_fields target = flag:parshanut } } }
			add_to_variable_list = { name = kehillah_apprentice_fields target = flag:parshanut }
		}
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = kehillah_apprentice_fields target = flag:talmudics } } }
			add_to_variable_list = { name = kehillah_apprentice_fields target = flag:talmudics }
		}

		trigger_event = kehillah_semicha_debug.1
		debug_log = "kehillah_semicha_debug.3: grant the curriculum -- END"
	}
}

kehillah_semicha_debug.4 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.4: build a study hall -- START"
		if = {
			limit = { exists = domicile }
			domicile = {
				if = {
					limit = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_01 } }
					add_domicile_building = kehillah_beit_midrash_01
					# FOUND LIVE 2026-10-10 (Shabbat heartbeat 4): this used to
					# log "added kehillah_beit_midrash_01" right here,
					# unconditionally, and it was LYING. add_domicile_building
					# consults the building's own can_construct, and
					# kehillah_beit_midrash_01's can_construct requires
					# kehillah_charter_allows_new_communal_building_trigger --
					# which is FALSE on the stock Worms 1066 start, because the
					# community sits under kehillah_christian_host_charter_
					# prototype with no construction right cached yet (V16
					# regional charters; the gate fails closed on purpose).
					#
					# So the engine threw "Domicile owner failed to meet
					# triggered requirements" into error.log, nothing was built,
					# and this probe reported success. A later .8 run is what
					# exposed it, by still reporting every branch FAIL after a
					# supposedly successful rebuild.
					#
					# The shim guide's "What ck3-tiger does NOT catch" section had
					# already written this exact trap down -- add_domicile_building
					# consults can_construct and tiger cannot simulate it, and the
					# Worms start had silently been a no-op above synagogue tier 1
					# for an unknown length of time. Second time this shape of bug
					# has bitten; hence an assertion rather than a log line.
					if = {
						limit = { has_domicile_building_or_higher = kehillah_beit_midrash_01 }
						debug_log = "kehillah_semicha_debug.4: ADDED kehillah_beit_midrash_01, and confirmed present"
					}
					else = {
						debug_log = "kehillah_semicha_debug.4: FAILED to add kehillah_beit_midrash_01 -- can_construct refused it; see error.log. Nothing was built"
						# root, not a domicile -> owner link: this is root's own
						# domicile (the enclosing block is `domicile = { ... }`
						# on root), so the owner is root by construction. Said
						# explicitly because `domicile_owner` looks like the
						# natural link and is NOT one -- ck3-tiger flagged it as
						# an unknown token and tiger was RIGHT. The vanilla hits
						# that seemed to confirm it were `nearby_domicile_owners`
						# and `top_realm_domicile_owners`, which are
						# `ai_recipients` VALUES in character_interactions, not a
						# scope link. A substring match is not a verification.
						if = {
							limit = { root = { kehillah_charter_allows_new_communal_building_trigger = yes } }
							debug_log = "kehillah_semicha_debug.4:   the charter DOES allow new communal buildings, so the refusal is something else (slots? cost?)"
						}
						else = {
							debug_log = "kehillah_semicha_debug.4:   CAUSE: the host charter does not (yet) allow new communal buildings. This is correct V16 behaviour, not a bug -- a study hall cannot be conjured under an unresolved Christian host charter"
						}
					}
				}
				else = { debug_log = "kehillah_semicha_debug.4: a Beit Midrash is already here" }
			}
		}
		else = { debug_log = "kehillah_semicha_debug.4: BLOCKED root has no domicile" }
		trigger_event = kehillah_semicha_debug.1
		debug_log = "kehillah_semicha_debug.4: build a study hall -- END"
	}
}


# ---------------------------------------------------------------------
# V21 slice 4 (the rabbinic apprenticeship), added 2026-10-10.
#   event kehillah_semicha_debug.5  -- READ-ONLY: guardian, accumulated
#                                      years, credit, endorsement
#   event kehillah_semicha_debug.6  -- grant a completed apprenticeship
#                                      directly (sets the years counter
#                                      to the required number), then
#                                      report
#   event kehillah_semicha_debug.11 -- the same three credit bands, but
#                                      on a subject already IN each band;
#                                      use this one, not .10, unless root
#                                      has no Learning modifiers
#   event kehillah_semicha_debug.10 -- drive Learning through all three
#                                      credit bands and check which
#                                      credit each one awards
#   event kehillah_semicha_debug.7  -- run the coming-of-age sorter
#                                      (kehillah_semicha.0002) NOW,
#                                      without waiting for a birthday
#
# The ordinary way to test slice 4 without waiting ten in-game years, on
# a child: .6, then .7. On an adult, .6 then .7 also works -- the sorter
# itself does not check age, only the records.
#
# .6 CHANGES STATE. .7 can grant the Rabbi trait, which is the point.
# ---------------------------------------------------------------------

kehillah_semicha_debug.5 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.5: apprenticeship state -- START"
		debug_log_scopes = yes

		if = {
			limit = {
				any_relation = {
					type = guardian
					always = yes
				}
			}
			debug_log = "kehillah_semicha_debug.5: HAS A GUARDIAN relation"
			random_relation = {
				type = guardian
				debug_log_scopes = yes
				if = {
					limit = { kehillah_rabbinic_guardian_trigger = yes }
					debug_log = "kehillah_semicha_debug.5: GUARDIAN QUALIFIES rabbi, rabbinic-authority, gender-eligible"
				}
				else = { debug_log = "kehillah_semicha_debug.5: GUARDIAN DOES NOT QUALIFY" }
			}
		}
		else = { debug_log = "kehillah_semicha_debug.5: NO GUARDIAN relation at all" }

		if = {
			limit = { kehillah_has_rabbinic_guardian_trigger = yes }
			debug_log = "kehillah_semicha_debug.5: QUALIFYING GUARDIAN YES a birthday now would add a year"
		}
		else = { debug_log = "kehillah_semicha_debug.5: QUALIFYING GUARDIAN NO a birthday now would add nothing" }

		if = {
			limit = { has_variable = kehillah_apprentice_rabbi_years }
			debug_log = "kehillah_semicha_debug.5: years record EXISTS"
			if = {
				limit = { var:kehillah_apprentice_rabbi_years >= kehillah_ordination_apprenticeship_years_value }
				debug_log = "kehillah_semicha_debug.5: APPRENTICESHIP COMPLETE at or above the required years"
			}
			else = { debug_log = "kehillah_semicha_debug.5: APPRENTICESHIP INCOMPLETE below the required years" }
		}
		else = { debug_log = "kehillah_semicha_debug.5: no years record -- never had a qualifying guardian on a birthday" }

		if = {
			limit = { has_variable = kehillah_apprenticeship_curriculum }
			debug_log = "kehillah_semicha_debug.5: FULL CURRICULUM CREDIT held (Learning 8+ band at adulthood)"
		}
		else = { debug_log = "kehillah_semicha_debug.5: no full curriculum credit" }

		if = {
			limit = { is_target_in_variable_list = { name = kehillah_apprentice_fields target = flag:rabbinic_apprenticeship } }
			debug_log = "kehillah_semicha_debug.5: FOUNDATION FIELD CREDIT held (Learning 6-7 band at adulthood)"
		}
		else = { debug_log = "kehillah_semicha_debug.5: no foundation field credit" }

		if = {
			limit = { kehillah_rabbi_ordination_curriculum_met_trigger = yes }
			debug_log = "kehillah_semicha_debug.5: CURRICULUM MET by one route or the other"
		}
		else = { debug_log = "kehillah_semicha_debug.5: CURRICULUM NOT MET" }

		if = {
			limit = {
				any_relation = {
					type = guardian
					kehillah_rabbinic_guardian_trigger = yes
					kehillah_mentor_endorses_trigger = { PUPIL = root }
				}
			}
			debug_log = "kehillah_semicha_debug.5: ENDORSEMENT AVAILABLE a qualifying guardian would endorse"
		}
		else = { debug_log = "kehillah_semicha_debug.5: NO ENDORSEMENT no qualifying guardian willing (opinion <= -50, or cynical+arbitrary, or none qualifies)" }

		debug_log = "kehillah_semicha_debug.5: apprenticeship state -- END"
	}
}

# V21 section 7 gaps -- institution branch coverage and losing the study
# hall, added 2026-10-10 (Shabbat heartbeat 4).
#
#   event kehillah_semicha_debug.8  -- READ-ONLY. Runs kehillah_semicha_
#                                      institution_available_trigger
#                                      against one subject per branch and
#                                      reports each verdict, with enough
#                                      detail to tell a real FAIL from a
#                                      correct one.
#   event kehillah_semicha_debug.9  -- CHANGES STATE. Destroys the
#                                      community's Beit Midrash, whatever
#                                      tier it is, then re-runs .8.
#                                      **OFTEN A ONE-WAY DOOR** -- .4 can
#                                      only rebuild where the host charter
#                                      allows new communal buildings, which
#                                      the stock Worms 1066 start does not.
#                                      .9 reports REVERSIBLE or WARNING,
#                                      ONE-WAY before it removes anything.
#                                      Reload rather than relying on .4.
#
# Why these exist: V21 section 7 lists the landed-ruler branch, the
# landless-adventurer branch, and losing access to the Beit Midrash as
# never tested. Heartbeat 4 is also the session that found the COURTIER
# branch had been false for every courtier in the game, because
# government_has_flag is true for a ruler's whole court (implementation
# doc section 14). So "this branch has never been exercised" is a
# statement with a track record behind it here, not a formality -- two of
# the three remaining branches have never had a single live verdict.
#
# Note the institution trigger carries NO faith gate of its own (faith
# lives in kehillah_rabbi_ordination_candidate_trigger), which is what
# makes the host county's Christian holder a legitimate subject for the
# MECHANICS of branch 2. It is not asserting that a Christian count may
# be ordained -- the candidate trigger refuses him long before this.
# ---------------------------------------------------------------------

kehillah_semicha_debug.8 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_semicha_debug.8: institution branch coverage -- START"

		# The community's own province, which every branch below compares
		# against. Guarded: a chained reference through an unset domicile
		# is the ~990-errors-a-load mistake recorded in the library
		# trigger's header.
		if = {
			limit = { exists = domicile.domicile_location }
			domicile.domicile_location = { save_scope_as = kds8_here }
			debug_log = "kehillah_semicha_debug.8: community province (scopes dump follows)"
			domicile.domicile_location = { debug_log_scopes = yes }
		}
		else = { debug_log = "kehillah_semicha_debug.8: BLOCKED root has no domicile_location -- run this as a Kehillah leader" }

		# --- BRANCH 1: the leader's own quarter. Already proven live, kept
		# here as the control: if this one FAILs while a study hall is
		# standing, the probe itself or the start is wrong, not the branch
		# under test.
		#
		# WORDING FIX 2026-10-10, same session: the FAIL branch used to say
		# "CONTROL BROKEN, distrust the rest of this run" unconditionally,
		# which is actively misleading when .9 has just demolished the study
		# hall on purpose -- that FAIL is the expected result, and the log
		# was telling the reader to throw out a passing test. The two cases
		# are now distinguished by the one thing that separates them.
		if = {
			limit = { kehillah_semicha_institution_available_trigger = { CHARACTER = root } }
			debug_log = "kehillah_semicha_debug.8: BRANCH 1 PASS the leader's own quarter (control)"
		}
		else_if = {
			limit = {
				OR = {
					NOT = { exists = domicile }
					domicile = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_01 } }
				}
			}
			debug_log = "kehillah_semicha_debug.8: BRANCH 1 FAIL the leader's own quarter -- EXPECTED, there is no Beit Midrash here right now (e.g. .9 has just removed it). Not a control break"
		}
		else = { debug_log = "kehillah_semicha_debug.8: BRANCH 1 FAIL the leader's own quarter WITH a Beit Midrash standing -- CONTROL BROKEN, distrust the rest of this run" }

		# --- BRANCH 2: a landed ruler whose capital province holds a
		# community. Subject derived from the community's own province
		# rather than hardcoding c_worms, so this probe keeps working at
		# any start and in any community.
		if = {
			limit = { exists = scope:kds8_here.county.holder }
			scope:kds8_here.county.holder = { save_scope_as = kds8_landed }
			debug_log = "kehillah_semicha_debug.8: BRANCH 2 subject is the host county's holder (scopes dump follows)"
			scope:kds8_landed = {
				debug_log_scopes = yes
				if = {
					limit = { is_ruler = yes }
					debug_log = "kehillah_semicha_debug.8:   is_ruler YES"
				}
				else = { debug_log = "kehillah_semicha_debug.8:   is_ruler NO" }
				if = {
					limit = { exists = capital_province }
					debug_log = "kehillah_semicha_debug.8:   HAS a capital_province"
					if = {
						limit = { capital_province = scope:kds8_here }
						debug_log = "kehillah_semicha_debug.8:   capital_province IS the community's province (expect BRANCH 2 PASS)"
					}
					else = { debug_log = "kehillah_semicha_debug.8:   capital_province is ELSEWHERE -- e.g. a duke holding this county from another seat. A BRANCH 2 FAIL below is then CORRECT, not a bug" }
				}
				else = { debug_log = "kehillah_semicha_debug.8:   has NO capital_province -- a BRANCH 2 FAIL below is CORRECT" }
			}
			if = {
				limit = { kehillah_semicha_institution_available_trigger = { CHARACTER = scope:kds8_landed } }
				debug_log = "kehillah_semicha_debug.8: BRANCH 2 PASS a landed ruler at the community's province"
			}
			else = { debug_log = "kehillah_semicha_debug.8: BRANCH 2 FAIL a landed ruler at the community's province -- read the three lines above before calling it a bug" }
		}
		else = { debug_log = "kehillah_semicha_debug.8: BRANCH 2 BLOCKED the community's province has no county holder" }

		# --- BRANCH 2b: the same branch, but hunting a subject that SHOULD
		# pass, anywhere in the world.
		#
		# ADDED 2026-10-10, after 2's first live verdict. At Worms the host
		# county's holder rules from another seat, so branch 2 returned FAIL
		# -- correctly, but a correct FAIL is a negative control, not proof
		# the branch can ever say yes. With a dozen-plus registered
		# communities, somewhere there is probably a count actually seated in
		# the province their community occupies; that character is the
		# positive test.
		#
		# Scanning for the subject instead of hardcoding one keeps this
		# honest: if nobody in the world qualifies, the probe says NOT
		# OBSERVED rather than manufacturing a ruler, exactly as 3b does.
		# Written as an effect-side scan with save_temporary_scope_as rather
		# than a `limit` that passes the character into the trigger. Textual
		# substitution of `this` into kehillah_semicha_institution_available_
		# trigger would re-bind inside its own any_in_global_list / holder
		# blocks and silently compare the wrong characters -- the same class
		# of scope mistake this trigger has already produced once today.
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { exists = domicile.domicile_location }
			}
