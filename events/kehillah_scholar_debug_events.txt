namespace = kehillah_scholar_debug

# Console only. Use a disposable Jewish adult with neither trait.
kehillah_scholar_debug.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		faith.religion = religion:judaism_religion
		age >= 16
		NOR = {
			has_trait = kehillah_rabbi_trait
			has_trait = lifestyle_scholar
		}
	}

	immediate = {
		add_trait = lifestyle_scholar
		add_trait_xp = { trait = lifestyle_scholar value = 83 }
		add_trait = kehillah_rabbi_trait
		trigger_event = { id = kehillah_scholar_debug.0002 days = 1 }
	}
}

kehillah_scholar_debug.0002 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_trait = lifestyle_scholar }
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = 20 }
				has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 20 }
				has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = 43 }
			}
			debug_log = "kehillah_scholar: 83 Scholar XP transferred and trait removed"
		}
		else = {
			debug_log = "kehillah_scholar: unexpected ordination transfer result"
		}
	}
}
