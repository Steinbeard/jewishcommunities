namespace = kehillah_mutual_aid
# =====================================================================
# MUTUAL AID (2026-11-20). common/scripted_effects/kehillah_mutual_aid_effects.txt.
#   .0001 An emissary from the Land of Israel (a shadar) asks for the
#         communities there. The standing need: no deadline, no list of
#         who was expected; the answer goes straight to the reputation.
# =====================================================================

kehillah_mutual_aid.0001 = {
	type = character_event
	title = kehillah_mutual_aid.0001.t
	desc = kehillah_mutual_aid.0001.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_rational
	}

	trigger = {
		exists = primary_title
		is_imprisoned = no
		NOT = { has_variable = kehillah_emissary_cooldown }
	}

	immediate = {
		set_variable = { name = kehillah_emissary_cooldown value = yes years = 3 }
	}

	# A generous gift.
	option = {
		name = kehillah_mutual_aid.0001.a
		trigger = { gold >= kehillah_emissary_generous_gold }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = kehillah_emissary_generous_gold
		add_piety = medium_piety_gain
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_aid_full_greatness }
		custom_tooltip = kehillah_giving_rep_up_tt
		hidden_effect = { kehillah_giving_rep_change_effect = { AMOUNT = kehillah_giving_rep_emissary_generous } }
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = generous }
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = -30 has_trait = greedy }
		}
	}

	# What the community can spare.
	option = {
		name = kehillah_mutual_aid.0001.b
		trigger = { gold >= kehillah_emissary_modest_gold }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = kehillah_emissary_modest_gold
		add_piety = minor_piety_gain
		custom_tooltip = kehillah_giving_rep_up_tt
		hidden_effect = { kehillah_giving_rep_change_effect = { AMOUNT = kehillah_giving_rep_emissary_modest } }
		ai_chance = { base = 50 }
	}

	# Nothing this year.
	option = {
		name = kehillah_mutual_aid.0001.c
		custom_tooltip = kehillah_giving_rep_down_tt
		hidden_effect = { kehillah_giving_rep_change_effect = { AMOUNT = kehillah_giving_rep_emissary_decline } }
		ai_chance = {
			base = 15
			modifier = { add = 30 has_trait = greedy }
		}
	}
}
