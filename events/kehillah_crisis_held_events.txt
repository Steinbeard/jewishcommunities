namespace = kehillah_crisis_held

# Quarter Held is a real military deliverance, not merely the absence of a
# breach. This single, pre-judgment scene lets the player attend to its actual
# costs without sending a non-struck community into Recovery.
kehillah_crisis_held.0001 = {
	type = character_event
	title = kehillah_crisis_held.0001.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { primary_title = { has_variable = kehillah_crisis_outer_damage } } desc = kehillah_crisis_held.0001.desc_damage }
			desc = kehillah_crisis_held.0001.desc_clear
		}
	}
	theme = stewardship
	left_portrait = { character = root animation = worry }
	trigger = { kehillah_crisis_held_accounting_open_trigger = yes }

	# The outer-damage flag is set only from the actual near-Struck track in
	# kehillah_crisis_held_start_effect; this event never invents a breach.
	option = {
		name = kehillah_crisis_held.0001.repair
		trigger = {
			primary_title = { has_variable = kehillah_crisis_outer_damage }
			gold >= minor_gold_value
		}
		show_as_unavailable = { primary_title = { has_variable = kehillah_crisis_outer_damage } }
		remove_short_term_gold = minor_gold_value
		kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = 5 }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = 10 }
		primary_title = { set_variable = { name = kehillah_crisis_held_accounted value = yes } }
	}
	option = {
		name = kehillah_crisis_held.0001.watch
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = 10 }
		add_stress = minor_stress_loss
		primary_title = { set_variable = { name = kehillah_crisis_held_accounted value = yes } }
	}
	option = {
		name = kehillah_crisis_held.0001.account
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = 5 }
		primary_title = { set_variable = { name = kehillah_crisis_held_accounted value = yes } }
	}
}
