# =====================================================================
# Kehillah Bet Din Conference -- the docket.
#
# Source of truth: docs/spec/v5-bet-din-conference.md. Fired from
# common/activities/activity_types/kehillah_bet_din_conference.txt's
# on_phase_active. PART 1 (TEST BUILD), 2026-09-08 -- three cases
# proving the mechanic, not the dozens/hundreds-case pool v5 Part 2
# wants (blocked on a case-idea draft, per that spec's section 5).
#
# =====================================================================
# THE LOOP BUG, AND WHAT THE DOCKET LOOKS LIKE NOW (fixed 2026-09-08,
# NOT YET LIVE-TESTED -- the mechanics below are unverified against a
# real boot, the same caution the rest of this build carries).
#
# SYMPTOM: every event of the docket fired at once, and then the
# activity never ended -- the conference sat open with nothing left to
# say and no way to finish.
#
# TWO CAUSES, both structural:
#   1. THE CHAIN. Every event fired the next one immediately, and
#      resolutions fired the NEXT CASE's host event. So one
#      on_phase_active produced all ten events of the whole docket
#      inside a single day. A docket is supposed to be a session worked
#      through, not a stack of ten popups.
#   2. NOTHING ENDED THE PHASE. The activity had one phase and never
#      called progress_activity_phase_after anywhere. That call is what
#      ends a phase in CK3 -- every vanilla activity_type makes it
#      (feast, wedding, coronation, tournament, local_examination were
#      all checked; no exceptions). Without it the phase never ended,
#      so the activity never completed.
#
# THE FIX, in three places:
#   * The activity now has kehillah_bet_din_docket_case_count (3)
#     phases, one case each, and each phase's on_phase_active DRAWS a
#     case at random from those this session has not heard yet
#     (kehillah_bet_din_draw_case_effect). The phase count is what makes
#     the docket end after three cases.
#   * Within a case, the three events are now
#     kehillah_bet_din_case_step_days apart instead of instantaneous.
#   * A case's resolution ends its phase (kehillah_bet_din_advance_
#     docket_effect) instead of summoning the next case, and the
#     activity's on_complete -- not an event -- fires the closing event
#     kehillah_bet_din.0099, which scores the whole session.
# See common/scripted_effects/kehillah_bet_din_scripted_effects.txt's
# "THE DOCKET STATE MACHINE" section for the state this all sits on.
# =====================================================================
#
# THE SHAPE OF A CASE, EVERY TIME (three events):
#   HOST event   (.0001/.0011/.0021) -- root = the convening leader.
#     Presents the case (real courtiers, for the cases that use them --
#     see below) and two options representing a genuine legal/moral
#     fork, not a pass/fail prompt. Whichever option is picked, the
#     HOST's own stat is checked against a threshold (two tiers, pass/
#     fail -- see kehillah_bet_din_judge_pass_tally/fail_tally in
#     common/script_values/kehillah_script_values.txt for why two, not
#     the three kehillah_dispute_events.txt uses) and the result is
#     added to involved_activity.var:kehillah_bet_din_tally. Chains to the co-
#     judge event, kehillah_bet_din_case_step_days later.
#   CO-JUDGE event (.0002/.0012/.0022) -- root = judge2 (global_var:
#     kehillah_bet_din_judge2), fired via trigger_event the same way
#     the activity's on_phase_active fires .0001. Presents the SAME
#     case, judge2 picks via ai_chance (this mod has no live 2-player
#     scenario yet, so this event only ever resolves through AI choice
#     in practice, but is written as a real event, not a disguised
#     effect block, so a future multiplayer Sh'um session sees a real
#     prompt). judge2's tier check adds to the tally the same way the
#     host's did. THE THIRD JUDGE, judge3, is folded into this SAME
#     event's effect as a direct stat check rather than getting its own
#     fourth event -- a deliberate Part 1 scope cut (see this file's
#     footer note) to keep the chain at 3 events per case instead of 4;
#     revisit if Part 2's larger case volume can afford the extra hop.
#     Chains to the resolution event, again after
#     kehillah_bet_din_case_step_days.
#   RESOLUTION event (.0003/.0013/.0023) -- root = the convening leader
#     again (involved_activity.var:kehillah_bet_din_convening_title.holder).
#     Reads the case's final tally, applies Prosperity/Stability/
#     Greatness to the CONVENING community only (see the script-values
#     file's own note on why not all three symmetrically, unlike the
#     retired takkanot), records the verdict's tier in the SESSION score
#     (kehillah_bet_din_record_case_result_effect), and then ends the
#     activity phase. It chains to no other event: the next case comes
#     from the next phase's own draw, and the closing event comes from
#     the activity completing.
#
# REAL CHARACTERS. Cases 1-2 pick two real courtiers from the convening
# leader's own court via kehillah_bet_din_pick_litigants_effect (common/
# scripted_effects/kehillah_bet_din_scripted_effects.txt), mirroring
# kehillah_dispute_pick_parties_effect exactly. Case 3 deliberately has
# none -- it is a communal policy question (does the community forbid
# the practice at all), not one household's dispute, the same shape the
# retired takkanot always had. This mirrors the historical Takkanot
# Shu"m themselves: communal ordinances, not individual verdicts.
#
# THE TENET-ADDING CASE. Case 3, "The Second Wife," is built on
# Rabbeinu Gershom's real historical herem against polygamy. Checked
# against the installed vanilla files (common/religion/religion_types/
# 00_judaism.txt): rabbinism, the faith this mod's Kehillot actually
# use, ships with doctrine = doctrine_polygamy as its starting marriage
# doctrine (that file's own header comment: "Polygamy was legal until
# around 1000 AD"). A great ruling that also chooses to forbid the
# practice removes doctrine_polygamy and adds doctrine_monogamy on the
# ruling character's faith -- both confirmed vanilla doctrine keys in
# the same marriage doctrine group (common/religion/doctrine_group_
# types', doctrine_marriage_type), not invented ones, via add_doctrine/
# remove_doctrine (confirmed vanilla effect precedent: common/scripted_
# effects/00_game_rule_effects.txt's game_rule_full_gender_equality_
# effect uses the identical remove-then-add shape on a religion's
# doctrine group).
#
# PART 1 SCOPE CUTS, FOR THE RECORD (v5 spec section 6 material):
#   - judge3 never gets its own event (see the co-judge event note
#     above). This is the one place this test build does NOT give
#     every judge a real choice-driven moment -- it gets a stat check
#     instead. Worth revisiting once Part 2's case count makes the
#     cost of a 4th event per case worth weighing for real.
#   - (NO LONGER TRUE, fixed 2026-09-08 with the loop fix: the docket
#     was a fixed sequence with no random draw. It now draws each case
#     at random from the ones the session has not heard yet -- see
#     kehillah_bet_din_draw_case_effect. What is still missing is Part
#     2's case CONTENT: with only these three cases in the pool, a
#     session hears all three, just in a random order.)
#   - Effects land on the convening community only, not all three
#     symmetrically. Flagged as a default, not a resolved design call,
#     in the script-values file.
# =====================================================================

# ---------------------------------------------------------------------
# kehillah_bet_din.0001 -- CASE 1, HOST RULING
# ---------------------------------------------------------------------

kehillah_bet_din.0001 = {
	type = character_event
	title = kehillah_bet_din.0001.t
	desc = kehillah_bet_din.0001.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = involved_activity.var:kehillah_bet_din_litigant_a
		animation = disapproval
	}
	lower_right_portrait = {
		character = involved_activity.var:kehillah_bet_din_litigant_b
		animation = worry
	}

	trigger = {
		government_has_flag = government_is_kehillah
	}

	immediate = {
		involved_activity = { set_variable = { name = kehillah_bet_din_tally value = 0 } }
		kehillah_bet_din_pick_litigants_effect = yes
		# Re-saved as plain scope names so loc can reference them the same
		# way kehillah_dispute_events.txt's [kehillah_party_a.GetFirstName]
		# does -- loc text uses a save_scope_as name directly, not a
		# global_var: prefix (no vanilla loc precedent was found for
		# referencing global_var: from inside a .yml string, so this mod
		# doesn't invent one).
		# Guarded -- litigant picking has a known small-court failure mode
		# (see kehillah_bet_din_pick_litigants_effect's header); an unset
		# read here would hit the same tooltip-rebuild error-storm risk as
		# the "after" blocks above.
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_litigant_a }
			involved_activity.var:kehillah_bet_din_litigant_a = { save_scope_as = kbd_litigant_a }
		}
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_litigant_b }
			involved_activity.var:kehillah_bet_din_litigant_b = { save_scope_as = kbd_litigant_b }
		}
	}

	# HOLD TO THE LETTER OF THE PROMISE.
	option = {
		name = kehillah_bet_din.0001.a

		if = {
			limit = { stewardship >= 10 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } }
			custom_tooltip = kehillah_bet_din.0001.a.pass.tt
		}
		else = {
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
			custom_tooltip = kehillah_bet_din.0001.a.fail.tt
		}

		ai_chance = { base = 50 }
	}

	# EXCUSE THE SHORTFALL.
	option = {
		name = kehillah_bet_din.0001.b

		if = {
			limit = { stewardship >= 10 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } }
			custom_tooltip = kehillah_bet_din.0001.b.pass.tt
		}
		else = {
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
			custom_tooltip = kehillah_bet_din.0001.b.fail.tt
		}

		ai_chance = { base = 50 }
	}

	# Guarded with exists = -- BUG FOUND LIVE, 2026-09-08: an unguarded
	# global_var read here, when the var is unset, doesn't just log once.
	# CK3 re-evaluates an option's effect (this "after" block included) on
	# EVERY hover-tooltip rebuild while the player's mouse rests on that
	# option, so an unset global_var here produced a live error-log storm
	# (~20MB in well under a minute in the session that found this) rather
	# than a single clean failure. Real effect execution (actually picking
	# the option) only errors once and moves on -- it was the SPECULATIVE
	# tooltip-preview evaluation that made this dangerous. Every "after"
	# block in this file got the same guard once this was found; see this
	# comment's twin on the first one where it's explained, not repeated.
	after = {
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_judge2 }
			involved_activity.var:kehillah_bet_din_judge2 = {
				trigger_event = {
					id = kehillah_bet_din.0002
					days = kehillah_bet_din_case_step_days
				}
			}
		}
	}
}

		animation = worry
	}

	trigger = {
		government_has_flag = government_is_kehillah
	}

	immediate = {
		# Guarded -- litigant picking has a known small-court failure mode
		# (see kehillah_bet_din_pick_litigants_effect's header); an unset
		# read here would hit the same tooltip-rebuild error-storm risk as
		# the "after" blocks above.
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_litigant_a }
			involved_activity.var:kehillah_bet_din_litigant_a = { save_scope_as = kbd_litigant_a }

			limit = { exists = involved_activity.var:kehillah_bet_din_litigant_b }
			involved_activity.var:kehillah_bet_din_litigant_b = { save_scope_as = kbd_litigant_b }
		}
		# The third judge is folded in here as a direct check -- see this
		# file's header for why. Guarded with an outer exists check -- same
		# unset-global_var risk as the "after" blocks (see kehillah_bet_din.

		# on every tooltip rebuild if judge3 somehow never got set.

			limit = { exists = involved_activity.var:kehillah_bet_din_judge3 }
			if = {
				limit = { involved_activity.var:kehillah_bet_din_judge3.learning >= 10 }
				involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } }
			}
			else = {
				involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
			}
		}
		else = {
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
		}
	}

	option = {
		name = kehillah_bet_din.0002.a

		if = {
			limit = { stewardship >= 10 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } }
		}
		else = {
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
		}

		ai_chance = {
			base = 10
			modifier = { add = 10 stewardship >= 10 }
		}
	}

	option = {
		name = kehillah_bet_din.0002.b

		if = {
			limit = { stewardship >= 10 }
			involved_activity = { change_variable = { name = kehillah_bet_din_tally add = kehillah_bet_din_judge_pass_tally } }
		}
		else = {
			involved_activity = { change_variable = { name = kehillah_bet_din_tally subtract = kehillah_bet_din_judge_fail_tally } }
		}

		ai_chance = {
			base = 10
			modifier = { add = 10 stewardship >= 10 }
		}
	}

	# Guarded -- see kehillah_bet_din.0001's "after" for why.
	after = {
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_convening_title }
			involved_activity.var:kehillah_bet_din_convening_title.holder = {
				trigger_event = {
					id = kehillah_bet_din.0003
					days = kehillah_bet_din_case_step_days
				}
			}
		}
	}
}

kehillah_bet_din.0003 = {
	type = character_event
	title = kehillah_bet_din.0003.t
	desc = kehillah_bet_din.0003.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	# BUG FOUND LIVE, 2026-09-24: the resolution event dropped the litigant
	# portraits its own case's ruling events (0001/0002) show, even though
	# the litigant vars are still in scope here (used just below by
	# kehillah_bet_din_record_case_result_effect's TARGET). Restored,
	# matching 0001's own animations.
	right_portrait = {
		character = involved_activity.var:kehillah_bet_din_litigant_a
		animation = disapproval
	}
	lower_right_portrait = {
		character = involved_activity.var:kehillah_bet_din_litigant_b
		animation = worry
	}

	trigger = {
		government_has_flag = government_is_kehillah
	}

	immediate = {
		if = {
			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great }
			involved_activity.var:kehillah_bet_din_convening_title = {
				change_variable = { name = kehillah_var_stability add = kehillah_bet_din_case1_stability_gain_great }
				change_variable = { name = kehillah_var_prosperity add = kehillah_bet_din_case1_prosperity_gain_great }
			}
			kehillah_bet_din_record_case_result_effect = { SCORE = kehillah_bet_din_case_score_great CASE = 1 TIER = great TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			custom_tooltip = kehillah_bet_din.0003.great.tt
		}
		else_if = {
			limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_good }
			involved_activity.var:kehillah_bet_din_convening_title = {
				change_variable = { name = kehillah_var_stability add = kehillah_bet_din_case1_stability_gain_good }
				change_variable = { name = kehillah_var_prosperity add = kehillah_bet_din_case1_prosperity_gain_good }
			}
			kehillah_bet_din_record_case_result_effect = { SCORE = kehillah_bet_din_case_score_good CASE = 1 TIER = good TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			custom_tooltip = kehillah_bet_din.0003.good.tt
		}
		else = {
			involved_activity.var:kehillah_bet_din_convening_title = {
				change_variable = { name = kehillah_var_stability subtract = kehillah_bet_din_case1_stability_loss_poor }
			}
			kehillah_bet_din_record_case_result_effect = { SCORE = kehillah_bet_din_case_score_poor CASE = 1 TIER = poor TARGET = involved_activity.var:kehillah_bet_din_litigant_a }
			custom_tooltip = kehillah_bet_din.0003.poor.tt
		}

		kehillah_bet_din_case1_memory_effect = yes
	}

	option = {
		name = kehillah_bet_din.0003.a
	}

	# END OF THE CASE, NOT OF THE DOCKET. This used to chain straight
	# into case 2's host event, which is half of why the whole docket
	# fired at once; the case now ends its own activity phase instead and
	# the NEXT phase draws the next case. See this file's header.
	after = {
		kehillah_bet_din_advance_docket_effect = yes
	}
}

# CASE 6 -- THE SILVERSMITHS' QUARREL
# ADDED 2026-09-15 at user request. Two real courtiers (kehillah_bet_
# din_pick_litigants_effect, reused verbatim, same as case 5 above):
# litigant_a the accuser (his shop was smashed), litigant_b the rival
# silversmith he says put the mob up to it.
#
# THE FACTS ARE NOT IN DOUBT HERE, UNLIKE CASE 5 -- the shop really was
# smashed by a gentile mob; there is no hidden truth roll. What is in
# genuine dispute is which of three real, distinct halakhic FRAMEWORKS
# actually applies, and the three options below draw on three different
# real Talmudic principles rather than three facings of one question:
#   a) MESIRAH -- treating this as informing on / inciting gentiles
#      against a fellow Jew, historically the community's gravest
#      betrayal (grave enough that a moser could, in real halakhic
#      history, be resisted by the community directly). Intrigue-led
#      (14): proving actual INTENT to incite, not mere coincidence of
#      timing, is the hard part.
#   b) GERAMA -- Bava Kamma's own distinction between direct damage
#      (the mob's own, adam hamazik) and damage caused only INDIRECTLY,
#      through an intermediate agent one set in motion. A real,
#      substantial halakhic liability, just a lesser and more ordinary
#      one than mesirah. Learning-led (12): correctly applying the
#      direct/indirect distinction is a scholarly question.
#   c) DISMISS FOR WANT OF PROOF -- kin'at sofrim tarbeh chochma
#      ("the envy of scholars/craftsmen increases wisdom"): fierce
#      trade rivalry between two silversmiths is not, on its own,
#      evidence that one incited a mob. Diplomacy-led (10): the skill
#      here is holding the community together around an unresolved
#      grievance, not scholarship or investigation.

# ---------------------------------------------------------------------
# kehillah_bet_din.0099 -- THE DOCKET CONCLUDES, AND IS SCORED
#
# Fired from the activity's own on_complete (common/activities/activity_
# types/kehillah_bet_din_conference.txt), NOT from the last case's
# "after" the way it used to be. Two things follow from that: the
# session is always scored exactly once, and it is scored even if a case
# was skipped for want of eligible litigants or a case's own chain never
# finished.
#
# WHAT IS BEING SCORED. involved_activity.var:kehillah_bet_din_session_score is the
# sum of the three cases' verdict tiers (great +2, good +1, poor -1 --
# kehillah_bet_din_case_score_* in common/script_values/kehillah_script_
# values.txt), so it runs +6 down to -3. That is a different thing from
# the per-case tally, which measures the three judges' skill checks
# WITHIN one case and is reset for each case; see the script-values
# file's own note on the two scores.
#
# The reward here sits ON TOP of the per-case effects each resolution
# already applied -- it is the standing the panel earned by ruling a
# whole docket well (or badly), not a re-payment of the rulings. Hence
# Greatness-led: a docket ruled well is what other communities hear
# about. A net-negative docket is a real loss, not a smaller gain, in
# keeping with how kehillah_bet_din.0023 already treats a herem the
# panel could not carry.
#
# The piety/prestige values are vanilla's own named tiers (major_piety_
# gain, minor_prestige_loss, etc. -- common/script_values/00_basic_
# values.txt), the same ones this mod already uses in kehillah_
# protection_events.txt and kehillah_decisions.txt.
# ---------------------------------------------------------------------

kehillah_bet_din.0099 = {
	type = character_event
	title = kehillah_bet_din.0099.t
	theme = faith

	# GENERALIZED 2026-09-10 (v6 spec section 1): the closing desc text
	# below used to name "Speyer and Mainz" directly, correct only when
	# Worms convened -- now that any of the four regions' own Greatness
	# leader can convene (v6 spec section 4), the desc needs to name
	# WHICHEVER two communities' leaders actually sat as co-judges this
	# session, not a hardcoded pair. Re-saved here as plain scope names so
	# loc can reference them directly -- loc text uses a save_scope_as
	# name, not a global_var:/involved_activity.var: prefix (no vanilla
	# loc precedent was found for referencing either from inside a .yml
	# string, matching kehillah_bet_din.0001's own identical note on this
	# same limitation). Guarded with exists -- this event can be reached
	# after a session that never properly opened a docket.
	immediate = {
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_judge2 }
			involved_activity.var:kehillah_bet_din_judge2 = { save_scope_as = kbd_judge2 }
		}
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_judge3 }
			involved_activity.var:kehillah_bet_din_judge3 = { save_scope_as = kbd_judge3 }
		}
	}

	# WHY THE SESSION REWARD LIVES IN THE OPTION, NOT IN "immediate".
	# Effects in an immediate block are applied but never SHOWN -- which
	# means every custom_tooltip in this file's immediate blocks (the
	# per-case verdict tiers included) is invisible to the player. That is
	# tolerable for a per-case verdict, whose text carries the meaning
	# anyway; it is not tolerable for the docket's own payout, which is
	# the whole point of scoring the session. So the reward is applied
	# from the option, where CK3 renders the full effect list plus the
	# custom_tooltip on hover, and the score's TIER is additionally told
	# in plain prose by the triggered desc below -- so a player who never
	# hovers anything still learns how the docket went.
	#
	# (The per-case resolutions were deliberately left alone here. Moving
	# their tooltips would mean re-testing all three cases' verdict paths,
	# which belongs with a pass that can actually boot the game; noted as
	# a real, separate cosmetic gap rather than fixed blind.)
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_landmark } }
				desc = kehillah_bet_din.0099.desc.landmark
			}
			triggered_desc = {
				trigger = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_strong } }
				desc = kehillah_bet_din.0099.desc.strong
			}
			triggered_desc = {
				trigger = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_adequate } }
				desc = kehillah_bet_din.0099.desc.adequate
			}
			triggered_desc = {
				trigger = { kehillah_bet_din_session_heard_any_case_trigger = yes }
				desc = kehillah_bet_din.0099.desc.failed
			}
			# Fallback: a session that reached its close without a single
			# case ever being ruled on -- no case was eligible in a very
			# thin court, or every case's popup went unanswered until the
			# activity ran out of phases. The original, tier-free closing
			# text, kept for exactly this case, and no effects go with it.
			desc = kehillah_bet_din.0099.desc
		}
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		government_has_flag = government_is_kehillah
	}

	option = {
		name = kehillah_bet_din.0099.a

		# Guarded as a whole -- unlike the case events, this one can be
		# reached (by on_complete) after a session that never opened a
		# docket properly, e.g. one where no case was ever eligible. And
		# because this IS an option effect, an unguarded global_var read
		# here is the tooltip-rebuild error storm this file's other
		# comments describe, not a single clean error.
		if = {
			limit = { exists = involved_activity.var:kehillah_bet_din_convening_title }

			if = {
				limit = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_landmark } }
				involved_activity.var:kehillah_bet_din_convening_title = {
					change_variable = { name = kehillah_var_greatness add = kehillah_bet_din_session_greatness_gain_landmark }
					change_variable = { name = kehillah_var_stability add = kehillah_bet_din_session_stability_gain_landmark }
					change_variable = { name = kehillah_var_prosperity add = kehillah_bet_din_session_prosperity_gain_landmark }
				}
				kehillah_bet_din_apply_session_stability_to_panel_effect = { OP = add VALUE = kehillah_bet_din_session_stability_gain_landmark }
				add_piety = major_piety_gain
				add_prestige = major_prestige_gain
				custom_tooltip = kehillah_bet_din.0099.landmark.tt
			}
			else_if = {
				limit = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_strong } }
				involved_activity.var:kehillah_bet_din_convening_title = {
					change_variable = { name = kehillah_var_greatness add = kehillah_bet_din_session_greatness_gain_strong }
					change_variable = { name = kehillah_var_stability add = kehillah_bet_din_session_stability_gain_strong }
				}
				kehillah_bet_din_apply_session_stability_to_panel_effect = { OP = add VALUE = kehillah_bet_din_session_stability_gain_strong }
				add_piety = medium_piety_gain
				add_prestige = medium_prestige_gain
				custom_tooltip = kehillah_bet_din.0099.strong.tt
			}
			else_if = {
				limit = { kehillah_bet_din_session_score_at_least_trigger = { THRESHOLD = kehillah_bet_din_session_threshold_adequate } }
				involved_activity.var:kehillah_bet_din_convening_title = {
					change_variable = { name = kehillah_var_greatness add = kehillah_bet_din_session_greatness_gain_adequate }
					change_variable = { name = kehillah_var_stability add = kehillah_bet_din_session_stability_gain_adequate }
				}
				kehillah_bet_din_apply_session_stability_to_panel_effect = { OP = add VALUE = kehillah_bet_din_session_stability_gain_adequate }
				add_piety = minor_piety_gain
				custom_tooltip = kehillah_bet_din.0099.adequate.tt
			}
			# A net-negative docket -- a real loss. Requires that at least
			# one case actually got a verdict: a session that closed with
			# NOTHING ruled on (thin court with no eligible case, or a
			# player who never answered a single case) falls through this
			# whole chain and gets no effects at all, which is the honest
			# outcome for a docket that never happened rather than a
			# punishment for one.
			else_if = {
				limit = { kehillah_bet_din_session_heard_any_case_trigger = yes }
				involved_activity.var:kehillah_bet_din_convening_title = {
					change_variable = { name = kehillah_var_greatness subtract = kehillah_bet_din_session_greatness_loss_failed }
					change_variable = { name = kehillah_var_stability subtract = kehillah_bet_din_session_stability_loss_failed }
				}
				kehillah_bet_din_apply_session_stability_to_panel_effect = { OP = subtract VALUE = kehillah_bet_din_session_stability_loss_failed }
				add_prestige = minor_prestige_loss
				custom_tooltip = kehillah_bet_din.0099.failed.tt
			}

			# Tier-independent: the co-judges sat on the panel whether the
			# docket went well or badly. See kehillah_bet_din_log_session_
			# participants_effect's own header (common/scripted_effects/
			# kehillah_bet_din_scripted_effects.txt) for why this needed
			# adding at all -- their faces never appeared anywhere in this
			# activity's UI before this.
			kehillah_bet_din_log_session_participants_effect = yes
		}
	}

	# The session's scratch state is cleared HERE, after the score has
	# been read -- not in on_complete, which would race this event's own
	# immediate block. The list of cases already heard is deliberately not
	# cleared here; see kehillah_bet_din_close_docket_effect's header.
	after = {
		kehillah_bet_din_close_docket_effect = yes
	}
}
