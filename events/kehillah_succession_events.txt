# =====================================================================
# Kehillah succession events -- ROADMAP item S3, added 2026-09-26
# (Sukkot heartbeat 3).
#
# One event today: the deferred government handoff for S3(a).
#
# WHY THIS EXISTS. kehillah_on_title_gain (common/on_action/
# kehillah_on_actions.txt) has to give an appointed successor the
# kehillah government, because nothing else does. But
# can_get_government (common/governments/kehillah_government.txt)
# requires the new leader to hold a landless-type title, and on every
# appointment succession since 2026-09-06 that call has thrown
# `change_government effect [ Trying to set illegal government ]` --
# three separate live tests, two "fixes" of the predicate itself,
# identical error each time (docs/implementation/
# v1-kehillah-implementation.md section 8).
#
# The on_title_gain caller now guards the inline attempt with that same
# predicate, and routes here when it would fail, one day later, by which
# point the title is certainly committed. Read that caller's own comment
# block for why deferring cannot regress anything: on the appointment
# path the inline call is already a no-op today, which is exactly what
# the error means.
#
# THIS EVENT IS ALSO THE VERDICT ON WHETHER THE GUARD WAS RIGHT. Its
# debug_log lines are unconditional (there are at most a handful per
# succession, not per tick) and say plainly whether the government still
# needed setting a day later and whether setting it then worked. If the
# real cause was never the timing, this event will report finding the
# government already correct, and that is the answer, not a failure.
# =====================================================================

namespace = kehillah_succession


# ---------------------------------------------------------------------
# kehillah_succession.0010 -- DEFERRED GOVERNMENT HANDOFF
#
# root = the new leader. scope:kq_title = their community title, saved by
# kehillah_on_title_gain before this is triggered.
# ---------------------------------------------------------------------
kehillah_succession.0010 = {
	type = character_event
	hidden = yes

	trigger = {
		exists = scope:kq_title
	}

	immediate = {
		# The single most important line this event logs, added
		# 2026-09-26 (heartbeat 7). The refuted-timing trace showed the
		# successor has no domicile at the instant on_title_gain fires,
		# and kehillah_government.txt requires one
		# (`domicile_type = kehillah_quarter`). Whether the quarter has
		# arrived by a day later is the whole question this deferral
		# rests on, so it is reported before anything is attempted.
		if = {
			limit = { exists = domicile }
			debug_log = "kehillah_succession.0010: the new leader HAS a domicile one day after title gain -- the quarter travelled with the title"
		}
		else = {
			debug_log = "kehillah_succession.0010: the new leader STILL has NO domicile one day after title gain -- deferring longer would not help either"
		}

		if = {
			limit = { government_has_flag = government_is_kehillah }
			# The engine propagated it on its own after on_title_gain
			# returned. Nothing to do, and worth logging: it means the
			# inline call was redundant all along, not merely mistimed.
			debug_log = "kehillah_succession.0010: government was ALREADY kehillah one day after title gain -- engine propagated it, no action needed"
		}
		else_if = {
			limit = {
				any_held_title = { is_landless_type_title = yes }
				exists = domicile
			}
			debug_log = "kehillah_succession.0010: government was NOT kehillah, and both a landless title and a domicile are now present -- setting it (deferral was the fix)"
			change_government = kehillah_government

			# Both of these were already run inline by
			# kehillah_on_title_gain and are documented there as safe to
			# call again (kehillah_restore_quarter_effect is a no-op with
			# no stored building state; kehillah_maintain_host_charter_
			# effect's own header says it is safe and cheap to call
			# unconditionally, every time). They are repeated here
			# because the inline pass ran before the domicile that
			# change_government creates existed.
			kehillah_restore_quarter_effect = yes
			kehillah_maintain_host_charter_effect = yes

			# Whether the deferred call was actually accepted, stated by
			# the engine rather than assumed. If this reads FAIL, the
			# error at the on_title_gain call site has simply moved here
			# and the domicile theory is wrong too -- report it, do not
			# patch a fifth time.
			if = {
				limit = { government_has_flag = government_is_kehillah }
				debug_log = "kehillah_succession.0010: CONFIRMED the government IS kehillah after the deferred call -- deferral is the fix"
			}
			else = {
				debug_log = "kehillah_succession.0010: FAIL the deferred change_government was refused as well -- the domicile theory is wrong; report, do not patch."
			}
		}
		else = {
			# Neither timing nor the predicate explains it. Do NOT guess
			# from here -- this is the case the 2026-09-23 note meant
			# when it said a fourth blind fix must not be attempted.
			if = {
				limit = { any_held_title = { is_landless_type_title = yes } }
				debug_log = "kehillah_succession.0010: STILL cannot get the kehillah government a day later -- root holds a landless title but has NO domicile, so the engine has nowhere to put a kehillah_quarter. Report, do not patch."
			}
			else = {
				debug_log = "kehillah_succession.0010: STILL cannot get the kehillah government a day later -- root holds no landless-type title. Neither timing nor the predicate explains this; report, do not patch."
			}
		}
	}
}
