kehillah_household_economy.10 = {
 type = character_event
 hidden = yes
 immediate = {
  kehillah_household_refresh_effect = yes
  if = {
   limit = { kehillah_household_can_manage_trigger = yes
 NOT = { scope:family.var:kehillah_household_assignment = 0 }
  }
   scope:family = {
    set_variable = { name = kehillah_household_assignment value = 0 }
    set_variable = { name = kehillah_household_assignment_cooldown value = yes years = kehillah_household_assignment_years }
   }
   kehillah_household_refresh_effect = yes
  }
 }
}
