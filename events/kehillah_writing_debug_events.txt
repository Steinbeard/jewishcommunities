# Console-only probes for the multi-phase writing project.
# Fire on the intended author: event kehillah_writing_debug.1
# Use after opening the decision, after each scheme phase, and after finish.
namespace = kehillah_writing_debug

kehillah_writing_debug.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_writing_debug.1 START"
		if = { limit = { kehillah_writing_torah_trigger = yes } debug_log = "kehillah_writing_debug.1 Torah commentary available" }
		else = { debug_log = "kehillah_writing_debug.1 Torah commentary unavailable" }
		if = { limit = { kehillah_writing_talmud_trigger = yes } debug_log = "kehillah_writing_debug.1 Bavli commentary available" }
		else = { debug_log = "kehillah_writing_debug.1 Bavli commentary unavailable" }
		if = { limit = { has_variable = kehillah_writing_form } debug_log = "kehillah_writing_debug.1 Draft form selected" }
		if = { limit = { has_variable = kehillah_writing_phase } debug_log = "kehillah_writing_debug.1 Writing scheme active or recently resolved" }
		if = { limit = { has_variable = kehillah_writing_success } debug_log = "kehillah_writing_debug.1 Completion succeeded" }
		if = { limit = { has_variable = kehillah_writing_landmark } debug_log = "kehillah_writing_debug.1 Landmark niche filled" }
		if = { limit = { has_variable = kehillah_writing_ordinary_form } debug_log = "kehillah_writing_debug.1 Ordinary manuscript, no ledger work" }
		if = { limit = { has_variable = kehillah_sefer_my_latest_work } debug_log = "kehillah_writing_debug.1 Ledger work registered" }
		debug_log = "kehillah_writing_debug.1 END"
	}
}

# 2026-11-24 sparks (docs/spec/2026-11-24-writing-sparks-v2.md).
# .2 logs each discipline slot. .3 fills every empty slot the character can
# hold, as if from study. .4 raises every held spark one tier (Insight to
# Chiddush ignores its routes), for checking the Learning window's cards.
kehillah_writing_debug.2 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "kehillah_writing_debug.2 START"
		if = { limit = { has_variable = kehillah_spark_parshanut_form } debug_log = "kehillah_writing_debug.2 parshanut: [ROOT.Char.Custom('KehillahSparkForm_parshanut')] tier [ROOT.Char.MakeScope.Var('kehillah_spark_parshanut_tier').GetValue|0], paths [ROOT.Char.MakeScope.ScriptValue('kehillah_spark_debug_paths_parshanut_value')|0]" }
		else = { debug_log = "kehillah_writing_debug.2 parshanut: empty" }
		if = { limit = { has_variable = kehillah_spark_talmudics_form } debug_log = "kehillah_writing_debug.2 talmudics: [ROOT.Char.Custom('KehillahSparkForm_talmudics')] tier [ROOT.Char.MakeScope.Var('kehillah_spark_talmudics_tier').GetValue|0], paths [ROOT.Char.MakeScope.ScriptValue('kehillah_spark_debug_paths_talmudics_value')|0]" }
		else = { debug_log = "kehillah_writing_debug.2 talmudics: empty" }
		if = { limit = { has_variable = kehillah_spark_hashkafa_form } debug_log = "kehillah_writing_debug.2 hashkafa: [ROOT.Char.Custom('KehillahSparkForm_hashkafa')] tier [ROOT.Char.MakeScope.Var('kehillah_spark_hashkafa_tier').GetValue|0], paths [ROOT.Char.MakeScope.ScriptValue('kehillah_spark_debug_paths_hashkafa_value')|0]" }
		else = { debug_log = "kehillah_writing_debug.2 hashkafa: empty" }
	}
}

kehillah_writing_debug.3 = {
	type = character_event
	hidden = yes
	immediate = {
		kehillah_spark_from_parshanut_pool_effect = { WORK = flag:none }
		kehillah_spark_from_talmudics_pool_effect = { WORK = flag:none }
		kehillah_spark_from_hashkafa_pool_effect = { WORK = flag:none }
		trigger_event = kehillah_writing_debug.2
	}
}
