namespace = kehillah_kohen_gadol_debug

# Console only. Report the High Priesthood's state and the player's eligibility.
kehillah_kohen_gadol_debug.0001 = {
	type = character_event
	hidden = yes
	immediate = {
		if = { limit = { kehillah_kohen_gadol_tenet_gate_trigger = yes } debug_log = "KOHEN_GADOL: tenet gate=yes" }
		else = { debug_log = "KOHEN_GADOL: tenet gate=no" }
		if = { limit = { kehillah_kohen_gadol_holds_jerusalem_trigger = yes } debug_log = "KOHEN_GADOL: holds Jerusalem=yes" }
		else = { debug_log = "KOHEN_GADOL: holds Jerusalem=no" }
		if = { limit = { kehillah_kohen_gadol_can_seat_trigger = yes } debug_log = "KOHEN_GADOL: can seat=yes" }
		else = { debug_log = "KOHEN_GADOL: can seat=no" }
		if = { limit = { title:c_jerusalem = { has_variable = kehillah_temple_service_restored } } debug_log = "KOHEN_GADOL: temple service restored=yes" }
		else = { debug_log = "KOHEN_GADOL: temple service restored=no" }
		if = {
			limit = { exists = global_var:kehillah_kohen_gadol_title }
			global_var:kehillah_kohen_gadol_title = { save_scope_as = kehillah_kg_title }
			global_var:kehillah_kohen_gadol_title.holder ?= { save_scope_as = kehillah_kg_kohen }
			debug_log = "KOHEN_GADOL: office exists"
			debug_log_scopes = yes
		}
		else = { debug_log = "KOHEN_GADOL: no office" }
		if = { limit = { exists = global_var:kehillah_kohen_gadol_is_head } debug_log = "KOHEN_GADOL: heads the faith=yes" }
		else = { debug_log = "KOHEN_GADOL: heads the faith=no" }
	}
}
