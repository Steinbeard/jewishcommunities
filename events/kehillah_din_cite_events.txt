# =====================================================================
# On Whose Authority? (2026-11-11, Daniel). After a judge rests a ruling on
# the sources, the court names the work it relies on: the most established
# authority, a newer one, or one whose teaching the judge holds. See
# common/scripted_effects/kehillah_din_citation_effects.txt and
# docs/spec/2026-11-11-rulings-sermons-and-gates.md section 2.
# =====================================================================
namespace = kehillah_din_cite

kehillah_din_cite.0001 = {
	type = character_event
	title = kehillah_din_cite.0001.t
	desc = kehillah_din_cite.0001.desc
	theme = learning
	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		exists = involved_activity
		exists = involved_activity.var:kehillah_din_cite_grounded
	}

	immediate = {
		involved_activity = { save_scope_as = kehillah_din_cite_activity }
	}

	# The most established work he has mastered on this order.
	option = {
		name = kehillah_din_cite.0001.grounded
		kehillah_din_cite_grounded_effect = yes
		ai_chance = {
			base = 40
			modifier = { add = 40 scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= { OR = { has_trait = zealous has_trait = stubborn } } }
		}
	}

	# A newer, lesser-known work: a bolder ruling.
	option = {
		name = kehillah_din_cite.0001.innovative
		trigger = {
			exists = scope:kehillah_din_cite_activity.var:kehillah_din_cite_innovative
			NOT = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_innovative = scope:kehillah_din_cite_activity.var:kehillah_din_cite_grounded }
		}
		kehillah_din_cite_innovative_effect = yes
		ai_chance = {
			base = 20
			modifier = { add = 40 scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= { OR = { has_trait = ambitious has_trait = arrogant } } }
		}
	}

	# A work whose teaching he holds.
	option = {
		name = kehillah_din_cite.0001.conviction
		trigger = {
			exists = scope:kehillah_din_cite_activity.var:kehillah_din_cite_conviction
			NOT = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_conviction = scope:kehillah_din_cite_activity.var:kehillah_din_cite_grounded }
		}
		kehillah_din_cite_conviction_effect = yes
		ai_chance = {
			base = 30
			modifier = { add = 30 scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= { has_trait = zealous } }
		}
	}
}
