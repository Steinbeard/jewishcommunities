namespace = kehillah_crisis_intro_debug

# Inspect the state while the opening event is still unanswered: since 2026-11-06 the crisis is live.
kehillah_crisis_intro_debug.1 = {
 type = character_event
 hidden = yes
 orphan = yes
 immediate = {
  if = {
   limit = {
    primary_title ?= {
     NOT = { has_variable = kehillah_crisis_intro_pending }
     OR = {
      exists = var:kehillah_crisis_situation
      exists = var:kehillah_trial_situation
      exists = var:kehillah_decree_situation
      has_variable = kehillah_expulsion_quarters
     }
    }
   }
   debug_log = "INTRO PROBE: PASS crisis active with its opening popup"
  }
  else = { debug_log = "INTRO PROBE: FAIL opening still pending or no crisis; inspect scopes" }
  primary_title ?= { debug_log_scopes = yes }
 }
}
# After responding, capture the active instance. No state is fabricated by the probe.
kehillah_crisis_intro_debug.2 = {
 type = character_event
 hidden = yes
 orphan = yes
 immediate = {
  primary_title = {
   remove_variable = kehillah_intro_probe_situation
   if = { limit = { exists = var:kehillah_trial_situation } set_variable = { name = kehillah_intro_probe_situation value = var:kehillah_trial_situation } }
   else_if = { limit = { exists = var:kehillah_decree_situation } set_variable = { name = kehillah_intro_probe_situation value = var:kehillah_decree_situation } }
   else_if = { limit = { exists = var:kehillah_crisis_situation } set_variable = { name = kehillah_intro_probe_situation value = var:kehillah_crisis_situation } }
   if = {
    limit = { NOT = { has_variable = kehillah_crisis_intro_pending } OR = { exists = var:kehillah_intro_probe_situation has_variable = kehillah_expulsion_quarters } }
    debug_log = "INTRO PROBE: PASS acknowledgment activated a crisis"
   }
   else = { debug_log = "INTRO PROBE: FAIL acknowledgment did not activate a crisis" }
   var:kehillah_intro_probe_situation ?= { debug_log_scopes = yes }
   debug_log_scopes = yes
  }
 }
}
# Repeat acknowledgment without a choice: its same situation reference and epoch must survive.
kehillah_crisis_intro_debug.3 = {
 type = character_event
 hidden = yes
 orphan = yes
 trigger = { primary_title ?= { NOT = { has_variable = kehillah_crisis_intro_pending } has_variable = kehillah_crisis_intro_epoch } }
 immediate = {
  primary_title = {
   save_scope_as = kehillah_crisis_intro_title
   save_scope_value_as = { name = kehillah_crisis_intro_epoch value = var:kehillah_crisis_intro_epoch }
  }
  kehillah_crisis_acknowledge_effect = { FAMILY = attack }
  kehillah_crisis_acknowledge_effect = { FAMILY = trial }
  kehillah_crisis_acknowledge_effect = { FAMILY = decree }
  kehillah_crisis_acknowledge_effect = { FAMILY = expulsion }
  remove_variable = kehillah_crisis_intro_accepted
  if = {
   limit = {
    primary_title = {
     var:kehillah_crisis_intro_epoch = scope:kehillah_crisis_intro_epoch
     OR = { var:kehillah_crisis_situation ?= var:kehillah_intro_probe_situation var:kehillah_trial_situation ?= var:kehillah_intro_probe_situation var:kehillah_decree_situation ?= var:kehillah_intro_probe_situation has_variable = kehillah_expulsion_quarters }
    }
   }
   debug_log = "INTRO PROBE: PASS repeated acknowledgment kept instance and epoch"
  }
  else = { debug_log = "INTRO PROBE: FAIL repeated acknowledgment changed instance or epoch" }
 }
}
