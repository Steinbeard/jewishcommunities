# =====================================================================
# THE LANDED KEHILLAH ("Kehillah of the Land"; V47 phase B,
# docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).
#
# A community leader who also holds land: rare, by extraordinary grant
# (Daniel 2026-10-19: Kehillot may rule land, it should just be difficult).
# The Nasi of the Sanhedrin is the main case: live tests on 2026-10-19
# showed CK3 forces a LANDLESS head of faith onto vanilla theocracy
# (refusing any Kehillah government and stripping his Quarter), while a
# landed Kehillah leader on this government kept his Quarter and stayed a
# playable temporal head of faith. Began as kehillah_nasi_government.
#
# kehillah_government copied field for field, plus mechanic_type =
# theocracy, the government_is_kehillah_landed flag, castle/city/temple
# holdings, and NO cannot_be_vassal_or_liege: that flag makes the engine
# refuse land to a Kehillah leader (it hands the county to a generated
# lowborn instead, live 2026-10-19), so a grant must switch the leader onto
# this government FIRST (kehillah_grant_community_county_effect). It keeps
# domicile_type = kehillah_quarter and the government_is_kehillah flag, so
# community systems and playability still treat him as a community leader.
# It does NOT carry government_is_theocracy (the council overrides and
# vanilla theocracy content key on it).
#
# kehillah_reconcile_landed_government_effect moves characters on and off
# it. Keep the shared fields in sync with kehillah_government.txt.
# =====================================================================
kehillah_landed_government = {
	government_rules = {
		landless_playable = yes
		rulers_should_have_dynasty = yes
		create_cadet_branches = no
		dynasty_named_realms = no
		council = yes
		inherit_from_dynastic_government = no
		use_title_tier_modifiers = no
		use_maa_maintenance = no
		court_generate_spouses = yes
		allow_accolades = no
	}

	domicile_type = kehillah_quarter
	mechanic_type = theocracy

	# Leads a community (holds a Kehillah title). Whether he also holds land
	# decides which of the two Kehillah governments he is on, and that is
	# kehillah_reconcile_landed_government_effect's job, not this trigger's.
	can_get_government = {
		any_held_title = { is_kehillah_title_trigger = yes }
	}

	valid_holdings = { castle_holding city_holding church_holding }

	court_generate_commanders = no

	character_modifier = {
		levy_size = -1
		men_at_arms_cap = -10
		men_at_arms_limit = -10
		knight_limit = -10
		monthly_learning_lifestyle_xp_gain_mult = 0.15
		monthly_stewardship_lifestyle_xp_gain_mult = 0.15
		short_reign_duration_mult = -0.5
		monthly_piety = 0.5
	}

	ai = {
		use_goals = yes
		use_legends = yes
		perform_religious_reformation = no
	}

	# No cannot_be_vassal_or_liege: he holds his county of his liege.
	flags = {
		government_is_kehillah
		government_is_kehillah_landed
	}

	min_appointment_tier = county
	color = { 230 205 120 }
}
