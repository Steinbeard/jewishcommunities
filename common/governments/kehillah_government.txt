# =====================================================================
# Kehillah government -- Phase 1, Track A.
#
# Source of truth: docs/spec/v1-kehillah-community.md (sections 2, 3, 6)
# and docs/scenarios/worms-1066.md.
#
# ---------------------------------------------------------------------
# DESIGN DEVIATION FROM SPEC SECTION 6 -- READ BEFORE CHANGING.
#
# The spec proposes modelling the Kehillah on the *administrative*
# pattern: a landless title held as a VASSAL inside the host realm.
# This implementation makes the Kehillah leader an INDEPENDENT landless
# ruler instead. Reason, checked against the installed 1.19 files:
#
#   - common/governments/_governments.info documents the "administrative"
#     government rule as the thing that lets a top liege "have landless
#     vassal who are house heads of noble families". Landless vassalage
#     is tied to the LIEGE having administrative government. In the
#     Worms 1066 scenario the host is Bishop Siegfried, a theocratic
#     vassal of the feudal HRE -- not an administrative realm. The
#     vassal route would therefore require either converting the HRE to
#     administrative government, or setting administrative = yes here,
#     which needs the admin_gov DLC flag and drags in the whole
#     Byzantine mechanic set the spec explicitly does not want.
#
#   - The independent route has direct vanilla precedent for the one
#     thing the spec actually needs from it: acclamation_succession_law
#     in common/laws/00_succession_laws.txt uses
#     order_of_succession = appointment together with
#     is_independent_ruler = yes. Appointment succession therefore
#     demonstrably works WITHOUT a liege doing the appointing, which is
#     what spec section 5 requires.
#
# Consequence: in v1 the host-realm relationship is flavor only (spec
# section 3b's light protection events), which is exactly the scope the
# spec assigns to v1 anyway -- the real host system is Phase 4. Revisit
# when Phase 4 lands, or if the "By God Alone" ecclesiastical titles
# noted in ROADMAP.md turn out to be a better substrate.
# ---------------------------------------------------------------------
#
# NOT YET VERIFIED IN A RUNNING GAME. See
# docs/implementation/v1-kehillah-implementation.md for everything that
# still needs a playtest to confirm.
# =====================================================================

kehillah_government = {
	government_rules = {
		# Landless and playable -- the whole point. Requires the
		# landless_playable DLC feature (Roads to Power), same as the
		# adventurer and administrative governments.
		landless_playable = yes

		# The community leader is a real person with a lineage, and
		# notable families matter (spec section 3, step 5).
		rulers_should_have_dynasty = yes

		# ...but the community is not a dynastic possession. Leadership
		# passes by appointment on merit (spec section 5), so no cadet
		# branches and no dynasty-named realms: the Kehillah of Worms
		# is never "the HaLevi Realm".
		create_cadet_branches = no
		dynasty_named_realms = no

		# Council ON since 2026-10-01 (docs/spec/v26-kehillah-council-
		# and-baalei-batim.md). It was off on the grounds that landless
		# rulers have no counties for councillor tasks to act on, as with
		# landless_adventurer. 1.20 undercuts that: task_type_general
		# tasks need no county, and ecclesiastical/admin landless rulers
		# run councils. The Kehillah seats its own lay board
		# (common/council_positions/kehillah_council_positions.txt);
		# vanilla's five classic seats are excluded for it in
		# kehillah_vanilla_council_overrides.txt. The salaried officers
		# remain court positions (kehillah_officers.txt). Councillors and
		# officers are both in the succession candidate pool
		# (holder_councilor, holder_court_position), so a board member
		# succeeding the old rabbi is now a real outcome even before the
		# quarter's office buildings exist.
		council = yes

		# Not a dynastic government. This blocks Kehillah leaders from
		# inheriting feudal or clan realms sideways, which would
		# instantly break the premise.
		inherit_from_dynastic_government = no

		# No passive prestige from title tier -- a Kehillah is not a
		# duchy in any meaningful sense, it only borrows the tier.
		use_title_tier_modifiers = no

		# RULES DELIBERATELY OMITTED RATHER THAN SET TO "no":
		#   royal_court, use_as_base_on_landed, use_as_base_on_rank_up,
		#   legitimacy
		#
		# The 1.19 parser rejects these with "Unexpected enum found"
		# when written as "= no", and that rejection desynced the whole
		# file on the first playtest, taking the character_modifier and
		# flag blocks down with it. Vanilla never writes any of these
		# four as "no" -- only ever "yes" -- which is the tell.
		#
		# Omitting them gives the behaviour we wanted anyway: all three
		# of royal_court, use_as_base_on_landed and use_as_base_on_rank_up
		# default to off. legitimacy defaults to ON, which we did not
		# want, but it is inert here because legitimacy only applies
		# when a legitimacy type is valid for the ruler and none is
		# defined for this government.
		#
		# Rule of thumb for anything added here later: only write a rule
		# as "no" if vanilla writes that same rule as "no" somewhere in
		# 00_government_types.txt. Every "no" in this block passes that
		# test. Otherwise omit it and rely on the default.

		# The community has no standing army and pays no MaA upkeep.
		use_maa_maintenance = no

		# Spouses still generate normally -- family life is a live part
		# of the fantasy.
		court_generate_spouses = yes

		# Live 1.19 expresses this as a government rule, not as an
		# active_accolades character modifier.
		allow_accolades = no
	}

	domicile_type = kehillah_quarter

	# Deliberately NOT a fallback government. Nothing should land in
	# this government type by accident; it is only ever assigned
	# explicitly, via title history today and via a found-a-community
	# decision in a later phase.

	can_get_government = {
		# Keep this tier-agnostic. A previous title_tier = duchy gate
		# invalidated the county-tier Worms start at a monthly government
		# check (2026-09-20 playtest correction). All community titles are
		# duchy-tier as of 2026-09-20, but the actual government criterion
		# is that the leader holds a landless title. Do not reintroduce a
		# rank gate as part of this retier.
		any_held_title = {
			is_landless_type_title = yes
		}
	}

	# Only relevant if a Kehillah ever ends up holding a holding, which
	# v1 never does. Listed so the game has a sane answer rather than
	# none.
	valid_holdings = { city_holding }

	court_generate_commanders = no

	character_modifier = {
		# A diaspora community is not a military power (spec section 3,
		# the "Martial is deliberately not a full pillar" note).
		levy_size = -1
		men_at_arms_cap = -10
		men_at_arms_limit = -10
		knight_limit = -10

		# Competence and knowledge, not scheming (spec section 2).
		# These two lines are the mechanical statement of the player
		# fantasy.
		monthly_learning_lifestyle_xp_gain_mult = 0.15
		monthly_stewardship_lifestyle_xp_gain_mult = 0.15

		# Leadership turns over by appointment and elder sages get
		# appointed late in life, so reigns are often short. Soften the
		# short-reign penalty rather than punishing normal play.
		short_reign_duration_mult = -0.5
	}

	ai = {
		use_goals = no
		use_legends = no
		perform_religious_reformation = no
	}

	# Use flags rather than has_government where possible, per vanilla's
	# own guidance in 00_government_types.txt.
	flags = {
		government_is_kehillah

	# government_has_influence REMOVED 2026-09-07. It did what it was
	# supposed to (confirmed live: it works fine off this government),
	# but playtest feedback was that Influence reads to players as
	# Byzantine/administrative political-scheme currency regardless of
	# what this mod actually spends it on, because that is what the
	# resource means everywhere else it appears. Replaced with two
	# tracked variables on the title itself (not the character --
	# d_kehillah_worms IS the community, same principle as the building-
	# continuity system), kehillah_var_stability and
	# kehillah_var_greatness -- see kehillah_quarterly_pillars_effect in
	# common/scripted_effects/kehillah_scripted_effects.txt for the full
	# reasoning and docs/implementation/v1-kehillah-implementation.md for
	# the decision record. Prosperity needed no equivalent: it was
	# always read off Gold, never off Influence.

	# Keeps the community independent and un-vassalizable. This is the
	# same code-referenced flag vanilla puts on
	# landless_adventurer_government. Without it a host realm could
	# absorb the Kehillah as a vassal, and the prototype would fall
	# apart in exactly the ways Phase 4 is meant to handle on purpose.
		cannot_be_vassal_or_liege
	}

	# ADDED 2026-10-09. CK3 1.20 only runs appointment succession on a
	# title at or above the government's min_appointment_tier
	# (_governments.info: "Title tier that enables appointment succession
	# mechanics"); every vanilla appointment government sets it. Without
	# it the Kehillah's appointment law had no effect: the succession
	# window listed no candidates and the title passed to the leader's
	# eldest daughter by plain inheritance, ignoring kehillah_leadership's
	# male-only lock and merit scoring (live 2026-10-01 and 2026-10-09).
	# County, as the theocratic governments use; community titles are
	# duchy tier.
	min_appointment_tier = county

	# ADDED 2026-10-09 (Shabbat heartbeat 1). THE TRAIT-FREE CLERICAL
	# CONTINUATION FIX -- ROADMAP item H1. READ BEFORE CHANGING.
	#
	# Problem: CK3 1.20 only hands a player a NON-DYNASTIC heir when the
	# dying character belongs to a faith organization -- vanilla's own
	# words, ECCLESIASTICAL_RULER_SUCCESSION_DESCRIPTION: "When dying as
	# a member of an Organization, your Heir will be the next person to
	# receive your Primary Title. This is usually not a member of your
	# Dynasty." The Kehillah's appointed successor is almost always from
	# another dynasty, so without this the player gets Game Over the
	# moment a leader dies (live 2026-10-09, docs/testing/
	# 2026-10-09-clerical-succession-live-test-log.md).
	#
	# A trait with clergy = yes fixes it, because clergy "automatically
	# join faith organization" (common/traits/_traits.info). That is why
	# kehillah_rabbi_trait got clergy = yes and why the RABBINIC path
	# already passes. It does nothing for a LAY (Parnas) leader, and
	# Daniel's instruction was explicitly to solve the lay case WITHOUT
	# adding a trait.
	#
	# A scripted join_organization is NOT a substitute. It does make the
	# character a member for that instant (probe kehillah_household_
	# debug.7 confirmed "PASS lay leader is a member"), but he still had
	# no player heir, and after an appointment succession all three lay
	# appointees read as non-members again. Vanilla says why, in
	# common/on_action/game_start.txt: "A faith's organization members
	# are exactly its clergy, so this never touches the laity." The
	# engine keeps membership synced to clergy status, so a script join
	# onto a non-clergy character is reverted and buys nothing.
	#
	# The trait-free lever is this one line. _governments.info documents
	# mechanic_type = theocracy as, verbatim, "Should rulers be
	# considered clergy". That is the mechanism by which vanilla's
	# non-monk clerical rulers -- bishops under ecclesiastical_government
	# and theocracy_government holders, none of whom carry the devoted
	# (monk) trait -- are clergy at all. Both of those governments set
	# it; nothing else about them is needed here.
	#
	# WHAT THIS DELIBERATELY DOES NOT DO. mechanic_type is not the
	# government_is_theocracy FLAG, and that distinction is the whole
	# reason this is safe. Vanilla gates its theocratic *content* on the
	# flags (government_is_theocracy, government_is_ecclesiastical,
	# government_uses_church_authority), which this government does not
	# have and must not get:
	#   - church_authority law group needs the flag (common/law_groups/
	#     00_realm_law_groups.txt) -- not gained.
	#   - bishop_theocratic_succession_law needs the flag
	#     (can_have_bishop_theocratic_succession_law_trigger) -- not
	#     gained.
	#   - theocratic_elective_succession_law needs the
	#     has_clerical_electors doctrine, which rabbinism does not have
	#     -- not gained.
	# Only eight places in all of vanilla read government_has_mechanic =
	# theocracy, and seven are harmless here (AI head-of-faith request
	# availability, PAM claim-beseeching, the church-treasury budget
	# events, which need has_treasury = yes, and the administrative
	# puppet letter). One is actively WANTED: common/important_actions/
	# 00_inheritance_actions.txt suppresses the "your primary heir is of
	# another dynasty" nag for exactly this mechanic.
	#
	# THE ONE REAL COLLISION, and its guard: vanilla's
	# clerical_appointment_law has should_start_with =
	# should_use_clerical_appointment_law_trigger, which is only
	# `government_has_mechanic = theocracy` plus a county-tier floor. Per
	# common/laws/_laws.info, default laws are "checked in definition
	# order" (NOT index order -- index is display order only), and
	# vanilla's file loads before this mod's, so clerical_appointment_law
	# would be checked first and would take the Kehillah's succession
	# law slot away from kehillah_appointment_succession_law. See
	# common/scripted_triggers/kehillah_clerical_appointment_law_override.txt
	# for the one-branch fork that excludes this government. Do not
	# remove that file while this line is here.
	mechanic_type = theocracy

	color = { 58 84 130 }
}
