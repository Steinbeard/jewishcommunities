# =====================================================================
# THE NASI'S GOVERNMENT (V47 phase B,
# docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).
#
# The government of the Nasi of the Sanhedrin, who holds the seat county
# in the Land as Rabbinic Judaism's head of faith. LANDED, which is the
# point: live tests on 2026-10-19 showed CK3 forces a landless head of
# faith onto vanilla theocracy (refusing any Kehillah government and
# stripping his Quarter), while a landed Nasi on this government kept his
# Quarter and stayed a playable temporal head of faith.
#
# kehillah_government copied field for field, plus mechanic_type =
# theocracy, the government_is_kehillah_nasi flag, castle/city/temple
# holdings, and no cannot_be_vassal_or_liege (the Nasi holds his county of
# his liege). It keeps domicile_type = kehillah_quarter and the
# government_is_kehillah flag, so community systems and playability still
# treat him as a community leader. It does NOT carry government_is_
# theocracy (the council overrides and vanilla theocracy content key on it).
# Daniel plans no pillars for it (V47); not yet done.
#
# kehillah_reconcile_nasi_government_effect moves characters on and off
# it. Keep the shared fields in sync with kehillah_government.txt.
# =====================================================================
kehillah_nasi_government = {
	government_rules = {
		landless_playable = yes
		rulers_should_have_dynasty = yes
		create_cadet_branches = no
		dynasty_named_realms = no
		council = yes
		inherit_from_dynastic_government = no
		use_title_tier_modifiers = no
		use_maa_maintenance = no
		court_generate_spouses = yes
		allow_accolades = no
	}

	domicile_type = kehillah_quarter
	mechanic_type = theocracy

	# V47 phase B: the holder of the Sanhedrin's seat county in the Land. A
	# LANDED government (the spike of 2026-10-19 proved a landed Nasi on it
	# keeps his Quarter and stays a playable head of faith), so it accepts
	# the county's holdings and may be the granting liege's vassal.
	can_get_government = {
		faith = faith:rabbinic_faith
		kehillah_is_sanhedrin_head_trigger = yes
	}

	valid_holdings = { castle_holding city_holding church_holding }

	court_generate_commanders = no

	character_modifier = {
		levy_size = -1
		men_at_arms_cap = -10
		men_at_arms_limit = -10
		knight_limit = -10
		monthly_learning_lifestyle_xp_gain_mult = 0.15
		monthly_stewardship_lifestyle_xp_gain_mult = 0.15
		short_reign_duration_mult = -0.5
		monthly_piety = 0.5
	}

	ai = {
		use_goals = yes
		use_legends = yes
		perform_religious_reformation = no
	}

	# No cannot_be_vassal_or_liege: the Nasi holds his county of his liege.
	flags = {
		government_is_kehillah
		government_is_kehillah_nasi
	}

	min_appointment_tier = county
	color = { 230 205 120 }
}
