# =====================================================================
# THE NASI'S GOVERNMENT (V46, docs/spec/v46-the-sanhedrin.md section 9).
#
# Why it exists. Live tests on 2026-10-19 showed that CK3 1.20 forces a
# LANDLESS head of faith onto a theocracy government, whatever the
# head-of-faith and clergy doctrines. change_government back to
# kehillah_government is refused ("Trying to set illegal government").
# Vanilla's own landless head of faith, the Pope, survives this because
# his government (ecclesiastical_government) has mechanic_type =
# theocracy, is landless_playable and has its own domicile, and his
# religion names it as its theocracy_government_type. This is the same
# pattern for the Nasi: rabbinic_faith's theocracy_government_type points
# here, so the engine's forced theocracy lands on a Kehillah.
#
# What it is. kehillah_government, copied field for field, plus
# mechanic_type = theocracy and the government_is_kehillah_nasi flag. It
# keeps:
#   - domicile_type = kehillah_quarter, so the Nasi keeps his Quarter
#     (to be confirmed live: section 9's first check);
#   - the government_is_kehillah flag, so every community system, from
#     pillars and succession to playability (is_playable_character
#     override) and the laws' can_have, still treats him as a community
#     leader.
#
# What it deliberately does NOT carry: the government_is_theocracy flag.
# The mod's council overrides and vanilla theocracy content key on that
# flag; the engine's head-of-faith rule only needs the mechanic.
#
# Who has it. Only a landless Nasi, and only while Nasi.
# kehillah_reconcile_nasi_government_effect moves characters on and off
# it. Keep the shared fields in sync with kehillah_government.txt.
# =====================================================================
kehillah_nasi_government = {
	government_rules = {
		landless_playable = yes
		rulers_should_have_dynasty = yes
		create_cadet_branches = no
		dynasty_named_realms = no
		council = yes
		inherit_from_dynastic_government = no
		use_title_tier_modifiers = no
		use_maa_maintenance = no
		court_generate_spouses = yes
		allow_accolades = no
	}

	domicile_type = kehillah_quarter
	mechanic_type = theocracy

	# A community leader who holds the Sanhedrin.
	can_get_government = {
		any_held_title = {
			is_landless_type_title = yes
		}
		has_title = title:d_kehillah_sanhedrin
	}

	valid_holdings = { city_holding }

	court_generate_commanders = no

	character_modifier = {
		levy_size = -1
		men_at_arms_cap = -10
		men_at_arms_limit = -10
		knight_limit = -10
		monthly_learning_lifestyle_xp_gain_mult = 0.15
		monthly_stewardship_lifestyle_xp_gain_mult = 0.15
		short_reign_duration_mult = -0.5
		monthly_piety = 0.5
	}

	ai = {
		use_goals = yes
		use_legends = yes
		perform_religious_reformation = no
	}

	flags = {
		government_is_kehillah
		government_is_kehillah_nasi
		cannot_be_vassal_or_liege
	}

	min_appointment_tier = county
	color = { 230 205 120 }
}
