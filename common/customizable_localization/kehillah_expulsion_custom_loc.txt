# KehillahExpulsionTimeLeft: scope a Kehillah leader with an edict
# pending. How long remains before it is enforced, in plain words.
KehillahExpulsionTimeLeft = {
	type = character
	text = {
		trigger = { primary_title.var:kehillah_expulsion_quarters ?= 0 }
		localization_key = kehillah_expulsion_time_year
	}
	text = {
		trigger = { primary_title.var:kehillah_expulsion_quarters ?= 1 }
		localization_key = kehillah_expulsion_time_nine_months
	}
	text = {
		trigger = { primary_title.var:kehillah_expulsion_quarters ?= 2 }
		localization_key = kehillah_expulsion_time_six_months
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_expulsion_time_months
		fallback = yes
	}
}
