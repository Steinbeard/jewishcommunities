# Jewish Feast presentation stays additive to vanilla activity script.
# These pickers are called only by the Feast localization overrides below.
# Non-Jewish hosts resolve to their original vanilla wording and food pickers.

KehillahFeastInvitation = {
    type = character
    text = {
        trigger = { is_jewish_character_trigger = yes }
        localization_key = jewish_holiday_feast_invitation
    }
    text = {
        trigger = { always = no }
        localization_key = jewish_holiday_feast_invitation_vanilla
        fallback = yes
    }
}

KehillahFeastHostEnd = {
    type = character
    text = {
        trigger = {
            is_jewish_character_trigger = yes
            involved_activity = {
                has_activity_type = activity_feast
                has_activity_option = {
                    category = special_type
                    option = feast_type_generic
                }
            }
        }
        localization_key = jewish_holiday_feast_end
    }
    text = {
        trigger = { always = no }
        localization_key = jewish_holiday_feast_end_vanilla
        fallback = yes
    }
}
