# =====================================================================
# Faith organization names (Daniel, 2026-10-11).
#
# Vanilla names every faith's organization with one global string,
# FAITH_ORGANIZATION_NAME = "[FAITH.GetAdjectiveNoTooltip] Clergy"
# (localization/english/organizations_l_english.yml). That read as
# "Rabbinic Clergy" -- a churchly word for the body of rabbis and
# dayyanim. localization/english/replace/kehillah_organization_names_
# l_english.yml points that string at this custom loc instead; every
# faith not named here falls through to the vanilla wording.
#
# Hakhamim ("the Sages") is the period word for the learned of the
# community, and the Karaites used it for their scholars too.
# =====================================================================

KehillahFaithOrganizationName = {
	type = faith

	text = {
		trigger = { this = faith:rabbinic_faith }
		localization_key = kehillah_faith_organization_rabbinism
	}
	text = {
		trigger = { this = faith:karaism }
		localization_key = kehillah_faith_organization_karaism
	}
	text = {
		fallback = yes
		localization_key = kehillah_faith_organization_vanilla
	}
}
