# =====================================================================
# Responsa question selector. ROADMAP S5, added 2026-09-27.
#
# type = character, and correctly so: var:kehillah_responsa_question
# lives on the LEADER who received the letter, not on the title. It is
# per-question working state for one open event, not part of the
# community's record -- the record is kehillah_var_responsa_count, which
# IS on the title (see kehillah_responsa_effects.txt's header for that
# split). Called as [ROOT.Char.Custom('KehillahResponsaQuestion')] from
# kehillah_responsa.0001.desc.
#
# has_variable guards EVERY branch, not just the first, per the
# 2026-09-26 heartbeat-4 lesson: an unguarded variable read in a
# custom-loc branch throws every time the tooltip system rebuilds the
# text, which in that case meant roughly 2,000 error.log lines a second
# while the player merely hovered.
#
# The fallback is a real question, not an empty string. If the variable
# somehow is not set, an event whose whole body is the question must not
# render a blank space where the question goes -- the oven case is the
# most self-contained of the six, so it is the safe default.
# =====================================================================

KehillahResponsaQuestion = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:oven
		}
		localization_key = kehillah_responsa_question_oven
	}
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:agunah
		}
		localization_key = kehillah_responsa_question_agunah
	}
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:wine
		}
		localization_key = kehillah_responsa_question_wine
	}
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:orphan
		}
		localization_key = kehillah_responsa_question_orphan
	}
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:sabbath_lamp
		}
		localization_key = kehillah_responsa_question_sabbath_lamp
	}
	text = {
		trigger = {
			has_variable = kehillah_responsa_question
			var:kehillah_responsa_question = flag:convert
		}
		localization_key = kehillah_responsa_question_convert
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_responsa_question_oven
		fallback = yes
	}
}
