# 2026-11-10, Daniel. The threat-grade tooltip in the crisis window (kehillah_crisis_grade_tt).
# Scope: the community's leader (Situation.Var('kehillah_crisis_title').Title.GetHolder), the
# same path the situation description uses. The grade names come from
# kehillah_crisis_chronicle_custom_loc.txt (KehillahArmedGradeBefore, KehillahStruckGradeBefore).

# How the grade got here: the reckoning that last lowered it.
KehillahCrisisGradeHistory = {
 type = character
 text = {
  trigger = { primary_title ?= { var:kehillah_crisis_struck_grade_before > var:kehillah_crisis_threat_grade } }
  localization_key = kehillah_crisis_grade_history_struck
 }
 text = {
  trigger = { primary_title ?= { var:kehillah_crisis_armed_grade_before > var:kehillah_crisis_threat_grade } }
  localization_key = kehillah_crisis_grade_history_armed
 }
 text = { trigger = { always = yes } localization_key = kehillah_crisis_grade_history_none fallback = yes }
}

# What can still change it, by phase. Quarter struck and Recovery have no reckoning left.
KehillahCrisisGradeOutlook = {
 type = character
 text = {
  trigger = { primary_title.var:kehillah_crisis_situation ?= { situation_current_phase = kehillah_crisis_warning } }
  localization_key = kehillah_crisis_grade_outlook_warning
 }
 text = {
  trigger = { primary_title.var:kehillah_crisis_situation ?= { situation_current_phase = kehillah_crisis_armed } }
  localization_key = kehillah_crisis_grade_outlook_armed
 }
 text = { trigger = { always = yes } localization_key = kehillah_crisis_grade_outlook_none fallback = yes }
}
