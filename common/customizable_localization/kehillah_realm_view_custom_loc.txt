# Their rite's teaching on the Jews, by name (V31 doctrines; Western
# Christian, Eastern Christian and Sunni rites without an explicit
# teaching default to Witness, Imperial Law and Dhimma).
KehillahRvTeaching = {
	type = character
	text = { trigger = { rite ?= { kehillah_rite_teaching_witness_trigger = yes } } localization_key = kehillah_rv_teaching_witness }
	text = { trigger = { rite ?= { kehillah_rite_teaching_imperial_law_trigger = yes } } localization_key = kehillah_rv_teaching_imperial_law }
	text = { trigger = { rite ?= { kehillah_rite_teaching_dhimma_trigger = yes } } localization_key = kehillah_rv_teaching_dhimma }
	text = { trigger = { rite ?= { rite_has_doctrine = kehillah_teaching_separation_mission } } localization_key = kehillah_rv_teaching_separation_mission }
	text = { trigger = { rite ?= { rite_has_doctrine = kehillah_teaching_strict_pact } } localization_key = kehillah_rv_teaching_strict_pact }
	text = { trigger = { rite ?= { rite_has_doctrine = kehillah_teaching_exclusion } } localization_key = kehillah_rv_teaching_exclusion }
	text = { trigger = { rite ?= { rite_has_doctrine = kehillah_teaching_abolition } } localization_key = kehillah_rv_teaching_abolition }
	text = { trigger = { always = yes } localization_key = kehillah_rv_teaching_none fallback = yes }
}

# The policy map's hover line (2026-10-28). Scope: a realm's primary
# title. A ruler who cannot hold the settlement law sets no policy; the
# map paints that grey (kehillah_pick_realm_policy_map_color_effect).
KehillahPolicyMapLine = {
	type = landed_title
	text = {
		trigger = {
			NOT = { has_variable = kehillah_jewish_settlement_policy }
			holder ?= { NOT = { kehillah_can_hold_settlement_law_trigger = yes } }
		}
		localization_key = kehillah_policy_map_line_none
	}
	text = { trigger = { kehillah_jewish_settlement_policy_encouraged_trigger = yes } localization_key = kehillah_settlement_policy_encouraged }
	text = { trigger = { kehillah_jewish_settlement_policy_banned_trigger = yes } localization_key = kehillah_settlement_policy_banned }
	text = { trigger = { kehillah_jewish_settlement_policy_discouraged_trigger = yes } localization_key = kehillah_settlement_policy_discouraged }
	text = { trigger = { always = yes } localization_key = kehillah_settlement_policy_allowed fallback = yes }
}
