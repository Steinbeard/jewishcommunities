# =====================================================================
# The "New Realm Priest" notice, re-flavored for a Kehillah's Dayyan
# (Daniel, 2026-10-11).
#
# CK3 sends msg_new_theocracy_lesee_approve / _disapprove (common/
# messages/01_religious_messages.txt) whenever a court chaplain is
# seated, with the new chaplain as LEFT_CHARACTER. Its title and text are
# global strings about a "Realm Priest" and "church property", neither of
# which a Kehillah has. localization/english/replace/
# kehillah_realm_priest_message_l_english.yml points those strings here;
# every realm that is not a Kehillah falls through to vanilla's wording.
#
# Scope: the new chaplain (LEFT_CHARACTER).
# =====================================================================

KehillahRealmPriestMsgTitle = {
	type = character
	text = {
		trigger = { liege ?= { government_has_flag = government_is_kehillah } }
		localization_key = kehillah_dayyan_msg_title
	}
	text = {
		fallback = yes
		localization_key = kehillah_realm_priest_msg_title_vanilla
	}
}
