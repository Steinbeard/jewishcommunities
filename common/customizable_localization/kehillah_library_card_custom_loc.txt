# The county view's Jewish cards (2026-11-08): the yeshiva line on the
# community card and the Beit Midrash card. Scope: a library or community
# title with a yeshiva (kehillah_card_has_yeshiva_trigger). Record:
# docs/implementation/2026-11-08-library-cards.md.
# =====================================================================

# Its fame tier (the thresholds of KehillahYeshivaFameTier), or dormant.
KehillahCardYeshiva = {
	type = landed_title
	text = {
		trigger = { has_variable = kehillah_yeshiva_dormant }
		localization_key = kehillah_card_yeshiva_dormant
	}
	text = {
		trigger = {
			has_variable = kehillah_yeshiva_fame
			var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_illustrious
		}
		localization_key = kehillah_yeshiva_tier_illustrious
	}
	text = {
		trigger = {
			has_variable = kehillah_yeshiva_fame
			var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_renowned
		}
		localization_key = kehillah_yeshiva_tier_renowned
	}
	text = {
		trigger = {
			has_variable = kehillah_yeshiva_fame
			var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_respected
		}
		localization_key = kehillah_yeshiva_tier_respected
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_yeshiva_tier_modest
		fallback = yes
	}
}

# Who heads it: a landed yeshiva's Rosh Yeshiva, a community's appointed
# one, or its leader.
KehillahCardYeshivaHead = {
	type = landed_title
	text = {
		trigger = { exists = var:kehillah_landed_rosh }
		localization_key = kehillah_card_yeshiva_head_landed
	}
	text = {
		trigger = { holder ?= { employs_court_position = rosh_yeshiva_kehillah_position } }
		localization_key = kehillah_card_yeshiva_head_appointed
	}
	text = {
		trigger = { holder ?= { kehillah_ordained_or_karaite_trigger = yes } }
		localization_key = kehillah_card_yeshiva_head_leader
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_card_yeshiva_head_none
		fallback = yes
	}
}
