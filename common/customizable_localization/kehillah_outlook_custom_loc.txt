KehillahOutlookBand = {
	type = character
	text = {
		trigger = { kehillah_outlook_on_jews_value >= 40 }
		localization_key = kehillah_outlook_band_protective
	}
	text = {
		trigger = { kehillah_outlook_on_jews_value >= 15 }
		localization_key = kehillah_outlook_band_well_disposed
	}
	text = {
		trigger = { kehillah_outlook_on_jews_value <= -40 }
		localization_key = kehillah_outlook_band_hostile
	}
	text = {

		localization_key = kehillah_outlook_band_cold
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_outlook_band_indifferent
	}
}

# Scope: a county offered to settlers (kehillah_settlers.0001). Why it
# draws them: an open invitation, or a realm that encourages Jews.
KehillahSettlersDestinationNote = {
	type = landed_title
	text = {
		trigger = { holder.top_liege.primary_title ?= { has_variable = kehillah_invitation_open } }
		localization_key = kehillah_settlers_dest_note_invited
	}
	text = {
		trigger = { holder.top_liege.primary_title ?= { kehillah_jewish_settlement_policy_encouraged_trigger = yes } }
		localization_key = kehillah_settlers_dest_note_encouraged
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_settlers_dest_note_none
	}
}

# V37 (2026-10-13). Scope: a Kehillah leader. The accuser's tale: the
# blood libel under a Christian lord, blasphemy under a Muslim one.
KehillahAccusationTale = {
	type = character
	text = {
		trigger = { domicile.domicile_location.county.holder.top_liege.faith.religion ?= religion:islam_religion }
		localization_key = kehillah_accusation_tale_blasphemy
	}
	text = {
		trigger = { has_variable = kehillah_accusation_desecration }
		localization_key = kehillah_accusation_tale_desecration
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_accusation_tale_blood_libel
	}
}

# V40 slice 1: the host's faith, alongside his teaching in the HUD.
# Other religions have no council/teaching system; omit their line.
KehillahHostFaithIncidents = {
	type = character
	text = {
		trigger = {
			OR = {
				faith.religion = religion:christianity_religion
				faith.religion = religion:islam_religion
			}
		}
		localization_key = kehillah_host_faith_incidents
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_host_faith_incidents_none
	}
}
