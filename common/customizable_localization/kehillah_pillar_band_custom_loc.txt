# =====================================================================
# Kehillah pillar band names.
#
# THE PROBLEM. v1 spec section 3's shared-rules list asks for "a visible
# score, a qualitative band ... and a compact breakdown" for each
# pillar. kehillah_view_standing_decision (common/decisions/
# kehillah_decisions.txt) already shows the raw numbers via dynamic
# localization ([ROOT.GetPrimaryTitle.Var('...').GetValue|V0]), but a
# raw number alone is not the qualitative band v1 spec asked for, and
# plain localization has no conditional branching of its own -- there is
# no in-string "if score >= X show Y" construct.
#
# THE FIX. The same customizable_localization system this mod already
# uses for KehillahDomicileName (common/customizable_localization/
# kehillah_domicile_custom_loc.txt) -- and which vanilla itself uses for
# tiered text keyed to a numeric variable threshold, e.g.
# common/customizable_localization/00_artifact_court_custom_loc.txt's
# book-quality tiers, which check "var:book_content_quality >= 80" the
# same shape as the checks below. type = landed_title is used rather
# than character (unlike KehillahDomicileName) because these are invoked
# as [ROOT.GetPrimaryTitle.Custom('...')] -- GetPrimaryTitle returns a
# landed_title, and the pillar variables live on that title, not the
# character (see kehillah_quarterly_pillars_effect's "WHY THE TITLE, NOT
# THE CHARACTER" note, common/scripted_effects/
# kehillah_scripted_effects.txt).
#
# One function per pillar, not one shared parameterized function:
# customizable_localization functions take no caller-supplied parameter
# the way a scripted trigger's $PILLAR$ can (confirmed absent from
# common/customizable_localization/_custom_loc.info's own format
# section -- "type"/"parent"/"suffix" are the only per-entry options),
# so each pillar's variable name has to be written out in its own
# function rather than substituted in.
#
# Checked highest band first, matching the artifact book-quality
# precedent's own ordering (>= 80 checked before the lower tiers) --
# the first matching text wins, so band checks must run top-down.
# Thresholds are common/script_values/kehillah_script_values.txt's
# kehillah_band_*_threshold values (v2 spec section 1.1), not
# reinvented here.
# =====================================================================

KehillahProsperityBandName = {
	type = landed_title

	text = {
		trigger = { var:kehillah_var_prosperity >= kehillah_band_legendary_threshold }
		localization_key = kehillah_band_legendary
	}
	text = {
		trigger = { var:kehillah_var_prosperity >= kehillah_band_flourishing_threshold }
		localization_key = kehillah_band_flourishing
	}
	text = {
		trigger = { var:kehillah_var_prosperity >= kehillah_band_healthy_threshold }
		localization_key = kehillah_band_healthy
	}
	text = {
		trigger = { var:kehillah_var_prosperity >= kehillah_band_strained_threshold }
		localization_key = kehillah_band_strained
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_band_crisis
		fallback = yes
	}
}

# ---------------------------------------------------------------------
# Kehillah<Pillar>Score -- ADDED 2026-09-15 (v12 section 6f).
#
# The pillar's current value as band-COLOURED text, for anywhere a bare
# number needs to read as good or bad at a glance (first user: the map
# view roster). Same thresholds, same top-down order as the BandName
# functions above; the difference is that each returned key embeds the
# value itself, with a format tag around it, so the caller gets one
# finished string: [Title.Custom('KehillahStandingScore')].
#
# Why the value is embedded in the key rather than concatenated by the
# caller: a format tag opened by one substitution and closed by a
# literal #! after it would depend on the loc parser resolving data
# functions before it parses formatting. Vanilla's own custom-loc keys
# carry complete #X ... #! spans (00_interactions_custom_loc.txt's
# imprison_decline_summary_* family), so a self-contained span is the
# proven shape, at the cost of twenty small keys instead of five.
#
# The colours are vanilla's court-position aptitude ramp
# (gui/preload/textformatting.gui: aptitude_terrible = red through
# aptitude_excellent = green), one step per band, so the roster reads
# with the same red-is-bad instinct every other CK3 screen trains.
# ---------------------------------------------------------------------

KehillahStandingScore = {
	type = landed_title

	text = {
		trigger = { var:kehillah_var_standing >= kehillah_band_legendary_threshold }
		localization_key = kehillah_score_legendary_standing
	}
	text = {
		trigger = { var:kehillah_var_standing >= kehillah_band_flourishing_threshold }
		localization_key = kehillah_score_flourishing_standing
	}
	text = {
		trigger = { var:kehillah_var_standing >= kehillah_band_healthy_threshold }
		localization_key = kehillah_score_healthy_standing
	}
	text = {
		trigger = { var:kehillah_var_standing >= kehillah_band_strained_threshold }
		localization_key = kehillah_score_strained_standing
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_score_crisis_standing
		fallback = yes
	}
}
