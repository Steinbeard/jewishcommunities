# Context for the opening of a book controversy. Most work-specific history
# belongs in the catalogue; this short text tells the player why this dispute
# is happening now, before the later letters ask for a side.
KehillahMachloketContext = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
		}
		localization_key = kehillah_sefer.0101.context_emunot_vedeot
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0101.context_generic
	}
}

KehillahMachloketCriticText = {
	type = character
	text = {
		trigger = { exists = scope:kehillah_machloket_critic }
		localization_key = kehillah_sefer.0101.critic
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0101.no_critic
	}
}

# The decision event must be a dossier, not an unexplained referendum. These
# components identify the book's actual claim, the scholars who circulated
# the letters, and the way it intersects with the reader's and rite's tenets.
KehillahMachloketLettersDossier = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
		}
		localization_key = kehillah_sefer.0102.dossier_emunot_vedeot
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0102.dossier_generic
	}
}

KehillahMachloketLettersScholars = {
	type = character
	text = {
		trigger = {
			exists = scope:kehillah_machloket_champion
			exists = scope:kehillah_machloket_critic
		}
		localization_key = kehillah_sefer.0102.scholars_both
	}
	text = {
		trigger = { exists = scope:kehillah_machloket_critic }
		localization_key = kehillah_sefer.0102.scholars_critic
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0102.scholars_generic
	}
}

KehillahMachloketLettersPersonalStance = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			has_personal_tenet = tenet_baalei_hamiqra
		}
		localization_key = kehillah_sefer.0102.personal_baalei_hamiqra
	}
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			OR = {
				has_personal_tenet = tenet_kehillah_hokhmah
				has_personal_tenet = tenet_talmud_torah
			}
		}
		localization_key = kehillah_sefer.0102.personal_favourable
	}
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
		}
		localization_key = kehillah_sefer.0102.personal_neutral
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0102.personal_generic
	}
}

KehillahMachloketLettersRiteStance = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			rite ?= { rite_has_tenet = tenet_baalei_hamiqra }
		}
		localization_key = kehillah_sefer.0102.rite_baalei_hamiqra
	}
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			rite ?= { has_tenet_status = { tenet = tenet_kehillah_hokhmah status = prohibited } }
		}
		localization_key = kehillah_sefer.0102.rite_hokhmah_prohibited
	}
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			NOT = { rite ?= { has_at_least_tenet_status = { tenet = tenet_kehillah_hokhmah status = core } } }
		}
		localization_key = kehillah_sefer.0102.rite_hokhmah_not_core
	}
	text = {
		trigger = {
			has_variable = kehillah_sefer_named_work
			var:kehillah_sefer_named_work = flag:work_emunot_vedeot
			NOT = { rite ?= { has_at_least_tenet_status = { tenet = tenet_talmud_torah status = core } } }
		}
		localization_key = kehillah_sefer.0102.rite_talmud_torah_not_core
	}
	text = {
		fallback = yes
		localization_key = kehillah_sefer.0102.rite_generic
	}
}
