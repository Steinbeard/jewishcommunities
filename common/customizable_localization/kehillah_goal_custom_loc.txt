# =====================================================================
# COMMUNITY GOALS -- ROADMAP S6, added 2026-09-28.
#
# type = landed_title on every entry here, and every branch carries its
# own has_variable. Both are the standing rule in this mod for anything
# reading a pillar-adjacent variable, and both were learned from the
# 2026-09-26 breakdown bug: title variables read through a
# character-scoped custom loc that hopped back with primary_title fell
# through to their fallback branch AND threw every frame while hovered,
# at roughly 2,000 error.log lines a second. See KehillahBdResponsa's own
# header in kehillah_breakdown_custom_loc.txt for the full account.
#
# Called as [Title.Custom('...')] from the standing tooltip
# (KEHILLAH_BD_STANDING_TOOLTIP) and as
# [ROOT.Char.GetPrimaryTitle.Custom('...')] from the two interface
# messages, which is the idiom kehillah_band_change_*_desc already uses.
# =====================================================================

# The active goal, by name. Ordering is irrelevant here -- the four
# branches are mutually exclusive by construction, since one variable
# holds one flag -- but each still checks has_variable first so that a
# community with no goal reaches the fallback rather than throwing.
KehillahGoalName = {
	type = landed_title
	text = {
		trigger = {
			has_variable = kehillah_var_goal
			var:kehillah_var_goal = flag:yeshiva
		}
		localization_key = kehillah_goal_name_yeshiva
	}
	text = {
		trigger = {
			has_variable = kehillah_var_goal
			var:kehillah_var_goal = flag:renown
		}
		localization_key = kehillah_goal_name_renown
	}
	text = {
		trigger = {
			has_variable = kehillah_var_goal
			var:kehillah_var_goal = flag:rights
		}
		localization_key = kehillah_goal_name_rights
	}
	text = {
		trigger = {
			has_variable = kehillah_var_goal
			var:kehillah_var_goal = flag:daughter
		}
		localization_key = kehillah_goal_name_daughter
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_goal_name_none
		fallback = yes
	}
}

# The standing-tooltip row. Three states, most specific first: a goal is
# running; no goal but this community has finished some before; nothing at
# all, which renders as no row, the same as every other optional
# contributor in that tooltip.
KehillahBdGoal = {
	type = landed_title
	text = {
		trigger = { has_variable = kehillah_var_goal }
		localization_key = kehillah_bd_goal_active
	}
	text = {
		trigger = {
			has_variable = kehillah_var_goals_completed
			var:kehillah_var_goals_completed >= 1
		}
		localization_key = kehillah_bd_goal_record_only
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_bd_none
		fallback = yes
	}
}
