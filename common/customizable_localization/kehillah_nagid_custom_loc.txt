# Scope: the ruler whose office it is. Same mechanics everywhere; the
# name follows the court (section 13). The Exilarch is a later slice.
KehillahNagidOfficeName = {
	type = character
	text = {
		trigger = {
			religion = religion:christianity_religion
			culture ?= { this = culture:portuguese }
		}
		localization_key = kehillah_nagid_name_arrabi_mor
	}
	text = {
		trigger = {
			religion = religion:christianity_religion
			culture ?= { has_cultural_pillar = heritage_iberian }
		}
		localization_key = kehillah_nagid_name_rab_de_la_corte
	}
	text = {
		trigger = {
			religion = religion:christianity_religion
			# The crown of England (vanilla 1.20 has no "english" culture).
			primary_title ?= { this = title:k_england }
		}
		localization_key = kehillah_nagid_name_presbyter
	}
	text = {
		trigger = {
			religion = religion:christianity_religion
			culture ?= { has_cultural_pillar = heritage_central_germanic }
		}
		localization_key = kehillah_nagid_name_hochmeister
	}
	text = {
		trigger = { religion = religion:christianity_religion }
		localization_key = kehillah_nagid_name_master
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_nagid_name_nagid
		fallback = yes
	}
}
