		localization_key = kehillah_nagid_name_rab_de_la_corte
	}
	text = {
		trigger = { kehillah_nagid_name_presbyter_trigger = yes }
		localization_key = kehillah_nagid_name_presbyter
	}
	text = {
		trigger = { kehillah_nagid_name_hochmeister_trigger = yes }
		localization_key = kehillah_nagid_name_hochmeister
	}
	text = {
		trigger = { kehillah_nagid_name_master_trigger = yes }
		localization_key = kehillah_nagid_name_master

	text = {
		trigger = { always = yes }
		localization_key = kehillah_nagid_name_nagid
		fallback = yes
	}

# Scope: a ruler. How well his Nagid serves: the tier of the service
# modifier he carries (kehillah_nagid_served).
KehillahNagidServiceTierName = {
	type = character
	text = {
		trigger = { has_variable = kehillah_nagid_served var:kehillah_nagid_served >= 4 }
		localization_key = kehillah_nagid_service_tier_4
	}
	text = {
		trigger = { has_variable = kehillah_nagid_served var:kehillah_nagid_served >= 3 }
		localization_key = kehillah_nagid_service_tier_3
	}
	text = {
		trigger = { has_variable = kehillah_nagid_served var:kehillah_nagid_served >= 2 }
		localization_key = kehillah_nagid_service_tier_2
	}
	text = {
		trigger = { has_variable = kehillah_nagid_served }
		localization_key = kehillah_nagid_service_tier_1
	}
	text = { trigger = { always = yes } localization_key = kehillah_nagid_service_none fallback = yes }
}

# Scope: a Nagid, or a ruler whose office is filled. The badge's word:
# the service, or the office's name when there is none.
KehillahNagidChipText = {
	type = character
	text = {
		trigger = {
			kehillah_is_nagid_trigger = yes
			var:kehillah_nagid_of ?= { kehillah_nagid_in_service_trigger = yes }
		}
		localization_key = kehillah_nagid_chip_nagid_service
	}
	text = {
		trigger = {
			kehillah_is_nagid_trigger = yes
			exists = var:kehillah_nagid_of
		}
		localization_key = kehillah_nagid_chip_nagid_office
	}
	text = {
		trigger = { kehillah_nagid_in_service_trigger = yes }
		localization_key = kehillah_nagid_chip_ruler_service
	}
	text = { trigger = { always = yes } localization_key = kehillah_nagid_chip_ruler_office fallback = yes }
}
