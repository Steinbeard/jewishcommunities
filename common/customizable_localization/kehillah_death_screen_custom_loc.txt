# Kehillah death-screen lines. Vanilla selects the existing intro, outro and
# heir keys; these wrappers replace only feudal-sounding lines for characters
# who have led a Kehillah. kehillah_leader_flag is set when office begins and
# remains on the deceased character after succession. Other characters get
# the exact vanilla text copied below.
KehillahSuccessionHeirMotion = {
	type = character
	text = {
		trigger = { has_character_flag = kehillah_leader_flag }
		localization_key = kehillah_succession_heir_motion
	}
	text = {
		trigger = { always = no }
		localization_key = kehillah_vanilla_succession_heir_motion
		fallback = yes
	}
}

KehillahSuccessionText = {
	type = character
	text = {
		trigger = { has_character_flag = kehillah_leader_flag }
		localization_key = kehillah_succession_text
	}
	text = {
		trigger = { always = no }
		localization_key = kehillah_vanilla_succession_text
		fallback = yes
	}
}
