# Ba'alei batim (V26 part 2). Scope: a household head (character), read
# in kehillah_household.0010. Standing thresholds are in
# common/script_values/kehillah_household_values.txt.

KehillahHouseholdBand = {
	type = character
	text = {
		trigger = {
			has_variable = kehillah_household_standing
			var:kehillah_household_standing >= kehillah_household_standing_leading_threshold
		}
		localization_key = kehillah_household_band_leading
	}
	text = {
		trigger = {
			has_variable = kehillah_household_standing
			var:kehillah_household_standing >= kehillah_household_standing_respected_threshold
		}
		localization_key = kehillah_household_band_respected
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_household_band_minor
		fallback = yes
	}
}

KehillahHouseholdTrade = {
	type = character
	text = {
		trigger = { house ?= { var:kehillah_household_trade ?= flag:scholar } }
		localization_key = kehillah_household_trade_scholar
	}
	text = {
		trigger = { house ?= { var:kehillah_household_trade ?= flag:merchant } }
		localization_key = kehillah_household_trade_merchant
	}
	text = {
		trigger = { house ?= { var:kehillah_household_trade ?= flag:advocate } }
		localization_key = kehillah_household_trade_advocate
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_household_trade_none
		fallback = yes
	}
}
