# Mutual aid (2026-11-20, v2). Status lines in the pillar breakdown
# tooltips (localization/english/kehillah_breakdown_l_english.yml): what a
# community of this Prosperity owes the others, and its name for giving.

# Scope: the community title. Under Prosperity.
KehillahBdAidDuty = {
	type = landed_title
	text = {
		trigger = {
			has_variable = kehillah_var_prosperity
			var:kehillah_var_prosperity >= kehillah_band_legendary_threshold
		}
		localization_key = kehillah_bd_aid_duty_legendary
	}
	text = {
		trigger = {
			has_variable = kehillah_var_prosperity
			var:kehillah_var_prosperity >= kehillah_band_flourishing_threshold
		}
		localization_key = kehillah_bd_aid_duty_flourishing
	}
	text = {
		trigger = {
			has_variable = kehillah_var_prosperity
			var:kehillah_var_prosperity >= kehillah_band_healthy_threshold
		}
		localization_key = kehillah_bd_aid_duty_healthy
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_bd_aid_duty_poor
		fallback = yes
	}
}
