# =====================================================================
# Hover cards for the Community Crises and Jewish Libraries map modes
# (2026-11-08, docs/implementation/2026-11-08-jewish-map-category.md).
# The cards themselves are KEHILLAH_CRISES_MAP_TOOLTIP_COMMUNITY and
# KEHILLAH_LIBRARIES_MAP_TOOLTIP_LIBRARY; these fill their variable lines.
# In each key, ROOT is the title Custom() was called on.
# =====================================================================

# --- Community Crises: scope is a d_kehillah_* community title. ---
# Same order as kehillah_pick_community_crises_map_color_effect, worst first.
KehillahCrisisMapStatus = {
	type = landed_title
	text = {
		trigger = { has_variable = kehillah_mob_threat }
		localization_key = kehillah_crisis_map_status_mob
	}
	text = {
		trigger = { kehillah_map_crisis_armed_trigger = yes }
		localization_key = kehillah_crisis_map_status_armed
	}
	text = {
		trigger = { kehillah_map_crisis_agitation_trigger = yes }
		localization_key = kehillah_crisis_map_status_agitation
	}
	text = {
		trigger = { has_variable = kehillah_expulsion_quarters }
		localization_key = kehillah_crisis_map_status_expulsion
	}
	text = {
		trigger = { exists = var:kehillah_decree_situation }
		localization_key = kehillah_crisis_map_status_decree
	}
	text = {
		trigger = { exists = var:kehillah_trial_situation }
		localization_key = kehillah_crisis_map_status_trial
	}
	text = {
		trigger = { kehillah_map_crisis_aftermath_trigger = yes }
		localization_key = kehillah_crisis_map_status_aftermath
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_crisis_map_status_none
		fallback = yes
	}
}

# The lines under the status: the situation and its phase, and for an
# attack its cause. Each branch checks the variable its key reads.
KehillahCrisisMapDetail = {
	type = landed_title
	text = {
		trigger = {
			exists = var:kehillah_crisis_situation
			NOT = { has_variable = kehillah_crisis_situation_done }
		}
		localization_key = kehillah_crisis_map_detail_attack
	}
	text = {
		trigger = {
			OR = {
				has_variable = kehillah_crisis_intro_pending
				AND = {
					has_variable = kehillah_crisis_stage
					NOT = { has_variable = kehillah_crisis_situation_done }
				}
			}
		}
		localization_key = kehillah_crisis_map_detail_cause
	}
	text = {
		trigger = { exists = var:kehillah_decree_situation }
		localization_key = kehillah_crisis_map_detail_decree
	}
	text = {
		trigger = { exists = var:kehillah_trial_situation }
		localization_key = kehillah_crisis_map_detail_trial
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_crisis_map_detail_none
		fallback = yes
	}
}

# --- Jewish Libraries: scope is a library title (kehillah_libraries). ---
# Same order as kehillah_paint_libraries_map_effect.
KehillahLibraryMapKind = {
	type = landed_title
	text = {
		trigger = { has_variable = kehillah_library_sealed }
		localization_key = kehillah_library_map_kind_sealed
	}
	text = {
		trigger = {
			OR = {
				has_variable = kehillah_landed_yeshiva
				holder ?= { kehillah_has_yeshiva_trigger = yes }
			}
		}
		localization_key = kehillah_library_map_kind_yeshiva
	}
	text = {
		trigger = { kehillah_lib_is_haymanot_library_trigger = yes }
		localization_key = kehillah_library_map_kind_haymanot
	}
	text = {
		trigger = { kehillah_lib_is_karaite_library_trigger = yes }
		localization_key = kehillah_library_map_kind_karaite
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_library_map_kind_rabbanite
		fallback = yes
	}
}
