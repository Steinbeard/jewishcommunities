# V29 section 9.2 (2026-10-11): the yeshiva's fame tier, by
# kehillah_yeshiva_fame_tier_value. Scope: a Kehillah leader.
KehillahYeshivaFameTier = {
	type = character
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 3 }
		localization_key = kehillah_yeshiva_tier_illustrious
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 2 }
		localization_key = kehillah_yeshiva_tier_renowned
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 1 }
		localization_key = kehillah_yeshiva_tier_respected
	}
	text = {
		fallback = yes
		localization_key = kehillah_yeshiva_tier_modest
	}
}
