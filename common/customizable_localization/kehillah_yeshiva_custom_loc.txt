# V29 section 9.2 (2026-10-11): the yeshiva's fame tier, by
# kehillah_yeshiva_fame_tier_value. Scope: a Kehillah leader.
KehillahYeshivaFameTier = {
	type = character
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 3 }
		localization_key = kehillah_yeshiva_tier_illustrious
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 2 }
		localization_key = kehillah_yeshiva_tier_renowned
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 1 }
		localization_key = kehillah_yeshiva_tier_respected
	}
	text = {
		fallback = yes
		localization_key = kehillah_yeshiva_tier_modest
	}
}

# Progress toward the next tier, in words -- "growing toward Renowned",
# "nearly Renowned" (within 10 of it), or the top. The number stays
# behind the scenes (Daniel, 2026-10-11). Scope: a Kehillah leader.
KehillahYeshivaNextTier = {
	type = character
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value >= 3 }
		localization_key = kehillah_yeshiva_next_tier_top
	}
	text = {
		trigger = {
			kehillah_yeshiva_fame_tier_value = 2
			kehillah_yeshiva_fame_to_next_value <= 10
		}
		localization_key = kehillah_yeshiva_next_nearly_illustrious
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value = 2 }
		localization_key = kehillah_yeshiva_next_toward_illustrious
	}
	text = {
		trigger = {
			kehillah_yeshiva_fame_tier_value = 1
			kehillah_yeshiva_fame_to_next_value <= 10
		}
		localization_key = kehillah_yeshiva_next_nearly_renowned
	}
	text = {
		trigger = { kehillah_yeshiva_fame_tier_value = 1 }
		localization_key = kehillah_yeshiva_next_toward_renowned
	}
	text = {
		trigger = { kehillah_yeshiva_fame_to_next_value <= 10 }
		localization_key = kehillah_yeshiva_next_nearly_respected
	}
	text = {
		fallback = yes
		localization_key = kehillah_yeshiva_next_toward_respected
	}
}
