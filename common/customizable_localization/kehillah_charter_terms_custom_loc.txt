# Host Charter terms for the realm window's Laws tab and the community map
# view (2026-11-08). Scope: a Kehillah leader. Like the notice rows in
# kehillah_charter_notification_custom_loc.txt, these read only the title
# cache that kehillah_refresh_charter_state_cache_effect writes, never live
# contract flags.
#
# The *Desc entries return the same _desc key the contract window shows
# for the level, so each term's effects are written once.

# The whole terms block, by charter package. The V15 fallback charter
# (hosts neither Christian nor Muslim) holds only lending, the walled
# quarter and tribute; its cache_lending marks it.
KehillahCharterTermsBlock = {
	type = character
	text = { trigger = { kehillah_cached_christian_charter_trigger = yes } localization_key = kehillah_charter_terms_block_christian }
	text = { trigger = { kehillah_cached_muslim_charter_trigger = yes } localization_key = kehillah_charter_terms_block_muslim }
	text = {
		trigger = { primary_title = { has_variable = kehillah_charter_cache_lending } }
		localization_key = kehillah_charter_terms_block_fallback
	}
	text = { trigger = { always = yes } localization_key = kehillah_charter_notice_unresolved fallback = yes }
}
