# V36 (2026-10-13): the host's protection in violence, shown when a war,
# crusade or rising term is active and the multiplier is not 1.
KehillahBdHostProtection = {
	type = character
	text = {
		trigger = {
			OR = {
				kehillah_bd_war_stability_value < 0
				kehillah_bd_holy_war_value < 0
			}
			kehillah_host_protection_mult_value < 1
		}
		localization_key = kehillah_bd_host_protection_good
	}
	text = {
		trigger = {
			OR = {
				kehillah_bd_war_stability_value < 0
				kehillah_bd_holy_war_value < 0
			}
			kehillah_host_protection_mult_value > 1
		}
		localization_key = kehillah_bd_host_protection_bad
	}
	text = {
		trigger = { always = yes }
		localization_key = kehillah_outlook_bd_none
	}
}
