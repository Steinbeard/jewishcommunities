# =====================================================================
# OVERRIDE of ImprisonCharacterDeclineSummary (CK3 1.20, common/
# customizable_localization/00_interactions_custom_loc.txt), the warning
# the imprison interaction shows for a refusal. Vanilla's entries are
# copied verbatim below the two Kehillah entries; refresh the copy if
# vanilla changes this block.
#
# Without the Kehillah entries a community leader would match vanilla's
# imprison_decline_summary_war ("will rise up in rebellion"), but a
# refusing leader goes into exile instead (kehillah_arrest_refused_
# effect, common/scripted_effects/kehillah_host_arrest_effects.txt), or,
# with no heir to leave the community to, is imprisoned regardless.
# Courtiers keep vanilla's text: vanilla already sends a refusing
# courtier to the wandering pool. Design: docs/spec/v32-host-arrest-and-
# exile.md.
# =====================================================================
ImprisonCharacterDeclineSummary = {
	type = character

	text = {
		localization_key = kehillah_imprison_decline_summary_exile
		trigger = {
			government_has_flag = government_is_kehillah
			exists = primary_title.current_heir
		}
	}

	text = {
		localization_key = kehillah_imprison_decline_summary_no_heir
		trigger = {
			government_has_flag = government_is_kehillah
		}
	}

	text = {
		localization_key = imprison_decline_summary_war_bloc
		trigger = {
			is_playable_character = yes
			primary_title.tier > tier_barony
			exists = house.house_confederation
			trigger_if = {
				limit = { exists = root.liege.house.house_confederation }
				NOT = { root.liege.house.house_confederation = house.house_confederation }
			}
		}
	}

	text = {
		localization_key = imprison_decline_summary_war
		trigger = {
			is_playable_character = yes
			primary_title.tier > tier_barony
		}
	}

	text = {
		localization_key = imprison_decline_summary_no_court
		trigger = {
			is_playable_character = yes
			primary_title.tier = tier_barony
		}
	}

	text = {
		localization_key = imprison_decline_summary_cannot_fire_court_chaplain
		trigger = {
			has_council_position = councillor_court_chaplain
			NOT = { can_be_fired_from_council_trigger = { COURT_OWNER = root.liege } }
		}
	}

	text = {
		localization_key = imprison_decline_summary_cannot_fire_HoF
		trigger = {
			exists = root.liege
			root.liege = { exists = faith.religious_head }
			root.liege.faith.religious_head = { this = root }
		}
	}

	text = {
		localization_key = imprison_decline_summary_no_war_your_courtier
		trigger = { host = scope:second }
	}

	text = {
		localization_key = imprison_decline_summary_no_war
		fallback = yes
	}
}
