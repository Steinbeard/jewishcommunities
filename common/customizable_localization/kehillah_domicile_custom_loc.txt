# =====================================================================
# Kehillah domicile naming override.
#
# THE PROBLEM. Vanilla composes every domicile's default display name
# from a single SHARED loc key, `domicile_name` (defined in vanilla's
# own localization/english/dlc/ep3/ep3_laamps_l_english.yml):
#
#   domicile_name: "[CHARACTER.GetDynastyHouseNameNoTooltip] [DOMICILE_TYPE.GetNameNoTooltip]"
#
# -- used for every domicile type in the game, camps and yurts and
# estates and ours alike. There is no per-type override key (confirmed:
# no "estate_domicile_name", "camp_domicile_name" etc. exist anywhere in
# vanilla) and no `name = ` field on the domicile_type definition itself
# (see common/domiciles/types/_domicile_types.info). The shared key is
# the only lever that exists, which is why overriding it naively would
# rename every domicile in the game, not just ours.
#
# THE FIX. Vanilla itself solves "different text depending on a real
# trigger check inside a loc string" with the customizable_localization
# system -- see GetRealmOrDomicile in
# common/customizable_localization/00_government_custom_loc.txt, the
# direct model for the structure below. localization/english/
# kehillah_domicile_name_l_english.yml redefines domicile_name to
# delegate here via CHARACTER.Custom('KehillahDomicileName'); this
# function is the only place that actually branches.
#
# The fallback branch reproduces vanilla's own template exactly, via
# paths reachable from CHARACTER alone (CHARACTER.GetDomicile.GetType in
# place of the separately-bound DOMICILE_TYPE vanilla's version uses --
# confirmed as a valid chain by vanilla's own
# CHARACTER.GetDomicile.GetType.GetTitleText(CHARACTER.Self) usage in
# localization/*/messages_l_*.yml). So this is additive: every domicile
# in the game that isn't a Kehillah looks exactly as it did before.
# =====================================================================

KehillahDomicileName = {
	type = character

	# Any Kehillah: the domicile is named after the held Kehillah title
	# ("Kehillah of Worms", "Kehillah of York"). The institution outlives
	# its leader (spec section 5), so the name follows the office, not
	# vanilla's dynasty-house default; and since rename_window =
	# primary_title on the domicile type already routes the player's
	# rename to the title, title and domicile can never drift apart.
	#
	# HISTORY, 2026-09-15 (live findings, in order):
	#  1. CHARACTER.GetDomicile.GetLocation.* rendered empty. The chain
	#     itself is valid (vanilla uses GetDomicile.GetLocation.
	#     GetBaronyNameExplicitly in dlc_ep3_decisions) and script reads
	#     domicile.domicile_location fine elsewhere in this mod -- but at
	#     the moment the engine evaluates domicile_name the location is
	#     not available, so any location-derived name is empty here.
	#  2. <title>.GetCountyData.GetCapital.* rendered empty. c_kehillah_*
	#     titles are landless = yes with no baronies, so they have no
	#     county data at all; vanilla's county_1.GetCountyData precedent
	#     is a landed county. The title's own capital = c_x field is not
	#     reachable through GetCountyData.
	#  3. A primary_title saved in setup_scope DID resolve (its
	#     GetNameNoTooltip rendered "Kehillah of York"), which is what
	#     this branch now uses. Three per-community branches
	#     (Worms/Speyer/Mainz, hand-written "<Town> Jewish Community")
	#     were deleted at the same time: with the title supplying the
	#     name they only produced an inconsistent second format.
	#
	# setup_scope (not the trigger, which only gates whether this branch
	# is picked) is what makes the title available to the localization_key
	# text as a bare saved-scope name. Only plain scope-links + save_scope_as
	# are permitted there ("interface effects", per _custom_loc.info and
	# every vanilla example); an every_held_title loop silently never saved.
	# primary_title ?= follows vanilla's house ?= / location ?= form and is
	# safe to equate with the held Kehillah county: the only landless title
	# that could compete for "primary" is d_kehillah_shum, which no single
	# leader holds under normal circumstances.
	text = {
		setup_scope = {
			primary_title ?= { save_scope_as = kehillah_generic_title }
		}
		trigger = {
			exists = scope:kehillah_generic_title
			scope:kehillah_generic_title = { is_kehillah_title_trigger = yes }
		}
		localization_key = kehillah_generic_domicile_name
	}

	# Not a Kehillah at all -- every other domicile in the game, players
	# and AI alike. Reproduces vanilla's own domicile_name text exactly.
	#
	# trigger = { always = no } is copied deliberately from vanilla's own
	# GetRealmOrDomicile fallback entry, not a mistake: fallback = yes is
	# what tells the engine to use this text when nothing above matched,
	# and the trigger itself is never meant to be reached on its own.
	text = {
		trigger = { always = no }
		localization_key = kehillah_domicile_name_vanilla_fallback
		fallback = yes
	}
}
