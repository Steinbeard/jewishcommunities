# =====================================================================
# Jewish Settlement Policy as a realm law (2026-10-28, Daniel: "Let's
# convert it to a law!"). Replaces V16's title variable
# kehillah_jewish_settlement_policy. The laws are in common/laws/
# kehillah_jewish_settlement_laws.txt; the four band triggers in
# kehillah_settlement_policy_triggers.txt remain the one way script reads
# the policy. Design: docs/spec/2026-10-28-jewish-settlement-law.md.
#
# flag = realm_law puts the group in vanilla's laws tab (window_my_realm
# lists MyRealmWindow.GetRealmLaws filtered on it); realm_law_no_number
# shows the law's name rather than a level number (vanilla crowned_laws).
#
# Only an independent, non-Jewish, non-Kehillah ruler holds it -- the
# same ruler V15/V16 resolve as a community's host (domicile county ->
# holder -> top_liege). Anyone else has no law, which the triggers read
# as Allowed, the old "variable absent" default.
# =====================================================================

kehillah_jewish_settlement = {
	default = kehillah_jewish_settlement_allowed
	flag = realm_law
	flag = realm_law_no_number

	# A Jewish ruler sets no policy toward Jews (Daniel, 2026-10-19).
	can_have_group = {
		kehillah_can_hold_settlement_law_trigger = yes
	}
}
