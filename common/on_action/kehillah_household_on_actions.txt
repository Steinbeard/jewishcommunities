# =====================================================================
# Ba'alei batim (V26 part 2): household pulses. Own file so these merge
# into vanilla's pulses without touching kehillah_on_actions.txt.
# Effects: common/scripted_effects/kehillah_household_effects.txt.
# =====================================================================

# Yearly: arrivals, departures and the two-household floor.
yearly_playable_pulse = {
	on_actions = { kehillah_household_yearly_pulse }
}

kehillah_household_yearly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}
	effect = {
		kehillah_household_yearly_effect = yes
	}
}

# Quarterly: keep the register, standing and pillar contributions
# current between yearly passes.
quarterly_playable_pulse = {
	on_actions = { kehillah_household_quarterly_pulse }
}

kehillah_household_quarterly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}
	effect = {
		kehillah_household_bootstrap_effect = yes
		# AI leaders fill empty lay council seats from the householders
		# (v30), before the recount so the new seat counts this quarter.
		if = {
			limit = { is_ai = yes }
			kehillah_household_seat_householders_effect = yes
		}
		kehillah_household_refresh_effect = yes
	}
}
