# =====================================================================
# OUTLOOK ON THE JEWS -- V34 section 7c (docs/spec/v34-outlook-
# teachings-and-founding.md), 2026-10-13. Yearly, an independent ruler's
# outlook drifts with what his chaplain preaches
# (kehillah_chaplain_sermons_effect).
# =====================================================================

yearly_playable_pulse = {
	on_actions = { kehillah_outlook_yearly }
}

kehillah_outlook_yearly = {
	trigger = {
		is_landed = yes
		is_independent_ruler = yes
		NOT = { government_has_flag = government_is_kehillah }
		NOT = { is_jewish_character_trigger = yes }
	}
	effect = {
		kehillah_chaplain_sermons_effect = yes
	}
}

# V35 (2026-10-13): the "Outlook on the Jews" marker on the character
# window, refreshed yearly for every landed character, and set for every
# ruler (and the player's host strip) at game start.
yearly_playable_pulse = {
	on_actions = { kehillah_outlook_marker_yearly }
}

kehillah_outlook_marker_yearly = {
	trigger = { is_landed = yes }
	effect = {
		kehillah_outlook_marker_refresh_effect = yes
	}
}

on_game_start_after_lobby = {
	on_actions = { kehillah_outlook_marker_game_start }
}

kehillah_outlook_marker_game_start = {
	effect = {
		every_ruler = {
			kehillah_outlook_marker_refresh_effect = yes
		}
		every_player = {
			limit = { government_has_flag = government_is_kehillah }
			kehillah_host_outlook_watch_effect = yes
		}
	}
}
