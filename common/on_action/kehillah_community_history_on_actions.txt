# =====================================================================
# Community history: losses the dissolution path cannot see. 2026-11-15.
# A Kehillah leader who dies with no successor ends the title through the
# engine, not kehillah_dissolve_community_effect. on_death notes the place;
# the yearly pulse records the loss if the seat stayed empty
# (kehillah_community_history_effects.txt).
# =====================================================================

on_death = {
	on_actions = { kehillah_history_on_death }
}

kehillah_history_on_death = {
	trigger = {
		exists = primary_title
		primary_title = { is_kehillah_title_trigger = yes }
	}
	effect = {
		kehillah_history_note_leader_death_effect = yes
	}
}

yearly_global_pulse = {
	on_actions = { kehillah_history_yearly }
}

kehillah_history_yearly = {
	effect = {
		kehillah_history_check_pending_effect = yes
	}
}
