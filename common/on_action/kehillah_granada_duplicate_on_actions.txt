# The 1066 Granada bookmark uses 9000216 for Yosef ibn Naghrilla and
# 9000233/9000234 for his wife and son. Vanilla separately seeds the same
# three people at Badis's court (3221, sephardi_0004, sephardi_0005).
# Retire the vanilla copies only in a start where our Yosef holds Granada.
# This avoids replacing vanilla's entire sephardi.txt character-history file.
on_game_start_after_lobby = {
	on_actions = { kehillah_granada_retire_vanilla_duplicates }
}

kehillah_granada_retire_vanilla_duplicates = {
	effect = {
		if = {
			limit = { title:d_kehillah_granada ?= { holder ?= { this = character:9000216 } } }
			character:3221 ?= {
				if = { limit = { is_alive = yes } death = { death_reason = death_vanished } }
			}
			character:sephardi_0004 ?= {
				if = { limit = { is_alive = yes } death = { death_reason = death_vanished } }
			}
			character:sephardi_0005 ?= {
				if = { limit = { is_alive = yes } death = { death_reason = death_vanished } }
			}
		}
	}
}
