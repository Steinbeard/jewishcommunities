# Appends a Jewish household beat to an ordinary Feast.
# feast_start_events fires for every attendee; the child on_action admits
# only a Jewish host, so the overlay appears once and leaves vanilla intents intact.
feast_start_events = {
    on_actions = { jewish_holiday_feast_start }
}

jewish_holiday_feast_start = {
    trigger = {
        this = scope:activity.activity_host
        is_jewish_character_trigger = yes
        scope:activity = {
            has_activity_option = {
                category = special_type
                option = feast_type_generic
            }
        }
    }
    events = { jewish_holiday_feast.0001 }
}
