# Ordination on_actions -- V21 slice 4, added 2026-10-10.
#
# Its own file rather than an addition to kehillah_on_actions.txt, the
# same reasoning kehillah_library_on_actions.txt records: on_actions with
# the same name in different files merge, so this work does not have to
# touch a file another line of work is editing.
#
# IMPORTANT: every guard this needs is restated in the mod's OWN
# sub-on_action trigger below, rather than relying on vanilla's trigger
# on on_birthday_childhood / on_16th_birthday. Merged on_action
# definitions share a name, not a trigger, and this code must not depend
# on which file's trigger the engine happens to apply to whose entries.
# =====================================================================

on_birthday_childhood = {
	on_actions = { kehillah_ordination_childhood_birthday }
}

# The cheap faith check first: this fires for every child in the world
# on every childhood birthday, and the overwhelming majority are not
# Jewish at all. kehillah_has_rabbinic_guardian_trigger (inside the
# effect) is the expensive half -- it walks the pupil's relations.
kehillah_ordination_childhood_birthday = {
	trigger = {
		is_adult = no
		age >= childhood_education_start_age
		is_rabbinic_authority_jewish_trigger = yes
	}
	effect = {
		kehillah_observe_rabbinic_education_effect = yes
	}
}

on_16th_birthday = {
	on_actions = { kehillah_ordination_on_adulthood }
}

# Delayed five days on purpose. Vanilla's own on_16th_birthday list runs
# childhood_education.0007 -- "make sure any education trait is applied"
# -- behind a delay = { days = 2 }, and the education trait is often what
# carries a student across Learning 8. Checking Learning on the birthday
# itself would read it before the trait lands and systematically
# under-credit exactly the students section 3.3 is about.
kehillah_ordination_on_adulthood = {
	trigger = {
		age = 16
		is_rabbinic_authority_jewish_trigger = yes
		kehillah_rabbinic_apprenticeship_complete_trigger = yes
	}
	effect = {
		trigger_event = {
			id = kehillah_semicha.0002
			days = 5
		}
	}
}
