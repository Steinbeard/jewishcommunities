# =====================================================================
# AI Bet Din hosting nudge -- 2026-10-16 soak (docs/testing/
# 2026-10-16-20-year-soak-report.md, "Probe: why no AI hosts a Bet Din").
#
# In 20 simulated years no AI community leader created a Bet Din. Every
# gate the mod controls passes and AI Watch lists the activity as valid at
# priority 100, yet the engine's planner never selects it. Vanilla forces
# an AI host for its own activities with a character flag plus
# ai_attempt_to_host_activity (yearly_on_actions.txt: feast_ai_override,
# tournament_ai_override); the activity's ai_will_do adds 10000 while the
# flag is set. This does the same, yearly, with a chance per eligible AI
# leader. The flag lasts 60 days, so it cannot keep stealing the AI's
# other activity choices.
# =====================================================================

yearly_playable_pulse = {
	on_actions = { kehillah_bet_din_ai_yearly }
}

kehillah_bet_din_ai_yearly = {
	trigger = {
		is_ai = yes
		kehillah_can_convene_bet_din_trigger = yes
	}
	effect = {
		random = {
			chance = 15
			add_character_flag = {
				flag = kehillah_bet_din_ai_override
				days = 60
			}
			ai_attempt_to_host_activity = activity_kehillah_bet_din_conference
		}
	}
}
