# =====================================================================
# THE NAGID OFFICE -- the Nagid spec, sections 8 and 13, 2026-10-30,
# slice 4. The ruler-side yearly check is in kehillah_status_on_actions.
# =====================================================================

# Each quarter, a Kehillah leader who holds the office or leads a seat
# keeps the office in step (kehillah_nagid_quarterly_effect).
quarterly_playable_pulse = {
	on_actions = { kehillah_nagid_quarterly }
}

kehillah_nagid_quarterly = {
	trigger = {
		government_has_flag = government_is_kehillah
		OR = {
			kehillah_is_nagid_trigger = yes
			kehillah_is_realm_seat_trigger = yes
		}
	}
	effect = { kehillah_nagid_quarterly_effect = yes }
}

# Starts (section 13). Five days in, after the settlement laws and the
# day-three pillar snapshot. 1066: Joseph ibn Naghrilla is Nagid of
# Badis's realm, and the Fatimid office is founded, vacant. 1178: the
# Ayyubid office is founded, vacant. Baghdad's Exilarchate is a later
# slice (section 12).
on_game_start_after_lobby = {
	on_actions = { kehillah_nagid_game_start }
}

kehillah_nagid_game_start = {
	effect = {
		if = {
			limit = {
				current_date >= 1066.1.1
				current_date < 1067.1.1
			}
			title:d_kehillah_granada.holder ?= { trigger_event = { id = kehillah_nagid.0090 days = 5 } }
			title:d_kehillah_fustat.holder ?= { trigger_event = { id = kehillah_nagid.0091 days = 5 } }
		}
		else_if = {
			limit = {
				current_date >= 1178.1.1
				current_date < 1179.1.1
			}
			title:d_kehillah_fustat.holder ?= { trigger_event = { id = kehillah_nagid.0091 days = 5 } }
		}
	}
}
