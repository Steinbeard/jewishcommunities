# =====================================================================
# The AI convenes its Bet Din (2026-10-15, 20-year soak: no AI community
# held a conference in 20 years, although a live probe found all nineteen
# eligible -- region, panel, rabbi, no cooldown). Once a year an eligible
# AI leader may be asked to plan one now; ai_attempt_to_host_activity runs
# the activity's own AI evaluation (ai_will_do, intents, location) at once
# instead of waiting on ai_check_interval.
# =====================================================================

yearly_playable_pulse = {
	on_actions = { kehillah_bet_din_ai_convene_yearly }
}

kehillah_bet_din_ai_convene_yearly = {
	trigger = {
		is_ai = yes
		kehillah_can_convene_bet_din_trigger = yes
	}
	effect = {
		random = {
			chance = kehillah_bet_din_ai_convene_chance
			ai_attempt_to_host_activity = activity_type:activity_kehillah_bet_din_conference
			debug_log = "kehillah_bet_din_ai_convene_yearly: an AI leader is asked to convene the Bet Din"
		}
	}
}
