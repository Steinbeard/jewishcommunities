# Vanilla code on-action: root is the prisoner, scope:imprisoner the captor.

	on_actions = { kehillah_aid_on_imprison }
}
kehillah_aid_on_imprison = {
	trigger = { is_jewish_character_trigger = yes }
	effect = { kehillah_aid_on_imprison_effect = yes }
}

# The emissary from the Land of Israel (kehillah_mutual_aid.0001). The
# great communities see him twice as often. Chances are literals here
# (random_events cannot take a script value).
yearly_playable_pulse = {
	on_actions = { kehillah_aid_emissary_pulse kehillah_aid_emissary_pulse_great }
}
kehillah_aid_emissary_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		kehillah_aid_expected_giver_trigger = yes
		NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_flourishing_threshold } }
		NOT = { has_variable = kehillah_emissary_cooldown }
	}
	random_events = {
		chance_to_happen = 10
		100 = kehillah_mutual_aid.0001
	}
}
kehillah_aid_emissary_pulse_great = {
	trigger = {
		government_has_flag = government_is_kehillah
		kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_flourishing_threshold }
		NOT = { has_variable = kehillah_emissary_cooldown }
	}
	random_events = {
		chance_to_happen = 20
		100 = kehillah_mutual_aid.0001
	}
}
