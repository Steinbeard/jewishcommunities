# =====================================================================
# The tribute a Kehillah pays its host (v31 §8.2). ADDED 2026-10-30.
# quarterly_playable_pulse is the finest pulse there is, so the tribute
# is paid each quarter; the Domain tab shows it per month.
# =====================================================================

quarterly_playable_pulse = {
	on_actions = { kehillah_tribute_quarterly_pulse }
}

kehillah_tribute_quarterly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		is_tributary = yes
	}
	effect = {
		kehillah_pay_host_tribute_effect = yes
	}
}
