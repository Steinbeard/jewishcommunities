# Runs once at game start, after the Kehillah titles have their holders
# (see kehillah_granada_duplicate_on_actions.txt for the same hook).
on_game_start_after_lobby = {
	on_actions = { kehillah_vanilla_historical_jews kehillah_vanilla_spawn_guard }
}

kehillah_vanilla_historical_jews = {
	effect = {
		kehillah_seed_vanilla_historical_jews_effect = yes
	}
}

# Vanilla's yearly pulse (spawn_historical_characters_effect) creates a new
# twenty-year-old of each famous figure inside a six-year window unless the
# global variable hchar_<name> already exists. Where the mod or vanilla history
# already holds the same person, register them as the existing one, as
# kehillah_rashi_duplicate_on_actions.txt does for Rashi in a 1066 start.
kehillah_vanilla_spawn_guard = {
	effect = {
		kehillah_register_historical_character_effect = { VAR = hchar_rashi PERSON = character:9000201 }
		kehillah_register_historical_character_effect = { VAR = hchar_saadia PERSON = character:9000224 }
		kehillah_register_historical_character_effect = { VAR = hchar_shaprut PERSON = character:9002077 }
		kehillah_register_historical_character_effect = { VAR = hchar_gabriol PERSON = character:9002076 }
		kehillah_register_historical_character_effect = { VAR = hchar_yehuda_halevi PERSON = character:9000410 }
		kehillah_register_historical_character_effect = { VAR = hchar_maimonides PERSON = character:maimonides_1 }
	}
}
