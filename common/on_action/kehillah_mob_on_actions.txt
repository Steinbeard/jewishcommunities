# S3: does on_army_enter_province see the mob's army arrive?
# Root is the army's owner; scope:army the army.
on_army_enter_province = {
	on_actions = { kehillah_mob_army_enter }
}
kehillah_mob_army_enter = {
	trigger = { has_character_flag = kehillah_mob_leader }
	effect = {
		scope:army.location ?= { save_scope_as = kms_entered }
		debug_log = "kehillah_mob_army_enter: the mob's army entered a province"
		if = {
			limit = {
				exists = global_var:kehillah_spike_target_province
				scope:army.location ?= global_var:kehillah_spike_target_province
			}
			debug_log = "kehillah_mob_army_enter: ARRIVED at the target community's province"
		}
		debug_log_scopes = yes
	}
}

# S2: does the mob complete a siege in the target county?
# Root is the occupant; scope:barony, scope:county, scope:war.
on_siege_completion = {
	on_actions = { kehillah_mob_siege_done }
}
kehillah_mob_siege_done = {
	trigger = { has_character_flag = kehillah_mob_leader }
	effect = {
		debug_log = "kehillah_mob_siege_done: the mob COMPLETED A SIEGE"
		debug_log_scopes = yes
	}
}
