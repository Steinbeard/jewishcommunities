# =====================================================================
# A converted Kehillah leader leaves office. Design: docs/spec/
# v31-host-teachings-stances-and-factions.md, section 10.1.
#
# Two hooks, because no single one sees every conversion:
#   - kehillah_apostasy_on_actions.txt (on_faith_conversion) fires for
#     the convert window and the effect that shares its logic, and sends
#     kehillah_conversion.0001 a day later;
#   - vanilla's Demand Conversion converts with set_character_rite
#     (11_dlc_pam_scripted_effects.txt, convert_to_rite_with_consequences
#     _effect: "avoids cascade"), which fires no on_action. This quarterly
#     sweep catches that, and anything else that changes faith silently.
# The event re-checks, so a leader seen by both is handled once.
# =====================================================================

quarterly_playable_pulse = {
	on_actions = {
		kehillah_conversion_quarterly_pulse
		kehillah_host_conversion_demand_pulse
	}
}

# V31 section 10.3: under an expulsion edict, the AI host may demand the
# leader's conversion ("convert or leave"). See
# kehillah_host_conversion_ai_check_effect.
kehillah_host_conversion_demand_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		is_jewish_character_trigger = yes
		exists = primary_title
		exists = domicile
		kehillah_expulsion_pending_trigger = yes
	}
	effect = {
		kehillah_host_conversion_ai_check_effect = yes
	}
}

kehillah_conversion_quarterly_pulse = {
	trigger = {
		kehillah_leader_left_judaism_trigger = yes
	}
	effect = {
		# A day later, never inside this pulse. Live 2026-10-12: fired
		# immediately, the step-down ran mid-pulse and the same pulse's
		# kehillah_quarterly_pulse then ran charter maintenance on the new
		# wayfarer, giving his camp a Host Charter.
		trigger_event = { id = kehillah_conversion.0001 days = 1 }
	}
}
