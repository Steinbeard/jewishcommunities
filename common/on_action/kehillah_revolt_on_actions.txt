# FALLEN REALMS. Vanilla's on_war_won_attacker effect turns a ruler who
# lost the last of his land into an adventurer (ep3_laamps.0002) in the
# same tick. This appended on_action only records what was lost and asks
# the Jewish loser three days later (kehillah_fallen_realm.0001), after
# that handling has settled, so it never races vanilla's title moves.
on_war_won_attacker = {
	on_actions = { kehillah_fallen_realm_war_won }
}

kehillah_fallen_realm_war_won = {
	effect = {
		scope:defender ?= {
			if = {
				limit = { kehillah_fallen_ruler_trigger = yes }
				kehillah_fallen_realm_record_effect = yes
			}
		}
	}
}

# JEWISH MAJORITY. Yearly, a very, very large community may come to
# outnumber the rest of its county.
yearly_playable_pulse = {
	on_actions = { kehillah_majority_yearly }
}

kehillah_majority_yearly = {
	trigger = {
		kehillah_majority_community_trigger = yes
	}
	effect = {
		random = {
			chance = kehillah_majority_yearly_chance_value
			trigger_event = kehillah_majority.0001
		}
	}
}
