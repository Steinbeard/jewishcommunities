# =====================================================================
# Kehillah on_actions.
#
# Every vanilla on_action touched here is extended using the pattern
# Paradox documents in common/on_action/_on_actions.info:
#
#   "you cannot append an effect block directly to an on_action which
#    already has an effect block, as this creates a conflict. If a
#    modder wishes to append their own effects to an on_action without
#    overriding a file, they may do it as follows:
#      some_vanilla_on_action = { on_actions = { some_modded_on_action } }
#      some_modded_on_action = { effect = { ... } }"
#
# All three vanilla on_actions below already carry effect blocks, so the
# indirection is required, not stylistic. No vanilla file is overridden
# and no vanilla effect is displaced.
# =====================================================================

# ---------------------------------------------------------------------
# TITLE GAIN -- the estate continuity fix (spec section 5).
#
# root = the character who gained the title
# scope:title = the title gained
#
# When a new leader is appointed, rebuild the community's quarter on
# their domicile from the state mirrored on the title. Without this the
# successor inherits an empty lot and the community's entire building
# history is lost, because CK3 domiciles are owned per character rather
# than per title.
# ---------------------------------------------------------------------

on_title_gain = {
	on_actions = { kehillah_on_title_gain kehillah_host_charter_steward_on_title_gain kehillah_map_view_on_title_gain }
}

kehillah_on_title_gain = {
	trigger = {
		scope:title = {
			is_kehillah_title_trigger = yes
		}
	}
	effect = {
		scope:title = { save_scope_as = kq_title }

		# Make the government follow the office, explicitly.
		#
		# This is NOT redundant belt-and-braces. Government type in CK3
		# is a property of the character, not of the title, and there is
		# no government rule that reliably hands a custom government to
		# an appointed successor who was previously an ordinary courtier
		# with no government at all. If this line is missing and the
		# engine does not propagate it on its own, the new leader holds
		# the Kehillah title while being mechanically something else --
		# no quarter, no pillar standing, no officers, and no way back.
		#
		# Setting it here means the rule is simply: whoever holds this
		# title leads a Kehillah. Which is what the title means.
		if = {
			limit = {
				NOT = { government_has_flag = government_is_kehillah }
			}
			change_government = kehillah_government
		}

		# Rebuild the community's quarter. Deliberately ordered AFTER
		# change_government, because changing government is what creates
		# the new leader's domicile in the first place -- restoring into
		# a domicile that does not exist yet would silently do nothing.
		kehillah_restore_quarter_effect = yes

		# HOST CHARTER, ADDED 2026-09-23 (Phase 4, v15). Deliberately
		# ordered AFTER kehillah_restore_quarter_effect, same reason as
		# that effect's own ordering note above -- the host-resolution
		# trigger reads through domicile.domicile_location, which does not
		# exist until the domicile does. tributary_heir_succession = yes
		# (kehillah_host_charter_groups.txt) already carries an existing
		# charter across this succession on its own; this call is the
		# safety net for the appointed-successor case where no charter
		# existed yet (kehillah_maintain_host_charter_effect no-ops
		# instantly if one already does).
		kehillah_maintain_host_charter_effect = yes

		# A new leader takes office with the community's confidence.
		# Small and deliberate, and separate from the institution's own
		# accumulated Stability/Greatness -- those live on the title
		# itself (kehillah_quarterly_pillars_effect's "WHY THE TITLE,
		# NOT THE CHARACTER" note) and so are already exactly as they
		# were, needing no restoration; this is a personal vote of
		# confidence added on top, not a duplicate of anything.
		# kehillah_init_pillars_effect first -- change_variable is a
		# silent no-op on a variable that has never been set (see that
		# effect's header in kehillah_scripted_effects.txt for the full
		# live-test finding). Matters most here of anywhere: an appointed
		# successor's quarter may not have had a quarterly tick yet.
		scope:title = {
			kehillah_init_pillars_effect = yes
			change_variable = { name = kehillah_var_stability add = 50 }
		}

		# Re-evaluate Rav vs. Parnas for the incoming leader -- see
		# common/flavorization/kehillah_title_holders.txt.
		kehillah_update_leader_flavor_effect = yes

		# SH'UM BOND, ADDED 2026-09-10 (v6 spec section 1). Refreshed
		# whenever a NEW leader takes office in Worms, Speyer or Mainz
		# specifically -- not all sixteen Kehillah titles, hence the
		# explicit guard here rather than folding this into kehillah_
		# update_leader_flavor_effect above, which every Kehillah runs.
		# See kehillah_refresh_shum_bond_effect's own header (common/
		# scripted_effects/kehillah_scripted_effects.txt) for the full
		# account, including why root is deliberately not used inside it.
		if = {
			limit = {
				OR = {
					scope:title = title:d_kehillah_worms
					scope:title = title:d_kehillah_speyer
					scope:title = title:d_kehillah_mainz
				}
			}
			kehillah_refresh_shum_bond_effect = yes
		}
	}
}


# ---------------------------------------------------------------------
# HOST-CHARTER STEWARDSHIP -- a host ruler, rather than a community
# leader, has succeeded to one or more existing tributary agreements.
#
# CK3's tributary_heir_succession carries the agreement itself. There is
# no contract rewrite here: scan the deliberately small registered-title
# list only to tell the successor and each affected player-community what
# survived. The permanent ruler flag debounces inheritance of a bundle of
# titles on the same date and intentionally does not suppress the separate
# fresh-charter report used after conquest.
#
# root = a character who just gained any title. The scan becomes nonzero
# only if that character is already the suzerain of a V16 regional charter;
# it is therefore inert for ordinary title gains and for the pre-charter
# part of game startup.
# ---------------------------------------------------------------------

kehillah_host_charter_steward_on_title_gain = {
	trigger = {
		NOT = { has_character_flag = kehillah_charter_steward_introduced }
		# Do not mistake an established host gaining a minor extra title for
		# a change of charter steward. Vanilla scope comparison syntax is
		# used here (see game/common/scripted_effects/00_major_decisions_
		# scripted_effects.txt): only the incoming ruler's current primary
		# title can introduce this one-time succession report.
		scope:title = { this = root.primary_title }
	}

	effect = {
		set_variable = { name = kehillah_charter_steward_community_count value = 0 }
		save_scope_as = kehillah_new_charter_steward

		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = {
					is_tributary = yes
					suzerain = scope:kehillah_new_charter_steward
					OR = {
						has_subject_contract_group = kehillah_christian_host_charter_prototype
						has_subject_contract_group = kehillah_muslim_host_charter_prototype
					}
				}
			}
			scope:kehillah_new_charter_steward = {
				change_variable = { name = kehillah_charter_steward_community_count add = 1 }
			}
			# A player leading one of these communities needs the same
			# continuity report as the successor. AI leaders immediately
			# acknowledge the event, so this is not an AI-event flood.
			holder = {
				trigger_event = { id = kehillah_settlement_conditions.0007 days = 1 }
			}
		}

		if = {
			limit = { var:kehillah_charter_steward_community_count >= 1 }
			add_character_flag = kehillah_charter_steward_introduced
			trigger_event = { id = kehillah_settlement_conditions.0006 days = 1 }
		}
		else = {
			remove_variable = kehillah_charter_steward_community_count
		}
	}
}


# ---------------------------------------------------------------------
# DOMICILE BUILDING COMPLETED -- keep the title's mirror current.
#
# root = the domicile
# scope:owner = the domicile's owner
#
# This is the write half of the continuity system. Every time the
# community finishes a building, the title records it, so the record
# survives the leader's death.
# ---------------------------------------------------------------------

on_domicile_building_completed = {
	on_actions = { kehillah_on_domicile_building_completed }
}

kehillah_on_domicile_building_completed = {
	trigger = {
		exists = scope:owner
		scope:owner = {
			government_has_flag = government_is_kehillah
		}
	}
	effect = {
		scope:owner = {
			kehillah_record_quarter_effect = yes
		}
	}
}


# ---------------------------------------------------------------------
# YEARLY PULSE -- the light protection layer (spec section 3b).
#
# root = the character
#
# Hooked to random_yearly_playable_pulse rather than to on_yearly_events
# for two reasons. It is the point where vanilla fans out to all its
# themed yearly pools, so it is the natural place to add another one.
# And unlike on_yearly_events it carries no random_events block of its
# own that ours would sit alongside -- our events get their own roll
# rather than competing for a slot in vanilla's pool, which matters
# because a Kehillah leader is excluded from most vanilla yearly
# content anyway and would otherwise have very quiet decades.
#
# chance_to_happen is deliberately modest. These are occasional
# incidents. A community harassed every single year is a different,
# much darker game than v1 is trying to be.
# ---------------------------------------------------------------------

random_yearly_playable_pulse = {
	on_actions = { kehillah_yearly_pulse }
}


# ---------------------------------------------------------------------
# QUARTERLY PILLAR TICK -- Stability and Greatness accrual.
#
# root = the character
#
# quarterly_playable_pulse is the finest-grained on_action that exists
# for playable characters -- there is no monthly equivalent to hook a
# custom effect from, which is why this runs quarterly rather than
# monthly. See kehillah_quarterly_pillars_effect in
# common/scripted_effects/kehillah_scripted_effects.txt for the full
# reasoning and every rate.
# ---------------------------------------------------------------------

quarterly_playable_pulse = {
	on_actions = { kehillah_quarterly_pulse }
}

kehillah_quarterly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
	}
	effect = {
		# V16 reads live tributary flags only in this deliberately scheduled
		# character effect. All pillars and UI consume its title cache, so a
		# manual charter edit is reflected before this quarter's convergence
		# without evaluating subject contracts during bootstrap/rendering.
		kehillah_refresh_charter_state_cache_effect = yes
		kehillah_quarterly_pillars_effect = yes

		# HOST CHARTER, ADDED 2026-09-23 (Phase 4, v15). This is the
		# backstop for a host changing hands by conquest (spike section 5
		# point 5, docs/spec/spike-host-charter-interaction.md) -- on_title_
		# gain and the founding effect both establish a charter, but
		# neither fires when the COUNTY the community sits in changes
		# holder. This quarterly re-check is what actually re-points it,
		# at most one quarter late; see kehillah_maintain_host_charter_
		# effect's own header for why that lag is a deliberate trade
		# against the tick-deferral constraint, not an oversight.
		kehillah_maintain_host_charter_effect = yes
	}
}


# ---------------------------------------------------------------------
# COURT ARRIVALS -- keep the court from filling up with strangers.
#
# root = the character joining
# scope:new_employer = who they are joining
# scope:old_employer = their previous employer, if any (unset for a
# character's first-ever employment)
#
# PARTIAL MITIGATION, NOT A FIX. Playtesting turned up a court full of
# courtiers who have nothing to do with the community -- wrong faith,
# wrong culture, apparently vanilla's generic "unemployed character
# seeks a court" behaviour pulling from the surrounding (Christian,
# German) population and the wider world regardless of what this
# government actually is. The exact vanilla mechanism responsible
# was not pinned down (see docs/implementation/v1-kehillah-implementation.md
# for what was and wasn't chased down). This on_action only catches the
# case it can catch safely and verifiably: a character who already has
# an employer and is about to switch to this one is sent back where
# they came from if they do not share the community's faith. It does
# nothing for a character seeking their very first employment (no
# old_employer to return them to) -- that case needs whatever the root
# vanilla mechanism turns out to be, once found.
# ---------------------------------------------------------------------

on_join_court = {
	on_actions = { kehillah_on_join_court }
}

kehillah_on_join_court = {
	trigger = {
		scope:new_employer = {
			government_has_flag = government_is_kehillah
		}
	}
	effect = {
		if = {
			limit = {
				exists = scope:old_employer
				NOT = { faith = scope:new_employer.faith }
			}
			set_employer = scope:old_employer
		}
		else = {
			# QUALITY (Wave 3 item 3, v2 spec's "courtier quality... skew
			# toward better skills at higher Greatness bands"). See
			# kehillah_apply_courtier_quality_effect's own header
			# (common/scripted_effects/kehillah_scripted_effects.txt) for
			# why this hooks HERE rather than editing
			# common/scripted_character_templates/
			# kehillah_pool_repopulation_patch.txt as originally suggested.
			kehillah_apply_courtier_quality_effect = yes
		}
	}
}

# though both hang off the same vanilla anchor. This is mundane
# internal governance, not the external-threat "protection" layer, and
# giving it its own chance_to_happen means the two rates can be tuned
# independently -- a community could plausibly face threats often and
# disputes rarely, or the reverse, and folding both into one roll would
# make that impossible to express. Two same-key
# random_yearly_playable_pulse blocks in one file merge exactly the way
# this file's own header describes for the three vanilla on_actions it
# already extends (on_actions = { ... } lists concatenate by key,
# regardless of which file or how many times the key appears).
#
# chance_to_happen is set looser than protection's 30: family disputes
# are ordinary communal friction, not a threat from outside, so they
# should come up somewhat more often. 65 is a first guess, per this
# file's and kehillah_script_values.txt's own "prototype loosely, tune
# by playtest" norm -- not a claim about the historically correct rate
# of Jewish communal litigation.
# ---------------------------------------------------------------------

random_yearly_playable_pulse = {
	on_actions = { kehillah_dispute_pulse }
}

kehillah_dispute_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		is_adult = yes
		is_imprisoned = no
		kehillah_has_dispute_candidates_trigger = yes
	}
	random_events = {
		chance_to_happen = 65

		# A meaningful chance nothing comes to you even when the roll
		# happens, same reasoning as kehillah_yearly_pulse above.
		100 = 0

		50 = kehillah_dispute.0001	# A Broken Partnership
		50 = kehillah_dispute.0002	# A Question of Dowry
	}
}


# ---------------------------------------------------------------------
# EPIDEMIC PULSE -- hooks the existing vanilla epidemic/physician system
# into Prosperity and Stability (v2 spec's "epidemic pulse" input on
# both pillars; Wave 1 item 2).
#
# root = the character. epidemic_ongoing_events is a GLOBAL vanilla
# on_action (common/on_action/ce1_on_actions.txt), confirmed root-bound
# to the affected character (its own effect block reads
# capital_province, a character-scope link, directly at the top level)
# and confirmed to already fire for landless governments explicitly --
# its own effect block special-cases
# OR = { has_government = landless_adventurer_government
#        is_landless_administrative = yes } for the Black Death
# splash-screen check. It also hands the effect block scope:epidemic,
# the specific epidemic instance driving this firing, already bound --
# confirmed by that same vanilla effect block reading
# capital_province ?= { any_province_epidemic = { this = scope:epidemic } ... }
# with no save_scope_as of its own beforehand. No iteration or extra
# scope-saving was needed to reach it, unlike the doc note that
# prompted this task assumed might be necessary.
#
# Gated on has_ce1_dlc_trigger (common/scripted_triggers/
# 00_has_dlc_scripted_triggers.txt) since epidemics are Legends of the
# Dead content; without the DLC this on_action's own vanilla trigger
# conditions mean it will not fire in the first place, but the explicit
# gate here is kept for clarity and consistency with this file's own
# style of never assuming a DLC silently.
#
# location = { any_province_epidemic = { this = scope:epidemic } }
# confirms the SPECIFIC epidemic driving this pulse is actually present
# at the character's own current location -- not merely somewhere in
# the wider world -- mirroring the exact trigger shape used throughout
# common/scripted_triggers/06_ce1_epidemic_triggers.txt and
# common/epidemics/00_epidemics.txt.
#
# kehillah_var_prosperity is introduced HERE, as a title variable
# (primary_title, not the character -- see kehillah_quarterly_pillars_
# effect's "WHY THE TITLE, NOT THE CHARACTER" note for why every pillar
# variable in this mod follows that pattern). It did not exist before
# this task; v2 spec section 1.1 flags creating it as the one piece of
# new plumbing this whole Prosperity/Stability/Greatness design
# depends on, so it is added properly here rather than assumed.
#
# UPDATED 2026-09-07 (Wave 2): this drain is no longer the only writer.
# kehillah_quarterly_pillars_effect (common/scripted_effects/
# kehillah_scripted_effects.txt) now also applies baseline convergence
# to this variable every quarter, per v2 spec section 8's Wave 2 item 3.
# This on_action's drain is unchanged -- it remains the one
# INSTANTANEOUS negative input on Prosperity (v2 spec section 2's
# epidemic input row); convergence pulling the variable back toward its
# structural baseline afterward is the intended interaction between the
# two, not a conflict.
#
# Drain amounts and physician mitigation are computed as script values
# in CHARACTER scope (kehillah_epidemic_prosperity_drain_value /
# kehillah_epidemic_stability_drain_value, common/script_values/
# kehillah_script_values.txt) and saved via save_scope_value_as BEFORE
# switching into primary_title -- the same pattern
# events/kehillah_protection_events.txt already uses for
# kehillah_shakedown_gold_cost, and necessary here for the same reason:
# employs_court_position and scope:epidemic need to resolve against the
# character, not the title, and a title-scoped block cannot read a
# character trigger directly.
# ---------------------------------------------------------------------

epidemic_ongoing_events = {
	on_actions = { kehillah_epidemic_pulse }
}

kehillah_epidemic_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		has_ce1_dlc_trigger = yes
		exists = scope:epidemic
		location = {
			any_province_epidemic = { this = scope:epidemic }
		}
	}
	effect = {
		save_scope_value_as = {
			name = kehillah_epidemic_prosperity_drain
			value = kehillah_epidemic_prosperity_drain_value
		}
		save_scope_value_as = {
			name = kehillah_epidemic_stability_drain
			value = kehillah_epidemic_stability_drain_value
		}

		primary_title = {
			change_variable = {
				name = kehillah_var_prosperity
				subtract = scope:kehillah_epidemic_prosperity_drain
			}
			change_variable = {
				name = kehillah_var_stability
				subtract = scope:kehillah_epidemic_stability_drain
			}
		}

		# Recovery-beat bookkeeping (see kehillah_quarterly_pillars_effect
		# for the detection side). Flag stays set for the whole time the
		# epidemic is active at this location; the "no physician" flag,
		# once set, is never cleared mid-epidemic -- a physician who
		# leaves partway through should not retroactively count as
		# "throughout".
		add_character_flag = kehillah_epidemic_active_flag
		if = {
			limit = {
				NOT = { employs_court_position = court_physician_court_position }
			}
			add_character_flag = kehillah_epidemic_no_physician_flag
		}
	}
}

# MAP VIEW REFRESH (v12, 2026-09-14).
#
# Keeps each player's mirrored community roster (see common/
# scripted_effects/kehillah_map_view_effects.txt) in step with the
# registry and with the live pillar numbers. Two hooks, deliberately
# separate from the pillar-economy pulse above rather than bolted onto
# it -- this file's own established convention (see the "Deliberately a
# SEPARATE on_action block" note on kehillah_dispute_pulse) -- because
# the refresh is a UI cache with no gameplay effect and must not become
# entangled with the tuning of anything that does.
#
# CADENCE. Quarterly, matching kehillah_quarterly_pulse, because that is
# exactly when pillar values change (kehillah_quarterly_pillars_effect);
# refreshing more often would only re-copy identical numbers, and less
# often would show a roster stale by up to a year. Note the trigger does
# NOT require the player to lead a Kehillah: the roster is a view of the
# world, useful to any player in a game where these communities exist,
# and the widget gates its own visibility on the mirrored count instead.
# ---------------------------------------------------------------------

quarterly_playable_pulse = {
	on_actions = { kehillah_map_view_quarterly_pulse }
}

kehillah_map_view_quarterly_pulse = {
	trigger = {
		is_ai = no
	}
	effect = {
		kehillah_refresh_map_view_effect = yes
	}
}

# in the roster the moment it resolves instead of up to three months

# root = the character who gained the title
# scope:title = the title gained

kehillah_map_view_on_title_gain = {
	trigger = {
		scope:title = {
			is_kehillah_title_trigger = yes
		}
	}

	}
}
