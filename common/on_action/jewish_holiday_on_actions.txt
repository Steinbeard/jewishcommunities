# =====================================================================
# Jewish holiday on_actions.
#
# Same indirection pattern as kehillah_on_actions.txt (see that file's
# header for the quote from _on_actions.info): a modded on_action of our
# own, appended to the vanilla hook via on_actions = { }, so this file
# and kehillah_on_actions.txt can both extend
# random_yearly_playable_pulse without either overriding the other.
#
# Scope for is_jewish_character_trigger is any character of any faith in
# judaism_religion -- not just the Kehillah leader -- so this fires for
# ordinary Jewish courtiers, other rulers, and Track B landed characters
# alike, wherever they are.
#
# chance_to_happen and the quiet-year weight are tuned independently of
# kehillah_yearly_pulse: this pool is much larger (every Jewish
# character on the map, not one government seat), so a given character
# should see one of these events less often, not more, even though there
# are more entries to pick from.
#
# NOTE on faith-gated entries (Hanukkah, Purim): the weight below is
# unconditional. A character whose faith fails that event's own trigger
# (a Samaritan rolling Hanukkah, say) simply gets a quiet year out of
# that roll rather than a reroll -- the same way vanilla's own weighted
# event lists behave. This is deliberately not corrected for, to avoid
# building a second selection layer for eight events; it just means
# faiths outside is_rabbinic_authority_jewish_trigger /
# is_hebrew_canon_jewish_trigger see flavor slightly less often overall,
# which is a reasonable reflection of having fewer holidays in the pool
# to begin with.
# =====================================================================

random_yearly_playable_pulse = {
	on_actions = { jewish_holiday_yearly_pulse }
}

quarterly_playable_pulse = {
	on_actions = { jewish_holiday_calendar_pulse }
}

jewish_holiday_yearly_pulse = {
	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
	}
	random_events = {
		chance_to_happen = 25

		# A quiet year is still the most likely outcome.
		160 = 0

		20 = jewish_holidays.0101	# Queue An Omen in the Honey (Rosh Hashanah)
		20 = jewish_holidays.0102	# Queue The New Year's Reckoning (Rosh Hashanah)
		20 = jewish_holidays.0103	# Queue The Four Questions (Passover)
		20 = jewish_holidays.0104	# Queue Bedikat Chametz (Passover)
		20 = jewish_holidays.0105	# Queue The Dreidel Game (Hanukkah)
		20 = jewish_holidays.0106	# Queue Light in the Window (Hanukkah)
		20 = jewish_holidays.0107	# Queue Ad Lo Yada (Purim)
		20 = jewish_holidays.0108	# Queue Mishloach Manot (Purim)
	}
}

jewish_holiday_calendar_pulse = {
	trigger = {
		is_jewish_character_trigger = yes
		is_adult = yes
		is_imprisoned = no
	}
	effect = {
		# These broad windows account for the Hebrew calendar moving through
		# CK3's fixed Gregorian-style months and for quarterly pulse timing.
		if = {
			limit = {
				OR = {
					has_character_flag = jewish_holiday_0001_pending
					has_character_flag = jewish_holiday_0002_pending
				}
				OR = {
					current_month = 8
					current_month = 9
					current_month = 10
				}
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0001_pending }
				remove_character_flag = jewish_holiday_0001_pending
				trigger_event = jewish_holidays.0001
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0002_pending }
				remove_character_flag = jewish_holiday_0002_pending
				trigger_event = jewish_holidays.0002
			}
		}
		if = {
			limit = {
				OR = {
					has_character_flag = jewish_holiday_0003_pending
					has_character_flag = jewish_holiday_0004_pending
				}
				OR = {
					current_month = 3
					current_month = 4
					current_month = 5
				}
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0003_pending }
				remove_character_flag = jewish_holiday_0003_pending
				trigger_event = jewish_holidays.0003
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0004_pending }
				remove_character_flag = jewish_holiday_0004_pending
				trigger_event = jewish_holidays.0004
			}
		}
		if = {
			limit = {
				OR = {
					has_character_flag = jewish_holiday_0005_pending
					has_character_flag = jewish_holiday_0006_pending
				}
				OR = {
					current_month = 11
					current_month = 12
					current_month = 1
				}
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0005_pending }
				remove_character_flag = jewish_holiday_0005_pending
				trigger_event = jewish_holidays.0005
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0006_pending }
				remove_character_flag = jewish_holiday_0006_pending
				trigger_event = jewish_holidays.0006
			}
		}
		if = {
			limit = {
				OR = {
					has_character_flag = jewish_holiday_0007_pending
					has_character_flag = jewish_holiday_0008_pending
				}
				OR = {
					current_month = 2
					current_month = 3
					current_month = 4
				}
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0007_pending }
				remove_character_flag = jewish_holiday_0007_pending
				trigger_event = jewish_holidays.0007
			}
			if = {
				limit = { has_character_flag = jewish_holiday_0008_pending }
				remove_character_flag = jewish_holiday_0008_pending
				trigger_event = jewish_holidays.0008
			}
		}
	}
}
