# =====================================================================
# Expulsion (ROADMAP S9b) and the AI ruler's settlement-policy review.
# Own file so these merge into vanilla's pulses without touching
# kehillah_on_actions.txt. Effects: common/scripted_effects/
# kehillah_expulsion_effects.txt.
# =====================================================================

# Quarterly: start, advance, enforce or lift an edict of expulsion.
quarterly_playable_pulse = {
	on_actions = { kehillah_expulsion_quarterly_pulse }
}

kehillah_expulsion_quarterly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		exists = primary_title
		exists = domicile
	}
	effect = {
		kehillah_expulsion_quarterly_effect = yes
	}
}

# Yearly: give the community's host ruler its chance to review the
# realm's Jewish Settlement Policy (AI rulers only; once a year per ruler).
yearly_playable_pulse = {
	on_actions = { kehillah_policy_review_yearly_pulse }
}

kehillah_policy_review_yearly_pulse = {
	trigger = {
		government_has_flag = government_is_kehillah
		exists = primary_title
		exists = domicile
	}
	effect = {
		kehillah_policy_ai_review_try_effect = yes
	}
}
