# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
# A separate child on_action, not an edit to kehillah_on_game_start: it only
# needs to run after vanilla's on_game_start, which every after-lobby hook
# does.
on_game_start_after_lobby = {
	on_actions = { kehillah_synod_on_game_start }
}

kehillah_synod_on_game_start = {
	effect = {
		kehillah_restore_general_rite_polygamy_effect = yes

		# The 1066 Rhineland synod: Worms convenes, Mainz if Worms is gone.
		if = {
			limit = { game_start_date >= 1066.1.1 }
			if = {
				limit = { exists = title:d_kehillah_worms.holder }
				title:d_kehillah_worms.holder = { trigger_event = { id = kehillah_synod.0001 days = 30 } }
			}
			else_if = {
				limit = { exists = title:d_kehillah_mainz.holder }
				title:d_kehillah_mainz.holder = { trigger_event = { id = kehillah_synod.0001 days = 30 } }
			}
		}
	}
}

# ---------------------------------------------------------------------
# V46 the Sanhedrin (docs/spec/v46-the-sanhedrin.md).
# ---------------------------------------------------------------------
yearly_global_pulse = {
	on_actions = { kehillah_sanhedrin_yearly_pulse }
}

kehillah_sanhedrin_yearly_pulse = {
	effect = {
		kehillah_sanhedrin_lapse_check_effect = yes
		kehillah_sanhedrin_refresh_seats_effect = yes
		kehillah_gaonate_yearly_effect = yes # V47
		kehillah_head_of_rite_yearly_effect = yes # V47 Recognition of the Sages
		# Catch any government transition the title hooks missed (spec s.9).
		every_in_global_list = {
			variable = kehillah_registered_communities
			holder ?= { kehillah_reconcile_nasi_government_effect = yes }
		}
	}
}

on_title_gain = {
	on_actions = {
		kehillah_sanhedrin_on_title_gain
		kehillah_nasi_community_on_title_gain
		kehillah_gaonate_on_title_gain
	}
}

# V47: a new holder of the Gaonate; and the academy of an established
# Gaon's office passing to a new leader, who receives the office a day
# later (kehillah_synod.0004), once the previous Gaon's title has been
# destroyed with him.
kehillah_gaonate_on_title_gain = {
	effect = {
		if = {
			limit = { scope:title = title:d_kehillah_gaonate }
			kehillah_gaonate_on_gain_effect = yes
		}
		if = {
			limit = {
				exists = global_var:kehillah_gaonate_office_academy
				scope:title = global_var:kehillah_gaonate_office_academy
			}
			debug_log = "KEHILLAH_GAONATE: the academy changed hands, office check queued"
			debug_log_scopes = yes
			trigger_event = { id = kehillah_synod.0004 days = 1 }
		}
	}
}

# V47 phase B: a new Nasi, elected to the Sanhedrin's seat county by its
# members. He gives up his own seat and community, takes the seat
# community, and moves onto the Nasi government
# (kehillah_sanhedrin_install_nasi_effect). The founder's own grant does not
# pass here: the seat variable is set only after his grant resolves.
kehillah_sanhedrin_on_title_gain = {
	trigger = {
		scope:title = { has_variable = kehillah_sanhedrin_seat_county }
	}
	effect = {
		kehillah_sanhedrin_install_nasi_effect = yes
		trigger_event = { id = kehillah_synod.0003 days = 1 }
		scope:previous_holder ?= {
			if = {
				limit = { is_alive = yes }
				kehillah_reconcile_nasi_government_effect = yes
			}
		}
	}
}

# Whoever takes a community, above all the successor of a Nasi, who
# inherits the Nasi's government with the title. Runs alongside the
# community's own kehillah_on_title_gain, which only acts when the
# government is not a Kehillah one at all.
kehillah_nasi_community_on_title_gain = {
	trigger = {
		scope:title = { is_kehillah_title_trigger = yes }
		OR = {
			has_government = kehillah_nasi_government
			kehillah_is_sanhedrin_head_trigger = yes
		}
	}
	effect = {
		kehillah_reconcile_nasi_government_effect = yes
		trigger_event = { id = kehillah_synod.0003 days = 1 }
	}
}
