# Religious Scholar and the Rabbi trait: see kehillah_scholar_effects.txt.
#
# 2026-11-11: the yearly pulse that stripped Religious Scholar from rabbis
# is gone. The Rabbi trait's opposites block new grants, and every script
# grant of the Rabbi trait converts Scholar on the spot. What remains:

# A new game: anyone generated with both traits converts before the lobby,
# so no player ever sees a "Trait Lost" message for it.
on_game_start = {
	on_actions = { kehillah_rabbi_scholar_game_start }
}

kehillah_rabbi_scholar_game_start = {
	effect = {
		every_living_character = {
			limit = { kehillah_has_redundant_scholar_trait_trigger = yes }
			kehillah_absorb_religious_scholar_effect = yes
		}
	}
}

# Safety net, should a grant reach a rabbi despite the opposite: convert
# at once, beside the grant that caused it, not months later.
on_trait_gained = {
	on_actions = { kehillah_rabbi_scholar_trait_gained }
}

kehillah_rabbi_scholar_trait_gained = {
	trigger = {
		scope:trait = trait:lifestyle_scholar
		kehillah_has_redundant_scholar_trait_trigger = yes
	}
	effect = { kehillah_absorb_religious_scholar_effect = yes }
}
