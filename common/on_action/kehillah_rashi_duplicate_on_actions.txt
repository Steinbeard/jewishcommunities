# Vanilla's annual historical-character pulse creates a new Rashi in Troyes
# during 1066-1072 unless hchar_rashi already exists. Our historical Rashi
# (9000201) is present at the 1066 start, so register him as the existing one.
# The theology-focus event has a separate once-only historical-Rashi gate.
on_game_start_after_lobby = {
	on_actions = { kehillah_rashi_register_existing_start }
}

kehillah_rashi_register_existing_start = {
	effect = {
		if = {
			limit = { title:d_kehillah_troyes ?= { holder ?= { this = character:9000201 } } }
			set_global_variable = { name = hchar_rashi value = character:9000201 }
			add_to_global_variable_list = {
				name = special_character_list
				target = flag:flag_theology_1101_shlomo_yitzchaki
			}
		}
	}
}
