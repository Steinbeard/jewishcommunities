# =====================================================================
# STATUS OF THE JEWS, yearly -- the Nagid spec section 3, 2026-10-30.
# Keeps the "favoring the Jews beyond the teaching" modifier in step with
# a ruler's teaching and culture, and ends Favored under Exclusion or
# Abolition (kehillah_status_yearly_effect).
# =====================================================================

yearly_playable_pulse = {
	on_actions = { kehillah_status_yearly }
}

kehillah_status_yearly = {
	trigger = {
		OR = {
			kehillah_ruler_favors_jews_trigger = yes
			has_character_modifier = kehillah_favored_beyond_teaching_1_modifier
			has_character_modifier = kehillah_favored_beyond_teaching_2_modifier
			has_character_modifier = kehillah_favored_beyond_teaching_3_modifier
			has_character_modifier = kehillah_favored_beyond_teaching_4_modifier
			# Slice 4: a realm with the Nagid office (on any held title).
			any_held_title = { has_variable = kehillah_nagid_office }
		}
	}
	effect = { kehillah_status_yearly_effect = yes }
}
