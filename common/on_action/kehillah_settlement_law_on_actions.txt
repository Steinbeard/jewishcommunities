# =====================================================================
# Jewish Settlement Policy as a realm law: making the laws authoritative.
# docs/spec/2026-10-28-jewish-settlement-law.md, "Old saves".
#
# Until the global flag kehillah_settlement_laws_ready is set the band
# triggers ignore the law and read the old title variable (absent =
# Allowed): loading a pre-law save, the engine hands every existing ruler
# the group's FIRST law, Banned. kehillah_settlement_laws_normalize_effect
# sets each eligible ruler's law from their variable and sets the flag.
#
# It runs at game start, and on the first yearly pulse after a load --
# random_yearly_playable_pulse fires for some playable character within
# days of any date, so this is effectively at once. The realm view and
# the policy map also run it before reading anything. Once the flag is
# set every call is a no-op.
#
# Several files' blocks for the same vanilla on_action merge (see the
# note in kehillah_on_actions.txt).
# =====================================================================

on_game_start_after_lobby = {
	on_actions = { kehillah_settlement_laws_normalize_on_action }
}

random_yearly_playable_pulse = {
	on_actions = { kehillah_settlement_laws_normalize_on_action }
}

kehillah_settlement_laws_normalize_on_action = {
	trigger = {
		NOT = { has_global_variable = kehillah_settlement_laws_ready }
	}
	effect = {
		kehillah_settlement_laws_normalize_effect = yes
	}
}
