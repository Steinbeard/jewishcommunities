# =====================================================================
# Jewish Settlement Policy as a realm law (2026-10-28): converting saves
# from before the law. docs/spec/2026-10-28-jewish-settlement-law.md.
#
# A pre-law save keeps each realm's policy in the title variable
# kehillah_jewish_settlement_policy. The four band triggers read that
# variable first while it exists, so nothing reads wrongly in between;
# once a year each ruler's variable becomes the matching law and is
# removed. A ruler who passes a settlement law sooner converts at once
# (kehillah_settlement_policy_changed_effect removes the variable).
# New games never write the variable.
#
# random_yearly_playable_pulse blocks in several files merge (see the
# note in kehillah_on_actions.txt).
# =====================================================================

random_yearly_playable_pulse = {
	on_actions = { kehillah_settlement_law_migration_pulse }
}

kehillah_settlement_law_migration_pulse = {
	trigger = {
		exists = primary_title
		primary_title = { has_variable = kehillah_jewish_settlement_policy }
	}
	effect = {
		kehillah_migrate_settlement_policy_effect = yes
	}
}
