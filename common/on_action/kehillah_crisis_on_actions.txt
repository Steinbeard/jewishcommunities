# Retry an unanswered settlement if its original relay ruler died.
quarterly_playable_pulse = { on_actions = { kehillah_crisis_settlement_retry kehillah_crisis_situation_reconcile } }
kehillah_crisis_situation_reconcile = {
 trigger = {
  government_has_flag = government_is_kehillah
  primary_title ?= { has_variable = kehillah_crisis_stage OR = { var:kehillah_crisis_stage = 1 var:kehillah_crisis_stage = 2 var:kehillah_crisis_stage = 3 } }
 }
 effect = { primary_title = { kehillah_crisis_situation_sync_effect = yes } }
}
kehillah_crisis_settlement_retry = {
 trigger = {
  government_has_flag = government_is_kehillah
  primary_title ?= { has_variable = kehillah_crisis_stage var:kehillah_crisis_stage = 3 NOT = { has_variable = kehillah_crisis_judgement_sent } NOT = { has_variable = kehillah_crisis_judgement_grace } }
 }
 effect = { kehillah_crisis_request_judgement_effect = yes }
}
