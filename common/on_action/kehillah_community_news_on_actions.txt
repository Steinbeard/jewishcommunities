# =====================================================================
# COMMUNITY NEWS -- 2026-10-12 (docs/spec/v30-community-news.md).
# Your host lord dies: a player Kehillah leader hears of it when the dead
# man held his community's county or was that county's top liege. Added
# to vanilla's on_death alongside the mod's other on_death entries.
# =====================================================================

on_death = {
	on_actions = { kehillah_on_death_host_news }
}

kehillah_on_death_host_news = {
	trigger = {
		is_landed = yes
	}
	effect = {
		save_scope_as = kehillah_dead_lord
		primary_heir ?= { save_scope_as = kehillah_new_lord }
		every_player = {
			limit = {
				government_has_flag = government_is_kehillah
				exists = domicile.domicile_location.county.holder
				OR = {
					domicile.domicile_location.county.holder = scope:kehillah_dead_lord
					domicile.domicile_location.county.holder.top_liege = scope:kehillah_dead_lord
				}
			}
			if = {
				limit = { exists = scope:kehillah_new_lord }
				send_interface_message = {
					type = msg_kehillah_community_news_bad
					title = kehillah_news_host_death_t
					desc = kehillah_news_host_death_d
					left_icon = scope:kehillah_dead_lord
					right_icon = scope:kehillah_new_lord
				}
			}
			else = {
				send_interface_message = {
					type = msg_kehillah_community_news_bad
					title = kehillah_news_host_death_t
					desc = kehillah_news_host_death_noheir_d
					left_icon = scope:kehillah_dead_lord
				}
			}
		}
	}
}
