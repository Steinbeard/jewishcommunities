# =====================================================================
# Community library on_actions. ADDED 2026-09-17. Its own file, not an
# addition to kehillah_on_actions.txt, so that this work does not touch
# a file another line of work is editing; on_actions with the same
# name in different files merge (this is how every mod adds to vanilla's
# on_game_start_after_lobby, including kehillah_on_actions.txt itself),
# so both kehillah_on_game_start and kehillah_library_on_game_start
# fire, in no guaranteed order -- nothing here depends on the order.
# =====================================================================

on_game_start_after_lobby = {
	on_actions = { kehillah_library_on_game_start }
}

kehillah_library_on_game_start = {
	effect = {
		kehillah_seed_beit_midrash_effect = yes
		kehillah_seed_historical_sofer_workshops_effect = yes
	}
}

# ---------------------------------------------------------------------
# Sefer redesign (2026-11-08, docs/spec/2026-11-08-sefer-system-redesign.md).
# A new game sets tenet statuses, registers every library (Kehillot and
# landed temple libraries), marks the named works already written by the
# start date and seeds every Jewish rite's standings. An older save does
# the same the first year it runs (kehillah_sefer_yearly_effect calls the
# same once-only migration).
# ---------------------------------------------------------------------
on_game_start_after_lobby = {
	on_actions = { kehillah_sefer_on_game_start }
}

kehillah_sefer_on_game_start = {
	effect = {
		kehillah_sefer_migrate_save_effect = yes
	}
}

yearly_global_pulse = {
	on_actions = { kehillah_sefer_yearly_pulse }
}

kehillah_sefer_yearly_pulse = {
	effect = {
		kehillah_sefer_yearly_effect = yes
	}
}
