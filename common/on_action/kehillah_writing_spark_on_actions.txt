# Writing sparks (2026-11-23, docs/spec/2026-11-23-writing-sparks.md).
# Every writer starts with one spark so Write a Sefer is not empty on day
# one. An existing save seeds its writers on the first yearly pulse, and
# AI writers, who rarely study or preach, occasionally find a spark.

on_game_start_after_lobby = { on_actions = { kehillah_spark_new_game } }

kehillah_spark_new_game = {
	effect = {
		every_living_character = {
			limit = {
				is_jewish_character_trigger = yes
				kehillah_sefer_can_write_trigger = yes
			}
			kehillah_spark_seed_effect = yes
		}
	}
}

yearly_playable_pulse = { on_actions = { kehillah_spark_yearly } }

kehillah_spark_yearly = {
	trigger = {
		is_jewish_character_trigger = yes
		kehillah_sefer_can_write_trigger = yes
	}
	effect = {
		kehillah_spark_seed_effect = yes
		if = {
			limit = { is_ai = yes }
			random = { chance = 15 kehillah_spark_from_any_effect = yes }
		}
	}
}
