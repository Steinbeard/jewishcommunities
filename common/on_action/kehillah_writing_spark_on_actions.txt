# Writing sparks (2026-11-24, docs/spec/2026-11-24-writing-sparks-v2.md).
# A game starts with no sparks; mastering a sefer sparks the first. Once a
# year a writer's held sparks are checked for their next tier, since some
# routes (aptitude, mastered works) can change without any spark action. AI
# writers, who rarely study, occasionally find an idea for an empty slot.

yearly_playable_pulse = { on_actions = { kehillah_spark_yearly } }

kehillah_spark_yearly = {
	trigger = {
		is_jewish_character_trigger = yes
		kehillah_sefer_can_write_trigger = yes
	}
	effect = {
		kehillah_spark_ripen_all_effect = yes
		if = {
			limit = { is_ai = yes }
			# 2026-11-24: a historical author takes up his own work first.
			kehillah_spark_ai_author_effect = yes
			random = { chance = 15 kehillah_spark_ai_find_effect = yes }
		}
	}
}

# 2026-11-24: dated landmarks no longer appear on their catalog year; they
# exist once written. New games mark the ones dated before the start (the
# sefer migration at game start); an older save marks, once, every work whose
# year it has already reached, since those were circulating in it.
on_game_start_after_lobby = { on_actions = { kehillah_spark_dated_works_new_game } }
kehillah_spark_dated_works_new_game = {
	effect = { set_global_variable = { name = kehillah_sefer_dated_marked value = yes } }
}
yearly_global_pulse = { on_actions = { kehillah_spark_dated_works_yearly } }
kehillah_spark_dated_works_yearly = {
	trigger = { NOT = { has_global_variable = kehillah_sefer_dated_marked } }
	effect = {
		set_global_variable = { name = kehillah_sefer_dated_marked value = yes }
		kehillah_sefer_mark_historical_works_effect = yes
	}
}
