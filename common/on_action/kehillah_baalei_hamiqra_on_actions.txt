# Faith-level tenet statuses are initialized after the lobby, alongside the
# mod's other game-start state. Only the conflicting authority claims are
# prohibited: Talmud Torah and ordinary biblical interpretation remain open.
on_game_start_after_lobby = {
	on_actions = { kehillah_baalei_hamiqra_status_start }
}

kehillah_baalei_hamiqra_status_start = {
	effect = {
		faith:rabbinic_faith = {
			set_tenet_status = { tenet = tenet_baalei_hamiqra status = prohibited }
		}
		faith:karaism = {
			set_tenet_status = { tenet = tenet_religious_legal_pronouncements status = prohibited }
		}
	}
}
