# Landless community claims cannot use CK3's ordinary title-claim storage.
# Pass each recorded claim to the claimant's primary heir at death.
on_death = {
	on_actions = { kehillah_community_claim_inheritance }
}

kehillah_community_claim_inheritance = {
	trigger = {
		has_variable_list = kehillah_community_claims
		exists = primary_heir
	}
	effect = {
		primary_heir = { save_scope_as = kehillah_claim_heir }
		every_in_list = {
			variable = kehillah_community_claims
			save_temporary_scope_as = kehillah_inherited_seat
			scope:kehillah_claim_heir = {
				if = {
					limit = {
						NOT = {
							is_target_in_variable_list = {
								name = kehillah_community_claims
								target = scope:kehillah_inherited_seat
							}
						}
					}
					add_to_variable_list = {
						name = kehillah_community_claims
						target = scope:kehillah_inherited_seat
					}
				}
			}
		}
	}
}
