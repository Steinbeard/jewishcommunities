# Temple Service starts unknown (common/scripted_effects/
# kehillah_temple_service_tenet_effects.txt). Fresh games read history; only an
# existing save runs the one-time migration, on the next yearly pulse.
on_game_start_after_lobby = { on_actions = { kehillah_temple_service_tenet_new_game } }
kehillah_temple_service_tenet_new_game = {
	effect = { set_global_variable = { name = kehillah_temple_service_unknown_20261124 value = yes } }
}
yearly_global_pulse = { on_actions = { kehillah_temple_service_tenet_yearly } }
kehillah_temple_service_tenet_yearly = { effect = { kehillah_temple_service_unknown_migrate_effect = yes } }
