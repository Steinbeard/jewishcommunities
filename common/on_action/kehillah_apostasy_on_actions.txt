# Generated characters (1066 courtiers and guests) can be handed Kohen or
# Levite although they were never Jewish, because potential does not stop them.
# Sweep non-Jews once at game start, when nobody has had a chance to convert.
# Deliberately NOT repeated in play: priestly status passes by descent and
# survives conversion, forced or not, in Jewish law and in the converso
# families that kept the tradition (decision 2026-10-29).
on_game_start_after_lobby = {
	on_actions = { kehillah_priestly_traits_game_start_sweep }
}

kehillah_priestly_traits_game_start_sweep = {
	effect = {
		every_living_character = {
			limit = {
				faith.religion != religion:judaism_religion
				OR = {
					has_trait = kohen
					has_trait = levite
				}
			}
			kehillah_strip_priestly_traits_effect = yes
		}
	}
}

# Root is the converted character; scope:old_faith is the faith they left.
on_faith_conversion = {
	on_actions = { kehillah_apostasy_on_faith_conversion }
}

kehillah_apostasy_on_faith_conversion = {
	effect = {
		if = {
			limit = {
				scope:old_faith.religion = religion:judaism_religion
				faith.religion != religion:judaism_religion
			}
			add_trait = kehillah_apostate_trait
			# Priestly status belongs to Jews only (2026-10-29).
			kehillah_strip_priestly_traits_effect = yes
		}
		else_if = {
			limit = {
				faith.religion = religion:judaism_religion
				has_trait = kehillah_apostate_trait
			}
			remove_trait = kehillah_apostate_trait
		}

		# V32 HOST ARREST: someone from the host's own religion converting to
		# Judaism at a community's court is a crime the host holds against
		# the community's leader. age >= 1 keeps V27's at-birth descent
		# assignment of a mixed-parent child's faith out of it. A pool
		# character has no liege, so template generation never counts.
		if = {
			limit = {
				age >= 1
				faith.religion = religion:judaism_religion
				NOT = { scope:old_faith.religion = religion:judaism_religion }
				NOT = { government_has_flag = government_is_kehillah }
				liege ?= {
					government_has_flag = government_is_kehillah
					kehillah_host_charter_host_available_trigger = yes
					domicile.domicile_location.county.holder.top_liege.faith.religion = scope:old_faith.religion
				}
			}
			liege = {
				kehillah_host_crime_effect = { CRIME = kehillah_crime_proselytism_opinion }
			}
		}

		# V31 section 10.1: a Kehillah leader who leaves Judaism leaves
		# office. A day later, outside the conversion. Vanilla's forced
		# conversion uses set_character_rite and does NOT fire this
		# on_action, so kehillah_conversion_quarterly_pulse
		# (kehillah_conversion_on_actions.txt) catches that case.
		if = {
			limit = { kehillah_leader_left_judaism_trigger = yes }
			trigger_event = { id = kehillah_conversion.0001 days = 1 }
		}
	}
}
