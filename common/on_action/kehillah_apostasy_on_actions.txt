# Root is the converted character; scope:old_faith is the faith they left.
on_faith_conversion = {
	on_actions = { kehillah_apostasy_on_faith_conversion }
}

kehillah_apostasy_on_faith_conversion = {
	effect = {
		if = {
			limit = {
				scope:old_faith.religion = religion:judaism_religion
				faith.religion != religion:judaism_religion
			}
			add_trait = kehillah_apostate_trait
		}
		else_if = {
			limit = {
				faith.religion = religion:judaism_religion
				has_trait = kehillah_apostate_trait
			}
			remove_trait = kehillah_apostate_trait
		}
	}
}
