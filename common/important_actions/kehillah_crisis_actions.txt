# Persistent, clickable alert after acknowledgment. Pending introductions never create an alert.
action_kehillah_active_crisis = {
 type = alert
 icon = action_dangerous_faction_targeting_me
 is_dangerous = yes
 priority = 100
 check_create_action = {
  if = {
   limit = {
    government_has_flag = government_is_kehillah
    primary_title ?= { NOT = { has_variable = kehillah_crisis_intro_pending } }
    OR = {
     exists = primary_title.var:kehillah_trial_situation
     exists = primary_title.var:kehillah_decree_situation
     primary_title ?= {
      exists = var:kehillah_crisis_situation
      has_variable = kehillah_crisis_stage
      var:kehillah_crisis_stage < 4
     }
    }
   }
   try_create_important_action = { important_action_type = action_kehillah_active_crisis }
  }
 }
 effect = {
  # The alert check and the click have separate script contexts.
  if = {
   limit = { exists = primary_title.var:kehillah_trial_situation }
   primary_title.var:kehillah_trial_situation = { open_view_data = situation }
  }
  else_if = {
   limit = { exists = primary_title.var:kehillah_decree_situation }
   primary_title.var:kehillah_decree_situation = { open_view_data = situation }
  }
  else_if = {
   limit = { primary_title ?= { has_variable = kehillah_crisis_stage var:kehillah_crisis_stage < 4 } }
   primary_title.var:kehillah_crisis_situation ?= { open_view_data = situation }
  }
 }
}
# Expulsion still uses its original edict rather than a situation window.

    primary_title ?= { has_variable = kehillah_expulsion_quarters NOT = { has_variable = kehillah_crisis_intro_pending } }
   }
   try_create_important_action = { important_action_type = action_kehillah_active_expulsion }
  }
 }
 effect = { decision:kehillah_answer_edict_decision = { open_view_data = decision_detail } }
}
