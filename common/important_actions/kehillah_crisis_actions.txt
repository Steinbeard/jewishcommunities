# Persistent, clickable alert after acknowledgment. Pending introductions never create an alert.
action_kehillah_active_crisis = {
 type = alert
 icon = action_dangerous_faction_targeting_me
 is_dangerous = yes
 priority = 100
 check_create_action = {
  if = {
   limit = { government_has_flag = government_is_kehillah primary_title ?= { NOT = { has_variable = kehillah_crisis_intro_pending } } }
   if = {
    limit = { exists = primary_title.var:kehillah_trial_situation }
    primary_title.var:kehillah_trial_situation = { save_temporary_scope_as = kehillah_alert_situation }
    try_create_important_action = { important_action_type = action_kehillah_active_crisis }
   }
   else_if = {
    limit = { exists = primary_title.var:kehillah_decree_situation }
    primary_title.var:kehillah_decree_situation = { save_temporary_scope_as = kehillah_alert_situation }
    try_create_important_action = { important_action_type = action_kehillah_active_crisis }
   }
   else_if = {
    limit = { exists = primary_title.var:kehillah_crisis_situation primary_title ?= { has_variable = kehillah_crisis_stage var:kehillah_crisis_stage < 4 } }
    primary_title.var:kehillah_crisis_situation = { save_temporary_scope_as = kehillah_alert_situation }
    try_create_important_action = { important_action_type = action_kehillah_active_crisis }
   }
  }
 }
 effect = { scope:kehillah_alert_situation ?= { open_view_data = situation } }
}
# Expulsion still uses its original edict rather than a situation window.
action_kehillah_active_expulsion = {
 type = alert
 icon = action_dangerous_faction_targeting_me
 is_dangerous = yes
 priority = 100
 check_create_action = {
  if = {
   limit = {
    government_has_flag = government_is_kehillah
    primary_title ?= { has_variable = kehillah_expulsion_quarters NOT = { has_variable = kehillah_crisis_intro_pending } }
   }
   try_create_important_action = { important_action_type = action_kehillah_active_expulsion }
  }
 }
 effect = { decision:kehillah_answer_edict_decision = { open_view_data = decision_detail } }
}
