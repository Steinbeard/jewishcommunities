# One top-bar alert per crisis action that can be taken now (2026-11-25, Daniel).
# Each opens its own decision; the condition is the decision itself, through
# kehillah_crisis_action_alert_effect (common/scripted_effects/kehillah_crisis_action_alert_effects.txt).
# The general action_kehillah_active_crisis alert (kehillah_crisis_actions.txt) still opens the window.
# The aftermath and remembrance decisions are left out, as Commemorate the Faith was (9c1efc1c).

action_kehillah_crisis_prepare = {
	type = alert
	icon = action_take_decision
	is_dangerous = yes
	priority = 105
	check_create_action = { kehillah_crisis_action_alert_effect = { ACTION = action_kehillah_crisis_prepare DECISION = kehillah_crisis_prepare_decision } }
	effect = { scope:decision = { open_view_data = decision_detail } }
}
