# Ask for a Head of the Jews (2026-11-25, Daniel: "a top-bar notice when you
# can ask for a head of the Jews"). While the player's community could
# petition the realm ruler (kehillah_ask_nagid_interaction in
# common/character_interactions/kehillah_nagid_interactions.txt), the top bar
# says so; a click opens the interaction aimed at that ruler, as vanilla's
# action_end_regency opens its interaction.
#
# is_character_interaction_valid covers the interaction's is_shown and
# is_valid (seat, qualified leader, Healthy standing, cooldown, law, policy),
# so the notice disappears the moment any of them fails. The ruler's answer
# is not checked: whether he is likely to agree is the interaction's own
# acceptance breakdown.
action_kehillah_ask_nagid = {
 type = alert
 icon = action_take_decision
 priority = 200
 check_create_action = {
  if = {
   limit = {
    is_ai = no
    government_has_flag = government_is_kehillah
    exists = domicile.domicile_location.county.holder.top_liege
   }
   domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_nagid_petition_ruler }
   if = {
    limit = {
     is_character_interaction_valid = {
      interaction = kehillah_ask_nagid_interaction
      recipient = scope:kehillah_nagid_petition_ruler
     }
    }
    try_create_important_action = {
     important_action_type = action_kehillah_ask_nagid
     actor = root
     recipient = scope:kehillah_nagid_petition_ruler
    }
   }
  }
 }
 effect = {
  open_interaction_window = {
   interaction = kehillah_ask_nagid_interaction
   actor = scope:actor
   recipient = scope:recipient
  }
 }
}
