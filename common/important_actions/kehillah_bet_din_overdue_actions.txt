# =====================================================================
# The Bet Din Is Overdue (2026-11-08). While the long gap costs Stability,
# this alert replaces the plain "Cases Await the Bet Din" suggestion
# (kehillah_bet_din_actions.txt) and a click opens the activity planner,
# which names anything still in the way (no rabbi, already busy).
# =====================================================================

action_kehillah_bet_din_overdue = {
	type = alert
	icon = action_can_host_activity
	priority = 350

	check_create_action = {
		if = {
			limit = { kehillah_bet_din_overdue_trigger = yes }
			try_create_important_action = {
				important_action_type = action_kehillah_bet_din_overdue
			}
		}
	}

	effect = {
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:activity_kehillah_bet_din_conference
		}
	}
}
