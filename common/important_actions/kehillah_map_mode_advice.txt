# A one-time reactive tip for a Jewish player with a populated community map.
# Modelled on vanilla's action_reactive_advice_clerical_regions.
action_kehillah_map_mode_advice = {
	type = tutorial
	icon = generic_reactive_advice
	check_create_action = {
		if = {
			limit = {
				should_show_tutorial_alert = { LESSON = kehillah_map_mode_advice }
				is_jewish_character_trigger = yes
				has_variable = kehillah_map_view_count
				var:kehillah_map_view_count >= 1
				NOT = { has_map_mode = kehillah_communities_map }
			}
			try_create_important_action = {
				important_action_type = action_kehillah_map_mode_advice
				actor = root
			}
		}
	}
	effect = {
		start_tutorial_lesson = kehillah_map_mode_advice
	}
}
