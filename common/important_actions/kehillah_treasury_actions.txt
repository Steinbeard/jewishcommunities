# =====================================================================
# The treasury is full (2026-10-15, 20-year soak: the player's gold rose
# from 126 to about 480 and nothing pointed at the Quarter's buildings, the
# main sink; the soak agent found the upgrade by its own exploration after
# thirteen years). While a Kehillah leader holds more than
# kehillah_treasury_nudge_gold, a suggestion offers the Quarter (My Realm,
# Domain), where the buildings are raised; its text names Tzedakah too.
# =====================================================================

action_kehillah_treasury_full = {
	type = action
	icon = action_lifestyle_stewarship
	priority = 100

	check_create_action = {
		if = {
			limit = {
				government_has_flag = government_is_kehillah
				is_adult = yes
				exists = domicile
				gold >= kehillah_treasury_nudge_gold
			}
			try_create_important_action = {
				important_action_type = action_kehillah_treasury_full
			}
		}
	}

	effect = {
		open_view_data = {
			view = my_realm
			data = flag:bookmark_domain
		}
	}
}
