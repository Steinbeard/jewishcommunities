# Open Bet Din sittings do not use vanilla's invited-activity alert. Offer
# explicitly invited players a direct route to the normal RSVP screen when
# they have not enabled the generic open-activity notification.
action_kehillah_bet_din_invitation = {
	type = alert
	icon = action_invited_to_activity
	check_create_action = {
		every_invited_activity = {
			limit = {
				has_activity_type = activity_kehillah_bet_din_conference
				is_open_invite_activity = yes
				save_temporary_scope_as = activity
				root = {
					NOT = { should_notify_can_join_activity = scope:activity.activity_type }
					can_join_activity = scope:activity
					can_arrive_in_time_to_activity_minimum = scope:activity
				}
			}
			try_create_important_action = {
				important_action_type = action_kehillah_bet_din_invitation
				activity = this
			}
		}
	}
	effect = {
		scope:activity = { open_view_data = activity_list_detail_invite_window }
	}
}
