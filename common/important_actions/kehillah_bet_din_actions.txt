# =====================================================================
# Cases Await the Bet Din (2026-10-15, 20-year soak). The vanilla
# "you could host this" suggestion, as for the hunt and the coronation:
# while the leader could convene a Bet Din, a suggestion offers it, and a
# click opens the activity planner on the conference.
# =====================================================================

action_kehillah_convene_bet_din = {
	type = action
	icon = action_can_host_activity
	priority = 300

	check_create_action = {
		if = {
			limit = { kehillah_can_convene_bet_din_trigger = yes }
			try_create_important_action = {
				important_action_type = action_kehillah_convene_bet_din
			}
		}
	}

	effect = {
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:activity_kehillah_bet_din_conference
		}
	}
}
