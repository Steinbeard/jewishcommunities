# Top-bar alerts that open the lessons in
# common/tutorial_lessons/kehillah_welcome_advice.txt.
# Modelled on vanilla's action_reactive_advice_landless_adventurer.

action_kehillah_welcome_advice = {
	type = tutorial
	icon = generic_reactive_advice
	soundeffect = "event:/SFX/UI/Notifications/TopBar/sfx_ui_notifications_topbar_reactive_advice"
	check_create_action = {
		if = {
			limit = {
				should_show_tutorial_alert = { LESSON = kehillah_welcome_advice }
				has_government = kehillah_government
			}
			try_create_important_action = {
				important_action_type = action_kehillah_welcome_advice
				actor = root
			}
		}
	}
	effect = {
		start_tutorial_lesson = kehillah_welcome_advice
	}
}
