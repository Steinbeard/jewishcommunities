# An Idea to Write (2026-11-24, writing sparks). While the player holds a
# spark and could begin writing, a quiet top-bar suggestion opens the same
# choice as the Write a Sefer decision.
action_kehillah_spark_ready = {
	type = action
	icon = action_take_decision
	priority = 120

	check_create_action = {
		if = {
			limit = {
				kehillah_has_any_spark_trigger = yes
				kehillah_sefer_can_write_trigger = yes
				is_adult = yes
				is_imprisoned = no
				NOT = { any_scheme = { type = kehillah_writing } }
				piety >= kehillah_rabbi_book_piety_cost
			}
			# 2026-11-24 fix: vanilla passes the actor and fires on scope:actor; without
			# it the click had no character to open the writing screen for.
			try_create_important_action = {
				important_action_type = action_kehillah_spark_ready
				actor = root
			}
		}
	}

	effect = { scope:actor = { trigger_event = kehillah_spark.0001 } }
}
