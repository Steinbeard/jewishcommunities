# Missing Foundational Works (2026-11-25, Daniel: "a notice if your library is
# missing foundational works as a kehillah leader too and you're not currently
# acquiring a sefer"). While the player's community library lacks a work that
# is Foundational or higher in the leader's rite
# (kehillah_lib_foundational_missing_count_value, the count that already costs
# Greatness through kehillah_bd_foundational_works_value) and no copy is on
# its way (kehillah_library_commission_work), the top bar says so.
#
# The label names the work when one is missing, else gives the count
# (custom loc KehillahLibMissingFoundLabel, generated). A click opens Acquire a
# Foundational Work (common/decisions/kehillah_library_foundation_decisions.txt,
# generated), which lists the missing works to choose from and goes on to the
# usual acquisition confirmation. Until 2026-11-25 (same day) the click opened
# the library window through gui/kehillah_library_opener.gui, kept for reuse.
action_kehillah_library_missing_foundations = {
 type = alert
 icon = action_take_decision
 priority = 120
 check_create_action = {
  if = {
   limit = {
    is_ai = no
    government_has_flag = government_is_kehillah
    is_adult = yes
    is_imprisoned = no
    primary_title ?= {
     kehillah_lib_foundational_missing_count_value > 0
     NOT = { has_variable_list = kehillah_library_commission_work }
    }
   }
   # 2026-11-25 (Daniel): name the missing works in the notice; the effect
   # (generated) keeps the first few on the character for its description.
   kehillah_lib_note_missing_foundations_effect = yes
   try_create_important_action = {
    important_action_type = action_kehillah_library_missing_foundations
    actor = root
    decision = decision:kehillah_acquire_foundational_work_decision
   }
  }
 }
 effect = {
  scope:decision = { open_view_data = decision_detail }
 }
}
