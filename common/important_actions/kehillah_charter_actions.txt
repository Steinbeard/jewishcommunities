# A harsh charter warrants attention when a petition is available. Use the
# same overall grade and interaction as the Laws dashboard.
action_kehillah_harsh_charter = {
	type = alert
	icon = action_take_decision
	priority = 200
	check_create_action = {
		if = {
			limit = {
				is_ai = no
				government_has_flag = government_is_kehillah
				exists = suzerain
				kehillah_charter_favour_overall_value < 1.75
			}
			suzerain = { save_scope_as = kehillah_charter_alert_suzerain }
			if = {
				limit = {
					is_character_interaction_valid = {
						interaction = subject_modify_tributary_contract_interaction
						recipient = scope:kehillah_charter_alert_suzerain
					}
				}
				try_create_important_action = {
					important_action_type = action_kehillah_harsh_charter
					actor = root
					recipient = scope:kehillah_charter_alert_suzerain
				}
			}
		}
	}
	effect = {
		open_interaction_window = {
			interaction = subject_modify_tributary_contract_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
