# The "Beit Midrash" chip in the character window: shown for a Jewish
# landed ruler who has a library (his own temple library, or one in his
# realm) and for a Jewish theocrat holding one. 2026-11-08 (Daniel's first
# look): also for a Kehillah leader whose community has a library, so the
# window opens from any community leader's character window -- one's own,
# or another community's, to see what it holds. The domicile's Books tab
# still shows the same library body.
kehillah_lib_window_chip = {
	scope = character
	is_shown = {
		OR = {
			AND = {
				kehillah_is_jewish_landed_ruler_trigger = yes
				OR = {
					exists = var:kehillah_own_library
					any_held_title = { kehillah_lib_is_library_title_trigger = yes }
				}
			}
			AND = {
				government_has_flag = government_is_kehillah
				primary_title ?= { kehillah_lib_is_library_title_trigger = yes }
			}
		}
	}
}

# Open the window on scope:target's library. root = the player.
kehillah_lib_window_open = {
	scope = character
	saved_scopes = { target }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:target = { kehillah_lib_refresh_own_library_effect = yes }
		set_variable = { name = kehillah_lib_view_owner value = scope:target }
		if = {
			limit = { exists = scope:target.var:kehillah_own_library }
			set_variable = { name = kehillah_lib_view_title value = scope:target.var:kehillah_own_library }
		}
		if = {
			limit = { this = scope:target }
			kehillah_lib_refresh_own_library_effect = yes
		}
	}
}

# The Libraries map mode's button (gui/shared/mapmodes.gui).
kehillah_libraries_map_paint = {
	scope = character
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = { kehillah_paint_libraries_map_effect = yes }
}
