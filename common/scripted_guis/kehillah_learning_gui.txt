# =====================================================================
# 2026-11-11: a character's Learning (gui/kehillah_learning_window.gui), opened
# from the round button beside Memories in the character window. The window
# lists var list kehillah_mastered_works in the order the works were mastered;
# a work learned away from home names the place
# (var:kehillah_mastered_at_<work>, kehillah_lib_record_mastery_place_effect).
# =====================================================================

# Scope: the character shown. The button shows for a Jew who has mastered a work.
kehillah_learning_shown = {
	scope = character
	is_shown = {
		is_jewish_character_trigger = yes
		kehillah_mastered_works_count_value > 0
	}
}

# Scope: the player; scope:target = the character whose Learning opens.
kehillah_learning_open = {
	scope = character
	saved_scopes = { target }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		set_variable = { name = kehillah_learning_view value = scope:target }
	}
}
