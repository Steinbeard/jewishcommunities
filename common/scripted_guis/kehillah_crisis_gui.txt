# Compact shortcut beside the situation's Actions heading; the three standing decisions
# remain the full action menus. The player must lead the active community crisis.
kehillah_crisis_recommend = {
 scope = character
 is_shown = { kehillah_crisis_actions_open_trigger = yes is_ai = no }
 is_valid = {
  kehillah_crisis_leader_free_trigger = yes
  NOT = { has_variable = kehillah_recommend_open }
 }
 effect = {
  # Synchronous click guard. A short expiry is a backstop if an event is dismissed
  # without running its after block; that block normally clears it immediately.
  set_variable = { name = kehillah_recommend_open value = yes days = 1 }
  if = { limit = { kehillah_crisis_warning_trigger = yes } trigger_event = kehillah_crisis_recommend.0001 }
  else = { trigger_event = kehillah_crisis_recommend.0002 }
 }
}
