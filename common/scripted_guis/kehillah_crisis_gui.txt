# Military reads live title state. Each mutation rechecks eligibility.
kehillah_crisis_visible = { scope = character is_shown = { government_has_flag = government_is_kehillah } }
kehillah_crisis_council = {
 scope = character
 is_shown = { government_has_flag = government_is_kehillah }
 is_valid = { primary_title.var:kehillah_mob_threat ?= { is_alive = yes } NOT = { has_variable = kehillah_mob_away } is_imprisoned = no }
 effect = { set_variable = { name = kehillah_mob_ref value = primary_title.var:kehillah_mob_threat } trigger_event = kehillah_mob.0100 }
}

kehillah_crisis_watch = {
 scope = character
 is_shown = { government_has_flag = government_is_kehillah }
 is_valid = {
  kehillah_crisis_warning_trigger = yes
  is_imprisoned = no
  NOT = { has_variable = kehillah_mob_away }
  primary_title = { NOT = { has_variable = kehillah_crisis_action } NOT = { has_variable = kehillah_mob_defended } }
  gold >= kehillah_crisis_watch_cost
 }
 effect = {
  remove_short_term_gold = kehillah_crisis_watch_cost
  primary_title = { set_variable = { name = kehillah_crisis_action value = 1 } set_variable = { name = kehillah_crisis_action_days value = kehillah_crisis_preparation_days } }
 }
}
