# =====================================================================
# How the Nagid office shows -- 2026-11-01 (the Nagid spec; Daniel: "I
# want it to feel 'real' in the world of the game"). is_shown only, never
# executed: the GUI asks these questions with the character as root.
#
#   kehillah_nagid_chip           the office's badge in the character
#                                 window (gui/window_character.gui), for
#                                 a Nagid and for a ruler whose office
#                                 is filled
#   kehillah_nagid_chip_<service> which icon the badge shows
#   kehillah_nagid_rv_has_holder  the realm view shows the holder's
#                                 portrait beside the ruler (root: the
#                                 top liege)
#   kehillah_nagid_rv_no_office   its empty frame opens the founding or
#                                 asking decision (root: the top liege)
#   kehillah_nagid_rv_seat        a community card or the detail card
#                                 marks the realm's seat (root: its leader)
# =====================================================================

kehillah_nagid_chip = {
	scope = character
	is_shown = {
		OR = {
			kehillah_is_nagid_trigger = yes
			kehillah_realm_nagid_filled_trigger = yes
		}
	}
}

kehillah_nagid_chip_vizier = {
	scope = character
	is_shown = { kehillah_nagid_badge_service_trigger = { SERVICE = vizier } }
}

kehillah_nagid_chip_treasurer = {
	scope = character
	is_shown = { kehillah_nagid_badge_service_trigger = { SERVICE = treasurer } }
}

kehillah_nagid_chip_physician = {
	scope = character
	is_shown = { kehillah_nagid_badge_service_trigger = { SERVICE = physician } }
}

kehillah_nagid_chip_envoy = {
	scope = character
	is_shown = { kehillah_nagid_badge_service_trigger = { SERVICE = envoy } }
}

kehillah_nagid_chip_none = {
	scope = character
	is_shown = {
		NOR = {
			kehillah_nagid_badge_service_trigger = { SERVICE = vizier }
			kehillah_nagid_badge_service_trigger = { SERVICE = treasurer }
			kehillah_nagid_badge_service_trigger = { SERVICE = physician }
			kehillah_nagid_badge_service_trigger = { SERVICE = envoy }
		}
	}
}

kehillah_nagid_rv_has_holder = {
	scope = character
	is_shown = { kehillah_realm_nagid_filled_trigger = yes }
}

kehillah_nagid_rv_no_office = {
	scope = character
	is_shown = { NOT = { kehillah_realm_has_nagid_office_trigger = yes } }
}

kehillah_nagid_rv_seat = {
	scope = character
	is_shown = { kehillah_is_realm_seat_trigger = yes }
}
