# =====================================================================
# How the Nagid office shows in the realm view -- 2026-11-01 (the Nagid
# spec; Daniel: "I want it to feel 'real' in the world of the game").
# is_shown only, never executed. (A badge in the character window was
# built the same day and removed at Daniel's request: "looks pretty bad".)
#
#   kehillah_nagid_rv_has_holder  the holder's portrait beside the ruler
#                                 (root: the top liege)
#   kehillah_nagid_rv_no_office   its empty frame opens the founding or
#                                 asking decision (root: the top liege)
#   kehillah_nagid_rv_seat        a community card or the detail card
#                                 marks the realm's seat (root: its leader)
# =====================================================================

kehillah_nagid_rv_has_holder = {
	scope = character
	is_shown = { kehillah_realm_nagid_filled_trigger = yes }
}

kehillah_nagid_rv_no_office = {
	scope = character
	is_shown = { NOT = { kehillah_realm_has_nagid_office_trigger = yes } }
}

kehillah_nagid_rv_seat = {
	scope = character
	is_shown = { kehillah_is_realm_seat_trigger = yes }
}
