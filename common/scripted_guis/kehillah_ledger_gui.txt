# =====================================================================
# World ledger of Jewish communities (2026-10-29).
# Design: docs/spec/v52-world-ledger-and-standing-rank.md.
# root = the clicking player. Opening builds the list; each column
# header re-sorts it (1 Standing, 2 Prosperity, 3 Stability,
# 4 Greatness). is_shown on a sort entry = that column is the sort, so
# the header can highlight it.
# =====================================================================

kehillah_ledger_open = {
	scope = character
	is_shown = { always = yes }
	effect = { kehillah_ledger_build_effect = yes }
}

kehillah_ledger_sort_standing = {
	scope = character
	is_shown = { OR = { NOT = { has_variable = kehillah_ledger_sort } var:kehillah_ledger_sort = 1 } }
	effect = {
		set_variable = { name = kehillah_ledger_sort value = 1 }
		kehillah_ledger_build_effect = yes
	}
}

kehillah_ledger_sort_prosperity = {
	scope = character
	is_shown = { has_variable = kehillah_ledger_sort var:kehillah_ledger_sort = 2 }
	effect = {
		set_variable = { name = kehillah_ledger_sort value = 2 }
		kehillah_ledger_build_effect = yes
	}
}

kehillah_ledger_sort_stability = {
	scope = character
	is_shown = { has_variable = kehillah_ledger_sort var:kehillah_ledger_sort = 3 }
	effect = {
		set_variable = { name = kehillah_ledger_sort value = 3 }
		kehillah_ledger_build_effect = yes
	}
}

kehillah_ledger_sort_greatness = {
	scope = character
	is_shown = { has_variable = kehillah_ledger_sort var:kehillah_ledger_sort = 4 }
	effect = {
		set_variable = { name = kehillah_ledger_sort value = 4 }
		kehillah_ledger_build_effect = yes
	}
}
