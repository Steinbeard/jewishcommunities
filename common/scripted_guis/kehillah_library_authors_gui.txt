# Resolve the history character when a catalog work is opened. Doing this
# on demand also initializes older saves without a game-start migration.
kehillah_lib_cache_saadia_author = {
	scope = character
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		# Read-before-write also keeps CK3's unused-variable lint quiet.
		if = { limit = { exists = var:kehillah_lib_saadia_author } remove_variable = kehillah_lib_saadia_author }
		if = {
			limit = { exists = character:9000224 }
			set_variable = { name = kehillah_lib_saadia_author value = character:9000224 }
		}
	}
}
