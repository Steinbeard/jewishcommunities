# Reader roster for selected books. The global index is populated on mastery.
# A player's older-save mastery is indexed when that player opens a roster.

kehillah_lib_readers_work_targum_onkelos = {
	scope = character
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		clear_variable_list = kehillah_lib_readers
		kehillah_lib_index_yeshiva_readers_effect = yes
		set_variable = { name = kehillah_lib_readers_work_id value = 1 }
		if = {
			limit = {
				is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
				NOT = { is_target_in_global_variable_list = { name = kehillah_sefer_masters target = this } }
			}
			add_to_global_variable_list = { name = kehillah_sefer_masters target = this }
		}
		save_scope_as = kehillah_lib_reader_viewer
		every_in_global_list = {
			variable = kehillah_sefer_masters
			limit = {
				is_alive = yes
				has_trait = kehillah_rabbi_trait
				is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
			}
			save_scope_as = kehillah_lib_reader_candidate
			scope:kehillah_lib_reader_viewer = {
				add_to_variable_list = { name = kehillah_lib_readers target = scope:kehillah_lib_reader_candidate }
			}
		}
	}
}
