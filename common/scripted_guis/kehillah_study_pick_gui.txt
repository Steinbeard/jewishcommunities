# =====================================================================
# 2026-11-08. The Learn Torah decision's library picker (gui/decision_view_
# widgets/kehillah_decision_widget_study_library.gui). Picking a library
# stores it on the player so the widget can show its shelf; the books on
# that shelf run kehillah_lib_study_<work> (kehillah_library_study_gui.txt).
# =====================================================================

kehillah_study_pick_library = {
	scope = character
	saved_scopes = { library }
	is_shown = { always = yes }
	is_valid = {
		kehillah_study_can_read_library_trigger = { CHARACTER = root LIBRARY = scope:library }
	}
	effect = {
		set_variable = { name = kehillah_study_pick_library value = scope:library }
	}
}
