# =====================================================================
# THE YESHIVA'S TEACHING FOCUS -- V29 section 11.3 (2026-10-12). The
# library panel's Yeshiva section has a row of three buttons (vanilla's
# button_standard_list filter row, jomini_achievements_window.gui), one
# per track. kehillah_yeshiva_focus_<track> sets it; kehillah_yeshiva_
# focus_is_<track> only answers "is this the current focus" for the
# button's `down` state. root = the player.
# =====================================================================

kehillah_yeshiva_focus_talmudics = {
	scope = character
	is_shown = { always = yes }
	is_valid = {
		kehillah_yeshiva_can_set_focus_trigger = { TRACK = talmudics }
	}
	effect = {
		kehillah_yeshiva_set_focus_effect = { TRACK = talmudics }
	}
}

kehillah_yeshiva_focus_is_talmudics = {
	scope = character
	is_shown = { always = yes }
	is_valid = {
		primary_title ?= {
			NOT = { var:kehillah_yeshiva_specialization ?= flag:parshanut }
			NOT = { var:kehillah_yeshiva_specialization ?= flag:hashkafa }
		}
	}
}
