# The Library acquisition-catalog opener (V48). It rebuilds a derived list of
# missing works that have already circulated. Named-work confirmation actions
# are generated in kehillah_library_acquire_work_gui.txt.

kehillah_lib_acquire_more = {
	scope = character

	is_shown = { always = yes }

	is_valid = {
		kehillah_lib_can_commission_trigger = yes
		is_adult = yes
		is_imprisoned = no
		# 2026-11-08: any library the character holds (a Kehillah's, or a
		# landed temple library), not only a Kehillah title.
		OR = {
			AND = {
				government_has_flag = government_is_kehillah
				primary_title = { NOT = { has_variable_list = kehillah_library_commission_work } }
			}
			var:kehillah_own_library ?= { NOT = { has_variable_list = kehillah_library_commission_work } }
		}
	}

	effect = { kehillah_library_refresh_acquisition_candidates_effect = yes }
}
