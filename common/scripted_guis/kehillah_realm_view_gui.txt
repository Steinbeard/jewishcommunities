# =====================================================================
# Realm view of Jewish communities (2026-10-28).
# Design: docs/spec/2026-10-28-realm-jewish-communities-view.md.
#
# Two scripted_guis, because the chip asks two questions with two
# different roots (vanilla does the same: pam_has_schism_stance is
# IsShown on Character.MakeScope, window_character.gui):
#   kehillah_realm_view_chip  root = the viewed character. is_shown
#                             only; never executed.
#   kehillah_realm_view_open  root = the clicking player, scope:target
#                             = the viewed character (vanilla's
#                             AddScope( 'child', Character.MakeScope )).
#                             Builds the view onto the player.
# =====================================================================

kehillah_realm_view_chip = {
	scope = character
	is_shown = { kehillah_rv_has_chip_trigger = yes }
}

# A portrait in the gallery was clicked: show that community in the
# detail card. root = the player, scope:target = the community title
# (Title.MakeScope on the card -- a title scope, not a bare list item,
# which is the form that failed for the library's donate buttons).
kehillah_realm_view_select = {
	scope = character
	saved_scopes = { target }
	is_shown = { always = yes }
	is_valid = { exists = scope:target }
	effect = { set_variable = { name = kehillah_rv_selected value = scope:target } }
}

kehillah_realm_view_open = {
	scope = character
	saved_scopes = { target }
	is_shown = { always = yes }
	is_valid = { exists = scope:target }
	effect = { kehillah_realm_view_build_effect = yes }
}
