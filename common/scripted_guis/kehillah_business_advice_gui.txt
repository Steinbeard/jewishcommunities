# One request at a time. Consumption must work even if the original condition changed.
kehillah_business_open_request_consume = {
	scope = character
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		remove_variable = kehillah_business_open_request
		remove_variable = kehillah_business_households_request
	}
}
kehillah_business_request_open = {
	scope = character
	is_shown = { government_has_flag = government_is_kehillah exists = domicile }
	is_valid = { government_has_flag = government_is_kehillah exists = domicile }
	effect = {
		remove_variable = kehillah_business_households_request
		set_variable = { name = kehillah_business_open_request value = yes days = 1 }
	}
}
kehillah_households_request_open = {
	scope = character
	is_shown = { government_has_flag = government_is_kehillah exists = domicile }
	is_valid = { government_has_flag = government_is_kehillah exists = domicile }
	effect = {
		remove_variable = kehillah_business_open_request
		set_variable = { name = kehillah_business_households_request value = yes days = 1 }
	}
}
