# Selected-book starters for the existing Learn Torah personal scheme.
# Keep IDs synchronized with kehillah_study_torah_pick_next_work_effect.
# The decision's script-value cost is currently 10 piety; this button uses
# that same amount and requires the same scheme/access conditions.

kehillah_lib_study_work_targum_onkelos = {
	scope = character
	is_shown = { always = yes }
	is_valid = {
		government_has_flag = government_is_kehillah
		has_trait = kehillah_rabbi_trait
		is_adult = yes
		is_imprisoned = no
		is_jewish_character_trigger = yes
		piety >= 10
		NOT = { any_scheme = { type = kehillah_study_torah } }
		can_start_scheme = { type = kehillah_study_torah target_character = root }
		primary_title = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_targum_onkelos } }
	}
	effect = {
		add_piety = -10
		start_scheme = { type = kehillah_study_torah target_character = root }
		kehillah_study_torah_resolve_library_effect = { CHARACTER = root }
		set_variable = { name = kehillah_study_current_work_id value = 1 }
		trigger_event = kehillah_study_torah.0002
	}
}
