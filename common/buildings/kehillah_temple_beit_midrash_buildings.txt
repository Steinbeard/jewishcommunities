# =====================================================================
# The Beit Midrash on a temple holding (2026-11-08, Sefer redesign,
# Daniel's decision 1). A Jewish holder of a temple holding -- the ruler
# himself or a Jewish theocratic vassal -- builds it; its barony title
# becomes a library (kehillah_lib_register_library_effect), tied to the
# map. Level 2 can seat a yeshiva (spec section 8); level 3 copies faster.
# The level never limits how many books the library holds (V42).
# =====================================================================

kehillah_temple_beit_midrash_01 = {
	construction_time = standard_construction_time

	can_construct_potential = {
		has_holding_type = church_holding
		scope:holder = { faith.religion = religion:judaism_religion }
	}

	is_enabled = {
		custom_tooltip = {
			text = kehillah_temple_beit_midrash_jewish_holder_tt
			barony.holder ?= { faith.religion = religion:judaism_religion }
		}
	}
	show_disabled = no

	cost_gold = normal_building_tier_1_cost

	character_modifier = {
		monthly_piety = 0.2
		monthly_learning_lifestyle_xp_gain_mult = 0.05
	}
	county_modifier = {
		development_growth_factor = 0.02
	}

	on_complete = {
		barony = {
			kehillah_lib_register_library_effect = yes
			if = {
				limit = { NOT = { has_variable = kehillah_library_seeded } }
				set_variable = { name = kehillah_library_seeded value = yes }
				kehillah_seed_landed_library_effect = yes
			}
		}
	}

	next_building = kehillah_temple_beit_midrash_02
	type_icon = "icon_building_monastic_schools.dds"

	ai_value = {
		base = 15
		ai_tier_1_building_modifier = yes
		ai_tier_1_pious_building_preference_modifier = yes
	}
}
