# =====================================================================
# The yeshiva's public face on the map -- V29 section 3.5
# (docs/spec/v29-yeshiva.md). Placed by kehillah_place_yeshiva_map_
# building_effect, never built from a menu (can_construct_potential is
# always no).
#
# The INSTITUTION is the domicile building kehillah_beit_midrash_03; this
# is only its visible face in the host's province. Its modifiers go to
# the BARONY HOLDER -- the host lord -- which is the point: hosting a
# renowned academy is worth something to him (V29 section 3.5, "what it
# gives the host lord"). Kept modest.
#
# is_enabled: the building works only while a Kehillah with a yeshiva
# still has its quarter in this province. If the community leaves or
# loses the yeshiva, the hall stands empty: the great building drops to
# its fallback modifiers, the special building to none.
#
# Two keys because the engine needs a building type per slot kind:
# kehillah_yeshiva_hall is the GREAT building (own slot, never collides
# with a province's landmark); kehillah_yeshiva_hall_special is the
# fallback if great buildings will not place outside All Under Heaven
# (V29 spike 1), used only into an EMPTY special slot.
#
# MODELS: vanilla ships no Jewish building meshes. These reuse the base
# game's generic temple-holding meshes per graphical region as
# placeholders (no DLC flag needed); a custom model is optional polish.
# =====================================================================

kehillah_yeshiva_hall = {
	type = great_building
	type_icon = "icon_building_university.dds"

	can_construct_potential = { always = no }

	is_enabled = {
		custom_tooltip = {
			text = kehillah_yeshiva_hall_enabled_tt
			any_province_domicile = {
				has_domicile_building_or_higher = kehillah_beit_midrash_03
			}
		}
	}
	show_disabled = yes

	asset = {
		type = pdxmesh
		name = "building_western_temple_generic_02_mesh"
		graphical_regions = { graphical_western }
	}
	asset = {
		type = pdxmesh
		name = "building_mediterranean_temple_generic_01_mesh"
		graphical_regions = { graphical_mediterranean }
	}
	asset = {
		type = pdxmesh
		name = "building_mena_temple_generic_01_mesh"
		graphical_regions = { graphical_mena graphical_india graphical_steppe graphical_east_asia graphical_siberia }
	}

	# Host benefits: scholars, students, copyists and their trade come to a
	# yeshiva town, and a renowned academy is a small source of prestige.
	character_modifier = {
		monthly_prestige = 0.2
	}
	county_modifier = {
		development_growth = 0.1
	}
	fallback = {
		county_modifier = {
			development_growth = 0.02
		}
	}

	next_building = kehillah_great_yeshiva_hall
	flag = travel_point_of_interest_learning
	flag = kehillah_yeshiva_building
}
