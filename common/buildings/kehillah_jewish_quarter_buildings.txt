# =====================================================================
# THE JEWISH QUARTER DISTRICT'S PRIMARY BUILDING (2026-11-21).
# common/districts/kehillah_jewish_quarter_districts.txt.
#
# What the host gets for his Jews: a guild-sized artisan tax (vanilla's
# pb_guild_01 pays 0.25). What the Quarter asks of his market
# (2026-11-23, reworked to vanilla's pattern, Daniel): like vanilla's
# religious community district, broad categories for a people's
# consumption, comforts for its table and books for its learning; and,
# like vanilla's communion wine, specific goods only for religious reasons,
# wine for Kiddush and cattle for the shochet (kashrut,
# docs/spec/2026-11-23-kashrut.md). Vanilla rural-tier amounts. A need the
# market meets pays a little more. The community's own Affluence reads the
# same market (kehillah_bd_market_supply_prosperity_value).
# =====================================================================

kehillah_jewish_quarter_01 = {
	construction_time = district_tier_1_construction_time
	cost_gold = district_tier_1_cost
	type_icon = "district_buildings/religious_community_district.dds"

	province_modifier = {
		monthly_artisan_tax = 0.25
		# Room for the community's yeshiva (2026-11-22,
		# docs/spec/2026-11-22-yeshiva-rework.md section 2).
		building_slots_for_host_district = 1
	}

	# Kashrut (2026-11-23): meat is salted to draw out its blood, so salt is
	# in demand where Jews live: vanilla's in-demand size (1%), as its diet
	# doctrines make their foods dearer. Shown on the Kashrut doctrine.
	market_modifier = {
		trade_good_salt_sell_price_mult = 0.01
	}

	# The community's table: any comfort (cheese, fruit, oil, honey, tea).
	trade_good_need = {
		trade_good_category = trade_category_comforts
		amount = rural_resource_need
		modifiers = {
			province_modifier = {
				monthly_artisan_tax = 0.05
			}
		}
	}
	# Its learning.
	trade_good_need = {
		trade_good_category = trade_category_books
		amount = rural_resource_need
		modifiers = {
			province_modifier = {
				monthly_artisan_tax = 0.05
			}
		}
	}
	# Wine for Kiddush, as vanilla's communion wine asks wine of a temple.
	trade_good_need = {
		trade_good = trade_good_wine
		amount = rural_resource_need
		modifiers = {
			province_modifier = {
				monthly_artisan_tax = 0.05
			}
		}
	}
	# Kashrut (2026-11-23, docs/spec/2026-11-23-kashrut.md). Kosher meat is
	# not bought at the market: the community's shochet slaughters live
	# beasts. So the quarter asks the market for cattle, not meat, as
	# vanilla's diet doctrines swap one food for another. Vanilla would key
	# that to the county's rite; a Jewish quarter is in another rite's county,
	# so it is always on here.
	trade_good_need = {
		trade_good = trade_good_cattle
		amount = rural_resource_need
		modifiers = {
			province_modifier = {
				monthly_artisan_tax = 0.05
			}
		}
	}

	can_construct_potential = {
		always = no
	}

	ai_value = {
		base = 0
	}
}

# =====================================================================
# THE YESHIVA IN THE JEWISH QUARTER (2026-11-22).
# docs/spec/2026-11-22-yeshiva-rework.md section 2; V29 section 3.5.
#
# The community's yeshiva, seen from the host's side: placed by
# kehillah_place_yeshiva_map_building_effect, raised to the Great Yeshiva by
# kehillah_raise_great_yeshiva_effect, never built from a menu. It goes with
# the quarter district when the community ends. The institution itself
# (students, the Rosh Yeshiva, renown) lives in the domicile
# (kehillah_beit_midrash_03). The host's benefits are the old yeshiva hall's,
# moved across unchanged.
# =====================================================================

kehillah_quarter_yeshiva_01 = {
	construction_time = district_tier_1_construction_time
	cost_gold = district_tier_1_cost
	type_icon = "special_buildings/university.dds"

	# add_building honours this even from script (live 2026-11-22: "cannot
	# be constructed in district"). Only the placement effects set the
	# global flag, around their add_building call; no player ever sees it.
	can_construct_potential = { has_global_variable = kehillah_placing_quarter_yeshiva }

	is_enabled = {
		custom_tooltip = {
			text = kehillah_yeshiva_hall_enabled_tt
			any_province_domicile = {
				has_domicile_building_or_higher = kehillah_beit_midrash_03
			}
		}
	}
	show_disabled = yes

	character_modifier = {
		monthly_prestige = 0.2
	}
	county_modifier = {
		prosperity_growth = 0.1
	}

	next_building = kehillah_quarter_yeshiva_02
	flag = kehillah_yeshiva_building

	ai_value = {
		base = 0
	}
}
