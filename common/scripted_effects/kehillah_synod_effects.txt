# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).

# ---------------------------------------------------------------------
# Game start, before the lobby (kehillah_synod_before_lobby), so no
# doctrine-change toast is posted. Vanilla's on_game_start (game_start.txt,
# "Rabbene Gershom outlaws polygamy") gives every start on or after
# 1000.1.1 a monogamous Rabbinic main rite. That herem bound Ashkenaz
# only, and in this mod Ashkenaz becomes its own rite at the 1066 synod,
# so the general rite gets its polygamy back. Both of vanilla's marker
# variables go on the rite: the yearly check in yearly_on_actions.txt
# reads them there (NOT = { a b }, so it needs both), and would otherwise
# fire jewish_events.0001 every year and undo this. Pre-1000 starts are
# left to vanilla's own event.
# ---------------------------------------------------------------------
kehillah_restore_general_rite_polygamy_effect = {
	if = {
		limit = {
			game_start_date >= 1000.1.1
			exists = rite:rabbinism
		}
		rite:rabbinism = {
			change_rite_doctrine = doctrine_polygamy
			set_variable = rabbinicism_accepted_monogamy
			set_variable = rabbinicism_rejected_monogamy
		}
	}
}

# ---------------------------------------------------------------------
# Seed Minhag Ashkenaz before play begins, without a news toast. Scope is the
# global start hook, so a living Rabbinic character performs the creation.
# ---------------------------------------------------------------------
kehillah_seed_minhag_ashkenaz_effect = {
	if = {
		limit = {
			game_start_date >= 1000.1.1
			NOT = { exists = global_var:kehillah_minhag_ashkenaz_founded }
			exists = title:d_kehillah_worms.holder
		}
		title:d_kehillah_worms.holder = {
			if = {
				limit = { NOT = { exists = rite:minhag_ashkenaz } }
			create_rite_from_type = {
				type = minhag_ashkenaz
				convert = yes
			}
			}
		}
		set_global_variable = { name = kehillah_minhag_ashkenaz_founded value = yes }
		rite:minhag_ashkenaz = { add_rite_culture = culture:ashkenazi }
		rite:rabbinism ?= { remove_rite_culture = culture:ashkenazi }
		every_living_character = {
			limit = {
				faith = faith:rabbinic_faith
				NOT = { rite = rite:minhag_ashkenaz }
				OR = {
					culture = culture:ashkenazi
					kehillah_ashkenazi_kehillah_member_trigger = yes
				}
			}
			set_character_rite_history = rite:minhag_ashkenaz
		}
		every_county = {
			limit = {
				faith = faith:rabbinic_faith
				culture = culture:ashkenazi
			}
			set_county_rite = rite:minhag_ashkenaz
		}
	}
}

# Scope: a just-created character. Script that creates Ashkenazi Jews with
# a hard-coded rite:rabbinism (the Norman-conquest founders, the pool
# repopulation patch) calls this so they join Minhag Ashkenaz once it
# exists. Births need nothing: children take their parents' rite.
kehillah_apply_ashkenaz_rite_effect = {
	if = {
		limit = {
			exists = global_var:kehillah_minhag_ashkenaz_founded
			faith = faith:rabbinic_faith
			rite = rite:rabbinism
			OR = {
				culture = culture:ashkenazi
				kehillah_ashkenazi_kehillah_member_trigger = yes
			}
		}
		set_character_rite = rite:minhag_ashkenaz
	}
}

# ---------------------------------------------------------------------
# Seed the other regional Rabbinic minhagim for the selected bookmark.
# In 867, the Palestinian-influenced zone includes the represented Italian
# and Byzantine communities as well as Jerusalem/Tiberias. This records
# local custom without implying that every community had a separate Talmudic
# academy; Geonic authority over interpretation remains centralized. Sepharad
# is assigned to Sephardi characters and counties from 1066 on. Each place
# still has only one community title.
# ---------------------------------------------------------------------
kehillah_set_community_minhag_yerushalmi_effect = {
	holder ?= {
		if = { limit = { faith = faith:rabbinic_faith } set_character_rite_history = rite:minhag_yerushalmi }
		every_courtier = {
			limit = { faith = faith:rabbinic_faith }
			set_character_rite_history = rite:minhag_yerushalmi
		}
	}
	title_capital_county ?= {
		if = { limit = { faith = faith:rabbinic_faith } set_county_rite = rite:minhag_yerushalmi }
	}
}

kehillah_seed_minhag_yerushalmi_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:kehillah_minhag_yerushalmi_founded }
			exists = title:d_kehillah_jerusalem.holder
		}
		title:d_kehillah_jerusalem.holder = {
			if = {
				limit = { NOT = { exists = rite:minhag_yerushalmi } }
			create_rite_from_type = {
				type = minhag_yerushalmi
				convert = yes
			}
			}
		}
		set_global_variable = { name = kehillah_minhag_yerushalmi_founded value = yes }
	}
	if = {
		limit = { exists = rite:minhag_yerushalmi }
		# Jerusalem's community is present at all three bookmarks.
		title:d_kehillah_jerusalem = { kehillah_set_community_minhag_yerushalmi_effect = yes }
		# Tiberias and the represented Italian/Byzantine communities follow
		# Palestinian-influenced custom in the 867 bookmark only.
		if = {
			limit = { game_start_date < 900.1.1 }
			if = {
				limit = { exists = title:d_kehillah_tiberias.holder }
				title:d_kehillah_tiberias = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			# Italy: Rome, Lucca, Bari, Oria and Palermo. Byzantine
			# Constantinople. These are the extant 867 seats in the roster.
			if = {
				limit = { exists = title:d_kehillah_rome.holder }
				title:d_kehillah_rome = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			if = {
				limit = { exists = title:d_kehillah_lucca.holder }
				title:d_kehillah_lucca = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			if = {
				limit = { exists = title:d_kehillah_bari.holder }
				title:d_kehillah_bari = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			if = {
				limit = { exists = title:d_kehillah_oria.holder }
				title:d_kehillah_oria = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			if = {
				limit = { exists = title:d_kehillah_palermo.holder }
				title:d_kehillah_palermo = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
			if = {
				limit = { exists = title:d_kehillah_constantinople.holder }
				title:d_kehillah_constantinople = { kehillah_set_community_minhag_yerushalmi_effect = yes }
			}
		}
	}
}

# ---------------------------------------------------------------------
# Open a synod inside a Sitting of the Bet Din. Scope: the host, from
# phase_bet_din_case_3's on_phase_active after two ordinary rulings
# (scope:activity). The host's own vote is the 1 seeded into votes_for.
# The cooldown starts now, and kehillah_synod_in_session waives the
# intent's requirements for the rest of the sitting.
# ---------------------------------------------------------------------
kehillah_synod_open_session_effect = {
	scope:activity = {
        clear_variable_list = kehillah_synod_voted
        clear_variable_list = kehillah_synod_voted_for
        clear_variable_list = kehillah_synod_voted_against
        clear_variable_list = kehillah_synod_electorate
        add_to_variable_list = { name = kehillah_synod_voted target = root }
        add_to_variable_list = { name = kehillah_synod_voted_for target = root }
        add_to_variable_list = { name = kehillah_synod_electorate target = root }

		set_variable = { name = kehillah_synod_votes_for value = 1 }
		set_variable = { name = kehillah_synod_votes_against value = 0 }
	}
	if = {
		limit = { has_activity_intent = kehillah_bet_din_sanhedrin_intent }
		scope:activity = { set_variable = { name = kehillah_synod_level value = flag:sanhedrin } }
	}
	else = {
		scope:activity = { set_variable = { name = kehillah_synod_level value = flag:regional } }
	}
	set_variable = { name = kehillah_synod_in_session days = 365 }
	set_variable = { name = kehillah_synod_cooldown years = kehillah_synod_cooldown_years }
	trigger_event = { id = kehillah_synod.0100 days = kehillah_synod_motion_delay_days }
}

# Put the chosen motion before the sitting. Scope: the host, in a motion
# event. Like vanilla's Ecumenical Council doctrine debate, the activity
# stores the proposed doctrine as a scope as well as the local motion flag.
# The doctrine is set by the doctrine-motion event before this shared helper.
# MOTION: flag:<takkanah>.
kehillah_synod_put_motion_effect = {
	involved_activity ?= {
        set_variable = { name = kehillah_synod_takkanah value = $MOTION$ }
        add_activity_log_entry = { key = kehillah_synod_vote_for_log character = root tags = { completed } show_in_conclusion = yes }
    }
	add_prestige = {
		value = kehillah_synod_prestige_cost
		multiply = -1
	}
	trigger_event = kehillah_synod.0101
}

# ---------------------------------------------------------------------
# Write a takkanah to the scoped leader's community and to every
# registered community it binds: every community of the binder's rite at a
# Synod (2026-11-15, was the minhag region), every community at a
# Sanhedrin (scope:kehillah_synod_level). VAR: the variable name.
# ---------------------------------------------------------------------
kehillah_synod_bind_effect = {
	save_scope_as = kehillah_synod_binder
	primary_title ?= { set_variable = $VAR$ }
	if = {
		limit = { scope:kehillah_synod_level ?= flag:sanhedrin }
		custom_tooltip = kehillah_sanhedrin_binds_all_tt
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				holder ?= {
					NOT = { this = scope:kehillah_synod_binder }
					government_has_flag = government_is_kehillah
				}
			}
			set_variable = $VAR$
		}
	}
	else = {
		custom_tooltip = kehillah_synod_binds_region_tt
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				holder ?= {
					NOT = { this = scope:kehillah_synod_binder }
					government_has_flag = government_is_kehillah
					kehillah_shares_rite_trigger = { OTHER = scope:kehillah_synod_binder }
				}
			}
			set_variable = $VAR$
		}
	}
}

# ---------------------------------------------------------------------
# 2026-11-25 dynamic agenda: a common undertaking. Marks the host's
# community and every community it binds (the same reach as
# kehillah_synod_bind_effect) for kehillah_synod_undertaking_cooldown_years,
# and runs EFFECT on each bound community's leader other than the host.
# VAR: the cooldown variable. EFFECT: a scripted effect, in the leader's
# scope; scope:kehillah_synod_binder is the host.
# ---------------------------------------------------------------------
kehillah_synod_undertake_effect = {
	save_scope_as = kehillah_synod_binder
	primary_title ?= { set_variable = { name = $VAR$ years = kehillah_synod_undertaking_cooldown_years } }
	if = {
		limit = { scope:kehillah_synod_level ?= flag:sanhedrin }
		custom_tooltip = kehillah_sanhedrin_undertaking_all_tt
		hidden_effect = {
			every_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					holder ?= {
						NOT = { this = scope:kehillah_synod_binder }
						government_has_flag = government_is_kehillah
					}
				}
				set_variable = { name = $VAR$ years = kehillah_synod_undertaking_cooldown_years }
				holder = { $EFFECT$ = yes }
			}
		}
	}
	else = {
		custom_tooltip = kehillah_synod_undertaking_rite_tt
		hidden_effect = {
			every_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					holder ?= {
						NOT = { this = scope:kehillah_synod_binder }
						government_has_flag = government_is_kehillah
						kehillah_shares_rite_trigger = { OTHER = scope:kehillah_synod_binder }
					}
				}
				set_variable = { name = $VAR$ years = kehillah_synod_undertaking_cooldown_years }
				holder = { $EFFECT$ = yes }
			}
		}
	}
}

# ---------------------------------------------------------------------
# 2026-11-25 law-and-custom doctrine motions. Scope: the host.
# Score how controversial it would be for his rite to take $DOCTRINE$, into
# var:$VAR$ on him (values in kehillah_synod_values.txt). STEP: the
# motion's base controversy.
# ---------------------------------------------------------------------
kehillah_synod_score_doctrine_effect = {
	set_variable = { name = $VAR$ value = $STEP$ }
	if = {
		limit = { faith.main_rite ?= { rite_has_doctrine = $DOCTRINE$ } }
		change_variable = { name = $VAR$ multiply = 0.5 }
	}
	if = {
		limit = {
			exists = rite
			exists = faith.main_rite
			rite = { this = prev.faith.main_rite }
		}
		change_variable = { name = $VAR$ add = kehillah_synod_controversy_main_rite_add }
	}
	else_if = {
		limit = { exists = rite exists = faith.main_rite }
		change_variable = {
			name = $VAR$
			add = {
				value = "rite.divergence(faith.main_rite)"
				multiply = 0.5
			}
		}
	}
}

# Show the controversy band of var:$VAR$ (scope: the host).
kehillah_synod_controversy_tt_effect = {
	if = {
		limit = { var:$VAR$ ?= 0 }
		custom_tooltip = kehillah_synod_controversy_none_tt
	}
	else_if = {
		limit = { var:$VAR$ < kehillah_synod_controversy_moderate }
		custom_tooltip = kehillah_synod_controversy_minor_tt
	}
	else_if = {
		limit = { var:$VAR$ < kehillah_synod_controversy_major }
		custom_tooltip = kehillah_synod_controversy_moderate_tt
	}
	else_if = {
		limit = { var:$VAR$ < kehillah_synod_controversy_grave }
		custom_tooltip = kehillah_synod_controversy_major_tt
	}
	else = { custom_tooltip = kehillah_synod_controversy_grave_tt }
}

# ---------------------------------------------------------------------
# Enact the takkanah a synod passed. Scope: the host.
# scope:kehillah_synod_takkanah: the motion flag.
# scope:kehillah_synod_proposed_doctrine: the doctrine scope saved from the
# activity, following vanilla's Ecumenical Council proposal pattern.
# scope:kehillah_synod_level: flag:regional or flag:sanhedrin.
# ---------------------------------------------------------------------
kehillah_synod_enact_takkanah_effect = {
	# 2026-11-08 Sefer redesign: a question of belief (kehillah_reform_*).
	if = {
		limit = { scope:kehillah_synod_takkanah = flag:tenet_reform }
		kehillah_reform_enact_effect = yes
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:herem_polygamy }
		custom_tooltip = kehillah_synod_renews_herem_tt
		rite:minhag_ashkenaz ?= { change_rite_doctrine = scope:kehillah_synod_proposed_doctrine }
		set_global_variable = { name = kehillah_gershom_herem_renewed value = yes }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:herem_all_israel }
		custom_tooltip = kehillah_sanhedrin_herem_all_israel_tt
		rite:rabbinism ?= { change_rite_doctrine = scope:kehillah_synod_proposed_doctrine }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
	}

    # A personal sanction is revalidated when acknowledged.
    else_if = {
        limit = { scope:kehillah_synod_takkanah = flag:herem_person }
        if = {
            limit = { scope:kehillah_synod_herem_target ?= { is_alive = yes kehillah_synod_herem_target_trigger = { HOST = root } } }
            save_scope_value_as = { name = kehillah_synod_applied value = yes }
            scope:kehillah_synod_herem_target = {
                custom_tooltip = kehillah_synod_herem_person_tt
                hidden_effect = {
                    save_scope_value_as = { name = kehillah_cherem_pronounced_by value = flag:synod }
                    kehillah_cherem_pronounce_effect = yes
                }
            }
            kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
        }
        else = { custom_tooltip = kehillah_reform_lapsed_tt }
    }
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:ketubbah_clause }
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_ketubbah_clause }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_ketubbah_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_ketubbah_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:gentile_courts }
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_gentile_courts }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_gentile_courts_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:informers }
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_informers }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_informers_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:hayishuv }
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_hayishuv }
		kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = kehillah_synod_hayishuv_prosperity }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_hayishuv_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_hayishuv_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:letters }
		kehillah_synod_bind_effect = { VAR = kehillah_takkanah_letters }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_letters_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_letters_greatness }
	}

    # Practice proposals revalidate the exact rite and its previous doctrine.
    else_if = {
        limit = { scope:kehillah_synod_takkanah = flag:rite_doctrine }
        if = {
            limit = {
                exists = scope:kehillah_synod_proposal_rite
                exists = scope:kehillah_synod_previous_doctrine
                rite = scope:kehillah_synod_proposal_rite
                rite = {
                    rite_has_doctrine = scope:kehillah_synod_previous_doctrine
                    NOT = { rite_has_doctrine = scope:kehillah_synod_proposed_doctrine }
                    rite_can_choose_doctrine = scope:kehillah_synod_proposed_doctrine
                }
            }
            rite = { change_rite_doctrine = scope:kehillah_synod_proposed_doctrine }
            save_scope_value_as = { name = kehillah_synod_applied value = yes }
            custom_tooltip = kehillah_synod_rite_doctrine_tt
            kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
            if = {
                limit = { exists = scope:kehillah_synod_controversy scope:kehillah_synod_controversy >= kehillah_synod_controversy_moderate }
                kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_controversy_stability_value }
            }
        }
        else = { custom_tooltip = kehillah_reform_lapsed_tt }
    }
	# 2026-11-25 dynamic agenda: the gathering names a head of the rite.
	# Re-checked: he may have died, or someone else been raised, since the vote.
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:head_of_rite }
		if = {
			limit = {
				exists = scope:kehillah_synod_nominee
				scope:kehillah_synod_nominee = {
					kehillah_synod_head_candidate_trigger = { HOST = root }
					kehillah_synod_rite_headship_open_trigger = yes
				}
			}

            scope:kehillah_synod_nominee = {
                hidden_effect = { kehillah_gaonate_grant_effect = yes }
                if = {
                    limit = { has_title = title:d_kehillah_gaonate exists = rite.head_of_rite rite.head_of_rite = this }
                    save_scope_value_as = { name = kehillah_synod_applied value = yes }
                    custom_tooltip = kehillah_synod_head_named_tt
                    debug_log = "KEHILLAH_SYNOD: title and engine headship confirmed"
                }
                else = { custom_tooltip = kehillah_synod_head_unconfirmed_tt debug_log = "KEHILLAH_SYNOD: headship grant not confirmed" }
            }
            if = {
                limit = { scope:kehillah_synod_applied ?= yes }
                kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_passed_greatness }
            }
		}
		else = { custom_tooltip = kehillah_synod_head_lapsed_tt }
	}
	# 2026-11-25 dynamic agenda: common undertakings.
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:captives }
		custom_tooltip = kehillah_synod_captives_tt
		kehillah_synod_undertake_effect = { VAR = kehillah_undertaking_captives EFFECT = kehillah_synod_captives_give_effect }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_captives_stability }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_captives_greatness }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:fast }
		custom_tooltip = kehillah_synod_fast_tt
		kehillah_synod_undertake_effect = { VAR = kehillah_undertaking_fast EFFECT = kehillah_synod_fast_keep_effect }
		add_piety = minor_piety_gain
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_synod_fast_stability }
	}
	else_if = {
		limit = { scope:kehillah_synod_takkanah = flag:levy }
		custom_tooltip = kehillah_synod_levy_tt
		kehillah_synod_undertake_effect = { VAR = kehillah_undertaking_levy EFFECT = kehillah_synod_levy_give_effect }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_synod_levy_greatness }
	}
	if = {
		limit = { scope:kehillah_synod_level ?= flag:sanhedrin scope:kehillah_synod_applied ?= yes }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sanhedrin_passed_greatness }
	}
}

# Snapshot the ruling in on_complete, before the activity link is cleared.
# Scope: host; scope:activity: the completion callback's activity.
kehillah_synod_prepare_closing_ruling_effect = {
	save_scope_value_as = { name = kehillah_synod_applied value = no }
	scope:activity.var:kehillah_reform_topic ?= { save_scope_as = kehillah_reform_topic }
	scope:activity.var:kehillah_reform_replacement ?= { save_scope_as = kehillah_reform_replacement }
	if = {
		limit = { scope:activity.var:kehillah_synod_takkanah ?= flag:tenet_reform exists = scope:activity.var:kehillah_reform_topic }
		set_variable = { name = kehillah_reform_topic value = scope:activity.var:kehillah_reform_topic }
		set_variable = { name = kehillah_reform_rite value = scope:activity.var:kehillah_reform_rite }
		set_variable = { name = kehillah_reform_action value = scope:activity.var:kehillah_reform_action }
		remove_variable = kehillah_reform_replacement
		if = { limit = { exists = scope:activity.var:kehillah_reform_replacement } set_variable = { name = kehillah_reform_replacement value = scope:activity.var:kehillah_reform_replacement } }
	}
	scope:activity.var:kehillah_synod_proposal_rite ?= { save_scope_as = kehillah_synod_proposal_rite }
	scope:activity.var:kehillah_synod_previous_doctrine ?= { save_scope_as = kehillah_synod_previous_doctrine }
	save_scope_value_as = {
		name = kehillah_synod_takkanah
		value = scope:activity.var:kehillah_synod_takkanah
	}
	scope:activity.var:kehillah_synod_proposed_doctrine ?= {
		save_scope_as = kehillah_synod_proposed_doctrine
	}
	scope:activity.var:kehillah_synod_herem_target ?= { save_scope_as = kehillah_synod_herem_target }
	scope:activity.var:kehillah_synod_nominee ?= { save_scope_as = kehillah_synod_nominee }
	save_scope_value_as = {
		name = kehillah_synod_level
		value = scope:activity.var:kehillah_synod_level
	}
	# 2026-11-25 law and custom: the controversy, for the ruling's cost.
	if = {
		limit = { exists = scope:activity.var:kehillah_synod_controversy }
		save_scope_value_as = {
			name = kehillah_synod_controversy
			value = scope:activity.var:kehillah_synod_controversy
		}
	}
	if = {
		limit = { kehillah_synod_closing_quorum_trigger = no }
		save_scope_value_as = { name = kehillah_synod_outcome value = flag:no_quorum }
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = kehillah_synod_votes_for
				has_variable = kehillah_synod_votes_against
				var:kehillah_synod_votes_for > var:kehillah_synod_votes_against
			}
		}
		save_scope_value_as = { name = kehillah_synod_outcome value = flag:passed }
	}
	else = {
		save_scope_value_as = { name = kehillah_synod_outcome value = flag:failed }
	}
	remove_variable = kehillah_synod_in_session

    set_variable = { name = kehillah_synod_result_for value = scope:activity.var:kehillah_synod_votes_for }
    set_variable = { name = kehillah_synod_result_against value = scope:activity.var:kehillah_synod_votes_against }
    set_variable = { name = kehillah_synod_result_uncast value = 0 }
    clear_variable_list = kehillah_synod_roll_for
    clear_variable_list = kehillah_synod_roll_against
    clear_variable_list = kehillah_synod_roll_uncast
    scope:activity = {
        every_in_list = {
            variable = kehillah_synod_voted_for
            root = { add_to_variable_list = { name = kehillah_synod_roll_for target = prev } }
        }
        every_in_list = {
            variable = kehillah_synod_voted_against
            root = { add_to_variable_list = { name = kehillah_synod_roll_against target = prev } }
        }
        every_in_list = {
            variable = kehillah_synod_electorate
            limit = { NOT = { scope:activity = { is_target_in_variable_list = { name = kehillah_synod_voted target = prev } } } }
            root = {
                add_to_variable_list = { name = kehillah_synod_roll_uncast target = prev }
                change_variable = { name = kehillah_synod_result_uncast add = 1 }
            }
        }
    }
	debug_log = "KEHILLAH_SYNOD: closing ruling captured"
}
