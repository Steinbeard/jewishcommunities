# The Talmud on trial (docs/spec/2026-11-01-talmud-trial-situation.md), 2026-11-01.
# Replaces the one-event trial (kehillah_incident.0001) with the kehillah_talmud_trial situation.

# Scope: a Kehillah leader (the incident director's Talmud pick, or the console). The baptized former
# student (Donin) denounces the books; the trial begins and its first scene opens.
kehillah_trial_begin_effect = {
 if = {
  limit = { kehillah_trial_can_begin_trigger = yes }
  kehillah_incident_scopes_effect = yes
  primary_title = { save_scope_as = kehillah_trial_new_title }
		create_character = {
			age = { 28 50 }
			gender = male
			faith = scope:kehillah_incident_host.faith
			culture = root.culture
			location = scope:kehillah_incident_town
			dynasty = none
			trait = zealous
			learning = { 12 20 }
			intrigue = { 6 14 }
			save_scope_as = kehillah_trial_new_accuser
		}
		scope:kehillah_incident_host = { add_courtier = scope:kehillah_trial_new_accuser }

  kehillah_crisis_reserve_intro_effect = { FAMILY = trial }
  primary_title = {
   set_variable = { name = kehillah_trial_accuser value = scope:kehillah_trial_new_accuser }
   remove_variable = kehillah_trial_patron_char
   kehillah_trial_roll_effect = yes
  }
  trigger_event = kehillah_talmud_trial.0001
  debug_log = "kehillah_trial_begin_effect: denunciation awaiting acknowledgment"
 }
}

# Scope: leader, only after the denunciation is acknowledged. Copy the already-read details.
kehillah_trial_activate_effect = {
 kehillah_incident_scopes_effect = yes
 primary_title = { save_scope_as = kehillah_trial_new_title }
 start_situation = { type = kehillah_talmud_trial start_phase = kehillah_trial_denunciation save_scope_as = kehillah_trial_new_situation }
 scope:kehillah_trial_new_situation ?= {
  set_variable = { name = kehillah_trial_title value = scope:kehillah_trial_new_title }
  scope:kehillah_trial_new_title = { if = { limit = { has_variable = kehillah_trial_accuser } scope:kehillah_trial_new_situation = { set_variable = { name = kehillah_trial_accuser value = scope:kehillah_trial_new_title.var:kehillah_trial_accuser } } } }
  scope:kehillah_trial_new_title = { if = { limit = { has_variable = kehillah_trial_story } scope:kehillah_trial_new_situation = { set_variable = { name = kehillah_trial_story value = scope:kehillah_trial_new_title.var:kehillah_trial_story } } } }
  scope:kehillah_trial_new_title = { if = { limit = { has_variable = kehillah_trial_patron } scope:kehillah_trial_new_situation = { set_variable = { name = kehillah_trial_patron value = scope:kehillah_trial_new_title.var:kehillah_trial_patron } } } }
  scope:kehillah_trial_new_title = { if = { limit = { has_variable = kehillah_trial_patron_char } scope:kehillah_trial_new_situation = { set_variable = { name = kehillah_trial_patron_char value = scope:kehillah_trial_new_title.var:kehillah_trial_patron_char } } } }
  set_situation_center_province = scope:kehillah_trial_new_title.holder.domicile.domicile_location
  situation_sub_region:kehillah_trial_area = { add_dejure_title_to_sub_region = scope:kehillah_trial_new_title.holder.domicile.domicile_location.county }
  scope:kehillah_trial_new_title = { set_variable = { name = kehillah_trial_situation value = scope:kehillah_trial_new_situation } }
  kehillah_trial_participants_effect = yes
  trigger_situation_catalyst = kehillah_trial_opening_catalyst
 }
 if = {
  limit = { exists = primary_title.var:kehillah_trial_situation }
  kehillah_crisis_clear_intro_effect = yes
  primary_title = { set_variable = { name = kehillah_talmud_cooldown value = yes years = kehillah_talmud_cooldown_years } }
  domicile.domicile_location.county.faith ?= { kehillah_faith_pressure_add_effect = { AMOUNT = 1 } }
  kehillah_realm_news_add_effect = { KIND = talmud_trial }
  scope:kehillah_incident_host = {
   if = { limit = { is_ai = no } set_variable = { name = kehillah_trial_target value = prev years = 1 } trigger_event = { id = kehillah_talmud_trial.0100 days = 2 } }
  }
  debug_log = "kehillah_trial_activate_effect: acknowledged; trial clock starts"
 }
}

# Scope: the situation. The community's leader, its sovereign and the accuser, as they are now.
kehillah_trial_participants_effect = {
	save_scope_as = kehillah_trial_sit
	if = {
		limit = { exists = var:kehillah_trial_leader }
		remove_manual_participant = var:kehillah_trial_leader
		remove_variable = kehillah_trial_leader
	}
	if = {
		limit = { exists = var:kehillah_trial_judge }
		remove_manual_participant = var:kehillah_trial_judge
		remove_variable = kehillah_trial_judge
	}
	if = {
		limit = {
			exists = var:kehillah_trial_accuser
			any_situation_participant = { this = scope:kehillah_trial_sit.var:kehillah_trial_accuser }
		}
		remove_manual_participant = var:kehillah_trial_accuser
	}
	var:kehillah_trial_title.holder ?= {
		scope:kehillah_trial_sit = {
			set_variable = { name = kehillah_trial_leader value = prev }
			add_manual_participant = prev
		}
	}
	var:kehillah_trial_title.holder.domicile.domicile_location.county.holder.top_liege ?= {
		scope:kehillah_trial_sit = {
			set_variable = { name = kehillah_trial_judge value = prev }
			add_manual_participant = prev
		}
	}
	var:kehillah_trial_accuser ?= {
		if = {
			limit = { is_alive = yes }
			scope:kehillah_trial_sit = { add_manual_participant = prev }
		}
	}
	# The accuser's patron, when he is not the judge.
	var:kehillah_trial_patron_char ?= {
		save_scope_as = kehillah_trial_patron_now
		if = {
			limit = {
				is_alive = yes
				NOT = { this = scope:kehillah_trial_sit.var:kehillah_trial_judge }
				NOT = { scope:kehillah_trial_sit = { any_situation_participant = { this = scope:kehillah_trial_patron_now } } }
			}
			scope:kehillah_trial_sit = { add_manual_participant = scope:kehillah_trial_patron_now }
		}
	}
}

# Scope: the situation, as it begins; scope:kehillah_incident_host saved. Rolled once:
#   story   colour: student (a former student), son_in_law (a broken betrothal), rival (a scholar
#           who lost a ruling to the rabbi)
#   patron  terms: friar (the accuser presses harder; outside the hall in the hearing), bishop (the
#           lord's chaplain, named: a monthly push toward Condemnation in the hearing), official (the
#           lord's chancellor, named: silver goes further)
kehillah_trial_roll_effect = {
	random_list = {
		2 = { set_variable = { name = kehillah_trial_story value = flag:student } }
		1 = { set_variable = { name = kehillah_trial_story value = flag:son_in_law } }
		1 = { set_variable = { name = kehillah_trial_story value = flag:rival } }
	}
	random_list = {
		10 = { set_variable = { name = kehillah_trial_patron value = flag:friar } }
		10 = {
			trigger = { exists = scope:kehillah_incident_host.cp:councillor_court_chaplain }
			set_variable = { name = kehillah_trial_patron value = flag:bishop }
			set_variable = { name = kehillah_trial_patron_char value = scope:kehillah_incident_host.cp:councillor_court_chaplain }
		}
		8 = {
			trigger = { exists = scope:kehillah_incident_host.cp:councillor_chancellor }
			set_variable = { name = kehillah_trial_patron value = flag:official }
			set_variable = { name = kehillah_trial_patron_char value = scope:kehillah_incident_host.cp:councillor_chancellor }
		}
	}
	debug_log = "kehillah_trial_roll_effect: story and patron rolled"
}

# Root: the situation, monthly. The accuser's pressure in Denunciation; the judges' lean in the
# hearing. An accuser who has died takes his case with him.
kehillah_trial_monthly_effect = {
	if = {
		limit = { situation_current_phase = kehillah_trial_closed }
		kehillah_trial_close_effect = yes
	}
	else_if = {
		limit = { exists = var:kehillah_trial_title }
		kehillah_trial_participants_effect = yes
		if = {
			limit = {
				OR = {
					situation_current_phase = kehillah_trial_denunciation
					situation_current_phase = kehillah_trial_hearing
				}
				OR = {
					NOT = { exists = var:kehillah_trial_accuser }
					var:kehillah_trial_accuser = { is_alive = no }
				}
			}
			trigger_situation_catalyst = kehillah_trial_accuser_gone_catalyst
		}
		else_if = {
			limit = { situation_current_phase = kehillah_trial_denunciation }
			trigger_situation_catalyst = kehillah_trial_accuser_catalyst
			kehillah_trial_toast_effect = { KEY = accuser }
		}
		else_if = {
			limit = { situation_current_phase = kehillah_trial_hearing }
			if = {
				limit = { kehillah_trial_judges_acquittal_value > 0 }
				trigger_situation_catalyst = kehillah_trial_judges_acquittal_catalyst
			}
			if = {
				limit = { kehillah_trial_judges_expurgation_value > 0 }
				trigger_situation_catalyst = kehillah_trial_judges_expurgation_catalyst
			}
			if = {
				limit = { kehillah_trial_patron_presses_value > 0 }
				trigger_situation_catalyst = kehillah_trial_patron_presses_catalyst
			}
			if = {
				limit = { kehillah_trial_judges_condemnation_value > 0 }
				trigger_situation_catalyst = kehillah_trial_judges_condemnation_catalyst
				kehillah_trial_toast_effect = { KEY = judges_condemnation }
			}
			else_if = {
				limit = { kehillah_trial_judges_acquittal_value > 0 }
				kehillah_trial_toast_effect = { KEY = judges_acquittal }
			}
			else_if = {
				limit = { kehillah_trial_judges_expurgation_value > 0 }
				kehillah_trial_toast_effect = { KEY = judges_expurgation }
			}
		}
	}
}

# Root: the situation, monthly. A toast to a player leader: the month's push and where it went.
# KEY: accuser, judges_acquittal, judges_expurgation, judges_condemnation. Flavor only, no numbers
# (Daniel, 2026-11-01).
kehillah_trial_toast_effect = {
	var:kehillah_trial_title ?= { save_scope_as = kehillah_trial_toast_title }
	var:kehillah_trial_accuser ?= { save_scope_as = kehillah_trial_toast_accuser }
	var:kehillah_trial_judge ?= { save_scope_as = kehillah_trial_toast_judge }
	scope:kehillah_trial_toast_title.holder ?= {
		if = {
			limit = { is_ai = no }
			save_scope_as = kehillah_trial_toast_leader
			kehillah_flavor_toast_effect = { KEY = kehillah_trial_pressure_$KEY$ TYPE = msg_kehillah_crisis_pressure ICON = scope:kehillah_trial_toast_title }
		}
	}
}

# Scope: the community title. A catalyst toward the trial's race, while the trial runs.
kehillah_trial_add_effect = {
	var:kehillah_trial_situation ?= { trigger_situation_catalyst = $CATALYST$ }
}

# Scope: a Kehillah leader. A sub-action is spent for this trial.
kehillah_trial_mark_done_effect = {
	primary_title = { set_variable = { name = kehillah_trial_did_$ONCE$ value = yes } }
}

# Root: the situation, as the seizure begins. The lord's men take the books.
kehillah_trial_seizure_start_effect = {
	debug_log = "kehillah_talmud_trial: the books are seized"
	var:kehillah_trial_title.holder ?= {
		if = {
			limit = { is_ai = no }
			# A day on, so the phase is current when the event checks it.
			trigger_event = { id = kehillah_talmud_trial.0040 days = 1 }
		}
	}
}

# Root: the situation, as the hearing begins. What the seizure prepared counts once; three sessions
# are set, about thirty days apart.
kehillah_trial_hearing_start_effect = {
	debug_log = "kehillah_talmud_trial: the hearing begins"
	var:kehillah_trial_title ?= { save_scope_as = kehillah_trial_hearing_title }
	if = {
		limit = { exists = scope:kehillah_trial_hearing_title }
		if = {
			limit = { scope:kehillah_trial_hearing_title = { has_variable = kehillah_trial_prepared } }
			trigger_situation_catalyst = kehillah_trial_seed_prepared_catalyst
		}
		if = {
			limit = { scope:kehillah_trial_hearing_title = { has_variable = kehillah_trial_letter } }
			trigger_situation_catalyst = kehillah_trial_seed_letter_catalyst
		}
		if = {
			limit = { scope:kehillah_trial_hearing_title = { has_variable = kehillah_trial_books_found } }
			trigger_situation_catalyst = kehillah_trial_seed_found_catalyst
		}
		scope:kehillah_trial_hearing_title = { set_variable = { name = kehillah_trial_session value = 0 } }
		# 2026-11-01: a player judge decides how he will lean (kehillah_talmud_trial.0110).
		var:kehillah_trial_judge ?= {
			if = {
				limit = { is_ai = no }
				set_variable = { name = kehillah_trial_target value = scope:kehillah_trial_hearing_title.holder years = 1 }
				trigger_event = { id = kehillah_talmud_trial.0110 days = 1 }
			}
		}
		scope:kehillah_trial_hearing_title.holder ?= {
			trigger_event = { id = kehillah_talmud_trial.0050 days = 3 }
			trigger_event = { id = kehillah_talmud_trial.0050 days = 33 }
			trigger_event = { id = kehillah_talmud_trial.0050 days = 63 }
		}
	}
}

# Root: the situation, as a verdict phase begins. VERDICT: quashed, acquittal, expurgation,
# condemnation. The verdict falls on the community's leader; a player hears it.
kehillah_trial_verdict_start_effect = {
	debug_log = "kehillah_talmud_trial: a verdict"
	# The judge hears the verdict given in his name.
	var:kehillah_trial_title ?= { save_scope_as = kehillah_ruler_notice_title }
	var:kehillah_trial_judge ?= {
		if = {
			limit = {
				is_ai = no
				NOT = { this = scope:kehillah_ruler_notice_title.holder }
			}
			send_interface_message = {
				type = msg_kehillah_crisis_ruler
				title = kehillah_trial_ruler_$VERDICT$_t
				desc = kehillah_trial_ruler_$VERDICT$_d
				left_icon = scope:kehillah_ruler_notice_title
			}
		}
	}
	var:kehillah_trial_title ?= {
		set_variable = { name = kehillah_trial_verdict value = flag:$VERDICT$ }
		holder ?= {
			kehillah_trial_verdict_$VERDICT$_effect = yes
			if = {
				limit = { is_ai = no }
				trigger_event = kehillah_talmud_trial.0060
			}
		}
	}
}

# Scope: the leader. The verdicts.
kehillah_trial_verdict_quashed_effect = {
	custom_tooltip = kehillah_incident_talmud_safe_tt
	kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_trial_quashed_stability }
}
kehillah_trial_verdict_acquittal_effect = {
	custom_tooltip = kehillah_incident_talmud_safe_tt
	kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_talmud_acquitted_greatness }
}
kehillah_trial_verdict_expurgation_effect = {
	custom_tooltip = kehillah_trial_expurgated_tt
	kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_talmud_expurgated_greatness }
	kehillah_charter_censor_text_effect = yes
}
kehillah_trial_verdict_condemnation_effect = {
	# Hidden or copied volumes: the burning is real, but the library keeps its Talmud.
	if = {
		limit = {
			primary_title = {
				OR = {
					has_variable = kehillah_trial_books_hidden
					has_variable = kehillah_trial_books_copied
				}
			}
		}
		custom_tooltip = kehillah_trial_burned_saved_tt
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_talmud_expurgated_greatness }
		add_stress = minor_stress_gain
		kehillah_charter_censor_text_effect = yes
	}
	else = {
		kehillah_talmud_burned_effect = yes
	}
}

# Root: the situation, as its Closed phase begins. Live test 2026-11-01: end_situation inside the
# situation's own phase on_start crashed the game (crash ck3_20261101_161102, the same second). The
# situation is ended a day later by a hidden event on the leader (kehillah_talmud_trial.0099), the way
# the attack's situation is ended from events. If no leader can take it, the monthly pass retries.
kehillah_trial_close_effect = {
	debug_log = "kehillah_talmud_trial: closing"
	var:kehillah_trial_title.holder ?= { trigger_event = { id = kehillah_talmud_trial.0099 days = 1 } }
}

# Scope: the community title (kehillah_talmud_trial.0099). The trial ends; the next crisis can begin.
kehillah_trial_end_effect = {
	debug_log = "kehillah_talmud_trial: closed"
	kehillah_crisis_post_end_cooldown_effect = yes
	var:kehillah_trial_situation ?= { end_situation = yes }
	save_scope_as = kehillah_trial_ending_title

		remove_variable = kehillah_trial_situation
		remove_variable = kehillah_trial_did_answer
		remove_variable = kehillah_trial_did_plead
		remove_variable = kehillah_trial_did_silver
		remove_variable = kehillah_trial_did_letter
		remove_variable = kehillah_trial_did_prepare
		remove_variable = kehillah_trial_did_hide
		remove_variable = kehillah_trial_did_copy
		remove_variable = kehillah_trial_prepared
		remove_variable = kehillah_trial_letter
		remove_variable = kehillah_trial_books_hidden
		remove_variable = kehillah_trial_books_found
		remove_variable = kehillah_trial_books_copied
		remove_variable = kehillah_trial_disputant

}
