}

  kehillah_crisis_armed_race_add_effect = { CATALYST = kehillah_crisis_efforts_catalyst POINTS = kehillah_crisis_held_silver_points }
  var:kehillah_mob_threat ?= {
   if = {
    limit = { has_variable = kehillah_mob_march_delay var:kehillah_mob_march_delay > 0 }

   }
  }
 }
}
