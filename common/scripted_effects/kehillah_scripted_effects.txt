# =====================================================================
# Kehillah scripted effects.
#
# Source of truth: docs/spec/v1-kehillah-community.md sections 3c and 5.
#
# ---------------------------------------------------------------------
# THE ESTATE CONTINUITY FIX (spec section 5, "Estate continuity fix")
#
# The problem, confirmed by reading vanilla's own
# change_to_administrative_effect and set_up_domicile_estate_effect:
# domiciles are owned PER CHARACTER, not per title. When a Kehillah
# leader dies, the appointed successor would normally receive a fresh,
# empty, culture-seeded quarter -- and three generations of Synagogue
# construction would evaporate. That is obviously wrong: the synagogue
# belongs to the congregation, not to whoever last held the keys.
#
# The spec calls for copying the outgoing leader's buildings onto the
# incoming leader's domicile. This implementation does that indirectly,
# via the TITLE, and the indirection is deliberate:
#
#   - on_title_gain does expose scope:previous_holder, but vanilla's own
#     comment on it reads "Might be dead". A dead character is not a
#     reliable place to read a live domicile from.
#   - So instead the quarter's state is mirrored onto the Kehillah title
#     every time a building completes. The title always knows what the
#     community has built, independent of who is alive.
#
# The title is also the conceptually right home for this. The buildings
# are the community's, and the title IS the community.
#
# Because spec section 5 opens the candidate pool beyond the outgoing
# leader's family, this matters more than it would under bloodline
# succession, not less -- leadership can now pass to someone with no
# family connection to their predecessor at all.
# ---------------------------------------------------------------------
#
# The per-track helpers below exist because CK3 offers no "iterate every
# building in this domicile" list. Vanilla solves the same problem the
# same way, by enumerating building families explicitly (see
# destroy_random_estate_building_effect). Adding a new building family
# means adding it here too -- there is no way around that, so it is
# called out rather than hidden.
# =====================================================================


# ---------------------------------------------------------------------
# Recording helpers. Scope: the domicile. Requires scope:kq_title.
# Checks tiers high to low and stores the reached tier as a number.
# ---------------------------------------------------------------------

kehillah_record_track_2_effect = {
	if = {
		limit = { has_domicile_building = $B2$ }
		scope:kq_title = { set_variable = { name = $VAR$ value = 2 } }
	}
	else_if = {
		limit = { has_domicile_building = $B1$ }
		scope:kq_title = { set_variable = { name = $VAR$ value = 1 } }
	}
}

# ---------------------------------------------------------------------
# Restoring helpers. Scope: the domicile. Requires scope:kq_title.
# Buildings are added in ascending tier order so each upgrade has its
# previous_building already in place.
# ---------------------------------------------------------------------

kehillah_restore_track_2_effect = {
	if = {
		limit = {
			scope:kq_title = { exists = var:$VAR$ }
			NOT = { has_domicile_building_or_higher = $B1$ }
		}
		add_domicile_building = $B1$
	}
	if = {
		limit = {
			scope:kq_title = { var:$VAR$ >= 2 }
			NOT = { has_domicile_building_or_higher = $B2$ }
		}
		add_domicile_building = $B2$
	}
}

# ---------------------------------------------------------------------
# kehillah_record_quarter_effect
# Scope: the Kehillah leader (character).
#
# Writes the current state of the quarter onto the Kehillah title. Safe
# to call at any time; it only ever overwrites with the truth.
# ---------------------------------------------------------------------

kehillah_record_quarter_effect = {
	if = {
		limit = {
			government_has_flag = government_is_kehillah
			exists = domicile
			exists = primary_title
		}

		# The Kehillah title is the leader's only title, so it is always
		# their primary. If later phases give a leader more than one
		# title, this needs to become an explicit search.
		primary_title = { save_scope_as = kq_title }

		# NOTE: the Stability/Greatness pillar variables need no mirror
		# here. They live on primary_title itself (see
		# kehillah_quarterly_pillars_effect's "WHY THE TITLE, NOT THE
		# CHARACTER" note), so they are already exactly where the next
		# holder will find them -- unlike the buildings below, which
		# live on the domicile and do need copying onto the title.

		domicile = {
			kehillah_record_track_5_effect = {
				VAR = kq_synagogue

# ---------------------------------------------------------------------
# kehillah_setup_worms_start_effect
# Scope: the Kehillah leader (character).
#
# Establishes the 1066 starting state described in
# docs/scenarios/worms-1066.md section 4:
#
#   - Synagogue at TIER 1, not empty. The real Worms Synagogue was built
#     in 1034, thirty-two years before this start date, so the community
#     is modestly established rather than founded from nothing.
#   - NO internal buildings. A modest single-tier community. Growth is
#     the player's job, not a pre-built showcase.
#   - NO officers, which follows automatically: all three gate on
#     internal buildings that do not exist yet.
#
# Run once, from the game-start on_action.
# ---------------------------------------------------------------------

kehillah_setup_worms_start_effect = {
	if = {
		limit = {
			exists = domicile
			NOT = { has_character_flag = kehillah_start_configured }
		}
		add_character_flag = kehillah_start_configured

		# Named explicitly rather than trusted to root: on_game_start_after_lobby
		# is a global on_action with no bound character root, and this
		# effect is reached through title:c_kehillah_worms ?= { holder ?= { ... } },
		# which changes the acting scope but not root. root.faith/root.culture
		# happened to still resolve for the first (heir) create_character
		# call below, then failed with "Event target link 'root' returned
		# an unset scope" on the second (notable families) call inside a
		# while loop -- root's binding here is not reliable enough to use
		# at all. scope:kehillah_leader is.
		save_scope_as = kehillah_leader

		domicile = {
			if = {
				limit = { NOT = { has_domicile_building_or_higher = kehillah_synagogue_01 } }
				add_domicile_building = kehillah_synagogue_01
			}
		}

		# A modest communal chest. Enough to start one internal
		# building and absorb an early flavor incident, not enough to
		# buy the whole quarter on day one.
		add_gold = 120

		# Isaac has led the community since 1064 (see
		# history/titles/kehillah_titles.txt), so he does not begin
		# with nothing to his name. Greatness, not Stability -- his
		# standing is scholarly (chief rabbi, yeshiva head, Rashi's own
		# teacher), not administrative.
		# kehillah_init_pillars_effect first -- see that effect's header:
		# change_variable silently no-ops on a variable that has never
		# been set, which is exactly what made this +150 a no-op live.
		primary_title = {
			kehillah_init_pillars_effect = yes
			change_variable = { name = kehillah_var_greatness add = 150 }
		}

		kehillah_seed_community_effect = yes

		# Mirror the starting state onto the title immediately, so
		# continuity works even if he dies before building anything.
		kehillah_record_quarter_effect = yes

		# Isaac reads as the community's rabbi, not a lay elder -- see
		# common/flavorization/kehillah_title_holders.txt.
		kehillah_update_leader_flavor_effect = yes
	}
}


# ---------------------------------------------------------------------
# kehillah_seed_community_effect
# Scope: the Kehillah leader (character).
#
# Populates the community with people. This exists to close a real gap
# in how appointment succession interacts with this government, and the
# reasoning matters more than the code:
#
# THE GAP. The candidate categories CK3 offers for appointment
# succession are all structural -- close family, house members,
# councillors, court position holders, vassals. There is deliberately no
# "any courtier" category. A Kehillah leader has no vassals and no
# council, and at game start holds no buildings, so no officers can be
# appointed either. That leaves close family as the only populated
# category. If the starting leader also had no living close family, the
# candidate pool would be empty when he dies -- and an empty pool on the
# player's own title is a game-over, in a mod whose entire premise is
# that leadership survives the leader.
#
# THE FIX, AND WHY IT IS ALSO THE DESIGN. Two things are seeded:
#
#   1. An adult child, if the leader has none. This guarantees the pool
#      is never empty. A floor, not a feature.
#
#   2. Three unrelated notable community members, each with their own
#      family. These are spec section 3's "multiple prominent families
#      exist alongside the ruling one". They are NOT eligible for
#      succession on day one -- they become eligible only once you have
#      built the Beit Midrash, Countinghouse or Chambers and appointed
#      them to those offices.
#
# That produces a genuine progression rather than a workaround. Early
# on, an undeveloped community really does pass leadership within the
# family, because it has no institutions through which anyone else can
# earn standing. Build the quarter, fill the offices, and the
# succession opens up to the whole community. The open pool spec
# section 5 asks for is something the player constructs, which is a
# better answer than handing it to them at turn one.
# ---------------------------------------------------------------------

kehillah_seed_community_effect = {
	# The floor: never let the pool be empty.
	if = {
		limit = {
			NOT = {
				any_child = { is_adult = yes }
			}
		}
		create_character = {
			father = scope:kehillah_leader
			dynasty = inherit
			age = { 26 38 }
			faith = scope:kehillah_leader.faith
			culture = scope:kehillah_leader.culture
			# scope:kehillah_leader.location is unreliable here: this
			# effect runs the same tick as change_government creates the
			# domicile, and a landless character's own .location does not
			# always resolve by then (vanilla's own
			# create_landless_adventurer_title_effect guards the identical
			# case with exists = location). The domicile's own placement
			# is set synchronously with its creation, so read the
			# location from there instead.
			location = scope:kehillah_leader.domicile.domicile_location
			random_traits = yes
			gender_female_chance = 0
			save_scope_as = kehillah_seeded_heir
		}
		scope:kehillah_seeded_heir = {
			set_employer = scope:kehillah_leader
		}
	}

	# The notable families. Given their own dynasties on purpose: these
	# are other households of the community, not branches of the
	# leader's. Three distinct people, not three rolls of the same
	# template -- each skewed and (education-)traited toward a
	# specific communal role, so the community reads as populated by
	# particular people rather than a generic pool the moment the game
	# starts, per the "tied to different communal roles that feel
	# realistic" design goal.
	#
	# Only one can actually hold a position on day one: the Shtadlan,
	# because it is the only office that isn't building-gated (see
	# common/court_positions/types/kehillah_officers.txt). The other two
	# are deliberately NOT appointed to anything yet -- Worms starts
	# with only a Synagogue (kehillah_setup_worms_start_effect above),
	# so neither the Beit Midrash nor the Countinghouse their skills
	# point toward exists. This is the same "growth is the player's
	# job" principle the floor-heir comment below already documents,
	# not an oversight: build the Beit Midrash and the community's own
	# future Chief Rabbi is already standing there waiting.

	# The scholar. Reads as a plausible future Chief Rabbi or, later, a
	# second learned hand for the Sofer's Workshop.
	create_character = {
		dynasty = generate
		age = { 30 55 }
		faith = scope:kehillah_leader.faith
		culture = scope:kehillah_leader.culture
		location = scope:kehillah_leader.domicile.domicile_location
		random_traits = yes
		gender_female_chance = 0
		trait = education_learning_3
		learning = { 8 15 }
		stewardship = { 3 9 }
		diplomacy = { 3 9 }
		save_scope_as = kehillah_seeded_scholar
	}
	scope:kehillah_seeded_scholar = {
		set_employer = scope:kehillah_leader
	}

	# The merchant. Reads as a plausible future Treasurer once the
	# community has a Countinghouse to keep its books in.
	create_character = {
		dynasty = generate
		age = { 30 55 }
		faith = scope:kehillah_leader.faith
		culture = scope:kehillah_leader.culture
		location = scope:kehillah_leader.domicile.domicile_location
		random_traits = yes
		gender_female_chance = 0
		trait = education_stewardship_3
		stewardship = { 8 15 }
		learning = { 3 9 }
		diplomacy = { 3 9 }
		save_scope_as = kehillah_seeded_merchant
	}
	scope:kehillah_seeded_merchant = {
		set_employer = scope:kehillah_leader
	}

	# The advocate. Appointed Shtadlan immediately below -- the one
	# office this community can fill from day one, being a person and
	# a function rather than a building (spec section 3c).
	create_character = {
		dynasty = generate
		age = { 30 55 }
		faith = scope:kehillah_leader.faith
		culture = scope:kehillah_leader.culture
		location = scope:kehillah_leader.domicile.domicile_location
		random_traits = yes
		gender_female_chance = 0
		trait = education_diplomacy_3
		diplomacy = { 8 15 }
		stewardship = { 3 9 }
		learning = { 3 9 }
		save_scope_as = kehillah_seeded_advocate
	}
	scope:kehillah_seeded_advocate = {
		set_employer = scope:kehillah_leader
	}

	# CORRECTED 2026-09-07. This was assign_court_position_type, which is
	# not an effect the game has -- error.log carried "Unknown effect:
	# assign_court_position_type" (pdx_persistent_reader.cpp:216) on every
	# load, so the Shtadlan was never actually appointed and the community
	# had no office holders at all. That was invisible until
	# holder_court_position was added to kehillah_leadership's candidate
	# pool, at which point it would have silently contributed nobody.
	#
	# The real effect is appoint_court_position, and it runs in the
	# EMPLOYER's scope, not the appointee's -- see vanilla
	# common/scripted_effects/00_court_position_effects.txt and
	# common/character_interactions/00_vassal_interactions.txt:3929.
	scope:kehillah_leader = {
		appoint_court_position = {
			recipient = scope:kehillah_seeded_advocate
			court_position = shtadlan_kehillah_position
		}
	}
}


# =====================================================================
# kehillah_quarterly_pillars_effect
# Scope: the Kehillah leader (character), via quarterly_playable_pulse
# (see common/on_action/kehillah_on_actions.txt). Every contribution is
# credited to primary_title, not root -- see "WHY THE TITLE, NOT THE
# CHARACTER" below.
#
# Accumulates the two tracked pillar scores -- Stability and Greatness
# -- that replaced the vanilla Influence resource on 2026-09-07 (see
# docs/implementation/v1-kehillah-implementation.md for the full
# reasoning). Variables, not a vanilla resource: CK3 has no way to add
# a resource with its own top-bar display without real interface/GUI
# work, and none of the vanilla toggleable "extra resource" flags
# (Merit, Treasury, Herd) fit two independent scores without dragging
# in unrelated mechanics. This is the tracking half; there is no
# display yet beyond what a future GUI pass adds.
#
# WHY THE TITLE, NOT THE CHARACTER. Caught in review before this ever
# shipped: an earlier draft of this effect stored both variables on
# root (the leader) and then had to bolt a manual mirror-on-death
# system onto kehillah_record_quarter_effect / kehillah_restore_
# quarter_effect to carry them across succession, duplicating logic
# that already exists for buildings. That was solving a problem this
# mod had already solved the right way once: c_kehillah_worms IS the
# community (the same principle the buildings' record/restore system is
# built on -- see that section's own header). Storing the variables on
# primary_title instead means they are simply already there for
# whoever holds the title next, with no copying step, no mirror
# variables, and nothing that can desync between the two copies.
#
# PROSPERITY HAS NO VARIABLE. It does not need one: every
# Prosperity-tagged building (Countinghouse, Market Stalls, Craft
# Workshops, the Slaughterhouse) already generates real Gold directly,
# which IS the Prosperity pillar's own resource, complete with its own
# top-bar display -- more visible than a hidden variable would be.
# Nothing about Prosperity ran through the old Influence pool either,
# so this is a continuation of the actual prior behaviour, not a gap.
#
# WHY QUARTERLY, NOT MONTHLY. Vanilla resources like Gold/Piety/Prestige
# accrue every month via an engine-internal tick that character_modifier
# fields hook into directly (domicile_monthly_gold_add and its
# siblings). There is no equivalent monthly on_action for playable
# characters to hang a custom effect from -- the finest-grained one
# that exists is quarterly_playable_pulse. Every rate below is the old
# monthly Influence rate x3.
#
# WHY BUILDING/POSITION CONTRIBUTIONS ARE CHECKED HERE RATHER THAN
# DECLARED ON THE BUILDING. A character_modifier field can only credit
# a fixed vanilla resource (common/domiciles/buildings/_domicile_buildings.info:
# "please use domicile_monthly_gold_add... same for _prestige / _piety
# / _influence" -- note what is NOT on that list). A custom variable
# has no such field, so the only way to accumulate one is to check "do
# I have this building/position/timed modifier" here and add manually.
# Every check below mirrors an amount that used to be a
# domicile_monthly_influence_add line on that exact building tier,
# position, or timed character_modifier -- removed from those files in
# the same pass that added this effect. If a contribution needs
# retuning later, this is now the only place it lives.
#
# POSITION CONTRIBUTIONS ARE FLATTENED, NOT APTITUDE-SCALED. The
# original scaling_employer_modifiers gave terrible/poor/average/good/
# excellent officeholders different Influence rates. Reproducing that
# from script would mean re-deriving each position's own aptitude
# script value here, duplicating logic that already exists once in
# common/court_positions/types/kehillah_officers.txt and
# kehillah_minor_positions.txt. Simplified instead to a single flat
# rate -- the position's own "average" tier value -- awarded whenever
# the position is filled at all. Aptitude still matters for everything
# else the position grants (salary, stat bonuses, Piety/Gold in Chief
# Rabbi's and Gabbai Tzedakah's cases); only the pillar-point rate lost
# its fine-grained scaling. First-pass tuning throughout this effect;
# revisit after a playtest, same as every number in
# kehillah_script_values.txt.
#
# ---------------------------------------------------------------------
# BASELINE CONVERGENCE, 2026-09-07 (v2 spec section 1.2, Wave 2 items
# 3-5) -- REVISES THE PARAGRAPH ABOVE. READ BEFORE RE-ADDING A FLAT
# PER-QUARTER STRUCTURAL BONUS TO THIS EFFECT.
#
# WHAT CHANGED. Every flat, unconditional per-quarter Stability/
# Greatness addition described above and originally listed in this
# effect's body -- the Synagogue's five tiers, the Beit Midrash's three,
# the Sofer's Workshop, the Chief Rabbi being in post (Greatness); the
# Hekdesh's two tiers, the Shtadlan and the Gabbai Tzedakah being in
# post (Stability) -- has been REMOVED from the per-quarter accrual
# below and REPLACED by the BASELINE CONVERGENCE block at the end of
# this effect. Prosperity, which never had a flat accrual pass at all
# (see that pillar's own comment below), gets its first one the same
# way, built from scratch rather than migrated.
#
# WHY THIS IS A REPLACEMENT, NOT AN OVERSIGHT. v2 spec section 1.2:
# each pillar now has a STRUCTURAL BASELINE computed from the same
# durable assets (buildings, officers) every quarter, and the pillar's
# CURRENT value drifts toward that baseline by a fraction of the gap
# (kehillah_convergence_rate_value, common/script_values/
# kehillah_script_values.txt). If the flat per-quarter adds above were
# LEFT IN ALONGSIDE that new baseline, the exact same structural fact --
# "the Hekdesh is built", "the Chief Rabbi is in post" -- would be
# counted twice every quarter: once as a permanent stacking increment
# that never stops, and again as an upward pull on the number the first
# mechanism is also permanently inflating. v1 spec section 3's own
# shared rules explicitly warn against exactly this ("avoid
# double-counting... unless each layer has a deliberately small and
# documented weight") -- there is no such deliberate split here, so the
# correct fix is removal, not down-weighting one side.
#
# WHAT DID NOT CHANGE. Every INSTANTANEOUS, decision- or event-driven
# change_variable call in this effect and elsewhere in this mod is
# UNCHANGED and is NOT part of this replacement: the four has_character_
# modifier checks still further down this effect (settled dispute,
# tzedakah, shaken/emboldened community), the epidemic drain and
# recovery beat below, and every change_variable call inside
# events/kehillah_dispute_events.txt, events/kehillah_protection_
# events.txt and common/decisions/kehillah_decisions.txt. Those
# represent things that HAPPENED, not durable structural investment --
# convergence smoothing their effect out over subsequent quarters is
# v2 spec section 1.2's own stated intent ("cushions single bad
# events"), not a bug to fix by also converting them to baseline terms.
#
# WHERE THE REPLACEMENT LIVES. kehillah_prosperity_baseline_value,
# kehillah_stability_baseline_value and kehillah_greatness_baseline_
# value (all common/script_values/kehillah_script_values.txt) now own
# every structural building/office check this effect used to make
# directly. This effect's job for those checks is done -- see the
# BASELINE CONVERGENCE block at the end of this effect for the three
# lines that actually apply them, and see each of those three script
# values' own header comments for the exact simplifications made versus
# v1/v2 spec's full aspiration (the top-N aggregate is approximated as
# leader-plus-one-officeholder with diminishing weight, not a true
# sorted top-N; Stability's average-notable-family-opinion input is
# dropped entirely, both documented in place, not silently missing).
# ---------------------------------------------------------------------
# =====================================================================

# ---------------------------------------------------------------------
# kehillah_init_pillars_effect
# Scope: a Kehillah title (e.g. scope:kq_title, primary_title). Idempotent
# -- safe to call every tick, from every entry point.
#
# ROOT CAUSE FOUND 2026-09-07, via live-test + a dedicated diagnostic
# pass: change_variable is a silent no-op on a variable that has never
# been set. Vanilla NEVER calls change_variable on a pillar-shaped
# variable without a set_variable = { ... value = 0 } immediately before
# it the first time (confirmed against common/scripted_effects/
# 00_laamp_effects.txt:1230, common/scripted_effects/
# 04_dlc_ep2_tour_effects.txt's traveler_progress_point_gain_effect, and
# two more independent DLC examples with the identical
# if-not-exists-set-else-change shape). This mod never did that for
# kehillah_var_prosperity/_stability/_greatness anywhere -- every one of
# the ~33 writes across the whole mod was change_variable, so the three
# pillar variables were NEVER ACTUALLY CREATED, and every write to them,
# including Isaac's starting +150 Greatness and every quarterly
# convergence tick, has been silently discarded since Wave 1/2 shipped.
# Confirmed decisively live: a parallel set_variable-based flag on the
# same title (kq_synagogue, kehillah_record_quarter_effect) read back
# correctly while the pillars read 0 on the identical character in the
# identical session.
#
# This was NOT a primary_title scope-resolution problem -- primary_title
# as an effect-scope-opener is exactly how vanilla itself writes and
# reads title variables (see e.g. common/decisions/dlc_decisions/
# ep3_decisions.txt's founded_varangian_guard). Read that finding before
# ever suspecting primary_title again for a similar symptom.
# ---------------------------------------------------------------------

kehillah_init_pillars_effect = {
	if = {
		limit = { NOT = { exists = var:kehillah_var_prosperity } }
		set_variable = { name = kehillah_var_prosperity value = 0 }
	}
	if = {
		limit = { NOT = { exists = var:kehillah_var_stability } }
		set_variable = { name = kehillah_var_stability value = 0 }
	}
	if = {
		limit = { NOT = { exists = var:kehillah_var_greatness } }
		set_variable = { name = kehillah_var_greatness value = 0 }
	}
}


kehillah_quarterly_pillars_effect = {
	# Belt-and-braces: guarantees the three variables exist before
	# anything below tries to change_variable on them, and self-repairs
	# any existing save/community that predates this fix.
	primary_title = { kehillah_init_pillars_effect = yes }

	# ---------------------------------------------------------------
	# PROSPERITY. kehillah_var_prosperity was introduced 2026-09-07
	# (v2 spec section 1.1, Wave 1 item 2) by kehillah_epidemic_pulse
	# (common/on_action/kehillah_on_actions.txt), which drains the
	# variable when the community's location has an active epidemic.
	#
	# WAVE 2, 2026-09-07: Prosperity now also gets the full passive
	# accrual v2 spec section 8 called for, via the BASELINE CONVERGENCE
	# block at the end of this effect (see this effect's own header
	# above for the full account). There is no separate flat-add
	# if-block here the way Greatness and Stability had one to remove
	# below -- Prosperity never had one (it read off Gold directly
	# before kehillah_var_prosperity existed at all), so convergence is
	# this pillar's ENTIRE passive-growth story, not a replacement for
	# anything that used to be here.
	# ---------------------------------------------------------------

	# ---------------------------------------------------------------
	# GREATNESS -- structural (building/office) contributions REMOVED
	# 2026-09-07 (Wave 2). See this effect's "BASELINE CONVERGENCE"
	# header above for the full reasoning: the Synagogue, Beit Midrash/
	# Yeshiva, Sofer's Workshop and Chief-Rabbi-in-post flat per-quarter
	# adds that used to live here are now expressed through
	# kehillah_greatness_baseline_value (common/script_values/
	# kehillah_script_values.txt) instead, applied by the BASELINE
	# CONVERGENCE block at the end of this effect. Nothing below this
	# comment computes Greatness any more.
	# ---------------------------------------------------------------

	# ---------------------------------------------------------------
	# STABILITY. Structural (building/office) contributions REMOVED
	# 2026-09-07 (Wave 2) -- see "BASELINE CONVERGENCE" above. The
	# Hekdesh, Shtadlan-in-post and Gabbai-Tzedakah-in-post flat
	# per-quarter adds that used to live here are now expressed through
	# kehillah_stability_baseline_value instead, applied by the BASELINE
	# CONVERGENCE block at the end of this effect.
	#
	# The four blocks below are UNCHANGED and are NOT part of that
	# replacement: they read instantaneous, decision/event-driven
	# consequences (a dispute ruling, tzedakah, a botched or successful
	# protection incident) off time-limited character modifiers, not
	# durable structural investment. See "BASELINE CONVERGENCE" above for
	# why these stay exactly as they are.
	# ---------------------------------------------------------------
	if = {
		limit = { has_character_modifier = kehillah_settled_dispute_modifier }
		primary_title = { change_variable = { name = kehillah_var_stability add = 1.5 } }
	}
	if = {
		limit = { has_character_modifier = kehillah_tzedakah_modifier }
		primary_title = { change_variable = { name = kehillah_var_stability add = 0.75 } }
	}
	if = {
		limit = { has_character_modifier = kehillah_shaken_community_modifier }
		primary_title = { change_variable = { name = kehillah_var_stability subtract = 0.75 } }
	}
	if = {
		limit = { has_character_modifier = kehillah_emboldened_community_modifier }
		primary_title = { change_variable = { name = kehillah_var_stability add = 1.5 } }
	}

	# ---------------------------------------------------------------
	# EPIDEMIC RECOVERY BEAT (v2 spec's "epidemic pulse", Wave 1 item 2).
	#
	# There is no vanilla "epidemic just ended at this location" hook to
	# trigger off of directly -- confirmed absent from
	# common/on_action/ during this task's research (only
	# epidemic_endemic_capital_legitimacy_effect exists nearby, which is
	# a different concept -- ongoing endemic disease, not an end event).
	# So this is checked here, at the existing quarterly tick, instead:
	# if kehillah_epidemic_active_flag is set (kehillah_epidemic_pulse
	# sets it every time the epidemic pulse fires) but the location no
	# longer has an active epidemic at all, the epidemic that flag
	# tracked has cleared since the last check. This can lag the actual
	# clearance by up to ~2 months (one quarterly tick), an accepted
	# tolerance rather than a design goal -- the same kind of belt-and-
	# suspenders timing gap v2 spec section 4.4 accepts for dissolution
	# detection.
	#
	# kehillah_epidemic_no_physician_flag, once set by even one pulse
	# during the epidemic, means "not throughout" -- reward only fires
	# if that flag was never set.
	# ---------------------------------------------------------------
	if = {
		limit = {
			has_character_flag = kehillah_epidemic_active_flag
			NOT = {
				location = {
					any_province_epidemic = { }
				}
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = kehillah_epidemic_no_physician_flag }
			}
			primary_title = { change_variable = { name = kehillah_var_stability add = kehillah_epidemic_recovery_stability_gain } }
			add_character_modifier = {
				modifier = kehillah_epidemic_recovery_modifier
				years = 3
			}
		}
		remove_character_flag = kehillah_epidemic_active_flag
		remove_character_flag = kehillah_epidemic_no_physician_flag
	}

	# ---------------------------------------------------------------
	# BASELINE CONVERGENCE (v2 spec section 1.2, Wave 2 items 3-5). See
	# this effect's top header for the full account of what this
	# replaces and why. current = current + (baseline - current) *
	# convergence_rate, computed here in CHARACTER scope (root = the
	# Kehillah leader) via save_scope_value_as, then applied to the
	# TITLE -- the same "compute in character scope, apply in title
	# scope" split kehillah_epidemic_pulse already uses for its drain
	# values (common/on_action/kehillah_on_actions.txt), and for the
	# same reason: employs_court_position, has_domicile_building_or_
	# higher and the leader's own skills all need to resolve against the
	# CHARACTER, not the title, and a title-scoped block cannot read a
	# character trigger directly.
	#
	# Each delta guards the "current" read with an exists check rather
	# than assuming the variable is already there. kehillah_var_
	# stability and kehillah_var_greatness are normally initialised
	# elsewhere (on_title_gain's +50 Stability, Isaac's starting +150
	# Greatness), but kehillah_var_prosperity has no equivalent
	# unconditional initialisation anywhere -- it is created on demand
	# by kehillah_epidemic_pulse's change_variable call, so a community
	# that has never had an epidemic would otherwise have this effect
	# try to read a variable that does not exist yet on its very first
	# quarterly tick. Treating "does not exist" as "reads as 0" for the
	# purpose of this subtraction is the semantically correct behaviour
	# regardless of the reason it is missing, so the same guard is
	# applied to all three pillars for consistency rather than only the
	# one that strictly needs it.
	# ---------------------------------------------------------------

	save_scope_value_as = {
		name = kehillah_prosperity_convergence_delta
		value = {
			value = kehillah_prosperity_baseline_value
			if = {
				limit = { primary_title = { exists = var:kehillah_var_prosperity } }
				subtract = primary_title.var:kehillah_var_prosperity
			}
			multiply = kehillah_convergence_rate_value
		}
	}
	save_scope_value_as = {
		name = kehillah_stability_convergence_delta
		value = {
			value = kehillah_stability_baseline_value
			if = {
				limit = { primary_title = { exists = var:kehillah_var_stability } }
				subtract = primary_title.var:kehillah_var_stability
			}
			multiply = kehillah_convergence_rate_value
		}
	}
	save_scope_value_as = {
		name = kehillah_greatness_convergence_delta
		value = {
			value = kehillah_greatness_baseline_value
			if = {
				limit = { primary_title = { exists = var:kehillah_var_greatness } }
				subtract = primary_title.var:kehillah_var_greatness
			}
			multiply = kehillah_convergence_rate_value
		}
	}

	primary_title = {
		change_variable = {
			name = kehillah_var_prosperity
			add = scope:kehillah_prosperity_convergence_delta
		}
		change_variable = {
			name = kehillah_var_stability
			add = scope:kehillah_stability_convergence_delta
		}
		change_variable = {
			name = kehillah_var_greatness
			add = scope:kehillah_greatness_convergence_delta
		}
	}

	# ---------------------------------------------------------------
	# POPULATION PRESSURE (Wave 3 item 3, v2 spec section 2's "soft
	# courtier/population cap"). See kehillah_courtier_capacity_value's
	# own header (common/script_values/kehillah_script_values.txt) for
	# why this is pressure rather than an enforced limit -- CK3 has no
	# scriptable per-character courtier cap at all.
	# ---------------------------------------------------------------
	if = {
		limit = { kehillah_courtier_count_value > kehillah_courtier_capacity_value }
		add_character_modifier = {
			modifier = kehillah_overcrowded_quarter_modifier
		}
		primary_title = {
			change_variable = { name = kehillah_var_stability subtract = kehillah_overcrowding_stability_drain }
		}
	}
	else = {
		remove_character_modifier = kehillah_overcrowded_quarter_modifier
	}

	# ---------------------------------------------------------------
	# RETENTION (Wave 3 item 3, v2 spec's "courtier retention/defection
	# chance, inversely scaled to Stability band"). FIRST-PASS
	# APPROXIMATION, per this task's own brief: no vanilla "chance a
	# courtier leaves" hook exists, so this is a plain percentage roll,
	# gated to only the WORST band (Crisis), that sends one random
	# courtier of a DIFFERENT dynasty than the leader's own (the same
	# "protect the succession floor" exclusion kehillah_has_dispute_
	# candidates_trigger already uses) to take service with the
	# community's own local count/baron instead -- the same
	# domicile.domicile_location.county.holder target the loan feature
	# below uses, which reads as "a discontented courtier leaves the
	# community for the host's own court" rather than vanishing.
	# ---------------------------------------------------------------
	if = {
		limit = {
			NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_strained_threshold } }
			exists = domicile.domicile_location.county.holder
		}
		random = {
			chance = kehillah_courtier_defection_chance_value
			random_courtier = {
				limit = {
					is_adult = yes
					is_imprisoned = no
					NOT = { dynasty = root.dynasty }
				}
				# Scope inside random_courtier's body is the picked
				# courtier; root still refers to the leader, so this
				# reaches the leader's own local county holder without
				# any extra scope-saving.
				set_employer = root.domicile.domicile_location.county.holder
			}
		}
	}

	# ---------------------------------------------------------------
	# LOAN TICK (Wave 3 item 4). Full mechanism account:
	# common/script_values/kehillah_script_values.txt's "Light loan
	# contracts" header. Two passes over the SAME variable_list, both
	# read-only with respect to the list itself (neither pass ever
	# calls remove_list_variable while iterating it -- see that
	# header's note on why removal is deliberately confined to
	# kehillah_repay_loan_decision's own plain, non-iterating effect
	# instead): first increment every active debtor's elapsed time,
	# then default any debtor whose term has run out. A defaulted
	# debtor's own loan variables are cleared (so this guard cannot
	# re-fire on them next quarter) but they are left in the list --
	# a harmless, permanently-inert entry, not a leak, since every
	# check here and in kehillah_repay_loan_decision already requires
	# exists = var:kehillah_loan_amount_owed before doing anything.
	# ---------------------------------------------------------------
	primary_title = {
		save_scope_as = kehillah_loan_title
	}
	scope:kehillah_loan_title = {
		every_in_list = {
			variable = kehillah_loan_debtors
			limit = { exists = var:kehillah_loan_amount_owed }
			change_variable = { name = kehillah_loan_years_elapsed add = 0.25 }
		}
		every_in_list = {
			variable = kehillah_loan_debtors
			limit = {
				exists = var:kehillah_loan_amount_owed
				var:kehillah_loan_years_elapsed >= kehillah_loan_term_years
			}
			remove_variable = kehillah_loan_amount_owed
			remove_variable = kehillah_loan_lender_title
			remove_variable = kehillah_loan_years_elapsed
			scope:kehillah_loan_title = {
				change_variable = { name = kehillah_var_prosperity subtract = kehillah_loan_default_prosperity_loss }
			}
		}
	}
}


# ---------------------------------------------------------------------
# kehillah_update_leader_flavor_effect
# Scope: the Kehillah leader (character).
#
# Keeps the character flag that drives common/flavorization/
# kehillah_title_holders.txt's "Rav" vs "Parnas" choice in sync.
# Deliberately re-evaluated only at the moments a leader's public
# identity is actually established -- game start and taking the title
# -- rather than every quarter alongside the pillar tick. This mirrors
# vanilla's own pattern for the same kind of one-time-but-revisitable
# flag: landless adventurers choose their camp's flavor
# (flavourise_camp_purpose_scholars and siblings) via a realm law,
# reapplied at game start and on death, not continuously recomputed.
# ---------------------------------------------------------------------

kehillah_update_leader_flavor_effect = {
	if = {
		limit = { kehillah_leader_is_rabbinic_trigger = yes }
		add_character_flag = kehillah_rabbinic_leader_flag
	}
	else = {
		remove_character_flag = kehillah_rabbinic_leader_flag
	}
}

#

# kehillah_scripted_triggers.txt) guarantees a qualifying pair exists

# in the event file still use the safe-scope style (?=) already used

# =====================================================================
# kehillah_apply_courtier_quality_effect
# Scope: a character who has just joined a Kehillah court. Requires
# scope:new_employer (handed to kehillah_on_join_court by vanilla's own
# on_join_court on_action).
#
# Wave 3 item 3, v2 spec section 4's "courtier quality: better skill/
# trait weights... at higher Greatness bands."
#
# WHY THIS HOOKS HERE RATHER THAN EDITING common/scripted_character_
# templates/kehillah_pool_repopulation_patch.txt, AS THIS TASK'S OWN
# BRIEF SUGGESTED. That file's random_traits_list/skill weighting runs
# inside pool_repopulate_local_flavor, vanilla's WORLD-WIDE low-born
# character-pool generator -- it fires for any Jewish-culture pool
# character anywhere on the map, keyed off geography (root.capital_
# province's geographical_region), with no reliable scope link back to
# any ONE specific Kehillah's title variables. There is no "which
# Kehillah is this character being generated near" question the
# template can even ask. kehillah_on_join_court, by contrast, already
# hands this mod scope:new_employer bound to the EXACT Kehillah a
# character is joining, every single time -- the correct, already-
# proven scope to read a Greatness band from. This is a deliberate
# pivot from the suggested file, not a missed instruction: the
# suggested file was read in full during this task before choosing an
# alternative.
#
# WHY A ONE-TIME SKILL BUMP RATHER THAN A CONTINUOUS MODIFIER. A
# character_modifier boosting a stat lives on the character it targets,
# but this mod controls the LEADER, not each courtier -- there is no
# clean "apply this to everyone in my court" character_modifier field.
# A one-time add-on-arrival is what stays within reach of a plain
# effect: a courtier who joins a Legendary-band community is simply a
# somewhat more accomplished person than one joining a Crisis-band one,
# permanently, not a temporary bonus that would need per-courtier
# tracking to remove correctly if the band later falls.
#
# Cumulative across bands (Legendary gets Healthy's + Flourishing's +
# its own), mirroring the baseline-value / courtier-capacity pattern
# already used elsewhere in this mod for the same reason: consistency
# and confirmed safety.
# =====================================================================

kehillah_apply_courtier_quality_effect = {
	if = {
		limit = {
			scope:new_employer = {
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_healthy_threshold }
			}
		}
		add_learning_skill = kehillah_courtier_quality_healthy_learning_bonus
	}
	if = {
		limit = {
			scope:new_employer = {
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_flourishing_threshold }
			}
		}
		add_learning_skill = kehillah_courtier_quality_flourishing_learning_bonus
	}
	if = {
		limit = {
			scope:new_employer = {
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_legendary_threshold }
			}
		}
		add_learning_skill = kehillah_courtier_quality_legendary_learning_bonus
		add_stewardship_skill = kehillah_courtier_quality_legendary_stewardship_bonus
	}
}


kehillah_dispute_pick_parties_effect = {
	random_courtier = {
		limit = {
			is_adult = yes
			is_imprisoned = no
		}
