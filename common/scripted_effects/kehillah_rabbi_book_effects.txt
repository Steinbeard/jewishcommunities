# =====================================================================
# kehillah_rabbi_write_book_effect
#
# ADDED 2026-09-09 at user request: "completing one of the tracks to
# ungate writing a legendary book on that topic... each tier lets you
# write a book of different quality of that genre." Shared by all three
# kehillah_write_*_book_decision decisions (common/decisions/
# kehillah_rabbi_book_decisions.txt), one per kehillah_rabbi_trait track
# (parshanut/talmudics/halakha, common/traits/kehillah_traits.txt).
#
# Reuses this mod's OWN already-working book-writing pattern verbatim --
# kehillah_compose_commentary_decision (common/decisions/kehillah_
# decisions.txt) already does exactly this shape (create_artifact_book_
# effect, gated on has_ep1_dlc_trigger with a no-DLC prestige-only
# fallback, set_artifact_description + set_artifact_rarity picked by
# tier) for the Greatness pillar's own three top bands. This is the
# same mechanism pointed at a DIFFERENT gate -- kehillah_rabbi_trait's
# per-track XP instead of the community's Greatness pillar -- so a
# personal book about a rabbi's own growing expertise, not a communal
# one about the Kehillah's standing.
#
# THRESHOLDS ARE 30/65/100, NOT 50/100/150 -- ck3-tiger caught "trait xp
# only goes up to 100" (a hard engine cap) before this ever reached a
# live game; see kehillah_rabbi_trait's own tracks block (common/traits/
# kehillah_traits.txt) for the full account. 100 = fully completing the
# track, which reads as a MORE literal match for "completing one of the
# tracks" ungating the legendary/illustrious book than the original
# 150 ever would have.
#
# $TRACK$: parshanut / talmudics / halakha, used both for the has_
# trait_xp check (token substitution, same technique as this mod's Bet
# Din case log keys) and nowhere else -- the description/tooltip text is
# passed in per tier instead of being built from $TRACK$, since "a
# legendary code of law" and "a legendary commentary on scripture" are
# not the same sentence with one word swapped.
#
# Scope: root only -- called directly from each decision's own effect
# block, where root is the character taking the decision. is_valid on
# each decision already guarantees track XP >= 30, so the else branch
# below (30-64) needs no further guard of its own.
# =====================================================================

kehillah_rabbi_write_book_effect = {
	if = {
		limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value >= 100 } }
		add_piety = major_piety_gain
		add_prestige = major_prestige_gain
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_book_effect = {
				OWNER = root
				CREATOR = root
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
			scope:newly_created_artifact ?= {
				set_artifact_description = $DESC_100$
				set_artifact_rarity = illustrious
				set_owner = root
			}
			custom_tooltip = $TT_100$
		}
		else = {
			add_prestige = medium_prestige_gain
			custom_tooltip = $TT_100$
		}
	}
	else_if = {
		limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value >= 65 } }
		add_piety = medium_piety_gain
		add_prestige = medium_prestige_gain
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_book_effect = {
				OWNER = root
				CREATOR = root
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
			scope:newly_created_artifact ?= {
				set_artifact_description = $DESC_65$
				set_artifact_rarity = famed
				set_owner = root
			}
			custom_tooltip = $TT_65$
		}
		else = {
			add_prestige = minor_prestige_gain
			custom_tooltip = $TT_65$
		}
	}
	else = {
		# 30-64: the only remaining possibility, since is_valid on every
		# caller already required >= 30 to show the decision at all.
		add_piety = minor_piety_gain
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_book_effect = {
				OWNER = root
				CREATOR = root
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
			scope:newly_created_artifact ?= {
				set_artifact_description = $DESC_30$
				set_artifact_rarity = masterwork
				set_owner = root
			}
			custom_tooltip = $TT_30$
		}
		else = {
			add_prestige = miniscule_prestige_gain
			custom_tooltip = $TT_30$
		}
	}
}
