# New writing project lifecycle. A project is chosen in events and started
# one day later, so no previewable decision/effect reads variables it has
# only just written (the old responsa tooltip flood showed that failure).

kehillah_writing_clear_draft_effect = {
	remove_variable = kehillah_writing_form
	remove_variable = kehillah_writing_phase
	remove_variable = kehillah_writing_partner
	remove_variable = kehillah_writing_extra_time
	remove_variable = kehillah_writing_success
	remove_variable = kehillah_writing_landmark
	remove_variable = kehillah_book_genre
	remove_variable = kehillah_book_tally
	remove_variable = kehillah_book_named_work
	remove_variable = kehillah_book_comment_on
	remove_variable = kehillah_book_tenet_a
	remove_variable = kehillah_book_tenet_b
	remove_variable = kehillah_book_lang
	remove_variable = kehillah_book_work
	remove_variable = kehillah_writing_ordinary_form
	remove_variable = kehillah_writing_spark_tier
	remove_variable = kehillah_writing_original
}

kehillah_writing_choose_project_effect = {
	set_variable = { name = kehillah_writing_form value = flag:$FORM$ }
	set_variable = { name = kehillah_book_genre value = flag:$GENRE$ }
	set_variable = { name = kehillah_writing_spark_tier value = var:kehillah_spark_$DISC$_tier }
}

# 2026-11-24 sparks: an option's preview -- the tier, that writing uses the
# spark up and, where it matters, the landmark status. The Learning window
# shows the routes to the next tier.
kehillah_spark_option_hint_effect = {
	if = {
		limit = { var:kehillah_spark_$DISC$_tier ?= 1 }
		custom_tooltip = kehillah_spark_tier_1_tt
	}
	else_if = {
		limit = { var:kehillah_spark_$DISC$_tier ?= 2 }
		custom_tooltip = kehillah_spark_tier_2_tt
	}
	else = {
		custom_tooltip = kehillah_spark_tier_3_tt
		kehillah_writing_landmark_hint_effect = { FORM = $FORM$ }
	}
	# 2026-11-24: whether to write now or deepen first.
	if = {
		limit = { kehillah_spark_can_upgrade_$DISC$_trigger = yes }
		custom_tooltip = kehillah_spark_hint_ready
	}
	else_if = {
		limit = { NOT = { var:kehillah_spark_$DISC$_tier ?= 3 } }
		custom_tooltip = kehillah_spark_hint_can_grow
	}
	# 2026-11-24: what this spark could still become.
	if = {
		limit = { NOT = { var:kehillah_spark_$DISC$_tier ?= 3 } }
		if = { limit = { kehillah_spark_form_has_history_$FORM$_trigger = yes } custom_tooltip = kehillah_spark_potential_2 }
		else_if = { limit = { kehillah_spark_ledger_slot_free_trigger = yes } custom_tooltip = kehillah_spark_potential_1 }
		else = { custom_tooltip = kehillah_spark_potential_0 }
	}
	custom_tooltip = kehillah_spark_used_up_tt
}

kehillah_writing_start_effect = {
	if = {
		limit = {
			has_variable = kehillah_writing_form
			piety >= kehillah_rabbi_book_piety_cost
			can_start_scheme = { type = kehillah_writing target_character = root }
		}
		add_piety = { value = kehillah_rabbi_book_piety_cost multiply = -1 }
		# 2026-11-23: starting a project uses up its spark.
		kehillah_spark_consume_current_effect = yes
		set_variable = { name = kehillah_writing_recent value = yes days = 365 }
		set_variable = { name = kehillah_writing_phase value = 0 }
		set_variable = { name = kehillah_book_tally value = kehillah_writing_start_quality_value }
		if = {
			limit = { var:kehillah_book_genre ?= flag:talmudics knows_language = language_aramaic }
			set_variable = { name = kehillah_book_lang value = flag:aramaic }
		}
		else = { set_variable = { name = kehillah_book_lang value = flag:hebrew } }
		if = {
			limit = { var:kehillah_writing_form ?= flag:responsa kehillah_has_good_bet_din_memory_trigger = yes }
			set_variable = { name = kehillah_book_bet_din_inspiration value = yes }
		}
		if = {
			limit = { var:kehillah_writing_form ?= flag:torah }
			set_variable = { name = kehillah_book_comment_on value = flag:work_tanakh_torah }
		}
		else_if = {
			limit = { var:kehillah_writing_form ?= flag:midrash }
			set_variable = { name = kehillah_book_comment_on value = flag:work_genesis_rabbah }
		}
		else_if = {
			limit = { var:kehillah_writing_form ?= flag:talmud }
			if = { limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_bavli_nezikin } } set_variable = { name = kehillah_book_comment_on value = flag:work_bavli_nezikin } }
			else_if = { limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_bavli_nashim } } set_variable = { name = kehillah_book_comment_on value = flag:work_bavli_nashim } }
			else = { set_variable = { name = kehillah_book_comment_on value = flag:work_bavli_moed } }
		}
		else_if = {
			limit = { var:kehillah_writing_form ?= flag:yerushalmi }
			if = { limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_yerushalmi_nezikin } } set_variable = { name = kehillah_book_comment_on value = flag:work_yerushalmi_nezikin } }
			else_if = { limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_yerushalmi_zeraim } } set_variable = { name = kehillah_book_comment_on value = flag:work_yerushalmi_zeraim } }
			else_if = { limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_yerushalmi_nashim } } set_variable = { name = kehillah_book_comment_on value = flag:work_yerushalmi_nashim } }
			else = { set_variable = { name = kehillah_book_comment_on value = flag:work_yerushalmi_moed } }
		}
		start_scheme = { type = kehillah_writing target_character = root }
		trigger_event = { id = kehillah_writing.0011 days = 1 }
	}
	else = {
		kehillah_writing_clear_draft_effect = yes
	}
}

# A historical niche is an outcome, never a title selected at the start.
# The existing ambition trigger supplies the source and identity gates;
# composition quality and a separate roll supply the demanding finish.
# Landmark selection is generated from the dated catalog and project map.

kehillah_writing_finish_effect = {
	if = {
		limit = { has_variable = kehillah_writing_success }
		set_variable = { name = kehillah_writing_recent value = yes days = 3650 }
		# 2026-11-24: a historical author who qualifies writes his own work.
		if = {
			limit = { var:kehillah_writing_spark_tier ?= 3 kehillah_spark_is_historical_author_trigger = yes }
			kehillah_spark_author_claim_effect = yes
		}
		if = {
			limit = {
				NOT = { has_variable = kehillah_book_named_work }
				var:kehillah_writing_spark_tier ?= 3
				var:kehillah_book_tally >= kehillah_writing_landmark_quality_value
			}
			random = {
				chance = kehillah_writing_landmark_chance_value
				kehillah_writing_select_landmark_effect = yes
			}
		}
		remove_variable = kehillah_sefer_my_latest_work
		if = {
			limit = { has_variable = kehillah_book_named_work }
			if = { limit = { var:kehillah_book_tally >= 55 } set_variable = { name = kehillah_writing_artifact_tier value = flag:illustrious } }
			else_if = { limit = { var:kehillah_book_tally >= 30 } set_variable = { name = kehillah_writing_artifact_tier value = flag:famed } }
			else = { set_variable = { name = kehillah_writing_artifact_tier value = flag:masterwork } }
			kehillah_sefer_finish_composition_effect = yes
			kehillah_writing_create_manuscript_effect = yes
		}
		# 2026-11-24: a Great Work that fills no historical niche becomes an
		# original work in one of the ledger's authored slots.
		else_if = {
			limit = {
				var:kehillah_writing_spark_tier ?= 3
				var:kehillah_book_tally >= kehillah_writing_landmark_quality_value
				kehillah_spark_ledger_slot_free_trigger = yes
			}
			set_variable = { name = kehillah_writing_original value = yes }
			set_variable = { name = kehillah_writing_artifact_tier value = flag:illustrious }
			kehillah_sefer_finish_composition_effect = yes
			kehillah_writing_create_manuscript_effect = yes
		}
		else = {
			set_variable = { name = kehillah_writing_ordinary_form value = var:kehillah_writing_form }
			kehillah_book_complete_effect = yes
			scope:newly_created_artifact ?= {
				kehillah_writing_set_ordinary_artifact_identity_effect = yes
			}
		}
		trigger_event = { id = kehillah_book.0006 days = 10 }
	}
	else = {
		set_variable = { name = kehillah_writing_recent value = yes days = 730 }
		add_learning_lifestyle_xp = 150
		remove_variable = kehillah_book_tally
	}
	remove_variable = kehillah_writing_form
	remove_variable = kehillah_writing_phase
	remove_variable = kehillah_writing_partner
	remove_variable = kehillah_writing_extra_time
	remove_variable = kehillah_book_comment_on
	remove_variable = kehillah_book_lang
	remove_variable = kehillah_writing_spark_tier
}

kehillah_writing_create_manuscript_effect = {
	if = {
		limit = { has_ep1_dlc_trigger = yes }
		save_scope_as = inspiration_sponsor
		create_artifact_book_effect = { OWNER = root CREATOR = root SET_SUBJECT = flag:no SET_TOPIC = flag:no }
		scope:newly_created_artifact ?= {
			set_variable = { name = kehillah_artifact_work value = root.var:kehillah_sefer_my_latest_work }
			kehillah_writing_register_manuscript_effect = yes
			kehillah_writing_set_landmark_artifact_identity_effect = yes
			if = {
				limit = { root = { has_variable = kehillah_writing_landmark } }
				set_artifact_rarity = illustrious
				add_artifact_modifier = kehillah_writing_landmark_manuscript_modifier
			}
			else_if = {
				limit = { root = { var:kehillah_writing_artifact_tier ?= flag:illustrious } }
				set_artifact_rarity = illustrious
			}
			else_if = {
				limit = { root = { var:kehillah_writing_artifact_tier ?= flag:famed } }
				set_artifact_rarity = famed
			}
			else = {
				set_artifact_rarity = masterwork
			}
		}
	}
}
