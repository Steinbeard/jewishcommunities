# =====================================================================
# What a reading gave (2026-11-24, Daniel: "list the other rewards your
# received, maybe as a tooltip on the buttons (e.g. rabbinics xp if you're a
# rabbi and piety if you got piety)").
#
# kehillah_study_specific_work_effect (kehillah_library_effects.txt) records
# each reward as it gives it; kehillah_study_torah.0001's options show them
# again as ordinary effect lines (show_as_tooltip), so the player reads
# them where vanilla puts outcomes, not in the prose.
# =====================================================================

# Scope: the reader, as a reading resolves.
kehillah_study_reward_reset_effect = {
	remove_variable = kehillah_study_reward_xp
	remove_variable = kehillah_study_reward_lifestyle
	remove_variable = kehillah_study_reward_fulfillment
	remove_variable = kehillah_study_reward_piety
	clear_variable_list = kehillah_study_reward_tenets
	set_variable = { name = kehillah_study_reward_track value = flag:$TRACK$ }
	set_variable = { name = kehillah_study_reward_piety_before value = piety }
}

# Scope: the reader. Adds $XP$ to the experience this reading gave.
kehillah_study_reward_xp_effect = {
	if = {
		limit = { has_variable = kehillah_study_reward_xp }
		change_variable = { name = kehillah_study_reward_xp add = $XP$ }
	}
	else = {
		set_variable = { name = kehillah_study_reward_xp value = $XP$ }
	}
}

# Scope: the player, in an option of kehillah_study_torah.0001. Nothing here
# is given again: show_as_tooltip only shows it.
kehillah_study_torah_rewards_tt_effect = {
	if = {
		limit = { has_variable = kehillah_study_reward_track NOT = { has_variable = kehillah_study_choose_later } }
		custom_tooltip = kehillah_study_rewards_already_received_tt
	}
	show_as_tooltip = {
		if = {
			limit = { has_variable = kehillah_study_reward_xp var:kehillah_study_reward_xp > 0 }
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				if = {
					limit = { var:kehillah_study_reward_track ?= flag:parshanut }
					add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = var:kehillah_study_reward_xp }
				}
				else_if = {
					limit = { var:kehillah_study_reward_track ?= flag:talmudics }
					add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = var:kehillah_study_reward_xp }
				}
				else = {
					add_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value = var:kehillah_study_reward_xp }
				}
			}
			else_if = {
				limit = {
					kehillah_is_karaite_character_trigger = yes
					has_trait = lifestyle_scholar
				}
				add_trait_xp = { trait = lifestyle_scholar value = var:kehillah_study_reward_xp }
			}
		}
		if = {
			limit = { has_variable = kehillah_study_reward_lifestyle var:kehillah_study_reward_lifestyle > 0 }
			add_learning_lifestyle_xp = var:kehillah_study_reward_lifestyle
		}
		if = {
			limit = { has_variable = kehillah_study_reward_piety var:kehillah_study_reward_piety > 0 }
			add_piety = var:kehillah_study_reward_piety
		}
		if = {
			limit = { var:kehillah_study_reward_fulfillment ?= 1 }
			change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		}
		else_if = {
			limit = { var:kehillah_study_reward_fulfillment ?= 2 }
			change_spiritual_fulfillment = miniscule_spiritual_fulfillment_value
		}
	}
	# An apprentice's experience is banked toward his ordination.
	if = {
		limit = {
			has_variable = kehillah_study_reward_xp var:kehillah_study_reward_xp > 0
			NOT = { has_trait = kehillah_rabbi_trait }
			has_variable = kehillah_yeshiva_enrolled
		}
		custom_tooltip = kehillah_study_reward_banked_tt
	}
	# A teaching this reading made known to him (kehillah_sefer_teach_tenet_effect).
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_scholarly_study } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_scholarly_study }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_literalism } } custom_tooltip = kehillah_study_reward_tenet_tenet_literalism }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_hokhmah } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_hokhmah }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_unrelenting_faith } } custom_tooltip = kehillah_study_reward_tenet_tenet_unrelenting_faith }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_religious_legal_pronouncements } } custom_tooltip = kehillah_study_reward_tenet_tenet_religious_legal_pronouncements }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_communal_identity } } custom_tooltip = kehillah_study_reward_tenet_tenet_communal_identity }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_avelei_tziyon } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_avelei_tziyon }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_esotericism } } custom_tooltip = kehillah_study_reward_tenet_tenet_esotericism }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_asceticism } } custom_tooltip = kehillah_study_reward_tenet_tenet_asceticism }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_kiddush_hashem } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_kiddush_hashem }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_sacred_liturgy } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_sacred_liturgy }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_covenant_community } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_covenant_community }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_messianic_hope } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_messianic_hope }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_prophetic_revelation } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_prophetic_revelation }
	if = { limit = { is_target_in_variable_list = { name = kehillah_study_reward_tenets target = flag:tenet_kehillah_temple_service } } custom_tooltip = kehillah_study_reward_tenet_tenet_kehillah_temple_service }
	if = {
		limit = { has_variable = kehillah_spark_just_gained }
		custom_tooltip = kehillah_study_reward_spark_tt
	}
}
