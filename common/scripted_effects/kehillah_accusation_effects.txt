# Scope: a Kehillah leader, quarterly (kehillah_on_actions.txt).
kehillah_accusation_check_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = primary_title
			exists = domicile.domicile_location.county.holder.top_liege
			NOT = { primary_title = { has_variable = kehillah_accusation_cooldown } }
			domicile.domicile_location.county.holder.top_liege = {
				NOT = { is_jewish_character_trigger = yes }
				OR = {
					faith.religion = religion:christianity_religion
					faith.religion = religion:islam_religion
				}
			}
			kehillah_accusation_chance_value > 0
		}
		random = {
			chance = kehillah_accusation_chance_value
			primary_title = {
				set_variable = { name = kehillah_accusation_cooldown value = yes years = kehillah_accusation_cooldown_years }
			}
			kehillah_accusation_begin_effect = yes
		}
	}
}

# Scope: a Kehillah leader. 2026-11-01 (Daniel: "folding the accusations into the crisis as the
# opening event"): the accusation opens the community's crisis (opening: accusation) and
# kehillah_accusation.0001 is its first scene. Its answers push the race (kehillah_crisis_acc_*
# catalysts) instead of rolling the violence on the spot. With a crisis already live, the charge
# is its first scene no longer: it feeds the running race through the same answers.
kehillah_accusation_begin_effect = {
 if = { limit = { primary_title = { NOT = { has_variable = kehillah_crisis_intro_pending } } }
	if = {
		limit = { primary_title = { NOT = { has_variable = kehillah_mob_threat } } }
		primary_title = { set_variable = { name = kehillah_crisis_pending_opening value = flag:accusation } }
		kehillah_mob_local_warning_effect = yes
		# No town to agitate in: no crisis, and the opening is not left behind for another.
		primary_title = {
			if = {
				limit = { NOT = { has_variable = kehillah_mob_threat } has_variable = kehillah_crisis_pending_opening }
				remove_variable = kehillah_crisis_pending_opening
			}
		}
	}
	if = {
		limit = { primary_title = { has_variable = kehillah_mob_threat NOT = { has_variable = kehillah_crisis_intro_pending } } }
		trigger_event = kehillah_accusation.0001
	}
	debug_log = "kehillah_accusation_begin_effect: an accusation opens the crisis"

 }
}

# Scope: a Kehillah leader. The accusation is defused: fear, not blood.

# Scope: a Kehillah leader. The mob comes. V39 slice 5 (2026-10-14): the
# crowd becomes a real mob in the town itself (kehillah_mob_raise_effect,
# 1000 men, base severity 45), so the accusation ends in the same siege of
# the quarter and pogrom as any other host; its harm, martyrs, hostages
# and the faith's incident are the pogrom's. $MULT$ (1, or 1.25 when the
# community was caught questioning witnesses) shifts the severity:
# (MULT - 1) x 40. Barred gates skip this effect: always a riot (.0003).
kehillah_accusation_violence_effect = {
	primary_title ?= {
		if = {
			limit = { NOT = { has_variable = kehillah_mob_severity_adjust } }
			set_variable = { name = kehillah_mob_severity_adjust value = 0 }
		}
		change_variable = {
			name = kehillah_mob_severity_adjust
			add = {
				value = $MULT$
				add = -1
				multiply = 40
			}
		}
	}
	custom_tooltip = kehillah_accusation_violence_tt
	set_variable = { name = kehillah_accusation_mult value = $MULT$ days = 30 }
	# A day later, in a hidden event (raising a mob inside the option made
	# the tooltip builder evaluate the raise -- live 2026-10-14): a mob
	# siege, or a riot in the street (soak 2026-10-14).
	trigger_event = { id = kehillah_accusation.0003 days = 1 }
}

# Scope: a Kehillah leader. The riot in the street (V37's blow): Stability
# and Prosperity fall, scaled by the lord's protection and the barricade;
# perhaps a martyr; +2 incidents in the county's faith.
kehillah_accusation_riot_effect = {
	save_scope_value_as = {
		name = kehillah_accusation_mult
		value = {
			value = kehillah_host_protection_mult_value
			if = {
				limit = { has_variable = kehillah_accusation_mult }
				multiply = var:kehillah_accusation_mult
			}
		}
	}
	set_variable = {
		name = kehillah_accusation_last_stability
		value = {
			value = kehillah_accusation_violence_stability
			multiply = scope:kehillah_accusation_mult
		}
	}
	set_variable = {
		name = kehillah_accusation_last_prosperity
		value = {
			value = kehillah_accusation_violence_prosperity
			multiply = scope:kehillah_accusation_mult
		}
	}
	primary_title = {
		change_variable = { name = kehillah_var_stability add = root.var:kehillah_accusation_last_stability }
		change_variable = { name = kehillah_var_prosperity add = root.var:kehillah_accusation_last_prosperity }
		clamp_variable = { name = kehillah_var_stability min = 0 max = 100000 }
		clamp_variable = { name = kehillah_var_prosperity min = 0 max = 100000 }
	}
	random = {
		chance = 30
		random_courtier = {
			limit = {
				is_adult = yes
				is_jewish_character_trigger = yes
				NOT = { root.player_heir ?= this }
				NOT = { is_spouse_of = root }
			}
			save_scope_as = kehillah_accusation_martyr
			death = { death_reason = death_murder }
		}
	}
	kehillah_notify_region_players_effect = {
		TYPE = msg_kehillah_community_news_bad
		TITLE = kehillah_news_accusation_t
		DESC = kehillah_news_accusation_d
	}
	domicile.domicile_location.county.faith ?= { kehillah_faith_pressure_add_effect = { AMOUNT = 2 } }
	kehillah_realm_news_add_effect = { KIND = accusation_riot }
	debug_log = "kehillah_accusation: a riot"
}

# Scope: a Kehillah leader, in kehillah_accusation.0001's immediate; scope:kehillah_accusation_town and
# scope:kehillah_accusing_host saved, var:kehillah_accusation_tale set. 2026-11-01 (docs/spec/2026-11-01-
# crisis-variety-and-consistency.md, step 4): the accusation's details, rolled once and stored on the
# community title for every later scene:
#   kehillah_acc_place      colour: where it happened, by the charge
#   kehillah_acc_witness    a named, generated townsperson who swears to it; a participant of the
#                           crisis, and the one questioned in The Witness Wavers. Their trade (colour)
#                           fits the charge; their cunning comes with the trade (terms, through the duel).
#   kehillah_acc_accused    the member of the community the witness names (a courtier, else the leader)
#   kehillah_acc_holy_week  terms: it happened in Holy Week; the crisis opens with a push toward the host
kehillah_accusation_roll_effect = {
	primary_title = { save_scope_as = kehillah_acc_title }
	scope:kehillah_accusation_town = {
		faith = { save_scope_as = kehillah_acc_faith }
		culture = { save_scope_as = kehillah_acc_culture }
	}
	# The one the witness names.
	random_courtier = {
		limit = {
			is_adult = yes
			is_jewish_character_trigger = yes
			NOT = { is_spouse_of = root }
		}
		save_scope_as = kehillah_acc_accused_new
	}
	if = {
		limit = { NOT = { exists = scope:kehillah_acc_accused_new } }
		save_scope_as = kehillah_acc_accused_new
	}
	# The place and the witness, by the charge.
	if = {
		limit = { var:kehillah_accusation_tale = flag:blood_libel }
		random_list = {
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:walls } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:river } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:well } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:woods } } }
		}
		random_list = {
			1 = {
			set_variable = { name = kehillah_acc_trade value = flag:carter }
			create_character = {
				age = { 25 55 }
				gender = male
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 4 10 }
				save_scope_as = kehillah_acc_witness_new
			}
			}
			1 = {
			set_variable = { name = kehillah_acc_trade value = flag:washerwoman }
			create_character = {
				age = { 25 55 }
				gender = female
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 4 10 }
				save_scope_as = kehillah_acc_witness_new
			}
			}
			1 = {
			set_variable = { name = kehillah_acc_trade value = flag:watchman }
			create_character = {
				age = { 25 50 }
				gender = male
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 8 14 }
				martial = { 8 14 }
				save_scope_as = kehillah_acc_witness_new
			}
			}
			1 = {
			set_variable = { name = kehillah_acc_trade value = flag:boy }
			create_character = {
				age = { 11 14 }
				gender = male
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 2 6 }
				save_scope_as = kehillah_acc_witness_new
			}
			}
		}
	}
	else_if = {
		limit = { var:kehillah_accusation_tale = flag:mockery }
		scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:church } }
			set_variable = { name = kehillah_acc_trade value = flag:deacon }
			create_character = {
				age = { 30 60 }
				gender = male
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 6 12 }
				learning = { 8 14 }
				save_scope_as = kehillah_acc_witness_new
			}
	}
	else_if = {
		limit = { var:kehillah_accusation_tale = flag:desecration }
		random_list = {
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:altar } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:chapel } } }
		}
			set_variable = { name = kehillah_acc_trade value = flag:maidservant }
			create_character = {
				age = { 16 40 }
				gender = female
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 3 8 }
				save_scope_as = kehillah_acc_witness_new
			}
	}
	else = {
		random_list = {
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:market } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:bathhouse } } }
			1 = { scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_place value = flag:well } } }
		}
			set_variable = { name = kehillah_acc_trade value = flag:porter }
			create_character = {
				age = { 20 50 }
				gender = male
				faith = scope:kehillah_acc_faith
				culture = scope:kehillah_acc_culture
				location = scope:kehillah_accusation_town
				dynasty = none
				intrigue = { 4 10 }
				save_scope_as = kehillah_acc_witness_new
			}
	}
	# The trade was set on the leader by the witness block; it belongs on the title.
	scope:kehillah_acc_title = { set_variable = { name = kehillah_acc_trade value = root.var:kehillah_acc_trade } }
	remove_variable = kehillah_acc_trade
	# The witness lives at the local lord's court, in the town.
	scope:kehillah_accusation_town.county.holder ?= { add_courtier = scope:kehillah_acc_witness_new }
	scope:kehillah_acc_title = {
		set_variable = { name = kehillah_acc_witness value = scope:kehillah_acc_witness_new }
		set_variable = { name = kehillah_acc_accused value = scope:kehillah_acc_accused_new }
		if = {
			limit = { has_variable = kehillah_acc_holy_week }
			remove_variable = kehillah_acc_holy_week
		}
		if = {
			limit = {
				kehillah_crisis_holy_week_season_trigger = yes
				NOT = { root.var:kehillah_accusation_tale = flag:blasphemy }
			}
			set_variable = { name = kehillah_acc_holy_week value = yes }
		}
	}
	debug_log = "kehillah_accusation_roll_effect: place, witness and accused rolled"
}
