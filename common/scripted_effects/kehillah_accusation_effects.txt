# Scope: a Kehillah leader, quarterly (kehillah_on_actions.txt).
kehillah_accusation_check_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = primary_title
			exists = domicile.domicile_location.county.holder.top_liege
			NOT = { primary_title = { has_variable = kehillah_accusation_cooldown } }
			domicile.domicile_location.county.holder.top_liege = {
				NOT = { is_jewish_character_trigger = yes }
				OR = {
					faith.religion = religion:christianity_religion
					faith.religion = religion:islam_religion
				}
			}
			kehillah_accusation_chance_value > 0
		}
		random = {
			chance = kehillah_accusation_chance_value
			primary_title = {
				set_variable = { name = kehillah_accusation_cooldown value = yes years = kehillah_accusation_cooldown_years }
			}
			trigger_event = kehillah_accusation.0001
		}
	}
}

# Scope: a Kehillah leader. The accusation is defused: fear, not blood.
kehillah_accusation_defused_effect = {
	primary_title = {
		change_variable = { name = kehillah_var_stability add = kehillah_accusation_defused_stability }
		clamp_variable = { name = kehillah_var_stability min = 0 max = 100000 }
	}
	custom_tooltip = kehillah_accusation_defused_tt
	debug_log = "kehillah_accusation: defused"
}

# Scope: a Kehillah leader. The mob comes. V39 slice 5 (2026-10-14): the
# crowd becomes a real mob in the town itself (kehillah_mob_raise_effect,
# 1000 men, base severity 45), so the accusation ends in the same siege of
# the quarter and pogrom as any other host; its harm, martyrs, hostages
# and the faith's incident are the pogrom's. $MULT$ (1, or the barricade's
# 0.6) shifts the severity: (MULT - 1) x 40, so the barricade is -16.
kehillah_accusation_violence_effect = {
	primary_title ?= {
		if = {
			limit = { NOT = { has_variable = kehillah_mob_severity_adjust } }
			set_variable = { name = kehillah_mob_severity_adjust value = 0 }
		}
		change_variable = {
			name = kehillah_mob_severity_adjust
			add = {
				value = $MULT$
				add = -1
				multiply = 40
			}
		}
	}
	custom_tooltip = kehillah_accusation_violence_tt
	# The mob rises a day later, in a hidden event: raising it inside the
	# option made the tooltip builder evaluate the raise (live 2026-10-14:
	# 18,000 "no character" lines while the options were hovered).
	trigger_event = { id = kehillah_mob.0030 days = 1 }
}
