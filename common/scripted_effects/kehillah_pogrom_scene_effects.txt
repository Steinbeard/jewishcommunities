# When the siege ends (kehillah_mob_tick_effect). The host waits while
# the scenes play (kehillah_mob_striking), at most 60 days.
# Report the choice's immediate result before the next distinct scene arrives.
# Final losses are determined later by the attack report.
kehillah_pogrom_scene_result_toast_effect = {
	send_interface_toast = {
		type = $TYPE$
		title = kehillah_toast_$TYPE$
		desc = kehillah_pogrom_result_$KEY$_d
		left_icon = root
	}
}

kehillah_pogrom_scenes_start_effect = {
	scope:kehillah_mob_leader = {
		add_character_flag = kehillah_mob_striking
		set_variable = { name = kehillah_mob_wait value = 0 }
	}
	set_variable = { name = kehillah_mob_ref value = scope:kehillah_mob_leader }
	# A Christian neighbour: a firewood seller, a carter, a widow with a
	# loom -- whoever lives next to the quarter. His character decides
	# whether he hides the children or gives them up.
	domicile.domicile_location ?= {
		save_scope_as = kehillah_pogrom_town
		create_character = {
			location = scope:kehillah_pogrom_town
			faith = scope:kehillah_pogrom_town.county.faith
			culture = scope:kehillah_pogrom_town.county.culture
			age = { 25 60 }
			gender_female_chance = 30
			dynasty = none
			save_scope_as = kehillah_pogrom_neighbor
		}
	}
	scope:kehillah_pogrom_neighbor ?= {
		root = { set_variable = { name = kehillah_pogrom_neighbor_ref value = scope:kehillah_pogrom_neighbor } }
	}
 primary_title = { set_variable = { name = kehillah_crisis_scene_count value = 0 } clear_variable_list = kehillah_crisis_scenes_seen }
 kehillah_crisis_next_attack_scene_effect = yes
	debug_log = "kehillah_pogrom_scenes_start_effect: the pogrom begins (scenes)"
}

# The leader baptized by force (after the report, kehillah_mob.0010).
# Scope: the leader; scope:kehillah_mob_leader. He takes the preacher's
# faith; kehillah_conversion.0001 then makes him step down to the
# appointment heir, and a player chooses whom to play on as
# (kehillah_conversion.0002, set_player_character).
kehillah_pogrom_leader_baptized_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = scope:kehillah_mob_leader
		}
		kehillah_anus_baptize_effect = yes
		primary_title ?= { save_scope_as = kehillah_baptized_leader_title }
		if = {
			limit = { exists = scope:kehillah_baptized_leader_title }
			set_variable = { name = kehillah_anus value = scope:kehillah_baptized_leader_title }
		}
		add_character_flag = kehillah_baptized_by_force
		trigger_event = { id = kehillah_conversion.0001 days = 1 }
		debug_log = "kehillah_pogrom_leader_baptized_effect: the leader was baptized by force"
	}
}

# V41 step 1b: a Jew baptized by force. Daniel, 2026-10-14: Christianity
# has a crypto-Jewish rite -- vanilla's Conversos (faith:conversos, a
# Western Christian faith with Jewish Syncretism: "originated when Jewish
# communities were forcibly converted"). Under a Christian mob the convert
# becomes a Converso; under a Muslim one, the mob's faith. Then the crypto
# roll (kehillah_anus_crypto_effect). Scope: the convert;
# scope:kehillah_mob_leader the preacher.
kehillah_anus_baptize_effect = {
	faith = { save_scope_as = kehillah_anus_old_faith }
	rite ?= { save_scope_as = kehillah_anus_old_rite }
	if = {
		limit = { scope:kehillah_mob_leader.faith.religion = religion:christianity_religion }
		set_character_faith = faith:conversos
	}
	else = {
		set_character_faith = scope:kehillah_mob_leader.faith
	}
	kehillah_anus_crypto_effect = yes
}

# V41 step 1b (Daniel, 2026-10-14: "Anusim already exists in game"):
# vanilla's Anusim is the Jewish name of tenet_false_conversion_sanction,
# and its converts keep their old faith as a crypto-religionist secret.
# A Jew baptized by force does the same: publicly the preacher's faith,
# secretly still a Jew -- always if his rite holds the Anusim tenet or he
# is zealous, about 70% otherwise, less if cynical. Scope: the convert,
# just baptized; scope:kehillah_anus_old_faith (and _old_rite) saved
# before the baptism.
kehillah_anus_crypto_effect = {
	if = {
		limit = { exists = scope:kehillah_anus_old_faith }
		random = {
			chance = kehillah_anus_crypto_chance_value
			make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:kehillah_anus_old_faith }
		}
	}
}

# One to four attack encounters by intrinsic threat grade; distinct families within an attack, with the
# previous attack's opening excluded from the next opening when possible.

 if = {
  limit = { primary_title = { has_variable = kehillah_crisis_scene_count var:kehillah_crisis_scene_count >= kehillah_crisis_harm_scene_cap_value } }
  trigger_event = kehillah_pogrom.0090
 }
 else = {
  primary_title = {
   if = { limit = { NOT = { has_variable = kehillah_crisis_scene_count } } set_variable = { name = kehillah_crisis_scene_count value = 1 } }
   change_variable = { name = kehillah_crisis_scene_count add = 1 }
  }
  random_list = {
   15 = {
    trigger = { primary_title = { NOT = { has_variable = kehillah_mob_castle_fell } NOT = { is_target_in_variable_list = { name = kehillah_crisis_scenes_seen target = flag:crisis_scene_1 } } OR = { var:kehillah_crisis_scene_count > 1 NOT = { has_variable = kehillah_crisis_last_opening } NOT = { var:kehillah_crisis_last_opening = 1 } } } }
    primary_title = { add_to_variable_list = { name = kehillah_crisis_scenes_seen target = flag:crisis_scene_1 } if = { limit = { var:kehillah_crisis_scene_count = 1 } set_variable = { name = kehillah_crisis_last_opening value = 1 } } }
    trigger_event = { id = kehillah_pogrom.0001 days = 1 }
   }
   15 = {
    trigger = { primary_title = {  NOT = { is_target_in_variable_list = { name = kehillah_crisis_scenes_seen target = flag:crisis_scene_2 } } OR = { var:kehillah_crisis_scene_count > 1 NOT = { has_variable = kehillah_crisis_last_opening } NOT = { var:kehillah_crisis_last_opening = 2 } } } }
