# =====================================================================
# Temple Service is learned, not assumed (2026-11-24, Daniel: "Should we
# adjust the Temple Service tenet so that maybe it's just not known and you can
# discover it personally through studying?").
#
# New games: scripts/kehillah_faith_revision_build.py no longer lists it among
# the familiar teachings, so outside the Samaritans' Gerizim rite (where it is a
# core tenet) every Jewish faith starts with it unknown. A reader learns it from
# a book that argues for it (Seder Kodashim, or a book written in play): the
# first mastery's kehillah_sefer_teach_tenet_effect adds it to his known tenets.
# Restoring the service in Jerusalem (kehillah_restore_temple_service_decision)
# makes it permitted for the faith; can_pick still requires the service.
#
# Existing saves: once, while the service is not restored, a faith that holds it
# exactly "known" returns it to unknown. CK3 has no effect that removes a tenet a
# character already knows, so a reader who knew it keeps it.
# =====================================================================

kehillah_temple_service_unknown_migrate_effect = {
	if = {
		limit = { NOT = { has_global_variable = kehillah_temple_service_unknown_20261124 } }
		set_global_variable = { name = kehillah_temple_service_unknown_20261124 value = yes }
		if = {
			limit = {
				NOT = { title:c_jerusalem = { has_variable = kehillah_temple_service_restored } }
			}
			faith:rabbinic_faith ?= { kehillah_temple_service_unknown_faith_effect = yes }
			faith:karaism ?= { kehillah_temple_service_unknown_faith_effect = yes }
			faith:haymanot ?= { kehillah_temple_service_unknown_faith_effect = yes }
			faith:malabarism ?= { kehillah_temple_service_unknown_faith_effect = yes }
			faith:kabarism ?= { kehillah_temple_service_unknown_faith_effect = yes }
		}
	}
}

# Scope: a faith.
kehillah_temple_service_unknown_faith_effect = {
	if = {
		limit = { main_rite = { has_tenet_status = { tenet = tenet_kehillah_temple_service status = known } } }
		set_tenet_status = { tenet = tenet:tenet_kehillah_temple_service status = unknown }
	}
}
