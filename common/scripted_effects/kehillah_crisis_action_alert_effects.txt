# 2026-11-25 (Daniel): "When you're in a crisis the top bar should show you a
# notification that opens an action decision when you have available actions.
# One for each available action." Used by
# common/important_actions/kehillah_crisis_action_alerts.txt.
#
# Scope: the player. Creates the ACTION alert while DECISION can be taken
# (vanilla's can_execute_decision: shown and valid), so a spent or blocked
# action, or one whose phase has passed, drops out of the top bar by itself.
kehillah_crisis_action_alert_effect = {
	if = {
		limit = {
			is_ai = no
			government_has_flag = government_is_kehillah
			primary_title ?= { NOT = { has_variable = kehillah_crisis_intro_pending } }
			can_execute_decision = $DECISION$
		}
		try_create_important_action = {
			important_action_type = $ACTION$
			actor = root
			decision = decision:$DECISION$
		}
	}
}
