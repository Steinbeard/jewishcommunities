# =====================================================================
# EVERY KEHILLAH STARTS WITH A BEIT MIDRASH. ADDED 2026-09-17 at user
# request ("give all Kehillahs a level 1 bet midrash and don't lock it
# behind synagogue level"). Called once per game from
# kehillah_library_on_game_start (common/on_action/kehillah_library_
# on_actions.txt).
#
# WHY: the Beit Midrash is the community library (docs/spec/v13-
# domicile-library-panel.md) -- the library panel exists only where the
# building does -- and the Learn Torah loop (study here, travel to study
# elsewhere, commission works) presumes every community has somewhere
# to keep a book. A house of study is also the one institution even a
# small medieval kehillah had; it is the Synagogue's growth that is the
# arc, not the existence of a study room.
#
# NOT LOCKED BEHIND THE SYNAGOGUE: the quarter's base_external_slots
# went 2 -> 3 for this (common/domiciles/types/kehillah_domicile_
# types.txt), so the seeded building occupies a slot that is unlocked
# at Synagogue tier 1 and leaves the original two free. It is added
# BEFORE anything else this effect touches, so it takes the lowest free
# slot (add_domicile_building fills in slot order) -- the developed
# Worms start (kehillah_worms_developed_start_effect) is guarded with
# NOT has_domicile_building_or_higher, so the two never collide.
#
# Scope: none (a game-start effect). Walks every living Kehillah leader
# rather than the registry (kehillah_registered_communities): this runs
# from its own on_action file, and on_action merge order between files
# is not something to lean on, so it must not assume the registry is
# built yet. One pass at game start; new games only -- an existing save
# does not re-fire on_game_start_after_lobby (v12 spec section 6).
# =====================================================================

kehillah_seed_beit_midrash_effect = {
	every_living_character = {
		limit = {
			government_has_flag = government_is_kehillah
			exists = domicile
			domicile = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_01 } }
		}
		domicile = {
			add_domicile_building = kehillah_beit_midrash_01
		}
	}
}
