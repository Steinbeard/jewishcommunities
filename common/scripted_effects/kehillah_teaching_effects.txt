# =====================================================================
# "Teaching on the Jews" -- 1066 assignment. Design: docs/spec/
# v31-host-teachings-stances-and-factions.md, section 4.3.
#
# Written onto each rite at game start with change_rite_doctrine (the
# rite-scope effect vanilla's councils use), so no vanilla religion or
# rite file is overridden. Rites that already hold a teaching are left
# alone; rites created later fall back by religion
# (kehillah_teaching_triggers.txt).
# =====================================================================

kehillah_assign_teachings_effect = {
	religion:christianity_religion = {
		every_faith = {
			every_faith_rite = {
				limit = { NOT = { kehillah_rite_has_teaching_trigger = yes } }
				if = {
					limit = { kehillah_rite_eastern_christian_trigger = yes }
					change_rite_doctrine = kehillah_teaching_imperial_law
				}
				else = {
					change_rite_doctrine = kehillah_teaching_witness
				}
			}
		}
	}
	religion:islam_religion = {
		every_faith = {
			every_faith_rite = {
				limit = { NOT = { kehillah_rite_has_teaching_trigger = yes } }
				change_rite_doctrine = kehillah_teaching_dhimma
			}
		}
	}
	debug_log = "kehillah_assign_teachings_effect: Christian and Muslim rites given their teaching on the Jews"
}
