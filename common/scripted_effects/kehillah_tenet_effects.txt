# =====================================================================
# Mechanics of the Sefer tenets outside the book system (2026-11-08;
# docs/spec/2026-11-08-sefer-tenet-mechanics.md). Most hooks sit in the
# systems they change (track values, sermons, the yeshiva tick, feasts,
# the head-of-rite acclamation); this file holds what has no home there.
# =====================================================================

# Once a year, from kehillah_sefer_yearly_effect. Ge'onut: the head of a
# rite that holds it -- or a head who holds it himself -- speaks with the
# academies' authority for the coming year. The modifier is renewed, not
# stacked, and lapses within a year once the rite or the head lets it go.
kehillah_tenet_yearly_effect = {
	religion:judaism_religion = {
		every_faith = {
			every_faith_rite = {
				limit = { exists = head_of_rite }
				save_temporary_scope_as = kehillah_geonut_rite
				head_of_rite = {
					if = {
						limit = { has_character_modifier = kehillah_geonut_head_modifier }
						remove_character_modifier = kehillah_geonut_head_modifier
					}
					if = {
						limit = {
							OR = {
								scope:kehillah_geonut_rite = { rite_has_parameter = kehillah_tenet_geonut_academy }
								has_personal_tenet_flag = kehillah_tenet_geonut_academy_personal
							}
						}
						add_character_modifier = { modifier = kehillah_geonut_head_modifier years = 1 }
					}
				}
			}
		}
	}
}

# Scope: the giver, from Send Alms to Jerusalem (Avelei Tziyon). The alms
# go to the Jerusalem community while it stands, and its yeshiva gains fame.
kehillah_tenet_send_zion_alms_effect = {
	save_temporary_scope_value_as = { name = kehillah_zion_alms value = kehillah_tenet_zion_alms_gold }
	add_piety = medium_piety_gain
	if = {
		limit = { government_has_flag = government_is_kehillah }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_tenet_zion_alms_greatness }
	}
	title:d_kehillah_jerusalem.holder ?= {
		if = {
			limit = { NOT = { this = root } }
			add_gold = scope:kehillah_zion_alms
			kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_tenet_zion_alms_fame }
		}
	}
}
