# Removes Kohen and Levite. Scope: character. Used when someone is outside Judaism.
kehillah_strip_priestly_traits_effect = {
	if = {
		limit = { has_trait = kohen }
		remove_trait = kohen
	}
	if = {
		limit = { has_trait = levite }
		remove_trait = levite
	}
}

# Game start only. Hands Kohen (5%) or Levite (4%) to unnamed Jewish men, the
# stand-in for the removed random_creation (which could not be kept off other
# faiths). Scope: none. Skips:
#  - women and children (the traits pass down the male line, and a boy's
#    descent is his father's);
#  - anyone authored in this mod's history files, whose descent is documented
#    or deliberately left open;
#  - anyone with a recorded father: generated start characters have none, and
#    a man with a father would otherwise outrank the line he descends from.
# Generated characters after game start get nothing: see ROADMAP.
kehillah_assign_priestly_traits_at_start_effect = {
	every_living_character = {
		limit = {
			is_male = yes
			is_adult = yes
			faith.religion = religion:judaism_religion
			NOT = { has_trait = kohen }
			NOT = { has_trait = levite }
			NOT = { exists = father }
			NOT = { kehillah_is_history_file_character_trigger = yes }
		}
		random_list = {
			5 = { add_trait = kohen }
			4 = { add_trait = levite }
			91 = { }
		}
	}
}
