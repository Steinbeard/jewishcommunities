# Scope: a Kehillah leader. Pays the host (the charter's suzerain) one
# quarter's tribute. Run by kehillah_tribute_quarterly_pulse and by the
# probe kehillah_tribute_debug.2.
kehillah_pay_host_tribute_effect = {
	if = {
		limit = {
			government_has_flag = government_is_kehillah
			is_tributary = yes
			exists = suzerain
			kehillah_tribute_quarter_paid_value > 0
		}
		save_scope_value_as = {
			name = kehillah_tribute_paid
			value = kehillah_tribute_quarter_paid_value
		}
		pay_short_term_gold = {
			target = suzerain
			gold = scope:kehillah_tribute_paid
		}
		primary_title = {
			set_variable = {
				name = kehillah_tribute_last_paid
				value = scope:kehillah_tribute_paid
			}
		}
	}
}
