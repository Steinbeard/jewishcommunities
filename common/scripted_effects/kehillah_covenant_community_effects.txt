# Scope: the giver; an actual gift/relief payment has already been made.
# The same reward hook serves individual aid and neighbouring-community relief.
kehillah_covenant_service_reward_effect = {
    save_temporary_scope_as = kehillah_covenant_aid_giver
    if = {
        limit = {
            kehillah_has_covenant_community_trigger = yes
            NOT = { has_character_flag = kehillah_covenant_service_rewarded }
            exists = scope:kehillah_covenant_aid_recipient
        }
        add_character_flag = { flag = kehillah_covenant_service_rewarded years = 1 }
        add_piety = minor_piety_gain
        if = {
            limit = { has_personal_tenet = tenet_kehillah_covenant_community }
            add_piety = minor_piety_gain
            add_stress = minor_stress_loss
        }
        scope:kehillah_covenant_aid_recipient = {
            add_opinion = { target = scope:kehillah_covenant_aid_giver modifier = kehillah_covenant_aid_opinion years = 5 }
            if = {
                limit = { culture = { has_same_culture_heritage = scope:kehillah_covenant_aid_giver.culture } }
                add_opinion = { target = scope:kehillah_covenant_aid_giver modifier = kehillah_covenant_shared_heritage_opinion years = 5 }
            }
        }
    }
}
