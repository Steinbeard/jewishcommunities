# =====================================================================
# VISIT THE JEWISH COMMUNITY -- household recruitment. 2026-11-16.
#
# Scope: the visiting adventurer. Requires saved scopes
# kehillah_visit_leader and kehillah_visit_title from kehillah_visit.0001.
# Every effect creates a random household of a deliberate vocation, then
# uses the ordinary household code to move that household into the camp.
# =====================================================================

kehillah_visit_finish_household_recruitment_effect = {
	kehillah_household_add_family_effect = {
		HEAD = scope:kh_new_head
		LEADER = root
	}
	scope:kehillah_visit_title = {
		set_variable = {
			name = kehillah_visit_recruit_cooldown
			value = yes
			years = kehillah_visit_recruit_cooldown_years_value
		}
	}
}

# A learned household: teacher, copyist, or careful student of the texts.
kehillah_visit_recruit_scholar_household_effect = {
	kehillah_save_courtier_culture_effect = { LEADER = scope:kehillah_visit_leader }
	create_character = {
		dynasty = generate
		age = { 28 50 }
		faith = scope:kehillah_visit_leader.faith
		rite = scope:kehillah_visit_leader.rite
		culture = scope:kehillah_courtier_culture
		location = root.location
		random_traits = yes
		gender_female_chance = 0
		trait = education_learning_3
		learning = { 12 18 }
		diplomacy = { 4 10 }
		stewardship = { 3 8 }
		save_scope_as = kh_new_head
	}
	scope:kh_new_head = {
		house = { set_variable = { name = kehillah_household_trade value = flag:scholar } }
	}
	kehillah_visit_finish_household_recruitment_effect = yes
}
