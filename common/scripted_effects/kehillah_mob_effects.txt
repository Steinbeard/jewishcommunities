# Spike 0 (2026-10-13): start_war fails for a created character who holds
# nothing ("not a ruler"). Vanilla's peasant revolt gives its leader a
# dynamic landless title (00_faction_effects.txt); the same here. Scope:
# the mob leader. COUNTY: the county the title is named from.
kehillah_mob_make_ruler_effect = {
	save_scope_as = kehillah_mob_new_leader
	create_dynamic_title = {
		tier = duchy
		name = KEHILLAH_MOB_TITLE_NAME
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = kehillah_mob_change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_capital_county = $COUNTY$
		set_landless_title = yes
		set_destroy_on_succession = yes
		set_delete_on_destroy = yes
		set_no_automatic_claims = yes
		set_definitive_form = yes
		set_can_be_named_after_dynasty = no
		change_title_holder = {
			holder = scope:kehillah_mob_new_leader
			change = scope:kehillah_mob_change
		}
	}
	resolve_title_and_vassal_change = scope:kehillah_mob_change
	scope:new_title = { generate_coa = factions }
	set_variable = { name = kehillah_mob_title value = scope:new_title }
}

# Spike 0b: march the army one county toward its target, by script
# (the AI roams to the nearest enemy holdings instead). Scope: the mob
# leader. TARGET: the target province.
kehillah_mob_step_toward_effect = {
	$TARGET$ = { save_scope_as = kehillah_mob_target_province }
	random_army = {
		save_scope_as = kehillah_mob_stepping_army
		location = { save_scope_as = kehillah_mob_here }
	}
	if = {
		limit = {
			exists = scope:kehillah_mob_here
			NOT = { scope:kehillah_mob_here = scope:kehillah_mob_target_province }
		}
		scope:kehillah_mob_here.county = {
			ordered_title_to_title_neighboring_county = {
				order_by = {
					value = "title_province.squared_distance(scope:kehillah_mob_target_province)"
					multiply = -1
				}
				position = 0
				save_scope_as = kehillah_mob_next_county
			}
		}
		if = {
			limit = { exists = scope:kehillah_mob_next_county }
			scope:kehillah_mob_stepping_army = {
				set_army_location = scope:kehillah_mob_next_county.title_province
			}
		}
	}
}
