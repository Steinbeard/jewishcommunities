# Character scope, called only after a successful Examine Scripture course.
# Three successful courses confer the vanilla Religious Scholar trait; later
# courses progress it. This is intentionally separate from rabbinic semicha.
kehillah_record_miqra_study_effect = {
	if = {
		limit = { is_playable_character = yes }
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
	}
	if = {
		limit = { has_trait = lifestyle_scholar }
		add_trait_xp = { trait = lifestyle_scholar value = $XP$ }
	}
	else = {
		if = {
			limit = { NOT = { has_variable = kehillah_miqra_successful_courses } }
			set_variable = { name = kehillah_miqra_successful_courses value = 0 }
		}
		change_variable = { name = kehillah_miqra_successful_courses add = 1 }
		if = {
			limit = { var:kehillah_miqra_successful_courses >= 3 }
			add_trait = lifestyle_scholar
			remove_variable = kehillah_miqra_successful_courses
		}
	}
}
