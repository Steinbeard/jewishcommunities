# =====================================================================
# FOUNDING NEW COMMUNITIES -- V34 section 1 (docs/spec/v34-outlook-
# teachings-and-founding.md), 2026-10-13. Daniel first asked for about
# one new community every five years; the same day he replaced the
# random yearly pulse with a decision ("Send Out Settlers",
# kehillah_settlers_decisions.txt): an overcrowded community sends one of
# its own to found a daughter community. Player and AI leaders both take
# it; an AI-wide gap of ~3 years keeps the world rate near one founding
# every few years.
# =====================================================================

# Scope: the founder (an existing character -- a courtier, or a debug
# spawn). scope:kehillah_founding_county, scope:kehillah_mother_leader
# and scope:kehillah_mother_title saved. Moves him to the county and
# founds a Kehillah there the way the Norman-Conquest foundings do.
kehillah_found_with_founder_effect = {
	save_scope_as = kehillah_natural_founder
	save_scope_as = kehillah_founder
	move_to_pool_at = scope:kehillah_founding_county.title_province
	create_adventurer_title = {
		name = kehillah_found_community_title_name
		holder = scope:kehillah_natural_founder
		government = kehillah_government
		article = kehillah_found_community_title_article
		save_scope_as = kehillah_new_community_title
	}
	hidden_effect = { set_primary_title_to = scope:kehillah_new_community_title }
	if = {
		limit = { NOT = { government_has_flag = government_is_kehillah } }
		change_government = kehillah_government
	}
	if = {
		limit = { NOT = { has_realm_law = kehillah_appointment_succession_law } }
		add_realm_law = kehillah_appointment_succession_law
	}
	hidden_effect = {
		scope:kehillah_new_community_title = {
			kehillah_init_pillars_effect = yes
			change_variable = { name = kehillah_var_stability add = 50 }
			change_variable = { name = kehillah_var_prosperity add = 50 }
			set_variable = { name = kehillah_naturally_founded value = yes }
			set_variable = { name = kehillah_mother_community value = scope:kehillah_mother_title }
		}
		kehillah_update_leader_flavor_effect = yes
		add_to_global_variable_list = {
			name = kehillah_registered_communities
			target = scope:kehillah_new_community_title
		}
		kehillah_refresh_all_settlement_networks_effect = yes
		scope:kehillah_new_community_title = {
			kehillah_seed_starting_library_effect = yes
		}
		kehillah_refresh_map_view_for_players_effect = yes
		kehillah_maintain_host_charter_effect = yes
		kehillah_goal_credit_daughter_community_effect = yes
	}
	kehillah_notify_region_players_effect = {
		TYPE = msg_kehillah_community_news
		TITLE = kehillah_news_founded_t
		DESC = kehillah_news_founded_d
	}
	debug_log = "kehillah_found_with_founder_effect: a new community was founded"
	debug_log_scopes = yes
}

# Scope: the mother community's leader. Saves up to three destination
# counties (kehillah_settlers_dest_1..3) and three leader candidates
# (kehillah_settlers_lead_1..3).
kehillah_settlers_pick_effect = {
	save_scope_as = kehillah_mother_leader
	primary_title = { save_scope_as = kehillah_mother_title }
	clear_saved_scope = kehillah_settlers_dest_1
	clear_saved_scope = kehillah_settlers_dest_2
	clear_saved_scope = kehillah_settlers_dest_3
	clear_saved_scope = kehillah_settlers_lead_1
	clear_saved_scope = kehillah_settlers_lead_2
	clear_saved_scope = kehillah_settlers_lead_3
	random_county = {
		limit = { kehillah_settlers_destination_trigger = yes }
		weight = {
			base = 1
			modifier = {
				add = 2
				holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
			}
		}
		save_scope_as = kehillah_settlers_dest_1
	}
	random_county = {
		limit = {
			kehillah_settlers_destination_trigger = yes
			NOT = { this = scope:kehillah_settlers_dest_1 }
		}
		save_scope_as = kehillah_settlers_dest_2
	}
	random_county = {
		limit = {
			kehillah_settlers_destination_trigger = yes
			NOT = { this = scope:kehillah_settlers_dest_1 }
			trigger_if = {
				limit = { exists = scope:kehillah_settlers_dest_2 }
				NOT = { this = scope:kehillah_settlers_dest_2 }
			}
		}
		save_scope_as = kehillah_settlers_dest_3
	}
	ordered_courtier = {
		limit = { kehillah_settlers_leader_candidate_trigger = yes }
		order_by = { value = stewardship add = learning }
		position = 0
		save_scope_as = kehillah_settlers_lead_1
	}
	ordered_courtier = {
		limit = { kehillah_settlers_leader_candidate_trigger = yes }
		order_by = { value = stewardship add = learning }
		position = 1
		save_scope_as = kehillah_settlers_lead_2
	}
	ordered_courtier = {
		limit = { kehillah_settlers_leader_candidate_trigger = yes }
		order_by = { value = stewardship add = learning }
		position = 2
		save_scope_as = kehillah_settlers_lead_3
	}
}

# Scope: the mother leader; scope:kehillah_settlers_leader and scope:
# kehillah_founding_county chosen. The leader goes, with his household and
# up to kehillah_settlers_max_followers others; the community founded.
kehillah_settlers_go_effect = {
	save_scope_as = kehillah_mother_leader
	primary_title = {
		save_scope_as = kehillah_mother_title
		set_variable = { name = kehillah_settlers_cooldown value = yes years = kehillah_settlers_cooldown_years }
	}
	# Who goes with him: his own household first, then others.
	scope:kehillah_settlers_leader = {
		every_close_family_member = {
			limit = {
				is_courtier_of = scope:kehillah_mother_leader
				NOT = { this = scope:kehillah_mother_leader }
			}
			add_to_list = kehillah_settlers_party
		}
	}
	ordered_courtier = {
		limit = {
			is_adult = yes
			is_jewish_character_trigger = yes
			is_imprisoned = no
			NOT = { this = scope:kehillah_settlers_leader }
			NOT = { is_in_list = kehillah_settlers_party }
			NOT = { has_variable = kehillah_yeshiva_enrolled }
			NOT = { is_councillor = yes }
			NOT = { scope:kehillah_mother_leader.player_heir ?= this }
			NOT = { is_close_family_of = scope:kehillah_mother_leader }
		}
		order_by = age
		max = kehillah_settlers_max_followers
		add_to_list = kehillah_settlers_party
	}
	scope:kehillah_settlers_leader = { kehillah_found_with_founder_effect = yes }
	every_in_list = {
		list = kehillah_settlers_party
		save_scope_as = kehillah_settler
		scope:kehillah_natural_founder = { add_courtier = scope:kehillah_settler }
	}
	# The mother community's credit: a daughter is greatness.
	scope:kehillah_mother_title = {
		change_variable = { name = kehillah_var_greatness add = 20 }
	}
	if = {
		limit = { is_ai = yes }
		set_global_variable = { name = kehillah_settlers_ai_gap value = yes years = 3 }
	}
	debug_log = "kehillah_settlers_go_effect: settlers left to found a daughter community"
}

# Debug only (kehillah_founding_debug.1): a founder is created rather than
# sent from a court -- the old random founding, kept as a probe.
kehillah_natural_founding_effect = {
	clear_saved_scope = kehillah_mother_title
	clear_saved_scope = kehillah_founding_county
	random_in_global_list = {
		variable = kehillah_registered_communities
		limit = {
			exists = holder
			exists = holder.domicile.domicile_location
		}
		save_scope_as = kehillah_mother_title
	}
	if = {
		limit = { exists = scope:kehillah_mother_title }
		scope:kehillah_mother_title.holder = { save_scope_as = kehillah_mother_leader }
		random_county = {
			limit = { kehillah_settlers_destination_trigger = yes }
			save_scope_as = kehillah_founding_county
		}
	}
	if = {
		limit = { exists = scope:kehillah_founding_county }
		create_character = {
			faith = scope:kehillah_mother_leader.faith
			culture = scope:kehillah_mother_leader.culture
			age = { 28 45 }
			location = scope:kehillah_founding_county.title_province
			random_traits = yes
			gender_female_chance = 0
			dynasty = generate
			save_scope_as = kehillah_debug_founder
		}
		scope:kehillah_debug_founder = { kehillah_found_with_founder_effect = yes }
	}
	else = {
		debug_log = "kehillah_natural_founding_effect: no mother community or no welcoming county"
	}
}
