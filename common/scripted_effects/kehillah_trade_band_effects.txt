# =====================================================================
# TRADE FROM AFFLUENCE (2026-11-26)
# Spec: docs/spec/2026-11-26-pillar-ceilings-deeds-and-founding.md, section 5
#
# Daniel (2026-11-26): "higher prosperity unlocks more trade routes maybe?
# Without ever letting you really compete with merchants -- maybe just one
# or two. And if you don't have Silk and Silver we could make it give you
# some additional gold once in a while."
#
# WITH SILK AND SILVER. The Kehillah government already allows Profit
# Routes at -1 (so 0 by default) and smaller than a merchant's (two
# markets, shorter, less capacity; crafts spec 4a/4b). The routes now come
# from the community's Affluence band instead of from the Factor's House
# and Market Hall:
#     Healthy        1 route
#     Flourishing+   2 routes
#   + Commercial leadership style: 1 more (kehillah_style_commercial_modifier)
# The Factor's House / Market Hall keep their capacity and competitiveness.
#
# WITHOUT IT. The same slots become "trading ventures": each quarter, each
# venture has a chance to bring home some gold (kehillah_trade_venture_
# pulse_effect). The player sees a modifier naming the ventures and a
# toast when one pays.
#
# Scope of everything here: the Kehillah leader.
# =====================================================================

kehillah_trade_route_band_effect = {
	if = { limit = { has_character_modifier = kehillah_trade_routes_1_modifier } remove_character_modifier = kehillah_trade_routes_1_modifier }
	if = { limit = { has_character_modifier = kehillah_trade_routes_2_modifier } remove_character_modifier = kehillah_trade_routes_2_modifier }
	if = { limit = { has_character_modifier = kehillah_trade_ventures_1_modifier } remove_character_modifier = kehillah_trade_ventures_1_modifier }
	if = { limit = { has_character_modifier = kehillah_trade_ventures_2_modifier } remove_character_modifier = kehillah_trade_ventures_2_modifier }

	if = {
		limit = { kehillah_trade_band_slots_value >= 1 }
		if = {
			limit = { has_dlc_feature = silk_and_silver }
			if = {
				limit = { kehillah_trade_band_slots_value >= 2 }
				add_character_modifier = kehillah_trade_routes_2_modifier
			}
			else = {
				add_character_modifier = kehillah_trade_routes_1_modifier
			}
		}
		else = {
			if = {
				limit = { kehillah_trade_band_slots_value >= 2 }
				add_character_modifier = kehillah_trade_ventures_2_modifier
			}
			else = {
				add_character_modifier = kehillah_trade_ventures_1_modifier
			}
		}
	}
}

# Quarterly, only without Silk and Silver: each venture may pay.
kehillah_trade_venture_pulse_effect = {
	kehillah_household_refresh_effect = yes
	kehillah_household_abstract_ventures_effect = yes
}
