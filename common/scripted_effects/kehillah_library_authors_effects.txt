# Scope: the player. Stores each library author on the player as
# var:kehillah_lib_author_<slug>, which the library panel's hover cards read with
# [GetPlayer.MakeScope.Var('kehillah_lib_author_<slug>').Char.GetShortUIName] to link to
# the historical character (history/characters/kehillah_library_authors.txt and
# kehillah_saadia_gaon.txt). Run by the scripted_gui kehillah_lib_cache_authors, which the
# panel fires when the library tab is clicked and when a shelf row is clicked.
#
# Remove-then-set: reading var: in a limit also keeps the engine's unused-variable lint quiet.

kehillah_lib_cache_authors_effect = {
