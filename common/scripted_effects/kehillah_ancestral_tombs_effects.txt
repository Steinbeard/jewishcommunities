# Register only with Rabbinic Judaism; faith splits inherit its native registry.
# Use separate travel PoIs rather than the generic tomb/relic event chain.
kehillah_initialize_ancestral_tombs_effect = {
    faith:rabbinic_faith = {
        if = {
            limit = { NOT = { has_saint_character = character:kehillah_rachel_imenu } }
            add_saint = { character = character:kehillah_rachel_imenu burial_province = province:5965 }
        }
        # CK3 1.21 made Hebron (5966) an empty barony until 1178, and
        # add_saint needs a holding. Register once Hebron has one; the
        # yearly re-run (kehillah_ancestral_tombs_on_actions.txt) catches it.
        if = {
            limit = {
                NOT = { has_saint_character = character:kehillah_avraham_avinu }
                province:5966 = { has_holding = yes }
            }
            add_saint = { character = character:kehillah_avraham_avinu burial_province = province:5966 }
        }
    }
    # Empty the native relic pool too: the generic extraction decision must not
    # make either ancestor a source of bodily relics. Safe to repeat on loading.
    while = {
        limit = { character:kehillah_rachel_imenu = { remaining_saint_relics > 0 } }
        character:kehillah_rachel_imenu = { consume_saint_relic_charge = character:kehillah_rachel_imenu }
    }
    while = {
        limit = { character:kehillah_avraham_avinu = { remaining_saint_relics > 0 } }
        character:kehillah_avraham_avinu = { consume_saint_relic_charge = character:kehillah_avraham_avinu }
    }
    province:5965 = {
        if = {
            limit = { NOT = { has_travel_point_of_interest = kehillah_poi_kever_rachel } }
            add_travel_point_of_interest = kehillah_poi_kever_rachel
        }
    }
    province:5966 = {
        if = {
            limit = { NOT = { has_travel_point_of_interest = kehillah_poi_machpelah } }
            add_travel_point_of_interest = kehillah_poi_machpelah
        }
    }
}
