# V40 slice 1, 2026-10-14. Scope: a faith, once yearly from its main rite.
# Incidents and goodwill remember roughly the last five years. Missing
# variables stay absent; the HUD values treat them as zero.
kehillah_faith_incidents_yearly_effect = {
	if = {
		limit = { has_variable = kehillah_faith_pressure }
		change_variable = { name = kehillah_faith_pressure multiply = 0.8 }
		if = {
			limit = {
				var:kehillah_faith_pressure >= kehillah_faith_pressure_head_mood_threshold
				exists = religious_head
			}
			religious_head = { kehillah_shift_outlook_effect = { AMOUNT = -1 } }
		}
	}
	if = {
		limit = { has_variable = kehillah_faith_goodwill }
		change_variable = { name = kehillah_faith_goodwill multiply = 0.8 }
	}
}
