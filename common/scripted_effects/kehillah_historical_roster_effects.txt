# Places living, named families at the court of their 1066 community.
# Called before generic households are generated. $PERSON$ is a character ID.
kehillah_roster_recruit_effect = {
	$PERSON$ ?= {
		if = {
			limit = { is_alive = yes NOT = { is_courtier_of = scope:kehillah_leader } }
			set_employer = scope:kehillah_leader
		}
	}
}

# $PERSON$ is the ID of an already recruited courtier, $SEAT$ the council
# position, $TRADE$ their house's communal work.
kehillah_roster_seat_effect = {
	$PERSON$ ?= {
		if = {
			limit = { is_alive = yes is_courtier_of = scope:kehillah_leader }
			add_character_flag = kehillah_household_founder
			house = { set_variable = { name = kehillah_household_trade value = flag:$TRADE$ } }
		}
	}
	scope:kehillah_leader = {
		if = {
			limit = { NOT = { exists = cp:$SEAT$ } $PERSON$ ?= { is_alive = yes is_courtier_of = scope:kehillah_leader } }
			assign_councillor_type = { type = $SEAT$ target = $PERSON$ remove_existing_councillor = no }
		}
	}
}
