# =====================================================================
# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs
# a gift to the answering community and is limited by a cooldown; answering
# is the reward of Greatness. Content: the Bet Din docket's 14 cases
# (kehillah_din_letter_N, kehillah_din_answer_N, kehillah_din_answer_careful_N).
# =====================================================================

# Scope: the asking leader. scope:kehillah_sheelah_posek is the posek.
kehillah_sheelah_send_effect = {
	save_scope_as = kehillah_sheelah_asker
	set_variable = { name = kehillah_sheelah_cooldown value = yes days = kehillah_sheelah_cooldown_days }
	random_list = {
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 1 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 2 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 3 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 4 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 5 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 6 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 7 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 8 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 9 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 10 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 11 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 12 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 13 } }
		1 = { save_scope_value_as = { name = kehillah_sheelah_case value = 14 } }
	}
	# The caller has taken the gift from the asker (the interaction shows it;
	# the AI pulse takes it); here it reaches the posek's community.
	save_scope_value_as = { name = kehillah_sheelah_gift value = kehillah_sheelah_gift_value }
	scope:kehillah_sheelah_posek = {
		add_gold = scope:kehillah_sheelah_gift
		set_variable = { name = kehillah_sheelah_pending_from value = scope:kehillah_sheelah_asker days = kehillah_sheelah_pending_days }
		set_variable = { name = kehillah_sheelah_case value = scope:kehillah_sheelah_case days = kehillah_sheelah_pending_days }
		trigger_event = { id = kehillah_sheelah.0001 days = { 10 25 } }
		if = {
			limit = { is_ai = no }
			set_variable = { name = kehillah_sheelah_player_letter_recent value = yes days = kehillah_sheelah_player_letter_gap_days }
		}
	}
	debug_log = "kehillah_sheelah_send_effect: a she'elah is sent"
	debug_log_scopes = yes
}

# Scope: the posek, answering (kehillah_sheelah.0001). QUALITY: 0 declined,
# 1 answered, 2 answered from the sources.
kehillah_sheelah_reply_effect = {
	scope:kehillah_sheelah_asker = {
		set_variable = { name = kehillah_sheelah_reply_from value = scope:kehillah_sheelah_posek days = kehillah_sheelah_pending_days }
		set_variable = { name = kehillah_sheelah_reply_case value = scope:kehillah_sheelah_posek.var:kehillah_sheelah_case days = kehillah_sheelah_pending_days }
		set_variable = { name = kehillah_sheelah_reply_quality value = $QUALITY$ days = kehillah_sheelah_pending_days }
		trigger_event = { id = kehillah_sheelah.0002 days = { 10 25 } }
	}
	remove_variable = kehillah_sheelah_pending_from
	remove_variable = kehillah_sheelah_case
}

# Scope: an AI Kehillah leader (yearly). Below Flourishing Greatness, now and
# then, it writes to a posek of its faith: likelier a great one, and one of
# its own rite.
kehillah_sheelah_ai_pulse_effect = {
	if = {
		limit = {
			is_ai = yes
			kehillah_sheelah_can_ask_trigger = yes
			NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_flourishing_threshold } }
			gold >= kehillah_sheelah_gift_value
		}
		save_scope_as = kehillah_sheelah_ai_asker
		random = {
			chance = kehillah_sheelah_ai_chance
			random_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					exists = holder
					holder = {
						kehillah_sheelah_posek_trigger = { ASKER = scope:kehillah_sheelah_ai_asker }
						in_diplomatic_range = scope:kehillah_sheelah_ai_asker
					}
				}
				weight = {
					base = 1
					modifier = {
						add = 2
						kehillah_pillar_at_least_trigger_title = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_flourishing_threshold }
					}
					modifier = {
						add = 3
						kehillah_pillar_at_least_trigger_title = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_legendary_threshold }
					}
					modifier = {
						add = 1
						holder.rite ?= scope:kehillah_sheelah_ai_asker.rite
					}
				}
				holder = { save_scope_as = kehillah_sheelah_posek }
			}
			if = {
				limit = { exists = scope:kehillah_sheelah_posek }
				remove_short_term_gold = kehillah_sheelah_gift_value
				kehillah_sheelah_send_effect = yes
			}
		}
	}
}

# Scope: the student. scope:kehillah_study_teacher is the rabbi. The caller
# (the interaction) takes the gift; here it reaches the rabbi.
kehillah_study_help_send_effect = {
	save_scope_as = kehillah_study_student
	set_variable = { name = kehillah_study_help_cooldown value = yes days = kehillah_study_help_cooldown_days }
	set_variable = { name = kehillah_study_help_work value = var:kehillah_study_current_work_id days = kehillah_study_help_pending_days }
	save_scope_value_as = { name = kehillah_study_help_gift value = kehillah_study_help_gift_value }
	scope:kehillah_study_teacher = {
		add_gold = scope:kehillah_study_help_gift
		set_variable = { name = kehillah_study_help_pending_from value = scope:kehillah_study_student days = kehillah_study_help_pending_days }
		trigger_event = { id = kehillah_sheelah.0011 days = { 10 20 } }
	}
	debug_log = "kehillah_study_help_send_effect: a student asks for help with his reading"
}

# Scope: the rabbi answering (kehillah_sheelah.0011). QUALITY: 0 declined,
# 1 explained, 2 a full explanation.
kehillah_study_help_reply_effect = {
	scope:kehillah_study_student = {
		set_variable = { name = kehillah_study_help_reply_from value = scope:kehillah_study_teacher days = kehillah_study_help_pending_days }
		set_variable = { name = kehillah_study_help_reply_quality value = $QUALITY$ days = kehillah_study_help_pending_days }
		trigger_event = { id = kehillah_sheelah.0012 days = { 10 20 } }
	}
	remove_variable = kehillah_study_help_pending_from
}
