# Scope: a county. A historical loss known at the bookmark. STATUS 2 =
# destroyed, 3 = abandoned or dissolved. CAUSE is a kehillah_history_*
# flag suffix. A later refounding helper call marks the place living again.
kehillah_history_seed_historical_loss_effect = {
	set_variable = { name = kehillah_history_last_loss_year value = $YEAR$ }
	set_variable = { name = kehillah_history_status value = $STATUS$ }
	set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_$CAUSE$ }
	kehillah_history_seed_former_site_effect = { PEAK = $PEAK$ }
}

# Scope: a county with a seeded historical loss. The community came back
# around YEAR; the year is approximate, so the summary says "around".
kehillah_history_seed_historical_refounding_effect = {
	set_variable = { name = kehillah_history_last_refounded_year value = $YEAR$ }
	set_variable = { name = kehillah_history_last_refounded_approx value = yes }
	set_variable = { name = kehillah_history_status value = 1 }
	if = {
		limit = { is_target_in_global_variable_list = { name = kehillah_former_sites target = this } }
		remove_list_global_variable = { name = kehillah_former_sites target = this }
	}
}

# Scope: a county. Remembers how high the place once stood and lists it
# among the world's former communities (the map and the ledger read the
# list). PEAK only ever raises the stored height.
kehillah_history_seed_former_site_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = kehillah_history_peak }
				var:kehillah_history_peak < $PEAK$
			}
		}
		set_variable = { name = kehillah_history_peak value = $PEAK$ }
	}
	if = {
		limit = { NOT = { is_target_in_global_variable_list = { name = kehillah_former_sites target = this } } }
		add_to_global_variable_list = { name = kehillah_former_sites target = this }
	}
}

# Scope: a living community title. Raises its county's remembered height
# to the Standing band the community holds now (1 crisis .. 5 legendary).
# Called from the rank pass, so it runs quarterly and on every change.
kehillah_history_update_peak_effect = {
	if = {
		limit = {
			has_variable = kehillah_history_county
			has_variable = kehillah_var_standing
		}
		if = {
			limit = { var:kehillah_var_standing >= kehillah_band_legendary_threshold }
			var:kehillah_history_county = { kehillah_history_raise_peak_effect = { PEAK = 5 } }
		}
		else_if = {
			limit = { var:kehillah_var_standing >= kehillah_band_flourishing_threshold }
			var:kehillah_history_county = { kehillah_history_raise_peak_effect = { PEAK = 4 } }
		}
		else_if = {
			limit = { var:kehillah_var_standing >= kehillah_band_healthy_threshold }
			var:kehillah_history_county = { kehillah_history_raise_peak_effect = { PEAK = 3 } }
		}
		else_if = {
			limit = { var:kehillah_var_standing >= kehillah_band_strained_threshold }
			var:kehillah_history_county = { kehillah_history_raise_peak_effect = { PEAK = 2 } }
		}
		else = {
			var:kehillah_history_county = { kehillah_history_raise_peak_effect = { PEAK = 1 } }
		}
	}
}

# 2026-11-15: a leader who dies with no successor. The engine ends such a
# title without passing through kehillah_dissolve_community_effect, so
# nothing would record the loss. on_death notes the place (pending);
# the yearly check records the loss if the seat is still empty, and
# clears the note if someone took it up.
# ---------------------------------------------------------------------

# Scope: a dying Kehillah leader (on_death).
kehillah_history_note_leader_death_effect = {
	save_scope_as = kehillah_history_dying_leader
	primary_title = {
		save_scope_as = kehillah_history_dying_title
		kehillah_history_update_peak_effect = yes
		if = {
			limit = { exists = var:kehillah_history_county }
			var:kehillah_history_county = {
				set_variable = { name = kehillah_history_pending_title value = scope:kehillah_history_dying_title }
				set_variable = { name = kehillah_history_pending_leader value = scope:kehillah_history_dying_leader }
				set_variable = { name = kehillah_history_pending_year value = current_year }
				if = {
					limit = { exists = holder.top_liege }
					set_variable = { name = kehillah_history_pending_host value = holder.top_liege }
				}
				if = {
					limit = { NOT = { is_target_in_global_variable_list = { name = kehillah_history_pending_counties target = this } } }
					add_to_global_variable_list = { name = kehillah_history_pending_counties target = this }
				}
			}
		}
	}
}

# Scope: none (yearly_global_pulse).
kehillah_history_check_pending_effect = {
	every_in_global_list = {
		variable = kehillah_history_pending_counties
		if = {
			limit = {
				OR = {
					NOT = { exists = var:kehillah_history_pending_title }
					NOT = { exists = var:kehillah_history_pending_title.holder }
				}
			}
			set_variable = { name = kehillah_history_last_loss_year value = var:kehillah_history_pending_year }
			set_variable = { name = kehillah_history_status value = 3 }
			set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_no_successor }
			if = {
				limit = { exists = var:kehillah_history_pending_leader }
				set_variable = { name = kehillah_history_last_leader value = var:kehillah_history_pending_leader }
			}
			if = {
				limit = { exists = var:kehillah_history_pending_host }
				set_variable = { name = kehillah_history_last_host value = var:kehillah_history_pending_host }
			}
			kehillah_history_seed_former_site_effect = { PEAK = 1 }
			debug_log = "kehillah_history: heirless community recorded as lost"
		}
		remove_variable = kehillah_history_pending_title
		remove_variable = kehillah_history_pending_leader
		remove_variable = kehillah_history_pending_year
		remove_variable = kehillah_history_pending_host
	}
	# Cleared after the walk, not inside it.
	clear_global_variable_list = kehillah_history_pending_counties
}

# Scope: a county.
kehillah_history_clear_pending_effect = {
	remove_variable = kehillah_history_pending_title
	remove_variable = kehillah_history_pending_leader
	remove_variable = kehillah_history_pending_year
	remove_variable = kehillah_history_pending_host
	if = {
		limit = { is_target_in_global_variable_list = { name = kehillah_history_pending_counties target = this } }
		remove_list_global_variable = { name = kehillah_history_pending_counties target = this }
	}
}

# Scope: a county.
kehillah_history_raise_peak_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = kehillah_history_peak }
				var:kehillah_history_peak < $PEAK$
			}
		}
		set_variable = { name = kehillah_history_peak value = $PEAK$ }
	}
}

# Scope: an authored Kehillah title with kehillah_founded_year and a capital.
# Called at new-game start for the complete historical roster, held or vacant.
kehillah_history_seed_authored_title_effect = {
	if = {
		limit = {
			has_variable = kehillah_founded_year
			exists = title_capital_county
		}
		set_variable = { name = kehillah_history_county value = title_capital_county }
		title_capital_county = {
			if = {
				limit = { NOT = { has_variable = kehillah_history_original_year } }
				set_variable = { name = kehillah_history_original_year value = prev.var:kehillah_founded_year }
				# Most authored years are the earliest record, not a founding;
				# a title whose history sets kehillah_founded_exact has a real one.
				if = {
					limit = { prev = { NOT = { has_variable = kehillah_founded_exact } } }
					set_variable = { name = kehillah_history_original_approx value = yes }
				}
			}
		}
	}
}

# Scope: the founder. Caller saves kehillah_founding_county and
# kehillah_new_community_title before this call; optional
# kehillah_history_origin_county denotes the settlers' actual source.
kehillah_history_record_founding_effect = {
	save_scope_as = kehillah_history_founder
	if = {
		limit = {
			exists = scope:kehillah_founding_county
			exists = scope:kehillah_new_community_title
		}
		scope:kehillah_founding_county = {
			if = {
				limit = { NOT = { has_variable = kehillah_history_original_year } }
				set_variable = { name = kehillah_history_original_year value = current_year }
				set_variable = { name = kehillah_history_original_founder value = scope:kehillah_history_founder }
				if = {
					limit = { exists = scope:kehillah_history_origin_county }
					set_variable = { name = kehillah_history_original_origin value = scope:kehillah_history_origin_county }
				}
			}
			else = {
				set_variable = { name = kehillah_history_last_refounded_year value = current_year }
				set_variable = { name = kehillah_history_last_refounder value = scope:kehillah_history_founder }
				if = {
					limit = { exists = scope:kehillah_history_origin_county }
					set_variable = { name = kehillah_history_last_refounding_origin value = scope:kehillah_history_origin_county }
				}
				else = { remove_variable = kehillah_history_last_refounding_origin }
				# A refounding now has an exact year.
				if = {
					limit = { has_variable = kehillah_history_last_refounded_approx }
					remove_variable = kehillah_history_last_refounded_approx
				}
			}
			set_variable = { name = kehillah_history_status value = 1 }
			# 2026-11-14: no longer a former community.
			if = {
				limit = { is_target_in_global_variable_list = { name = kehillah_former_sites target = this } }
				remove_list_global_variable = { name = kehillah_former_sites target = this }
			}
		}
		scope:kehillah_new_community_title = {
			set_variable = { name = kehillah_history_county value = scope:kehillah_founding_county }
			set_variable = { name = kehillah_founded_year value = scope:kehillah_founding_county.var:kehillah_history_original_year }
		}
		debug_log = "kehillah_history: founding/refounding recorded on county"
		# 2026-11-09: in Iberia, a founding eases the struggle.
		kehillah_iberia_struggle_push_effect = { CATALYST = founded KIND = flourish }
	}
}

# Scope: the departing leader, BEFORE destroy_title. The reason is derived
# from existing cause markers; the attacker is optional and must be a real
# saved character, never inferred from a hostile realm's current ruler.
kehillah_history_record_loss_effect = {
	save_scope_as = kehillah_history_loss_leader
	# 2026-11-09: in Iberia, a lost community hardens the struggle.
	kehillah_iberia_struggle_push_effect = { CATALYST = lost KIND = ruin }
	# 2026-11-14: the place remembers how high it stood before the end.
	primary_title ?= { kehillah_history_update_peak_effect = yes }
	if = {
		limit = { exists = domicile.domicile_location.county }
		domicile.domicile_location.county = {
			kehillah_history_seed_former_site_effect = { PEAK = 1 }
			# 2026-11-15: who led it and who ruled it at the end, for the
			# inspector's account. Both stay readable after either dies.
			set_variable = { name = kehillah_history_last_leader value = scope:kehillah_history_loss_leader }
			if = {
				limit = { exists = holder.top_liege }
				set_variable = { name = kehillah_history_last_host value = holder.top_liege }
			}
			else = { remove_variable = kehillah_history_last_host }
			if = {
				limit = { has_variable = kehillah_history_pending_title }
				kehillah_history_clear_pending_effect = yes
			}
			set_variable = { name = kehillah_history_last_loss_year value = current_year }
			set_variable = { name = kehillah_history_last_barony value = scope:kehillah_history_loss_leader.domicile.domicile_location.barony }
			if = { limit = { has_variable = kehillah_history_last_destroyer } remove_variable = kehillah_history_last_destroyer }
			if = {
				limit = { scope:kehillah_history_loss_leader.primary_title = { has_variable = kehillah_pogrom_recent_news } }
				set_variable = { name = kehillah_history_status value = 2 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_mob }
				if = {
					limit = { scope:kehillah_history_loss_leader.primary_title = { has_variable = kehillah_mob_threat } }
					set_variable = { name = kehillah_history_last_destroyer value = scope:kehillah_history_loss_leader.primary_title.var:kehillah_mob_threat }
				}
			}
			else_if = {
				limit = { scope:kehillah_history_loss_leader = { has_character_flag = kehillah_baptized_by_force } }
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_forced_conversion }
			}
			else_if = {
				limit = { scope:kehillah_history_loss_leader = { has_variable = kehillah_exiles_from } }
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_expulsion }
			}
			else = {
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_collapse }
			}
		}
		debug_log = "kehillah_history: loss recorded on county before title teardown"
	}
}
