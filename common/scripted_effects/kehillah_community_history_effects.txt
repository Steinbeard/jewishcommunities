# Scope: an authored Kehillah title with kehillah_founded_year and a capital.
# Called at new-game start for the complete historical roster, held or vacant.
kehillah_history_seed_authored_title_effect = {
	if = {
		limit = {
			has_variable = kehillah_founded_year
			exists = title_capital_county
		}
		set_variable = { name = kehillah_history_county value = title_capital_county }
		title_capital_county = {
			if = {
				limit = { NOT = { has_variable = kehillah_history_original_year } }
				set_variable = { name = kehillah_history_original_year value = prev.var:kehillah_founded_year }
				set_variable = { name = kehillah_history_original_approx value = yes }
			}
		}
	}
}

# Scope: the founder. Caller saves kehillah_founding_county and
# kehillah_new_community_title before this call; optional
# kehillah_history_origin_county denotes the settlers' actual source.
kehillah_history_record_founding_effect = {
	save_scope_as = kehillah_history_founder
	if = {
		limit = {
			exists = scope:kehillah_founding_county
			exists = scope:kehillah_new_community_title
		}
		scope:kehillah_founding_county = {
			if = {
				limit = { NOT = { has_variable = kehillah_history_original_year } }
				set_variable = { name = kehillah_history_original_year value = current_year }
				set_variable = { name = kehillah_history_original_founder value = scope:kehillah_history_founder }
				if = {
					limit = { exists = scope:kehillah_history_origin_county }
					set_variable = { name = kehillah_history_original_origin value = scope:kehillah_history_origin_county }
				}
			}
			else = {
				set_variable = { name = kehillah_history_last_refounded_year value = current_year }
				set_variable = { name = kehillah_history_last_refounder value = scope:kehillah_history_founder }
				if = {
					limit = { exists = scope:kehillah_history_origin_county }
					set_variable = { name = kehillah_history_last_refounding_origin value = scope:kehillah_history_origin_county }
				}
				else = { remove_variable = kehillah_history_last_refounding_origin }
			}
			set_variable = { name = kehillah_history_status value = 1 }
		}
		scope:kehillah_new_community_title = {
			set_variable = { name = kehillah_history_county value = scope:kehillah_founding_county }
			set_variable = { name = kehillah_founded_year value = scope:kehillah_founding_county.var:kehillah_history_original_year }
		}
		debug_log = "kehillah_history: founding/refounding recorded on county"
	}
}

# Scope: the departing leader, BEFORE destroy_title. The reason is derived
# from existing cause markers; the attacker is optional and must be a real
# saved character, never inferred from a hostile realm's current ruler.
kehillah_history_record_loss_effect = {
	save_scope_as = kehillah_history_loss_leader
	if = {
		limit = { exists = domicile.domicile_location.county }
		domicile.domicile_location.county = {
			set_variable = { name = kehillah_history_last_loss_year value = current_year }
			if = { limit = { has_variable = kehillah_history_last_destroyer } remove_variable = kehillah_history_last_destroyer }
			if = {
				limit = { scope:kehillah_history_loss_leader.primary_title = { has_variable = kehillah_pogrom_recent_news } }
				set_variable = { name = kehillah_history_status value = 2 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_mob }
				if = {
					limit = { scope:kehillah_history_loss_leader.primary_title = { has_variable = kehillah_mob_threat } }
					set_variable = { name = kehillah_history_last_destroyer value = scope:kehillah_history_loss_leader.primary_title.var:kehillah_mob_threat }
				}
			}
			else_if = {
				limit = { scope:kehillah_history_loss_leader = { has_character_flag = kehillah_baptized_by_force } }
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_forced_conversion }
			}
			else_if = {
				limit = { scope:kehillah_history_loss_leader = { has_variable = kehillah_exiles_from } }
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_expulsion }
			}
			else = {
				set_variable = { name = kehillah_history_status value = 3 }
				set_variable = { name = kehillah_history_last_loss_cause value = flag:kehillah_history_collapse }
			}
		}
		debug_log = "kehillah_history: loss recorded on county before title teardown"
	}
}
