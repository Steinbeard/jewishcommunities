# =====================================================================
# COMMUNITY NEWS -- 2026-10-12 (docs/spec/v30-community-news.md).
#
# kehillah_notify_region_players_effect. Scope: the Kehillah leader the
# news is about. Sends a feed message to every OTHER player who leads a
# Kehillah in the same minhag region (kehillah_shares_minhag_region_
# trigger, the Bet Din's own "nearby"). A character flagged
# kehillah_news_skip (e.g. a co-judge who was there) is passed over.
#   TYPE   msg_kehillah_community_news or msg_kehillah_community_news_bad
#   TITLE  / DESC  loc keys; they may use [kehillah_news_leader...] and
#          [kehillah_news_title...]
# =====================================================================

kehillah_notify_region_players_effect = {
	save_scope_as = kehillah_news_leader
	primary_title = { save_scope_as = kehillah_news_title }
	every_player = {
		limit = {
			NOT = { this = scope:kehillah_news_leader }
			NOT = { has_character_flag = kehillah_news_skip }
			government_has_flag = government_is_kehillah
			kehillah_shares_minhag_region_trigger = { OTHER = scope:kehillah_news_leader }
		}
		send_interface_message = {
			type = $TYPE$
			title = $TITLE$
			desc = $DESC$
			left_icon = scope:kehillah_news_leader
		}
	}
}
