# =====================================================================
# Host Charter maintenance. See docs/spec/v15-host-charter.md.
#
# kehillah_maintain_host_charter_effect is the single entry point for
# establishing and keeping a Kehillah's charter with its host current.
# Called from four places (common/on_action/kehillah_on_actions.txt's
# kehillah_on_game_start, kehillah_on_title_gain and
# kehillah_quarterly_pulse; common/scripted_effects/
# kehillah_found_community_effects.txt's kehillah_found_community_effect)
# so it must be idempotent and safe to call on a Kehillah that already
# has a correct, current charter -- which is the common case, not the
# exception, since it runs every quarter for every Kehillah in the game.
#
# TICK-DEFERRAL CONSTRAINT, load-bearing, do not "simplify" this into an
# end_tributary immediately followed by start_tributary in the same
# block. Both docs/spec/spike-host-charter-interaction.md section 3
# (end_tributary) and section 10.2 (tributary_contract_set_obligation_
# level) found, live, that a subject-contract write does not commit
# within the effect block that issues it -- it needs at least one further
# tick. The host-mismatch branch below therefore only ever calls
# end_tributary and stops; the NEXT scheduled call to this effect (at
# most one quarter later, via kehillah_quarterly_pulse, which exists
# specifically as the backstop for a host changing hands by conquest --
# spike section 5 point 5) sees is_tributary = no and starts the new
# charter fresh. This trades a possible one-quarter lag after a host
# changes for never risking the same-tick failure mode the spike found.
# =====================================================================

kehillah_maintain_host_charter_effect = {
	if = {
		limit = { NOT = { kehillah_host_charter_host_available_trigger = yes } }
		debug_log = "kehillah_maintain_host_charter_effect: no host resolvable yet, skipping"
	}
	else = {
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_host }

		if = {
			limit = {
				is_tributary = yes
				# A current host may legitimately use the legacy V15 group or
				# either V16 regional fresh-charter group. Treat all three as
				# stable: otherwise the quarterly maintenance pass would end a
				# perfectly valid regional charter every time it ran.
				OR = {
					has_subject_contract_group = kehillah_host_charter
					has_subject_contract_group = kehillah_christian_host_charter_prototype
					has_subject_contract_group = kehillah_muslim_host_charter_prototype
				}
				suzerain = scope:kehillah_host
			}
			# Already correct -- the common case on every quarterly pulse
			# and every ordinary succession. Nothing to do.
		}
		else_if = {
			limit = { is_tributary = yes }
			# Wrong host (the county changed hands) or, defensively, some
			# other contract group entirely. End now; see the tick-
			# deferral note above for why this does not also start_tributary
			# in the same block. Remember the transition on the community
			# title: after the later fresh contract has committed, its report
			# can correctly say that a new liege has replaced the old charter
			# instead of presenting it as an ordinary first settlement.
			# .0004 clears this marker on acknowledgement.
			primary_title = {
				set_variable = { name = kehillah_charter_suzerain_changed value = 1 }
			}
			debug_log = "kehillah_maintain_host_charter_effect: host mismatch, ending stale charter"
			end_tributary = yes
		}
		else = {
			kehillah_start_fresh_host_charter_effect = yes
		}
	}
}

# ---------------------------------------------------------------------
# Scope: a Kehillah leader, with scope:kehillah_host saved by the
# maintenance effect above. This is called only when no existing
# tributary contract exists. It is therefore the one safe location for
# regional selection and policy-derived defaults: it does not touch a
# saved contract, and every additional community resolves the SAME
# primary-title policy from its own host at this moment of creation.
#
# Do not fold this back into the existing-charter branch. A community
# moving across realms follows the documented end -> later start path;
# on that later start it rightly receives the new host's fresh offer.
# ---------------------------------------------------------------------
kehillah_start_fresh_host_charter_effect = {
	if = {
		limit = {
			scope:kehillah_host = { religion = religion:christianity_religion }
		}
		start_tributary = {
			contract_group = kehillah_christian_host_charter_prototype
			suzerain = scope:kehillah_host
		}
		trigger_event = { id = kehillah_settlement_conditions.0002 days = 1 }
		debug_log = "kehillah_start_fresh_host_charter_effect: Christian policy charter established"
	}
	else_if = {
		limit = {
			scope:kehillah_host = { religion = religion:islam_religion }
		}
		start_tributary = {
			contract_group = kehillah_muslim_host_charter_prototype
			suzerain = scope:kehillah_host
		}
		trigger_event = { id = kehillah_settlement_conditions.0002 days = 1 }
		debug_log = "kehillah_start_fresh_host_charter_effect: Muslim policy charter established"
	}
	else = {
		# Universal policy still gates founding in every realm. Only the
		# regional contract vocabulary awaits further historical work.
		start_tributary = {
			contract_group = kehillah_host_charter
			suzerain = scope:kehillah_host
		}
		debug_log = "kehillah_start_fresh_host_charter_effect: legacy fallback charter established"
	}
}

# Runs only after start_tributary has committed. Contract is_valid cannot
# resolve the host realm while CK3 creates a tributary, so policy defaults
# are written here from the stable, concrete contract API instead.
kehillah_apply_fresh_charter_policy_effect = {
	domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_policy_host }
	if = {
		limit = { has_subject_contract_group = kehillah_christian_host_charter_prototype }
		if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes } }
			tributary_contract_set_obligation_level = { type = kehillah_christian_quarter_building_rights level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_community_security level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_jurisdiction level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 0 }
		}
		else_if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_discouraged_trigger = yes } }
			tributary_contract_set_obligation_level = { type = kehillah_christian_quarter_building_rights level = 1 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_jurisdiction level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 1 }
		}
		else_if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_banned_trigger = yes } }
			# Restrictive defaults already match Banned; write them explicitly
			# to make a later policy flip deterministic.
			tributary_contract_set_obligation_level = { type = kehillah_christian_quarter_building_rights level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_jurisdiction level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 2 }
		}
		else = {
			tributary_contract_set_obligation_level = { type = kehillah_christian_quarter_building_rights level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_jurisdiction level = 1 }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 0 }
		}
	}
	else_if = {
		limit = { has_subject_contract_group = kehillah_muslim_host_charter_prototype }
		if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes } }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_quarter_building_rights level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_community_security level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_jurisdiction level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_market_access level = 2 }
		}
		else_if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_discouraged_trigger = yes } }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_quarter_building_rights level = 1 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_jurisdiction level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_market_access level = 0 }
		}
		else_if = {
			limit = { scope:kehillah_policy_host.primary_title = { kehillah_jewish_settlement_policy_banned_trigger = yes } }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_quarter_building_rights level = 2 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_jurisdiction level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_market_access level = 0 }
		}
		else = {
			tributary_contract_set_obligation_level = { type = kehillah_muslim_quarter_building_rights level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_community_security level = 0 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_jurisdiction level = 1 }
			tributary_contract_set_obligation_level = { type = kehillah_muslim_market_access level = 1 }
		}
	}
}

# ---------------------------------------------------------------------
# V16 live-charter cache writer.
#
# Scope: a Kehillah leader / tributary. This is deliberately the ONLY
# ordinary-script reader of subject_contract_has_flag. Tributary flags do
# not work through vassal_contract_has_flag, while putting the correct
# subject trigger into tooltip, bootstrap, or pillar code stalled a fresh
# load during Initializing Game. This effect runs only from a later hidden
# event after a fresh writer has committed, and then quarterly before pillar
# convergence. It copies the five package-neutral rungs onto the enduring
# community title, which every ordinary consumer reads safely.
#
# Cache is cleared before every reconciliation. Thus an old regional cache
# cannot leak across a host change into V15/future fallback charters.
# Christian charters write construction/security/jurisdiction/text; Muslim
# charters write construction/security/jurisdiction/market. A concrete
# contract always supplies one flag per row; restrictive defaults are used
# defensively if a malformed contract has none.
# ---------------------------------------------------------------------
kehillah_refresh_charter_state_cache_effect = {
	primary_title = {
		remove_variable = kehillah_charter_cache_construction
		remove_variable = kehillah_charter_cache_security
		remove_variable = kehillah_charter_cache_jurisdiction
		remove_variable = kehillah_charter_cache_text_study
		remove_variable = kehillah_charter_cache_market_access
	}

	if = {
		limit = { has_subject_contract_group = kehillah_christian_host_charter_prototype }
		if = {
			limit = { subject_contract_has_flag = kehillah_christian_building_free }
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 0 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_christian_building_permission }
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 2 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_christian_security_recognized_watch }
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_christian_security_licensed_watch }
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 0 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_christian_jurisdiction_bet_din_discipline }
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_christian_jurisdiction_royal_appeal }
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 0 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_christian_texts_unrestricted }
			primary_title = { set_variable = { name = kehillah_charter_cache_text_study value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_christian_texts_licensed }
			primary_title = { set_variable = { name = kehillah_charter_cache_text_study value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_text_study value = 0 } }
		}
	}
	else_if = {
		limit = { has_subject_contract_group = kehillah_muslim_host_charter_prototype }
		if = {
			limit = { subject_contract_has_flag = kehillah_muslim_building_free }
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 0 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_muslim_building_authorized }
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_construction value = 2 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_muslim_security_host_protection }
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_muslim_security_licensed_watch }
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_security value = 0 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_muslim_jurisdiction_bet_din_discipline }
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_muslim_jurisdiction_communal_arbitration }
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_jurisdiction value = 0 } }
		}

		if = {
			limit = { subject_contract_has_flag = kehillah_muslim_market_chartered }
			primary_title = { set_variable = { name = kehillah_charter_cache_market_access value = 2 } }
		}
		else_if = {
			limit = { subject_contract_has_flag = kehillah_muslim_market_regional }
			primary_title = { set_variable = { name = kehillah_charter_cache_market_access value = 1 } }
		}
		else = {
			primary_title = { set_variable = { name = kehillah_charter_cache_market_access value = 0 } }
		}
	}
}

# Queues player-facing notification only after .0003 has copied the
# committed contract into the safe title cache. Every community leader gets
# its own notice (only a player-controlled leader will see it), while a host
# ruler gets one delayed digest for every charter that resolved on that day.
# The temporary character flag is a debounce: it is cleared, along with the
# count, by the ruler event's sole acknowledgement option.
kehillah_queue_charter_notification_effect = {
	trigger_event = { id = kehillah_settlement_conditions.0004 days = 1 }

	scope:liege = {
		if = {
			limit = { has_character_flag = kehillah_charter_notice_pending }
			change_variable = { name = kehillah_charter_notice_count add = 1 }
		}
		else = {
			set_variable = { name = kehillah_charter_notice_count value = 1 }
			add_character_flag = kehillah_charter_notice_pending
			trigger_event = { id = kehillah_settlement_conditions.0005 days = 1 }
		}
	}
}
