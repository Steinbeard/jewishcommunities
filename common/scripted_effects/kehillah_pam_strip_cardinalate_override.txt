# =====================================================================
# Overrides vanilla's pam_strip_illegal_cardinalate_effect
# (common/scripted_effects/11_dlc_pam_scripted_effects.txt, CK3 1.20.1).
#
# V46 (docs/spec/v46-the-sanhedrin.md). Seats on the Sanhedrin are clerical
# elector titles, as cardinalates are, and they are held by Kehillah
# leaders. Vanilla strips an elector title from any ruler who is not a
# theocracy or holy order, with no check of faith, from three places:
# on_title_gain, every government change, and the yearly playable pulse.
# Without this exemption every seat would be destroyed within a year.
#
# Copied verbatim; the one change is the third NOR branch. Keep in sync
# with the installed game after updates.
# =====================================================================
pam_strip_illegal_cardinalate_effect = {
	if = {
		limit = {
			is_ruler = yes
			NOR = {
				government_has_flag = government_is_theocracy
				government_has_flag = government_is_holy_order
				government_has_flag = government_is_kehillah # V46: Sanhedrin seats
			}
			exists = clerical_elector_title
		}
		pam_destroy_cardinal_title_effect = { CARDINALATE = clerical_elector_title }
	}
}
