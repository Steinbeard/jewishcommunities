# The petition uses the title-change wrapper from kehillah_step_down_effect.
# A separate office grant follows the community transfer; it cannot become
# primary and cannot convert the claimant into a theocrat.
kehillah_press_exilarch_claim_effect = {
	save_scope_as = kehillah_exilarch_claimant
	primary_title = { save_scope_as = kehillah_exilarch_old_camp }
	title:d_kehillah_baghdad.holder = { save_scope_as = kehillah_exilarch_caretaker }
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = kehillah_exilarch_baghdad_change
		add_claim_on_loss = no
	}
	title:d_kehillah_baghdad = {
		change_title_holder = {
			holder = scope:kehillah_exilarch_claimant
			change = scope:kehillah_exilarch_baghdad_change
			government_base = scope:kehillah_exilarch_caretaker
		}
	}
	resolve_title_and_vassal_change = scope:kehillah_exilarch_baghdad_change
	set_primary_title_to = title:d_kehillah_baghdad
	if = {
		limit = { has_title = scope:kehillah_exilarch_old_camp }
		destroy_title = scope:kehillah_exilarch_old_camp
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = kehillah_exilarch_office_change
		add_claim_on_loss = no
	}
	title:d_kehillah_exilarchate = {
		change_title_holder = {
			holder = scope:kehillah_exilarch_claimant
			change = scope:kehillah_exilarch_office_change
		}
	}
	resolve_title_and_vassal_change = scope:kehillah_exilarch_office_change
	set_primary_title_to = title:d_kehillah_baghdad
	kehillah_update_leader_flavor_effect = yes
	if = {
		limit = { character:9003160 ?= { is_alive = yes is_courtier_of = root } }
		assign_councillor_type = {
			type = councillor_court_chaplain
			target = character:9003160
			remove_existing_councillor = yes
		}
	}
	kehillah_refresh_map_view_for_players_effect = yes
	debug_log = "KEHILLAH_EXILARCH: Baghdad and the separate Exilarchate granted to David; old camp removed"
	debug_log_scopes = yes
}
