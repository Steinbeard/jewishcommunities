# The Exilarchate is a lineage office: recognition changes neither the
# holder's community nor the host realm's Nagid and court appointments.
kehillah_grant_exilarchate_effect = {
	if = {
		limit = {
			is_jewish_character_trigger = yes
			has_trait = ben_david
			government_has_flag = government_is_kehillah
			primary_title = { is_kehillah_title_trigger = yes }
			NOT = { exists = title:d_kehillah_exilarchate.holder }
		}
		save_scope_as = kehillah_exilarch_claimant
		primary_title = { save_scope_as = kehillah_exilarch_community }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = kehillah_exilarch_office_change
			add_claim_on_loss = no
		}
		title:d_kehillah_exilarchate = {
			change_title_holder = {
				holder = scope:kehillah_exilarch_claimant
				change = scope:kehillah_exilarch_office_change
			}
		}
		resolve_title_and_vassal_change = scope:kehillah_exilarch_office_change
		set_primary_title_to = scope:kehillah_exilarch_community
		kehillah_update_leader_flavor_effect = yes
		debug_log = "KEHILLAH_EXILARCH: vacant office recognized for a Ben David community leader"
		debug_log_scopes = yes
	}
}
