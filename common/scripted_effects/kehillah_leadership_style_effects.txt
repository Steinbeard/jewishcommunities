kehillah_leadership_choose_scholarly_effect = {
 if = {
  limit = {
   kehillah_leadership_style_available_trigger = yes
   NOT = { primary_title = { has_variable = kehillah_leadership_style kehillah_leadership_is_scholarly_trigger = yes } }
  }
  primary_title = {
   set_variable = { name = kehillah_leadership_style value = 1 }
   set_variable = { name = kehillah_leadership_style_lock value = yes years = 10 }
   set_variable = { name = kehillah_leadership_style_until value = current_year }
   change_variable = { name = kehillah_leadership_style_until add = 10 }
  }
 }
}

# 2026-11-20 (pillars spec R2). Scope: a Kehillah leader. Exactly one style
# modifier, matching the community's chosen style (none before a style is
# chosen). Refreshed quarterly (kehillah_quarter_on_actions.txt).
kehillah_leadership_style_apply_effect = {
	if = { limit = { has_character_modifier = kehillah_style_scholarly_modifier } remove_character_modifier = kehillah_style_scholarly_modifier }
	if = { limit = { has_character_modifier = kehillah_style_political_modifier } remove_character_modifier = kehillah_style_political_modifier }
	if = { limit = { has_character_modifier = kehillah_style_commercial_modifier } remove_character_modifier = kehillah_style_commercial_modifier }
	if = {
		limit = { government_has_flag = government_is_kehillah }
		primary_title ?= { save_temporary_scope_as = kehillah_style_seat }
		if = {
			limit = { scope:kehillah_style_seat ?= { kehillah_leadership_mandate_is_greatness_trigger = yes } }
			add_character_modifier = kehillah_style_scholarly_modifier
		}
		else_if = {
			limit = { scope:kehillah_style_seat ?= { kehillah_leadership_mandate_is_stability_trigger = yes } }
			add_character_modifier = kehillah_style_political_modifier
		}
		else_if = {
			limit = { scope:kehillah_style_seat ?= { kehillah_leadership_mandate_is_prosperity_trigger = yes } }
			add_character_modifier = kehillah_style_commercial_modifier
		}
	}
}
