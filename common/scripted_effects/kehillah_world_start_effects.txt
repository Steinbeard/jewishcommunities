# First-pass multi-bookmark bootstrap.

# Scope: a historical community title. Unheld titles are deliberately
# ignored, so the global registry and ledger describe the chosen bookmark
# rather than every title that exists in the database.
kehillah_register_start_community_effect = {
	if = {
		limit = { exists = holder }
		add_to_global_variable_list = {
			name = kehillah_registered_communities
			target = this
		}
		holder = { kehillah_setup_world_start_effect = yes }
	}
}

# Scope: a leader not handled by one of the richer named 1066 setup
# effects. This intentionally supplies only the common playable floor.
# Buildings, officers, host conditions and local development then flow
# through the same visible baseline contributors as every other Kehillah.
kehillah_setup_world_start_effect = {
	if = {
		limit = {
			exists = domicile
			NOT = { has_character_flag = kehillah_start_configured }
		}
		add_character_flag = kehillah_start_configured
		save_scope_as = kehillah_leader
		add_gold = 60
		kehillah_seed_community_effect = yes
		kehillah_record_quarter_effect = yes
		kehillah_update_leader_flavor_effect = yes
		if = {
			limit = { primary_title = title:d_kehillah_kairouan }
			kehillah_seed_yeshiva_effect = { FAME = kehillah_yeshiva_fame_seed_kairouan_867 }
		}
		if = {
			limit = { primary_title = title:d_kehillah_fustat }
			kehillah_seed_yeshiva_effect = { FAME = kehillah_yeshiva_fame_seed_fustat }
		}
	}
}
