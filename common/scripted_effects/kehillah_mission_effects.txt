# Scope: a Kehillah leader, quarterly. Disputation or sermons may come.
kehillah_mission_check_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = primary_title
			exists = domicile.domicile_location.county.holder.top_liege
			domicile.domicile_location.county.holder.top_liege = { NOT = { is_jewish_character_trigger = yes } }
		}
		if = {
			limit = {
				NOT = { primary_title = { has_variable = kehillah_disputation_cooldown } }
				kehillah_disputation_chance_value > 0
			}
			random = {
				chance = kehillah_disputation_chance_value
				primary_title = {
					set_variable = { name = kehillah_disputation_cooldown value = yes years = kehillah_disputation_cooldown_years }
				}
				trigger_event = kehillah_mission.0001
			}
		}
		if = {
			limit = {
				NOT = { primary_title = { has_variable = kehillah_sermon_cooldown } }
				kehillah_sermon_chance_value > 0
			}
			random = {
				chance = kehillah_sermon_chance_value
				primary_title = {
					set_variable = { name = kehillah_sermon_cooldown value = yes years = kehillah_sermon_cooldown_years }
				}
				trigger_event = kehillah_mission.0010
			}
		}
	}
}

# Scope: a Kehillah leader with a Christian charter. The Talmud on trial:
# the text term one step stricter (cache text_study 2 = unrestricted).
kehillah_charter_censor_text_effect = {
	if = {
		limit = { has_subject_contract_group = kehillah_christian_host_charter_prototype }
		if = {
			limit = { primary_title = { var:kehillah_charter_cache_text_study ?= 2 } }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 1 }
		}
		else_if = {
			limit = { primary_title = { var:kehillah_charter_cache_text_study ?= 1 } }
			tributary_contract_set_obligation_level = { type = kehillah_christian_text_censorship level = 2 }
		}
		primary_title = { set_variable = { name = kehillah_charter_forced_revision value = yes years = 1 } }
	}
}

# Scope: a Kehillah leader; scope:kehillah_mission_host saved. A
# courtier listens too well to the friars: he converts and goes to the
# lord's court.
kehillah_mission_conversion_effect = {
	random_courtier = {
		limit = {
			is_adult = yes
			is_jewish_character_trigger = yes
			NOT = { root.player_heir ?= this }
			NOT = { is_spouse_of = root }
			NOT = { is_councillor = yes }
		}
		save_scope_as = kehillah_mission_convert
		set_character_faith = scope:kehillah_mission_host.faith
		scope:kehillah_mission_host = { add_courtier = scope:kehillah_mission_convert }
	}
}
