# V46 the Sanhedrin (docs/spec/v46-the-sanhedrin.md).

# ---------------------------------------------------------------------
# Restore the Sanhedrin. Scope: the founder, from
# kehillah_restore_sanhedrin_decision. The founder keeps his community as
# primary title; the Sanhedrin becomes the faith's head-of-faith title and
# passes by its members' election (kehillah_sanhedrin_succession_law).
# The head-of-faith doctrine group is main_rite only
# (00_doctrine_group_types.txt), so the main rite's doctrine is the
# faith's, Minhag Ashkenaz included.
# ---------------------------------------------------------------------
kehillah_restore_sanhedrin_effect = {
	save_scope_as = kehillah_sanhedrin_founder
	primary_title = { save_scope_as = kehillah_founder_primary }

	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:d_kehillah_sanhedrin = {
		change_title_holder = {
			holder = scope:kehillah_sanhedrin_founder
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	set_primary_title_to = scope:kehillah_founder_primary
	kehillah_sanhedrin_sit_with_nasi_effect = yes

	title:d_kehillah_sanhedrin = {
		add_title_law = kehillah_sanhedrin_succession_law
	}
	# ORDER (spec section 9): doctrines, then government, then the head.
	# A landless Nasi moves onto kehillah_nasi_government BEFORE he
	# becomes head, so the engine's forced theocracy finds him already on
	# the faith's theocracy government and has nothing to change. Setting
	# the head also moves the faith's main rite to his rite, so the
	# doctrines are applied to every rite, before and after.
	kehillah_sanhedrin_apply_head_doctrines_effect = yes
	kehillah_reconcile_nasi_government_effect = yes
	faith:rabbinic_faith = {
		set_religious_head_title = title:d_kehillah_sanhedrin
	}
	kehillah_sanhedrin_apply_head_doctrines_effect = yes
	kehillah_reconcile_nasi_government_effect = yes
	# The engine settles governments after effects return; check again.
	trigger_event = { id = kehillah_synod.0003 days = 1 }
	set_global_variable = { name = kehillah_sanhedrin_formed value = yes }
	if = {
		limit = { government_has_flag = government_is_kehillah }
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_sanhedrin_founding_greatness }
		kehillah_update_leader_flavor_effect = yes
	}
	kehillah_sanhedrin_refresh_seats_effect = yes
}

# ---------------------------------------------------------------------
# The head-of-faith doctrines for the scoped Nasi, on every rite.
#   A landless Nasi (a community rabbi): a SPIRITUAL head, the Pope's
#   model, with Judaism's vanilla theocratic clergy. He sits on
#   kehillah_nasi_government, the faith's theocracy government.
#   A landed Nasi (a Jewish king): a TEMPORAL head, the Caliph's model,
#   with lay clergy, which doctrine_temporal_head requires. A landed
#   ruler keeps his own government. Not yet live-tested.
# Run again for every new Nasi, since the next one may be the other kind.
# ---------------------------------------------------------------------
kehillah_sanhedrin_apply_head_doctrines_effect = {
	if = {
		limit = { is_landed = yes }
		kehillah_sanhedrin_set_rite_doctrines_effect = { HEAD = doctrine_temporal_head CLERGY = doctrine_theocracy_lay_clergy }
	}
	else = {
		kehillah_sanhedrin_set_rite_doctrines_effect = { HEAD = doctrine_spiritual_head CLERGY = doctrine_theocracy_temporal }
	}
}

# ---------------------------------------------------------------------
# THE GOVERNMENT TRANSITIONS (spec section 9). Scope: any character. One
# place that decides, run wherever Nasi status can change:
#   - restoration (above), before and after the head is set;
#   - kehillah_sanhedrin_on_title_gain: the new Nasi, and the previous
#     holder if he is still alive;
#   - kehillah_nasi_community_on_title_gain: whoever takes a community,
#     typically the successor of a Nasi, who inherits the Nasi's
#     government along with the title;
#   - kehillah_synod.0003, a day after each of those;
#   - the yearly pulse, for every community leader.
#
# Rules:
#   1. A landless Nasi who leads a community -> kehillah_nasi_government.
#   2. On kehillah_nasi_government but not Nasi -> kehillah_government.
#   3. Leads a community, not Nasi, on some other government (a forced
#      generic theocracy, say) -> kehillah_government.
# change_government needs a domicile already there (kehillah_on_title_gain
# notes, 2026-09); both governments share kehillah_quarter, so a switch
# between them should keep it. kehillah_restore_quarter_effect runs after
# each switch to rebuild from the recorded buildings if it did not.
# Each switch logs a KEHILLAH_NASI_GOV line for live verification.
# ---------------------------------------------------------------------
kehillah_reconcile_nasi_government_effect = {
	if = {
		limit = {
			is_ruler = yes
			is_landed = no
			kehillah_is_sanhedrin_head_trigger = yes
			any_held_title = { is_kehillah_title_trigger = yes }
			NOT = { has_government = kehillah_nasi_government }
		}
		change_government = kehillah_nasi_government
		kehillah_nasi_government_restore_quarter_effect = yes
		if = {
			limit = { has_government = kehillah_nasi_government }
			debug_log = "KEHILLAH_NASI_GOV: to Nasi government, accepted"
		}
		else = { debug_log = "KEHILLAH_NASI_GOV: to Nasi government, REFUSED" }
		debug_log_scopes = yes
	}
	else_if = {
		limit = {
			is_ruler = yes
			NOT = { kehillah_is_sanhedrin_head_trigger = yes }
			any_held_title = { is_kehillah_title_trigger = yes }
			OR = {
				has_government = kehillah_nasi_government
				NOT = { government_has_flag = government_is_kehillah }
			}
		}
		change_government = kehillah_government
		kehillah_nasi_government_restore_quarter_effect = yes
		if = {
			limit = { has_government = kehillah_government }
			debug_log = "KEHILLAH_NASI_GOV: back to Kehillah government, accepted"
		}
		else = { debug_log = "KEHILLAH_NASI_GOV: back to Kehillah government, REFUSED" }
		debug_log_scopes = yes
	}
}

# kehillah_restore_quarter_effect reads scope:kq_title.
kehillah_nasi_government_restore_quarter_effect = {
	random_held_title = {
		limit = { is_kehillah_title_trigger = yes }
		save_scope_as = kq_title
	}
	if = {
		limit = { exists = domicile }
		kehillah_restore_quarter_effect = yes
	}
	else = { debug_log = "KEHILLAH_NASI_GOV: no domicile after the switch" }
}

# ---------------------------------------------------------------------
# Head-of-faith and clergy doctrines on every Rabbinic rite. Both groups
# act through the main rite only, but the main rite follows the head of
# faith's own rite, so every rite a Nasi might belong to carries them.
# HEAD: doctrine_temporal_head / doctrine_no_head. CLERGY:
# doctrine_theocracy_lay_clergy / doctrine_theocracy_temporal.
# ---------------------------------------------------------------------
kehillah_sanhedrin_set_rite_doctrines_effect = {
	faith:rabbinic_faith.main_rite ?= {
		if = { limit = { NOT = { rite_has_doctrine = $CLERGY$ } } change_rite_doctrine = $CLERGY$ }
		if = { limit = { NOT = { rite_has_doctrine = $HEAD$ } } change_rite_doctrine = $HEAD$ }
	}
	rite:rabbinism ?= {
		if = { limit = { NOT = { rite_has_doctrine = $CLERGY$ } } change_rite_doctrine = $CLERGY$ }
		if = { limit = { NOT = { rite_has_doctrine = $HEAD$ } } change_rite_doctrine = $HEAD$ }
	}
	rite:merkabah ?= {
		if = { limit = { NOT = { rite_has_doctrine = $CLERGY$ } } change_rite_doctrine = $CLERGY$ }
		if = { limit = { NOT = { rite_has_doctrine = $HEAD$ } } change_rite_doctrine = $HEAD$ }
	}
	# Guarded by the founding flag, not ?=: rite:minhag_ashkenaz errors
	# before the rite exists (see kehillah_found_minhag_ashkenaz_effect).
	if = {
		limit = { exists = global_var:kehillah_minhag_ashkenaz_founded }
		rite:minhag_ashkenaz = {
			if = { limit = { NOT = { rite_has_doctrine = $CLERGY$ } } change_rite_doctrine = $CLERGY$ }
			if = { limit = { NOT = { rite_has_doctrine = $HEAD$ } } change_rite_doctrine = $HEAD$ }
		}
	}
}

# ---------------------------------------------------------------------
# The Sanhedrin sits wherever its Nasi leads a community (Daniel,
# 2026-10-19): restored in Toledo, it sits in Toledo, and it moves when a
# Nasi from elsewhere is elected, as the ancient court moved from Yavne to
# Usha to Tiberias. Scope: the Nasi. Saves scope:kehillah_sanhedrin_city.
# ---------------------------------------------------------------------
kehillah_sanhedrin_sit_with_nasi_effect = {
	if = {
		limit = { exists = primary_title.title_capital_county }
		primary_title.title_capital_county = { save_scope_as = kehillah_sanhedrin_city }
	}
	else_if = {
		limit = { exists = capital_county }
		capital_county = { save_scope_as = kehillah_sanhedrin_city }
	}
	else_if = {
		limit = { exists = location.county }
		location.county = { save_scope_as = kehillah_sanhedrin_city }
	}
	if = {
		limit = { exists = scope:kehillah_sanhedrin_city }
		title:d_kehillah_sanhedrin = {
			set_capital_county = scope:kehillah_sanhedrin_city
			set_variable = { name = kehillah_sanhedrin_city value = scope:kehillah_sanhedrin_city }
		}
	}
}

# ---------------------------------------------------------------------
# Give the scoped character a seat. Vanilla's cardinalate pattern
# (11_dlc_pam_scripted_effects.txt): a dynamic, landless duchy destroyed on
# succession, marked as a clerical elector of Rabbinic Judaism.
#
# ORDER MATTERS: the title is marked as an elector only AFTER the transfer
# resolves. on_title_gain (title_on_actions.txt) treats a gained elector
# title as a cardinalate, flagging the holder a cardinal and enrolling him
# in the Christian Church situation. Marking it afterwards skips that.
# ---------------------------------------------------------------------
kehillah_sanhedrin_grant_seat_effect = {
	save_scope_as = kehillah_seat_recipient
	primary_title = { save_scope_as = kehillah_seat_primary }
	create_dynamic_title = {
		tier = duchy
		name = KEHILLAH_SANHEDRIN_SEAT_NAME
		adj = KEHILLAH_SANHEDRIN_SEAT_ADJ
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:new_title ?= {
		save_scope_as = kehillah_new_seat
		set_variable = kehillah_sanhedrin_seat
		# Capital first: set_landless_title errors on a title with no
		# preferred capital (live test 2026-10-19), and PAM sets it first.
		set_capital_county = title:d_kehillah_sanhedrin.title_capital_county
		set_coa = scope:kehillah_seat_primary
		set_color_from_title = scope:kehillah_seat_primary
		set_landless_title = yes
		set_destroy_on_succession = yes
		set_no_automatic_claims = yes
		set_definitive_form = yes
		set_can_be_named_after_dynasty = no
		set_can_use_nomadic_naming = no
		change_title_holder = {
			holder = scope:kehillah_seat_recipient
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	scope:kehillah_new_seat ?= { add_clerical_elector = faith:rabbinic_faith }
	set_primary_title_to = scope:kehillah_seat_primary
}

# ---------------------------------------------------------------------
# Seat every eligible rabbi who leads a community, up to the cap. Run at
# restoration and yearly while the Sanhedrin exists: seats are destroyed
# when their holders die, and vanilla also destroys one when its holder
# changes rite (religion_on_actions.txt).
# ---------------------------------------------------------------------
kehillah_sanhedrin_refresh_seats_effect = {
	if = {
		limit = { exists = title:d_kehillah_sanhedrin.holder }
		set_global_variable = { name = kehillah_sanhedrin_seat_count value = 0 }
		faith:rabbinic_faith = {
			every_clerical_elector_title = {
				limit = { has_variable = kehillah_sanhedrin_seat }
				change_global_variable = { name = kehillah_sanhedrin_seat_count add = 1 }
			}
		}
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				holder ?= { kehillah_sanhedrin_seat_eligible_trigger = yes }
			}
			if = {
				limit = { global_var:kehillah_sanhedrin_seat_count < kehillah_sanhedrin_seat_cap }
				holder = { kehillah_sanhedrin_grant_seat_effect = yes }
				change_global_variable = { name = kehillah_sanhedrin_seat_count add = 1 }
			}
		}
		# Landed Jewish rulers who are rabbis (a king of Israel, say).
		every_ruler = {
			limit = {
				is_landed = yes
				faith = faith:rabbinic_faith
				has_trait = kehillah_rabbi_trait
				kehillah_sanhedrin_seat_eligible_trigger = yes
			}
			if = {
				limit = { global_var:kehillah_sanhedrin_seat_count < kehillah_sanhedrin_seat_cap }
				kehillah_sanhedrin_grant_seat_effect = yes
				change_global_variable = { name = kehillah_sanhedrin_seat_count add = 1 }
			}
		}
		remove_global_variable = kehillah_sanhedrin_seat_count
	}
}

# ---------------------------------------------------------------------
# The Sanhedrin lapses when it has no valid successor (the title is
# destroy_if_invalid_heir). Restore the faith to having no head, dissolve
# the seats, and let a later rabbi restore it again.
# ---------------------------------------------------------------------
kehillah_sanhedrin_lapse_check_effect = {
	if = {
		limit = {
			exists = global_var:kehillah_sanhedrin_formed
			NOT = { exists = title:d_kehillah_sanhedrin.holder }
		}
		remove_global_variable = kehillah_sanhedrin_formed
		kehillah_sanhedrin_set_rite_doctrines_effect = { HEAD = doctrine_no_head CLERGY = doctrine_theocracy_temporal }
		faith:rabbinic_faith = {
			every_clerical_elector_title = {
				limit = { has_variable = kehillah_sanhedrin_seat }
				add_to_list = kehillah_lapsed_seats
			}
		}
		if = {
			limit = { faith:rabbinic_faith = { any_clerical_elector_title = { has_variable = kehillah_sanhedrin_seat } } }
			every_in_list = {
				list = kehillah_lapsed_seats
				holder ?= { destroy_title = prev }
			}
		}
		every_player = {
			limit = { faith = faith:rabbinic_faith }
			send_interface_message = {
				type = msg_kehillah_community_news
				title = kehillah_news_sanhedrin_lapsed_t
				desc = kehillah_news_sanhedrin_lapsed_d
			}
		}
	}
}
