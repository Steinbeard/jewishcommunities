# Scope: a Kehillah leader. Quarterly. A lending community marks its county
# holder and his top liege as served by Jewish banking for a year.
kehillah_banking_services_refresh_effect = {
	if = {
		limit = {
			kehillah_community_is_lending_trigger = yes
			exists = domicile.domicile_location.county.holder
		}
		domicile.domicile_location.county.holder = {
			set_variable = { name = kehillah_banking_services value = yes years = 1 }
			top_liege ?= {
				set_variable = { name = kehillah_banking_services value = yes years = 1 }
			}
		}
	}
}

# Yearly, once for the world. Counts the non-Jewish lending houses based in
# each ruler's realm (kehillah_rival_credit on the top liege). Cleared and
# recounted each year; one every_ruler pass a year.
kehillah_rival_credit_yearly_effect = {
	every_in_global_list = {
		variable = kehillah_rival_credit_hosts
		if = {
			limit = { has_variable = kehillah_rival_credit }
			remove_variable = kehillah_rival_credit
		}
	}
	clear_global_variable_list = kehillah_rival_credit_hosts
	every_ruler = {
		limit = { kehillah_is_rival_lender_trigger = yes }
		save_temporary_scope_as = kehillah_rival_lender
		# Where the house is based: a merchant company's domicile, else
		# the holy order's (or any ruler's) capital county.
		if = {
			limit = { exists = domicile.domicile_location.county.holder.top_liege }
			domicile.domicile_location.county.holder.top_liege = { kehillah_rival_credit_add_effect = yes }
		}
		else_if = {
			limit = { exists = capital_county.holder.top_liege }
			capital_county.holder.top_liege = { kehillah_rival_credit_add_effect = yes }
		}
	}
	debug_log = "kehillah_rival_credit_yearly_effect: recounted non-Jewish lending houses"
}

# Scope: the top liege of a rival lender's base.
kehillah_rival_credit_add_effect = {
	if = {
		limit = { NOT = { this = scope:kehillah_rival_lender } }
		if = {
			limit = { has_variable = kehillah_rival_credit }
			change_variable = { name = kehillah_rival_credit add = 1 }
		}
		else = {
			set_variable = { name = kehillah_rival_credit value = 1 }
			add_to_global_variable_list = { name = kehillah_rival_credit_hosts target = this }
		}
	}
}
