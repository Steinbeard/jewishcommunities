# Root: community title. Since 2026-11-01 the situation owns the Agitation clock (the race);
# events that decide a transition outright still set the title's stage, and the situation follows.
kehillah_crisis_situation_close_effect = {
 save_scope_as = kehillah_crisis_close_title
 if = {
  limit = { exists = var:kehillah_crisis_situation }
  var:kehillah_crisis_situation = {
   if = {
    limit = { situation_type = kehillah_community_crisis var:kehillah_crisis_title ?= scope:kehillah_crisis_close_title var:kehillah_crisis_epoch = scope:kehillah_crisis_close_title.var:kehillah_crisis_epoch }
    end_situation = yes
   }
  }
  remove_variable = kehillah_crisis_situation
 }
 kehillah_crisis_race_clear_mirror_effect = yes
 set_variable = { name = kehillah_crisis_situation_done value = yes }
}

kehillah_crisis_situation_sync_inner_effect = {
 save_scope_as = kehillah_crisis_situation_title
 if = {
  limit = { NOT = { has_variable = kehillah_crisis_intro_pending } NOT = { has_variable = kehillah_crisis_situation_done } }
 if = {
  limit = { OR = { NOT = { exists = var:kehillah_crisis_situation } var:kehillah_crisis_situation = { NOT = { situation_type = kehillah_community_crisis var:kehillah_crisis_title ?= root var:kehillah_crisis_epoch = root.var:kehillah_crisis_epoch } } } }
  remove_variable = kehillah_crisis_situation
  if = {
   limit = { exists = holder.domicile.domicile_location.county }
   if = {
    limit = { var:kehillah_crisis_stage = 2 }
    start_situation = { type = kehillah_community_crisis start_phase = kehillah_crisis_armed save_scope_as = kehillah_crisis_new_situation }
   }
   else_if = {
    limit = { var:kehillah_crisis_stage = 3 }
    start_situation = { type = kehillah_community_crisis start_phase = kehillah_crisis_judgement save_scope_as = kehillah_crisis_new_situation }
   }
   else_if = {
    limit = { var:kehillah_crisis_stage = 4 }
    start_situation = { type = kehillah_community_crisis start_phase = kehillah_crisis_recovery save_scope_as = kehillah_crisis_new_situation }
   }
   else = { start_situation = { type = kehillah_community_crisis start_phase = kehillah_crisis_warning save_scope_as = kehillah_crisis_new_situation } }
   scope:kehillah_crisis_new_situation ?= {
    set_variable = { name = kehillah_crisis_title value = scope:kehillah_crisis_situation_title }
    set_variable = { name = kehillah_crisis_epoch value = scope:kehillah_crisis_situation_title.var:kehillah_crisis_epoch }
    set_situation_center_province = scope:kehillah_crisis_situation_title.holder.domicile.domicile_location
    situation_sub_region:kehillah_crisis_area = { add_dejure_title_to_sub_region = scope:kehillah_crisis_situation_title.holder.domicile.domicile_location.county }
    scope:kehillah_crisis_situation_title = { set_variable = { name = kehillah_crisis_situation value = scope:kehillah_crisis_new_situation } }
    # The race opens with the crowd already stirred (an old save without a recorded opening gets
    # the generic one).
    if = {
     limit = {
      scope:kehillah_crisis_situation_title.var:kehillah_crisis_stage = 1
      scope:kehillah_crisis_situation_title = { NOT = { has_variable = kehillah_crisis_opening } NOT = { has_variable = kehillah_crisis_fresh } }
     }
     add_to_temporary_list = kehillah_crisis_opening_list
    }
   }
  }
 }
 var:kehillah_crisis_situation ?= {
  save_scope_as = kehillah_crisis_situation_instance
  if = { limit = { exists = var:kehillah_crisis_leader } remove_manual_participant = var:kehillah_crisis_leader remove_variable = kehillah_crisis_leader }
  if = { limit = { exists = var:kehillah_crisis_sovereign } remove_manual_participant = var:kehillah_crisis_sovereign remove_variable = kehillah_crisis_sovereign }
  if = {
   limit = { exists = var:kehillah_crisis_agitator }
   if = {
    limit = { any_situation_participant = { this = scope:kehillah_crisis_situation_instance.var:kehillah_crisis_agitator } }
    remove_manual_participant = var:kehillah_crisis_agitator
   }
   remove_variable = kehillah_crisis_agitator
  }
  if = {
   limit = { exists = var:kehillah_crisis_witness }
   if = {
    limit = { any_situation_participant = { this = scope:kehillah_crisis_situation_instance.var:kehillah_crisis_witness } }
    remove_manual_participant = var:kehillah_crisis_witness
   }
   remove_variable = kehillah_crisis_witness
  }
  # 2026-11-01: an accusation's witness, while the charge stands.
  scope:kehillah_crisis_situation_title ?= {
   if = {
    limit = {
     var:kehillah_crisis_opening ?= flag:accusation
     var:kehillah_acc_witness ?= { is_alive = yes }
    }
    var:kehillah_acc_witness = {
     scope:kehillah_crisis_situation_instance = {
      set_variable = { name = kehillah_crisis_witness value = prev }
      add_manual_participant = prev
     }
    }
   }
  }
  scope:kehillah_crisis_situation_title.holder ?= {
   scope:kehillah_crisis_situation_instance = {
    set_variable = { name = kehillah_crisis_leader value = prev }
    add_manual_participant = prev
   }
  }
  scope:kehillah_crisis_situation_title.holder.domicile.domicile_location.county.holder.top_liege ?= {
   scope:kehillah_crisis_situation_instance = {
    set_variable = { name = kehillah_crisis_sovereign value = prev }
    add_manual_participant = prev
   }
  }
  scope:kehillah_crisis_situation_title.var:kehillah_mob_threat ?= {
   if = { limit = { is_alive = yes }
    scope:kehillah_crisis_situation_instance = {
     set_variable = { name = kehillah_crisis_agitator value = prev }
     add_manual_participant = prev
    }
   }
  }
  if = {
   limit = { is_in_list = kehillah_crisis_opening_list }
   kehillah_crisis_store_push_effect = yes
   set_variable = { name = kehillah_crisis_opening_catalyst_pts value = kehillah_crisis_opening_points_value }
   kehillah_crisis_opening_catalyst_trigger_effect = { NOTICE = yes }
  }
  # Step 1e: a recorded opening counts once, by name.
  scope:kehillah_crisis_situation_title = { kehillah_crisis_credit_opening_effect = yes }
  # Event-decided transitions. In Agitation the judgement catalyst carries Spared to 100: any
  # prevention, payment or dispersal before the host rose means the danger has passed.
  if = { limit = { situation_current_phase = kehillah_crisis_warning scope:kehillah_crisis_situation_title.var:kehillah_crisis_stage = 2 } trigger_situation_catalyst = kehillah_crisis_armed_catalyst }
  if = { limit = { situation_current_phase = kehillah_crisis_warning scope:kehillah_crisis_situation_title.var:kehillah_crisis_stage = 3 } trigger_situation_catalyst = kehillah_crisis_judgement_catalyst }
  # Step 1d: an event resolved the Armed race. A struck quarter is Struck; anything else is Held.
  if = {
   limit = { situation_current_phase = kehillah_crisis_armed scope:kehillah_crisis_situation_title.var:kehillah_crisis_stage = 3 }
   if = {
    limit = { scope:kehillah_crisis_situation_title.var:kehillah_crisis_outcome ?= 4 }
    trigger_situation_catalyst = kehillah_crisis_struck_catalyst
   }
   else = { trigger_situation_catalyst = kehillah_crisis_judgement_catalyst }
  }
  if = {
   limit = {
    scope:kehillah_crisis_situation_title.var:kehillah_crisis_stage = 4
    OR = {
     situation_current_phase = kehillah_crisis_struck
     AND = { situation_current_phase = kehillah_crisis_judgement scope:kehillah_crisis_situation_title.var:kehillah_crisis_outcome = 4 }
    }
   }
   trigger_situation_catalyst = kehillah_crisis_recovery_catalyst
  }
  kehillah_crisis_race_mirror_effect = yes
 }
 }
}

# Root: the situation, monthly. While the race runs, the named drivers add their points. Since
# 2026-11-01 (docs/spec/2026-11-01-crisis-catalyst-groups.md) each group fires once, with the sum
# of its drivers: the agitator's work, ill times, those who shield us.
kehillah_crisis_race_monthly_effect = {
 if = {
  limit = { kehillah_crisis_race_running_trigger = yes }
  kehillah_crisis_store_push_effect = yes
  # The month's push toward the host, told to the leader (kehillah_crisis_pressure_toast_effect).
  set_variable = { name = kehillah_crisis_month_push value = kehillah_crisis_preaching_points_value }
  if = {
   limit = { kehillah_crisis_following_points_value > 0 }
   change_variable = { name = kehillah_crisis_month_push add = kehillah_crisis_following_points_value }
  }
  set_variable = { name = kehillah_crisis_agitator_catalyst_pts value = var:kehillah_crisis_month_push }
  kehillah_crisis_agitator_catalyst_trigger_effect = { NOTICE = no }
  set_variable = { name = kehillah_crisis_ill_times_catalyst_pts value = 0 }
  if = {
   limit = {
    kehillah_crisis_holy_week_season_trigger = yes
    OR = {
     var:kehillah_crisis_sovereign.faith.religion ?= religion:christianity_religion
     var:kehillah_crisis_title.holder.domicile.domicile_location.county.faith.religion ?= religion:christianity_religion
    }
   }
   change_variable = { name = kehillah_crisis_month_push add = kehillah_crisis_holy_week_points }
   change_variable = { name = kehillah_crisis_ill_times_catalyst_pts add = kehillah_crisis_holy_week_points }
   kehillah_crisis_driver_notice_effect = { KEY = holy_week }
  }
  if = {
   limit = { var:kehillah_crisis_sovereign ?= { kehillah_outlook_band_value <= -1 } }
   change_variable = { name = kehillah_crisis_month_push add = kehillah_crisis_cold_lord_points_value }
   change_variable = { name = kehillah_crisis_ill_times_catalyst_pts add = kehillah_crisis_cold_lord_points_value }
   kehillah_crisis_driver_notice_effect = { KEY = cold_lord }
  }
  if = {
   limit = {
    var:kehillah_crisis_title.holder.domicile.domicile_location.county ?= {
     OR = {
      faith ?= { exists = great_holy_war }
      holder.top_liege.faith ?= { exists = great_holy_war }
     }
    }
   }
   change_variable = { name = kehillah_crisis_month_push add = kehillah_crisis_holy_war_points_value }
   change_variable = { name = kehillah_crisis_ill_times_catalyst_pts add = kehillah_crisis_holy_war_points_value }
   kehillah_crisis_driver_notice_effect = { KEY = holy_war }
  }
  if = {
   limit = { var:kehillah_crisis_ill_times_catalyst_pts > 0 }
   kehillah_crisis_ill_times_catalyst_trigger_effect = { NOTICE = yes }
  }
  # Those who shield us: the month's protection and the lord's goodwill.
  set_variable = { name = kehillah_crisis_shield_catalyst_pts value = 0 }
  if = {
   limit = { kehillah_crisis_protection_points_value > 0 }
   change_variable = { name = kehillah_crisis_shield_catalyst_pts add = kehillah_crisis_protection_points_value }
  }
  if = {
   limit = { var:kehillah_crisis_sovereign ?= { kehillah_outlook_band_value >= 1 } }
   change_variable = { name = kehillah_crisis_shield_catalyst_pts add = kehillah_crisis_goodwill_points_value }
  }
  if = {
   limit = { var:kehillah_crisis_shield_catalyst_pts > 0 }
   kehillah_crisis_shield_catalyst_trigger_effect = { NOTICE = yes }
  }
  # Step 1c: once for each kind of protection, as it is first won.
  kehillah_crisis_credit_protection_effect = { FLAG = kehillah_crisis_royal_protection }
  kehillah_crisis_credit_protection_effect = { FLAG = kehillah_mob_condemned }
  kehillah_crisis_credit_protection_effect = { FLAG = kehillah_mob_men_by }
  kehillah_crisis_credit_protection_effect = { FLAG = kehillah_mob_armed }
  kehillah_crisis_credit_protection_effect = { FLAG = kehillah_mob_refuge }
  kehillah_crisis_race_mirror_effect = yes
  kehillah_crisis_pressure_toast_effect = yes
 }
}

# Root: the situation, as the Armed host phase begins. If the race (not an event) brought it,
# the preacher's following becomes the host, exactly as the old end-of-warning roll did.
kehillah_crisis_armed_start_effect = {
 debug_log = "kehillah_community_crisis: Armed host phase entered"
 var:kehillah_crisis_title ?= {
  if = { limit = { has_variable = kehillah_crisis_spared_points } set_variable = { name = kehillah_crisis_prior_spared_progress value = var:kehillah_crisis_spared_points } }
 }
 var:kehillah_crisis_title ?= {
  if = { limit = { NOT = { has_variable = kehillah_crisis_paid_points } } set_variable = { name = kehillah_crisis_paid_points value = 0 } }
 }
 var:kehillah_crisis_title ?= { kehillah_crisis_race_clear_mirror_effect = yes }
 if = {
  limit = { kehillah_crisis_situation_link_valid_trigger = yes }
  var:kehillah_crisis_title = { save_scope_as = kehillah_crisis_phase_title }
  if = {
   limit = { scope:kehillah_crisis_phase_title.var:kehillah_crisis_stage = 1 }
   if = {
    limit = { scope:kehillah_crisis_phase_title.var:kehillah_mob_threat ?= { is_alive = yes has_character_flag = kehillah_mob_preaching } }
    scope:kehillah_crisis_phase_title.var:kehillah_mob_threat = { kehillah_mob_preaching_mobilize_effect = yes }
    # Step 1d: the second race starts from what was prepared.
    if = {
     limit = { scope:kehillah_crisis_phase_title.var:kehillah_crisis_stage = 2 }
     kehillah_crisis_armed_seed_effect = yes
    }
   }
   else = {
    # No preacher to lead a host: the danger passes after all, and the phase moves on.
    debug_log = "kehillah_community_crisis: Armed won the race but no preacher remains; resolving as prevented"
    scope:kehillah_crisis_phase_title.holder ?= { kehillah_crisis_resolve_effect = { OUTCOME = 1 } }
   }
  }
  else_if = {
   limit = { scope:kehillah_crisis_phase_title.var:kehillah_crisis_stage = 2 }
   kehillah_crisis_armed_seed_effect = yes
  }
  else_if = {
   limit = { scope:kehillah_crisis_phase_title.var:kehillah_crisis_stage >= 3 }
   debug_log = "kehillah_community_crisis: Armed phase began after the crisis was already resolved; moving on"
   if = {
    limit = { scope:kehillah_crisis_phase_title.var:kehillah_crisis_outcome ?= 4 }
    trigger_situation_catalyst = kehillah_crisis_struck_catalyst
   }
   else = { trigger_situation_catalyst = kehillah_crisis_judgement_catalyst }
  }
