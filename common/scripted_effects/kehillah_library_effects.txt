# ---------------------------------------------------------------------
# kehillah_seed_starting_library_effect
# Scope: a title (a Kehillah community's primary title). Called once per
# registered community from kehillah_on_game_start (common/on_action/
# kehillah_on_actions.txt), AFTER every add_to_global_variable_list call
# that builds kehillah_registered_communities -- mirrors kehillah_
# refresh_map_view_for_players_effect's own placement there exactly, and
# for the same reason: this needs the full registry already built to
# iterate over it.
#
# RANDOMIZED, 1-3 WORKS, UPDATED 2026-09-19 at user request ("mix it up,
# random books from the initial list, expand the initial list") --
# previously every community got the literal same two (Targum Onkelos +
# the Mishnah). Now each community rolls a count of 1-3 (equal odds) and
# draws that many DISTINCT works from the 8-work Parshanut+Talmudics pool
# via kehillah_seed_one_random_torah_work_effect below, called once per
# roll. HASHKAFA IS DELIBERATELY STILL EXCLUDED from this pool -- kept at
# user's explicit choice, preserving the scripted-triggers file's own
# "Hashkafa starts empty on purpose" design (its header, just above the
# six kehillah_owns_/missing_<track>_work_trigger entries) rather than
# silently overriding it.
# ---------------------------------------------------------------------

kehillah_seed_starting_library_effect = {
	random_list = {
		1 = { set_variable = { name = kehillah_starting_book_count value = 1 } }
		1 = { set_variable = { name = kehillah_starting_book_count value = 2 } }
		1 = { set_variable = { name = kehillah_starting_book_count value = 3 } }
	}
	kehillah_seed_one_random_torah_work_effect = yes
	if = {
		limit = { var:kehillah_starting_book_count >= 2 }
		kehillah_seed_one_random_torah_work_effect = yes
	}
	if = {
		limit = { var:kehillah_starting_book_count >= 3 }
		kehillah_seed_one_random_torah_work_effect = yes
	}
	remove_variable = kehillah_starting_book_count
}

# =====================================================================
# kehillah_study_specific_work_effect -- ADDED 2026-09-17 at user request
# (Yeshiva Visit's own picker: "when you get there it gives you a
# sampling of books to choose from?"). Scope: root = the studying
# character. Originally journey-only (called from kehillah_learn_torah_
# journey.0002/0003/0004, events/kehillah_learn_torah_journey_events.txt,
# one per track, each offering the specific works that track's random_
# list branch used to pick among blind) while local study went through
# its own three now-retired per-track effects instead (this file's own
# retirement note just above).
#
# AS OF 2026-09-19, ALSO LOCAL STUDY'S OWN RESOLUTION -- kehillah_study_
# torah_resolve_current_work_effect (below) dispatches to this same
# effect, by literal WORK/TRACK, for whichever work the kehillah_study_
# torah scheme's own numeric-ID bridge says is currently being read. No
# changes were needed to this effect itself to support that: it was
# already exactly "resolve one named work for one named character",
# which is exactly what both callers need.
#
# $WORK$/$TRACK$, THE SAME PARAMETERIZED-SCRIPTED-EFFECT IDIOM $LIBRARY$
# already established (kehillah_scripted_triggers.txt's own header) --
# $WORK$ substitutes the literal flag (e.g. flag:work_targum_onkelos) and
# $TRACK$ the literal trait track (parshanut/talmudics/hashkafa), so one
# effect definition serves all twelve works, called once per work from
# each caller's own per-work branch rather than twelve near-duplicate
# effect bodies.
#
# NO $LIBRARY$ HERE: "does the relevant library own this work" is checked
# by the CALLER (the journey's own per-option trigger; the scheme's own
# pick-next-work effect, which is what decides which work becomes
# "current" in the first place) -- this effect only ever touches the
# STUDYING character's own kehillah_studied_works, never a library.
# random_list and a custom_tooltip-per-branch reveal are also absent here
# for the same reason: whichever caller already knows the specific work
# (the journey's own clicked option; the scheme's own numeric-ID
# dispatch) has already revealed it, via the same kehillah_study_torah.
# <track>.<work>_tt lines, before this effect ever runs.
# =====================================================================

kehillah_study_specific_work_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = kehillah_studied_works target = $WORK$ } }
		set_variable = { name = kehillah_learn_torah_repeat value = 1 }
	}
	else = {
		set_variable = { name = kehillah_learn_torah_repeat value = 0 }
		add_to_variable_list = { name = kehillah_studied_works target = $WORK$ }
	}

	if = {
		limit = { var:kehillah_learn_torah_repeat = 1 }
		if = {
			limit = { learning >= 10 }
			add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_repeat_pass }
			custom_tooltip = kehillah_study_torah.repeat_pass.tt
		}
		else = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_repeat_fail }
			custom_tooltip = kehillah_study_torah.repeat_fail.tt
		}
	}
	else = {
		if = {
			limit = { learning >= 10 }
			add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_pass }
			custom_tooltip = kehillah_study_torah.pass.tt
		}
		else = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_fail }
			custom_tooltip = kehillah_study_torah.fail.tt
		}
	}
	add_piety = minor_piety_gain
	remove_variable = kehillah_learn_torah_repeat
}

# THE NUMERIC-ID BRIDGE, AND WHY IT EXISTS AT ALL. A scheme phase and its
# own on_phase_completed hook are two different script moments, months
# apart (base_progress_goal), with nothing but the scheme's own state to
# carry information between them -- there is no "phase started" hook to
# pair with on_phase_completed, so "which work am I currently reading"
# has to be decided once (at pick time) and read back later (at resolve

# kind of flag reference and compare against it later -- was already
# investigated and explicitly rejected once in kehillah_study_specific_

# twin of the same bridge). Twenty-three if/elif branches is more lines
# than a generic lookup would be, but it is the confirmed-safe shape, not
# an invented one.
#

# BEYOND KEHILLAH LEADERS -- ADDED 2026-09-19, same-day follow-up, at user
# request. See kehillah_study_torah_has_accessible_library_trigger's own
# header (common/scripted_triggers/kehillah_scripted_triggers.txt) for the
# three populations and why a landless adventurer's anchor is their own
# CURRENT LOCATION, not a domicile (confirmed absent from vanilla for that
# government type).
#
# kehillah_study_torah_resolve_library_effect is the one place that
# actually decides WHICH community's library applies, and it is called
# FRESH every single time a library is needed (the decision's own effect,
# every phase completion, every monthly tick below) rather than once and
# remembered -- the same "live query, not snapshot" choice kehillah_
# study_torah_pick_next_work_effect's own header already made, now
# extended to "which library" as well as "which work". This is not just
# consistency for its own sake: it is what lets a landed ruler's capital
# moving, or an adventurer wandering off, be handled by simply re-running
# the same search next time, rather than needing separate move-detection
# code of its own. A character variable holding the resolved title (the
# retired kehillah_visit_library_interaction's own approach, see events/
# kehillah_learn_torah_journey_events.txt's header for that precedent) was
# considered and rejected here specifically because that mod's own reason
# for persisting one -- surviving a gap between two DIFFERENT effect
# chains separated by real time -- does not apply: every caller below
# already needs a fresh answer at the moment it asks, not a memory of an
# old one.
#
# scope:kehillah_study_torah_library is left UNSET if nothing matches
# (no Kehillah, no ruler capital with a nearby community, no adventurer
# location with one) -- every caller checks `exists =` before using it;
# see kehillah_study_torah_monthly_effect below for what happens when it
# comes up empty (nothing, this tick -- the scheme's own `valid` trigger,
# common/schemes/scheme_types/kehillah_study_torah_scheme.txt, is what
# actually ends the scheme if this is not transient).
#
# $CHARACTER$, NOT BARE root -- ck3-tiger caught this: root is UNSAFE when
# this effect is called from kehillah_study_torah_monthly_effect below,
# which itself runs inside the scheme's own on_monthly hook where root is
# the SCHEME, not the owner (root does not change through a nested scope-
# opening block, so it stays the scheme even several layers of scheme_
# owner = { ... } deep). $CHARACTER$ substitutes literal caller-supplied
# text into every reference below, including the ones nested inside
# holder = { ... } -- root from the decision's own effect or the phase-
# completion event's option (both stable there, character_event/decision
# effect root never shifts); scope:owner from kehillah_study_torah_
# monthly_effect (already saved one level up by the scheme's own on_
# monthly, and -- unlike root or this -- a saved scope stays correct no
# matter how deep it is later referenced).
# =====================================================================

kehillah_study_torah_resolve_library_effect = {
	if = {
		limit = { $CHARACTER$ = { government_has_flag = government_is_kehillah } }
		$CHARACTER$.primary_title = { save_scope_as = kehillah_study_torah_library }
	}
	else_if = {
		limit = { $CHARACTER$ = { is_ruler = yes } }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { domicile.domicile_location = $CHARACTER$.capital_province }
			}
			save_scope_as = kehillah_study_torah_library
		}
	}
	else_if = {
		limit = { $CHARACTER$ = { has_government = landless_adventurer_government } }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { domicile.domicile_location = $CHARACTER$.location }
			}
			save_scope_as = kehillah_study_torah_library
		}
	}
}

# Scope: the studying character. Called from the scheme's own on_monthly
# (common/schemes/scheme_types/kehillah_study_torah_scheme.txt) -- the
# "refresh periodically... if you don't own a book anymore" half of the
# user's request; the "detect if there are new books" half needs no extra
# code at all, since kehillah_study_torah_pick_next_work_effect above
# already queries the library's CURRENT contents live, every single time
# it runs (at scheme start and at every phase completion) rather than
# working from any kind of snapshot -- see that effect's own header.
#
# If the specific work currently being read is no longer owned (donated
# away mid-read, per the library panel's own new donate-for-piety action
# -- OR simply because a moved capital/wandered-off adventurer now resolves
# a DIFFERENT community's library entirely), toast the player and re-pick
# immediately rather than let scheme_progress keep building toward a book
# that no longer exists to reward. Progress itself is left untouched -- it
# represents time spent at the Beit Midrash, not attachment to one
# specific volume, so switching what is being read does not reset or
# penalize it.
#
# If the current work is still owned, this is instead where the
# occasional flavor-only "insight" event (kehillah_study_torah.0002,
# events/kehillah_study_torah_events.txt) fires -- a flat 12%/month
# chance, pure flavor, no mechanical reward (the real reward is still
# only ever paid at phase completion) so as not to double up with it.
kehillah_study_torah_monthly_effect = {
	if = {
		limit = { has_variable = kehillah_study_current_work_id }
		# CHARACTER = scope:owner, NOT root or this -- see kehillah_study_
		# torah_resolve_library_effect's own header just above for why:
		# root here is the SCHEME (this effect runs inside the scheme's own
		# on_monthly), and scope:owner is already saved one level up by
		# that same on_monthly, before this effect was ever called.
		kehillah_study_torah_resolve_library_effect = { CHARACTER = scope:owner }
		if = {
			limit = { exists = scope:kehillah_study_torah_library }
			if = {
				limit = { NOT = { kehillah_study_torah_current_work_still_owned_trigger = { LIBRARY = scope:kehillah_study_torah_library } } }
				send_interface_toast = {
					title = kehillah_study_torah_book_lost_title
					custom_tooltip = kehillah_study_torah_book_lost_tt
					left_icon = this
				}
				kehillah_study_torah_pick_next_work_effect = { LIBRARY = scope:kehillah_study_torah_library }
			}
			else = {
				random = {
					chance = 12
					trigger_event = kehillah_study_torah.0002
				}
			}
		}
	}
}
