# ---------------------------------------------------------------------
# kehillah_seed_starting_library_effect
# Scope: a title (a Kehillah community's primary title). Called once per
# registered community from kehillah_on_game_start (common/on_action/
# kehillah_on_actions.txt), AFTER every add_to_global_variable_list call
# that builds kehillah_registered_communities -- mirrors kehillah_
# refresh_map_view_for_players_effect's own placement there exactly, and
# for the same reason: this needs the full registry already built to
# iterate over it.
#
# RANDOMIZED, 1-3 WORKS, UPDATED 2026-09-19 at user request ("mix it up,
# random books from the initial list, expand the initial list") --
# previously every community got the literal same two (Targum Onkelos +
# the Mishnah). Now each community rolls a count of 1-3 (equal odds) and
# draws that many DISTINCT works from the Parshanut+Talmudics pool
# via kehillah_seed_one_random_torah_work_effect below, called once per
# roll. HASHKAFA IS DELIBERATELY STILL EXCLUDED from this pool -- kept at
# user's explicit choice, preserving the scripted-triggers file's own
# "Hashkafa starts empty on purpose" design (its header, just above the
# six kehillah_owns_/missing_<track>_work_trigger entries) rather than
# silently overriding it. 2026-10-14: a title whose holder already has a
# tier-2 Beit Midrash starts with 3-4; a tier-3 yeshiva starts with 4-6.
# Upgrading later does not conjure manuscripts: acquisition remains the route.
# ---------------------------------------------------------------------

# 2026-11-08 SHAS-FIRST SEEDING (docs/spec/2026-11-08-sefer-system-
# redesign.md section 5.2). Supersedes the random 1-6 draw above. Every
# library is registered and gets the Tanakh. A Rabbanite yeshiva seat
# (tier 3) starts with a complete Shas, a tier-2 hall with three to five
# orders, a small community one or two -- Moed, Nashim and Nezikin before
# Zeraim, Kodashim and Tahorot -- plus one to three other works. A Karaite
# community starts with its founding works in circulation.
kehillah_seed_starting_library_effect = {
	kehillah_lib_register_library_effect = yes
	if = {
		limit = { kehillah_lib_is_karaite_library_trigger = yes }
		add_to_variable_list = { name = kehillah_library_works target = flag:work_anan_sefer_hamitzvot }
		if = {
			limit = { current_date >= 937.1.1 }
			add_to_variable_list = { name = kehillah_library_works target = flag:work_kitab_alanwar }
		}
		if = {
			limit = { current_date >= 990.1.1 }
			add_to_variable_list = { name = kehillah_library_works target = flag:work_yefet_commentary }
		}
	}
	else = {
		if = {
			limit = { holder = { domicile ?= { has_domicile_building_or_higher = kehillah_beit_midrash_03 } } }
			set_variable = { name = kehillah_starting_book_count value = 6 }
			random_list = {
				1 = { set_variable = { name = kehillah_starting_other_count value = 2 } }
				1 = { set_variable = { name = kehillah_starting_other_count value = 3 } }
			}
		}
		else_if = {
			limit = { holder = { domicile ?= { has_domicile_building_or_higher = kehillah_beit_midrash_02 } } }
			random_list = {
				1 = { set_variable = { name = kehillah_starting_book_count value = 3 } }
				1 = { set_variable = { name = kehillah_starting_book_count value = 4 } }
				1 = { set_variable = { name = kehillah_starting_book_count value = 5 } }
			}
			random_list = {
				1 = { set_variable = { name = kehillah_starting_other_count value = 1 } }
				1 = { set_variable = { name = kehillah_starting_other_count value = 2 } }
			}
		}
		else = {
			random_list = {
				1 = { set_variable = { name = kehillah_starting_book_count value = 1 } }
				1 = { set_variable = { name = kehillah_starting_book_count value = 2 } }
			}
			random_list = {
				1 = { set_variable = { name = kehillah_starting_other_count value = 1 } }
				1 = { set_variable = { name = kehillah_starting_other_count value = 2 } }
			}
		}
		# The orders, in the order a community would want them.
		add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_moed }
		if = {
			limit = { var:kehillah_starting_book_count >= 2 }
			add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_nashim }
		}
		if = {
			limit = { var:kehillah_starting_book_count >= 3 }
			add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_nezikin }
		}
		if = {
			limit = { var:kehillah_starting_book_count >= 4 }
			random_list = {
				1 = { add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_zeraim } }
				1 = { add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim } }
			}
		}
		if = {
			limit = { var:kehillah_starting_book_count >= 5 }
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim } } }
				add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim }
			}
			else = {
				add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_zeraim }
			}
		}
		if = {
			limit = { var:kehillah_starting_book_count >= 6 }
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_zeraim } } }
				add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_zeraim }
			}
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim } } }
				add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim }
			}
			add_to_variable_list = { name = kehillah_library_works target = flag:work_bavli_tahorot }
		}
		kehillah_seed_one_other_work_effect = yes
		if = {
			limit = { var:kehillah_starting_other_count >= 2 }
			kehillah_seed_one_other_work_effect = yes
		}
		if = {
			limit = { var:kehillah_starting_other_count >= 3 }
			kehillah_seed_one_other_work_effect = yes
		}
		remove_variable = kehillah_starting_book_count
		remove_variable = kehillah_starting_other_count
	}
	kehillah_lib_refresh_cache_effect = yes
}

# kehillah_study_specific_work_effect -- ADDED 2026-09-17 at user request
# (Yeshiva Visit's own picker: "when you get there it gives you a
# sampling of books to choose from?"). Scope: root = the studying
# character. Originally journey-only (called from kehillah_learn_torah_
# journey.0002/0003/0004, events/kehillah_learn_torah_journey_events.txt,
# one per track, each offering the specific works that track's random_
# list branch used to pick among blind) while local study went through
# its own three now-retired per-track effects instead (this file's own
# retirement note just above).
#
# AS OF 2026-09-19, ALSO LOCAL STUDY'S OWN RESOLUTION -- kehillah_study_
# torah_resolve_current_work_effect (below) dispatches to this same
# effect, by literal WORK/TRACK, for whichever work the kehillah_study_
# torah scheme's own numeric-ID bridge says is currently being read. No
# changes were needed to this effect itself to support that: it was
# already exactly "resolve one named work for one named character",
# which is exactly what both callers need.
#
# $WORK$/$TRACK$, THE SAME PARAMETERIZED-SCRIPTED-EFFECT IDIOM $LIBRARY$
# already established (kehillah_scripted_triggers.txt's own header) --
# $WORK$ substitutes the literal flag (e.g. flag:work_targum_onkelos) and
# $TRACK$ the literal trait track (parshanut/talmudics/hashkafa), so one
# effect definition serves all twelve works, called once per work from
# each caller's own per-work branch rather than twelve near-duplicate
# effect bodies.
#
# NO $LIBRARY$ HERE: "does the relevant library own this work" is checked
# by the CALLER (the journey's own per-option trigger; the scheme's own
# pick-next-work effect, which is what decides which work becomes
# "current" in the first place) -- this effect only ever touches the
# STUDYING character's own encountered and mastered work lists, never a library.
# random_list and a custom_tooltip-per-branch reveal are also absent here
# for the same reason: whichever caller already knows the specific work
# (the journey's own clicked option; the scheme's own numeric-ID
# dispatch) has already revealed it, via the same kehillah_study_torah.
# <track>.<work>_tt lines, before this effect ever runs.
# =====================================================================

# ---------------------------------------------------------------------
# V21 SLICE 2, 2026-10-09 (Shabbat heartbeat 1, ROADMAP H4). TWO
# CHANGES, BOTH INERT UNTIL THE LEARN TORAH GATE IS WIDENED.
#
# 1. Every add_trait_xp below is guarded on actually holding
#    kehillah_rabbi_trait. docs/spec/v21-rabbi-ordination-paths.md
#    section 3.2: "Trait XP is only meaningful after a character has the
#    lifestyle trait. An apprentice must therefore use separate
#    character-scoped study records, not attempt to add
#    kehillah_rabbi_trait XP before the trait exists." Today the Learn
#    Torah decision requires the trait outright, so these guards cannot
#    change anything yet; they are what makes it safe for slice 3 to let
#    an unordained candidate in.
#
# 2. A FIRST mastery also records the work's FIELD in
#    kehillah_apprentice_fields. The spec's curriculum is "2 distinct
#    completed works, from 2 distinct fields", and the distinct-works
#    half is already recorded -- kehillah_mastered_works is exactly
#    that list -- so this adds only the part that was missing rather
#    than a parallel work list that could drift out of step with it.
#    flag:$TRACK$ works because macro substitution is textual: $TRACK$
#    is already the literal track token (parshanut/talmudics/hashkafa)
#    this effect's callers pass.
#
#    Recorded for a rabbi too, not just an apprentice. It is one list
#    append, it keeps the branch count down, and section 3.4 says the
#    apprentice records survive ordination as biography anyway; the
#    readiness trigger is what refuses to count them twice, not this.
# ---------------------------------------------------------------------

kehillah_study_specific_work_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = $WORK$ } }
		set_variable = { name = kehillah_learn_torah_repeat value = 1 }
	}
	else = {
		set_variable = { name = kehillah_learn_torah_repeat value = 0 }
	}
	# This older list means encountered. Preserve it for travel discovery and
	# existing saves; a failed first reading does not confer mastery.
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = kehillah_studied_works target = $WORK$ } } }
		add_to_variable_list = { name = kehillah_studied_works target = $WORK$ }
	}

	if = {
		limit = { var:kehillah_learn_torah_repeat = 1 }
		if = {
			limit = {
				OR = {
					learning >= 10
					has_variable = kehillah_study_torah_mastery_bonus
					kehillah_talmud_torah_mastery_ready_trigger = yes
				}
			}
			set_variable = { name = kehillah_study_torah_last_result value = 2 }
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_repeat_pass }
				if = {
					limit = { kehillah_talmud_torah_follower_trigger = yes }
					add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_talmud_torah_rabbi_xp_repeat_mastery }
				}
			}
			custom_tooltip = kehillah_study_torah.repeat_pass.tt
		}
		else = {
			set_variable = { name = kehillah_study_torah_last_result value = 3 }
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_repeat_fail }
			}
			custom_tooltip = kehillah_study_torah.repeat_fail.tt
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					learning >= 10
					has_variable = kehillah_study_torah_mastery_bonus
					kehillah_talmud_torah_mastery_ready_trigger = yes
				}
			}
			set_variable = { name = kehillah_study_torah_last_result value = 1 }
			add_to_variable_list = { name = kehillah_mastered_works target = $WORK$ }
			# Index successful readers once. This lets the shelf show living
			# rabbis who mastered a selected work without scanning the world.
			if = {
				limit = { NOT = { is_target_in_global_variable_list = { name = kehillah_sefer_masters target = this } } }
				add_to_global_variable_list = { name = kehillah_sefer_masters target = this }
			}
			# The field half of V21's curriculum -- see this effect's header.
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = kehillah_apprentice_fields target = flag:$TRACK$ } } }
				add_to_variable_list = { name = kehillah_apprentice_fields target = flag:$TRACK$ }
			}
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_pass }
				if = {
					limit = { kehillah_talmud_torah_follower_trigger = yes }
					add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_talmud_torah_rabbi_xp_first_mastery }
				}
			}
			custom_tooltip = kehillah_study_torah.pass.tt
		}
		else = {
			set_variable = { name = kehillah_study_torah_last_result value = 0 }
			if = {
				limit = { has_trait = kehillah_rabbi_trait }
				add_trait_xp = { trait = kehillah_rabbi_trait track = $TRACK$ value = kehillah_learn_torah_xp_fail }
			}
			custom_tooltip = kehillah_study_torah.fail.tt
		}
	}
	# V21 slice 3: an ordination candidate is told, every time, that he is an
	# apprentice and where he stands. Section 3.1 asks for exactly this --
	# the completion "should explain that an unordained reader is an
	# apprentice, not award the lifestyle trait immediately" -- and without
	# it a candidate would study for years with no visible signal at all
	# that it leads anywhere.
	if = {
		limit = {
			NOT = { has_trait = kehillah_rabbi_trait }
			kehillah_rabbi_study_eligible_trigger = yes
		}
		if = {
			limit = { kehillah_rabbi_ordination_curriculum_met_trigger = yes }
			custom_tooltip = kehillah_study_torah.apprentice_ready.tt
		}
		else = {
			custom_tooltip = kehillah_study_torah.apprentice.tt
		}
	}

	# Torah learning is open to lay readers under this tenet. The bonus is
	# paid for understanding a work, not for acquiring the Rabbi trait.
	if = {
		limit = { kehillah_talmud_torah_follower_trigger = yes }
		if = {
			limit = { var:kehillah_study_torah_last_result = 1 }
			add_piety = kehillah_talmud_torah_first_mastery_piety
			add_learning_lifestyle_xp = kehillah_talmud_torah_first_mastery_learning_xp
		}
		else_if = {
			limit = { var:kehillah_study_torah_last_result = 2 }
			add_piety = kehillah_talmud_torah_repeat_mastery_piety
			add_learning_lifestyle_xp = kehillah_talmud_torah_repeat_mastery_learning_xp
		}
	}

	# Spiritual Fulfillment follows actual understanding, not merely starting
	# a course. A newly mastered work is worth more than a rereading; failed
	# readings do not advance it. This applies to eligible readers with or
	# without the Talmud Torah tenet.
	if = {
		limit = {
			is_playable_character = yes
			var:kehillah_study_torah_last_result = 1
		}
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
	}
	else_if = {
		limit = {
			is_playable_character = yes
			var:kehillah_study_torah_last_result = 2
		}
		change_spiritual_fulfillment = miniscule_spiritual_fulfillment_value
	}

	add_piety = minor_piety_gain
	# 2026-11-08 Sefer redesign: standing, tenets, rounds and the siyum.
	# $KEY$ is the work id without flag: (every caller is generated by
	# scripts/kehillah_library_catalog.py and passes it).
	kehillah_sefer_after_study_effect = { KEY = $KEY$ }
	remove_variable = kehillah_study_torah_mastery_bonus
	remove_variable = kehillah_learn_torah_repeat
}

# STUDY TORAH AS A SCHEME -- ADDED 2026-09-19 at user request ("implement
# Learn Torah as a continuous scheme... events during the scheme fire to
# tell you which text you're starting... refresh periodically to detect
# if there are new books or if you don't own a book anymore"). See
# common/schemes/scheme_types/kehillah_study_torah_scheme.txt for the
# scheme type these three effects serve, and events/kehillah_study_
# torah_events.txt for where they are actually called from.
#
# kehillah_learn_torah_decision now just starts this scheme (its own
# header, common/decisions/kehillah_learn_torah_decisions.txt, has the
# full retirement account of the old flat kehillah_learn_torah.0001 event
# this replaces). kehillah_study_specific_work_effect above is UNCHANGED
# and STILL SHARED with the travel journey (kehillah_learn_torah_journey.
# 0002/0003/0004) -- everything below is new plumbing that calls it, not
# a replacement for it, so the journey path is untouched by this pass.
#
# THE NUMERIC-ID BRIDGE, AND WHY IT EXISTS AT ALL. A scheme phase and its
# own on_phase_completed hook are two different script moments, months
# apart (base_progress_goal), with nothing but the scheme's own state to
# carry information between them -- there is no "phase started" hook to
# pair with on_phase_completed, so "which work am I currently reading"
# has to be decided once (at pick time) and read back later (at resolve
# time). The obvious way to do that -- store the specific work as some
# kind of flag reference and compare against it later -- was already
# investigated and explicitly rejected once in kehillah_study_specific_
# work_effect's own header above ("target = var:X where X holds a flag...
# was checked against the installed game and NOT found anywhere as real
# precedent, so it is not risked here"). This reuses that same finding
# rather than re-risking it: kehillah_study_current_work_id is a plain
# NUMBER variable (1-23, one literal work per number, mapping fixed by
# convention below and never computed), read back via the same OR-of-AND/
# if-elif-per-literal shape this file and kehillah_scripted_triggers.txt
# already use everywhere a "pick one of twenty-three real things" problem
# shows up (kehillah_study_torah_current_work_still_owned_trigger, common/
# scripted_triggers/kehillah_scripted_triggers.txt, is the trigger-side
# twin of the same bridge). Twenty-three if/elif branches is more lines
# than a generic lookup would be, but it is the confirmed-safe shape, not
# an invented one.
#
# THE MAPPING (fixed, referenced by every branch below and by the
# matching trigger). UPDATED 2026-09-19, SAME-DAY FOLLOW-UP: grew from
# twelve to twenty-three when the Bavli and Yerushalmi were each split
# into their own Sedarim (real print-volume divisions, not invented --
# see kehillah_scripted_triggers.txt's own updated header), and Hashkafa
# grew from four works to seven when Emunot ve-Deot/Chovot HaLevavot/
# Kuzari were date-gated behind their real historical composition years
# and backfilled with three pre-867 replacements (same header):
#   1  = work_targum_onkelos          (parshanut)
#   2  = work_mekhilta                (parshanut)
#   3  = work_genesis_rabbah          (parshanut)
#   4  = work_pirkei_derabbi_eliezer  (parshanut)
#   5  = work_mishnah                 (talmudics)
#   6  = work_bavli_zeraim            (talmudics)
#   7  = work_bavli_moed              (talmudics)

# CONTINUING WHEN OUT OF UNSTUDIED BOOKS, PER EXPLICIT USER REQUEST ("you
# should be able to continue the scheme even when out of books, but it
# should be very clear that the rewards are now much less"). The pick
# below draws from EVERY work the library owns, not just unstudied ones --

# BEYOND KEHILLAH LEADERS -- ADDED 2026-09-19, same-day follow-up, at user
# request. See kehillah_study_torah_has_accessible_library_trigger's own
# header (common/scripted_triggers/kehillah_scripted_triggers.txt) for the
# three populations and why a landless adventurer's anchor is their own
# CURRENT LOCATION, not a domicile (confirmed absent from vanilla for that
# government type).
#
# kehillah_study_torah_resolve_library_effect is the one place that
# actually decides WHICH community's library applies, and it is called
# FRESH every single time a library is needed (the decision's own effect,
# every phase completion, every monthly tick below) rather than once and
# remembered -- the same "live query, not snapshot" choice kehillah_
# study_torah_pick_next_work_effect's own header already made, now
# extended to "which library" as well as "which work". This is not just
# consistency for its own sake: it is what lets a landed ruler's capital
# moving, or an adventurer wandering off, be handled by simply re-running
# the same search next time, rather than needing separate move-detection
# code of its own. A character variable holding the resolved title (the
# retired kehillah_visit_library_interaction's own approach, see events/
# kehillah_learn_torah_journey_events.txt's header for that precedent) was
# considered and rejected here specifically because that mod's own reason
# for persisting one -- surviving a gap between two DIFFERENT effect
# chains separated by real time -- does not apply: every caller below
# already needs a fresh answer at the moment it asks, not a memory of an
# old one.
#
# scope:kehillah_study_torah_library is left UNSET if nothing matches
# (no Kehillah, no ruler capital with a nearby community, no adventurer
# location with one) -- every caller checks `exists =` before using it;
# see kehillah_study_torah_monthly_effect below for what happens when it
# comes up empty (nothing, this tick -- the scheme's own `valid` trigger,
# common/schemes/scheme_types/kehillah_study_torah_scheme.txt, is what
# actually ends the scheme if this is not transient).
#
# $CHARACTER$, NOT BARE root -- ck3-tiger caught this: root is UNSAFE when
# this effect is called from kehillah_study_torah_monthly_effect below,
# which itself runs inside the scheme's own on_monthly hook where root is
# the SCHEME, not the owner (root does not change through a nested scope-
# opening block, so it stays the scheme even several layers of scheme_
# owner = { ... } deep). $CHARACTER$ substitutes literal caller-supplied
# text into every reference below, including the ones nested inside
# holder = { ... } -- root from the decision's own effect or the phase-
# completion event's option (both stable there, character_event/decision
# effect root never shifts); scope:owner from kehillah_study_torah_
# monthly_effect (already saved one level up by the scheme's own on_
# monthly, and -- unlike root or this -- a saved scope stays correct no
# matter how deep it is later referenced).
# =====================================================================

kehillah_study_torah_resolve_library_effect = {
	if = {
		limit = { $CHARACTER$ = { government_has_flag = government_is_kehillah } }
		$CHARACTER$.primary_title = { save_scope_as = kehillah_study_torah_library }
	}
	else_if = {
		limit = { $CHARACTER$ = { is_ruler = yes } }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { domicile.domicile_location = $CHARACTER$.capital_province }
			}
			save_scope_as = kehillah_study_torah_library
		}
	}
	else_if = {
		limit = { $CHARACTER$ = { has_government = landless_adventurer_government } }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				exists = holder
				holder = { domicile.domicile_location = $CHARACTER$.location }
			}
			save_scope_as = kehillah_study_torah_library
		}
	}
}

# Scope: the studying character. Called from the scheme's own on_monthly
# (common/schemes/scheme_types/kehillah_study_torah_scheme.txt) -- the
# "refresh periodically... if you don't own a book anymore" half of the
# user's request; the "detect if there are new books" half needs no extra
# code at all, since kehillah_study_torah_pick_next_work_effect above
# already queries the library's CURRENT contents live, every single time
# it runs (at scheme start and at every phase completion) rather than
# working from any kind of snapshot -- see that effect's own header.
#
# If the specific work currently being read is no longer owned (donated
# away mid-read, per the library panel's own new donate-for-piety action
# -- OR simply because a moved capital/wandered-off adventurer now resolves
# a DIFFERENT community's library entirely), toast the player and re-pick
# immediately rather than let scheme_progress keep building toward a book
# that no longer exists to reward. Progress itself is left untouched -- it
# represents time spent at the Beit Midrash, not attachment to one
# specific volume, so switching what is being read does not reset or
# penalize it.
#
# If the current work is still owned, this is instead where the rare
# intervention events (kehillah_study_torah.0003/.0004) may fire. The
# old 12%/month flavor pulse was removed: at most one intervention can
# occur per course, and it can grant progress or a one-course mastery
# bonus.
kehillah_study_torah_monthly_effect = {
	if = {
		limit = { has_variable = kehillah_study_current_work_id }
		# CHARACTER = scope:owner, NOT root or this -- see kehillah_study_
		# torah_resolve_library_effect's own header just above for why:
		# root here is the SCHEME (this effect runs inside the scheme's own
		# on_monthly), and scope:owner is already saved one level up by
		# that same on_monthly, before this effect was ever called.
		kehillah_study_torah_resolve_library_effect = { CHARACTER = scope:owner }
		if = {
			limit = { exists = scope:kehillah_study_torah_library }
			if = {
				limit = { NOT = { kehillah_study_torah_current_work_still_owned_trigger = { LIBRARY = scope:kehillah_study_torah_library } } }
				send_interface_toast = {
					title = kehillah_study_torah_book_lost_title
					custom_tooltip = kehillah_study_torah_book_lost_tt
					left_icon = this
				}
				kehillah_study_torah_pick_next_work_effect = { LIBRARY = scope:kehillah_study_torah_library }
			}
			else = {
				# At most one interaction pulse per course, and only a small
				# chance each month. This replaces the old 12% monthly flavor
				# event, which made the same book-specific pulse appear too often.
				if = {
					limit = { NOT = { has_variable = kehillah_study_torah_pulse_seen } }
					random = {
						chance = 3
						set_variable = { name = kehillah_study_torah_pulse_seen value = 1 }
						random_list = {
							1 = { trigger_event = kehillah_study_torah.0003 }
							1 = { trigger_event = kehillah_study_torah.0004 }
							1 = { trigger_event = kehillah_study_torah.0005 }
							1 = { trigger_event = kehillah_study_torah.0006 }
							1 = { trigger_event = kehillah_study_torah.0007 }
						}
					}
				}
			}
		}
	}
}

# Scope: the player (the one viewing the roster). Adds the living students of
# their own yeshiva to the global reader index before a roster is built, so a
# graduate who mastered a work at the yeshiva before the yeshiva recorded
# itself in the index (older saves) still shows under "Rabbis who mastered it".
# Cheap: a yeshiva holds a handful of students.
kehillah_lib_index_yeshiva_readers_effect = {
	primary_title ?= {
		every_in_list = {
			variable = kehillah_yeshiva_students
			limit = {
				is_alive = yes
				NOT = { is_target_in_global_variable_list = { name = kehillah_sefer_masters target = this } }
			}
			add_to_global_variable_list = { name = kehillah_sefer_masters target = this }
		}
	}
}
