# ---------------------------------------------------------------------
# kehillah_seed_starting_library_effect
# Scope: a title (a Kehillah community's primary title). Called once per
# registered community from kehillah_on_game_start (common/on_action/
# kehillah_on_actions.txt), AFTER every add_to_global_variable_list call
# that builds kehillah_registered_communities -- mirrors kehillah_
# refresh_map_view_for_players_effect's own placement there exactly, and
# for the same reason: this needs the full registry already built to
# iterate over it.
#
# TWO WORKS, THE SAME TWO FOR EVERY COMMUNITY: Targum Onkelos
# (Parshanut) and the Mishnah (Talmudics) -- see the scripted-triggers
# file's own header for why these two specifically, and why Hashkafa
# starts empty on purpose.
# ---------------------------------------------------------------------

kehillah_seed_starting_library_effect = {
	add_to_variable_list = { name = kehillah_library_works target = flag:work_targum_onkelos }
	add_to_variable_list = { name = kehillah_library_works target = flag:work_mishnah }
}


# =====================================================================
# STUDYING A TRACK -- shared by kehillah_learn_torah.0001 (study locally,
# events/kehillah_learn_torah_events.txt) and kehillah_learn_torah_
# journey.0001 (study at a community you have travelled to, events/
# kehillah_learn_torah_journey_events.txt). ADDED 2026-09-15 at user
# request ("implement the travel to study a book... keep track of which
# books you've already studied... studying the same book a second time
# should give a much smaller boost").
#
# ONE EFFECT PER TRACK, NOT ONE GENERIC ONE -- each track's four works
# are a different set of literal flags, and CK3 parameter substitution
# cannot iterate a list, so three near-identical effects is the actual
# shape this takes, the same reason "Write a Book" and the Translation
# Contract both have one branch per genre/work rather than a single
# generic one. $LIBRARY$ is the one thing that DOES get parameterized
# (which title's kehillah_library_works to read from) -- see common/
# scripted_triggers/kehillah_scripted_triggers.txt's own header for why,
# and for the matching kehillah_owns_<track>_work_trigger these three
# effects assume has already gated their own caller (none of the three
# guards against being called with nothing owned; that is deliberately
# the CALLER's job, exactly like kehillah_translation_pick_work_effect
# assumes a valid pool upstream).
#
# ALREADY STUDIED TRACKING, scope: root (the studying character, NOT the
# library). kehillah_studied_works is a character variable_list -- unlike
# kehillah_library_works, this belongs to the PERSON, not the community,
# because what a scholar has personally read does not reset when they
# travel or when their own community's collection changes. A work
# already in this list on a repeat study pays kehillah_learn_torah_xp_
# repeat_pass/_fail instead of kehillah_learn_torah_xp_pass/_fail (common/
# script_values/kehillah_script_values.txt) -- both real pass/fail tiers
# still exist on a repeat, just much smaller, per explicit request
# ("a much smaller boost", not zero -- rereading a book still teaches
# something, just far less than a first read).
#
# THE "IS THIS A REPEAT" SIGNAL is a plain NUMBER variable (kehillah_
# learn_torah_repeat, 0 or 1), set inside whichever random_list entry
# actually fires, and read immediately after -- NOT is_target_in_
# variable_list called with a variable as its own target. That specific
# form (`target = var:X` where X holds a flag) was checked against the
# installed game and NOT found anywhere as real precedent, so it is not
# risked here; every is_target_in_variable_list call below takes a
# literal flag:X, the confirmed-safe form used everywhere else in this
# mod, and the cross-block signal is an ordinary number comparison
# instead (var:kehillah_learn_torah_repeat = 1), which has extensive
# precedent (e.g. every kehillah_bet_din_case*_direction check).
# =====================================================================

kehillah_study_parshanut_effect = {
	random_list = {
		25 = {
			trigger = { $LIBRARY$ = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_targum_onkelos } } }
			if = {
				limit = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_targum_onkelos } }
				set_variable = { name = kehillah_learn_torah_repeat value = 1 }
			}
			else = {
				set_variable = { name = kehillah_learn_torah_repeat value = 0 }
				add_to_variable_list = { name = kehillah_studied_works target = flag:work_targum_onkelos }
			}
			custom_tooltip = kehillah_study_torah.parshanut.targum_onkelos_tt
		}
		25 = {
			trigger = { $LIBRARY$ = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_mekhilta } } }
			if = {
				limit = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_mekhilta } }
				set_variable = { name = kehillah_learn_torah_repeat value = 1 }
			}
			else = {
				set_variable = { name = kehillah_learn_torah_repeat value = 0 }
				add_to_variable_list = { name = kehillah_studied_works target = flag:work_mekhilta }
			}
			custom_tooltip = kehillah_study_torah.parshanut.mekhilta_tt
		}
		25 = {
			trigger = { $LIBRARY$ = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_genesis_rabbah } } }
			if = {
				limit = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_genesis_rabbah } }
				set_variable = { name = kehillah_learn_torah_repeat value = 1 }
			}
			else = {
				set_variable = { name = kehillah_learn_torah_repeat value = 0 }
				add_to_variable_list = { name = kehillah_studied_works target = flag:work_genesis_rabbah }
			}
			custom_tooltip = kehillah_study_torah.parshanut.genesis_rabbah_tt
		}
		25 = {
			trigger = { $LIBRARY$ = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_pirkei_derabbi_eliezer } } }
			if = {
				limit = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_pirkei_derabbi_eliezer } }
				set_variable = { name = kehillah_learn_torah_repeat value = 1 }
			}
			else = {
				set_variable = { name = kehillah_learn_torah_repeat value = 0 }
				add_to_variable_list = { name = kehillah_studied_works target = flag:work_pirkei_derabbi_eliezer }
			}
			custom_tooltip = kehillah_study_torah.parshanut.pirkei_derabbi_eliezer_tt
		}
	}

	if = {
		limit = { var:kehillah_learn_torah_repeat = 1 }
		if = {
			limit = { learning >= 10 }
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_learn_torah_xp_repeat_pass }
			custom_tooltip = kehillah_study_torah.repeat_pass.tt
		}
		else = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_learn_torah_xp_repeat_fail }
			custom_tooltip = kehillah_study_torah.repeat_fail.tt
		}
	}
	else = {
		if = {
			limit = { learning >= 10 }
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_learn_torah_xp_pass }
			custom_tooltip = kehillah_study_torah.pass.tt
		}
		else = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_learn_torah_xp_fail }
			custom_tooltip = kehillah_study_torah.fail.tt
		}
	}
	add_piety = minor_piety_gain
	remove_variable = kehillah_learn_torah_repeat
}
