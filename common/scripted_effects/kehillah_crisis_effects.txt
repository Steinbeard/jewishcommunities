# 2026-10-29. State on the community title survives a change of leader.
# 2026-11-01: the crisis situation's race (Armed host vs Spared) decides when Agitation ends; the
# preacher story still supplies the weekly agitation, preparations and developments.
kehillah_crisis_begin_effect = {
 primary_title = {
  if = {
   limit = { OR = { NOT = { has_variable = kehillah_crisis_stage } AND = { NOT = { var:kehillah_crisis_stage = 1 } NOT = { var:kehillah_crisis_stage = 2 } } } }
   kehillah_crisis_situation_close_effect = yes
   remove_variable = kehillah_crisis_situation_done
   if = { limit = { NOT = { has_variable = kehillah_crisis_epoch } } set_variable = { name = kehillah_crisis_epoch value = 0 } }
   change_variable = { name = kehillah_crisis_epoch add = 1 }
   set_variable = { name = kehillah_crisis_stage value = 1 }
   # UX test 2026-11-01: the generic and the named opening both fired (+46 at the start).
   set_variable = { name = kehillah_crisis_fresh value = yes days = 1 }
   set_variable = { name = kehillah_crisis_agitation value = 45 }
   set_variable = { name = kehillah_crisis_elapsed value = 0 }
   set_variable = { name = kehillah_crisis_developments value = 0 }
   set_variable = { name = kehillah_crisis_followers value = 0 }
   remove_variable = kehillah_crisis_neighbours_willing
   remove_variable = kehillah_crisis_neighbours_spoke
   remove_variable = kehillah_crisis_answered
   remove_variable = kehillah_crisis_town_won
   remove_variable = kehillah_crisis_discredit_tried
   remove_variable = kehillah_crisis_appeal_sent
   remove_variable = kehillah_crisis_fasted
   remove_variable = kehillah_crisis_silver_sent
   remove_variable = kehillah_crisis_opening
   remove_variable = kehillah_crisis_opening_credited
   remove_variable = kehillah_crisis_forgiven
   remove_variable = kehillah_crisis_sheltered
   remove_variable = kehillah_crisis_barred
   remove_variable = kehillah_crisis_pleaded
   remove_variable = kehillah_granada_letter_found
   remove_variable = kehillah_crisis_struck_month
   remove_variable = kehillah_crisis_announced
   # The pressure toasts' first-time notices (kehillah_crisis_race_monthly_effect).
   remove_variable = kehillah_crisis_told_cold_lord
   remove_variable = kehillah_crisis_told_holy_war
   remove_variable = kehillah_crisis_told_holy_week
   clear_variable_list = kehillah_crisis_protection_credited
   remove_variable = kehillah_crisis_action
   remove_variable = kehillah_crisis_action_days
   remove_variable = kehillah_crisis_supplies
   remove_variable = kehillah_crisis_advocacy
   remove_variable = kehillah_crisis_response
   remove_variable = kehillah_crisis_judgement_sent
   remove_variable = kehillah_crisis_scene_count
   clear_variable_list = kehillah_crisis_developments_seen
   clear_variable_list = kehillah_crisis_scenes_seen
  }
 }
 primary_title = { kehillah_crisis_situation_sync_effect = yes }
}
# Scope: community leader. $DAYS$ is supplied by the owning story's tick.
kehillah_crisis_tick_effect = {
 primary_title = {
  if = {
   limit = { var:kehillah_crisis_stage = 1 }
   change_variable = { name = kehillah_crisis_agitation add = 3 } clamp_variable = { name = kehillah_crisis_agitation min = 5 max = 95 }
   change_variable = { name = kehillah_crisis_elapsed add = $DAYS$ }
   if = {
    limit = { has_variable = kehillah_crisis_action }
    change_variable = { name = kehillah_crisis_action_days add = -$DAYS$ }
    if = {
     limit = { var:kehillah_crisis_action_days <= 0 }
     # Each finished preparation also adds its Spared points to the race (2026-11-01).
     if = { limit = { var:kehillah_crisis_action = 1 } set_variable = { name = kehillah_mob_defended value = yes } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_watch_catalyst } }
     else_if = { limit = { var:kehillah_crisis_action = 2 } set_variable = { name = kehillah_crisis_supplies value = yes } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_supplies_catalyst } }
     else_if = { limit = { var:kehillah_crisis_action = 3 } set_variable = { name = kehillah_mob_scattered value = yes } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_routes_catalyst } }
     else_if = { limit = { var:kehillah_crisis_action = 4 } set_variable = { name = kehillah_crisis_advocacy value = yes } change_variable = { name = kehillah_crisis_agitation add = -20 } clamp_variable = { name = kehillah_crisis_agitation min = 5 max = 95 } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_advocacy_catalyst } }
     # Step 1c: the appeal arrives; its condemnation also protects each month.
     else_if = { limit = { var:kehillah_crisis_action = 5 } set_variable = { name = kehillah_mob_condemned value = yes } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_act_appeal_catalyst } }
     remove_variable = kehillah_crisis_action
     remove_variable = kehillah_crisis_action_days
     holder ?= { if = { limit = { is_ai = no } send_interface_toast = { type = event_toast_effect_good title = kehillah_crisis_prepared_t desc = kehillah_crisis_prepared_d } } }
    }
   }
   # Step 1b: one warning event every 21 days or more while the race runs, none repeated.
   if = {
    limit = { var:kehillah_crisis_elapsed >= kehillah_crisis_event_spacing_days var:kehillah_crisis_developments < 10 kehillah_crisis_title_race_running_trigger = yes }
    set_variable = { name = kehillah_crisis_elapsed value = 0 }
    change_variable = { name = kehillah_crisis_developments add = 1 }
    holder ?= { kehillah_crisis_development_effect = yes }
   }
  }
 }
}
# Scope: community leader. Picks one unseen warning event (step 1b). Players get the event; for an
# AI community, the agitator may gain a follower instead.
kehillah_crisis_development_effect = {
 primary_title.var:kehillah_mob_threat ?= { save_scope_as = kehillah_crisis_agitator_scope }
 if = {
  limit = { is_ai = yes }
  random = {
   chance = kehillah_crisis_ai_follower_chance
   primary_title = { change_variable = { name = kehillah_crisis_followers add = 1 } }
  }
 }
 else = {
  random_list = {
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = threats } } kehillah_crisis_pick_development_effect = { KEY = threats EVENT = kehillah_crisis.0010 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = market } } kehillah_crisis_pick_development_effect = { KEY = market EVENT = kehillah_crisis.0011 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = neighbours } } kehillah_crisis_pick_development_effect = { KEY = neighbours EVENT = kehillah_crisis.0012 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = watch } } kehillah_crisis_pick_development_effect = { KEY = watch EVENT = kehillah_crisis.0013 } }
   12 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = sermon } } kehillah_crisis_pick_development_effect = { KEY = sermon EVENT = kehillah_crisis.0020 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = child } } kehillah_crisis_pick_development_effect = { KEY = child EVENT = kehillah_crisis.0021 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = brawl } } kehillah_crisis_pick_development_effect = { KEY = brawl EVENT = kehillah_crisis.0022 } }
   14 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = follower } } kehillah_crisis_pick_development_effect = { KEY = follower EVENT = kehillah_crisis.0023 } }
   10 = { trigger = { kehillah_crisis_development_unseen_trigger = { KEY = convert } } kehillah_crisis_pick_development_effect = { KEY = convert EVENT = kehillah_crisis.0024 } }
   25 = { trigger = { kehillah_crisis_opening_is_trigger = { KEY = crusade } kehillah_crisis_development_unseen_trigger = { KEY = crusade_event } } kehillah_crisis_pick_development_effect = { KEY = crusade_event EVENT = kehillah_crisis.0030 } }
   25 = { trigger = { kehillah_crisis_opening_is_trigger = { KEY = accusation } kehillah_crisis_development_unseen_trigger = { KEY = accusation_event } } kehillah_crisis_pick_development_effect = { KEY = accusation_event EVENT = kehillah_crisis.0031 } }
   25 = { trigger = { kehillah_crisis_opening_is_trigger = { KEY = debts } kehillah_crisis_development_unseen_trigger = { KEY = debts_event } } kehillah_crisis_pick_development_effect = { KEY = debts_event EVENT = kehillah_crisis.0032 } }
   25 = { trigger = { kehillah_crisis_opening_is_trigger = { KEY = holy_week } kehillah_crisis_development_unseen_trigger = { KEY = holy_week_event } } kehillah_crisis_pick_development_effect = { KEY = holy_week_event EVENT = kehillah_crisis.0033 } }
   # Not in Granada, whose own scenes tell of the poem (2026-11-01, three standing actions).
   25 = { trigger = { kehillah_crisis_opening_is_trigger = { KEY = vizier } NOT = { kehillah_granada_chain_valid_trigger = yes } kehillah_crisis_development_unseen_trigger = { KEY = vizier_event } } kehillah_crisis_pick_development_effect = { KEY = vizier_event EVENT = kehillah_crisis.0034 } }
  }
 }
}
# Records the event as seen and sends it. Its random_list entry checks kehillah_crisis_development_unseen_trigger.

# Resolve once, before any old end effect clears preparations or kills the agitator.
# OUTCOME: 1 dispersed/prevented, 2 defended, 3 bought off, 4 quarter struck.
kehillah_crisis_resolve_effect = {
 if = { limit = { primary_title ?= { NOT = { has_variable = kehillah_crisis_stage } } } kehillah_crisis_begin_effect = yes }
 if = {
  limit = { primary_title ?= { OR = { var:kehillah_crisis_stage = 1 var:kehillah_crisis_stage = 2 } } }
  primary_title = {
   set_variable = { name = kehillah_crisis_stage value = 3 }
   set_variable = { name = kehillah_crisis_outcome value = $OUTCOME$ }
   set_variable = { name = kehillah_crisis_had_protection value = holder.kehillah_crisis_protection_value }
   remove_variable = kehillah_crisis_action
   remove_variable = kehillah_crisis_action_days
   set_variable = { name = kehillah_crisis_judgement_grace value = yes days = 30 }
   save_scope_as = kehillah_crisis_title
   kehillah_crisis_situation_sync_effect = yes
  }
  save_scope_value_as = { name = kehillah_crisis_epoch value = primary_title.var:kehillah_crisis_epoch }
  # Queue visible followups through a hidden relay: an option tooltip must
  # never evaluate events that depend on scopes created earlier in that option.
  domicile.domicile_location.county.holder ?= { trigger_event = { id = kehillah_crisis.0290 days = 1 } }
  debug_log = "kehillah_crisis_resolve_effect: resolved outcome $OUTCOME$; political judgement pending"
 }
}

# --- Step 1e: openings ---

# Scope: any; scope:kehillah_mob_leader, scope:kehillah_mob_victim_title. Records how this crisis
# began, then counts its opening points once. A pending opening (debts, Holy Week) set by the
# quarterly pulse wins; otherwise the agitator decides.
kehillah_crisis_set_opening_effect = {
 scope:kehillah_mob_victim_title = {
  if = {
   limit = { NOT = { has_variable = kehillah_crisis_opening } }
   if = {
    limit = { has_variable = kehillah_crisis_pending_opening }
    set_variable = { name = kehillah_crisis_opening value = var:kehillah_crisis_pending_opening }
   }
   else_if = {
    limit = { scope:kehillah_mob_leader ?= { has_character_flag = kehillah_crisis_granada_agitator } }
    set_variable = { name = kehillah_crisis_opening value = flag:vizier }
   }
   else_if = {
    limit = { scope:kehillah_mob_leader ?= { has_character_flag = kehillah_mob_crusade } }
    set_variable = { name = kehillah_crisis_opening value = flag:crusade }
   }
   else_if = {
    limit = { scope:kehillah_mob_leader ?= { has_character_flag = kehillah_mob_local_preaching } }
    set_variable = { name = kehillah_crisis_opening value = flag:accusation }
   }
   else_if = {
    limit = {
     holder.domicile.domicile_location.county ?= {
      OR = {
       faith ?= { exists = great_holy_war }
       holder.top_liege.faith ?= { exists = great_holy_war }
      }
     }
    }
    set_variable = { name = kehillah_crisis_opening value = flag:crusade }
   }
   else = { set_variable = { name = kehillah_crisis_opening value = flag:preacher } }
  }
  if = { limit = { has_variable = kehillah_crisis_pending_opening } remove_variable = kehillah_crisis_pending_opening }
  kehillah_crisis_credit_opening_effect = yes
 }
}

# Scope: community title. The opening's named catalyst, once, while the race runs.
kehillah_crisis_credit_opening_effect = {
 if = {
  limit = {
   has_variable = kehillah_crisis_opening
   NOT = { has_variable = kehillah_crisis_opening_credited }
   kehillah_crisis_title_race_running_trigger = yes
  }
  set_variable = { name = kehillah_crisis_opening_credited value = yes }
  if = { limit = { var:kehillah_crisis_opening = flag:accusation } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_accusation_catalyst } }
  else_if = { limit = { var:kehillah_crisis_opening = flag:crusade } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_crusade_catalyst } }
  else_if = { limit = { var:kehillah_crisis_opening = flag:debts } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_debts_catalyst } }
  else_if = { limit = { var:kehillah_crisis_opening = flag:holy_week } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_holy_week_catalyst } }
  else_if = { limit = { var:kehillah_crisis_opening = flag:vizier } kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_vizier_catalyst } }
  else = { kehillah_crisis_race_add_effect = { CATALYST = kehillah_crisis_opening_preacher_catalyst } }
  kehillah_crisis_announce_once_effect = yes
 }
}

  kehillah_crisis_announce_effect = yes
 }
}

# Scope: community title, as its crisis opens (2026-11-01). Tells a player leader that the danger
# has begun, how it began, and where to act. Worded by the opening.
kehillah_crisis_announce_effect = {
 save_scope_as = kehillah_crisis_announce_title
 holder.domicile.domicile_location.county ?= { save_scope_as = kehillah_crisis_announce_county }
 var:kehillah_mob_threat ?= { save_scope_as = kehillah_crisis_announce_agitator }
 holder ?= {
  if = {
   limit = { is_ai = no exists = scope:kehillah_crisis_announce_county }
   # A host raised with no warning: the crisis begins already armed.
   if = {
    limit = {
     scope:kehillah_crisis_announce_title.var:kehillah_crisis_stage ?= 2
     scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:rising
    }
    kehillah_crisis_send_announcement_effect = { KEY = rising }
   }
   else_if = {
    limit = {
     scope:kehillah_crisis_announce_title.var:kehillah_crisis_stage ?= 2
     scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:crusade
    }
    kehillah_crisis_send_announcement_effect = { KEY = crusade_host }
   }
   else_if = {
    limit = {
     scope:kehillah_crisis_announce_title.var:kehillah_crisis_stage ?= 2
     NOT = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:vizier }
    }
    kehillah_crisis_send_announcement_effect = { KEY = host }
   }
   else_if = {
    limit = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:vizier exists = scope:kehillah_crisis_announce_agitator }
    kehillah_crisis_send_announcement_effect = { KEY = vizier }
   }
   else_if = {
    limit = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:accusation }
    kehillah_crisis_send_announcement_effect = { KEY = accusation }
   }
   else_if = {
    limit = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:crusade }
    kehillah_crisis_send_announcement_effect = { KEY = crusade }
   }
   else_if = {
    limit = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:debts }
    kehillah_crisis_send_announcement_effect = { KEY = debts }
   }
   else_if = {
    limit = { scope:kehillah_crisis_announce_title.var:kehillah_crisis_opening ?= flag:holy_week }
    kehillah_crisis_send_announcement_effect = { KEY = holy_week }
   }
   else = { kehillah_crisis_send_announcement_effect = { KEY = preacher } }
  }
 }
}
# Scope: the leader. KEY: the opening.

  title = kehillah_crisis_begins_$KEY$_t
  desc = kehillah_crisis_begins_$KEY$_d
  left_icon = scope:kehillah_crisis_announce_title
 }
}

# Scope: community leader, quarterly (players). Debts after a dear year, or Holy Week, may raise a
# local agitator with no preacher in sight. Uses the accusation's local agitator, flavored.
kehillah_crisis_opening_pulse_effect = {
 if = {
  limit = { kehillah_crisis_quiet_trigger = yes }
  random_list = {
   100 = { }
   0 = {
    modifier = {
     add = kehillah_crisis_holy_week_chance
     kehillah_crisis_holy_week_season_trigger = yes
     domicile.domicile_location.county.holder.top_liege.faith.religion ?= religion:christianity_religion
     domicile.domicile_location.county.holder.top_liege ?= { kehillah_outlook_band_value <= 0 }
    }
    kehillah_crisis_start_local_opening_effect = { OPENING = holy_week }
   }
   0 = {
    modifier = { add = kehillah_crisis_debts_chance_value }
    kehillah_crisis_start_local_opening_effect = { OPENING = debts }
   }
  }
 }
}
