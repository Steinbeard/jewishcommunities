# =====================================================================
# A pillar change with its own tooltip line (2026-10-14). Scope: a
# Kehillah leader. PILLAR: greatness, prosperity or stability. AMOUNT: a
# number or a constant script value (it is read in the title's scope).
#
# 2026-11-01 (Daniel): pillar changes are shown as a size, not a number:
# "Stability: Minor penalty", "Greatness: Major bonus". The bands are
# kehillah_pillar_change_*_band below, on the pillars' 0-1000 scale
# (bands of the pillars themselves at 150/400/700/950). The old numeric
# lines (common/effect_localization/kehillah_pillar_effect_localization.txt)
# are no longer used here.
# =====================================================================
kehillah_change_pillar_effect = {
	primary_title ?= {
		save_temporary_scope_value_as = { name = kehillah_pillar_change value = $AMOUNT$ }
	}
	if = {
		limit = { exists = scope:kehillah_pillar_change }
		if = {
			limit = { scope:kehillah_pillar_change > kehillah_pillar_change_major_band }
			custom_tooltip = kehillah_$PILLAR$_bonus_great_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change > kehillah_pillar_change_moderate_band }
			custom_tooltip = kehillah_$PILLAR$_bonus_major_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change > kehillah_pillar_change_minor_band }
			custom_tooltip = kehillah_$PILLAR$_bonus_moderate_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change > 0 }
			custom_tooltip = kehillah_$PILLAR$_bonus_minor_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change >= kehillah_pillar_change_minor_band_neg }
			custom_tooltip = kehillah_$PILLAR$_penalty_minor_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change >= kehillah_pillar_change_moderate_band_neg }
			custom_tooltip = kehillah_$PILLAR$_penalty_moderate_tt
		}
		else_if = {
			limit = { scope:kehillah_pillar_change >= kehillah_pillar_change_major_band_neg }
			custom_tooltip = kehillah_$PILLAR$_penalty_major_tt
		}
		else = {
			custom_tooltip = kehillah_$PILLAR$_penalty_severe_tt
		}
	}
	hidden_effect = {
		primary_title ?= {
			change_variable = { name = kehillah_var_$PILLAR$ add = $AMOUNT$ }
			clamp_variable = { name = kehillah_var_$PILLAR$ min = 0 max = 100000 }
		}
	}
}
