# Scope: a ruler. Puts the right "beyond the teaching" modifier on a ruler
# whose realm holds the Jews Favored and clears it from anyone else. Run
# by kehillah_settlement_policy_changed_effect on every change of the law,
# and yearly (kehillah_status_yearly), which catches a changed teaching or
# culture.
kehillah_favored_cost_refresh_effect = {
	if = {
		limit = { has_character_modifier = kehillah_favored_beyond_teaching_1_modifier }
		remove_character_modifier = kehillah_favored_beyond_teaching_1_modifier
	}
	if = {
		limit = { has_character_modifier = kehillah_favored_beyond_teaching_2_modifier }
		remove_character_modifier = kehillah_favored_beyond_teaching_2_modifier
	}
	if = {
		limit = { has_character_modifier = kehillah_favored_beyond_teaching_3_modifier }
		remove_character_modifier = kehillah_favored_beyond_teaching_3_modifier
	}
	if = {
		limit = { has_character_modifier = kehillah_favored_beyond_teaching_4_modifier }
		remove_character_modifier = kehillah_favored_beyond_teaching_4_modifier
	}
	if = {
		limit = { kehillah_favored_cost_tier_value = 4 }
		add_character_modifier = { modifier = kehillah_favored_beyond_teaching_4_modifier }
	}
	else_if = {
		limit = { kehillah_favored_cost_tier_value = 3 }
		add_character_modifier = { modifier = kehillah_favored_beyond_teaching_3_modifier }
	}
	else_if = {
		limit = { kehillah_favored_cost_tier_value = 2 }
		add_character_modifier = { modifier = kehillah_favored_beyond_teaching_2_modifier }
	}
	else_if = {
		limit = { kehillah_favored_cost_tier_value = 1 }
		add_character_modifier = { modifier = kehillah_favored_beyond_teaching_1_modifier }
	}
}

# Scope: a ruler (yearly_playable_pulse). A realm whose rite has come to
# teach Exclusion or Abolition cannot hold the Jews Favored: it drops one
# rung to Tolerated, with the usual notices and cooldown. Then the
# modifier is refreshed.
kehillah_status_yearly_effect = {
	if = {
		limit = {
			kehillah_ruler_favors_jews_trigger = yes
			kehillah_teaching_exclusionary_trigger = yes
		}
		debug_log = "kehillah_status_yearly_effect: an exclusionary teaching ends Favored; lowering to Tolerated"
		kehillah_lower_jewish_settlement_policy_effect = yes
	}
	kehillah_favored_cost_refresh_effect = yes
}
