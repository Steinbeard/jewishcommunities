# English conquest invitation, revised 2026-10-29.
# The first post-Godwinson holder of England gets a one-time ruler decision.
# Taking it passes the existing Encouraged Jewish Settlement law, whose
# on_after_pass sends the ordinary invitation through its normal mechanic.
# The letter has special English-conquest wording. A hidden review one year
# later counts actual English communities and founds vacant named sites only
# until there are three. The S8 London player-adventurer offer still goes
# out immediately when an eligible player exists.
# See docs/spec/2026-10-29-conquest-invitation-revision.md. That addendum
# supersedes V18 sections 1.4, 1.6 and 1.7's immediate-four founding.
# Scope: the character who just gained title:k_england (root, per vanilla
# on_title_gain's own documented scopes).
kehillah_norman_conquest_evaluate_effect = {
	if = {
		limit = { dynasty = dynasty:756 }
		debug_log = "kehillah_norman_conquest_evaluate_effect: still House Godwinson, awaiting a real change of house"
	}
	else = {
		set_global_variable = { name = kehillah_norman_conquest_resolved value = yes }
		title:k_england = { set_variable = { name = kehillah_norman_invitation_decision_open value = yes years = 5 } }
		debug_log = "kehillah_norman_conquest_evaluate_effect: invitation decision opened for England's conqueror"
	}
}

# Scope: the ruler taking the decision. S8's player offer is made at once
# if possible. A newly eligible player may also receive it at the review
# if London remains vacant. Both routes use .0010-.0012 and their single
# pending-offer variable.
kehillah_norman_conquest_execute_effect = {
	debug_log = "kehillah_norman_conquest_execute_effect: English invitation issued; founding review in one year"
	save_scope_as = kehillah_norman_king
	save_scope_as = kehillah_policy_host
	kehillah_settlement_laws_normalize_effect = yes
	if = {
		limit = { dynasty = dynasty:752 }
		set_global_variable = { name = kehillah_norman_new_leader_dynasty_choice value = 1 }
	}
	else_if = {
		limit = { OR = { dynasty = dynasty:499 culture = culture:norwegian } }
		set_global_variable = { name = kehillah_norman_new_leader_dynasty_choice value = 2 }
	}
	else = { set_global_variable = { name = kehillah_norman_new_leader_dynasty_choice value = 3 } }
	primary_title = {
		set_variable = { name = kehillah_norman_special_invitation value = yes years = 2 }
		set_variable = { name = kehillah_policy_reason value = flag:towns }
	}
	if = {
		limit = { primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes } }
		primary_title = { save_scope_as = kehillah_policy_realm }
		kehillah_invitation_issue_effect = yes
	}
	else = {
		# A newly independent conqueror can inherit the first law of this
		# group (Banned). Establish the ordinary Allowed baseline silently,
		# then pass Encouraged through the law's full on_after_pass path.
		if = {
			limit = { NOT = { primary_title = { kehillah_jewish_settlement_policy_allowed_trigger = yes } } }
			if = {
				limit = { primary_title = { kehillah_jewish_settlement_policy_banned_trigger = yes } }
				kehillah_expulsion_lift_effect = yes
			}
			kehillah_set_settlement_law_effect = { RUNG = allowed VALUE = 2 }
		}
		add_realm_law = kehillah_jewish_settlement_encouraged
		# 2026-10-30 (the Nagid spec section 6): raising the law invites by
		# itself only for an AI ruler. This decision is itself the invitation,
		# so a player king who took it invites here.
		if = {
			limit = { NOT = { primary_title = { has_variable = kehillah_invitation_open } } }
			primary_title = { save_scope_as = kehillah_policy_realm }
			kehillah_invitation_issue_effect = yes
		}
	}
	if = {
		limit = {
			any_player = { kehillah_norman_london_invitation_eligible_trigger = yes }
			title:c_middlesex = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
		}
		set_global_variable = { name = kehillah_norman_inviting_king value = scope:kehillah_norman_king }
		random_player = {
			limit = { kehillah_norman_london_invitation_eligible_trigger = yes }
			set_global_variable = { name = kehillah_norman_london_offer_pending value = this }
			trigger_event = { id = kehillah_norman_conquest.0010 days = 3 }
			trigger_event = { id = kehillah_norman_conquest.0012 days = 60 }
		}
		trigger_event = { id = kehillah_norman_conquest.0012 days = 60 }
	}
	title:k_england = { set_variable = { name = kehillah_norman_founding_review_pending value = yes } }
	trigger_event = { id = kehillah_norman_conquest.0002 years = 1 }
	trigger_event = { id = kehillah_norman_conquest.0001 days = 2 }
}

# Runs only from the hidden one-year review, never from a decision tooltip.
# Count actual communities in England, including player and ordinary settler
# foundings; fill only vacant named sites until there are three.
kehillah_norman_conquest_fill_shortfall_effect = {
	title:k_england = { set_variable = { name = kehillah_norman_english_count value = 0 } }
	every_in_global_list = {
		variable = kehillah_registered_communities
		limit = {
			exists = holder.domicile.domicile_location.county
			holder.domicile.domicile_location.county = { target_is_de_jure_liege_or_above = title:k_england }
		}
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	# A pending London offer reserves one of the three places until the
	# player answers or the existing sixty-day fallback resolves it.
	if = {
		limit = {
			exists = global_var:kehillah_norman_london_offer_pending
			title:c_middlesex = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
		}
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	if = {
		limit = {
			title:k_england = { var:kehillah_norman_english_count < 3 }
			title:c_middlesex = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
			title:c_middlesex.holder.top_liege = root
			NOT = { exists = global_var:kehillah_norman_london_offer_pending }
		}
		if = {
			limit = { any_player = { kehillah_norman_london_invitation_eligible_trigger = yes } }
			set_global_variable = { name = kehillah_norman_inviting_king value = this }
			random_player = {
				limit = { kehillah_norman_london_invitation_eligible_trigger = yes }
				set_global_variable = { name = kehillah_norman_london_offer_pending value = this }
				trigger_event = { id = kehillah_norman_conquest.0010 days = 3 }
				trigger_event = { id = kehillah_norman_conquest.0012 days = 60 }
			}
			trigger_event = { id = kehillah_norman_conquest.0012 days = 60 }
		}
		else = { kehillah_norman_found_london_effect = yes }
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	if = {
		limit = {
			title:k_england = { var:kehillah_norman_english_count < 3 }
			title:c_north_riding = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
			title:c_north_riding.holder.top_liege = root
		}
		kehillah_norman_found_york_effect = yes
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	if = {
		limit = {
			title:k_england = { var:kehillah_norman_english_count < 3 }
			title:c_lincolnshire = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
			title:c_lincolnshire.holder.top_liege = root
		}
		kehillah_norman_found_lincoln_effect = yes
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	if = {
		limit = {
			title:k_england = { var:kehillah_norman_english_count < 3 }
			title:c_norfolk = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
			title:c_norfolk.holder.top_liege = root
		}
		kehillah_norman_found_norwich_effect = yes
		title:k_england = { change_variable = { name = kehillah_norman_english_count add = 1 } }
	}
	title:k_england = {
		remove_variable = kehillah_norman_english_count
		remove_variable = kehillah_norman_founding_review_pending
	}
}

# The four founding effects. Each mirrors kehillah_found_community_
# effect's own post-creation checklist (pillars, registry, network
# cache, library seed, map view, host charter) exactly, since a
# dynamically-created Kehillah needs precisely the same bookkeeping
# regardless of how it came to exist. None needs that effect's own
# "destroy the old adventurer title" step -- these are BRAND NEW
# characters with no prior title to dispose of.
#
# Numbers (gold/Prosperity/Greatness) are carried over verbatim from the
# old, now-removed kehillah_setup_london/york/lincoln/norwich_start_
# effect (common/scripted_effects/kehillah_scripted_effects.txt's own
# removal note) -- this retcon changes WHEN these communities start
# existing, not their relative standing once they do.
#
# DYNASTY, per the user's own explicit "personal connections to other
# Ashkenazi communities -- especially French communities if William
# wins" request. global_var:kehillah_norman_new_leader_dynasty_choice
# (set once, above): 1 = William branch, given dynn_Yitzhaki (9000003) --
# Rashi of Troyes's own real, already-defined dynasty (common/dynasties/
# ha_levi.txt) -- read as kin of his household crossing the Channel in
# William's own wake, the strongest, most concrete realization of
# "personal connection" this mod's existing infrastructure can give (a
# real, in-game-verifiable shared house with an existing community's own
# named leader). 2 = Harald branch and 3 = other branch both use
# `dynasty = generate` (no specific household record) -- see the FIXED
# note below for why the Harald branch's own originally-planned
# dynn_Bacharach tie was dropped, not just re-implemented differently.
#
# FIXED 2026-09-25, found live, after THREE failed runtime mechanisms in
# a row -- do not attempt a fourth guess at joining a pre-existing
# dynasty/house directly from inside create_character without a live
# re-test; use the mechanism below, which is grounded in two
# independently-proven primitives instead of a new one:
#   1. `dynasty = dynasty:9000003` inside create_character does NOT
#      reliably join a freshly created character to a pre-existing
#      dynasty -- confirmed live, it silently lands the character in an
#      unrelated generated dynasty instead, durably (re-checked a full
#      day later, still wrong, so not a same-tick read-timing issue).
#   2. `set_dynasty` (referenced in effect_localization/
#      00_character_effects.txt) is dead/vestigial localization, not a
#      real scriptable effect in this build -- confirmed live: "Unknown
#      effect: set_dynasty".
#   3. `dynasty_house = house:<key>` (real, repeated vanilla precedent:
#      common/scripted_effects/00_almohad_invasion_effects.txt:104,145)
#      still failed for a BRAND NEW, custom, zero-population house
#      defined only in common/dynasty_houses/ -- "Failed to fetch a
#      valid house", even after a full clean relaunch. Strong suspicion,
#      not fully proven: a house needs at least one history-file member
#      before the engine ever instantiates it as a queryable object.
# THE MECHANISM ACTUALLY USED instead combines two primitives already
# independently proven to work, rather than a fourth guess at the same
# untested runtime path: bare `dynasty = 9000003` in a HISTORY FILE (the
# exact syntax Rashi's own entry already uses successfully, never in
# doubt tonight -- only the RUNTIME script-effect equivalent was), plus
# `father = character:X` + `dynasty = inherit` inside create_character
# (real, confirmed vanilla precedent used in shipped, tested content:
# common/scripted_effects/00_mongol_invasion_effects.txt:1717-1723, the
# actual mechanism behind Genghis Khan's own dynamically generated
# children). history/characters/yitzhaki_anglia_anchor.txt adds
# character 9000218, "Meir" -- an elder kinsman of Rashi's own household,
# already deceased by 1066 on purpose (a living anchor would need to
# survive up to 24 years of unprotected simulation before ever being
# referenced) -- solely so the William branch below can set `father =
# character:9000218` + `dynasty = inherit` rather than reference a
# dynasty or house directly at all. See that history file's own header
# for the full account, including the one part still not independently
# confirmed live: whether `father =` referencing an already-deceased
# character resolves cleanly for a SCRIPTED (not organic) birth.
#
# THE HARALD BRANCH'S OWN dynn_Bacharach TIE WAS DROPPED, not reimplemented
# with the new mechanism -- it never had a living reference character to
# begin with (no existing community in this mod's roster has a natural
# North Sea/Scandinavian trade connection), so there was no equivalent
# "already-deceased anchor" to build for it the way Yitzhaki has Meir.
# It was always documented as the deliberately lighter of the two ties;
# now it is `dynasty = generate`, same as the other branch, rather than
# inventing a second anchor character for a connection this feature never
# actually needed to be concrete.
# ---------------------------------------------------------------------

# ---------------------------------------------------------------------
# Shared finishing effect. Scope: the newly created leader, immediately
# after create_adventurer_title -- the exact same scope shape kehillah_
# found_community_effect uses for its own equivalent block, so this
# reuses the identical exists-guard idiom for the identical reason (a
# decision confirmation-tooltip preview walks this effect against a title
# that does not exist yet during a dry run; there is no decision here,
# but create_character/create_adventurer_title inside a trigger_event's
# own immediate block CAN still be tooltip-previewed by a hovering
# player if this is ever surfaced through UI later, so the guard is kept
# as a defensive match to established practice, not dead code).
#
# primary_title is already set correctly by government = kehillah_
# government on create_adventurer_title above (this is the character's
# only title), so unlike kehillah_found_community_effect there is no
# earlier adventurer title to make way for and no set_primary_title_to
# call is needed here.
#
# Prosperity/Greatness/gold values are branch-INDEPENDENT (same four
# numbers regardless of which claimant won) and per-CITY, read off
# root's own held title name via a direct city check -- simpler than
# passing the number set as a parameter, and every one of these calls
# fires from a scope where root is unambiguous (the newly created
# leader, holding exactly one title).
# ---------------------------------------------------------------------
kehillah_norman_conquest_finish_founding_effect = {
	if = {
		limit = { exists = primary_title }
		hidden_effect = {
			if = {
				limit = {
					exists = scope:kehillah_norman_london_title
					primary_title = scope:kehillah_norman_london_title
				}
				add_gold = 60
				primary_title = {
					kehillah_init_pillars_effect = yes
					change_variable = { name = kehillah_var_prosperity add = 220 }
					change_variable = { name = kehillah_var_greatness add = 15 }
				}
			}
			else_if = {
				limit = {
					exists = scope:kehillah_norman_york_title
					primary_title = scope:kehillah_norman_york_title
				}
				add_gold = 30
				primary_title = {
					kehillah_init_pillars_effect = yes
					change_variable = { name = kehillah_var_prosperity add = 80 }
					change_variable = { name = kehillah_var_greatness add = 5 }
				}
			}
			else_if = {
				limit = {
					exists = scope:kehillah_norman_lincoln_title
					primary_title = scope:kehillah_norman_lincoln_title
				}
				add_gold = 30
				primary_title = {
					kehillah_init_pillars_effect = yes
					change_variable = { name = kehillah_var_prosperity add = 80 }
					change_variable = { name = kehillah_var_greatness add = 5 }
				}
			}
			else_if = {
				limit = {
					exists = scope:kehillah_norman_norwich_title
					primary_title = scope:kehillah_norman_norwich_title
				}
				add_gold = 30
				primary_title = {
					kehillah_init_pillars_effect = yes
					change_variable = { name = kehillah_var_prosperity add = 80 }
					change_variable = { name = kehillah_var_greatness add = 10 }
				}
			}

			kehillah_update_leader_flavor_effect = yes

			# Persistent marker, ADDED 2026-09-25 alongside the debug harness
			# (events/kehillah_debug_events.txt's kehillah_debug.93) -- a
			# saved effect scope (scope:kehillah_norman_london_title etc.)
			# does not survive past the effect call that created it, so
			# anything that needs to find these four titles again later
			# (a debug probe, or future flavor content) needs a real,
			# persistent way to recognize "this was founded by the Norman
			# Conquest event" that isn't just re-deriving it from the
			# title's own display name.
			primary_title = {
				kehillah_generate_community_coa_effect = yes
				set_variable = { name = kehillah_norman_founded value = yes }
				set_variable = { name = kehillah_founded_year value = current_year }
			}
			if = {
				limit = { exists = domicile.domicile_location.county }
				domicile.domicile_location.county = { save_scope_as = kehillah_founding_county }
				primary_title = { save_scope_as = kehillah_new_community_title }
				kehillah_history_record_founding_effect = yes
			}

			add_to_global_variable_list = {
				name = kehillah_registered_communities
				target = primary_title
			}
			kehillah_refresh_all_settlement_networks_effect = yes
			primary_title = {
				kehillah_seed_starting_library_effect = yes
			}
			kehillah_refresh_map_view_for_players_effect = yes

			# The decision already passed Encouraged before sending letters.
			# Founding never re-adds that law: repeated skip-effects calls
			# error when the ruler already holds it.
			kehillah_maintain_host_charter_effect = yes
			kehillah_invitation_answered_effect = yes
		}
	}
	else = {
		debug_log = "kehillah_norman_conquest_finish_founding_effect: WARNING no primary_title after create_adventurer_title -- founding failed"
	}
}
