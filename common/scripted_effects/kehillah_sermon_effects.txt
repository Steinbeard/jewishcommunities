# Character scope: the rabbinic leader, or any Karaite leader (2026-11-08:
# Karaites have no clergy; Religious Scholar depth stands in for Parshanut).
# PILLAR is greatness or stability.
# Assess Parshanut before hopping to the community title for the reward.
kehillah_deliver_sermon_effect = {
	if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 100 }
			}
		}
		kehillah_change_pillar_effect = { PILLAR = $PILLAR$ AMOUNT = kehillah_sermon_pillar_gain_master }
	}
	else_if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 65 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 65 }
			}
		}
		kehillah_change_pillar_effect = { PILLAR = $PILLAR$ AMOUNT = kehillah_sermon_pillar_gain_adept }
	}
	else_if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 30 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 30 }
			}
		}
		kehillah_change_pillar_effect = { PILLAR = $PILLAR$ AMOUNT = kehillah_sermon_pillar_gain_studied }
	}
	else = {
		kehillah_change_pillar_effect = { PILLAR = $PILLAR$ AMOUNT = kehillah_sermon_pillar_gain_novice }
	}
	add_piety = kehillah_sermon_piety_gain
	# 2026-11-08: a Karaite preacher (Karaites have no clergy) grows in
	# Religious Scholar instead, if he holds it.
	if = {
		limit = { has_trait = kehillah_rabbi_trait }
		add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_sermon_parshanut_xp }
	}
	else = {
		kehillah_karaite_scholar_xp_effect = { XP = kehillah_sermon_parshanut_xp }
	}
	# 2026-11-08 tenet mechanics: Emunah Temimah -- the plain faith preached
	# steadies the community, and the preacher who lives it gains more.
	if = {
		limit = { rite ?= { rite_has_parameter = kehillah_tenet_emunah_sermon } }
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_tenet_emunah_sermon_stability }
	}
	if = {
		limit = { has_personal_tenet_flag = kehillah_tenet_emunah_sermon_personal }
		add_piety = kehillah_tenet_emunah_sermon_piety
	}
}
