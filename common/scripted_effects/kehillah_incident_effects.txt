# =====================================================================
# V41 step 4a -- the incident director (docs/spec/v41-danger-incidents-
# and-the-pogrom-unfolding.md section 7), 2026-10-14. One quarterly roll
# per community against the danger of violence (kehillah_danger_value)
# picks one incident that fits the host and the time. It replaces the
# separate fixed-chance checks of V37 (accusation) and V38 (disputation,
# friars). At most one incident a year. A mob is rare by itself; most
# mobs grow out of accusations.
# Scope: a Kehillah leader, quarterly (kehillah_on_actions.txt).
# =====================================================================

kehillah_incident_director_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = primary_title
			exists = domicile.domicile_location.county.holder.top_liege
			domicile.domicile_location.county.holder.top_liege = {
				NOT = { is_jewish_character_trigger = yes }
				OR = {
					faith.religion = religion:christianity_religion
					faith.religion = religion:islam_religion
				}
			}
			primary_title = {
				NOT = { has_variable = kehillah_incident_cooldown }
				NOT = { has_variable = kehillah_mob_threat }
			}
			kehillah_incident_chance_value > 0
		}
		random = {
			chance = kehillah_incident_chance_value
			primary_title = { set_variable = { name = kehillah_incident_cooldown value = yes years = 1 } }
			domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_incident_host }
			# Which community (the soak test counts incidents per community).
			debug_log_scopes = yes
			random_list = {
				# An accusation: the blood libel, or blasphemy.
				30 = {
					trigger = { primary_title = { NOT = { has_variable = kehillah_accusation_cooldown } } }
					# Before the blood libel (Norwich, 1144) a Christian town's
					# charges against the Jews were rarer (2026-10-18).
					modifier = {
						factor = 0.5
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						current_year < kehillah_accusation_blood_libel_year
					}
					primary_title = {
						set_variable = { name = kehillah_accusation_cooldown value = yes years = kehillah_accusation_cooldown_years }
					}
					trigger_event = kehillah_accusation.0001
					debug_log = "kehillah_incident_director_effect: an accusation"
				}
				# A disputation (Christian hosts). Not before
				# kehillah_incident_christian_late_year either (2026-10-18:
				# the summoned public disputation is a 12th-13th century
				# form, like the friars and the Talmud trial).
				20 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						current_year >= kehillah_incident_christian_late_year
						primary_title = { NOT = { has_variable = kehillah_disputation_cooldown } }
					}
					modifier = {
						add = 15
						scope:kehillah_incident_host = {
							OR = {
								kehillah_teaching_strict_trigger = yes
								kehillah_teaching_exclusionary_trigger = yes
							}
						}
					}
					primary_title = {
						set_variable = { name = kehillah_disputation_cooldown value = yes years = kehillah_disputation_cooldown_years }
					}
					trigger_event = kehillah_mission.0001
					debug_log = "kehillah_incident_director_effect: a disputation"
				}
				# Friars in the synagogue (Christian hosts). Not before
				# kehillah_incident_christian_late_year (2026-10-15, 20-year soak:
				# the mendicant orders are 13th-century; 1110 is Daniel's floor).
				20 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						current_year >= kehillah_incident_christian_late_year
						primary_title = { NOT = { has_variable = kehillah_sermon_cooldown } }
					}
					modifier = {
						add = 15
						scope:kehillah_incident_host = {
							OR = {
								kehillah_teaching_strict_trigger = yes
								kehillah_teaching_exclusionary_trigger = yes
							}
						}
					}
					primary_title = {
						set_variable = { name = kehillah_sermon_cooldown value = yes years = kehillah_sermon_cooldown_years }
					}
					trigger_event = kehillah_mission.0010
					debug_log = "kehillah_incident_director_effect: friars in the synagogue"
				}
				# The Talmud on trial (V41 step 4b): Christian hosts, a library
				# with the Talmud; likelier under a strict teaching and after 1230.
				# Not before kehillah_incident_christian_late_year (2026-10-15,
				# 20-year soak: it fired in 1085; Paris was 1240).
				10 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						current_year >= kehillah_incident_christian_late_year
						primary_title = {
							kehillah_has_talmud_trigger = yes
							NOT = { has_variable = kehillah_talmud_cooldown }
						}
					}
					modifier = {
						add = 10
						scope:kehillah_incident_host = {
							OR = {
								kehillah_teaching_strict_trigger = yes
								kehillah_teaching_exclusionary_trigger = yes
							}
						}
					}
					modifier = {
						add = 10
						current_year >= 1230
					}
					trigger_event = kehillah_incident.0001
					debug_log = "kehillah_incident_director_effect: the Talmud on trial"
				}
				# The badge, or the ghiyar: likelier under a strict teaching and,
				# for Christian hosts, after Lateran IV (1215). A Christian host
				# not before kehillah_incident_christian_late_year (2026-10-15,
				# 20-year soak: a badge in 1080); the ghiyar has no floor
				# (al-Hakim's decrees, 1009).
				10 = {
					trigger = {
						primary_title = { NOT = { has_variable = kehillah_badge } }
						OR = {
							scope:kehillah_incident_host = { faith.religion = religion:islam_religion }
							current_year >= kehillah_incident_christian_late_year
						}
					}
					modifier = {
						add = 10
						scope:kehillah_incident_host = {
							OR = {
								kehillah_teaching_strict_trigger = yes
								kehillah_teaching_exclusionary_trigger = yes
							}
						}
					}
					modifier = {
						add = 10
						OR = {
							current_year >= 1215
							scope:kehillah_incident_host = { faith.religion = religion:islam_religion }
						}
					}
					trigger_event = kehillah_incident.0010
					debug_log = "kehillah_incident_director_effect: the badge"
				}
				# The servants' decree (2026-10-15): the early restriction --
				# no Christian servant or nurse in a Jewish house, no Jew in an
				# office over Christians (Gregory VII's Roman council, 1078; his
				# letter to Alfonso VI, 1081). Christian hosts, from the start;
				# likelier under a strict teaching, rarer once the badge exists.
				15 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						primary_title = {
							NOT = { has_variable = kehillah_servants_decree }
							NOT = { has_variable = kehillah_servants_decree_cooldown }
						}
					}
					modifier = {
						add = 10
						scope:kehillah_incident_host = {
							OR = {
								kehillah_teaching_strict_trigger = yes
								kehillah_teaching_exclusionary_trigger = yes
							}
						}
					}
					modifier = {
						add = -10
						current_year >= 1215
					}
					primary_title = {
						set_variable = { name = kehillah_servants_decree_cooldown value = yes years = kehillah_servants_decree_cooldown_years }
					}
					trigger_event = kehillah_incident.0040
					debug_log = "kehillah_incident_director_effect: the servants' decree"
				}
				# Holy Week: Christian hosts, March and April.
				40 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						current_month >= 3
						current_month <= 4
						primary_title = {
							NOT = { has_variable = kehillah_holy_week_redeemed }
							NOT = { has_variable = kehillah_holy_week_cooldown }
						}
					}
					trigger_event = kehillah_incident.0020
					debug_log = "kehillah_incident_director_effect: Holy Week"
				}
				# The bonds: a lord of the host's realm owes us; likelier in hard
				# times.
				10 = {
					trigger = {
						kehillah_has_host_realm_debtor_trigger = yes
						primary_title = { NOT = { has_variable = kehillah_bonds_cooldown } }
					}
					modifier = {
						add = 15
						kehillah_danger_hard_times_value >= 15
					}
					trigger_event = kehillah_incident.0030
					debug_log = "kehillah_incident_director_effect: the bonds"
				}
				# A mob, by itself: rare, likelier in hard times; at most once in
				# 15 years.
				5 = {
					trigger = { primary_title = { NOT = { has_variable = kehillah_mob_alone_cooldown } } }
					modifier = {
						add = 10
						kehillah_danger_hard_times_value >= 15
					}
					primary_title = { set_variable = { name = kehillah_mob_alone_cooldown value = yes years = 15 } }
					trigger_event = kehillah_mob.0030
					debug_log = "kehillah_incident_director_effect: a mob"
				}
				# A child stolen to the font (V43 slice 4; the legend of Elhanan
				# of Mainz): rare, Christian hosts, once in 20 years.
				3 = {
					trigger = {
						scope:kehillah_incident_host = { faith.religion = religion:christianity_religion }
						primary_title = { NOT = { has_variable = kehillah_child_theft_cooldown } }
						any_courtier = {
							is_adult = no
							age >= 2
							is_jewish_character_trigger = yes
							father ?= { is_alive = yes }
						}
					}
					modifier = {
						add = 4
						kehillah_danger_value >= 60
					}
					primary_title = { set_variable = { name = kehillah_child_theft_cooldown value = yes years = 20 } }
					kehillah_child_stolen_effect = yes
					debug_log = "kehillah_incident_director_effect: a child stolen"
				}
				# Talk in the market and nothing more, when nothing else fits.
				20 = {
					debug_log = "kehillah_incident_director_effect: only talk"
				}
			}
		}
	}
}
