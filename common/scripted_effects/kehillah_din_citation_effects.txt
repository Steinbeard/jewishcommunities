# The Bet Din docket by order of the Mishnah (2026-11-11, Daniel).
# docs/spec/2026-11-11-rulings-sermons-and-gates.md section 2.
#
# A source option calls kehillah_din_cite_source_effect: the ruling gains a
# strong tally, the citing judge's possible authorities are found among the
# works he has mastered on the case's order, and kehillah_din_cite.0001
# asks which one the court names: the most established (grounded), a newer
# one (innovative), or one whose tenets he holds (by conviction). The named
# work counts a citation toward its standing when the ruling is good
# (kehillah_bet_din_reward_judges_for_ruling_effect).
# =====================================================================

# Scope: the event's root, in the Bet Din. $JUDGE$ is the citing judge
# (scope:kehillah_av_beit_din or root).
#
# 2026-11-23 (Daniel; docs/spec/2026-11-23-bet-din-citation-strength-and-
# risk.md): naming a source no longer adds a flat tally. Each work the court
# could name gets two values, shown on kehillah_din_cite.0001's options:
#   strength    what the ruling gains if the argument holds. Higher for a
#               work few scholars have mastered (a view the parties have not
#               already weighed): kehillah_din_cite_strength_max, less
#               kehillah_din_cite_strength_per_master for each master in the
#               registered communities, never below _min.
#   reliability the chance it holds: the work's standing in the judge's rite
#               and the judge's Learning (kehillah_din_cite_reliability_*).
# A failed argument costs kehillah_din_cite_fail_tally and names no work.
kehillah_din_cite_source_effect = {
	involved_activity = {
		save_scope_as = kehillah_din_cite_activity
		set_variable = { name = kehillah_din_citing_judge value = $JUDGE$ }
	}
	$JUDGE$ = {
		kehillah_din_case_$CASE$_pick_effect = yes
		kehillah_din_cite_measure_effect = { SLOT = grounded }
		kehillah_din_cite_measure_effect = { SLOT = innovative }
		kehillah_din_cite_measure_effect = { SLOT = conviction }
	}
	trigger_event = kehillah_din_cite.0001
}

# Scope: the citing judge; scope:kehillah_din_cite_activity. Strength and
# reliability of the work in $SLOT$, if the pick found one.
kehillah_din_cite_measure_effect = {
	if = {
		limit = { exists = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ }
		save_scope_as = kehillah_din_measuring_judge
		# Evaluate the host community's rite, not the citing judge's rite.
		scope:kehillah_din_cite_activity = { remove_variable = kehillah_din_cite_$SLOT$_host_standing }
		scope:kehillah_din_cite_activity.var:kehillah_bet_din_convening_title.holder ?= {
			save_scope_as = kehillah_din_cite_host
			kehillah_din_cite_host_standing_dispatch_effect = { SOURCE = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ SLOT = $SLOT$ }
		}
		# (Learning - pivot) * per-learning, held on the judge for a moment.
		set_variable = { name = kehillah_din_cite_learning_bonus value = learning }
		change_variable = { name = kehillah_din_cite_learning_bonus subtract = kehillah_din_cite_reliability_learning_pivot }
		change_variable = { name = kehillah_din_cite_learning_bonus multiply = kehillah_din_cite_reliability_per_learning }
		scope:kehillah_din_cite_activity = {
			set_variable = { name = kehillah_din_cite_$SLOT$_masters value = 0 }
			every_in_global_list = {
				variable = kehillah_registered_communities
				limit = { exists = holder }
				holder = {
					if = {
						limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ } }
						scope:kehillah_din_cite_activity = { change_variable = { name = kehillah_din_cite_$SLOT$_masters add = 1 } }
					}
					every_courtier = {
						limit = { is_target_in_variable_list = { name = kehillah_mastered_works target = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ } }
						scope:kehillah_din_cite_activity = { change_variable = { name = kehillah_din_cite_$SLOT$_masters add = 1 } }
					}
				}
			}
			# Strength: max less per-master, floored at min.
			set_variable = { name = kehillah_din_cite_$SLOT$_strength value = var:kehillah_din_cite_$SLOT$_masters }
			change_variable = { name = kehillah_din_cite_$SLOT$_strength multiply = kehillah_din_cite_strength_per_master_neg }
			change_variable = { name = kehillah_din_cite_$SLOT$_strength add = kehillah_din_cite_strength_max }
			clamp_variable = { name = kehillah_din_cite_$SLOT$_strength min = kehillah_din_cite_strength_min max = kehillah_din_cite_strength_max }
			# Reliability: base + standing * per-standing + (Learning - pivot) * per-learning.
			# The pick left the work's standing in the judge's rite on the judge.
			set_variable = { name = kehillah_din_cite_$SLOT$_reliability value = scope:kehillah_din_measuring_judge.var:kehillah_din_cite_$SLOT$_standing }
			change_variable = { name = kehillah_din_cite_$SLOT$_reliability multiply = kehillah_din_cite_reliability_per_standing }
			change_variable = { name = kehillah_din_cite_$SLOT$_reliability add = kehillah_din_cite_reliability_base }
			change_variable = { name = kehillah_din_cite_$SLOT$_reliability add = scope:kehillah_din_measuring_judge.var:kehillah_din_cite_learning_bonus }
			clamp_variable = { name = kehillah_din_cite_$SLOT$_reliability min = kehillah_din_cite_reliability_min max = kehillah_din_cite_reliability_max }
		}
		remove_variable = kehillah_din_cite_learning_bonus
	}
	remove_variable = kehillah_din_cite_$SLOT$_standing
}

# Scope: root of kehillah_din_cite.0001; scope:kehillah_din_cite_activity.
# The argument resting on the work in $SLOT$ is tested: it holds with the
# slot's reliability and adds its strength, or fails and costs the fail
# tally. Either way the activity log records the citation. Call after the
# option's own effect has named the work (kehillah_din_cited_work).
kehillah_din_cite_roll_effect = {
	random_list = {
		0 = {
			modifier = { add = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_reliability }
			scope:kehillah_din_cite_activity = {
				change_variable = { name = kehillah_bet_din_tally add = var:kehillah_din_cite_$SLOT$_strength }
			}
			if = {
				limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_strength >= kehillah_din_cite_strength_band_decisive }
				custom_tooltip = kehillah_din_cite_holds_decisive_tt
			}
			else_if = {
				limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_strength >= kehillah_din_cite_strength_band_strong }
				custom_tooltip = kehillah_din_cite_holds_strong_tt
			}
			else_if = {
				limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_strength >= kehillah_din_cite_strength_band_solid }
				custom_tooltip = kehillah_din_cite_holds_solid_tt
			}
			else = { custom_tooltip = kehillah_din_cite_holds_modest_tt }
			hidden_effect = {
				scope:kehillah_din_cite_activity = {
					kehillah_din_cite_log_dispatch_effect = { SOURCE = var:kehillah_din_cite_$SLOT$ RESULT = held JUDGE = var:kehillah_din_citing_judge }
				}
			}
		}
		0 = {
			modifier = {
				add = {
					value = 100
					subtract = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_reliability
				}
			}
			scope:kehillah_din_cite_activity = {
				change_variable = { name = kehillah_bet_din_tally subtract = kehillah_din_cite_fail_tally }
			}
			custom_tooltip = kehillah_din_cite_fails_tt
			hidden_effect = {
				scope:kehillah_din_cite_activity = {
					kehillah_din_cite_log_dispatch_effect = { SOURCE = var:kehillah_din_cite_$SLOT$ RESULT = failed JUDGE = var:kehillah_din_citing_judge }
					# A failed argument names no authority, so no work gains
					# a citation from it (kehillah_bet_din_reward_judges_for_
					# ruling_effect).
					remove_variable = kehillah_din_cited_work
				}
			}
		}
	}
}

# The host rite judges the public citation; apply XP to its actual judge.
kehillah_din_cite_cherem_effect = {
 if = {
  limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_host_standing ?= 0 }
  custom_tooltip = kehillah_din_cite_anathema_risk_tt
  hidden_effect = {
   scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= {
    if = {
     limit = { is_alive = yes is_jewish_character_trigger = yes NOT = { kehillah_cherem_in_full_trigger = yes } }
     set_variable = { name = kehillah_cherem_cause_work value = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ }
     kehillah_cherem_add_effect = { AMOUNT = kehillah_cherem_cite_anathema_points }
    }
   }
  }
 }
 else_if = {
  limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_host_standing ?= 1 }
  custom_tooltip = kehillah_din_cite_banned_risk_tt
  hidden_effect = {
   scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= {
    if = {
     limit = { is_alive = yes is_jewish_character_trigger = yes NOT = { kehillah_cherem_in_full_trigger = yes } }
     set_variable = { name = kehillah_cherem_cause_work value = scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$ }
     kehillah_cherem_add_effect = { AMOUNT = kehillah_cherem_cite_banned_points }
    }
   }
  }
 }
}

# Scope: root of kehillah_din_cite.0001. How widely the work in $SLOT$ is
# studied, said in words (the reason for its strength).
kehillah_din_cite_rarity_tt_effect = {
	if = {
		limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_masters <= kehillah_din_cite_rare_masters }
		custom_tooltip = kehillah_din_cite_rare_tt
	}
	else_if = {
		limit = { scope:kehillah_din_cite_activity.var:kehillah_din_cite_$SLOT$_masters >= kehillah_din_cite_common_masters }
		custom_tooltip = kehillah_din_cite_common_tt
	}
	else = { custom_tooltip = kehillah_din_cite_known_tt }
}

# Scope: root of kehillah_din_cite.0001; scope:kehillah_din_cite_activity.
# Each names its work as the ruling's authority, then adds its own effect.
kehillah_din_cite_grounded_effect = {
	scope:kehillah_din_cite_activity = {
		set_variable = { name = kehillah_din_cited_work value = var:kehillah_din_cite_grounded }
	}
	scope:kehillah_din_cite_activity.var:kehillah_bet_din_convening_title.holder ?= {
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_din_cite_grounded_stability }
	}
}

kehillah_din_cite_innovative_effect = {
	scope:kehillah_din_cite_activity = {
		set_variable = { name = kehillah_din_cited_work value = var:kehillah_din_cite_innovative }
	}
	scope:kehillah_din_cite_activity.var:kehillah_din_citing_judge ?= {
		add_prestige = kehillah_din_cite_innovative_prestige
	}
	scope:kehillah_din_cite_activity.var:kehillah_bet_din_convening_title.holder ?= {
		kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_din_cite_innovative_greatness }
		random = {
			chance = kehillah_din_cite_innovative_unease_chance
			kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_din_cite_innovative_unease }
		}

kehillah_din_bench_case_effect = {
	involved_activity = {
		save_scope_as = kds_activity
		kehillah_din_bench_clear_effect = yes
		set_variable = { name = kehillah_din_bench_case value = $CASE$ }
		set_variable = { name = kehillah_din_bench_count value = 0 }
		set_variable = { name = kehillah_din_bench_judges value = 0 }
		var:kehillah_bet_din_av_beit_din ?= { save_scope_as = kds_bench_av }
		var:kehillah_bet_din_judge2 ?= {
			if = {
				limit = { NAND = { exists = scope:kds_bench_av this = scope:kds_bench_av } }
				save_scope_as = kds_bench_j2
			}
		}
		var:kehillah_bet_din_judge3 ?= {
			if = {
				limit = {
					NAND = { exists = scope:kds_bench_av this = scope:kds_bench_av }
					NAND = { exists = scope:kds_bench_j2 this = scope:kds_bench_j2 }
				}
				save_scope_as = kds_bench_j3
			}
		}
	}
	# The judges' names, for the case text (GetPlayer.MakeScope.Var).
	remove_variable = kehillah_din_bench_av
	remove_variable = kehillah_din_bench_j2
	remove_variable = kehillah_din_bench_j3
	if = { limit = { exists = scope:kds_bench_av } set_variable = { name = kehillah_din_bench_av value = scope:kds_bench_av days = 60 } }
	if = { limit = { exists = scope:kds_bench_j2 } set_variable = { name = kehillah_din_bench_j2 value = scope:kds_bench_j2 days = 60 } }
	if = { limit = { exists = scope:kds_bench_j3 } set_variable = { name = kehillah_din_bench_j3 value = scope:kds_bench_j3 days = 60 } }
	kehillah_din_case_$CASE$_bench_effect = yes
	scope:kds_activity = {
		if = { limit = { has_variable = kehillah_din_bench_av_knows } change_variable = { name = kehillah_din_bench_judges add = 1 } }
		if = { limit = { has_variable = kehillah_din_bench_j2_knows } change_variable = { name = kehillah_din_bench_judges add = 1 } }
		if = { limit = { has_variable = kehillah_din_bench_j3_knows } change_variable = { name = kehillah_din_bench_judges add = 1 } }
		set_variable = { name = kehillah_din_bench_tally value = var:kehillah_din_bench_judges }
		change_variable = { name = kehillah_din_bench_tally multiply = kehillah_din_bench_judge_tally }
		if = {
			limit = { var:kehillah_din_bench_judges >= 3 }
			change_variable = { name = kehillah_din_bench_tally add = kehillah_din_bench_full_tally }
		}
	}
	# Works past the five shown, for "and N more".
	set_variable = { name = kehillah_din_bench_more value = scope:kds_activity.var:kehillah_din_bench_count days = 60 }
	change_variable = { name = kehillah_din_bench_more subtract = 5 }
}

# Scope: the convening leader. One work on the order the faith knows (called
# from the generated kehillah_din_bench_<seder>_effect). The first five are
# shown, each with a mask of the judges who have mastered it: 1 the Av Beit
# Din, 2 the second judge, 4 the third.
kehillah_din_bench_add_effect = {
	scope:kds_activity = {
		change_variable = { name = kehillah_din_bench_count add = 1 }
		set_variable = { name = kehillah_din_bench_mask value = 0 }
	}
	if = {
		limit = { scope:kds_bench_av ?= { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:$WORK$ } } }
		scope:kds_activity = {
			change_variable = { name = kehillah_din_bench_mask add = 1 }
			set_variable = kehillah_din_bench_av_knows
		}
	}
	if = {
		limit = { scope:kds_bench_j2 ?= { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:$WORK$ } } }
		scope:kds_activity = {
			change_variable = { name = kehillah_din_bench_mask add = 2 }
			set_variable = kehillah_din_bench_j2_knows
		}
	}
	if = {
		limit = { scope:kds_bench_j3 ?= { is_target_in_variable_list = { name = kehillah_mastered_works target = flag:$WORK$ } } }
		scope:kds_activity = {
			change_variable = { name = kehillah_din_bench_mask add = 4 }
			set_variable = kehillah_din_bench_j3_knows
		}
	}
	scope:kds_activity = {
		if = {
			limit = { var:kehillah_din_bench_count = 1 }
			set_variable = { name = kehillah_din_bench_work_1 value = flag:$WORK$ }
			set_variable = { name = kehillah_din_bench_1_knowers value = var:kehillah_din_bench_mask }
		}
		else_if = {
			limit = { var:kehillah_din_bench_count = 2 }
			set_variable = { name = kehillah_din_bench_work_2 value = flag:$WORK$ }
			set_variable = { name = kehillah_din_bench_2_knowers value = var:kehillah_din_bench_mask }
		}
		else_if = {
			limit = { var:kehillah_din_bench_count = 3 }
			set_variable = { name = kehillah_din_bench_work_3 value = flag:$WORK$ }
			set_variable = { name = kehillah_din_bench_3_knowers value = var:kehillah_din_bench_mask }
		}
		else_if = {
			limit = { var:kehillah_din_bench_count = 4 }
			set_variable = { name = kehillah_din_bench_work_4 value = flag:$WORK$ }
			set_variable = { name = kehillah_din_bench_4_knowers value = var:kehillah_din_bench_mask }
		}
		else_if = {
			limit = { var:kehillah_din_bench_count = 5 }
			set_variable = { name = kehillah_din_bench_work_5 value = flag:$WORK$ }
			set_variable = { name = kehillah_din_bench_5_knowers value = var:kehillah_din_bench_mask }
		}
	}
}
