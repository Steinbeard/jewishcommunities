# ---------------------------------------------------------------------
# kehillah_save_rosh_yeshiva_effect. Scope: a Kehillah leader.
# Binds scope:kehillah_rosh_yeshiva to the head of the community's
# yeshiva: the appointed Rosh Yeshiva, or a Rav-leader, who IS the Rosh
# Yeshiva whenever he has not appointed one (V29 section 3.2, Daniel
# 2026-10-11). Leaves the scope unset when there is no yeshiva or no head.
# ---------------------------------------------------------------------
kehillah_save_rosh_yeshiva_effect = {
	if = {
		limit = { kehillah_has_yeshiva_trigger = yes }
		if = {
			limit = { employs_court_position = rosh_yeshiva_kehillah_position }
			court_position:rosh_yeshiva_kehillah_position = { save_scope_as = kehillah_rosh_yeshiva }
		}
		else_if = {
			limit = { has_trait = kehillah_rabbi_trait }
			save_scope_as = kehillah_rosh_yeshiva
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_establish_yeshiva_effect. Scope: a Kehillah leader.
# FAME = the starting fame.
#
# The shared core of founding (the decision) and seeding (Worms at game
# start): the building, its record on the title (so succession restores
# it -- kehillah_record_quarter_effect), starting fame, the map building,
# and a head.
# ---------------------------------------------------------------------
kehillah_establish_yeshiva_effect = {
	# Must come first: the Yeshiva tier's can_construct reads it, and
	# add_domicile_building honours can_construct.
	primary_title = { set_variable = kehillah_yeshiva_founded }
	domicile = {
		if = {
			limit = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_01 } }
			add_domicile_building = kehillah_beit_midrash_01
		}
		if = {
			limit = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_02 } }
			add_domicile_building = kehillah_beit_midrash_02
		}
		if = {
			limit = { NOT = { has_domicile_building_or_higher = kehillah_beit_midrash_03 } }
			add_domicile_building = kehillah_beit_midrash_03
		}
	}
	kehillah_record_quarter_effect = yes

	primary_title = {
		if = {
			limit = { NOT = { has_variable = kehillah_yeshiva_fame } }
			set_variable = { name = kehillah_yeshiva_fame value = $FAME$ }
		}
	}

	kehillah_place_yeshiva_map_building_effect = yes
	kehillah_yeshiva_check_tier_effect = yes

	# A head. A Rav-leader heads it himself (leaving the seat empty);
	# otherwise the most learned qualified courtier is appointed.
	if = {
		limit = {
			NOT = { has_trait = kehillah_rabbi_trait }
			NOT = { employs_court_position = rosh_yeshiva_kehillah_position }
			any_courtier = { kehillah_qualified_rosh_yeshiva_trigger = yes }
		}
		ordered_courtier = {
			limit = { kehillah_qualified_rosh_yeshiva_trigger = yes }
			order_by = learning
			save_scope_as = kehillah_new_rosh_yeshiva
		}
		appoint_court_position = {
			recipient = scope:kehillah_new_rosh_yeshiva
			court_position = rosh_yeshiva_kehillah_position
		}
	}
	debug_log = "kehillah_establish_yeshiva_effect: yeshiva established"
}

# ---------------------------------------------------------------------
# kehillah_place_yeshiva_map_building_effect. Scope: a Kehillah leader.
#
# The yeshiva's public face (V29 section 3.5): a building in the host's
# province, where the community's quarter stands. A GREAT building first
# -- its own slot type, so it never displaces Notre Dame or the
# Alhambra, and it falls back to reduced modifiers when disabled. If the
# engine will not place one (unproven outside All Under Heaven -- V29
# spike 1), a SPECIAL building, but only into an EMPTY special slot:
# add_special_building replaces whatever is there, and this must never
# demolish a province's own landmark. If neither can be placed, the
# yeshiva still works; it simply has no map building.
# ---------------------------------------------------------------------
kehillah_place_yeshiva_map_building_effect = {
	if = {
		limit = { exists = domicile.domicile_location }
		domicile.domicile_location = {
			if = {
				limit = {
					NOR = {
						has_building = kehillah_yeshiva_hall
						has_building = kehillah_great_yeshiva_hall
						has_building = kehillah_yeshiva_hall_special
					}
				}
				add_great_building = kehillah_yeshiva_hall
			}
			if = {
				limit = {
					NOR = {
						has_building = kehillah_yeshiva_hall
						has_building = kehillah_yeshiva_hall_special
						has_special_building = yes
					}
				}
				add_special_building = kehillah_yeshiva_hall_special
			}
			if = {
				limit = { has_building = kehillah_yeshiva_hall }
				debug_log = "kehillah_place_yeshiva_map_building_effect: GREAT building placed"
			}
			else_if = {
				limit = { has_building = kehillah_yeshiva_hall_special }
				debug_log = "kehillah_place_yeshiva_map_building_effect: SPECIAL building placed (great building refused)"
			}
			else = {
				debug_log = "kehillah_place_yeshiva_map_building_effect: NO map building (great refused, special slot occupied)"
			}
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_found_yeshiva_effect. Scope: a Kehillah leader. The decision's
# effect (common/decisions/kehillah_yeshiva_decisions.txt).
# ---------------------------------------------------------------------
kehillah_found_yeshiva_effect = {
	kehillah_establish_yeshiva_effect = { FAME = kehillah_yeshiva_fame_on_founding }
	trigger_event = kehillah_yeshiva.0001
	# A founding draws scholars (V29 section 3.1): ordained rabbis come
	# looking for a place, as Dayyan candidates and Bet Din judges.
	trigger_event = { id = kehillah_yeshiva.0010 days = { 180 400 } }
	trigger_event = { id = kehillah_yeshiva.0010 days = { 700 1100 } }
}

# ---------------------------------------------------------------------
# kehillah_add_yeshiva_fame_effect. Scope: a Kehillah leader.
# AMOUNT = how much. Clamped to [0, kehillah_yeshiva_fame_max].
# ---------------------------------------------------------------------
kehillah_add_yeshiva_fame_effect = {
	if = {
		limit = { kehillah_has_yeshiva_trigger = yes }
		primary_title = {
			if = {
				limit = { NOT = { has_variable = kehillah_yeshiva_fame } }
				set_variable = { name = kehillah_yeshiva_fame value = 0 }
			}
			change_variable = { name = kehillah_yeshiva_fame add = $AMOUNT$ }
			clamp_variable = { name = kehillah_yeshiva_fame min = 0 max = kehillah_yeshiva_fame_max }
		}
		kehillah_yeshiva_check_tier_effect = yes
	}
}

# ---------------------------------------------------------------------
# kehillah_yeshiva_quarterly_effect. Scope: a Kehillah leader. From
# kehillah_quarterly_pulse. Fame decays slowly, faster while Stability is
# below Healthy (V29 section 3.1). A learned head (a rabbi with Learning
# at kehillah_rosh_yeshiva_min_learning) sustains it by +1, which offsets
# the stable decay but not the troubled one. Learning alone, not the
# founding's track depth: Worms's seeded head (Isaac, Learning 35, low
# tracks) must be able to keep his own academy's name (live 2026-10-11).
# ---------------------------------------------------------------------
kehillah_yeshiva_quarterly_effect = {
	kehillah_save_rosh_yeshiva_effect = yes
	if = {
		limit = {
			kehillah_has_yeshiva_trigger = yes
			scope:kehillah_rosh_yeshiva ?= {
				has_trait = kehillah_rabbi_trait
				learning >= kehillah_rosh_yeshiva_min_learning
			}
		}
		kehillah_add_yeshiva_fame_effect = { AMOUNT = 1 }
	}
	if = {
		limit = {
			kehillah_has_yeshiva_trigger = yes
			primary_title ?= { has_variable = kehillah_yeshiva_fame }
		}
		if = {
			limit = {
				kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold }
			}
			primary_title = {
				change_variable = { name = kehillah_yeshiva_fame subtract = kehillah_yeshiva_fame_decay_stable }
				clamp_variable = { name = kehillah_yeshiva_fame min = 0 max = kehillah_yeshiva_fame_max }
			}
		}
		else = {
			primary_title = {
				change_variable = { name = kehillah_yeshiva_fame subtract = kehillah_yeshiva_fame_decay_troubled }
				clamp_variable = { name = kehillah_yeshiva_fame min = 0 max = kehillah_yeshiva_fame_max }
			}
		}
	}
}

# ---------------------------------------------------------------------
# kehillah_study_zman_effect. Scope: the student, arrived at the
# community whose yeshiva he is studying in. Requires
# scope:kehillah_journey_destination_title (kehillah_learn_torah_journey.
# 0001) and scope:kehillah_rosh_yeshiva.
#
# One zman (V29 section 3.3):
# - an ordained student gains track XP in his teacher's strongest
#   teaching track;
# - an unordained student earns zman credit, which the ordination
#   curriculum accepts (kehillah_rabbi_ordination_curriculum_met_trigger);
# - the yeshiva gains a little fame;
# - a student who is now ready is examined on the spot by the Rosh
#   Yeshiva (kehillah_semicha.0001, which examines by the community's
#   Rosh Yeshiva since V29).
# ---------------------------------------------------------------------
kehillah_study_zman_effect = {
	add_piety = minor_piety_gain
	add_learning_lifestyle_xp = minor_lifestyle_xp
	if = {
		limit = { has_trait = kehillah_rabbi_trait }
		if = {
			limit = {
				# Talmudics if the teacher is deep in it, otherwise Parshanut
				# (the other teaching track a Rosh Yeshiva can qualify by).
				scope:kehillah_rosh_yeshiva = {
					has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= kehillah_rosh_yeshiva_min_track_xp }
				}
			}
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = kehillah_zman_track_xp }
		}
		else = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value = kehillah_zman_track_xp }
		}
	}
	else = {
		kehillah_add_zman_term_effect = yes
	}
	scope:kehillah_journey_destination_title.holder ?= {
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_yeshiva_fame_per_zman }
	}
	if = {
		limit = { kehillah_rabbi_ordination_ready_trigger = yes }
		scope:kehillah_journey_destination_title = { save_scope_as = kehillah_semicha_community }
		trigger_event = kehillah_semicha.0001
	}
}

# ---------------------------------------------------------------------
# kehillah_seed_yeshiva_effect. Scope: a Kehillah leader, at game start
# (from his community's kehillah_setup_<town>_start_effect, which saves
# scope:kehillah_leader). FAME = starting fame.
#
# The yeshivot active in 1066 (Daniel, 2026-10-11): Worms (seeded from its
# developed-start effect), Narbonne, Baghdad -- the heir of Sura and
# Pumbedita, seeded at low fame because the Geonic academies had
# effectively ended about 1040 -- and Lucena once that community exists.
#
# A leader who is not a rabbi cannot head it, so the seed brings the
# academy's head: a learned rabbi deep in Talmud, appointed Rosh Yeshiva.
# Built here from scope:kehillah_leader rather than through
# kehillah_generate_rabbi_candidate_effect, which reads `root` -- not
# safe at game start, where root is not the leader.
# ---------------------------------------------------------------------
kehillah_seed_yeshiva_effect = {
	kehillah_establish_yeshiva_effect = { FAME = $FAME$ }
	if = {
		limit = {
			NOT = { has_trait = kehillah_rabbi_trait }
			NOT = { employs_court_position = rosh_yeshiva_kehillah_position }
		}
		create_character = {
			dynasty = generate
			age = { 45 65 }
			faith = scope:kehillah_leader.faith
			rite = scope:kehillah_leader.rite
			culture = scope:kehillah_leader.culture
			location = scope:kehillah_leader.domicile.domicile_location
			random_traits = yes
			gender_female_chance = 0
			trait = education_learning_4
			trait = kehillah_rabbi_trait
			learning = { 15 19 }
			save_scope_as = kehillah_seeded_rosh_yeshiva
		}
		scope:kehillah_seeded_rosh_yeshiva = {
			add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 70 }
		}
		add_courtier = scope:kehillah_seeded_rosh_yeshiva
		appoint_court_position = {
			recipient = scope:kehillah_seeded_rosh_yeshiva
			court_position = rosh_yeshiva_kehillah_position
		}
		debug_log = "kehillah_seed_yeshiva_effect: seeded a Rosh Yeshiva (the leader is not a rabbi)"
	}
	debug_log = "kehillah_seed_yeshiva_effect: yeshiva seeded"
}

# Scope: a student. One completed term toward semicha.
kehillah_add_zman_term_effect = {
	if = {
		limit = { NOT = { has_variable = kehillah_zman_terms } }
		set_variable = { name = kehillah_zman_terms value = 0 }
	}
	change_variable = { name = kehillah_zman_terms add = 1 }
}

# Scope: a Kehillah leader. Notices a fame tier crossed upward, sends the
# message, and at Illustrious raises the hall to the Great Yeshiva. The
# highest tier reached is remembered on the title, so falling back and
# climbing again does not repeat the notice.
kehillah_yeshiva_check_tier_effect = {
	# Not `root`: a travelling student's zman calls this through the
	# destination's holder, where root is the student.
	save_scope_as = kehillah_tier_leader
	if = {
		limit = { kehillah_has_yeshiva_trigger = yes }
		# First sight: record the current tier SILENTLY -- a seeded or
		# founded yeshiva starts at its tier without announcing it (live
		# 2026-10-11: a spurious notice fired on the first fame change).
		if = {
			limit = { primary_title = { NOT = { has_variable = kehillah_yeshiva_tier_seen } } }
			primary_title = { set_variable = { name = kehillah_yeshiva_tier_seen value = scope:kehillah_tier_leader.kehillah_yeshiva_fame_tier_value } }
		}
		if = {
			limit = {
				primary_title = { has_variable = kehillah_yeshiva_tier_seen }
				kehillah_yeshiva_fame_tier_value > primary_title.var:kehillah_yeshiva_tier_seen
			}
			primary_title = { set_variable = { name = kehillah_yeshiva_tier_seen value = scope:kehillah_tier_leader.kehillah_yeshiva_fame_tier_value } }
			send_interface_message = {
				type = msg_kehillah_band_improved
				title = kehillah_yeshiva_tier_msg_title
				desc = kehillah_yeshiva_tier_msg_desc
				left_icon = scope:kehillah_tier_leader
			}
			if = {
				limit = { kehillah_yeshiva_fame_tier_value >= 3 }
				kehillah_raise_great_yeshiva_effect = yes
			}
		}
	}
}

# Scope: a Kehillah leader. The hall becomes the Great Yeshiva.
kehillah_raise_great_yeshiva_effect = {
	if = {
		limit = { exists = domicile.domicile_location }
		domicile.domicile_location = {
			# One great building per province (live 2026-10-11: "Province
			# already has a great building"), so the hall is removed first.
			if = {
				limit = { has_building = kehillah_yeshiva_hall }
				remove_building = kehillah_yeshiva_hall
				add_great_building = kehillah_great_yeshiva_hall
			}
			if = {
				limit = { has_building = kehillah_great_yeshiva_hall }
				debug_log = "kehillah_raise_great_yeshiva_effect: the Great Yeshiva stands"
			}
			else = {
				debug_log = "kehillah_raise_great_yeshiva_effect: could NOT raise the Great Yeshiva"
			}
		}
	}
}

# Scope: a Kehillah leader, at the start of a zman. Flags this term's
# students: the court's eligible young men, plus visiting students drawn
# by the yeshiva's fame (1 / 2 / 3 / 4 by tier), who join the court.
kehillah_gather_zman_students_effect = {
	every_courtier = {
		limit = { kehillah_zman_student_candidate_trigger = yes }
		add_character_flag = { flag = kehillah_zman_student months = 6 }
	}
	kehillah_zman_visiting_student_effect = yes
	if = {
		limit = { kehillah_yeshiva_fame_tier_value >= 1 }
		kehillah_zman_visiting_student_effect = yes
	}
	if = {
		limit = { kehillah_yeshiva_fame_tier_value >= 2 }
		kehillah_zman_visiting_student_effect = yes
	}
	if = {
		limit = { kehillah_yeshiva_fame_tier_value >= 3 }
		kehillah_zman_visiting_student_effect = yes
	}
	set_variable = { name = kehillah_zman_student_count value = 0 }
	every_courtier = {
		limit = { has_character_flag = kehillah_zman_student }
		root = { change_variable = { name = kehillah_zman_student_count add = 1 } }
	}
}

# Scope: a Kehillah leader. One visiting student arrives for the term.
kehillah_zman_visiting_student_effect = {
	create_character = {
		dynasty = generate
		age = { 16 24 }
		faith = root.faith
		rite = root.rite
		culture = root.culture
		location = root.domicile.domicile_location
		random_traits = yes
		gender_female_chance = 0
		trait = education_learning_2
		learning = { 7 11 }
		save_scope_as = kehillah_new_student
	}
	add_courtier = scope:kehillah_new_student
	scope:kehillah_new_student = {
		add_character_flag = { flag = kehillah_zman_student months = 6 }
		add_character_flag = kehillah_visiting_student
		set_variable = { name = kehillah_alma_mater value = root.primary_title }
	}
}

# Scope: a Kehillah leader, at the end of a zman. scope:kehillah_rosh_yeshiva
# must be bound. Advances every student, ordains the ready, places
# ordained visiting students as Dayyanim elsewhere, and grows fame.
kehillah_conclude_zman_effect = {
	primary_title = { save_scope_as = kehillah_semicha_community }
	set_variable = { name = kehillah_zman_ordained_count value = 0 }
	set_variable = { name = kehillah_zman_alumni_count value = 0 }

	every_courtier = {
		limit = { has_character_flag = kehillah_zman_student }
		save_scope_as = kehillah_zman_this_student
		add_learning_skill = 1
		kehillah_add_zman_term_effect = yes
		if = {
			limit = { kehillah_rabbi_ordination_ready_trigger = yes }
			kehillah_grant_semicha_effect = yes
			root = { change_variable = { name = kehillah_zman_ordained_count add = 1 } }
			# Named in the end-of-term event (kehillah_yeshiva.0202).
			if = {
				limit = { NOT = { exists = scope:kehillah_zman_first_ordained } }
				save_scope_as = kehillah_zman_first_ordained
			}
			if = {
				limit = { has_character_flag = kehillah_visiting_student }
				root = { kehillah_place_alumnus_effect = yes }
			}
		}
		remove_character_flag = kehillah_zman_student
	}

	# Fame by the head's quality.
	scope:kehillah_rosh_yeshiva ?= {
		save_scope_value_as = { name = kehillah_zman_quality value = kehillah_judicial_learning_value }
	}
	if = {
		limit = {
			exists = scope:kehillah_zman_quality
			scope:kehillah_zman_quality >= kehillah_zman_quality_great
		}
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_zman_fame_great }
	}
	else_if = {
		limit = {
			exists = scope:kehillah_zman_quality
			scope:kehillah_zman_quality >= kehillah_zman_quality_good
		}
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_zman_fame_good }
	}
	else = {
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_zman_fame_poor }
	}
	if = {
		limit = {
			kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold }
		}
		kehillah_add_yeshiva_fame_effect = { AMOUNT = 2 }
	}
	remove_character_flag = kehillah_zman_in_progress
	debug_log = "kehillah_conclude_zman_effect: the zman is concluded"
}

# Scope: a Kehillah leader. scope:kehillah_zman_this_student is a newly
# ordained VISITING student: he goes home to a community with no Dayyan
# and takes the seat -- this yeshiva's alumnus. Stays if no seat is free.
kehillah_place_alumnus_effect = {
	# Live 2026-10-11: assigning the seat in the same effect as add_courtier
	# was refused ("doesn't have an open council position"), and all three
	# alumni went to the same community. So: reserve the destination (a
	# 60-day title variable, which other alumni skip), move him now, and
	# seat him a day later from the destination's own event
	# (kehillah_yeshiva.0210), which logs only a real seating.
	# Saved scopes live for the whole conclusion: without this the third
	# alumnus reused the second's destination when no seat was left
	# (live 2026-10-11).
	clear_saved_scope = kehillah_alumnus_destination
	random_in_global_list = {
		variable = kehillah_registered_communities
		limit = {
			exists = holder
			NOT = { this = root.primary_title }
			NOT = { has_variable = kehillah_alumnus_incoming }
			holder = {
				NOT = { exists = cp:councillor_court_chaplain }
				faith = scope:kehillah_zman_this_student.faith
			}
		}
		save_scope_as = kehillah_alumnus_destination
	}
	if = {
		limit = { exists = scope:kehillah_alumnus_destination }
		scope:kehillah_alumnus_destination = {
			set_variable = { name = kehillah_alumnus_incoming value = scope:kehillah_zman_this_student days = 60 }
		}
		scope:kehillah_alumnus_destination.holder = {
			add_courtier = scope:kehillah_zman_this_student
			trigger_event = { id = kehillah_yeshiva.0210 days = 1 }
		}
		scope:kehillah_zman_this_student = { remove_character_flag = kehillah_visiting_student }
		if = {
			limit = { NOT = { exists = scope:kehillah_zman_first_alumnus } }
			scope:kehillah_zman_this_student = { save_scope_as = kehillah_zman_first_alumnus }
			scope:kehillah_alumnus_destination = { save_scope_as = kehillah_zman_first_alumnus_home }
		}
		change_variable = { name = kehillah_zman_alumni_count add = 1 }
		kehillah_add_yeshiva_fame_effect = { AMOUNT = kehillah_yeshiva_fame_per_alumnus }
		debug_log = "kehillah_place_alumnus_effect: an alumnus is going home to take a Dayyan seat"
	}
}
