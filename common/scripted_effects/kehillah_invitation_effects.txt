# =====================================================================
# INVITATIONS -- V34 section 7 (docs/spec/v34-outlook-teachings-and-
# founding.md), 2026-10-13. A ruler with no Jews in his realm may still
# open the door: a yearly "open-door review", driven by his outlook on
# the Jews and his wish to fill towns and markets, can raise his Jewish
# Settlement Policy. Raising it to Encouraged -- by this review, the
# hosts' review or the player's decree -- IS the invitation: Kehillah
# players within diplomatic range hear of it, and for 20 years it draws
# settlers (Send Out Settlers, exiles) and lets a community that is not
# overcrowded answer the call.
# (The historical shape: Bela IV of Hungary after the Mongols, 1251;
# Boleslaw the Pious's Statute of Kalisz, 1264; the Rhineland bishops'
# charters of 1084/1090.)
# =====================================================================

# Scope: an AI ruler passing kehillah_open_door_candidate_trigger. Raise
# the policy one rung if the welcoming pressures carry it.
kehillah_open_door_review_effect = {
	save_scope_value_as = { name = kehillah_open_door_outlook value = kehillah_outlook_on_jews_value }
	save_scope_value_as = { name = kehillah_open_door_net value = kehillah_open_door_net_pressure_value }
	debug_log_scopes = yes
	if = {
		limit = { kehillah_open_door_net_pressure_value <= kehillah_policy_review_raise_threshold_negative }
		primary_title = { save_scope_as = kehillah_reason_realm }
		# 2026-11-20: a lord who needs credit says so first.
		if = {
			limit = { kehillah_wants_jewish_credit_trigger = yes }
			scope:kehillah_reason_realm = { set_variable = { name = kehillah_policy_reason value = flag:credit } }
		}
		else_if = {
			limit = { kehillah_invitation_motive_value > 0 }
			scope:kehillah_reason_realm = { set_variable = { name = kehillah_policy_reason value = flag:towns } }
		}
		else = {
			scope:kehillah_reason_realm = { set_variable = { name = kehillah_policy_reason value = flag:justice } }
		}
		debug_log = "kehillah_open_door_review_effect: a ruler with no Jews opens the door -- raising one rung"
		kehillah_raise_jewish_settlement_policy_effect = yes
	}
	else = {
		kehillah_start_policy_review_cooldown_effect = yes
		debug_log = "kehillah_open_door_review_effect: no reason to open the door -- policy unchanged"
	}
}

# Scope: the host ruler, just raised to Encouraged (end of kehillah_
# raise_jewish_settlement_policy_effect), or taking Invite Jewish Settlers,
# or the Norman conqueror; scope:kehillah_policy_realm saved. The
# invitation: open for 20 years across the realm.
# 2026-11-25 (docs/spec/2026-11-25-invitation-letters.md): it also names
# one county -- the best founding site of his own domain -- and calls the
# three communities nearest it by name. A player community called gets the
# letter (kehillah_invitation.0001) and answers in it; an AI community
# decides some months later (kehillah_invitation.0020), so a player can
# answer first. The first community founded there closes the call
# (kehillah_invitation_answered_effect). Other Kehillah players in
# diplomatic range hear of it in the feed.
kehillah_invitation_issue_effect = {
	save_scope_as = kehillah_policy_host
	primary_title = {
		save_scope_as = kehillah_policy_realm
		set_variable = { name = kehillah_invitation_open value = yes years = 20 }
	}
	clear_saved_scope = kehillah_call_county
	ordered_realm_county = {
		limit = {
			holder = scope:kehillah_policy_host
			kehillah_founding_site_base_trigger = yes
		}
		order_by = kehillah_invitation_site_score_value
		save_scope_as = kehillah_call_county
	}
	if = {
		limit = { NOT = { exists = scope:kehillah_call_county } }
		ordered_realm_county = {
			limit = { kehillah_founding_site_base_trigger = yes }
			order_by = kehillah_invitation_site_score_value
			save_scope_as = kehillah_call_county
		}
	}
	if = {
		limit = { exists = scope:kehillah_call_county }
		scope:kehillah_call_county.title_province = { save_scope_as = kehillah_call_province }
		scope:kehillah_policy_realm = {
			set_variable = {
				name = kehillah_invitation_county
				value = scope:kehillah_call_county
				years = kehillah_invitation_call_years
			}
		}
		ordered_in_global_list = {
			variable = kehillah_registered_communities
			limit = { kehillah_invitation_callable_community_trigger = yes }
			order_by = kehillah_invitation_call_nearness_value
			max = kehillah_invitation_called_communities
			check_range_bounds = no
			holder = { kehillah_invitation_deliver_letter_effect = yes }
		}
		# 2026-11-25 (Daniel: "send invites to adventurers and merchants
		# too"): the nearest Jewish adventurers who could found a community,
		# and the nearest Jewish merchant companies, who could send families.
		ordered_ruler = {
			government_type_filter = government_type:landless_adventurer_government
			limit = { kehillah_invitation_callable_band_trigger = yes }
			order_by = kehillah_invitation_char_nearness_value
			max = kehillah_invitation_called_bands
			check_range_bounds = no
			kehillah_invitation_deliver_letter_effect = yes
		}
		ordered_ruler = {
			government_type_filter = government_type:merchant_government
			limit = { kehillah_invitation_callable_merchant_trigger = yes }
			order_by = kehillah_invitation_char_nearness_value
			max = kehillah_invitation_called_merchants
			check_range_bounds = no
			kehillah_invitation_deliver_letter_effect = yes
		}
	}
	every_player = {
		limit = {
			government_has_flag = government_is_kehillah
			in_diplomatic_range = scope:kehillah_policy_host
			NOT = { is_in_list = kehillah_invitation_called }
		}
		send_interface_message = {
			type = msg_kehillah_community_news
			title = kehillah_news_invitation_t
			desc = kehillah_news_invitation_d
			left_icon = scope:kehillah_policy_host
		}
	}
	debug_log_scopes = yes
	debug_log = "kehillah_invitation_issue_effect: a ruler invites Jewish settlers"
}

# Scope: the mother community's leader; scope:kehillah_call_county,
# scope:kehillah_policy_host and scope:kehillah_settlers_leader saved.
# Answering a call: the costs and gains the letter's send option shows.
# The founding itself is kehillah_invitation.0021 (never inside an
# option: the tooltip dry run of a founding stalled the game, 2026-10-13).

	custom_tooltip = kehillah_invitation_send_party_tt
	kehillah_change_pillar_effect = { PILLAR = greatness AMOUNT = kehillah_invitation_send_greatness }

			target = root
			modifier = respect_opinion
			opinion = kehillah_invitation_send_opinion
		}
	}
	save_scope_value_as = { name = kehillah_settlers_invited value = yes }
	scope:kehillah_call_county = { save_scope_as = kehillah_founding_county }
}

# A player answers personally. The old community passes by the same
# change_title_holder route as voluntary abdication, then the former
# leader founds the new seat using the established founder effect.
# This must run from an event immediate, never an option tooltip.
kehillah_invitation_lead_self_effect = {
	save_scope_as = kehillah_mother_leader
	primary_title = { save_scope_as = kehillah_mother_title }
	primary_title.current_heir = { save_scope_as = kehillah_departure_successor }
	scope:kehillah_call_county = { save_scope_as = kehillah_founding_county }
	scope:kehillah_mother_title = {
		set_variable = { name = kehillah_settlers_cooldown value = yes years = kehillah_settlers_cooldown_years }
		change_variable = { name = kehillah_var_greatness add = kehillah_invitation_send_greatness }
	}
	add_prestige = medium_prestige_gain
	scope:kehillah_policy_host = {
		add_opinion = {
			target = root
			modifier = respect_opinion
			opinion = kehillah_invitation_send_opinion
		}
	}
	# Collect followers before the old court passes to the successor.
	every_close_family_member = {
		limit = {
			is_courtier_of = root
			NOT = { this = scope:kehillah_departure_successor }
		}
		add_to_list = kehillah_settlers_party
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = kehillah_invitation_handoff_change
		add_claim_on_loss = no
	}
	scope:kehillah_mother_title = {
		change_title_holder = {
			holder = scope:kehillah_departure_successor
			change = scope:kehillah_invitation_handoff_change
			government_base = scope:kehillah_mother_leader
		}
	}
	resolve_title_and_vassal_change = scope:kehillah_invitation_handoff_change
	debug_log = "kehillah_invitation_lead_self_effect: old seat passed to appointed successor"
	# Remove the departing leader's old realm office before establishing
	# the new seat; the successor can be appointed separately.
	if = {
		limit = { has_variable = kehillah_nagid_of }
		kehillah_nagid_step_down_effect = yes
	}
	kehillah_found_with_founder_effect = yes
	every_in_list = {
		list = kehillah_settlers_party
		limit = { NOT = { is_courtier_of = scope:kehillah_natural_founder } }
		save_scope_as = kehillah_settler
		scope:kehillah_natural_founder = { add_courtier = scope:kehillah_settler }
	}
	kehillah_refresh_map_view_for_players_effect = yes
	debug_log = "kehillah_invitation_lead_self_effect: leader founded the invited community personally"
}

# Scope: a community leader, adventurer or merchant called by name
# (kehillah_invitation_issue_effect). A player gets the letter now; the AI
# decides some months later, so a player called with it can answer first.
kehillah_invitation_deliver_letter_effect = {
	add_to_temporary_list = kehillah_invitation_called
	if = {
		limit = { is_ai = no }
		trigger_event = kehillah_invitation.0001
	}
	else = {
		trigger_event = {
			id = kehillah_invitation.0020
			days = { 30 240 }
		}
	}
}

# Scope: whoever was called, in kehillah_invitation.0001/.0020. Saves what
# the letter shows: the leader candidates, the party's size (a community)
# and the distance band to the called county (from a community's quarter,
# or from where an adventurer or merchant is).
kehillah_invitation_letter_setup_effect = {
	save_scope_as = kehillah_mother_leader
	clear_saved_scope = kehillah_settlers_lead_merchant
	if = {
		limit = { government_has_flag = government_is_kehillah }
		primary_title = { save_scope_as = kehillah_mother_title }
		kehillah_settlers_pick_leaders_effect = yes
		# A single volunteer is named in the letter. The existing three
		# candidate types remain available to Send Out Settlers elsewhere.
		ordered_courtier = {
			limit = { kehillah_settlers_leader_candidate_trigger = yes }
			order_by = { value = stewardship add = learning }
			save_scope_as = kehillah_invitation_volunteer
		}
		save_scope_value_as = { name = kehillah_invitation_party_size value = kehillah_invitation_party_size_value }
		domicile.domicile_location ?= { save_scope_as = kehillah_call_from }
	}
	else_if = {
		limit = { has_government = merchant_government }
		ordered_courtier = {
			limit = { kehillah_settlers_leader_candidate_trigger = yes }
			order_by = { value = stewardship add = learning }
			save_scope_as = kehillah_settlers_lead_merchant
		}
		location ?= { save_scope_as = kehillah_call_from }
	}
	else = {
		location ?= { save_scope_as = kehillah_call_from }
	}
	if = {
		limit = {
			exists = scope:kehillah_call_from
			scope:kehillah_call_county.title_province = {
				squared_distance = {
					target = scope:kehillah_call_from
					value <= squared_distance_small
				}
			}
		}
		save_scope_value_as = { name = kehillah_call_distance value = 1 }
	}
	else_if = {
		limit = {
			exists = scope:kehillah_call_from
			scope:kehillah_call_county.title_province = {
				squared_distance = {
					target = scope:kehillah_call_from
					value <= squared_distance_medium
				}
			}
		}
		save_scope_value_as = { name = kehillah_call_distance value = 2 }
	}
	else = {
		save_scope_value_as = { name = kehillah_call_distance value = 3 }
	}
}

# Scope: the merchant; scope:kehillah_settlers_leader and scope:kehillah_
# founding_county saved. His factor goes with his own household and up to
# kehillah_invitation_party_min Jewish families of the company, and founds
# the community. Not a daughter community: the merchant company is no
# Kehillah, so no mother title is recorded.
kehillah_invitation_merchant_go_effect = {
	save_scope_as = kehillah_mother_leader
	clear_saved_scope = kehillah_mother_title
	set_variable = { name = kehillah_settlers_to_send value = kehillah_invitation_party_min }
	scope:kehillah_settlers_leader = {
		every_close_family_member = {
			limit = {
				is_courtier_of = scope:kehillah_mother_leader
				NOT = { this = scope:kehillah_mother_leader }
				NOT = { is_close_family_of = scope:kehillah_mother_leader }
				NOT = { is_spouse_of = scope:kehillah_mother_leader }
			}
			add_to_list = kehillah_settlers_party
		}
	}
	while = {
		limit = {
			var:kehillah_settlers_to_send > 0
			any_courtier = { kehillah_settlers_follower_candidate_trigger = yes }
		}
		random_courtier = {
			limit = { kehillah_settlers_follower_candidate_trigger = yes }
			add_to_list = kehillah_settlers_party
		}
		change_variable = { name = kehillah_settlers_to_send add = -1 }
	}
	remove_variable = kehillah_settlers_to_send
	scope:kehillah_settlers_leader = { kehillah_found_with_founder_effect = yes }
	every_in_list = {
		list = kehillah_settlers_party
		limit = { NOT = { is_courtier_of = scope:kehillah_natural_founder } }
		save_scope_as = kehillah_settler
		scope:kehillah_natural_founder = { add_courtier = scope:kehillah_settler }
	}
	debug_log = "kehillah_invitation_merchant_go_effect: a merchant's factor founded a community"
}

# Scope: the leader of a community just founded (Send Out Settlers, the
# Found a Jewish Community decision, exiles arriving). If its lord's
# realm has an open invitation, the invitation is answered: the
# community starts with the privileges promised, remembers who invited
# it, and the lord warms to the Jews (his settlers came). A player lord
# is told; Kehillah players in range hear of it.
kehillah_invitation_answered_effect = {
	if = {
		limit = {
			exists = primary_title
			domicile.domicile_location.county.holder.top_liege.primary_title ?= { has_variable = kehillah_invitation_open }
		}
		save_scope_as = kehillah_answering_leader
		domicile.domicile_location.county.holder.top_liege = { save_scope_as = kehillah_inviter }
		domicile.domicile_location.county = { save_scope_as = kehillah_invited_county }
		primary_title = {
			change_variable = { name = kehillah_var_prosperity add = kehillah_invitation_answered_prosperity }
			change_variable = { name = kehillah_var_stability add = kehillah_invitation_answered_stability }
			set_variable = { name = kehillah_founded_by_invitation value = scope:kehillah_inviter }
		}
		# 2026-11-25: a community founded in the county the lord named
		# closes his call (docs/spec/2026-11-25-invitation-letters.md).
		scope:kehillah_inviter.primary_title = {
			if = {
				limit = { var:kehillah_invitation_county ?= scope:kehillah_invited_county }
				remove_variable = kehillah_invitation_county
			}
		}
		scope:kehillah_inviter = {
			kehillah_shift_outlook_effect = { AMOUNT = kehillah_invitation_answered_outlook }
			if = {
				limit = { is_ai = no }
				send_interface_message = {
					type = msg_kehillah_community_news
					title = kehillah_news_invitation_answered_t
					desc = kehillah_news_invitation_answered_d
					left_icon = scope:kehillah_answering_leader
				}
			}
		}
		kehillah_notify_region_players_effect = {
			TYPE = msg_kehillah_community_news
			TITLE = kehillah_news_invitation_answered_t
			DESC = kehillah_news_invitation_answered_d
		}
		debug_log_scopes = yes
		debug_log = "kehillah_invitation_answered_effect: settlers answered an invitation"
	}
}

# Scope: the Shtadlan performing Seek an Invitation; scope:councillor_
# liege set. Keeps (or picks: the warmest outlook) the lord he courts,
# warms him, and each month may win an audience. At the audience the
# lord raises his policy one rung if his open-door net pressure, with the
# embassy's arguments, carries it (to Encouraged: an invitation, sent by
# the ordinary raise effect); otherwise he declines and the question is
# closed for the review cooldown, and the Shtadlan turns to another lord.
kehillah_seek_invitation_monthly_effect = {
	save_scope_as = kehillah_suitor_shtadlan
	scope:councillor_liege = { save_scope_as = kehillah_suitor_leader }
	scope:kehillah_suitor_leader = {
		if = {
			limit = {
				exists = var:kehillah_invitation_suitor
				var:kehillah_invitation_suitor = { kehillah_invitation_suitor_target_trigger = yes }
			}
			var:kehillah_invitation_suitor = { save_scope_as = kehillah_invitation_suitor }
		}
		else = {
			if = {
				limit = { has_variable = kehillah_invitation_suitor }
				remove_variable = kehillah_invitation_suitor
			}
			ordered_ruler = {
				limit = { kehillah_invitation_suitor_target_trigger = yes }
				order_by = kehillah_invitation_suitor_score_value
				save_scope_as = kehillah_invitation_suitor
			}
			if = {
				limit = { exists = scope:kehillah_invitation_suitor }
				set_variable = { name = kehillah_invitation_suitor value = scope:kehillah_invitation_suitor }
				if = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_kehillah_community_news
						title = kehillah_seek_invitation_sets_out_t
						desc = kehillah_seek_invitation_sets_out_d
						left_icon = scope:kehillah_suitor_shtadlan
						right_icon = scope:kehillah_invitation_suitor
					}
				}
			}
		}
	}
	if = {
		limit = { exists = scope:kehillah_invitation_suitor }
		scope:kehillah_invitation_suitor = {
			add_opinion = {
				target = scope:kehillah_suitor_leader
				modifier = kehillah_shtadlan_petition_opinion
			}
		}
		random = {
			chance = kehillah_council_petition_chance_display_value
			kehillah_seek_invitation_audience_effect = yes
		}
	}
}

# The audience (kehillah_seek_invitation_monthly_effect, or the debug
# kehillah_outlook_debug.6). Scopes: kehillah_suitor_shtadlan,
# kehillah_suitor_leader, kehillah_invitation_suitor saved.
kehillah_seek_invitation_audience_effect = {
	scope:kehillah_invitation_suitor = {
		primary_title = { save_scope_as = kehillah_suitor_realm }
		if = {
			limit = { kehillah_open_door_net_pressure_value <= kehillah_invitation_embassy_threshold }
			primary_title = { set_variable = { name = kehillah_policy_reason value = flag:petition } }
			kehillah_raise_jewish_settlement_policy_effect = yes
			kehillah_shift_outlook_effect = { AMOUNT = 5 }
			debug_log = "kehillah_seek_invitation_monthly_effect: the lord heard the Shtadlan and raised his policy"
			scope:kehillah_suitor_leader = {
				if = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_kehillah_community_news
						title = kehillah_seek_invitation_granted_t
						desc = kehillah_seek_invitation_granted_d
						left_icon = scope:kehillah_suitor_shtadlan
						right_icon = scope:kehillah_invitation_suitor
					}
				}
			}
		}
		else = {
			kehillah_start_policy_review_cooldown_effect = yes
			debug_log = "kehillah_seek_invitation_monthly_effect: the lord heard the Shtadlan and declined"
			scope:kehillah_suitor_leader = {
				if = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_kehillah_community_news_bad
						title = kehillah_seek_invitation_declined_t
						desc = kehillah_seek_invitation_declined_d
						left_icon = scope:kehillah_suitor_shtadlan
						right_icon = scope:kehillah_invitation_suitor
					}
				}
			}
		}
	}
	scope:kehillah_suitor_leader = { remove_variable = kehillah_invitation_suitor }
}

# V44 (2026-10-15, Daniel). Scope: an AI Jewish landless adventurer. While
# a realm within reach keeps its invitation open, he may take his band
# there (kehillah_invitation.0010 on arrival), where Found a Jewish
# Community weighs heavily with him.
kehillah_invitation_adventurer_effect = {
	save_scope_as = kehillah_drawn_adventurer
	random_independent_ruler = {
		limit = {
			primary_title = { has_variable = kehillah_invitation_open }
			exists = capital_province
			capital_province = {
				squared_distance = {
					target = scope:kehillah_drawn_adventurer.location
					value <= squared_distance_large
				}
			}
		}
		random_realm_county = {
			limit = { kehillah_founding_site_trigger = yes }
			title_province = { save_scope_as = kehillah_invited_site }
		}
	}
	if = {
		limit = { exists = scope:kehillah_invited_site }
		start_travel_plan = {
			destination = scope:kehillah_invited_site
			return_trip = no
			on_arrival_event = kehillah_invitation.0010
		}
		debug_log = "kehillah_invitation_adventurer_effect: a Jewish adventurer sets out for an inviting realm"
	}
}
