# 2026-11-01. Struggle-style feedback for actual community-situation catalysts.
# We cannot reuse msg_catalyst_triggered: its text requires code-provided Struggle fields.
# Each group dispatch below reads the very same script value as its native future-phase binding.
# NOTICE=no is reserved for recurring drivers whose existing narrative report carries the notice.

# Scope: situation. Snapshot BEFORE firing: a threshold-crossing catalyst may change the phase,
# seed another catalyst and clear the live race counters before this effect returns.
kehillah_crisis_apply_catalyst_effect = {
 save_scope_value_as = { name = kehillah_notice_$CATALYST$_points value = $POINTS$ }
 trigger_situation_catalyst = $CATALYST$
 if = {
  limit = { always = $NOTICE$ scope:kehillah_notice_$CATALYST$_points > 0 }
  kehillah_crisis_catalyst_notice_effect = { CATALYST = $CATALYST$ POINTS = scope:kehillah_notice_$CATALYST$_points TRACK = $TRACK$ }
 }
}

# Scope: situation. A player-only notification; it never awards points itself.
kehillah_crisis_catalyst_notice_effect = {
 var:kehillah_crisis_title ?= { save_scope_as = kehillah_notice_title }
 save_scope_value_as = { name = kehillah_notice_points value = $POINTS$ }
 scope:kehillah_notice_title.holder ?= {
  if = {
   limit = { is_ai = no scope:kehillah_notice_points > 0 scope:kehillah_notice_title = { NOT = { has_variable = kehillah_crisis_intro_pending } } }
   send_interface_message = {
    type = msg_kehillah_crisis_catalyst
    title = $CATALYST$
    desc = kehillah_crisis_catalyst_progress_$TRACK$
    left_icon = scope:kehillah_notice_title
   }
  }
 }
}

# Scope: player leader. Merge the existing short narrative pool with a mechanical footer. Random
# choices and the existing per-pool last-variant variables remain the same as the flavor helper.
kehillah_crisis_flavor_catalyst_effect = {
 save_scope_value_as = { name = kehillah_notice_points value = $POINTS$ }
 random_courtier = { limit = { is_adult = yes is_jewish_character_trigger = yes } save_scope_as = kehillah_toast_elder }
 domicile.domicile_location.county.holder ?= {
  random_courtier = { limit = { is_adult = yes NOT = { is_jewish_character_trigger = yes } NOT = { this ?= scope:kehillah_crisis_announce_agitator } } save_scope_as = kehillah_toast_townsman }
 }
 random_list = {
  10 = {
   trigger = { NOT = { var:kehillah_toast_last_$KEY$ ?= 1 } }
   kehillah_crisis_flavor_catalyst_send_effect = { KEY = $KEY$ N = 1 PROGRESS = $PROGRESS$ }
  }
  10 = {
   trigger = { NOT = { var:kehillah_toast_last_$KEY$ ?= 2 } }
   kehillah_crisis_flavor_catalyst_send_effect = { KEY = $KEY$ N = 2 PROGRESS = $PROGRESS$ }
  }
  10 = {
   trigger = { exists = scope:kehillah_toast_townsman NOT = { var:kehillah_toast_last_$KEY$ ?= 3 } }
   kehillah_crisis_flavor_catalyst_send_effect = { KEY = $KEY$ N = 3 PROGRESS = $PROGRESS$ }
  }
  10 = {
   trigger = { exists = scope:kehillah_toast_elder NOT = { var:kehillah_toast_last_$KEY$ ?= 4 } }
   kehillah_crisis_flavor_catalyst_send_effect = { KEY = $KEY$ N = 4 PROGRESS = $PROGRESS$ }
  }
 }
}
kehillah_crisis_flavor_catalyst_send_effect = {
 set_variable = { name = kehillah_toast_last_$KEY$ value = $N$ }
 send_interface_message = {
  type = msg_kehillah_crisis_catalyst
  title = $KEY$_t$N$
  desc = { desc = $KEY$_d$N$ desc = kehillah_crisis_catalyst_report_$PROGRESS$ }
  left_icon = scope:kehillah_crisis_announce_title
 }
}
