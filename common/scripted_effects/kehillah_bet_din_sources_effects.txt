# Clear only the recipient's reservation. The activity owns the answer.
kehillah_din_clear_letter_effect = {
    if = { limit = { has_variable = kehillah_din_letter_activity } remove_variable = kehillah_din_letter_activity }
    if = { limit = { has_variable = kehillah_din_letter_case } remove_variable = kehillah_din_letter_case }
    if = { limit = { has_variable = kehillah_din_reply_quality } remove_variable = kehillah_din_reply_quality }
}
kehillah_din_start_case_effect = {
    if = { limit = { exists = involved_activity involved_activity.var:kehillah_din_case = $CASE$ }
        if = { limit = { involved_activity = { has_variable = kehillah_din_pending } } involved_activity = { remove_variable = kehillah_din_pending } }
        trigger_event = $EVENT$
    }
}
kehillah_din_adopt_advice_effect = {
    if = { limit = { exists = involved_activity.var:kehillah_din_reply_author NOT = { involved_activity = { has_variable = kehillah_din_advice_used } } }
        involved_activity = { set_variable = { name = kehillah_din_advice_used value = 1 } }
        involved_activity.var:kehillah_din_reply_author = { save_scope_as = kds_adopted_author }
        involved_activity.var:kehillah_bet_din_convening_title = { set_variable = { name = kehillah_din_precedent_$CASE$ value = scope:kds_adopted_author } }
        if = { limit = { scope:kds_adopted_author = { is_alive = yes } }
            scope:kds_adopted_author = {
                add_prestige = 10
                if = { limit = { has_trait = kehillah_rabbi_trait } add_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value = 1 } }
                add_opinion = { target = root modifier = kehillah_din_consulted_opinion }
                send_interface_message = { type = event_generic_good title = kehillah_din_adopted_title desc = kehillah_din_adopted_desc }
            }
        }
    }
}

# Correspondents for the four new case questions.

# Each litigant reacts to what the order does for them. The quality of the
# hearing then shifts both reactions: a careful hearing may be accepted even
# by a disappointed party, while a poorly argued one strains both relations.
# Root is the convening leader; the active case and direction are on activity.
kehillah_din_case_party_response_effect = {
	if = {
		limit = { involved_activity.var:$DIRECTION$ = 1 }
		if = {
			limit = { involved_activity.var:kehillah_din_case = 1 }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
		}
		else = {
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
		}
	}
	else_if = {
		limit = { involved_activity.var:$DIRECTION$ = 2 }
		if = {
			limit = { involved_activity.var:kehillah_din_case = 14 }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_din_case = 12 }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_din_case = 8 involved_activity.var:kehillah_din_charity_funded = 0 }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
		}
		else_if = {
			limit = { involved_activity.var:kehillah_din_case = 1 involved_activity.var:kehillah_bet_din_tally < kehillah_bet_din_tally_threshold_good }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
		}
		else = {
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
			if = {
				limit = { OR = { involved_activity.var:kehillah_din_case = 1 involved_activity.var:kehillah_din_case = 8 involved_activity.var:kehillah_din_case = 9 involved_activity.var:kehillah_din_case = 10 } }
				involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
			}
			else = { involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } } }
		}
	}
	else = {
		if = {
			limit = { OR = { involved_activity.var:kehillah_din_case = 8 involved_activity.var:kehillah_din_case = 12 } }
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
		}
		else = {
			involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_dispute_ruling_disfavor_opinion } }
			involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_dispute_ruling_favor_opinion } }
		}
	}
	if = {
		limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_great }
		involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_din_well_reasoned_opinion } }
		involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_din_well_reasoned_opinion } }
	}
	else_if = {
		limit = { involved_activity.var:kehillah_bet_din_tally >= kehillah_bet_din_tally_threshold_good }
		involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_din_workable_hearing_opinion } }
		involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_din_workable_hearing_opinion } }
	}
	else = {
		involved_activity.var:kehillah_bet_din_litigant_a = { add_opinion = { target = root modifier = kehillah_din_poor_hearing_opinion } }
		involved_activity.var:kehillah_bet_din_litigant_b = { add_opinion = { target = root modifier = kehillah_din_poor_hearing_opinion } }
	}
}
