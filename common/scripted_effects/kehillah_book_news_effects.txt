# The chosen identity is stored on the author through the delayed news event.
kehillah_book_choose_work_effect = {
	if = {
		limit = { var:kehillah_book_genre = flag:hashkafa }
		random_list = {
			100 = { set_variable = { name = kehillah_book_work value = flag:hash_discourses } }
			100 = { set_variable = { name = kehillah_book_work value = flag:hash_name } }
			100 = { set_variable = { name = kehillah_book_work value = flag:hash_depths } }
		}
	}
	else_if = {
		limit = { var:kehillah_book_genre = flag:parshanut }
		random_list = {
			100 = { set_variable = { name = kehillah_book_work value = flag:parsh_shemot } }
			100 = { set_variable = { name = kehillah_book_work value = flag:parsh_ladder } }
			100 = { set_variable = { name = kehillah_book_work value = flag:parsh_wells } }
		}
	}
	else_if = {
		limit = { var:kehillah_book_genre = flag:talmudics }
		if = {
			limit = { exists = var:kehillah_book_bet_din_inspiration any_memory = { has_memory_type = kehillah_bet_din_case1_memory } }
			set_variable = { name = kehillah_book_work value = flag:talm_dowry }
		}
		else_if = {
			limit = { exists = var:kehillah_book_bet_din_inspiration any_memory = { has_memory_type = kehillah_bet_din_case2_memory } }
			set_variable = { name = kehillah_book_work value = flag:talm_agunah }
		}
		else = {
			random_list = {
				100 = { set_variable = { name = kehillah_book_work value = flag:talm_damages } }
				100 = { set_variable = { name = kehillah_book_work value = flag:talm_found } }
				100 = { set_variable = { name = kehillah_book_work value = flag:talm_witness } }
			}
		}
	}
	else = {
		random_list = {
			100 = { set_variable = { name = kehillah_book_work value = flag:poetry_shabbat } }
			100 = { set_variable = { name = kehillah_book_work value = flag:poetry_rivers } }
			100 = { set_variable = { name = kehillah_book_work value = flag:poetry_dawn } }
		}
	}
}
