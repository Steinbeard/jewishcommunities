# Neighbouring Kehillot learn of a crisis through the same bounded local
# network used by the Settlement Conditions mutual-aid contribution: the
# vanilla squared_distance_medium band.  A crisis is therefore visible to
# communities that can plausibly send people, money, or news quickly, rather
# than to every Kehillah in a large empire.
#
# Scope: the leader of the community facing the crisis.
kehillah_crisis_neighbourhood_alert_effect = {
	primary_title = {
		save_scope_as = kehillah_network_crisis_title
		if = {
			limit = { exists = holder.domicile.domicile_location }
			holder.domicile.domicile_location = { save_scope_as = kehillah_network_crisis_location }
			every_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					NOT = { this = scope:kehillah_network_crisis_title }
					exists = holder
					holder = {
						is_alive = yes
						government_has_flag = government_is_kehillah
						NOT = { kehillah_trouble_crisis_active_trigger = yes }
						exists = domicile.domicile_location
						domicile.domicile_location = {
							squared_distance = {
								target = scope:kehillah_network_crisis_location
								value <= squared_distance_medium
							}
						}
					}
				}
				holder = {
					set_variable = { name = kehillah_network_crisis_title value = scope:kehillah_network_crisis_title }
					set_variable = { name = kehillah_network_crisis_epoch value = scope:kehillah_network_crisis_title.var:kehillah_crisis_epoch }
					trigger_event = { id = kehillah_crisis_network.0001 days = 1 }
				}
			}
		}
	}
}

# Scope: the leader of the community whose crisis has received its political
# settlement.  The same bounded neighbourhood receives a distinct report of
# the outcome, so aid can match an averted threat, a struck quarter, or an
# expulsion rather than arriving as an anonymous generic request.
kehillah_crisis_neighbourhood_resolution_effect = {
	primary_title = {
		save_scope_as = kehillah_network_crisis_title
		if = {
			limit = { exists = holder.domicile.domicile_location }
			holder.domicile.domicile_location = { save_scope_as = kehillah_network_crisis_location }
			every_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					NOT = { this = scope:kehillah_network_crisis_title }
					exists = holder
					holder = {
						is_alive = yes
						government_has_flag = government_is_kehillah
						NOT = { kehillah_trouble_crisis_active_trigger = yes }
						exists = domicile.domicile_location
						domicile.domicile_location = {
							squared_distance = {
								target = scope:kehillah_network_crisis_location
								value <= squared_distance_medium
							}
						}
					}
				}
				holder = {
					set_variable = { name = kehillah_network_crisis_title value = scope:kehillah_network_crisis_title }
					set_variable = { name = kehillah_network_crisis_epoch value = scope:kehillah_network_crisis_title.var:kehillah_crisis_epoch }
					trigger_event = { id = kehillah_crisis_network.0002 days = 1 }
					# Mutual aid v2: this report is their letter for the
					# need opened below; no second letter.
					set_variable = { name = kehillah_aid_letter_covered value = scope:kehillah_network_crisis_title days = 10 }
				}
			}
		}
	}
	# 2026-11-20, mutual aid v2: a struck quarter asks the communities in
	# reach for rebuilding, an expelled one for help to resettle.
	kehillah_aid_crisis_need_effect = yes
}
