# 2026-11-06, Daniel: a player may look at an option's odds and go back. An option in a category
# scene is only considered; the final choice in the next scene commits the category.
kehillah_crisis_consider_action_effect = {
 primary_title = {
  set_variable = { name = kehillah_crisis_pending_category value = flag:$CATEGORY$ }
  set_variable = { name = kehillah_crisis_pending_story value = flag:$STORY$ }
 }
}
kehillah_crisis_clear_considered_action_effect = {
 primary_title = {
  remove_variable = kehillah_crisis_pending_category
  remove_variable = kehillah_crisis_pending_story

}
