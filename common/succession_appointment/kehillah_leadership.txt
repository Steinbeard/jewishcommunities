# =====================================================================
# Kehillah leadership -- meritocratic appointment succession.
#
# Source of truth: docs/spec/v1-kehillah-community.md section 5.
#
# Forked in STRUCTURE from vanilla's admin_governor, but the scoring is
# written from scratch. Vanilla's governor score is a flat sum of all
# five skills plus a long tail of administrative, claim, war and scheme
# modifiers. None of that belongs here: the whole point of spec
# section 2 is that Kehillah leadership rewards competence and Torah
# knowledge, not court manoeuvring.
#
# =====================================================================
# THE POOL IS FAMILY-ONLY, AND THAT IS NOT WHAT THE SPEC ASKED FOR.
# READ THIS BEFORE TRYING TO WIDEN IT.
#
# Spec section 5 wants an open pool: notable family heads, not just the
# outgoing leader's kin. This entry cannot currently deliver that, and
# the reason is an engine limitation rather than a design choice.
#
# The candidate categories documented in
# common/succession_appointment/_succession_appointment.info are:
#
#   holder_close_family, holder_close_extended_family,
#   holder_house_member, landed_vassal, landed_vassal_close_family,
#   landed_vassal_close_extended_family, landed_vassal_house_member,
#   unlanded_noble_house_head, unlanded_noble_close_family,
#   unlanded_noble_close_extended_family, unlanded_noble_house_member,
#   holder_councilor, holder_court_position, direct_subject
#
# For an independent, landless, council-less Kehillah they sort into
# three groups:
#
#   1. Family categories. Work, but are all kin.
#   2. Vassal and noble-house categories. Administrative-government
#      constructs. They parse, but contribute nobody here.
#   3. holder_councilor, holder_court_position, direct_subject. These
#      are the three that WOULD have opened the pool to the community's
#      officers. **They are documented but not implemented.** Grepping
#      the 1.19 files shows no vanilla content using any of them, and
#      the first playtest confirmed it the hard way: including
#      holder_court_position made the parser reject the whole
#      invested_candidates statement, which left this entry undefined,
#      which left the succession law pointing at a missing appointment
#      type, which crashed the game on load with
#      "Failed build succession ... unhandled succession order [invalid]"
#      (succession_order.cpp:704).
#
# So the officers cannot be candidates through this system. What
# remains is family, which is an honest v1 state but is NOT the design.
#
# The route to the real thing is almost certainly to stop using
# succession_appointment for this and script the handover instead: on
# the leader's death, run an effect that scores every eligible member
# of the community and grants the title to the winner. That is more
# work, but it removes the category restriction entirely and it is the
# same machinery a player-chosen successor and a retirement decision
# would need anyway. Logged in docs/iteration/v1-iteration-notes.md.
#
# The candidate_score below is written as though the pool were already
# open -- the office-holding bonus, the culture check and the
# faith check all still make sense, and several will apply to family
# members who hold offices. None of it needs rewriting when the pool
# widens.
# =====================================================================

kehillah_leadership = {
	default_candidates = { holder_close_family }

	# holder_court_position belongs here and cannot go here. See the
	# header. Do not re-add it without testing that the game still
	# loads.

	# Communal leadership is not a child's office. A minor cannot lead a
	# congregation, and unlike vanilla's admin_emperor there is no realm
	# to hold together under a regency, so there is no reason to allow
	# it.
	allow_children = no

	candidate_score = {
		value = {
			# -----------------------------------------------------
			# THE MERIT CORE (spec section 5: Learning, communal
			# standing, dynasty prestige).
			#
			# Learning is weighted heaviest and Stewardship second.
			# Diplomacy contributes at half rate for the Shtadlan
			# path. Martial and Intrigue contribute NOTHING -- that
			# absence is the design, not an oversight. A gifted
			# schemer has no claim on this office.
			# -----------------------------------------------------
			add = {
				value = learning
				multiply = 3
				desc = kehillah_score_learning_desc
			}

			add = {
				value = stewardship
				multiply = 2
				desc = kehillah_score_stewardship_desc
			}

			add = {
				value = diplomacy
				multiply = 0.5
				desc = kehillah_score_diplomacy_desc
			}

			# -----------------------------------------------------
			# STANDING IN THE COMMUNITY -- REMOVED 2026-09-07, READ
			# BEFORE RE-ADDING SOMETHING SIMILAR.
			#
			# This used to read influence_level, a per-CHARACTER
			# bucketed tier off the vanilla Influence resource: each
			# candidate could have their own personal standing, so the
			# term genuinely differentiated candidates. When Influence
			# was retired (common/governments/kehillah_government.txt)
			# the review that caught the character-vs-title scoping bug
			# also caught this: kehillah_var_stability/
			# kehillah_var_greatness correctly moved onto the TITLE
			# (see kehillah_quarterly_pillars_effect's "WHY THE TITLE,
			# NOT THE CHARACTER" note) because they represent the
			# community's standing, not any one person's. But
			# candidate_score is evaluated once per candidate against
			# the SAME title -- scope:title never changes between
			# candidates in one succession event -- so a term reading
			# scope:title.var:... adds the identical number to every
			# candidate and changes nothing about their relative order.
			# An earlier draft of this section left that term in with a
			# 0.1 multiplier, which was dead code: consistent, harmless,
			# and pure waste. Removed rather than kept "just in case".
			# If a personal standing signal is wanted again, it needs a
			# per-candidate source (a character-scoped variable, a
			# trait, something scope:title.holder-independent) -- not
			# a resurrection of this term.
			# -----------------------------------------------------

			# Holding one of the community offices is the single
			# strongest non-skill signal. You earned a seat; the
			# congregation has watched you work.
			if = {
				limit = {
					scope:title.holder ?= {
						any_court_position_holder ?= { this = root }
					}
				}
				add = {
					value = 25
					desc = kehillah_score_officer_desc
				}
			}

			# Piety, as visible religious standing. Modest weight:
			# this is a learned office, not a saintly one.
			if = {
				limit = { piety_level >= 2 }
				add = {
					value = {
						value = piety_level
						multiply = 4
					}
					desc = kehillah_score_piety_desc
				}
			}

			# Dynasty prestige, per spec section 5. Kept deliberately
			# small next to Learning: lineage counts for something,
			# but it must never outweigh scholarship, or the whole
			# meritocratic premise collapses back into ordinary CK3
			# dynasty play.
			if = {
				limit = { exists = dynasty }
				add = {
					value = dynasty.dynasty_prestige_level
					multiply = 3
					desc = kehillah_score_dynasty_prestige_desc
				}
			}

			# -----------------------------------------------------
			# AGE AND EXPERIENCE
			#
			# Real communal leadership skewed old, and the Worms
			# scenario leans on that explicitly (scenario doc
			# section 3 picks an older birth year so the documented
			# ~1070 death reads as an elder's natural end). Reward
			# maturity, then taper: the very old are revered but
			# increasingly unable to serve.
			# -----------------------------------------------------
			if = {
				limit = { age >= 35 }
				add = {
					value = 10
					desc = kehillah_score_mature_desc
				}
			}
			if = {
				limit = { age >= 50 }
				add = {
					value = 10
					desc = kehillah_score_elder_desc
				}
			}
			if = {
				limit = { age >= 70 }
				subtract = {
					value = 15
					desc = kehillah_score_very_aged_desc
				}
			}

			# -----------------------------------------------------
			# TRAITS
			# -----------------------------------------------------

			# Scholarly and teacherly traits.
			if = {
				limit = { has_trait = theologian }
				add = {
					value = 15
					desc = kehillah_score_theologian_desc
				}
			}
			if = {
				limit = { has_trait = scholar }
				add = {
					value = 10
					desc = kehillah_score_scholar_desc
				}
			}

			# Character traits the congregation weighs. These are
			# drawn from Judaism's own listed virtue and sin traits
			# in vanilla's common/religion/religions/00_judaism.txt,
			# rather than a generic good-and-bad list, so the
			# community judges by its own faith's standards.
			if = {
				limit = { has_trait = just }
				add = {
					value = 8
					desc = kehillah_score_just_desc
				}
			}
			if = {
				limit = { has_trait = patient }
				add = {
					value = 6
					desc = kehillah_score_patient_desc
				}
			}
			if = {
				limit = { has_trait = compassionate }
				add = {
					value = 6
					desc = kehillah_score_compassionate_desc
				}
			}
			if = {
				limit = { has_trait = arbitrary }
				subtract = {
					value = 12
					desc = kehillah_score_arbitrary_desc
				}
			}
			if = {
				limit = { has_trait = greedy }
				subtract = {
					value = 10
					desc = kehillah_score_greedy_desc
				}
			}

			# Cannot lead in practice.
			if = {
				limit = {
					OR = {
						has_trait = incapable
						has_trait = infirm
					}
				}
				subtract = {
					value = 100
					desc = kehillah_score_incapable_desc
				}
			}

			# -----------------------------------------------------
			# FAITH AND CULTURE
			#
			# A Kehillah leader must be of the community's faith.
			# This is not snobbery about outsiders, it is what the
			# office IS -- the person who rules on religious law for
			# this congregation. A hard, very large penalty rather
			# than an outright exclusion, so that if the pool somehow
			# empties the game still has an answer instead of
			# deadlocking.
			# -----------------------------------------------------
			if = {
				limit = {
					NOT = { faith = scope:title.holder.faith }
				}
				subtract = {
					value = 500
					desc = kehillah_score_wrong_faith_desc
				}
			}

			if = {
				limit = {
					culture ?= scope:title.holder.culture
				}
				add = {
					value = 10
					desc = kehillah_score_same_culture_desc
				}
			}

			# -----------------------------------------------------
			# Personal ties to the outgoing leader. Small, and
			# smaller than the officer bonus above, so favouritism
			# never beats demonstrated competence.
			# -----------------------------------------------------
			if = {
				limit = {
					scope:title.holder ?= { has_relation_friend = root }
				}
				add = {
					value = 5
					desc = kehillah_score_friend_desc
				}
			}
		}
	}
}
