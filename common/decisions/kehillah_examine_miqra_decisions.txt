# Starts a continuing Karaite study course, parallel to but independent of
# the rabbinic Learn Torah library scheme.
kehillah_examine_miqra_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_tale.dds" }
	decision_group_type = kehillah
	sort_order = 95
	ai_check_interval = 60

	is_shown = {
		is_jewish_character_trigger = yes
		rite = { rite_has_tenet = tenet_baalei_hamiqra }
		OR = {
			government_has_flag = government_is_kehillah
			is_ruler = yes
			has_government = landless_adventurer_government
		}
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		custom_tooltip = {
			text = kehillah_examine_miqra_already_studying_tt
			NOT = { any_scheme = { type = kehillah_examine_miqra } }
		}
	}

	is_valid_showing_failures_only = {
		can_start_scheme = { type = kehillah_examine_miqra target_character = root }
	}

	cost = { piety = 0 }

	effect = {
		start_scheme = { type = kehillah_examine_miqra target_character = root }
	}
}
