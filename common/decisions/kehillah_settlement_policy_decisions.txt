# =====================================================================
# Raise / Lower Jewish Settlement Policy. ROADMAP S9a, added 2026-09-29.
#
# The host-ruler side of V16's realm policy: the first thing in ordinary
# play that writes kehillah_jewish_settlement_policy (before this, only
# debug events 72-75 did). This is V17 Slice B's *voluntary* half only --
# a ruler choosing to review the policy -- and deliberately nothing else:
#
# - ONE RUNG PER DECISION, never two (V17 §4.2).
# - FIVE-YEAR REVIEW COOLDOWN on the realm title, shared by both
#   decisions (V17 §4.1's recommended per-realm cooldown). It lives on the
#   title, not the ruler, so a successor inherits a policy AND the fact
#   that it was just reviewed.
# - LOWER STOPS AT DISCOURAGED. Banned belongs to S9b, which builds the
#   warning, counterplay and exit that V17 §6 says must exist before any
#   expulsion outcome can. Until then Banned is reachable only by debug
#   event 75; Raise still climbs out of it, so a debug-banned realm is
#   never stuck.
# - NO AI USE (ai_potential = always = no). V17 wants AI policy change to
#   be event-led with a named reason; that design is with Daniel.
# - EXISTING CHARTERS ARE NEVER REWRITTEN (V16 §4.1). Only the realm
#   policy moves. What changes for a community already in the realm is
#   its ledger badge and, where its rights now exceed the policy, the
#   grandfathered "charter review" line -- both already built and both
#   read the policy live -- plus the notice this decision sends.
#
# Scope of everything below: the ruler taking the decision. The policy
# is stored on the ruler's own primary title, which is exactly where a
# Kehillah resolves it (domicile county -> holder -> top_liege ->
# primary_title), hence the independence requirement: a vassal's title
# is never the one its communities read.
# =====================================================================

kehillah_raise_jewish_settlement_policy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}

	decision_group_type = realm
	sort_order = 40
	ai_check_interval = 0

	is_shown = {
		is_landed = yes
		is_independent_ruler = yes
		exists = primary_title
		NOT = { government_has_flag = government_is_kehillah }
		NOT = { primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes } }
	}

	is_valid = {
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_policy_review_cooldown_tt
			NOT = { primary_title = { kehillah_policy_review_on_cooldown_trigger = yes } }
		}
	}

	cost = {
		piety = kehillah_policy_raise_piety_cost
	}

	effect = {
		kehillah_raise_jewish_settlement_policy_effect = yes
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
