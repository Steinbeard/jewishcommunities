# =====================================================================
# VISIT THE JEWISH COMMUNITY -- 2026-11-14 (Daniel). A Jewish landless
# adventurer who is standing where a Jewish community lives may call on it.
# It replaces the "Seek Out Jewish Families" decision proposed the same day.
#
# The decision only finds the community and opens kehillah_visit.0001,
# where the choices are: pray with the congregation, take a family into
# the company, or speak with the leader about the seat. Families come from
# the community's local Jewish culture and rite, through the household code
# the communities already use for their own newcomers.
#
# NO AI USE. Adventurers the AI plays find their followers as before.
# =====================================================================

kehillah_visit_community_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 94
	cooldown = { months = 6 }

	is_shown = {
		has_dlc_feature = roads_to_power
		has_government = landless_adventurer_government
		is_jewish_character_trigger = yes
		exists = location.county
		any_in_global_list = {
			variable = kehillah_registered_communities
			kehillah_visit_title_here_trigger = { VISITOR = root }
		}
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		is_travelling = no
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	effect = {
		hidden_effect = {
			random_in_global_list = {
				variable = kehillah_registered_communities
				limit = {
					kehillah_visit_title_here_trigger = { VISITOR = root }
				}
				save_scope_as = kehillah_visit_title
			}
			if = {
				limit = { exists = scope:kehillah_visit_title }
				trigger_event = kehillah_visit.0001
			}
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
