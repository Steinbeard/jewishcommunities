# =====================================================================
# INVEST IN THE COMMUNITY'S LIVELIHOODS (2026-11-26)
# Spec: docs/spec/2026-11-26-pillar-ceilings-deeds-and-founding.md, section 4
#
# Daniel (2026-11-26): "Maybe you increase prosperity by spending money on
# things?" Affluence was the pillar a player could do least about: Greatness
# has sermons, books and the yeshiva, Stability has tzedakah and the Bet
# Din, and Affluence had only the commenda (which pays gold back, not
# standing). This is the direct lever: the leader puts communal gold into
# the community's own trades, once a year.
#
# The gain is a DEED (kehillah_pillar_model_values.txt): it counts at once
# and fades over the years, so keeping Affluence up by spending means
# spending again. A yearly moderate gift holds about five times its size;
# that is what makes it a real lever without making it permanent.
# kehillah_change_pillar_effect applies the Commercial style's boost.
# =====================================================================

kehillah_invest_in_trades_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds" }
	decision_group_type = kehillah
	sort_order = 86
	ai_check_interval = 24

	is_shown = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = kehillah_invest_needs_gold_tt
			gold >= kehillah_invest_modest_cost_value
		}
	}

	cooldown = { years = 1 }

	effect = {
		custom_tooltip = kehillah_invest_in_trades_tt
		trigger_event = kehillah_invest.0001
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 15
		modifier = { add = 15 has_character_modifier = kehillah_style_commercial_modifier }
		modifier = { add = 10 stewardship >= 12 }
		# Keep a purse: only invest with three times the modest stake in hand.
		modifier = {
			factor = 0
			gold < {
				value = kehillah_invest_modest_cost_value
				multiply = 3
			}
		}
	}
}
