# =====================================================================
# AFFLUENCE SINKS: SPECIFIC THINGS A COMMUNITY SPENDS ON (2026-11-26)
# Spec: docs/spec/2026-11-26-pillar-ceilings-deeds-and-founding.md, section 5
#
# Daniel (2026-11-26): raise Affluence by "spending on some specific
# sinks we flavorfully make". This replaces the first pass of the same
# day (one generic "Invest in the Community's Livelihoods" decision).
#
# Five sinks, each something a medieval kehillah really spent its money
# on, each with its own cooldown and size, and each paying more when the
# Quarter has the building that does that work. Every gain is a DEED
# (kehillah_pillar_model_values.txt): it counts at once and fades over
# the years, so standing kept by spending has to be kept up.
#
#   Stock the Free-Loan Chest   gemilut chasadim: interest-free loans to
#                               members. Affluence + a little Stability.
#                               More with the Kuppah.
#   Take Stalls at the Fair     the community's traders at the season's
#                               fair. More with Fair Stalls.
#   Send Apprentices Abroad     youths learn a trade in another community
#                               and come home years later. Pays LATE and
#                               large. More with a Craft Row.
#   Lay In Wheat and Wine       bought wholesale after the harvest, sold to
#                               members at cost before Passover. Affluence +
#                               Stability. More with the Matzah Bakery.
#   Pave and Drain the Street   a costly, rare public work. Affluence +
#                               Stability, both moderate.
#
# All available to a newly founded community: these are the founder's
# first levers, before any specialisation exists. Values:
# common/script_values/kehillah_trade_band_values.txt. All numbers are a
# first pass for the calibration model (spec section 7).
# =====================================================================

# ---------------------------------------------------------------------
kehillah_stock_free_loan_chest_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds" }
	decision_group_type = kehillah
	sort_order = 86
	ai_check_interval = 24

	is_shown = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}
	cost = { gold = kehillah_sink_free_loan_cost_value }
	cooldown = { years = 2 }

	effect = {
		if = {
			limit = { domicile ?= { has_domicile_building_or_higher = kehillah_kuppah_01 } }
			kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = kehillah_sink_free_loan_gain_built }
		}
		else = {
			kehillah_change_pillar_effect = { PILLAR = prosperity AMOUNT = kehillah_sink_free_loan_gain }
		}
		kehillah_change_pillar_effect = { PILLAR = stability AMOUNT = kehillah_sink_free_loan_stability }
	}

	ai_potential = { government_has_flag = government_is_kehillah }
	ai_will_do = {
		base = 15
		modifier = { add = 10 has_character_modifier = kehillah_style_commercial_modifier }
		modifier = { factor = 0 gold < { value = kehillah_sink_free_loan_cost_value multiply = 3 } }
	}
}

# Pays late: the apprentices come home in kehillah_invest.0010, three to
# four years on. The due year is stored on the TITLE and checked by the
# quarterly pulse (kehillah_apprentices_return_check_effect), so a new
# leader who takes office in the meantime still receives them.
kehillah_send_apprentices_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds" }
	decision_group_type = kehillah
	sort_order = 88
	ai_check_interval = 36

	is_shown = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = kehillah_apprentices_away_tt
			primary_title = { NOT = { has_variable = kehillah_apprentices_due_year } }
		}
	}
	cost = { gold = kehillah_sink_apprentices_cost_value }
	cooldown = { years = 5 }

	effect = {
		custom_tooltip = kehillah_send_apprentices_tt
		primary_title = {
			set_variable = {
				name = kehillah_apprentices_due_year
				value = {
					value = current_year
					add = { integer_range = { min = 3 max = 4 } }
				}
			}
		}
	}

	ai_potential = { government_has_flag = government_is_kehillah }
	ai_will_do = {
		base = 10
		modifier = { add = 10 domicile ?= { has_domicile_building_or_higher = kehillah_workshops_01 } }
		modifier = { factor = 0 gold < { value = kehillah_sink_apprentices_cost_value multiply = 3 } }
	}
}
