# V47 heads of rite (docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).

# A truly exceptional rabbi becomes head of a rite that has none, for his
# lifetime (Daniel, 2026-10-19). Shown to Rabbinic community rabbis while
# their rite has no head, so the goal is visible; the AI takes it when able.
kehillah_seek_head_of_rite_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_scholars.dds"
	}
	decision_group_type = major
	sort_order = 85
	ai_check_interval = 12

	is_shown = {
		kehillah_head_of_rite_leader_trigger = yes
		has_trait = kehillah_rabbi_trait
		kehillah_rite_has_no_head_trigger = yes
		NOT = { exists = title:d_kehillah_gaonate.holder }
	}

	is_valid = {
		custom_tooltip = {
			text = kehillah_head_of_rite_requires_yeshiva_tt
			kehillah_head_of_rite_illustrious_yeshiva_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_head_of_rite_requires_rosh_yeshiva_tt
			kehillah_qualified_rosh_yeshiva_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_head_of_rite_requires_tracks_tt
			kehillah_head_of_rite_two_tracks_trigger = yes
		}
		piety_level >= kehillah_head_of_rite_min_piety_level
	}

	cost = {
		prestige = kehillah_head_of_rite_prestige_cost
	}

	effect = {
		if = {
			limit = { kehillah_is_minhag_ashkenaz_trigger = yes }
			custom_tooltip = kehillah_seek_head_of_rite_effect_rabbenu_tt
		}
		else = {
			custom_tooltip = kehillah_seek_head_of_rite_effect_gaon_tt
		}
		hidden_effect = { kehillah_gaonate_grant_effect = yes }
	}

	ai_will_do = {
		base = 100
	}
}

# The head of rite, leading Sura (or the Exilarch's academy at Baghdad),
# makes his recognition an office of the academy: it passes to the
# academy's next head instead of lapsing.
kehillah_establish_gaon_office_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 84
	ai_check_interval = 24

	is_shown = {
		kehillah_is_gaonate_holder_trigger = yes
		NOT = { kehillah_gaonate_is_office_trigger = yes }
	}

	is_valid = {
		custom_tooltip = {
			text = kehillah_gaon_office_requires_academy_tt
			kehillah_leads_geonic_academy_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_head_of_rite_requires_yeshiva_tt
			kehillah_head_of_rite_illustrious_yeshiva_trigger = yes
		}
	}

	cost = {
		prestige = kehillah_gaon_office_prestige_cost
		piety = kehillah_gaon_office_piety_cost
	}

	effect = {
		custom_tooltip = kehillah_establish_gaon_office_effect_tt
		kehillah_gaonate_establish_office_effect = yes
	}

	ai_will_do = {
		base = 100
	}
}
