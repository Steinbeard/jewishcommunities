# =====================================================================
# SEEK A RABBI (Daniel, 2026-10-11).
#
# Replaces this file's three earlier decisions: "Serve as the Community's
# Rabbi", "Stand Down as Rabbi" (ROADMAP S4(2), 2026-09-26) and "Seek a
# Chief Rabbi" (S4(1)). The Chief Rabbi court position is gone; a
# Kehillah's rabbi is now its leader if he carries kehillah_rabbi_trait,
# otherwise its Dayyan (see kehillah_has_community_rabbi_trigger in
# common/scripted_triggers/kehillah_scripted_triggers.txt). So there is
# nothing left to toggle, and the search now fills the Dayyan seat.
#
# Modelled on vanilla's hire_physician_decision (common/decisions/
# 90_minor_decisions.txt): a cooldown flag, then an event a week or two
# later offering candidates at skill tiers and gold costs. The candidates
# and the selection event live in events/kehillah_rabbi_search_events.txt.
#
# A Rav-leader can seek a Dayyan too: the seat is a council seat with its
# own work (the Bet Din's preparation task), whoever the community's
# presiding rabbi is.
# =====================================================================

kehillah_seek_rabbi_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}

	decision_group_type = kehillah
	sort_order = 86
	ai_check_interval = 24

	is_shown = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no

		custom_description = {
			text = kehillah_seek_rabbi_seat_empty_tt
			NOT = { exists = cp:councillor_court_chaplain }
		}

		custom_description = {
			text = kehillah_seek_rabbi_cooldown_tt
			NOT = { has_character_flag = kehillah_seek_rabbi_cooldown }
		}
	}

	effect = {
		custom_tooltip = kehillah_seek_rabbi_tt

		hidden_effect = {
			add_character_flag = {
				flag = kehillah_seek_rabbi_cooldown
				years = 2
			}
			trigger_event = {
				id = kehillah_rabbi_search.0001
				days = { 7 14 }
			}
		}
	}

	# AI communities use the same search. Now that the Dayyan seat
	# requires kehillah_rabbi_trait, this is the main way a rabbi reaches
	# an AI community's council.
	ai_potential = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
	}

	ai_will_do = {
		base = 80

		# Don't bankrupt a poor community chasing the office. The event's
		# own per-option ai_chance already refuses individual candidates
		# it cannot afford; this stops the search being started at all
		# when nothing on the list could possibly be paid for.
		modifier = {
			factor = 0
			short_term_gold < kehillah_rabbi_hire_base_cost
		}
	}
}
