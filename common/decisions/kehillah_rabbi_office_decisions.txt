# =====================================================================
# THE RABBI'S OFFICE -- ROADMAP item S4(2), added 2026-09-26 (Sukkot
# heartbeat 3).
#
# "Serve as the Community's Rabbi", and standing down from it.
#
# WHY A DECISION AND NOT A COURT POSITION. A ruler cannot hold a court
# position in their own court -- that is an engine rule, not something
# this mod can gate around. So a Kehillah leader who is himself a rabbi
# (Worms's Isaac, Mainz's Isaac ben Judah, Troyes's Rashi -- all three
# ship with kehillah_rabbi_trait) could not fill his own community's
# Chief Rabbi seat, and the community lost that whole contribution. A
# leader like Rashi presiding over a community recorded as having "no
# rabbi" reads as a bug, not a design.
#
# Every consumer of the seat asks who is ACTING in the role -- see
# kehillah_has_acting_chief_rabbi_trigger's header (common/
# scripted_triggers/kehillah_scripted_triggers.txt) for the shared
# trigger/value/effect trio this is built on and why it is deliberately
# one place rather than a special case per site.
#
# =====================================================================
# UPDATED 2026-09-27 (S4b). THE OFFICE AND THE SEAT ARE NOW ONE THING.
#
# As first built, the toggle and the court position were independent: a
# leader could take up the office and then also appoint somebody to it,
# and the community read as having two rabbis. Two changes fixed that,
# and they are deliberately ASYMMETRIC:
#
#   * FROM THE SEAT'S SIDE -- appointment is simply refused while the
#     leader holds the office (chief_rabbi_kehillah_position's
#     valid_position, common/court_positions/types/kehillah_officers.txt),
#     and kehillah_seek_chief_rabbi_decision below is greyed out for the
#     same reason rather than running a search whose prize cannot be
#     seated.
#   * FROM THE DECISION'S SIDE -- taking up the office DEPOSES a sitting
#     Chief Rabbi (kehillah_replace_appointed_rabbi_effect), at the cost
#     of the ordinary revoked-position opinion hit. Deciding the
#     community's rabbi should be you is a real move a leader can make,
#     not something the presence of an incumbent forbids.
#
# The asymmetry is the design: the leader's own intent wins over the
# appointment system, and never the other way around.
#
# kehillah_clear_redundant_rabbi_toggle_effect (the quarterly sweep that
# used to be the ONLY thing reconciling the two states) is kept, but it is
# now unreachable in ordinary play -- nothing can produce the
# both-at-once state it cleans up. It stays as a save-migration and
# console-testing backstop, not as load-bearing logic.
# =====================================================================

kehillah_serve_as_rabbi_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}

	# An explicit desc, ADDED 2026-09-27 (S4b), replacing the implicit
	# kehillah_serve_as_rabbi_decision_desc this key used to resolve to on
	# its own. The decision now has two genuinely different readings --
	# filling an empty seat, or taking it off the man currently sitting in
	# it -- and the second one should not be described as "the seat stands
	# empty". Vanilla precedent for a first_valid decision desc:
	# unity_establish_futuwaa_lodges_decision (game/common/decisions/
	# 00_unity_decisions.txt).
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { employs_court_position = chief_rabbi_kehillah_position }
				desc = kehillah_serve_as_rabbi_decision_replace_desc
			}
			desc = kehillah_serve_as_rabbi_decision_desc
		}
	}

	decision_group_type = kehillah
	sort_order = 85
	ai_check_interval = 36

	# Shown to any Kehillah leader with rabbinic standing, so the option
	# is discoverable and its requirements readable, rather than hidden
	# until it happens to be takeable.
	is_shown = {
		government_has_flag = government_is_kehillah
		has_trait = kehillah_rabbi_trait
		NOT = { has_character_flag = kehillah_serving_as_own_rabbi }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no

		# NOTE, 2026-09-27 (S4b): a "the seat must be empty" refusal used
		# to sit here. It is gone on purpose -- taking up the office now
		# DEPOSES a sitting Chief Rabbi rather than being refused because
		# of him (see the effect below and kehillah_replace_appointed_
		# rabbi_effect). Two rabbis is still not a thing; the exclusion
		# now lives on the appointment side instead, in chief_rabbi_
		# kehillah_position's valid_position.

		# Recognized standing plus real learning. See
		# kehillah_serve_as_rabbi_learning_threshold's own comment
		# (common/script_values/kehillah_script_values.txt) for why 12.
		custom_description = {
			text = kehillah_serve_as_rabbi_learned_enough_tt
			kehillah_can_serve_as_own_rabbi_trigger = yes
		}
	}

	effect = {
		# ONE branch, and it comes FIRST -- before anything touches the
		# seat. Both halves of that matter:
		#
		#   * BEFORE the flag, because kehillah_serving_as_own_rabbi_
		#     trigger yields to an employed court-position holder, so the
		#     toggle does not take effect at all until the seat clears.
		#   * BEFORE any other read of the seat, because an effect block is
		#     walked in order both when it EXECUTES and when its tooltip is
		#     GENERATED. A branch placed after the revocation would see an
		#     empty seat at runtime and show the player the wrong line --
		#     the two would disagree, which is exactly the bug a single
		#     up-front branch cannot have.
		if = {
			limit = { employs_court_position = chief_rabbi_kehillah_position }

			# Bound here, ahead of the tooltip that names him, rather than
			# leaning on kehillah_replace_appointed_rabbi_effect below to
			# have bound it first -- the tooltip is rendered in this
			# position, so the scope has to exist in this position.
			court_position:chief_rabbi_kehillah_position = {
				save_scope_as = kehillah_deposed_rabbi
			}
			custom_tooltip = kehillah_serve_as_rabbi_replace_tt

			kehillah_replace_appointed_rabbi_effect = yes
		}
		else = {
			custom_tooltip = kehillah_serve_as_rabbi_tt
		}

		add_character_flag = kehillah_serving_as_own_rabbi

		# No duration: the arrangement lasts until the leader stands
		# down, a rabbi is appointed, or the leader dies -- all three of
		# which remove it explicitly (kehillah_end_serving_as_own_rabbi_
		# effect; a character modifier dies with its character).
		add_character_modifier = kehillah_serving_as_own_rabbi_modifier

		add_piety = medium_piety_gain
	}

	# AI communities take it up readily -- an AI leader who qualifies has
	# no reason to leave the seat empty, and this is the cheapest route to
	# the AI parity V22 section 6.6 asks for. The frequency is low because
	# the state it produces is permanent.
	ai_potential = {
		government_has_flag = government_is_kehillah
		has_trait = kehillah_rabbi_trait

		# S4b, 2026-09-27: the AI fills an EMPTY seat eagerly and never
		# deposes its own appointee. The replacement path the decision
		# gained is a player choice about who should be judging -- an AI
		# taking it would churn the office for no gain it can evaluate, and
		# would eat the opinion hit for nothing. (The second reason this
		# comment originally gave -- that it would strip its own rabbi's
		# semicha each time -- stopped being true the same day: S4c made
		# the trait permanent. The first reason is enough on its own.)
		NOT = { employs_court_position = chief_rabbi_kehillah_position }
	}

	ai_will_do = {
		base = 100
	}
}


kehillah_stand_down_as_rabbi_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}

	decision_group_type = kehillah
	sort_order = 84
	ai_check_interval = 120

	is_shown = {
		government_has_flag = government_is_kehillah
		has_character_flag = kehillah_serving_as_own_rabbi
	}

	is_valid = {
		is_adult = yes
	}

	effect = {
		kehillah_end_serving_as_own_rabbi_effect = yes
		custom_tooltip = kehillah_stand_down_as_rabbi_tt
	}

	# The AI never gives the office up on its own -- there is nothing to
	# gain by it, and the one situation that genuinely should end the
	# arrangement (a real rabbi being appointed) is handled by
	# kehillah_clear_redundant_rabbi_toggle_effect instead.
	ai_potential = {
		always = no
	}
}


# =====================================================================
# SEEK A CHIEF RABBI -- ROADMAP item S4(1), added 2026-09-26.
#
# Modelled on vanilla's hire_physician_decision (common/decisions/
# 90_minor_decisions.txt:47): a cooldown flag, then an event a week or two
# later offering candidates at skill tiers and gold costs. Vanilla reaches
# it through the combined recruit_court_position_decision widget; this mod
# has no such widget and one decision per office is clearer at this scale,
# so it is its own decision instead.
#
# The candidates and the whole selection event live in
# events/kehillah_rabbi_search_events.txt -- read that file's header for
# where the candidates come from, which is the part that differs from
# vanilla on purpose.
# =====================================================================

kehillah_seek_chief_rabbi_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}

	decision_group_type = kehillah
	sort_order = 86
	ai_check_interval = 24

	is_shown = {
		government_has_flag = government_is_kehillah

		# NO BUILDING REQUIREMENT, as of 2026-09-27 (S4c). This used to
		# repeat chief_rabbi_kehillah_position's Beit Midrash gate, on the
		# principle that showing a "find a rabbi" decision to a community
		# with nowhere to put one would be a promise the appointment then
		# refuses to keep. The principle stands; the gate it mirrored is
		# gone (see that valid_position's own note), so mirroring it would
		# now be the thing that lies. Every community can look for a rabbi.
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no

		custom_description = {
			text = kehillah_seek_chief_rabbi_seat_empty_tt
			NOT = { employs_court_position = chief_rabbi_kehillah_position }
		}

		# S4b, 2026-09-27: nor while the leader is the Chief Rabbi himself.
		# The search would otherwise run to completion and then hand the
		# player a candidate chief_rabbi_kehillah_position's own
		# valid_position now refuses to seat -- a dead end the player
		# cannot see into, exactly the failure this decision's header
		# already warns about for the Beit Midrash gate. Stand down first
		# (kehillah_stand_down_as_rabbi_decision) to go looking.
		custom_description = {
			text = kehillah_seek_chief_rabbi_not_serving_tt
			kehillah_has_taken_up_rabbinate_trigger = no
		}

		# Same custom_tooltip-wrapped cooldown flag shape vanilla's own
		# physician item uses (recruit_court_physician_cooldown_tt), so the
		# reason the decision is greyed out is readable rather than
		# mysterious.
		custom_description = {
			text = kehillah_seek_chief_rabbi_cooldown_tt
			NOT = { has_character_flag = kehillah_seek_chief_rabbi_cooldown }
		}
	}

	effect = {
		custom_tooltip = kehillah_seek_chief_rabbi_tt

		hidden_effect = {
			add_character_flag = {
				flag = kehillah_seek_chief_rabbi_cooldown
				years = 2
			}
			trigger_event = {
				id = kehillah_rabbi_search.0001
				days = { 7 14 }
			}
		}
	}

	# AI communities use the same search -- ROADMAP S4(1) asks for this
	# explicitly ("AI communities with an empty seat use the same
	# decision"), and it is the cheapest route to the AI parity V22
	# section 6.6 wants.
	ai_potential = {
		government_has_flag = government_is_kehillah
		kehillah_has_taken_up_rabbinate_trigger = no
	}

	ai_will_do = {
		base = 80

		# Don't bankrupt a poor community chasing the office. The event's
		# own per-option ai_chance already refuses individual candidates
		# it cannot afford; this stops the search being started at all
		# when nothing on the list could possibly be paid for.
		modifier = {
			factor = 0
			short_term_gold < kehillah_rabbi_hire_base_cost
		}
	}
}
