# =====================================================================
# FOUND A YESHIVA -- V29 section 3.1 (docs/spec/v29-yeshiva.md),
# 2026-10-11. Modelled on vanilla's Found a University (a major decision
# that grants the province its academy).
#
# A rare payoff (target: 10-25% of communities). Gated on STABILITY
# (Healthy -- order lets learning flourish; Daniel: gate on another
# pillar so the pillars support each other instead of snowballing), a
# modest Greatness (Strained), the House of Debate, and above all a
# QUALIFIED HEAD: a rabbi worth travelling to. No regional cap.
#
# This decision replaces building the third Beit Midrash tier from the
# domicile menu (that tier's can_construct now refuses, pointing here).
# =====================================================================

kehillah_found_yeshiva_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	decision_group_type = kehillah
	sort_order = 87
	ai_check_interval = 36

	is_shown = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		exists = domicile
		domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_01 }
		kehillah_has_yeshiva_trigger = no
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_description = {
			text = kehillah_found_yeshiva_house_of_debate_tt
			domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_02 }
		}
		custom_description = {
			text = kehillah_found_yeshiva_stability_tt
			kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold }
		}
		custom_description = {
			text = kehillah_found_yeshiva_greatness_tt
			kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_strained_threshold }
		}
		custom_description = {
			text = kehillah_found_yeshiva_head_tt
			kehillah_has_qualified_rosh_yeshiva_candidate_trigger = yes
		}
	}

	cost = {
		gold = kehillah_found_yeshiva_gold_cost
	}

	effect = {
		custom_tooltip = kehillah_found_yeshiva_tt
		hidden_effect = {
			kehillah_found_yeshiva_effect = yes
		}
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
	}

	ai_will_do = {
		base = 50
		modifier = {
			factor = 0
			short_term_gold < kehillah_found_yeshiva_gold_cost
		}
	}
}

# =====================================================================
# HOLD A ZMAN -- V29 section 9.1 (2026-10-11). The yeshiva's verb: a term
# of study at home. Students gather (the court's young men and visiting
# students drawn by fame), the head teaches, the ready are ordained, and
# the yeshiva's fame grows by his quality. Yearly. The AI holds them too,
# which is how AI yeshivot keep producing rabbis for the world.
# =====================================================================
kehillah_hold_zman_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	decision_group_type = kehillah
	sort_order = 88
	ai_check_interval = 12

	is_shown = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		kehillah_has_yeshiva_trigger = yes
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_description = {
			text = kehillah_hold_zman_head_tt
			kehillah_has_rosh_yeshiva_trigger = yes
		}
		custom_description = {
			text = kehillah_hold_zman_cooldown_tt
			NOT = { has_character_flag = kehillah_zman_cooldown }
			NOT = { has_character_flag = kehillah_zman_in_progress }
		}
		custom_description = {
			text = kehillah_hold_zman_not_crisis_tt
			kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_strained_threshold }
		}
	}

	cost = {
		gold = kehillah_hold_zman_gold_cost
	}

	effect = {
		custom_tooltip = kehillah_hold_zman_tt
		hidden_effect = {
			add_character_flag = { flag = kehillah_zman_cooldown years = 1 } # kehillah_hold_zman_cooldown_years
			add_character_flag = kehillah_zman_in_progress
			trigger_event = kehillah_yeshiva.0201
		}
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		kehillah_has_yeshiva_trigger = yes
	}

	ai_will_do = {
		base = 60
		modifier = {
			factor = 0
			short_term_gold < kehillah_hold_zman_gold_cost
		}
	}
}
