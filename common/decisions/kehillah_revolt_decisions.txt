# =====================================================================
# THE JEWISH REVOLT (2026-11-15, Daniel's request). Design and spike:
# docs/spec/2026-11-15-fallen-realms-revolts-and-jewish-majorities.md.
#
#   Raise the Banner     a landless leader whose county is Jewish-majority
#                        rises against the realm that hosts him, for the
#                        county or its duchy (kehillah_revolt.0001), and
#                        may proclaim the rising messianic.
#   Prepare the Rising   arms and drilled men in secret: heavier
#                        companies when the banner is raised (0-3).
# =====================================================================

kehillah_raise_the_banner_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = kehillah
	sort_order = 40
	ai_check_interval = 60

	is_shown = {
		kehillah_revolt_shown_trigger = yes
		kehillah_revolt_county_is_jewish_trigger = yes
	}

	is_valid = {
		kehillah_revolt_can_rise_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		prestige = 150
	}

	effect = {
		custom_tooltip = kehillah_raise_the_banner_tt
		trigger_event = kehillah_revolt.0001
	}

	ai_potential = {
		kehillah_revolt_shown_trigger = yes
		kehillah_revolt_county_is_jewish_trigger = yes
	}

	ai_will_do = {
		base = 0
		# A leader rises only when his host is distracted or weak, or when
		# he is a zealot, a messianic believer, or has prepared.
		modifier = {
			add = 10
			domicile.domicile_location.county.holder.top_liege ?= { is_at_war = yes }
		}
		modifier = {
			add = 10
			kehillah_revolt_preparation_level_value >= 2
		}
		modifier = {
			add = 10
			has_trait = zealous
		}
		modifier = {
			add = 10
			kehillah_revolt_can_be_messianic_trigger = yes
			has_character_flag = kehillah_messianic_mission_active
		}
		modifier = {
			factor = 0
			has_trait = craven
		}
		modifier = {
			factor = 0.25
			has_trait = content
		}
	}
}

kehillah_prepare_the_rising_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = kehillah
	sort_order = 41
	ai_check_interval = 60

	is_shown = {
		kehillah_revolt_shown_trigger = yes
		kehillah_revolt_county_is_jewish_trigger = yes
	}

	is_valid = {
		is_at_war = no
		custom_tooltip = {
			text = kehillah_prepare_the_rising_max_tt
			kehillah_revolt_preparation_level_value < kehillah_revolt_preparation_max
		}
		custom_tooltip = {
			text = kehillah_revolt_foreign_suzerain_tt
			kehillah_revolt_has_foreign_suzerain_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		gold = kehillah_revolt_preparation_gold
	}

	cooldown = { years = 2 }

	effect = {
		kehillah_revolt_prepare_effect = yes
	}

	ai_potential = {
		kehillah_revolt_shown_trigger = yes
		kehillah_revolt_county_is_jewish_trigger = yes
	}

	ai_will_do = {
		base = 0
		modifier = { add = 5 has_trait = zealous }
		modifier = { add = 5 has_trait = ambitious }
		modifier = { add = 5 kehillah_has_messianic_hope_trigger = yes }
		modifier = { factor = 0 has_trait = craven }
		modifier = { factor = 0 gold < 300 }
	}
}
