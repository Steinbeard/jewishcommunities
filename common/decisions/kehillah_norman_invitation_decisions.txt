# After the 1066 English crown leaves House Godwinson, its new holder
# has one chance to invite Jewish settlers. The invitation uses the
# ordinary Encouraged settlement law and its on_after_pass effects.
kehillah_invite_jews_to_england_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
	decision_group_type = major
	sort_order = 4
	is_shown = {
		title:k_england = {
			holder = root
			has_variable = kehillah_norman_invitation_decision_open
		}
	}
	is_valid = {
		custom_tooltip = {
			text = kehillah_invite_jews_to_england_decision_eligible_tt
			kehillah_can_hold_settlement_law_trigger = yes
		}
	}
	effect = {
		title:k_england = { remove_variable = kehillah_norman_invitation_decision_open }
		custom_tooltip = kehillah_invite_jews_to_england_decision_effect_tt
		hidden_effect = { trigger_event = { id = kehillah_norman_conquest.0003 days = 1 } }
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 30
		modifier = { add = 470 dynasty = dynasty:752 }
		modifier = { add = 70 dynasty = dynasty:499 }
	}
}
