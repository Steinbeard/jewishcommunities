# Seek Semicha -- V21 slice 3, added 2026-10-09 (Shabbat heartbeat 1,
# ROADMAP item H4).
#
# Source of truth: docs/spec/v21-rabbi-ordination-paths.md section 3.4.
# "It is a capstone decision/event, not an annual random notification."
#
# Everything this decision tests lives in named triggers
# (common/scripted_triggers/kehillah_ordination_triggers.txt), per
# section 6 slice 1's instruction not to paste the conditions into each
# of the decision, scheme, office and event. What is NOT in those
# triggers is the population gate below, which is copied from
# kehillah_learn_torah_decision deliberately: the two must cover exactly
# the same people, or a character could study somewhere he can never
# ordain.
#
# NOT YET LIVE-TESTED. See ROADMAP H4.
# =====================================================================

kehillah_seek_semicha_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 92
	ai_check_interval = 60

	# Same three populations as kehillah_learn_torah_decision's is_shown,
	# for the reason in this file's header. is_shown rather than is_valid
	# for the trait check: an ordained rabbi should not see a greyed-out
	# "Seek Semicha" for the rest of his life.
	is_shown = {
		NOT = { has_trait = kehillah_rabbi_trait }
		is_jewish_character_trigger = yes
		OR = {
			government_has_flag = government_is_kehillah
			is_ruler = yes
			has_government = landless_adventurer_government
		}
	}

	# CHARACTER = root is safe in a decision's own is_valid -- its root
	# never shifts, the same note kehillah_learn_torah_decision carries.
	#
	# The two halves are split into separate custom_tooltips on purpose:
	# "you have not finished the curriculum" and "there is no study hall
	# that can ordain you" are completely different problems for the
	# player, and a single combined failure line would not say which.
	is_valid = {
		# kehillah_rabbi_ordination_ready_trigger is candidate + curriculum
		# (minus the trait, which is_shown already handles). Calling it as
		# one tooltip would label a Learning-5 adult's failure "you have
		# not completed a course of study", which is not his problem -- so
		# its two halves get their own lines, same reasoning as the
		# curriculum/institution split below.
		custom_tooltip = {
			text = kehillah_seek_semicha_candidate_tt
			kehillah_rabbi_ordination_candidate_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_seek_semicha_curriculum_tt
			kehillah_rabbi_ordination_curriculum_met_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_seek_semicha_institution_tt
			kehillah_semicha_institution_available_trigger = { CHARACTER = root }
		}
	}

	# Section 3.4: "Piety may affect flavour, a modest Piety cost/reward".
	# Modest, and not a second hard gate -- the reward on ordination
	# (kehillah_grant_semicha_effect) is larger than this.
	cost = {
		piety = minor_piety_value
	}

	effect = {
		kehillah_semicha_resolve_institution_effect = { CHARACTER = root }
		trigger_event = kehillah_semicha.0001
	}

	# Same reasoning as kehillah_learn_torah_decision's own ai_potential
	# note: a personal study/ordination action is not something this mod
	# wants an AI evaluation loop running for every landed Jewish ruler
	# and landless adventurer on the map, only for AI-played Kehillah
	# leaders, the bounded population that was already tuned for.
	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 100
	}
}
