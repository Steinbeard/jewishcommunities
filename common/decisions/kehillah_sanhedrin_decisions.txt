# V47 phase B (docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md): restore
# the Sanhedrin in the Land of Israel. Replaces V46's landless restoration.
#
# Shown to every Rabbinic community rabbi while no Sanhedrin sits, so the
# goal is visible. Valid only for a rabbi of the highest learning and
# devotion who leads a community in the Land, whose liege holds that county
# directly and not as his capital. The decision is the petition: the liege
# grants the county, and the Sanhedrin is restored on it, all at once. A
# player liege answers by event; an AI liege by the odds shown.
kehillah_restore_sanhedrin_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 90
	ai_check_interval = 60

	is_shown = {
		government_has_flag = government_is_kehillah
		is_ruler = yes
		faith = faith:rabbinic_faith
		has_trait = kehillah_rabbi_trait
		NOT = { exists = global_var:kehillah_sanhedrin_seat }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_the_land_tt
			kehillah_community_in_the_land_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_grantable_county_tt
			kehillah_liege_can_grant_county_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_mastery_tt
			kehillah_masters_all_rabbinic_tracks_trigger = yes
		}
		piety_level >= 5
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_members_tt
			kehillah_sanhedrin_has_founding_members_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_no_head_tt
			NOT = { exists = faith.religious_head }
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_no_recent_refusal_tt
			NOT = { kehillah_sanhedrin_petition_refused_trigger = yes }
		}
	}

	cost = {
		piety = kehillah_sanhedrin_piety_cost
		prestige = kehillah_sanhedrin_prestige_cost
	}

	effect = {
		save_scope_as = kehillah_sanhedrin_petitioner
		if = {
			limit = { suzerain = { is_ai = no } }
			custom_tooltip = kehillah_sanhedrin_petition_player_liege_tt
			suzerain = { trigger_event = kehillah_synod.0006 }
		}
		else = {
			random_list = {
				20 = {
					desc = kehillah_sanhedrin_petition_granted
					# A Jewish liege, or one with land to spare, gives
					# more readily; a valuable county is harder to win.
					modifier = { add = 40 suzerain = { religion = religion:judaism_religion } }
					modifier = { add = 15 suzerain = { domain_size >= 8 } }
					modifier = { add = -10 suzerain = { domain_size <= 3 } }
					modifier = { add = -15 primary_title.title_capital_county = { development_level >= 20 } }
					modifier = { add = 10 primary_title.title_capital_county = { development_level <= 8 } }
					opinion_modifier = {
						who = suzerain
						opinion_target = root
						multiplier = 0.5
					}
					min = 1
					custom_tooltip = kehillah_restore_sanhedrin_effect_tt
					hidden_effect = { kehillah_sanhedrin_restore_effect = yes }
				}
				60 = {
					desc = kehillah_sanhedrin_petition_refused
					min = 1
					set_variable = {
						name = kehillah_sanhedrin_petition_refused
						years = kehillah_sanhedrin_petition_cooldown_years
					}
					custom_tooltip = kehillah_sanhedrin_petition_refused_tt
				}
			}
		}
	}

	ai_potential = {
		kehillah_masters_all_rabbinic_tracks_trigger = yes
		kehillah_community_in_the_land_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}
