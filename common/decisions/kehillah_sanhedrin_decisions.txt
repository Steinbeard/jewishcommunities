# V46 (docs/spec/v46-the-sanhedrin.md): restore the Sanhedrin as Rabbinic
# Judaism's head of faith. Shown to every Rabbinic Kehillah leader while no
# Sanhedrin sits, so the goal is visible from the start, and deliberately
# very hard to meet: mastery of all three rabbinic tracks and the highest
# level of devotion.
kehillah_restore_sanhedrin_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 90
	ai_check_interval = 60

	# Community leaders, and (Daniel, 2026-10-19) a landed Jewish ruler such
	# as a king of Israel who is himself a great rabbi.
	is_shown = {
		is_ruler = yes
		faith = faith:rabbinic_faith
		OR = {
			government_has_flag = government_is_kehillah
			is_landed = yes
		}
		NOT = { exists = title:d_kehillah_sanhedrin.holder }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_mastery_tt
			kehillah_masters_all_rabbinic_tracks_trigger = yes
		}
		piety_level >= 5
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_members_tt
			kehillah_sanhedrin_has_founding_members_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_no_head_tt
			NOT = { exists = faith.religious_head }
		}
	}

	cost = {
		piety = kehillah_sanhedrin_piety_cost
		prestige = kehillah_sanhedrin_prestige_cost
	}

	effect = {
		custom_tooltip = kehillah_restore_sanhedrin_effect_tt
		hidden_effect = { kehillah_restore_sanhedrin_effect = yes }
		every_player = {
			limit = { NOT = { this = root } }
			send_interface_message = {
				type = msg_kehillah_community_news
				title = kehillah_news_sanhedrin_restored_t
				desc = kehillah_news_sanhedrin_restored_d
				left_icon = root
			}
		}
	}

	ai_potential = {
		kehillah_masters_all_rabbinic_tracks_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}
