# V47 phase B (docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md): the
# Sanhedrin in the Land of Israel. Replaces V46's landless restoration.
#
# Two decisions (Daniel, 2026-10-19):
#   - Restore the Sanhedrin: the visible goal, shown to every Rabbinic
#     ruler while no Sanhedrin sits. Valid for one who holds a county in the
#     Land and has the highest learning and devotion. Kings, lords and
#     community leaders alike.
#   - Petition for Land in Eretz Yisrael: one way to get that county. A
#     community leader in the Land asks his liege for the county his
#     community lives in. Rare by design: Kehillot may rule land, but only
#     in extraordinary circumstances.

kehillah_restore_sanhedrin_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 90
	ai_check_interval = 60

	is_shown = {
		is_ruler = yes
		faith = faith:rabbinic_faith
		NOT = { exists = global_var:kehillah_sanhedrin_seat }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_county_in_the_land_tt
			kehillah_holds_county_in_the_land_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_mastery_tt
			kehillah_masters_all_rabbinic_tracks_trigger = yes
		}
		piety_level >= 5
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_members_tt
			kehillah_sanhedrin_has_founding_members_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_no_head_tt
			NOT = { exists = faith.religious_head }
		}
	}

	cost = {
		piety = kehillah_sanhedrin_piety_cost
		prestige = kehillah_sanhedrin_prestige_cost
	}

	effect = {
		custom_tooltip = kehillah_restore_sanhedrin_effect_tt
		hidden_effect = { kehillah_sanhedrin_restore_effect = yes }
	}

	ai_potential = {
		kehillah_masters_all_rabbinic_tracks_trigger = yes
		kehillah_holds_county_in_the_land_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}

kehillah_petition_for_land_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	sort_order = 89
	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		is_landed = no
		government_has_flag = government_is_kehillah
		kehillah_community_in_the_land_trigger = yes
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_land_requires_grantable_county_tt
			kehillah_liege_can_grant_county_trigger = yes
		}
		prestige_level >= 3
		custom_tooltip = {
			text = kehillah_land_requires_no_recent_refusal_tt
			NOT = { kehillah_land_petition_refused_trigger = yes }
		}
	}

	cost = {
		prestige = kehillah_land_petition_prestige_cost
	}

	effect = {
		save_scope_as = kehillah_land_petitioner
		if = {
			limit = { suzerain = { is_ai = no } }
			custom_tooltip = kehillah_land_petition_player_liege_tt
			suzerain = { trigger_event = kehillah_synod.0006 }
		}
		else = {
			random_list = {
				10 = {
					desc = kehillah_land_petition_granted
					# A Jewish liege, or one with land to spare, gives more
					# readily; a valuable county is harder to win.
					modifier = { add = 40 suzerain = { religion = religion:judaism_religion } }
					modifier = { add = 15 suzerain = { domain_size >= 8 } }
					modifier = { add = -10 suzerain = { domain_size <= 3 } }
					modifier = { add = -15 primary_title.title_capital_county = { development_level >= 20 } }
					modifier = { add = 10 primary_title.title_capital_county = { development_level <= 8 } }
					opinion_modifier = {
						who = suzerain
						opinion_target = root
						multiplier = 0.5
					}
					min = 1
					custom_tooltip = kehillah_land_petition_granted_tt
					hidden_effect = { kehillah_grant_community_county_effect = yes }
				}
				90 = {
					desc = kehillah_land_petition_refused
					set_variable = {
						name = kehillah_land_petition_refused
						years = kehillah_land_petition_cooldown_years
					}
					custom_tooltip = kehillah_land_petition_refused_tt
				}
			}
		}
	}

	ai_potential = {
		kehillah_masters_all_rabbinic_tracks_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}
