		kehillah_crisis_leader_free_trigger = yes

	effect = {
		if = { limit = { primary_title = { has_variable = kehillah_crisis_bargain_won } } custom_tooltip = kehillah_crisis_contribute_bargain_tt }
		kehillah_crisis_contribute_effect = yes
	}
	ai_will_do = { base = 5 modifier = { add = 30 gold >= major_gold_value } }
}
