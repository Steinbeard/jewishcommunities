# SPIKE (2026-11-01, disposable): the situation's decisions. Shown only to the leader whose title
# holds a live spike in Agitation, so an outside observer sees none. Free and repeatable on purpose.

kehillah_spike_post_watch_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds"
	}
	decision_group_type = kehillah
	ai_check_interval = 0

	is_shown = {
		primary_title.var:kehillah_spike_situation ?= { situation_current_phase = kehillah_spike_agitation }
	}

	effect = {
		primary_title.var:kehillah_spike_situation ?= {
			trigger_situation_catalyst = {
				catalyst = kehillah_spike_watch_catalyst
				character = root
			}
			kehillah_spike_log_effect = { WHEN = decision_watch }
		}
	}

	ai_will_do = { base = 0 }
}
