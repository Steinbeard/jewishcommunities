# =====================================================================
# Learn Torah. ADDED 2026-09-15 at user request ("implement the first
# version of" docs/spec/spike-book-inventory.md), and confirmed by the
# user as the general umbrella all three kehillah_rabbi_trait tracks
# (parshanut/talmudics/hashkafa) study through -- not three separate
# decisions, and not a standalone competitor to the earlier "Study a
# Work of Hashkafa" idea, which this supersedes before it was ever built.
#
# kehillah_learn_torah_decision REBUILT 2026-09-19 AS A SCHEME-STARTER,
# at user request: "implement Learn Torah as a continuous scheme... it
# seems like that's the parallel to the base game designs." It no longer
# resolves anything itself -- it starts kehillah_study_torah (common/
# schemes/scheme_types/kehillah_study_torah_scheme.txt), a long-running
# personal scheme that keeps reading through the community's own library
# indefinitely (one work every base_progress_goal, currently 3 years)
# rather than a single one-shot Learning check. kehillah_learn_torah.0001
# (events/kehillah_learn_torah_events.txt), the old flat resolution event,
# is RETIRED -- see that file's own retirement note for the full account,
# and events/kehillah_study_torah_events.txt for its replacement.
#
# kehillah_acquire_torah_work_decision is UNCHANGED by this pass -- still
# a flat, one-shot decision, commissioning a copy of a work the community
# does NOT yet own, gated on the Sofer's Workshop's own scriptorium tier
# (kehillah_has_scriptorium). "Commission a copy" is a single transaction
# with a clear beginning and end; there is no equivalent "continuous"
# framing for it the way there is for reading, so it had no reason to
# follow Learn Torah into scheme form.
#
# THE LIBRARY ITSELF is a `variable_list` on each community's title
# (kehillah_library_works), not a real artifact -- see common/scripted_
# triggers/kehillah_scripted_triggers.txt's own header (the six
# kehillah_owns_/missing_<track>_work_trigger entries both decisions
# below read) for the full account, including the twenty-three real
# works (UPDATED 2026-09-19: grew from twelve when the Talmud split into
# Sedarim and Hashkafa's date-gated backfill were added) and why every
# community starts owning a random 1-3 of the Parshanut+Talmudics ones.
# =====================================================================

kehillah_learn_torah_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 94
	ai_check_interval = 60

	# BROADENED 2026-09-19, same-day follow-up ("make this available to
	# non-Kehilla ruler Jews too... Jewish unlanded adventurers... Jewish
	# rulers can use the library of the kehilla in their capital"). Was
	# government_has_flag = government_is_kehillah alone; kehillah_study_
	# torah_has_accessible_library_trigger's own header (common/scripted_
	# triggers/kehillah_scripted_triggers.txt) has the three populations
	# now covered and why a landless adventurer's own CURRENT LOCATION,
	# not a domicile, is the right anchor for them specifically (domiciles
	# are confirmed absent from vanilla for that government type).
	# is_ruler = yes here is intentionally not narrowed to independent
	# rulers -- matches kehillah_bet_din_eligible_judge_trigger's own bare
	# is_ruler = yes precedent for "a Jewish ruler" generally.
	is_shown = {
		is_jewish_character_trigger = yes
		OR = {
			government_has_flag = government_is_kehillah
			is_ruler = yes
			has_government = landless_adventurer_government
		}
	}

	# has_trait = kehillah_rabbi_trait, NOT "OR learning >= 8" -- matches
	# kehillah_write_book_decision's own gate exactly (every one of its
	# three rabbi-genre options requires the trait outright, events/
	# kehillah_book_events.txt's kehillah_book.0001.a/b/c). Deliberate,
	# not an oversight: add_trait_xp only ever gets called on a track this
	# character already holds the trait for -- there is no established
	# precedent anywhere in this codebase for accruing trait_xp on a trait
	# not yet held, so this doesn't invent one. (Bet Din judging used to be
	# the other add_trait_xp caller enforcing this same rule; REMOVED
	# 2026-09-19 -- see kehillah_rabbi_trait's own header, common/traits/
	# kehillah_traits.txt.)
	#
	# The OR-of-owns clause is REPLACED, same 2026-09-19 follow-up, by
	# kehillah_study_torah_has_accessible_library_trigger -- "does the
	# community's own library hold anything at all to begin reading" is
	# now "does WHICHEVER community's library applies to this specific
	# character hold anything," the same three-population check the
	# scheme's own valid trigger uses. The custom_tooltip clause is
	# unchanged: reusing the real vanilla can_start_scheme guard pattern
	# (confirmed against tgp_china_study_confucian_classics_decision,
	# dlc_decisions/tgp/tgp_china_decisions.txt) so a rabbi already
	# mid-scheme sees an explanatory line here rather than the decision
	# just silently greying out with no visible reason.
	is_valid = {
		is_adult = yes
		is_imprisoned = no
		has_trait = kehillah_rabbi_trait
		is_jewish_character_trigger = yes
		# CHARACTER = root is safe here (unlike inside the scheme's own
		# valid block) -- a decision's own is_valid root never shifts, it
		# already is the character throughout.
		kehillah_study_torah_has_accessible_library_trigger = { CHARACTER = root }
		custom_tooltip = {
			text = kehillah_learn_torah_decision_already_studying_tt
			NOT = { any_scheme = { type = kehillah_study_torah } }
		}
	}

	is_valid_showing_failures_only = {
		can_start_scheme = {
			type = kehillah_study_torah
			target_character = root
		}
	}

	cost = {
		piety = kehillah_learn_torah_piety_cost
	}

	# NO cooldown, DELIBERATELY, unlike before this became a scheme-
	# starter: the old flat decision recurred every 3 years because it WAS
	# the recurring action; now the SCHEME recurs (one work every
	# base_progress_goal, common/schemes/scheme_types/kehillah_study_
	# torah_scheme.txt), and this decision only ever fires once per course
	# of study, to start it. can_start_scheme above already prevents a
	# second one stacking on top of a running scheme, which is the only
	# thing a cooldown here would otherwise have been guarding against.
	effect = {
		start_scheme = {
			type = kehillah_study_torah
			target_character = root
		}

		# Resolves WHICH library applies to THIS character (kehillah_
		# study_torah_resolve_library_effect's own header, common/
		# scripted_effects/kehillah_library_effects.txt, has the three
		# populations), then reveals which work the scheme begins on via
		# custom_tooltip, the same "You take up X" lines used since
		# 2026-09-15 -- see that effect's own header for why the pick is a
		# live query, not a snapshot. is_valid above already guarantees a
		# match exists by the time this runs. CHARACTER = root, same
		# reasoning as is_valid above -- a decision's own effect root
		# never shifts either.
		kehillah_study_torah_resolve_library_effect = { CHARACTER = root }
		kehillah_study_torah_pick_next_work_effect = { LIBRARY = scope:kehillah_study_torah_library }
		trigger_event = kehillah_study_torah.0002
	}

	# UNCHANGED, DELIBERATELY: ai_potential still reads government_has_flag
	# = government_is_kehillah alone, narrower than is_shown/is_valid above.
	# The AI-eligible population for THIS decision stays exactly what it
	# was -- see kehillah_learn_torah_journey.txt's own "AI NEVER HOSTS
	# THIS, ON PURPOSE" header for the identical reasoning already
	# established in this mod: a personal study action is not something
	# this mod wants an AI evaluation loop running for every landed Jewish
	# ruler and landless adventurer on the map, only for AI-played Kehillah
	# leaders, the bounded population this was already tuned for.
	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 50
	}
}

kehillah_acquire_torah_work_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds"
	}

	decision_group_type = kehillah
	sort_order = 93
	ai_check_interval = 60

	# Gated on the Sofer's Workshop's own scriptorium tier -- see this
	# file's own header for why this specific building/tier, not a new
	# one invented for the purpose.
	is_shown = {
		government_has_flag = government_is_kehillah
		domicile ?= { has_domicile_parameter = kehillah_has_scriptorium }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		OR = {
			kehillah_missing_parshanut_work_trigger = { LIBRARY = primary_title }
			kehillah_missing_talmudics_work_trigger = { LIBRARY = primary_title }
			kehillah_missing_hashkafa_work_trigger = { LIBRARY = primary_title }
		}
	}

	cost = {
		gold = medium_gold_value
	}

	cooldown = { years = 2 }

	effect = {
		trigger_event = kehillah_acquire_torah_work.0001
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
		domicile ?= { has_domicile_parameter = kehillah_has_scriptorium }
	}

	ai_will_do = {
		base = 40

		modifier = {
			factor = 0
			gold < 100
		}
	}
}
