# =====================================================================
# Learn Torah. ADDED 2026-09-15 at user request ("implement the first
# version of" docs/spec/spike-book-inventory.md), and confirmed by the
# user as the general umbrella all three kehillah_rabbi_trait tracks
# (parshanut/talmudics/hashkafa) study through -- not three separate
# decisions, and not a standalone competitor to the earlier "Study a
# Work of Hashkafa" idea, which this supersedes before it was ever built.
#
# kehillah_learn_torah_decision REBUILT 2026-09-19 AS A SCHEME-STARTER,
# at user request: "implement Learn Torah as a continuous scheme... it
# seems like that's the parallel to the base game designs." It no longer
# resolves anything itself -- it starts kehillah_study_torah (common/
# schemes/scheme_types/kehillah_study_torah_scheme.txt), a long-running
# personal scheme that keeps reading through the community's own library
# indefinitely (one work every base_progress_goal, currently 3 years)
# rather than a single one-shot Learning check. kehillah_learn_torah.0001
# (events/kehillah_learn_torah_events.txt), the old flat resolution event,
# is RETIRED -- see that file's own retirement note for the full account,
# and events/kehillah_study_torah_events.txt for its replacement.
#
# kehillah_acquire_torah_work_decision is UNCHANGED by this pass -- still
# a flat, one-shot decision, commissioning a copy of a work the community
# does NOT yet own, gated on the Sofer's Workshop's own scriptorium tier
# (kehillah_has_scriptorium). "Commission a copy" is a single transaction
# with a clear beginning and end; there is no equivalent "continuous"
# framing for it the way there is for reading, so it had no reason to
# follow Learn Torah into scheme form.
#
# THE LIBRARY ITSELF is a `variable_list` on each community's title
# (kehillah_library_works), not a real artifact -- see common/scripted_
# triggers/kehillah_scripted_triggers.txt's own header (the six
# kehillah_owns_/missing_<track>_work_trigger entries both decisions
# below read) for the full account, including the twelve real works and
# why every community starts owning exactly two of them.
# =====================================================================

kehillah_learn_torah_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 94
	ai_check_interval = 60

	is_shown = {
		government_has_flag = government_is_kehillah
	}

	# has_trait = kehillah_rabbi_trait, NOT "OR learning >= 8" -- matches
	# kehillah_write_book_decision's own gate exactly (every one of its
	# three rabbi-genre options requires the trait outright, events/
	# kehillah_book_events.txt's kehillah_book.0001.a/b/c). Deliberate,
	# not an oversight: add_trait_xp only ever gets called on a track this
	# character already holds the trait for, both here and everywhere
	# else in this mod (kehillah_rabbi_grant_case_xp_effect's own guard,
	# common/scripted_effects/kehillah_bet_din_scripted_effects.txt) --
	# there is no established precedent anywhere in this codebase for
	# accruing trait_xp on a trait not yet held, so this doesn't invent
	# one.
	# The OR-of-owns clause is unchanged from before this decision started
	# a scheme -- still "does the community's own library hold anything at
	# all to begin reading." The custom_tooltip clause is new: reusing the
	# real vanilla can_start_scheme guard pattern (confirmed against
	# tgp_china_study_confucian_classics_decision, dlc_decisions/tgp/
	# tgp_china_decisions.txt) so a rabbi already mid-scheme sees an
	# explanatory line here rather than the decision just silently greying
	# out with no visible reason.
	is_valid = {
		is_adult = yes
		is_imprisoned = no
		has_trait = kehillah_rabbi_trait
		OR = {
			kehillah_owns_parshanut_work_trigger = { LIBRARY = primary_title }
			kehillah_owns_talmudics_work_trigger = { LIBRARY = primary_title }
			kehillah_owns_hashkafa_work_trigger = { LIBRARY = primary_title }
		}
		custom_tooltip = {
			text = kehillah_learn_torah_decision_already_studying_tt
			NOT = { any_scheme = { type = kehillah_study_torah } }
		}
	}

	is_valid_showing_failures_only = {
		can_start_scheme = {
			type = kehillah_study_torah
			target_character = root
		}
	}

	cost = {
		piety = kehillah_learn_torah_piety_cost
	}

	# NO cooldown, DELIBERATELY, unlike before this became a scheme-
	# starter: the old flat decision recurred every 3 years because it WAS
	# the recurring action; now the SCHEME recurs (one work every
	# base_progress_goal, common/schemes/scheme_types/kehillah_study_
	# torah_scheme.txt), and this decision only ever fires once per course
	# of study, to start it. can_start_scheme above already prevents a
	# second one stacking on top of a running scheme, which is the only
	# thing a cooldown here would otherwise have been guarding against.
	effect = {
		start_scheme = {
			type = kehillah_study_torah
			target_character = root
		}

		# Reveals which work the scheme begins on, via custom_tooltip, the
		# same "You take up X" lines used since 2026-09-15 -- see this
		# effect's own header (common/scripted_effects/kehillah_library_
		# effects.txt) for why the pick is a live query, not a snapshot.
		kehillah_study_torah_pick_next_work_effect = { LIBRARY = primary_title }
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 50
	}
}

kehillah_acquire_torah_work_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds"
	}

	decision_group_type = kehillah
	sort_order = 93
	ai_check_interval = 60

	# Gated on the Sofer's Workshop's own scriptorium tier -- see this
	# file's own header for why this specific building/tier, not a new
	# one invented for the purpose.
	is_shown = {
		government_has_flag = government_is_kehillah
		domicile ?= { has_domicile_parameter = kehillah_has_scriptorium }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		OR = {
			kehillah_missing_parshanut_work_trigger = { LIBRARY = primary_title }
			kehillah_missing_talmudics_work_trigger = { LIBRARY = primary_title }
			kehillah_missing_hashkafa_work_trigger = { LIBRARY = primary_title }
		}
	}

	cost = {
		gold = medium_gold_value
	}

	cooldown = { years = 2 }

	effect = {
		trigger_event = kehillah_acquire_torah_work.0001
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
		domicile ?= { has_domicile_parameter = kehillah_has_scriptorium }
	}

	ai_will_do = {
		base = 40

		modifier = {
			factor = 0
			gold < 100
		}
	}
}
