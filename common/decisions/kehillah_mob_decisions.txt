# dead and baptized, more plundered, and some may be found.

	}
	desc = kehillah_mob_scatter_decision_desc
	selection_tooltip = kehillah_mob_scatter_decision_tooltip

	is_shown = {
		government_has_flag = government_is_kehillah
		primary_title ?= { has_variable = kehillah_mob_threat }

		custom_tooltip = {
			text = kehillah_mob_already_scattered_tt

		primary_title = { set_variable = { name = kehillah_mob_scattered value = yes } }

			random = {
				chance = 30
				primary_title = { set_variable = { name = kehillah_mob_scattered_found value = yes } }
			}
		}
	}

	ai_check_interval = 0
	ai_will_do = { base = 0 }
