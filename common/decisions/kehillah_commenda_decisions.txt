kehillah_invest_in_voyage_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds" }
	decision_group_type = kehillah
	sort_order = 85
	ai_check_interval = 36

	is_shown = {
		government_has_flag = government_is_kehillah
		exists = primary_title
		domicile ?= { has_domicile_building_or_higher = kehillah_funduq_01 }
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = kehillah_commenda_at_sea_tt
			primary_title = { NOT = { has_variable = kehillah_commenda_stake } }
		}
	}

	cost = {
		gold = kehillah_commenda_stake_value
	}

	effect = {
		custom_tooltip = kehillah_commenda_invest_tt
		primary_title = {
			set_variable = { name = kehillah_commenda_stake value = root.kehillah_commenda_stake_value }
		}
		trigger_event = { id = kehillah_commenda.0001 days = { 730 1095 } }
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 20
		modifier = { add = 20 stewardship >= 12 }
		modifier = { factor = 0 gold < kehillah_commenda_ai_gold_floor }
	}
}
