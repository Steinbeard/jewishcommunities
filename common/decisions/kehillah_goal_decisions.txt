# =====================================================================
# Set the Community's Goal. ROADMAP S6, added 2026-09-28.
#
# The decision does nothing but open kehillah_goals.0001, the same shape
# kehillah_seek_chief_rabbi_decision uses: the four goals are a CHOICE
# between four things with different costs and different odds, and an
# event with four options is how this mod already presents that, with room
# for the flavour text a bare decision list has nowhere to put.
#
# ONE GOAL AT A TIME is enforced here rather than in the event, so the
# decision reads as unavailable with a reason instead of opening a window
# that then refuses. The reason is a custom_tooltip and not a raw
# has_variable check for the reason the responsa compile decision records
# in its own header: the engine has no localisation for has_variable
# inside a primary_title block and prints its own internal complaint to
# the player in broken-loc magenta.
#
# NO COOLDOWN beyond "finish or fail the one you have". A community that
# completes a goal in three years should be free to set the next one in
# its fourth, and a community that missed one has already paid for it in
# Stability (see kehillah_fail_community_goal_effect's header).
# =====================================================================

kehillah_set_community_goal_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 93
	ai_check_interval = 60

	is_shown = {
		government_has_flag = government_is_kehillah
		exists = primary_title
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no

		custom_tooltip = {
			text = kehillah_goal_none_active_tt
			NOT = { kehillah_has_active_goal_trigger = yes }
		}
	}

	effect = {
		trigger_event = { id = kehillah_goals.0001 days = 0 }
	}

	ai_potential = {
		government_has_flag = government_is_kehillah
	}

	ai_will_do = {
		base = 60
	}
}
