# Write a Book.
#
# ADDED 2026-09-09; FLESHED OUT the same day at user request. Originally
# three separate decisions (one per kehillah_rabbi_trait track), each
# resolving in one instant effect. Replaced with ONE decision that opens
# a real event chain (events/kehillah_book_events.txt) -- genre choice,
# inspiration, a courtier discussion, a revision choice, then the book
# is finished and copies go out to nearby communities. See kehillah_
# has_eligible_book_genre_trigger's own header (common/scripted_
# triggers/kehillah_scripted_triggers.txt) for exactly who is offered
# this and why, and common/script_values/kehillah_script_values.txt's
# "WRITE A BOOK" section for the full tally/reward design.
#
# Was distinct from kehillah_compose_commentary_decision (kehillah_
# decisions.txt) on purpose: that one was gated on the COMMUNITY's
# Greatness pillar and produced one generic commentary in one step; this
# is gated on ONE PERSON's own accumulated expertise (or, for poetry,
# their own lifestyle_poet trait) and is a real multi-step chain about
# writing a specific book on a specific subject.
#
# UPDATE, 2026-09-15: kehillah_compose_commentary_decision is RETIRED --
# see its own retirement note in kehillah_decisions.txt. This is now the
# ONLY "compose a scholarly work" decision this mod offers. The same
# pass added a Talmudics/Halakha-specific inspiration option to kehillah_
# book.0002 (events/kehillah_book_events.txt) that draws on a character's
# own memory of a good Bet Din ruling, if they have one -- see that
# event's own header and kehillah_has_good_bet_din_memory_trigger
# (common/scripted_triggers/kehillah_scripted_triggers.txt).
# =====================================================================

kehillah_write_book_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_tale.dds"
	}

	decision_group_type = kehillah
	sort_order = 96
	ai_check_interval = 60

	is_shown = {
		kehillah_sefer_can_write_trigger = yes
		is_jewish_character_trigger = yes
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_tooltip = { text = kehillah_writing_no_active_project_tt NOT = { any_scheme = { type = kehillah_writing } } }
		# 2026-11-23: a spark replaces the old cooldown and project gates;
		# writing uses the spark up (docs/spec/2026-11-23-writing-sparks.md).
		custom_tooltip = { text = kehillah_spark_needed_tt kehillah_has_any_spark_trigger = yes }
		piety >= kehillah_rabbi_book_piety_cost
	}

	effect = {
		trigger_event = kehillah_spark.0001
	}

	ai_potential = {
		kehillah_sefer_can_write_trigger = yes
		kehillah_has_any_spark_trigger = yes
		NOT = { any_scheme = { type = kehillah_writing } }
	}

	# 2026-11-24 (Daniel: "AI won't always write an insight and will sometimes
	# wait to upgrade all the way to a historical work"): an Insight is rarely
	# written; a spark that could still become a historical landmark is held
	# longer; a Great Work is written; a historical author writes his own.
	ai_will_do = {
		base = 4
		modifier = { add = 20 OR = { var:kehillah_spark_parshanut_tier ?= 2 var:kehillah_spark_talmudics_tier ?= 2 var:kehillah_spark_hashkafa_tier ?= 2 } }
		modifier = { add = 80 OR = { var:kehillah_spark_parshanut_tier ?= 3 var:kehillah_spark_talmudics_tier ?= 3 var:kehillah_spark_hashkafa_tier ?= 3 } }
		modifier = {
			factor = 0.25
			NOT = { OR = { var:kehillah_spark_parshanut_tier ?= 3 var:kehillah_spark_talmudics_tier ?= 3 var:kehillah_spark_hashkafa_tier ?= 3 } }
			OR = {
				kehillah_spark_parshanut_potential_value = 2
				kehillah_spark_talmudics_potential_value = 2
				kehillah_spark_hashkafa_potential_value = 2
			}
		}
		modifier = {
			add = 100
			kehillah_spark_is_historical_author_trigger = yes
			OR = { var:kehillah_spark_parshanut_tier ?= 3 var:kehillah_spark_talmudics_tier ?= 3 var:kehillah_spark_hashkafa_tier ?= 3 }
		}
	}
}
