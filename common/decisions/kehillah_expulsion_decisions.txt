# Reopens the edict (kehillah_expulsion.0001) while a plea or a ransom
# has not been tried yet.
kehillah_answer_edict_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}

	decision_group_type = kehillah
	sort_order = 5

	is_shown = {
		government_has_flag = government_is_kehillah
		kehillah_expulsion_pending_trigger = yes
		NOT = { primary_title = { has_variable = kehillah_expulsion_defiant } }
	}

	is_valid = {
		custom_tooltip = {
			text = kehillah_answer_edict_exhausted_tt
			primary_title = {
				OR = {
					NOT = { has_variable = kehillah_expulsion_pleaded }
					NOT = { has_variable = kehillah_expulsion_ransomed }
				}
			}
		}
	}

	effect = {
		trigger_event = kehillah_expulsion.0001
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

# Leave now, in good order, before the edict is enforced.

	}

	decision_group_type = kehillah
	sort_order = 6

	is_shown = {
		government_has_flag = government_is_kehillah
		kehillah_expulsion_pending_trigger = yes
	}

	is_valid = {
		is_imprisoned = no
	}

	effect = {
		custom_tooltip = kehillah_leave_in_good_order_decision_effect_tt
		hidden_effect = { trigger_event = kehillah_expulsion.0020 }
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
