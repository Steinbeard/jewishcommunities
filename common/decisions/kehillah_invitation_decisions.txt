# =====================================================================
# INVITE JEWISH SETTLERS -- the Nagid spec, section 6 (docs/spec/
# 2026-10-30-the-nagid.md), 2026-10-30. Daniel: the top rung of the
# Status of the Jews (the Encouraged law, shown as "Jews Favored")
# unlocks this decision instead of inviting settlers by itself.
#
# It opens the same 20-year invitation V34 section 7 built
# (kehillah_invitation_issue_effect: kehillah_invitation_open on the
# realm title, the letter or news to Kehillah players in range). While
# it is open the decision is unavailable, so the invitation's own 20
# years are the cooldown. The settlers' and exiles' destination weights
# favour the inviting ruler's capital county (the "invited city").
#
# An AI ruler who RAISES the law to Favored still invites at once, as
# before (kehillah_settlement_policy_changed_effect), which is the open-
# door review "taking the decision in one step". This decision is how a
# player invites, and how an AI ruler renews an invitation that has run
# out while the realm stays Favored.
# =====================================================================

kehillah_invite_jewish_settlers_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
	decision_group_type = realm
	desc = kehillah_invite_jewish_settlers_decision_desc
	selection_tooltip = kehillah_invite_jewish_settlers_decision_tooltip
	ai_check_interval = 60

	is_shown = {
		kehillah_can_hold_settlement_law_trigger = yes
		primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	is_valid = {
		custom_tooltip = {
			text = kehillah_invite_jewish_settlers_already_open_tt
			NOT = { primary_title = { has_variable = kehillah_invitation_open } }
		}
	}

	effect = {
		save_scope_as = kehillah_policy_host
		primary_title = { save_scope_as = kehillah_policy_realm }
		custom_tooltip = kehillah_invite_jewish_settlers_decision_effect_tt
		hidden_effect = { kehillah_invitation_issue_effect = yes }
	}

	ai_potential = { always = yes }

	ai_will_do = {
		base = 10
		# "Towns and markets to fill" (V34 section 7's motive).
		modifier = { add = 20 stewardship >= 12 }
		modifier = { add = 10 has_trait = ambitious }
		modifier = { add = 10 has_trait = greedy }
		# A capital with no community of its own: the realm has no seat
		# of its Jews yet (Nagid spec section 4).
		modifier = {
			add = 30
			exists = capital_county
			capital_county = { NOT = { kehillah_inspector_county_occupied_trigger = yes } }
		}
		modifier = { add = 10 kehillah_outlook_protector_trigger = yes }
	}
}
