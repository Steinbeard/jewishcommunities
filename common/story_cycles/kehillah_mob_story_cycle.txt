# =====================================================================
# V39 -- the Mob's story (common/scripted_effects/kehillah_mob_effects.
# txt). Owned by the mob's leader; every 4 days the host marches one
# county, or its siege counts down. Ends with the leader's death (the
# mob's end kills him, kehillah_mob_end_effect).
# =====================================================================

story_kehillah_mob = {
	on_setup = {
		debug_log = "story_kehillah_mob: set up"
	}

	on_end = {
		debug_log = "story_kehillah_mob: ended"
	}

	# The preacher is dead (vanished at the mob's end, or killed in the
	# field): clear the community's state if no end did, and close.
	on_owner_death = {
		story_owner = {
			var:kehillah_mob_target_title ?= {
				if = {
					limit = { has_variable = kehillah_mob_threat }
					kehillah_mob_clear_title_effect = yes
				}
				holder ?= {
					if = {
						limit = { has_variable = kehillah_mob_ref }
						remove_variable = kehillah_mob_ref
					}
					if = {
						limit = { has_variable = kehillah_mob_band }
						remove_variable = kehillah_mob_band
					}
				}
			}
		}
		end_story = yes
	}

	effect_group = {
		days = 4
		triggered_effect = {
			trigger = {
				story_owner = { is_alive = yes }
			}
			effect = {
				story_owner = { kehillah_mob_tick_effect = yes }
			}
		}
	}
}
