# =====================================================================
# THE JEWISH QUARTER DISTRICT (2026-11-21). docs/spec/2026-11-20-kehillah-quarter-building-tracks.md section 16.
#
# A regular district in the host's holding, placed and removed only by
# script (kehillah_quarter_district_sync_effect / _sweep_effect): no ruler
# can build or replace it. It shows the community on the host's map and
# pays him for it; the community's own buildings stay in its domicile.
#
# Regular, not special: a province holds one special district (Jerusalem's
# Dome of the Rock, Rome's basilica), so a special Jewish quarter could
# never stand where Jews most needed one. When the holding has no free
# district slot, the community gets none and its tooltips say "no room".
# Authored starts get one extra slot (kehillah_quarter_room_modifier).
# =====================================================================

kehillah_jewish_quarter_district = {
	type = regular
	primary_building = kehillah_jewish_quarter_01
	color = district_religious

	#MENA
	background = {
		trigger = {
			OR = {
				culture ?= { has_building_gfx = iranian_building_gfx }
				culture ?= { has_building_gfx = mena_building_gfx }
				culture ?= { has_building_gfx = african_building_gfx }
			}
		}
		reference = "gfx/interface/illustrations/event_scenes/gdp_artisans_street_mena_crowd.dds"
	}

	#Western Fallback
	background = {
		reference = "gfx/interface/illustrations/event_scenes/gdp_artisans_street_western_crowd.dds"
	}

	can_be_replaced = no

	# The community's yeshiva (2026-11-22), placed by script.
	buildings = {
		kehillah_quarter_yeshiva_01
	}

	# Only script places it.
	can_construct_potential = {
		always = no
	}
}
