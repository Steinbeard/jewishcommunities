# =====================================================================
# By-name override of CK3 1.21's temple holding district
# (common/districts/holding_districts.txt), copied from 1.21.0 "Florin".
#
# 1.21 offers a holding only the buildings its district lists. The one
# change: kehillah_temple_beit_midrash_01 is added to the end of
# the buildings list, so a Jewish temple holder can build the Beit
# Midrash (common/buildings/kehillah_temple_beit_midrash_buildings.txt).
# Its own can_construct_potential keeps it to Jewish holders of a
# church_holding, so temple_citadel_holding_district is not overridden.
# Rebase these on each patch (scripts/vanilla_rebase).
# =====================================================================

temple_holding_district = {
	primary_building = temple_01
	
	#Vaticano
	background = {
		trigger = {
			this = title:b_vaticano.title_province
		}
		reference = "gfx/interface/illustrations/event_scenes/pam_papal_throne_room.dds"
	}

	#Islamic
	background = {
		trigger = {
			faith ?= { religion = religion:islam_religion }
		}
		reference = "gfx/interface/illustrations/event_scenes/mosque.dds"
	}

	#Christian
	background = {
		trigger = {
			faith ?= { religion = religion:christianity_religion }
		}
		reference = "gfx/interface/illustrations/event_scenes/church.dds"
	}

	#East Asian
	background = {
		trigger = {
			OR = {
				faith ?= { religion = religion:shintoism_religion }
				AND = {
					faith ?= {
						religion = {
							OR = {
								is_in_family = rf_eastern
								is_in_family = rf_sinitic
							}
						}
					}
					culture ?= { has_graphical_east_asia_culture_group_trigger = yes }
				}
			}
		}
		reference = "gfx/interface/illustrations/event_scenes/tgp_temple_asia.dds"
	}

	#Eastern/Sinitic
	background = {
		trigger = {
			faith ?= {
				religion = {
					OR = {
						is_in_family = rf_eastern
						is_in_family = rf_sinitic
					}
				}
			}
		}
		reference = "gfx/interface/illustrations/event_scenes/ep2_holysite_indian.dds"
	}

	#Fallback
	background = {
		reference = "gfx/interface/illustrations/event_scenes/temple.dds"
	}

	type = holding
	buildings = {
		oath_faithful_building_01
		temple_hostels_1
		temple_reliquary_1
		temple_religious_fairs_1
		oath_warriors_building_01
		oath_poor_building_01
		temple_hospices_1
		megalith_1
		kehillah_temple_beit_midrash_01 # Jewish Communities
	}
}
