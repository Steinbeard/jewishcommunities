# The yeshiva's public face on the map, as 1.21 districts.
#
# CK3 1.21 moved great and special buildings into districts: a building
# no longer carries a slot type, its district does, and scripts place
# the district (set_great_district / add_special_district) rather than
# the building. These wrap the three buildings in
# common/buildings/kehillah_yeshiva_buildings.txt one for one. Placement
# lives in kehillah_place_yeshiva_map_building_effect
# (common/scripted_effects/kehillah_yeshiva_effects.txt).
#
# The Great Yeshiva is a separate great district, not an upgrade level of
# the hall: kehillah_raise_great_yeshiva_effect removes the hall's
# district and sets this one, as the 1.20 code removed one great
# building and added the other.
# =====================================================================

kehillah_yeshiva_hall_district = {
	primary_building = kehillah_yeshiva_hall

	background = {
		trigger = {
			OR = {
				culture ?= { has_graphical_mena_culture_group_trigger = yes }
				culture ?= { has_graphical_iranian_culture_group_trigger = yes }
				culture ?= { has_graphical_african_culture_group_trigger = yes }
			}
		}
		reference = "gfx/interface/illustrations/event_scenes/bp2_university_mena.dds"
	}
	background = {
		reference = "gfx/interface/illustrations/event_scenes/bp2_university.dds"
	}

	type = great_district
}
