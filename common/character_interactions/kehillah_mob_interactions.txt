# =====================================================================
# V39 -- the Mob (docs/spec/v39-pogroms-and-the-popular-crusade.md),
# 2026-10-13. Daniel: the host should be something "player rulers could
# attack, but AI usually shouldn't". A player ruler joins the host's war
# on the community's side (add_defender) and may fight it. The AI never
# does this on its own; an AI lord joins only when the community moves
# him to raise his men (kehillah_mob.0002).
# =====================================================================

kehillah_defend_community_interaction = {
	icon = icon_combat
	category = interaction_category_diplomacy
	common_interaction = no
	use_diplomatic_range = no

	desc = kehillah_defend_community_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			is_landed = yes
			NOT = { government_has_flag = government_is_kehillah }
		}
		scope:recipient = {
			government_has_flag = government_is_kehillah
			primary_title ?= { has_variable = kehillah_mob_threat }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_available_adult = yes
			custom_tooltip = {
				text = kehillah_defend_community_already_tt
				NOT = { is_at_war_with = scope:recipient.primary_title.var:kehillah_mob_threat }
			}
		}
	}

	on_accept = {
		scope:recipient.primary_title.var:kehillah_mob_threat = {
			save_scope_as = kehillah_mob_leader
			random_character_war = {
				limit = { is_attacker = scope:kehillah_mob_leader }
				add_defender = scope:actor
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = grateful_opinion
				opinion = 40
			}
			if = {
				limit = { is_ai = no }
				send_interface_message = {
					type = msg_kehillah_community_news
					title = kehillah_defend_community_msg_t
					desc = kehillah_defend_community_msg_d
					left_icon = scope:actor
				}
			}
		}
		custom_tooltip = kehillah_defend_community_tt
	}

	ai_frequency = 0
}

# =====================================================================
# V39 slice 1b (Daniel, 2026-10-13): the crisis answers as interactions
# with the people involved, each tried once per host and target, each
# with a cost and odds shown before sending (a random_list in on_accept).
# Shown only while a host marches on the actor's community
# (kehillah_mob_threat on its title).
# =====================================================================

# Treat with the Preacher: silver for the host to turn aside.
kehillah_mob_treat_interaction = {
	icon = icon_gold
	category = interaction_category_diplomacy
	common_interaction = no
	use_diplomatic_range = no
	desc = kehillah_mob_treat_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = {
			government_has_flag = government_is_kehillah
			primary_title ?= { has_variable = kehillah_mob_threat }
		}
		scope:recipient = {
			has_character_flag = kehillah_mob_leader
			this = scope:actor.primary_title.var:kehillah_mob_threat
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_available_adult = yes
			custom_tooltip = {
				text = kehillah_mob_already_tried_tt
				primary_title = { NOT = { has_variable = kehillah_mob_tried_treat } }
			}
		}
	}

	send_option = {
		is_valid = { scope:actor = { gold >= kehillah_mob_treat_more_cost_value } }
		flag = kehillah_mob_more_silver
		localization = KEHILLAH_MOB_MORE_SILVER
	}
	send_options_exclusive = no

	cost = {
		gold = {
			value = scope:actor.kehillah_mob_treat_cost_value
		}
	}

	on_accept = {
		scope:actor = {
			primary_title = { set_variable = { name = kehillah_mob_tried_treat value = yes } }
			if = {
				limit = { scope:kehillah_mob_more_silver = yes }
				remove_short_term_gold = {
					value = kehillah_mob_treat_cost_value
					multiply = 0.5
				}
			}
		}
		random_list = {
			0 = {
				modifier = { add = scope:actor.kehillah_mob_treat_chance_value }
				desc = kehillah_mob_bribe_taken
				custom_tooltip = kehillah_mob_bribe_taken_tt
				scope:actor = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = kehillah_mob_bribe_taken
						left_icon = scope:recipient
					}
				}
				hidden_effect = {
					scope:recipient = { kehillah_mob_bought_off_effect = yes }
				}
			}
			0 = {
				modifier = { add = scope:actor.kehillah_mob_treat_fail_value }
				desc = kehillah_mob_bribe_kept
				custom_tooltip = kehillah_mob_bribe_kept_tt
				scope:actor = {
					send_interface_toast = {
						type = event_toast_effect_bad
						title = kehillah_mob_bribe_kept
						left_icon = scope:recipient
					}
				}
			}
		}
	}

	ai_frequency = 0
}

# Ask for Refuge in the Garrison: the county's lord, or the overlord.
kehillah_mob_ask_refuge_interaction = {
	icon = icon_liege
	category = interaction_category_diplomacy
	common_interaction = no
	use_diplomatic_range = no
	desc = kehillah_mob_ask_refuge_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = {
			government_has_flag = government_is_kehillah
			primary_title ?= { has_variable = kehillah_mob_threat }
		}
		scope:recipient = {
			is_landed = yes
			NOT = { this = scope:actor }
			OR = {
				this = scope:actor.domicile.domicile_location.county.holder
				this = scope:actor.domicile.domicile_location.county.holder.top_liege
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_available_adult = yes
			custom_tooltip = {
				text = kehillah_mob_already_asked_tt
				primary_title = {
					NOT = { is_target_in_variable_list = { name = kehillah_mob_asked_refuge target = scope:recipient } }
				}
			}
			custom_tooltip = {
				text = kehillah_mob_already_sheltered_tt
				primary_title = { NOT = { has_variable = kehillah_mob_refuge } }
			}
		}
	}

	send_option = {
		is_valid = { scope:actor = { gold >= medium_gold_value } }
		flag = kehillah_mob_gift
		localization = KEHILLAH_MOB_GIFT
	}
	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			primary_title = { add_to_variable_list = { name = kehillah_mob_asked_refuge target = scope:recipient } }
			if = {
				limit = { scope:kehillah_mob_gift = yes }
				remove_short_term_gold = medium_gold_value
			}
		}
		random_list = {
			0 = {
				modifier = { add = scope:actor.kehillah_mob_ask_refuge_chance_value }
				desc = kehillah_mob_refuge_granted
				custom_tooltip = kehillah_mob_refuge_granted_tt
				scope:actor = {
					primary_title = { set_variable = { name = kehillah_mob_refuge value = scope:recipient } }
					send_interface_toast = {
						type = event_toast_effect_good
						title = kehillah_mob_refuge_granted
						left_icon = scope:recipient
					}
				}
			}
			0 = {
				modifier = { add = scope:actor.kehillah_mob_ask_refuge_fail_value }
				desc = kehillah_mob_refuge_refused
				custom_tooltip = kehillah_mob_refuge_refused_tt
				scope:recipient = {
					hidden_effect = { kehillah_shift_outlook_effect = { AMOUNT = -3 } }
				}
				scope:actor = {
					send_interface_toast = {
						type = event_toast_effect_bad
						title = kehillah_mob_refuge_refused
						left_icon = scope:recipient
					}
				}
			}
		}
	}

	ai_frequency = 0
}
