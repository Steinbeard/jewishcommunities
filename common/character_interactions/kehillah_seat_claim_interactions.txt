# A Jewish adventurer with a recorded claim can press any Kehillah
# seat. Assent from its current leader and host is required. The office
# of Exilarch is a separate decision with a separate lineage condition.
kehillah_press_community_claim_interaction = {
	icon = heir
	category = interaction_category_diplomacy
	common_interaction = no
	use_diplomatic_range = no
	desc = kehillah_press_community_claim_interaction_desc
	auto_accept = yes
	cost = {
		gold = { value = 100 }
		prestige = { value = 150 }
	}
	is_shown = {
		exists = scope:recipient.primary_title
		scope:recipient.primary_title = { is_kehillah_title_trigger = yes }
		scope:actor = {
			is_jewish_character_trigger = yes
			has_government = landless_adventurer_government
			is_target_in_variable_list = {
				name = kehillah_community_claims
				target = scope:recipient.primary_title
			}
		}
	}
	is_valid = {
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
			is_at_war = no
			location.county = scope:recipient.primary_title.title_capital_county
		}
		scope:recipient = {
			is_imprisoned = no
			opinion = { target = scope:actor value >= -10 }
		}
		scope:recipient.primary_title.title_capital_county.holder.top_liege = {
			opinion = { target = scope:actor value >= -25 }
		}
	}
	on_auto_accept = {
		scope:recipient.primary_title = { save_scope_as = kehillah_claim_seat }
		scope:actor = { kehillah_press_community_claim_effect = yes }
	}
}
