kehillah_nagid_change_post_interaction = {
	icon = council
	category = interaction_category_diplomacy
	common_interaction = no
	desc = kehillah_nagid_change_post_interaction_desc
	auto_accept = yes
	cooldown = { years = 1 }

	is_shown = {
		scope:actor = {
			is_ai = no
			kehillah_ruler_has_own_nagid_trigger = yes
			primary_title.var:kehillah_nagid_holder ?= scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no }
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = kehillah_nagid_change_post_effect_tt
			trigger_event = kehillah_nagid.0003
		}
	}

	ai_will_do = { base = 0 }
}

kehillah_nagid_dismiss_interaction = {
	icon = revoke_title
	category = interaction_category_hostile
	common_interaction = no
	desc = kehillah_nagid_dismiss_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			kehillah_ruler_has_own_nagid_trigger = yes
			primary_title.var:kehillah_nagid_holder ?= scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no }
	}

	on_accept = {
		scope:recipient = { save_scope_as = kehillah_nagid_seat_leader }
		scope:actor = {
			custom_tooltip = kehillah_nagid_dismiss_effect_tt
			kehillah_nagid_dismiss_effect = yes
		}
	}

	ai_will_do = { base = 0 }
}
