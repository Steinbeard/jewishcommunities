# A discussion changes opinion and support for an already known teaching;
# it cannot convert its recipient or alter a rite's core tenets.
kehillah_reasoned_debate_interaction = {
    icon = religious
    category = interaction_category_diplomacy
    common_interaction = no
    desc = kehillah_reasoned_debate_interaction_desc
    auto_accept = yes
    cooldown_against_recipient = { years = 3 }
    ai_frequency = 24
    ai_targets = { ai_recipients = courtiers ai_recipients = vassals }
    is_shown = {
        scope:actor = { kehillah_has_reasoned_theology_trigger = yes }
        NOT = { scope:actor = scope:recipient }
        scope:recipient.faith = scope:actor.faith
    }
    is_valid = {
        scope:actor = {
            is_adult = yes is_imprisoned = no is_incapable = no
            learning >= 8
            NOT = { has_character_flag = kehillah_reasoned_debate_cooldown }
            NOT = { has_character_flag = kehillah_reasoned_debate_pending }
        }
        scope:recipient = { is_adult = yes is_imprisoned = no is_incapable = no learning >= 8 }
    }
    on_auto_accept = {
        scope:recipient = { save_scope_as = kehillah_debate_partner }
        scope:actor = {
            add_character_flag = kehillah_reasoned_debate_pending
            add_character_flag = { flag = kehillah_reasoned_debate_cooldown years = 1 }
            trigger_event = kehillah_faith_practice.0001
        }
    }
    ai_will_do = { base = 15 }
}
