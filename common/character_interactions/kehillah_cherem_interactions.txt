# =====================================================================
# PRONOUNCE A CHEREM (2026-11-11). docs/spec/2026-11-11-reviews-and-
# cherem.md section 3.3. The head of a rite or faith, or the Nasi of the
# Sanhedrin, places the full ban on one of the faith who has given grounds
# (Nezifah or worse on the cherem track). This is the one way, besides a
# synod, to ban a community leader, whose own score stops at 99.
#
# The AI does not use this interaction: an AI head of rite bans those deep
# in Niddui from the yearly pass (kehillah_cherem_yearly_effect).
# =====================================================================

kehillah_pronounce_cherem_interaction = {
	icon = excommunication
	category = interaction_category_hostile
	common_interaction = yes
	use_diplomatic_range = no

	desc = kehillah_pronounce_cherem_interaction_desc
	notification_text = kehillah_pronounce_cherem_interaction_notification

	auto_accept = yes

	cost = {
		piety = {
			value = kehillah_cherem_pronounce_piety_cost
			desc = kehillah_pronounce_cherem_cost_desc
		}
	}

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { kehillah_cherem_authority_trigger = yes }
		scope:recipient = {
			is_jewish_character_trigger = yes
			faith = scope:actor.faith
			NOT = { kehillah_cherem_in_full_trigger = yes }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
		}
		scope:recipient = {
			custom_description = {
				text = kehillah_pronounce_cherem_grounds_tt
				object = scope:recipient
				kehillah_cherem_has_grounds_trigger = yes
			}
		}
	}

	on_auto_accept = {
		scope:recipient = {
			custom_tooltip = kehillah_pronounce_cherem_effect_tt
			hidden_effect = {
				save_scope_value_as = { name = kehillah_cherem_pronounced_by value = flag:head }
				kehillah_cherem_pronounce_effect = yes
			}
			add_opinion = {
				target = scope:actor
				modifier = insulted_opinion
				opinion = -40
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
