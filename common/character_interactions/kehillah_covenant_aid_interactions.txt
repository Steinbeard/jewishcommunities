# Cross-rite and cross-culture mutual aid. No government, fixed culture, or
# Israelite-heritage gate; shared heritage is compared between the actual cultures.
kehillah_covenant_aid_interaction = {
    icon = icon_gold
    category = interaction_category_friendly
    common_interaction = no
    desc = kehillah_covenant_aid_interaction_desc
    auto_accept = yes
    cooldown_against_recipient = { years = 5 }
    ai_frequency = 24
    ai_targets = { ai_recipients = courtiers ai_recipients = vassals max = 10 }
    ai_targets = { ai_recipients = scripted_relations ai_recipients = neighboring_rulers max = 5 }
    is_shown = {
        always = no # Retired: use the existing crisis relief system.
        scope:actor = { kehillah_has_covenant_community_trigger = yes }
        NOT = { scope:actor = scope:recipient }
        OR = {
            scope:recipient.faith = scope:actor.faith
            scope:actor.culture = { has_same_culture_heritage = scope:recipient.culture }
        }
    }
    is_valid = {
        scope:actor = {
            is_adult = yes
            is_imprisoned = no
            is_incapable = no
            NOT = { has_character_flag = kehillah_covenant_service_rewarded }
        }
        scope:recipient = {
            is_alive = yes
            gold < scope:actor.kehillah_covenant_aid_gold_value
        }
    }
    cost = { gold = { value = scope:actor.kehillah_covenant_aid_gold_value } }
    on_auto_accept = {
        scope:recipient = { save_scope_as = kehillah_covenant_aid_recipient }
        scope:actor = {
            save_scope_value_as = { name = kehillah_covenant_gift value = kehillah_covenant_aid_gold_value }
            scope:kehillah_covenant_aid_recipient = { add_gold = scope:kehillah_covenant_gift }
            kehillah_covenant_service_reward_effect = yes
            stress_impact = { generous = minor_stress_impact_loss greedy = minor_stress_impact_gain }
        }
    }
    ai_will_do = {
        base = 15
        modifier = { factor = 2 scope:actor = { has_trait = generous } }
        modifier = { factor = 2 scope:actor = { has_trait = compassionate } }
        modifier = { factor = 0.1 scope:actor = { has_trait = greedy } }
    }
}
