# =====================================================================
# SEND A SHE'ELAH (2026-11-22). docs/spec/2026-11-22-sheelot.md.
#
# One interaction for every letter of question (Daniel, 2026-11-22: "combine
# the help with studies into the responsa interaction"). Two kinds, chosen
# with exclusive send options:
#   - A case from our court: a Kehillah leader writes to a posek of the same
#     faith (section 9). Era-scaled gift, three-year cooldown.
#   - A passage of the work you are reading: anyone studying (the Study Torah
#     scheme) writes to a rabbi of the same faith who has mastered it
#     (section 10). Small gift, one-year cooldown.
# Each kind keeps its own requirements, shown for the option selected. The
# AI sends court cases through kehillah_sheelah_ai_pulse_effect (no
# ai_frequency here) and never asks for help with its reading.
# =====================================================================

kehillah_send_sheelah_interaction = {
	category = interaction_category_religion
	icon = religious
	common_interaction = no
	desc = kehillah_send_sheelah_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:recipient = {
			NOT = { this = scope:actor }
			exists = faith
			faith = scope:actor.faith
		}
		OR = {
			# A case: leader to leader.
			AND = {
				scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
				scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
			}
			# A passage: a student to a rabbi.
			AND = {
				scope:actor = { kehillah_study_help_can_ask_trigger = yes }
				scope:recipient = { kehillah_study_help_teacher_trigger = { STUDENT = scope:actor } }
			}
		}
	}

	options_heading = kehillah_sheelah_options_heading
	send_options_exclusive = yes

	send_option = {
		flag = kehillah_sheelah_case
		localization = kehillah_sheelah_option_case
		current_description = kehillah_sheelah_option_case_desc
		is_shown = {
			scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
			scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
		}
		starts_enabled = {
			scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
			scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
		}
	}

	send_option = {
		flag = kehillah_sheelah_study
		localization = kehillah_sheelah_option_study
		current_description = kehillah_sheelah_option_study_desc
		is_shown = {
			scope:actor = { kehillah_study_help_can_ask_trigger = yes }
			scope:recipient = { kehillah_study_help_teacher_trigger = { STUDENT = scope:actor } }
		}
		# Selected whenever the case option is not offered (live 2026-11-22:
		# NOR left a leader asking a courtier rabbi with nothing selected).
		starts_enabled = {
			NOT = {
				AND = {
					scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
					scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = kehillah_sheelah_choose_kind_tt
			OR = {
				scope:kehillah_sheelah_case = yes
				scope:kehillah_sheelah_study = yes
			}
		}
		# A case from our court.
		trigger_if = {
			limit = {
				# Both must lead communities, not only the flag: a non-leader
				# rabbi got the case checks (live 2026-11-22).
				scope:kehillah_sheelah_case = yes
				scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
				scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
			}
			scope:actor = {
				custom_tooltip = {
					text = kehillah_sheelah_cooldown_tt
					NOT = { has_variable = kehillah_sheelah_cooldown }
				}
				gold >= kehillah_sheelah_gift_value
			}
			scope:recipient = {
				custom_tooltip = {
					text = kehillah_sheelah_needs_rabbi_tt
					kehillah_has_community_rabbi_trigger = yes
				}
				custom_tooltip = {
					text = kehillah_sheelah_needs_greatness_tt
					kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_healthy_threshold }
				}
				custom_tooltip = {
					text = kehillah_sheelah_busy_tt
					NOT = { has_variable = kehillah_sheelah_pending_from }
				}
			}
		}
		# Otherwise a passage of the work you are reading.
		trigger_else = {
			scope:actor = {
				custom_tooltip = {
					text = kehillah_study_help_cooldown_tt
					NOT = { has_variable = kehillah_study_help_cooldown }
				}
				gold >= kehillah_study_help_gift_value
			}
			scope:recipient = {
				custom_tooltip = {
					text = kehillah_study_help_knows_work_tt
					kehillah_study_help_knows_work_trigger = { STUDENT = scope:actor }
				}
				custom_tooltip = {
					text = kehillah_study_help_busy_tt
					NOT = { has_variable = kehillah_study_help_pending_from }
				}
			}
		}
	}

	on_accept = {
		if = {
			limit = {
				# Both must lead communities, not only the flag: a non-leader
				# rabbi got the case checks (live 2026-11-22).
				scope:kehillah_sheelah_case = yes
				scope:actor = { kehillah_sheelah_is_leader_trigger = yes }
				scope:recipient = { kehillah_sheelah_is_leader_trigger = yes }
			}
			scope:recipient = { save_scope_as = kehillah_sheelah_posek }
			scope:actor = {
				custom_tooltip = kehillah_sheelah_send_tt
				remove_short_term_gold = kehillah_sheelah_gift_value
				hidden_effect = { kehillah_sheelah_send_effect = yes }
			}
		}
		else_if = {
			limit = {
				scope:actor = { kehillah_study_help_can_ask_trigger = yes }
				scope:recipient = { kehillah_study_help_teacher_trigger = { STUDENT = scope:actor } }
			}
			scope:recipient = { save_scope_as = kehillah_study_teacher }
			scope:actor = {
				custom_tooltip = kehillah_study_help_send_tt
				remove_short_term_gold = kehillah_study_help_gift_value
				hidden_effect = { kehillah_study_help_send_effect = yes }
			}
		}
	}
}
