kehillah_messianic_appeal_interaction = {
    icon = religious
    category = interaction_category_diplomacy
    desc = kehillah_messianic_appeal_interaction_desc
    auto_accept = yes
    cooldown_against_recipient = { years = 3 }
    ai_frequency = 24
    ai_targets = { ai_recipients = courtiers ai_recipients = neighboring_rulers max = 3 }
    is_shown = {
        scope:actor = { kehillah_has_messianic_hope_trigger = yes has_character_flag = kehillah_messianic_mission_active }
        NOT = { scope:actor = scope:recipient }
        scope:recipient.faith = scope:actor.faith
    }
    is_valid = {
        scope:actor = { is_adult = yes is_imprisoned = no is_incapable = no NOT = { has_character_flag = kehillah_messianic_appeal_cooldown } }
        scope:recipient = { is_adult = yes is_imprisoned = no NOT = { has_variable = kehillah_messianic_patron } }
    }
    on_auto_accept = {
        scope:actor = { save_scope_as = kehillah_messianic_claimant add_character_flag = { flag = kehillah_messianic_appeal_cooldown months = 3 } }
        scope:recipient = { trigger_event = kehillah_revival.0002 }
    }
    ai_will_do = { base = 20 }
}
