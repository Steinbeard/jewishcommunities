kehillah_poi_kever_rachel = {
    build_province_list = {
        if = {
            limit = { exists = character:kehillah_rachel_imenu }
            province:5965 = { add_to_list = provinces }
        }
    }
    on_visit = {
        visiting_poi_effect = yes
        send_interface_toast = {
            title = kehillah_poi_kever_rachel.visit
            left_icon = root
            right_icon = character:kehillah_rachel_imenu
            custom_tooltip = kehillah_poi_kever_rachel.visit_desc
            if = {
                limit = { faith.religion = religion:judaism_religion }
                add_piety = point_of_interest_miniscule_piety_value
            }
            else = { add_learning_lifestyle_xp = travel_miniscule_lifestyle_xp }
        }
    }
}
