# =====================================================================
# The Kehillah Quarter -- the community's domicile.
#
# Source of truth: docs/spec/v1-kehillah-community.md section 3c.
#
# WHY A NEW DOMICILE TYPE RATHER THAN REUSING VANILLA "estate":
#
#  1. Vanilla's estate domicile gates itself with
#       allowed_for_character = { any_held_title = { is_noble_family_title = yes } }
#     Noble family titles are an administrative-government construct.
#     A Kehillah leader has no noble family title, so the vanilla estate
#     would simply never be granted.
#
#  2. Every vanilla estate building declares
#     allowed_domicile_types = { estate }. Defining our own type means
#     the Kehillah Quarter shows ONLY Kehillah buildings -- which is
#     precisely what spec section 3c asks for, since several vanilla
#     estate buildings grant Intrigue and scheme bonuses that the design
#     deliberately rejects. Reusing the estate type would have required
#     overriding vanilla buildings to hide them.
#
# Textures and backgrounds are vanilla base-game estate art, referenced
# by path. That is intentional for a first iteration: it looks like a
# walled urban quarter, which is close enough to read correctly, and it
# costs no art budget. Spec section 1 explicitly puts art out of scope
# for v1. Bespoke art is a later pass.
# =====================================================================

kehillah_quarter = {
	# Gated on the government flag rather than on holding a specific
	# title, so that later phases can add more Kehillah titles without
	# touching this file.
	allowed_for_character = {
		government_has_flag = government_is_kehillah
	}

	rename_window = primary_title

	map_pin_anchor = left

	# The quarter does not wander. A Jewish community is defined by
	# being embedded in a specific town -- this is the deliberate break
	# from the adventurer camp pattern called out in spec section 6.
	# Moving it at all is a Phase 4 concern (expulsion, migration), so
	# for v1 it is simply pinned.
	can_move_manually = no
	provisions = no
	travel = no
