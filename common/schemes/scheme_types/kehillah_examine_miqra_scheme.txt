# A self-directed Scripture course for followers of Ba'alei ha-Miqra.
# It uses the established self-targeted study-scheme shape, but has no
# rabbinic library, semicha, or Rabbi trait XP dependency.
kehillah_examine_miqra = {
	skill = learning
	desc = kehillah_examine_miqra_desc_general
	success_desc = "KEHILLAH_EXAMINE_MIQRA_SCHEME_SUCCESS_DESC"
	icon = icon_scheme_study_scriptures
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = personal
	target_type = character
	is_secret = no
	is_basic = yes
	hide_target_name = yes
	freeze_scheme_when_traveling = yes

	speed_per_skill_point = -2.5
	spymaster_speed_per_skill_point = 0
	uses_resistance = no
	base_progress_goal = 540
	base_maximum_success = 95
	minimum_success = 5

	allow = {
		age >= 16
		is_imprisoned = no
		NOT = { has_trait = incapable }
		is_jewish_character_trigger = yes
		rite = { rite_has_tenet = tenet_baalei_hamiqra }
	}

	valid = {
		scope:target = scope:owner
		is_imprisoned = no
		NOT = { has_trait = incapable }
		is_jewish_character_trigger = yes
		rite = { rite_has_tenet = tenet_baalei_hamiqra }
	}

	odds_prediction = {
		add = base_odds_prediction_target_is_char_value
		min = 0
	}

	base_success_chance = {
		base = 50
		modifier = {
			add = scope:owner.learning
			desc = kehillah_examine_miqra_learning_modifier
		}
	}

	on_phase_completed = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = { name = kehillah_miqra_success value = yes }
		}
		scope:owner = {
			if = {
				limit = { exists = scope:kehillah_miqra_success }
				trigger_event = kehillah_examine_miqra.0001
			}
			else = { trigger_event = kehillah_examine_miqra.0002 }
		}
		reset_scheme_progress = yes
	}

	on_invalidated = { }
}
