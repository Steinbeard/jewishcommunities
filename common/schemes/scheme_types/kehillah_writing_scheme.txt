# A self-targeted personal scheme, following kehillah_study_torah's proven
# study_confucian_classics pattern. Three phases form one manuscript; an
# Insight spark (2026-11-23) finishes after two.
kehillah_writing = {
	skill = learning
	desc = kehillah_writing_scheme_desc
	success_desc = kehillah_writing_scheme_success_desc
	icon = icon_scheme_learn_language
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = personal
	target_type = character
	is_secret = no
	is_basic = yes
	hide_target_name = yes
	uses_resistance = no
	base_progress_goal = 320
	speed_per_skill_point = -4
	spymaster_speed_per_skill_point = 0
	base_maximum_success = 100
	minimum_success = 0
	allow = {
		scope:target = scope:owner
		is_jewish_character_trigger = yes
		kehillah_sefer_can_write_trigger = yes
		has_variable = kehillah_writing_form
		is_imprisoned = no
		NOT = { has_trait = incapable }
	}
	valid = {
		scope:target = scope:owner
		is_jewish_character_trigger = yes
		has_variable = kehillah_writing_form
		is_imprisoned = no
		NOT = { has_trait = incapable }
	}
	odds_prediction = { add = base_odds_prediction_target_is_char_value min = 0 }
	base_success_chance = {
		base = 0
		modifier = {
			add = scope:owner.kehillah_writing_completion_chance_value
			desc = kehillah_writing_scheme_chance_desc
		}
		# 2026-11-23 sparks: a plain Insight is easier to finish, a Great Work harder.
		modifier = {
			add = 10
			desc = kehillah_writing_scheme_spark_insight_desc
			scope:owner = { var:kehillah_writing_spark_tier ?= 1 }
		}
		modifier = {
			add = -10
			desc = kehillah_writing_scheme_spark_great_desc
			scope:owner = { var:kehillah_writing_spark_tier ?= 3 }
		}
		modifier = {
			add = -15
			desc = kehillah_writing_scheme_language_desc
			scope:owner = { NOT = { kehillah_writing_knows_book_language_trigger = yes } }
		}
		# 2026-11-24: commentaries mastered in this project's discipline.
		modifier = {
			add = scope:owner.kehillah_spark_writing_craft_chance_value
			desc = KEHILLAH_SPARK_CRAFT_FACTOR
			scope:owner = { kehillah_spark_writing_craft_chance_value > 0 }
		}
	}
	# 2026-11-24: commentary craft speeds the writing too.
	on_monthly = {
		if = {
			limit = { scheme_owner = { kehillah_spark_writing_craft_speed_value > 0 } }
			add_scheme_progress = scheme_owner.kehillah_spark_writing_craft_speed_value
		}
	}
	on_phase_completed = {
		save_scope_as = kehillah_writing_scheme
		reset_scheme_progress = yes
		scheme_owner = {
			change_variable = { name = kehillah_writing_phase add = 1 }
			if = {
				limit = { var:kehillah_writing_phase = 1 }
				trigger_event = kehillah_writing.0100
			}
			# An Insight spark skips the objections phase.
			else_if = {
				limit = { var:kehillah_writing_phase = 2 NOT = { var:kehillah_writing_spark_tier ?= 1 } }
				trigger_event = kehillah_writing.0200
			}
			else_if = {
				limit = { has_variable = kehillah_writing_extra_time var:kehillah_writing_phase = 3 }
				trigger_event = kehillah_writing.0250
			}
			else = {
				remove_variable = kehillah_writing_success
				random = {
					chance = scope:kehillah_writing_scheme.scheme_success_chance
					set_variable = { name = kehillah_writing_success value = yes }
				}
				trigger_event = kehillah_writing.0300
				scope:kehillah_writing_scheme = { end_scheme = yes }
			}
		}
	}
	# 2026-11-24: marks this scheme for the scheme panel (gui/kehillah_scheme_widget.gui).
	on_start = { set_variable = kehillah_is_writing_scheme }
	on_invalidated = { }
}
