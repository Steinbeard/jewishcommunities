# =====================================================================
# Kehillah leader rank flavorization.
#
# Without this, the leader of a Kehillah shows the generic county-tier
# rank "Count"/"Countess" (see common/flavorization/00_title_holders.txt's
# own `count`/`count_female` entries, priority 15) -- correct
# mechanically (c_kehillah_worms is a county-tier title) but wrong in
# every other sense: a Jewish community's leader was never a count.
#
# The tier fields below were duchy until 2026-09-07, when the title
# moved to county tier (see common/landed_titles/
# kehillah_landed_titles.txt for why). A flavorization entry only
# applies to holders of a title at its declared tier, so these had to
# move with it or the entries would simply never match and every leader
# would read as "Count". The priorities need no adjustment: the generic
# rank they have to beat got LOWER, not higher (count is 15 where duke
# was 25), so 301/302 still win comfortably.
#
# HISTORICAL NOTE. Real Ashkenazi communal leadership in this period was
# not a single office. The rav (rabbi) held religious/halakhic
# authority; day-to-day communal governance -- finances, relations with
# the host authorities, representing the community outward -- was
# often held by a separate lay figure, most commonly called a parnas
# (plural parnasim), sometimes rosh ha-kahal ("head of the community").
# The two frequently overlapped in a small or scholarly-led community
# (Isaac ben Eliezer at Worms combined both kinds of standing in one
# person, per docs/scenarios/worms-1066.md), and diverged in others,
# where a wealthy or well-connected layman led communal affairs while a
# separate rabbi held religious authority. This mod models one leader,
# not two offices, so rather than force every leader into one identity,
# the flavor text follows which kind of standing THIS leader actually
# has -- see kehillah_leader_is_rabbinic_trigger in
# common/scripted_triggers/kehillah_scripted_triggers.txt.
#
# MECHANISM. Modeled directly on vanilla's own landless_adventurer
# "camp purpose" flavors (duke_landless_adventurer_camp_scholars and
# its siblings in 00_title_holders.txt): a government-gated, flag-gated
# character flavorization at a priority high enough to beat the generic
# `duke`/`duke_female` (25). Deliberately gated on the GOVERNMENT
# (kehillah_government) rather than on c_kehillah_worms by name, again
# matching the adventurer precedent (which gates on
# landless_adventurer_government + domicile_type = camp, never on any
# specific title): a later phase adding a second Kehillah title for a
# different region must not need this file touched to inherit the same
# flavor. The flag itself (kehillah_rabbinic_leader_flag) is kept in
# sync by kehillah_update_leader_flavor_effect, called at game start
# and on every change of leader (common/on_action/
# kehillah_on_actions.txt).
#
# GENDER NOTE. Both flavors below reuse the identical localized text for
# male and female holders. A female rav is anachronistic for this
# period (the game's own accommodation of female rulers already asks
# for more period license than that in other ways); rather than invent
# a shaky feminine-Hebrew neologism for either term, both genders read
# the same word, the way an English "Duke"-style borrowing sometimes
# does for genuinely unisex source titles.
# =====================================================================

kehillah_rav_male = {
	type = character
	gender = male
	special = holder
	tier = county
	domicile_type = kehillah_quarter
	priority = 302
	governments = { kehillah_government }
	flag = kehillah_rabbinic_leader_flag

	flavourization_rules = {
		top_liege = no

kehillah_rav_spouse_ashkenazi_female = {
    # ck3-tiger (2026-09-07): `type = spouse` isn't a real flavourization
    # type (valid values are character/title/domicile, per
    # common/flavorization/_flavourization.info in the base game) -- these
    # three entries work anyway because of how the type=character case is
    # actually resolved: "Contextual Character" is the owner of the Causing
    # Title, which for a wife IS her husband when his title outranks hers.
    # So `type = character` here, tested against a female Character, with
    # the government/domicile_type/flag/tier fields matching against the
    # husband as Contextual Character, is the correct -- and only -- way to
    # give the rav's wife her own flavor text distinct from his (which is
    # exactly why his own entries set spouse_takes_title = no: to leave
    # room for these to fire instead of the default same-title-for-spouse
    # behavior).
    type = character
    gender = female
    special = holder
    tier = county
    domicile_type = kehillah_quarter
    priority = 303
    governments = { kehillah_government }
    flag = kehillah_rabbinic_leader_flag

    # target_root_culture isn't a real field either; culture filtering is
    # via name_lists (matched against common/culture/name_lists) or
    # heritages, per the same schema doc. name_list_ashkenazi is Israelite
    # culture's actual name list (common/culture/cultures/00_israelite.txt).
    name_lists = { name_list_ashkenazi }

    flavourization_rules = {
        top_liege = no
    }
}
