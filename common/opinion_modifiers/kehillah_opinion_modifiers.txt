# =====================================================================
# Kehillah opinion modifiers.
#
# Small, targeted opinion swings for the organic dispute-ruling event
# (events/kehillah_dispute_events.txt, v2 spec section 3) -- a
# disputant's opinion of the LEADER specifically, which is a different
# thing from common/modifiers/kehillah_modifiers.txt's four
# community-wide self-modifiers on the leader.
#
# Applied via reverse_add_opinion (the TARGET's opinion of the acting
# scope changes), confirmed against vanilla's own reverse_opinion
# trigger semantics (common/scripted_effects/00_councillor_effects.txt,
# where reverse_opinion = { target = X } reads as "X's opinion of the
# current scope") and reverse_add_opinion effect usage (e.g.
# events/activities/tour_activity/tour_grounds_events_chad.txt).
#
# Both magnitude and duration are overridden per ruling tier at the
# call site (opinion = / years =), which vanilla's own add_opinion/
# reverse_add_opinion calls do routinely -- see e.g.
# events/activities/playdate_activity/playdate_events.txt
# (opinion = 20, years = 10 on top of a named modifier). That is why
# there are only two modifiers here rather than one per tier.
# =====================================================================

kehillah_dispute_ruling_favor_opinion = {
	opinion = 15
	years = 4
	decaying = yes
}

kehillah_dispute_ruling_disfavor_opinion = {
	opinion = -15
	years = 4
	decaying = yes
}

# ADDED 2026-09-09, Bet Din case 4 ("The Recalcitrant Husband"). A
# separate, far larger modifier rather than another override of the two
# above: those two exist specifically to be overridden per ruling tier
# (opinion = / years = at the call site, per this file's own header
# note) with magnitudes on the -20..+20 scale; this one is not tiered at
# all -- it fires once, at a fixed huge magnitude, only on the "beat him
# up" direction (kehillah_bet_din_recalcitrant_husband_resolution_effect,
# common/scripted_effects/kehillah_bet_din_scripted_effects.txt) -- and
# giving it its own name keeps that call site from silently inheriting
# whichever tier magnitude a future edit to the dispute modifiers above
# might introduce. -70, literal: every opinion_modifiers entry in
# vanilla's own game/common/opinion_modifiers uses a literal number for
# this field, not a script-value token, so this follows suit rather than
# assuming untested syntax.
kehillah_bet_din_beaten_husband_opinion = {
	opinion = -70
	years = 15
	decaying = yes
}

# ADDED 2026-09-09, the conference's own guest pulse_actions (common/
# activities/pulse_actions/kehillah_bet_din_pulse_actions.txt) -- small,
# repeatable opinion swings between two attending characters over the
# course of a docket, not applied to the leader specifically (unlike the
# dispute-ruling pair above) and not overridden per call the way those
# two are. Kept separate from kehillah_dispute_ruling_favor/disfavor_
# opinion on purpose: reusing a modifier literally named for dispute
# rulings on unrelated small talk between guests would read wrong on
# hover, even though the numbers could have been made to match.
kehillah_bet_din_pulse_favor_opinion = {
	opinion = 10
	years = 3
	decaying = yes
}

kehillah_bet_din_pulse_disfavor_opinion = {
	opinion = -10
	years = 3
	decaying = yes
}

# ADDED 2026-09-10 (v6 spec section 1). Sh'um preserved as FLAVOR, not
# structure, now that d_kehillah_shum (the titular duchy that used to
# group Worms/Speyer/Mainz) is gone -- see common/landed_titles/
# kehillah_landed_titles.txt's header. A small, PERMANENT bond between
# whoever currently leads the three specific communities: no years/
# decaying field at all, matching vanilla's own permanent-opinion
# precedent (common/opinion_modifiers/00_opinion_modifiers.txt's
# bitter_rivalry_opinion/raised_as_my_own -- plain `opinion = X`, no
# years, is confirmed vanilla shorthand for "does not decay" -- vanilla
# opinion modifiers already omit `decaying` entirely rather than write
# `decaying = no`, so this follows the same convention rather than
# inventing one). 20 is the exact magnitude, implementer's discretion
# within the spec's own +15 to +25 range, matching the scale of this
# mod's other communal opinion modifiers above (the dispute-ruling pair
# at +-15). Applied/refreshed via kehillah_refresh_shum_bond_effect
# (common/scripted_effects/kehillah_scripted_effects.txt), called from
# game start and from on_title_gain whenever a NEW leader takes office in
# Worms, Speyer or Mainz specifically (common/on_action/kehillah_on_
# actions.txt), so the bond survives leadership turnover the way the
# spec asks.
kehillah_shum_bond_opinion = {
	opinion = 20
}

# above -- a real but ordinary-sized slight, not the beaten-husband
# modifier's own one-off huge magnitude below.
kehillah_bet_din_dismissed_mysticism_opinion = {

	years = 3
	decaying = yes
}
