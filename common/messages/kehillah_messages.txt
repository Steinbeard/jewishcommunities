# =====================================================================
# Kehillah interface messages.
#
# ROADMAP item S2.2: a pillar crossing a band boundary should TELL the
# player, rather than silently changing a number and a modifier. Sent by
# kehillah_update_pillar_band_effect (common/scripted_effects/
# kehillah_pillar_band_effects.txt).
#
# TWO TYPES, NOT ONE, purely so `style` can differ -- CK3 colours a
# message from the type, not the call site, so one type could not render
# "Prosperity is now Flourishing" in green and "Prosperity is now
# Crisis" in red. Everything else about them is identical.
#
# icon = "domicile_upgrade": the closest existing message icon to what
# this actually is. A Kehillah's pillars are the state of its Jewish
# Quarter, which IS a domicile in engine terms, so this is the same
# visual language the player already associates with the community
# itself rather than a borrowed one.
#
# message_filter_type = court_grandeur: reusing the vanilla filter for
# "your court's standing tier has changed", which is the nearest
# existing analogue to a pillar band change, so these land in a filter
# category players already understand and can already mute. Defining a
# new filter type would add a settings-menu row for a single mod.
# =====================================================================

msg_kehillah_band_improved = {
	icon = "domicile_upgrade"
	title = event_message_title
	desc = event_message_effect
	style = good
	message_filter_type = court_grandeur
}

msg_kehillah_band_worsened = {
	icon = "domicile_upgrade"
	title = event_message_title
	desc = event_message_effect
	style = bad
	message_filter_type = court_grandeur
}
