# =====================================================================
# THE CHEREM TRAIT (2026-11-11, Daniel: "Let's do it as a single lifestyle
# trait! There are other negative progress traits, like health conditions
# and gallowsbait"). docs/spec/2026-11-11-reviews-and-cherem.md section 3
# and its build record.
#
# One trait, one track of 0-100 XP, read as the Talmud's gradation of bans:
#    1-24  Reproved   (Tokhachah): no effect yet, a warning
#   25-49  Nezifah    (rebuked)
#   50-99  Niddui     (shunned)
#     100  Cherem     (the full ban; replaces vanilla's excommunicated for
#                      Jews, whose effects it mirrors at this level)
# Track thresholds add up, as vanilla's gallowsbait's do. The XP is moved by
# common/scripted_effects/kehillah_cherem_effects.txt only; the icon reuses
# vanilla's heresiarch.dds, a book pierced by a dagger (Daniel, 2026-11-11;
# the mod's art policy).
# =====================================================================

kehillah_cherem_trait = {
	category = lifestyle
	icon = heresiarch.dds
	shown_in_ruler_designer = no

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_kehillah_cherem_trait
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 100 } }
				desc = trait_kehillah_cherem_trait_cherem
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 50 } }
				desc = trait_kehillah_cherem_trait_niddui
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 25 } }
				desc = trait_kehillah_cherem_trait_nezifah
			}
			desc = trait_kehillah_cherem_trait_reproved
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_kehillah_cherem_trait_desc
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 100 } }
				desc = trait_kehillah_cherem_trait_cherem_desc
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 50 } }
				desc = trait_kehillah_cherem_trait_niddui_desc
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = kehillah_cherem_trait value >= 25 } }
				desc = trait_kehillah_cherem_trait_nezifah_desc
			}
			desc = trait_kehillah_cherem_trait_reproved_desc
		}
	}

	track = {
		# Nezifah.
		25 = {
			zealot_opinion = -10
		}
		# Niddui.
		50 = {
			zealot_opinion = -15
			same_faith_opinion = -5
			monthly_piety_gain_mult = -0.1
		}
		# Cherem: vanilla excommunicated's own figures, less what Niddui
		# already gives.
		100 = {
			zealot_opinion = -25
			same_faith_opinion = -5
			monthly_piety_gain_mult = -0.4
			piety_standalone_gain_mult = -0.5
			domain_tax_same_faith_mult = -0.25
			levy_reinforcement_rate_same_faith = -0.5
		}
	}
}
