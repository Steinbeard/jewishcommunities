# Overrides vanilla's `excommunicated` trait (common/traits/00_traits.txt, CK3 1.20.0.4)
# so Jewish characters see "In Herem" instead of "Excommunicated". Vanilla gives the
# trait no `name` block, and a trait's name can only vary by faith through one, so the
# whole trait has to be redefined -- CK3 replaces database objects whole, by key.
# Everything below is vanilla UNCHANGED except the added `name` block and the two
# Jewish `desc` entries. Regenerate from vanilla after any game update, as with
# common/religion/religion_types/kehillah_judaism_reserved_names.txt.
excommunicated = {
	category = fame
	monthly_piety_gain_mult = -0.5
	piety_standalone_gain_mult = -0.5
	zealot_opinion = -50
	same_opinion = 50
	domain_tax_same_faith_mult = -0.25
	levy_reinforcement_rate_same_faith = -0.5
	army_toughness_mult = -0.1

	# Links the trait tooltip to the Recover from Excommunication game concept.
	flag = excommunicated_recovery_hint

	shown_in_ruler_designer = no

	triggered_opinion = {
		parameter = excommunication_active
		opinion_modifier = tenet_excommunicated_crime
		same_faith = yes
	}

	rite_modifier = {
		parameter = tenet_communion_denied_sacrament
		monthly_piety_gain_mult = -0.25
		piety_standalone_gain_mult = -0.25
		same_faith_opinion = -10
	}

	rite_modifier = {
		parameter = tenet_purgatory_excommunication_penance
		monthly_piety_gain_mult = 0.25
		piety_standalone_gain_mult = 0.25
		same_faith_opinion = 10
	}

	# ADDED by this mod.
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_excommunicated
			}
			triggered_desc = {
				trigger = {
					exists = faith.religion
					faith.religion = religion:judaism_religion
				}
				desc = trait_excommunicated_herem
			}
			desc = trait_excommunicated
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = { exists = this }
				}
				desc = trait_excommunicated_desc
			}
			# ADDED by this mod.
			triggered_desc = {
				trigger = {
					exists = faith.religion
					faith.religion = religion:judaism_religion
				}
				desc = trait_excommunicated_herem_character_desc
			}
			desc = trait_excommunicated_character_desc
		}
	}
}
