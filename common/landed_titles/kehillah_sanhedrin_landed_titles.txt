# V46 (docs/spec/v46-the-sanhedrin.md): the Sanhedrin, Rabbinic Judaism's
# head-of-faith title once restored.
#
# It has no history entry, so it does not exist at any game start; only
# kehillah_restore_sanhedrin_decision creates it. A landless duchy, the same
# tier as every Kehillah, held ALONGSIDE the head's own community: the
# never-primary score is vanilla's PAM cardinal pattern
# (07_pam_ecclesiastical_titles.txt), and kehillah_sanhedrin_on_title_gain
# also sets the community back as primary, because pillars, domicile and
# succession all read primary_title. Tiberias, the Sanhedrin's last seat in
# antiquity, is only the placeholder capital: the court sits wherever its
# Nasi leads a community (kehillah_sanhedrin_sit_with_nasi_effect), from
# the moment it is restored.
d_kehillah_sanhedrin = {
	capital = c_tiberias
	landless = yes
	allow_domicile = no
	definite_form = yes
	ruler_uses_title_name = no
	can_be_named_after_dynasty = no
	can_use_nomadic_naming = no
	no_automatic_claims = yes
	de_jure_drift_disabled = yes
	destroy_if_invalid_heir = yes

	ai_primary_priority = {
		add = -1000
	}

	color = { 230 205 120 }
}
