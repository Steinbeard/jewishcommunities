# =====================================================================
# Kehillah titles.
#
# Source of truth: docs/spec/v1-kehillah-community.md section 6 and
# docs/scenarios/worms-1066.md section 4 (which left the tag as "TBD in
# technical design" -- this file is that decision: c_kehillah_worms).
#
# Structure: a LANDLESS COUNTY-TIER title.
#
# CHANGED FROM DUCHY TIER, 2026-09-07. A Kehillah is a community inside
# a single town, so county tier is what it actually is. Duchy tier also
# made no room for the intended next step -- neighbouring communities in
# Speyer and Mainz, and a regional body (takkanot, rabbinical synods)
# sitting above them. That body wants to be the duchy-tier thing; the
# individual communities cannot already be occupying that rank.
#
# The previous version of this header claimed duchy tier was "the tier
# the landless_playable government rule expects". That was not true.
# common/governments/_governments.info documents landless_playable as
# nothing more than "Allow rulers be playable even when they do not have
# any county" -- it says nothing about tier. And vanilla ships 442
# county-tier landless titles: every c_nf_* noble-family title in
# common/landed_titles/01_japan_noble_family.txt and its Korean and
# Chinese siblings. c_nf_yamato is the same shape as this title --
# top-level in the file, not nested in any duchy, no baronies, landless,
# with capital pointing at a real county somebody else holds.
#
# The domicile survives the tier change because domiciles are attached
# to the GOVERNMENT, not the title: kehillah_government declares
# domicile_type = kehillah_quarter, and kehillah_quarter gates itself on
# government_has_flag = government_is_kehillah. Neither the domicile
# type database nor the landed_titles allow_domicile field (default yes,
# documented as applying "despite possibly landless") mentions tier at
# all.
#
# The government's use_title_tier_modifiers = no still strips the
# prestige that would otherwise ride along with the rank.
#
# The county under the community (c_worms) is untouched and still
# belongs to its vanilla holder, Bishop Siegfried. The Kehillah title
# is layered over it, not carved out of it. That is the whole premise:
# you do not own the ground you stand on.
#
# This file adds ONE new title and modifies no vanilla title, so it
# merges cleanly with the vanilla landed_titles database.
# =====================================================================

c_kehillah_worms = {
	# Deep blue, distinct from the vanilla Jewish religious-head titles
	# in the 0/100-140/255 range so the two are not confused on the map.
	color = { 45 70 125 }

	landless = yes
	# Marks this as a PLAYABLE landless ruler's title rather than a
	# titular one like k_papal_state. Without it the engine refuses to
	# hand the holder to a player -- pdx_assert.cpp:619 "Trying to
	# assign unplayable character" -- and the game crashes on Play.
	# Every vanilla d_laamp_* adventurer title carries this flag.
	require_landless = yes
	definite_form = yes

	# The holder is addressed by their own name and role, not as a duke
	# of anything -- same treatment vanilla gives religious head titles.
	ruler_uses_title_name = no

	# A Kehillah is an institution, never a family possession. This is
	# the title-level statement of the same principle the government
	# type makes by disabling cadet branches and dynasty-named realms.
	can_be_named_after_dynasty = no
	can_use_nomadic_naming = no

	# Nobody inherits, claims or drifts into leading a Jewish community.
	# Succession runs exclusively through
	# kehillah_appointment_succession_law.
	no_automatic_claims = yes
	de_jure_drift_disabled = yes

	# The community sits in Worms and stays there. Province 2731,
	# verified against the installed 1.19 files and matching
	# docs/scenarios/worms-1066.md section 1.
	capital = c_worms
}
