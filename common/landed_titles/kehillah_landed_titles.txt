# =====================================================================
# Kehillah titles.
#
# CURRENT TIER (2026-09-20): all fifteen pre-authored communities are
# flat, landless DUCHY titles (d_kehillah_*), matching the duchy title
# created by Found a Jewish Community. Only their tier/IDs changed in
# this pass; their capital still points at the real host county, whose
# vanilla holder continues to own the land. Earlier v7 county-tier
# reasoning below is retained as design history, not current tier.
#
# Source of truth: docs/spec/v1-kehillah-community.md section 6,
# docs/scenarios/worms-1066.md section 4, docs/spec/v4-regional-
# communities-and-batei-din.md (the original Sh'um duchy), docs/spec/
# v6-minhag-regions.md sections 1-3 (the four-duchy rewrite), and (for
# THIS rewrite) docs/spec/v7-minhag-geographic-tagging.md section 1,
# which supersedes v6 sections 1-2 entirely.
#
# REWRITTEN 2026-09-10 (v7 spec, same day as v6, following a design
# conversation that changed the load-bearing mechanism). The FOUR titular
# duchies v6 built four hours earlier in this same pass --
# d_kehillah_western_ashkenaz, d_kehillah_eastern_ashkenaz,
# d_kehillah_anglia, d_kehillah_provence -- are REMOVED. Every one of the
# fifteen community titles below were made FLAT: top-level
# landless title with no duchy parent at all, exactly the shape
# Worms had before v4/v6 ever added a regional parent, and the same
# flat arrangement vanilla's own c_nf_yamato landless title uses
# (game/common/landed_titles/01_japan_noble_family.txt: c_nf_yamato sits
# bare at file top level, no enclosing d_/k_/e_ title, confirmed directly
# in the installed files before this rewrite).
#
# WHY, SO SOON AFTER V6 BUILT THE DUCHY TIER CLEANLY: v6's own duchy
# structure worked for the fifteen communities authored
# once, correctly, in this file. It does not support the user's actual
# goal, surfaced in the SAME day's follow-up conversation: founding a new
# Kehillah literally anywhere, not just at one of a pre-authored set of
# locations. De jure parentage is fixed at file-load time -- no effect
# can dynamically reassign an existing title's de jure liege into a
# duchy computed at runtime, though a founded title's own CAPITAL can be
# moved freely (set_capital_county, confirmed vanilla, works on titular
# titles). A de-jure-nested minhag structure can therefore only ever
# correctly describe communities placed in this file in advance -- so
# widening it further (as v6 did, from one duchy to four) still could
# never reach "anywhere," no matter how many regions it eventually held.
# Full account: docs/spec/v7-minhag-geographic-tagging.md's own "Why this
# supersedes v6 section 1-2".
#
# THE REPLACEMENT MECHANISM: minhag membership is no longer a title-tree
# fact at all. It is now a computed fact about REAL, ALWAYS-PRESENT
# vanilla geography -- which real de jure duchy (or, for Anglia, kingdom/
# empire) a Kehillah's host county sits in, looked up against a
# mod-defined table, tagged onto the real vanilla duchy titles themselves
# via a one-time effect at game start (kehillah_tag_minhag_regions_effect,
# common/scripted_effects/kehillah_minhag_geography_effects.txt) rather
# than via title nesting. See that effect's own header for the full
# twelve-region table. common/scripted_triggers/kehillah_scripted_
# triggers.txt's kehillah_shares_minhag_region_trigger and the two
# triggers built on top of it read that tagging, not de jure parentage,
# for exactly this reason.
#
# Sh'um itself is UNCHANGED by this rewrite -- it was already preserved
# as FLAVOR, not structure, by the v6 pass (a permanent opinion bond
# between whoever currently leads Worms/Speyer/Mainz,
# kehillah_shum_bond_opinion, common/opinion_modifiers/kehillah_opinion_
# modifiers.txt), and nothing about removing the wider regional duchy
# tier changes that.
#
# EVERY COUNTY TAG BELOW is exactly as v6 verified it against the
# installed CK3 1.19 files (grep common/landed_titles/00_landed_titles.txt)
# -- this rewrite only removes the four enclosing d_kehillah_<region>
# blocks and un-indents their contents by one tab; not one county tag,
# capital line, or per-title field changed. Several of the user's raw
# place-names in the original roster turned out not to be county-tier
# titles at all -- major cities are frequently baronies inside a
# differently-named county -- and are still called out at each capital
# line below so nobody re-derives them incorrectly from the city name
# alone.
#
# The county under each community (c_worms, c_troyes, c_praha, ...) is
# untouched and still belongs to its vanilla holder. The Kehillah title
# is layered over it, not carved out of it -- you do not own the ground
# you stand on. This file adds new titles and modifies no vanilla title,
# so it merges cleanly with the vanilla landed_titles database.
# =====================================================================

# Dedicated test fixture for the adventurer -> Kehillah founding path.
# Vanilla landless-adventurer bookmarks always begin with a backing
# d_laamp_* title; the test bookmark uses this isolated equivalent so it
# cannot take ownership of a historical adventurer title.
#
# capital = c_frankfurt, NOT c_worms (FIXED 2026-09-23 live-test session).
# A landless title's `capital` is what a holder with no real domicile
# resolves as their `location` to, and the decision's own is_valid gate
# now refuses to found a second community at a location that already has
# one registered (custom_tooltip kehillah_found_community_requires_empty_
# location_tt, added after the 2026-09-20 pass). With capital = c_worms,
# this fixture's own Yitzhak always spawned standing exactly on top of
# Isaac's already-registered Kehillah of Worms (d_kehillah_worms, held
# from 1064 per history/characters/worms_1066.txt) -- confirmed live via
# a run/*.txt probe: the fixture's own is_valid failed the empty-location
# clause specifically, with every other clause (10 Jewish followers after
# seeding, rabbinic-leader, is_adult, etc.) passing. That made the fixture
# permanently unable to exercise its own decision once the gate shipped,
# which is exactly backwards for a test fixture. c_frankfurt is a real
# vanilla county (00_landed_titles.txt), in the same Rhineland area as the
# Sh'um trio for thematic consistency with "A Rabbi on the Road" ("Stop
# wandering... found a Jewish community"), and is not the capital of any
# other d_kehillah_* title in this file, so it cannot collide with a
# pre-authored community the way c_worms did.
d_kehillah_founder_test = {
	landless = yes
	require_landless = yes
	definite_form = yes
	ruler_uses_title_name = no
	no_automatic_claims = yes
	destroy_if_invalid_heir = yes
	capital = c_frankfurt
	color = { 100 100 100 }
}

d_kehillah_worms = {
	landless = yes
	# Marks this as a PLAYABLE landless ruler's title rather than a
	# titular one like k_papal_state. Without it the engine refuses
	# to hand the holder to a player -- pdx_assert.cpp:619 "Trying
	# to assign unplayable character" -- and the game crashes on
	# Play. Every vanilla d_laamp_* adventurer title carries this
	# flag.
	require_landless = yes
	definite_form = yes

	# The holder is addressed by their own name and role, not as a
	# count of anything -- same treatment vanilla gives religious
	# head titles.
	ruler_uses_title_name = no

	# A Kehillah is an institution, never a family possession.
	can_be_named_after_dynasty = no
	can_use_nomadic_naming = no

	# Nobody inherits, claims or drifts into leading a Jewish
	# community. Succession runs exclusively through
	# kehillah_appointment_succession_law.
	no_automatic_claims = yes
	de_jure_drift_disabled = yes

	# The community sits in Worms and stays there. Province 2731,
	# verified against the installed 1.19 files and matching
	# docs/scenarios/worms-1066.md section 1.
	capital = c_worms

	color = { 45 70 125 }
}
