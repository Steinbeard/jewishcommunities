# =====================================================================
# Kehillah titles.
#
# Source of truth: docs/spec/v1-kehillah-community.md section 6 and
# docs/scenarios/worms-1066.md section 4 (which left the tag as "TBD in
# technical design" -- this file is that decision: d_kehillah_worms).
#
# Structure, per spec section 6: a LANDLESS DUCHY-TIER title, the same
# shape vanilla uses for Byzantine administrative offices and for
# religious head titles such as d_karaism. Duchy tier is not a claim
# that a Kehillah is the peer of a duchy -- it is simply the tier the
# landless_playable government rule expects, and the government's
# use_title_tier_modifiers = no strips the unearned prestige that would
# otherwise come with it.
#
# The county under the community (c_worms) is untouched and still
# belongs to its vanilla holder, Bishop Siegfried. The Kehillah title
# is layered over it, not carved out of it. That is the whole premise:
# you do not own the ground you stand on.
#
# This file adds ONE new title and modifies no vanilla title, so it
# merges cleanly with the vanilla landed_titles database.
# =====================================================================

d_kehillah_worms = {
	# Deep blue, distinct from the vanilla Jewish religious-head titles
	# in the 0/100-140/255 range so the two are not confused on the map.
	color = { 45 70 125 }

	landless = yes
	# Marks this as a PLAYABLE landless ruler's title rather than a
	# titular one like k_papal_state. Without it the engine refuses to
	# hand the holder to a player -- pdx_assert.cpp:619 "Trying to
	# assign unplayable character" -- and the game crashes on Play.
	# Every vanilla d_laamp_* adventurer title carries this flag.
	require_landless = yes
	definite_form = yes

	# The holder is addressed by their own name and role, not as a duke
	# of anything -- same treatment vanilla gives religious head titles.
	ruler_uses_title_name = no

	# A Kehillah is an institution, never a family possession. This is
	# the title-level statement of the same principle the government
	# type makes by disabling cadet branches and dynasty-named realms.
	can_be_named_after_dynasty = no
	can_use_nomadic_naming = no

	# Nobody inherits, claims or drifts into leading a Jewish community.
	# Succession runs exclusively through
	# kehillah_appointment_succession_law.
	no_automatic_claims = yes
	de_jure_drift_disabled = yes

	# The community sits in Worms and stays there. Province 2731,
	# verified against the installed 1.19 files and matching
	# docs/scenarios/worms-1066.md section 1.
	capital = c_worms
}
