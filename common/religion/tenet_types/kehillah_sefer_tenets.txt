# =====================================================================
# Jewish tenets of the Sefer redesign (2026-11-08). Design:
# docs/spec/2026-11-08-sefer-system-redesign.md section 4.2, with Daniel's
# decision 6 (all tenets in the first pass; Talmud Torah is the Oral
# Torah tenet). Books argue for or against these (the catalog's
# argues_for / argues_against), and a tenet's status in a faith sets how
# high such a book may climb in that faith's rites. Faith statuses are set
# by kehillah_sefer_tenet_status_init_effect.
#
# Axes: Reason (Hokhmah / Emunah Temimah, exclusive as core tenets), Law
# (Pesak / Pilpul), Authority (Ge'onut / Minhag Avoteinu), and the Land
# (Avelei Tziyon). Kabbalah is vanilla Esotericism, which already has a
# Jewish name; Hasidut is vanilla Asceticism given a Jewish face in
# kehillah_asceticism_override.txt.
# =====================================================================

# Mechanics outside the books (2026-11-08, Daniel's first look; design in
# docs/spec/2026-11-08-sefer-tenet-mechanics.md). Each tenet has a core
# parameter read through rite_has_parameter and a personal one read through
# has_personal_tenet_flag; the numbers are in kehillah_tenet_values.txt:
#   Hokhmah          the Shtadlan's petitions to the host (track values)
#   Emunah Temimah   sermons also steady the community (sermon effects)
#   Pesak            judges rule better: Bet Din, disputes, responsa
#   Pilpul           the yeshiva's fame (yeshiva quarterly tick)
#   Ge'onut          the head of rite's authority; easier acclamation
#   Minhag Avoteinu  festival feasts steady the community
#   Avelei Tziyon    Send Alms to Jerusalem (kehillah_tenet_decisions.txt)
# Virtues and sins avoid Religious Scholar except where Karaites hold the
# tenet: for Rabbanites the Rabbi trait eclipses it.

tenet_kehillah_hokhmah = {
	icon = tenet_alexandrian_catechism
	name = { desc = tenet_kehillah_hokhmah_name }
	desc = { desc = tenet_kehillah_hokhmah_desc }

	can_pick = {
		religion = religion:judaism_religion
		custom_description = {
			text = kehillah_tenet_reason_exclusivity_desc
			NOT = { tenet:tenet_kehillah_emunah_temimah = { is_in_list = selected_doctrines } }
		}
	}
	can_pick_as_personal_tenet = {
		faith = { religion = religion:judaism_religion }
	}

	piety_cost = {
		value = faith_tenet_cost_low
		if = {
			limit = { rite_has_tenet = tenet_kehillah_hokhmah }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	traits = {
		virtues = {
			lifestyle_physician = { weight = 15 scale = 1 }
			lifestyle_scholar = { weight = 10 scale = 1 }
		}
		sins = {
			zealous = { weight = 15 scale = 1 }
		}
	}

	character_modifier = {
		different_faith_opinion = 5
		monthly_learning_lifestyle_xp_gain_mult = 0.05
	}
	personal_tenet_modifier = {
		learning = 1
		different_faith_opinion = 5
	}
	parameters = { kehillah_sefer_hokhmah kehillah_tenet_hokhmah_shtadlan }
	personal_tenet_parameters = { kehillah_sefer_hokhmah_personal kehillah_tenet_hokhmah_shtadlan_personal }
}
