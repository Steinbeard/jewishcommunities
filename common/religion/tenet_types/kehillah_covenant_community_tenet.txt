# The Jewish alternative to vanilla Communal Identity. Its static modifiers
# deliberately cross rites; vanilla's definition and non-Jewish users are intact.
tenet_kehillah_covenant_community = {
    icon = kehillah_covenant_community
    name = tenet_kehillah_covenant_community_name
    desc = tenet_kehillah_covenant_community_desc
    historical_glossary = {
        first_valid = {
            triggered_desc = {
                trigger = { rite_religion_tag = judaism_religion }
                desc = kehillah_am_kadosh_historical_glossary
            }
            desc = kehillah_covenant_community_historical_glossary
        }
    }
    can_pick_as_personal_tenet = { always = yes }
    piety_cost = {
        value = faith_tenet_cost_low
        if = {
            limit = { rite_has_tenet = tenet_kehillah_covenant_community }
            multiply = faith_unchanged_doctrine_cost_mult
        }
    }
    character_modifier = { same_faith_opinion = 10 }
    personal_tenet_modifier = { opinion_of_same_faith = 5 }
    parameters = { same_culture_conversion_bonus_active same_faith_promote_culture_bonus_active sanctify_artifacts_communal }
    county_modifier = { prosperity_growth_factor = 0.1 }
    personal_tenet_parameters = {
        tenet_communal_identity_free_council_sway
        tenet_communal_identity_sway_piety
        tenet_communal_identity_language_devotion
        tenet_communal_identity_language_reward
    }
    traits = {
        virtues = {
            generous = { weight = 20 scale = 1 }
            loyal = { weight = 15 scale = 1 }
        }
        sins = {
            callous = { weight = 20 scale = 1 }
            disloyal = { weight = 15 scale = 1 }
        }
    }
}
