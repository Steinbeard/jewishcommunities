# Torah study as an ordinary religious practice, rather than a synonym for
# ordination or the vanilla Scholar trait. The active reward is granted by
# kehillah_study_specific_work_effect when a work is mastered.
#
# 2026-11-08 (Sefer redesign, Daniel's decision 6): Talmud Torah is the
# Rabbanite tenet of the Oral Torah, the counter to Ba'alei ha-Miqra:
# Karaism prohibits it (kehillah_sefer_tenet_status_init_effect), the two
# cannot both be core, and it can now be held as a personal tenet. It also
# makes canon never wear out: a reread of the Tanakh or a Talmud order pays
# a follower as a first mastery does, and a siyum pays half again.
tenet_talmud_torah = {
	icon = tenet_pursuit_of_knowledge
	name = { desc = tenet_talmud_torah_name }
	desc = { desc = tenet_talmud_torah_desc }

	can_pick = {
		religion = religion:judaism_religion
		custom_description = {
			text = kehillah_tenet_oral_torah_exclusivity_desc
			NOT = { tenet:tenet_baalei_hamiqra = { is_in_list = selected_doctrines } }
		}
	}
	can_pick_as_personal_tenet = {
		faith = { religion = religion:judaism_religion }
	}
	personal_tenet_modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.1
	}
	personal_tenet_parameters = { kehillah_talmud_torah_personal }

	piety_cost = {
		value = faith_tenet_cost_low
		if = {
			limit = { rite_has_tenet = tenet_talmud_torah }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	traits = {
		virtues = {
			diligent = { weight = 20 scale = 1 }
			kehillah_rabbi_trait = { weight = 15 scale = 1 }
		}
		sins = {
			lazy = { weight = 20 scale = 1 }
		}
	}

	character_modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.10
	}

	parameters = {
		kehillah_talmud_torah_study_reward
	}
}
