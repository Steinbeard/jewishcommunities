# ordination or the vanilla Scholar trait. The active reward is granted by
# kehillah_study_specific_work_effect when a work is mastered.

	icon = tenet_pursuit_of_knowledge

	can_pick = {
		religion = religion:judaism_religion

			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	traits = {
		virtues = {
			diligent = { weight = 20 scale = 1 }
		}
		sins = {
			lazy = { weight = 20 scale = 1 }
		}
	}

	character_modifier = {
		lifestyle_scholar_xp_gain_mult = 0.10
	}

	parameters = {
		kehillah_talmud_torah_study_reward
	}
}
