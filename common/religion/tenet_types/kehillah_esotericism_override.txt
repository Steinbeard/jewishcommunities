# Vanilla mechanics; Kabbalah is a Rabbinic flavor, not every Jewish mysticism.
tenet_esotericism = {
	icon = tenet_esotericism

	historical_glossary = {
		first_valid = {
            triggered_desc = { trigger = { kehillah_rite_rabbanite_trigger = yes } desc = kehillah_esotericism_jewish_historical_glossary }
            desc = kehillah_esotericism_general_glossary
        }
	}

	piety_cost = {
		if = {
			limit = {
				rite_religion_tag = christianity_religion

			}
			value = faith_tenet_cost_high
		}
		else_if = {
			limit = {
				OR = {
					rite_religion_tag = hinduism_religion
					rite_religion_tag = jainism_religion
					rite_religion_tag = zoroastrianism_religion
					rite_has_doctrine = pagan_hostility_doctrine
				}
			}
			value = faith_tenet_cost_low
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_esotericism }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	traits = {
		virtues = {
			lifestyle_mystic = { weight = 15 scale = 1 }
			patient = { weight = 10 scale = 1 }
		}
		sins = {
			cynical = { weight = 15 scale = 1 }
		}
	}

	name = {
		first_valid = {
			triggered_desc = {
				trigger = {
					kehillah_rite_rabbanite_trigger = yes
				}
				desc = tenet_esotericism_name
			}
			desc = tenet_esotericism_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					kehillah_rite_rabbanite_trigger = yes
				}
				desc = tenet_esotericism_kabbalah_desc
			}
			desc = tenet_esotericism_desc
		}
	}

	parameters = {
		tenet_esotericism_mystic_education
		tenet_esotericism_eccentric_education
		tenet_esotericism_eccentric_bonuses
		sanctify_artifacts_occult
		pilgrimage_decision_active
	}

	personal_tenet_parameters = {
		tenet_esotericism_seek_spiritual_mysteries
	}

	personal_tenet_modifier = {
		lifestyle_mystic_xp_gain_mult = 0.5
		monthly_learning_lifestyle_xp_gain_mult = 0.1
	}
}
