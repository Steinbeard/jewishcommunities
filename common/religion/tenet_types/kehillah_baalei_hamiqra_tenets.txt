# Karaite scriptural authority. Study and public interpretation are provided
# by the separate Examine Scripture scheme, and since 2026-11-08 also by the
# library: every Karaite library holds the Tanakh and the founding Karaite
# works (docs/spec/2026-11-08-sefer-system-redesign.md). It is the counter
# to Talmud Torah, the Rabbanite tenet of the Oral Torah: the two cannot
# both be core, and it can be held as a personal tenet.
tenet_baalei_hamiqra = {
	icon = tenet_literalism
	name = { desc = tenet_baalei_hamiqra_name }
	desc = { desc = tenet_baalei_hamiqra_desc }

	can_pick = {
		religion = religion:judaism_religion
		custom_description = {
			text = kehillah_tenet_oral_torah_exclusivity_desc
			NOT = { tenet:tenet_talmud_torah = { is_in_list = selected_doctrines } }
		}
	}
	can_pick_as_personal_tenet = {
		faith = { religion = religion:judaism_religion }
	}

	piety_cost = {
		value = faith_tenet_cost_low
		if = {
			limit = { rite_has_tenet = tenet_baalei_hamiqra }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	traits = {
		virtues = {
			lifestyle_scholar = { weight = 15 scale = 1 }
		}
	}

	personal_tenet_modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.1
	}

	parameters = { kehillah_baalei_examine_scripture }
	personal_tenet_parameters = { kehillah_baalei_hamiqra_personal }
}
