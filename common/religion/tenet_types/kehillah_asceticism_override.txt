# CK3 1.20.1's Asceticism tenet, with a Jewish name and description
# (Hasidut, piety beyond the letter of the law) added for the 2026-11-08
# Sefer redesign. Every vanilla mechanical field is kept: meditation,
# laampdom and other scripts check this ID directly. Review against
# 00_tenet_types.txt after a game update, as kehillah_ritual_celebrations_
# override.txt is.
tenet_asceticism = {
	icon = tenet_asceticism

	historical_glossary = {
		first_valid = {
			triggered_desc = {
				trigger = { rite_religion_tag = judaism_religion }
				desc = tenet_kehillah_hasidut_historical_glossary
			}
			desc = tenet_asceticism_tapas_historical_glossary
		}
	}

	piety_cost = {
		if = {
			limit = {
				OR = {
					rite_religion_tag = zoroastrianism_religion
					rite_has_doctrine = pagan_hostility_doctrine
				}
			}
			value = faith_tenet_cost_high
		}
		else_if = {
			limit = {
				OR = {
					rite_religion_tag = christianity_religion
					rite_religion_tag = dualism_religion
					rite_religion_tag = buddhism_religion
					rite_religion_tag = jainism_religion
					rite_religion_tag = taoism_religion
				}
			}
			value = faith_tenet_cost_low
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_asceticism }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {
		meditation_mechanics_active
		unlock_voluntary_laampdom_faith
		sanctify_artifacts_humble
		tenet_asceticism_inappetetic_bonus
		tenet_asceticism_comfort_eater_penalty
	}

	character_modifier = {
		prowess = -2
		piety_standalone_gain_mult = 0.1
	}

	personal_tenet_modifier = {
		character_spiritual_fulfillment_gain_mult = 0.15
	}

	personal_tenet_parameters = {
		tenet_asceticism_inappetetic_trait
		unlock_voluntary_laampdom_faith
	}

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { rite_religion_tag = judaism_religion }
				desc = tenet_kehillah_hasidut_name
			}
			desc = tenet_asceticism_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { rite_religion_tag = judaism_religion }
				desc = tenet_kehillah_hasidut_desc
			}
			desc = tenet_asceticism_desc
		}
	}

	traits = {
		virtues = {
			temperate = { weight = 25 scale = 1 }
		}
		sins = {
			gluttonous = { weight = 25 scale = 1 }
			greedy = { weight = 10 scale = 1 }
			lifestyle_reveler = { weight = 15 scale = 1 }
		}
	}

	can_pick = {
		custom_description = {
			text = hesychasm_anachoresis_mortification_exclusivity_desc
			NOR = { has_tenet = tenet_anachoresis has_tenet = tenet_mortification_of_the_flesh has_tenet = tenet_hesychasm }
		}
	}
}
