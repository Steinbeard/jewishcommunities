# =====================================================================
# "Teaching on the Jews": what a Christian or Muslim rite teaches about
# Jewish subjects. Design: docs/spec/v31-host-teachings-stances-and-
# factions.md, section 4 (V31 Slice A).
#
# A teaching never decides an outcome. It sets the ceiling of the AI's
# Jewish Settlement Policy review, scales how hard zeal pushes, and gates
# forced conversion. Every rite holding one is assigned at game start
# (kehillah_assign_teachings_effect); a rite created later, or a save
# begun before this existed, falls back by religion (see
# common/scripted_triggers/kehillah_teaching_triggers.txt).
#
# Available when creating or reforming a Christian or Muslim faith, so a
# reformer may choose his rite's teaching.
# =====================================================================

kehillah_teaching_on_jews = {
	category = main_group
	divergence = sequence
	is_available_on_create = {
		OR = {
			religion_tag = christianity_religion
			religion_tag = islam_religion
		}
	}
}
