# =====================================================================
# KASHRUT (2026-11-23, Daniel). docs/spec/2026-11-23-kashrut.md.
# A fourth choice in vanilla 1.21's dietary_restrictions group, for every
# Jewish rite (replacing doctrine_diet_unrestricted).
#
# Vanilla holdings read their county's rite: in a Jewish-rite county they
# eat as with no restrictions. A Jewish ruler's lands raise more cattle and
# less meat (character_modifier below: no swine). What sets
# kashrut apart is the Jewish quarter: it buys cattle for its shochet, not
# meat (kehillah_jewish_quarter_01), and a market that sells cattle adds a
# little Affluence for a kashrut-keeping community
# (kehillah_bd_market_supply_prosperity_value, through
# kehillah_keeps_kashrut_trigger). Simplified 2026-11-23 (Daniel: the first
# cut was too convoluted; a missing good is no longer a penalty).
# =====================================================================

doctrine_diet_kashrut = {
	doctrine_group_type = dietary_restrictions
	index = 6
	divergence = 5

	icon = "special_doctrine_default"

	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = { rite_has_doctrine = doctrine_diet_kashrut }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else = {
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}

	can_pick = {
		custom_description = {
			text = kehillah_kashrut_jewish_only
			# can_pick runs in faith scope (live 2026-11-23: rite_religion_tag logged
			# 'Inconsistent trigger scopes (faith vs. rite)').
			religion_tag = judaism_religion
		}
	}

	# Read by vanilla buildings in a Jewish-rite county (holdings use their
	# county's rite): meat_permitted and leather_permitted, as No Dietary
	# Restrictions. Not fish_permitted (2026-11-23): vanilla gives that only
	# to doctrines that forbid meat, and castles then add a separate fish
	# food need beside the meat need.
	# kehillah_kashrut is this mod's: its tooltip line describes the quarter.
	parameters = {
		meat_permitted
		leather_permitted
		kehillah_kashrut
	}

	# Jewish lands (2026-11-23, Daniel: give it flavor): herds are raised for
	# the shochet and no swine are kept. Character modifiers with production
	# tags reach the holder's domain, as vanilla's merchant company buildings
	# use them. Applies to every character of the rite; for a Kehillah leader
	# or a merchant it touches only what they produce.
	character_modifier = {
		trade_good_cattle_production_mult = 0.15
		trade_good_meat_production_mult = -0.1
	}
}
