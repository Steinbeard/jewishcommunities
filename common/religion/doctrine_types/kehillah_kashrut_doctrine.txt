# =====================================================================
# KASHRUT (2026-11-23, Daniel). docs/spec/2026-11-23-kashrut.md.
# A fourth choice in vanilla 1.21's dietary_restrictions group, for every
# Jewish rite (replacing doctrine_diet_unrestricted).
#
# Vanilla holdings read their county's rite: in a Jewish-rite county they
# see meat, fish and leather permitted, as with no restrictions. What sets
# kashrut apart is carried by:
#   - kehillah_kashrut, a parameter this mod reads: a Jewish quarter needs
#     cattle for shechita and salt for koshering
#     (kehillah_jewish_quarter_01; kehillah_bd_kosher_table_value);
#   - the slaughterhouse, which sells to the town the meat Jews may not
#     eat (kehillah_slaughterhouse_01/02);
#   - character_modifier: Jewish traders deal in kosher cattle.
# =====================================================================

doctrine_diet_kashrut = {
	doctrine_group_type = dietary_restrictions
	index = 6
	divergence = 5

	icon = "special_doctrine_default"

	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = { rite_has_doctrine = doctrine_diet_kashrut }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else = {
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}

	can_pick = {
		custom_description = {
			text = kehillah_kashrut_jewish_only
			rite_religion_tag = judaism_religion
		}
	}

	parameters = {
		meat_permitted
		fish_permitted
		leather_permitted
		kehillah_kashrut
	}

	character_modifier = {
		trade_good_cattle_sell_price_mult = 0.02
	}
}
