# Hereditary priestly succession (kehillah_priestly_succession_law). Not a
# real vote: the holder is the only elector, and his score ranks the
# priestly line, as the Samaritan high priesthood passed within the
# priestly family. Only a man with the kohen trait (passed father to son,
# kehillah_traits.txt) of the holder's faith can be chosen.
# kehillah_samaritan_priesthood_yearly_effect keeps at least one such
# candidate at the holder's court, so the title (destroy_if_invalid_heir)
# never lacks an heir.
kehillah_priestly_elective = {
	elector_vote_strength = {
		base = 1
	}

	electors = {
		character_sets = {
			kehillah_priestly_holder = {
				type = holder
			}
		}
	}

	candidates = {
		limit = {
			kehillah_priestly_heir_candidate_trigger = { HOLDER = scope:holder }
		}
		character_sets = {
			kehillah_priestly_kin = {
				type = holder_close_or_extended_family
			}
			kehillah_priestly_council = {
				type = holder_council_members
			}
			kehillah_priestly_clergy = {
				type = same_faith_clergy
			}
		}
	}

	# root = the elector (the holder), scope:candidate = the candidate.
	# The line first: sons, then brothers, grandsons, nephews, other kin.
	# Within a tier, the eldest adult, then learning.
	candidate_score = {
		base = 0
		modifier = {
			add = 400
			scope:candidate = { is_child_of = root }
		}
		modifier = {
			add = 300
			scope:candidate = { is_sibling_of = root }
		}
		modifier = {
			add = 250
			scope:candidate = { is_grandchild_of = root }
		}
		modifier = {
			add = 200
			scope:candidate = { is_nibling_of = root }
		}
		modifier = {
			add = 100
			scope:candidate = { is_close_or_extended_family_of = root }
		}
		modifier = {
			add = 150
			scope:candidate = { is_adult = yes }
		}
		modifier = {
			add = scope:candidate.age
		}
		modifier = {
			add = scope:candidate.learning
		}
		modifier = {
			add = -300
			scope:candidate = { is_incapable = yes }
		}
		# A player holder's own choice among the Kohanim.
		modifier = {
			add = 50
			exists = scope:holder_candidate
			scope:candidate = scope:holder_candidate
		}
	}
}
