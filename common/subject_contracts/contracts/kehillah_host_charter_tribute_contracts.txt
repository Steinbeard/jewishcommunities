# =====================================================================
# The tribute a Kehillah pays its host (v31 §8.2, part 1: "the ordinary
# term"). ADDED 2026-10-30. Daniel, 2026-10-12: one unified system for
# every host, starting at 20% of the community's income.
#
# One term, listed in all three charter groups (the V15 fallback and
# both V16 regional groups), so every host draws it the same way.
#
# CHANGED 2026-11-08 (Daniel): each level carries an engine `tax =`
# share, as vanilla's tributary contracts do, so the tribute is a line
# in both treasuries' monthly breakdowns. This replaces the quarterly
# scripted transfer of 2026-10-30. Do not bring that transfer back
# alongside this, or the host is paid twice. v31 §8.2's engine risk --
# that the engine reads a Kehillah's domicile income as zero here -- is
# the first item of the 2026-11-08 test checklist; if it holds, see
# docs/implementation/2026-10-30-host-tribute.md for the fallback.
# The `flag` stays: the charter cache reads it for display. The rates
# are one script value per rung (kehillah_tribute_values.txt).
#
# REVERTED 2026-11-11 (live, Worms 1066): the engine's tributary tax is
# a share of HOLDING taxes, and a Kehillah has none -- its income is all
# domicile buildings -- so `tax =` charged 0 ("Monthly Expenses: No
# values"). Every rung is `tax = 0` again and the quarterly scripted
# transfer is back (kehillah_tribute_on_actions.txt). Never give these
# rungs a nonzero `tax` while that transfer runs.
#
# Customary is the default, and it is unconditional, as CK3 requires.
# The V16 day-one policy writer leaves this term alone, so every fresh
# charter starts at Customary; the negotiation window moves it.
# =====================================================================

kehillah_charter_tribute = {
	display_mode = tree
	icon = gold_icon
	defaults_to_highest_valid_level = no

	obligation_levels = {
		kehillah_tribute_none = {
			position = { 0 0 }
			flag = kehillah_tribute_none

			tax = 0
			subject_opinion = 5

			ai_liege_desire = 0
			ai_subject_desire = 10
			score = 2
		}
		kehillah_tribute_customary = {
			default = yes
			parent = kehillah_tribute_none
			position = { 1 0 }
			flag = kehillah_tribute_customary

			tax = 0

			ai_liege_desire = 4
			ai_subject_desire = 4
			score = 0
		}
		kehillah_tribute_heavy = {
			parent = kehillah_tribute_customary
			position = { 2 0 }
			flag = kehillah_tribute_heavy

			tax = 0
			subject_opinion = -15

			ai_liege_desire = 8
			ai_subject_desire = 0
			score = -2
		}
	}
}
