# =====================================================================
# V16 regional fresh Host Charter rows. These terms are selected for new
# Christian/Muslim charters while the shipped V15 group remains intact for
# existing contracts and other host traditions; see the matching groups file
# and docs/spec/v16-jewish-settlement-policy-and-charters.md §4.
#
# Shared design vocabulary:
# - quarter construction is about building new communal institutions;
# - community security is specifically a communal watch/guard, not a
#   claim that every Jewish individual has a general right to bear arms;
# - the highest armed-watch rung may unlock one *character* MaA regiment
#   (a small communal retinue), never title MaA or levies. The Kehillah
#   government's -10 MaA caps are deliberately offset only on that rung;
#   a live test must confirm the regiment persists through an eventual
#   transition to the vanilla adventurer government before this is shipped;
# - jurisdiction is civil/communal authority. "Bet Din discipline" means
#   compel appearance and impose communal penalties, not independent
#   capital jurisdiction or a generic police power;
# - text censorship is a policy category. A future Talmud-burning event
#   may attach individual works to a small proscribed list, but the
#   contract row must stay a readable regime, not an arbitrary list UI.
# =====================================================================

kehillah_christian_quarter_building_rights = {
	display_mode = tree
	icon = prestige_icon
	# Policy levels are written one day after creation, when the contract
	# exists. CK3 does not provide the liege scope while start_tributary
	# chooses defaults, so the unconditional default below is deliberately
	# restrictive until that deferred write commits.
	defaults_to_highest_valid_level = no
	obligation_levels = {
		kehillah_christian_building_free = {
			position = { 0 0 }
			flag = kehillah_christian_building_free
			ai_liege_desire = 0
			ai_subject_desire = 8
			score = 3
			# Encouraged and Allowed both make ordinary communal construction
			# free. Lesser rungs remain valid below it as local variation.
			is_valid = { always = yes }
		}
		kehillah_christian_building_permission = {
			parent = kehillah_christian_building_free
			position = { 1 0 }
			flag = kehillah_christian_building_permission
			ai_liege_desire = 5
			ai_subject_desire = 3
			score = 1
			is_valid = { always = yes }
		}
		kehillah_christian_building_forbidden = {
			parent = kehillah_christian_building_permission
			# CK3 requires the fallback default itself to be unconditional.
			# defaults_to_highest_valid_level still promotes a policy-valid
			# higher rung at fresh-contract creation; this is only Banned's
			# safe fallback.
			default = yes
			position = { 2 0 }
			flag = kehillah_christian_building_forbidden
			ai_liege_desire = 8
			ai_subject_desire = 0
			score = 0
		}
	}
}
