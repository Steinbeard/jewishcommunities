# =====================================================================
# Host Charter terms, v1 -- two dimensions. See
# docs/spec/v15-host-charter.md for the full account and the backlog of
# further terms (jurisdiction/self-governance, protection, residency)
# deliberately left for a later pass rather than guessed at here.
#
# Both terms are modelled on vanilla's own religious_rights
# (common/subject_contracts/contracts/special_contracts.txt:295-352,
# spike section 1): a display_mode, a default rung nobody has to
# negotiate for, `flag =` on every granted rung so the rest of the mod
# can read charter state with vassal_contract_has_flag, and asymmetric
# ai_liege_desire/ai_subject_desire/score so the negotiation window's
# own acceptance math (spike section 4) does the AI-weighing work this
# mod does not have to build.
#
# Faith doctrine is not wired into is_shown/is_valid for either term
# yet -- spike section 1 identifies this as "the cleanest possible
# landing zone for the By God Alone tenet restructure" once that
# restructure exists. Left unbuilt on purpose; do not invent tenet
# checks against today's tenet list, they would need redoing anyway.
# =====================================================================

# ---------------------------------------------------------------------
# Right to Lend at Interest. A three-rung ladder, checkbox would lose
# the "tolerated but limited" middle rung (pawnbroking) that most of
# real medieval Ashkenazi Jewish moneylending practice actually
# occupied -- Worms and Speyer's own historical charters graduated
# lending rights rather than granting or denying them outright.
# ---------------------------------------------------------------------

kehillah_charter_moneylending_rights = {
	display_mode = tree
	icon = gold_icon

	obligation_levels = {
		kehillah_moneylending_forbidden = {
			default = yes
			position = { 0 0 }

			ai_liege_desire = 3
			ai_subject_desire = 0
			score = 0
		}
		kehillah_moneylending_pawnbroking = {
			parent = kehillah_moneylending_forbidden
			position = { 1 0 }

			subject_opinion = 5
			subject_modifier = {
				monthly_income = 0.3
			}
			flag = kehillah_moneylending_pawnbroking_granted

			ai_liege_desire = 1
			ai_subject_desire = 6
			score = 2
		}
		kehillah_moneylending_full = {
			parent = kehillah_moneylending_pawnbroking
			position = { 2 0 }

			subject_opinion = 10
			subject_modifier = {
				monthly_income = 0.6
			}
			flag = kehillah_moneylending_full_granted

			ai_liege_desire = 0
			ai_subject_desire = 10
			score = 4
		}
	}
}

# ---------------------------------------------------------------------
# Walled Quarter. A flat right: either the community may fortify its
# own quarter against a mob, or it may not. checkbox is correct here --
# there is no meaningful middle tier the way there is for lending.
# ---------------------------------------------------------------------

kehillah_charter_quarter_rights = {
	display_mode = checkbox

	obligation_levels = {
		kehillah_quarter_open = {
			default = yes
			position = { 0 0 }

			ai_liege_desire = 2
			ai_subject_desire = 0
			score = 0
		}
		kehillah_quarter_walled = {
			parent = kehillah_quarter_open
			position = { 1 0 }

			subject_opinion = 8
			subject_modifier = {
				monthly_income = -0.1
				monthly_prestige = 0.1
			}
			flag = kehillah_quarter_walled_granted

			ai_liege_desire = 0
			ai_subject_desire = 9
			score = 3
		}
	}
}
