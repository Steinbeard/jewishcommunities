# =====================================================================
# Kehillah character modifiers.
#
# Kept small on purpose. These exist to make decisions and events leave
# a visible trace for a few years, not to become a second progression
# system running alongside the buildings.
#
# Each of the four below also feeds Stability while active -- removed
# from here 2026-09-07 when Influence was retired, since a
# character_modifier can only credit a fixed vanilla resource (see
# common/domiciles/buildings/_domicile_buildings.info) and Stability is
# a custom variable. The actual amounts now live in
# kehillah_quarterly_pillars_effect (common/scripted_effects/
# kehillah_scripted_effects.txt), gated on has_character_modifier
# checks against these same four keys.
# =====================================================================


# Applied after successfully mediating a dispute between notable
# families. The community runs more smoothly when its quarrels are
# settled -- and settled quarrels are worth standing, not gold.
kehillah_settled_dispute_modifier = {
	icon = diplomacy_positive
	courtier_and_guest_opinion = 10
}


# Applied after distributing tzedakah.
kehillah_tzedakah_modifier = {
	icon = piety_positive
	general_opinion = 5
	monthly_piety = 0.25
}


# Applied when a protection incident is settled badly, or simply
# endured. Deliberately mild: v1's protection layer is flavor and must
# not spiral (spec section 3b -- "no existential stakes"). The real
# consequences belong to Phase 4.
kehillah_shaken_community_modifier = {
	icon = stress_negative
	domicile_monthly_gold_mult = -0.15
}


# Applied when the community faces down a threat on its own terms --
# the Shtadlan talked it down, or the watch stood firm. Confidence.
kehillah_emboldened_community_modifier = {
	icon = prestige_positive
	monthly_prestige = 0.25
}


# Applied when an epidemic clears from the leader's location after a
# physician was in post throughout -- see the recovery-beat detection
# in kehillah_quarterly_pillars_effect (common/scripted_effects/
# kehillah_scripted_effects.txt). UNLIKE the four modifiers above, this
# one is NOT scanned for by kehillah_quarterly_pillars_effect to feed a
# pillar variable -- monthly_prestige here is a native vanilla
# character_modifier field, applied directly by the engine the same way
# it is for any other character_modifier. The Stability reward for this
# same event is a separate, one-time primary_title change_variable call
# at the point of detection, not routed through this modifier at all.
kehillah_epidemic_recovery_modifier = {
	icon = prestige_positive
	monthly_prestige = 0.2
}


# Applied while the community's courtier count exceeds
# kehillah_courtier_capacity_value (Wave 3 item 3, v2 spec's "soft
# courtier/population cap"). Refreshed every quarterly tick in
# kehillah_quarterly_pillars_effect -- added back if still overcrowded,
# removed the moment it is not, so this never lingers past the
# condition that caused it. Flavor/visibility only: the actual Stability
# cost is a direct change_variable call alongside this modifier, not
# routed through it, for the same reason kehillah_epidemic_recovery_
# modifier's own header gives (a character_modifier field cannot credit
# a custom variable).
kehillah_overcrowded_quarter_modifier = {
	icon = opinion_negative
	courtier_and_guest_opinion = -5
	stress_gain_mult = 0.1
}


# A time-limited host grant obtained through the V16 construction
# petition. It is intentionally informational as well as mechanical:
# the shared charter trigger reads its presence before allowing a new
# Jewish Quarter institution to be started.
kehillah_charter_construction_permission_modifier = {
	icon = prestige_positive
}


# =====================================================================
# S2: PER-BAND LEADER MODIFIERS. 3 pillars x 4 bands.
#
# ROADMAP item S2.1. Before this, a pillar band did almost nothing on
# its own: bands gated building tiers, courtier quality, one Crisis
# random event and a map colour, but simply BEING in a band had no
# ongoing effect, so the three headline numbers were mostly scenery.
# These are what make a band felt every month.
#
# Applied to and removed from the LEADER by kehillah_update_pillar_band_
# effect (common/scripted_effects/kehillah_pillar_band_effects.txt) on
# the quarterly tick, which swaps the old band's modifier for the new
# one and tells the player what changed.
#
# FOUR BANDS, NOT FIVE. Healthy is the neutral middle and deliberately
# has NO modifier -- ROADMAP's own "Healthy = small or none". An empty
# modifier definition would be a lie in the character sheet ("Healthy
# Community: nothing"), so Healthy is represented by the absence of any
# of these, and the band-change notice says so in words.
#
# EVERY KEY HERE WAS CHECKED against this build's own
# logs/modifiers.log before use, per S2's instruction: monthly_income_
# mult, stress_gain_mult, courtier_and_guest_opinion, monthly_prestige_
# gain_mult and learning all exist in 1.19.
#
# ONE SUBSTITUTION FROM S2's STARTING PROPOSAL, deliberate. S2 proposed
# Stability drive "influence gain and stress gain". INFLUENCE DOES NOT
# EXIST IN THIS MOD -- government_has_influence was removed 2026-09-07
# (common/governments/kehillah_government.txt) because "Influence reads
# to players as an administrative-government currency", and nothing has
# read it since. Stability instead drives stress_gain_mult (leading a
# community that is coming apart is wearing; leading a settled one is
# not) plus courtier_and_guest_opinion, which has direct precedent in
# this very file: kehillah_settled_dispute_modifier already expresses
# "the community runs more smoothly" as exactly that key. For a Kehillah
# the courtiers ARE the community, so it is the closest thing to a
# communal-mood modifier the engine offers.
#
# THE NUMBERS are intentionally modest and symmetric, and they are the
# first pass -- S10's soak test is the thing that should retune them.
# =====================================================================

# =====================================================================
# SERVING AS THE COMMUNITY'S OWN RABBI -- ROADMAP S4(2), added
# 2026-09-26. Worn for as long as the leader holds both offices; removed
# by kehillah_end_serving_as_own_rabbi_effect.
#
# THE DRAWBACK EXISTS SO THE TOGGLE IS A CHOICE, not free value --
# ROADMAP S4(2) asks for exactly that, "a stress or influence-gain cost
# for doing two jobs". Stress, not influence: Influence was retired from
# this government entirely (common/governments/kehillah_government.txt),
# so an influence cost would be a cost of nothing.
#
# SIZED TO BE FELT, NOT TO PUNISH. +20% stress gain is real pressure over
# a long reign -- a leader doing two jobs wears down -- while the
# scholarship the arrangement represents is acknowledged honestly with a
# small Learning-lifestyle bonus rather than pretending the work has no
# upside. The net is deliberately positive for a genuinely learned
# leader, negative for one who took the office to save the salary, which
# is the decision the toggle is meant to pose.
# =====================================================================

kehillah_serving_as_own_rabbi_modifier = {
	icon = learning_positive
	stress_gain_mult = 0.2
	monthly_learning_lifestyle_xp_gain_mult = 0.1
}
