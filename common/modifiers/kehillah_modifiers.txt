# =====================================================================
# Kehillah character modifiers.
#
# Kept small on purpose. These exist to make decisions and events leave
# a visible trace for a few years, not to become a second progression
# system running alongside the buildings.
#
# Each of the four below also feeds Stability while active -- removed
# from here 2026-09-07 when Influence was retired, since a
# character_modifier can only credit a fixed vanilla resource (see
# common/domiciles/buildings/_domicile_buildings.info) and Stability is
# a custom variable. The actual amounts now live in
# kehillah_quarterly_pillars_effect (common/scripted_effects/
# kehillah_scripted_effects.txt), gated on has_character_modifier
# checks against these same four keys.
# =====================================================================


# Applied after successfully mediating a dispute between notable
# families. The community runs more smoothly when its quarrels are
# settled -- and settled quarrels are worth standing, not gold.
kehillah_settled_dispute_modifier = {
	icon = diplomacy_positive
	courtier_and_guest_opinion = 10
}


# Applied after distributing tzedakah.
kehillah_tzedakah_modifier = {
	icon = piety_positive
	general_opinion = 5
	monthly_piety = 0.25
}


# Applied when a protection incident is settled badly, or simply
# endured. Deliberately mild: v1's protection layer is flavor and must
# not spiral (spec section 3b -- "no existential stakes"). The real
# consequences belong to Phase 4.
kehillah_shaken_community_modifier = {
	icon = stress_negative
	domicile_monthly_gold_mult = -0.15
}


# Applied when the community faces down a threat on its own terms --
# the Shtadlan talked it down, or the watch stood firm. Confidence.
kehillah_emboldened_community_modifier = {
	icon = prestige_positive
	monthly_prestige = 0.25
}


# Applied when an epidemic clears from the leader's location after a
# physician was in post throughout -- see the recovery-beat detection
# in kehillah_quarterly_pillars_effect (common/scripted_effects/
# kehillah_scripted_effects.txt). UNLIKE the four modifiers above, this
# one is NOT scanned for by kehillah_quarterly_pillars_effect to feed a
# pillar variable -- monthly_prestige here is a native vanilla
# character_modifier field, applied directly by the engine the same way
# it is for any other character_modifier. The Stability reward for this
# same event is a separate, one-time primary_title change_variable call
# at the point of detection, not routed through this modifier at all.
kehillah_epidemic_recovery_modifier = {
	icon = prestige_positive
	monthly_prestige = 0.2
}


# Applied while the community's courtier count exceeds
# kehillah_courtier_capacity_value (Wave 3 item 3, v2 spec's "soft
# courtier/population cap"). Refreshed every quarterly tick in
# kehillah_quarterly_pillars_effect -- added back if still overcrowded,
# removed the moment it is not, so this never lingers past the
# condition that caused it. Flavor/visibility only: the actual Stability
# cost is a direct change_variable call alongside this modifier, not
# routed through it, for the same reason kehillah_epidemic_recovery_
# modifier's own header gives (a character_modifier field cannot credit
# a custom variable).
kehillah_overcrowded_quarter_modifier = {
	icon = opinion_negative
	courtier_and_guest_opinion = -5
	stress_gain_mult = 0.1
}
