# =====================================================================
# Book artifact modifiers.
#
# ADDED 2026-09-09 at user request ("an interesting range of possible
# boosts"). Two per genre, applied to whoever OWNS the finished book
# artifact (common/scripted_effects/kehillah_book_effects.txt's
# completion step picks one of the two at random via random_list,
# 50/50) -- not tiered by rarity on purpose: masterwork/famed/
# illustrious already differ in reward magnitude and artifact
# description/name; the MODIFIER is what varies which of two directions
# a given book leans, so two books of the same genre and even the same
# tier do not read identically.
#
# All fields used here are confirmed vanilla character-modifier fields,
# not invented: learning/diplomacy/stewardship, monthly_piety(_gain_
# mult), monthly_prestige, same_faith_opinion, stress_loss_mult,
# attraction_opinion, courtly_opinion are all already used this way
# elsewhere in this mod (kehillah_rabbi_trait's own tracks, common/
# traits/kehillah_traits.txt) or in vanilla's lifestyle_hunter/falconer
# tracks (common/traits/00_traits.txt) this mod's tracks were modeled
# on.
# =====================================================================

# HALAKHA / RESPONSA -- legal-authority flavor: the book is consulted
# as a practical reference, not just read.
kehillah_book_halakha_modifier_1 = {
	stewardship = 1
	diplomacy = 1
	monthly_prestige = 0.2
}
kehillah_book_halakha_modifier_2 = {
	learning = 1
	same_faith_opinion = 5
	monthly_prestige = 0.15
}

# TORAH COMMENTARY -- devotional flavor: the book deepens the owner's
# own standing and piety more than their worldly reach.
kehillah_book_parshanut_modifier_1 = {
	learning = 1
	monthly_piety = 0.3
}
kehillah_book_parshanut_modifier_2 = {
	monthly_piety_gain_mult = 0.1
	same_faith_opinion = 5
}

# TALMUD COMMENTARY -- argumentative flavor: sharpens the owner's own
# reasoning and standing among scholars specifically.
kehillah_book_talmudics_modifier_1 = {
	learning = 1
	diplomacy = 1
}
kehillah_book_talmudics_modifier_2 = {
	learning = 2
	monthly_prestige = 0.15
}

# HEBREW POETRY -- courtly flavor, distinct from the three rabbinic
# genres above: this is read for pleasure and reputation at court, not
# consulted for authority.
kehillah_book_poetry_modifier_1 = {
	attraction_opinion = 5
	courtly_opinion = 5
	monthly_prestige = 0.2
}
kehillah_book_poetry_modifier_2 = {
	stress_loss_mult = 0.05
	same_faith_opinion = 5
	monthly_prestige = 0.15
}
