# =====================================================================
# Jewish holiday character modifiers.
#
# Two, deliberately, reused across every event in
# events/jewish_holiday_events.txt -- the same economy as
# kehillah_modifiers.txt. These are personal, not communal (no
# domicile_* fields: unlike the Kehillah modifiers, these have to work
# on any Jewish character, landed or not, Kehillah leader or not), and
# short (months, not years). A holiday is a mood for a season, not a
# standing condition.
# =====================================================================


# A holiday went well. Reused for every event's best outcome. Icon
# matches the field it actually moves (stress_gain_mult negative, i.e.
# less stress gained), not just a mood picture.
jewish_holiday_warm_memory_modifier = {
	icon = stress_negative
	stress_gain_mult = -0.15
	general_opinion = 5
}


# A holiday went awkwardly -- a mishap, a slight, an argument at the
# table. Never worse than that: see the tone note in
# events/jewish_holiday_events.txt. Reused for every event's worst
# outcome.
jewish_holiday_awkward_memory_modifier = {
	icon = stress_positive
	stress_gain_mult = 0.1
	general_opinion = -3
}
