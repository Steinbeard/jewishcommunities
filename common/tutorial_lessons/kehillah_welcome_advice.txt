# A short tip for a Jewish adventurer. Vanilla's adventurer lesson covers the
# camp and contracts; this adds the one thing it cannot know about.

	chain = reactive_advice
	start_automatically = no
	trigger = {
		has_government = landless_adventurer_government
		is_rabbinic_authority_jewish_trigger = yes
	}

	kehillah_road_advice_step = {
		text = "kehillah_road_advice_desc"
		animation = center
		highlight_widget = "tab_decisions"

		gui_transition = {
			button_id = "read_later"
			button_text = "tutorial_lesson_button_read_later"
			target = lesson_abort
		}
		gui_transition = {
			button_id = "next"
			button_text = "tutorial_lesson_button_complete"
			target = lesson_finish
		}
	}
}
