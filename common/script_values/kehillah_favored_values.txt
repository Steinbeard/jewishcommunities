# =====================================================================
# FAVORED BEYOND THE TEACHING -- the Nagid spec section 3, 2026-10-30.
# Scope: a ruler. How far favoring the Jews goes beyond his faith's
# teaching, as a multiplier on its cost:
#   0.5  Dhimma: the dhimma already deals with each community through a
#        recognised head, so favour is a small stretch
#   1    Witness, or a faith with no teaching on the Jews
#   2    Imperial Law, Separation and Mission, Strict Pact: open defiance
#        (Exclusion and Abolition cannot pass Favored at all)
# A tolerant culture (kehillah_favored_tolerant_culture_trigger) halves it.
# Read by the Favored law's pass cost and by the ruler's modifier tier.
# =====================================================================

kehillah_favored_teaching_factor_value = {
	value = 1
	if = {
		limit = {
			OR = {
				kehillah_teaching_strict_trigger = yes
				kehillah_teaching_exclusionary_trigger = yes
				rite ?= { kehillah_rite_teaching_imperial_law_trigger = yes }
			}
		}
		value = 2
	}
	else_if = {
		limit = { rite ?= { kehillah_rite_teaching_dhimma_trigger = yes } }
		value = 0.5
	}
	if = {
		limit = { kehillah_favored_tolerant_culture_trigger = yes }
		multiply = 0.5
	}
}

# Which of the four modifiers (common/modifiers/kehillah_favored_
# modifiers.txt) a Favored ruler carries: 1 (factor 0.25) to 4 (factor 2).
# 0 when the ruler does not favor the Jews.
kehillah_favored_cost_tier_value = {
	value = 0
	if = {
		limit = { kehillah_ruler_favors_jews_trigger = yes }
		if = {
			limit = { kehillah_favored_teaching_factor_value >= 2 }
			value = 4
		}
		else_if = {
			limit = { kehillah_favored_teaching_factor_value >= 1 }
			value = 3
		}
		else_if = {
			limit = { kehillah_favored_teaching_factor_value >= 0.5 }
			value = 2
		}
		else = {
			value = 1
		}
	}
}
