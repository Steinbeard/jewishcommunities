# =====================================================================
# INVITATIONS -- V34 section 7 (docs/spec/v34-outlook-teachings-and-
# founding.md), 2026-10-13. A ruler with no Jews in his realm may still
# open the door: a yearly "open-door review", driven by his outlook on
# the Jews and his wish to fill towns and markets, can raise his Jewish
# Settlement Policy. Raising it to Encouraged -- by this review, the
# hosts' review or the player's decree -- IS the invitation: Kehillah
# players within diplomatic range hear of it, and for 20 years it draws
# settlers (Send Out Settlers, exiles) and lets a community that is not
# overcrowded answer the call.
# (The historical shape: Bela IV of Hungary after the Mongols, 1251;
# Boleslaw the Pious's Statute of Kalisz, 1264; the Rhineland bishops'
# charters of 1084/1090.)
# =====================================================================

# Chance a year that an eligible ruler (kehillah_open_door_candidate_
# trigger) reviews his policy at all. Low: most years nothing happens,
# and most reviews find no reason to change.
# V44 (2026-10-15): was a flat 5 for every independent Christian or
# Muslim duke within "one France" of any community -- eleven invitations
# in fifteen years in the soak. Now 2, and 3 more for each motive, among
# far fewer candidates. Target: an invitation every three to five years
# world-wide.
kehillah_invitation_review_chance = {
	value = 2
	add = {
		value = kehillah_invitation_motive_value
		multiply = 3
	}
}
# V44: a ruler must be at least well-disposed toward the Jews
# (kehillah_outlook_band_value 1: outlook 15 or more) to invite them.
kehillah_invitation_min_outlook = 15

# Scope: a ruler. The concrete reasons to invite Jews (V44, 2026-10-15,
# Daniel: "rulers who don't have Jewish communities yet but have
# neighbors with Jewish communities ... a very high outlook ... their
# economic focus is increasing development or they're short on cash"):
# - a stewardship lifestyle aimed at wealth or the domain;
# - short of cash;
# - an able and driven steward (stewardship 14+ and diligent, ambitious
#   or greedy);
# - refugees at the border: a neighbour has just expelled its Jews
#   (kehillah_open_door_refugees_flag, kehillah_expulsion_proclaim_effect).
# 0..3. A candidate needs at least one.
kehillah_invitation_motive_value = {
	value = 0
	if = {
		limit = {
			OR = {
				has_focus = stewardship_wealth_focus
				has_focus = stewardship_domain_focus
			}
		}
		add = 1
	}
	if = {
		limit = { gold < kehillah_invitation_short_of_cash_gold }
		add = 1
	}
	if = {
		limit = {
			stewardship >= 14
			OR = {
				has_trait = diligent
				has_trait = ambitious
				has_trait = greedy
			}
		}
		add = 1
	}
	if = {
		limit = { has_character_flag = kehillah_open_door_refugees_flag }
		add = 2
	}
	max = 3
}
# V44: "short of cash" for the motive above.
kehillah_invitation_short_of_cash_gold = 50

# Scope: a ruler. The hosts' review's net pressure (kehillah_expulsion_
# values.txt; negative = welcoming), less the motive above. It must
# reach kehillah_policy_review_raise_threshold_negative (-4) to raise.
kehillah_open_door_net_pressure_value = {
	value = kehillah_policy_net_pressure_value
	subtract = kehillah_invitation_motive_value
}

# A community founded in a realm whose lord has an open invitation starts
# with the privileges he promised (kehillah_invitation_answered_effect).
kehillah_invitation_answered_prosperity = 60
kehillah_invitation_answered_stability = 30
# How much the inviting lord warms to the Jews when settlers come.
kehillah_invitation_answered_outlook = 5

# Seek an Invitation (task_kehillah_seek_invitation): at an audience the
# lord raises his policy if his open-door net pressure is at or below
# this -- the hosts' -4, less 2 for the embassy's arguments.
kehillah_invitation_embassy_threshold = -2

# Scope: a ruler; scope:kehillah_suitor_leader saved. Which lord the
# Shtadlan courts first: outlook on the Jews, plus half his opinion of
# the community's leader (live 2026-10-13: he went to Svend of Denmark,
# whose opinion of the leader was -74).
kehillah_invitation_suitor_score_value = {
	value = kehillah_outlook_on_jews_value
	add = {
		value = "opinion(scope:kehillah_suitor_leader)"
		divide = 2
	}
}

# V44 (2026-10-15): how long a neighbour's expulsion counts as a reason to
# invite the refugees (kehillah_open_door_refugees_flag).
kehillah_open_door_refugees_years = 3
# V44 (2026-10-15): a player community hears a lord's letter as an event at
# most once in this many years; other invitations are feed news.
kehillah_invitation_letter_years = 10
# V44: thanking the inviting lord ("Write to thank ..."): his opinion.
kehillah_invitation_thanks_opinion = 15
# V44: the yearly chance a Jewish adventurer near an open invitation sets
# out for the inviting realm.
kehillah_invitation_adventurer_chance = 25
