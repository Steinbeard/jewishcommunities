# The Agitation race (docs/spec/2026-11-01-the-attack-as-a-race.md). Root = the
# kehillah_community_crisis situation unless noted. Each catalyst's points are one of these.
# Provisional balance (2026-11-01): an unanswered warning reaches Armed in about three months;
# the four preparations alone are worth about half of Spared, so protection is needed to win.

kehillah_crisis_backstop_days = 365
# A transition an event already decided (the host rose, the preacher was bought off).
kehillah_crisis_decided_points = 100
# Console-only pushes (kehillah_crisis_debug.6/.7).
kehillah_crisis_debug_points = 50

# --- Toward the Armed host ---

# The preaching, monthly: half the title's agitation (45 at the start, +3 a week, at most 95).
kehillah_crisis_preaching_points_value = {
 value = 1
 if = {
  limit = { exists = var:kehillah_crisis_title.var:kehillah_crisis_agitation }
  value = var:kehillah_crisis_title.var:kehillah_crisis_agitation
  divide = 2
  round = yes
 }
 min = 1
}
# Once, as the race opens: the crowd is already stirred.
kehillah_crisis_opening_points_value = kehillah_crisis_preaching_points_value

# Monthly while the sovereign is Cold (8) or Hostile (15) toward the Jews.
kehillah_crisis_cold_lord_points_value = {
 value = 8
 if = {
  limit = { var:kehillah_crisis_sovereign ?= { kehillah_outlook_band_value <= -2 } }
  value = 15
 }
}
# Monthly while the county's or the sovereign's faith wages a great holy war.
kehillah_crisis_holy_war_points_value = 10

# --- Toward Spared ---

# Monthly, from protection secured in the elders' scenes: half of what each once took off the
# old outbreak roll (royal protection 5, condemnation 8, the lord's men 8, arms 5, refuge 3).
kehillah_crisis_protection_points_value = {
 value = 0
 if = { limit = { var:kehillah_crisis_title ?= { has_variable = kehillah_crisis_royal_protection } } add = 5 }
 if = { limit = { var:kehillah_crisis_title ?= { has_variable = kehillah_mob_condemned } } add = 8 }
 if = { limit = { var:kehillah_crisis_title ?= { has_variable = kehillah_mob_men_by } } add = 8 }
 if = { limit = { var:kehillah_crisis_title ?= { has_variable = kehillah_mob_armed } } add = 5 }
 if = { limit = { var:kehillah_crisis_title ?= { has_variable = kehillah_mob_refuge } } add = 3 }
}
# Monthly from a Protective (8) or Well-disposed (4) sovereign.
kehillah_crisis_goodwill_points_value = {
 value = 4
 if = {
  limit = { var:kehillah_crisis_sovereign ?= { kehillah_outlook_band_value >= 2 } }
  value = 8
 }
}
# Once, as each preparation is finished.
kehillah_crisis_watch_points = 15
kehillah_crisis_supplies_points = 10
kehillah_crisis_routes_points = 5
kehillah_crisis_advocacy_points = 20

# --- Shown in the window and in Military ---

# What the next monthly pulse adds to the Armed host from the preaching alone. 0 outside the
# race, which hides the window's pressure line.
kehillah_crisis_next_push_value = {
 value = 0
 if = {
  limit = {
   situation_current_phase = kehillah_crisis_warning
   var:kehillah_crisis_title ?= { var:kehillah_crisis_stage ?= 1 }
  }
  value = kehillah_crisis_preaching_points_value
 }
}
