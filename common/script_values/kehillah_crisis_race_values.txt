# The Agitation race (docs/spec/2026-11-01-the-attack-as-a-race.md). Root = the
# kehillah_community_crisis situation unless noted. Each catalyst's points are one of these.
# Provisional balance (2026-11-01): an unanswered warning reaches Armed in about three months;
# the four preparations alone are worth about half of Spared, so protection is needed to win.

kehillah_crisis_backstop_days = 365

kehillah_crisis_decided_points = 100

# --- Toward the Armed host ---

# The preaching, monthly: half the title's agitation (45 at the start, +3 a week, at most 95).
# kehillah_crisis_store_push_effect copies it onto the situation as var:kehillah_crisis_push just
# before each catalyst that uses it. Live test 2026-11-01: reading the title's variable through
# var:kehillah_crisis_title here added no points, while a situation's own variable works (spike).
kehillah_crisis_preaching_points_value = {
 value = 1
 if = {
  limit = { exists = var:kehillah_crisis_push }
  value = var:kehillah_crisis_push
 }
 min = 1
}
# Once, as the race opens: the crowd is already stirred. (Not an alias: CK3 rejects a script value
# that names another one directly, "referencing a named formula".)
kehillah_crisis_opening_points_value = { value = kehillah_crisis_preaching_points_value }

# Monthly while the sovereign is Cold (8) or Hostile (15) toward the Jews.
kehillah_crisis_cold_lord_points_value = {
 value = 8
 if = {
  limit = { var:kehillah_crisis_sovereign ?= { kehillah_outlook_band_value <= -2 } }
  value = 15
 }
}
# Monthly while the county's or the sovereign's faith wages a great holy war.
kehillah_crisis_holy_war_points_value = 10

# old outbreak roll (royal protection 5, condemnation 8, the lord's men 8, arms 5, refuge 3).

# --- Shown in the window and in Military ---

# What the next monthly pulse adds to the Armed host from the preaching alone. 0 outside the
# race, which hides the window's pressure line.

}
