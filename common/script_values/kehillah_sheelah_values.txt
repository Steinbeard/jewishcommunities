# =====================================================================
# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs
# a gift to the answering community and is limited by a cooldown; answering
# is the reward of Greatness. Content: the Bet Din docket's 14 cases
# (kehillah_din_letter_N, kehillah_din_answer_N, kehillah_din_answer_careful_N).
# =====================================================================

# Days before a community may send another she'elah (2026-11-23, Daniel:
# one year, was three).
kehillah_sheelah_cooldown_days = 365
# Days a posek holds one letter; he takes no other meanwhile.
kehillah_sheelah_pending_days = 120

# The cost of asking (2026-11-23, Daniel: prestige, not gold): to send a
# question is to admit another community knows better. Vanilla's minor
# prestige value; a plain number, since an alias value (a = b) is rejected.
kehillah_sheelah_prestige_cost = 75

# Greatness to the posek.
kehillah_sheelah_answer_greatness = 10
kehillah_sheelah_careful_greatness = 20
kehillah_sheelah_decline_greatness = -10
# Stability to the asking community, by the reply.
kehillah_sheelah_answer_stability = 10
kehillah_sheelah_careful_stability = 20

# Yearly chance (percent) that an AI community below Flourishing Greatness
# sends a she'elah.
kehillah_sheelah_ai_chance = 20

# ---------------------------------------------------------------------
# Ask for Help with Your Studies (2026-11-22, Daniel). A student of a work
# writes to a rabbi who has mastered it.
# ---------------------------------------------------------------------
# Days between requests.
kehillah_study_help_cooldown_days = 365
# Days a rabbi holds one request.
kehillah_study_help_pending_days = 90
# The gift to the rabbi: smaller than a community's she'elah.
kehillah_study_help_gift_value = {
	value = 10
	multiply = kehillah_aid_era_value
	round = yes
}
# Scheme progress an explanation gives the current phase. Shown in game as
# days of progress (live 2026-11-22); a phase needs about 240
# (base_progress_goal), so 60 is about a quarter.
kehillah_study_help_progress = 60
# A rabbi who can write a full explanation: Learning at least this.
kehillah_study_help_commentary_learning = 15

# The letter cap (Daniel: do not overwhelm players). A player posek gets at
# most one letter event in this many days; letters beyond it are answered by
# the court on its own and reported as notifications (2026-11-23).
kehillah_sheelah_player_letter_gap_days = 365

# =====================================================================
# SLICE 2: "follows the rulings of" (2026-11-22). Spec section 5 and 12.
# One tie per community, on the asking title: var:kehillah_sheelah_follows
# (the followed title) and var:kehillah_sheelah_follows_strength. Untuned
# guesses, for a soak run: an AI community asks about once in eight years,
# so a tie must outlast that gap to grow.
# =====================================================================
# Strength an answered letter adds; a letter answered from the sources adds more.
kehillah_sheelah_tie_answer_gain = 3
kehillah_sheelah_tie_careful_gain = 5
kehillah_sheelah_tie_max = 10
# Yearly fade (one point every four years).
kehillah_sheelah_tie_fade = 0.25
# A strong tie: its follower may take up the followed community's rite.
kehillah_sheelah_tie_strong = 6

# Scope: a Kehillah leader. Other communities whose tie runs to ours. prev,
# not root, inside the list (see kehillah_yeshiva_seated_alumni_value).
kehillah_sheelah_follower_count_value = {
	value = 0
	if = {
		limit = { exists = primary_title }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				NOT = { this = prev.primary_title }
				exists = holder
				var:kehillah_sheelah_follows ?= prev.primary_title
			}
			add = 1
		}
	}
}
kehillah_sheelah_follower_cap = 5
kehillah_sheelah_follower_counted_value = {
	value = kehillah_sheelah_follower_count_value
	max = kehillah_sheelah_follower_cap
}
# Greatness baseline per follower (a seated alumnus gives 25, at most 3).
kehillah_sheelah_follower_greatness_each = 10
kehillah_bd_follower_greatness_value = {
	value = kehillah_sheelah_follower_counted_value
	multiply = kehillah_sheelah_follower_greatness_each
}
# Each season, extra percent chance that a travelling student arrives at
# our yeshiva, per follower: followers send their students to us.
kehillah_sheelah_follower_arrival_each = 3
kehillah_sheelah_follower_arrival_value = {
	value = kehillah_sheelah_follower_counted_value
	multiply = kehillah_sheelah_follower_arrival_each
}
# Extra percent chance that an AI follower answers our mutual-aid appeal
# (beside the Greatness term, -15 to +20).
kehillah_aid_follower_chance = 25
# AI posek choice: weight for the community already followed, and for the
# yeshiva the asker's leader or Dayyan studied at.
kehillah_sheelah_ai_tie_weight = 3
kehillah_sheelah_ai_alma_mater_weight = 2
# 2026-11-23 (Daniel): a community it has written to before (posek title's
# var list kehillah_sheelah_written_by), a leader who holds semicha, and a
# leader whose lifestyle focus is Scholarship or Theology.
kehillah_sheelah_ai_wrote_before_weight = 1
kehillah_sheelah_ai_rabbi_leader_weight = 1
kehillah_sheelah_ai_scholar_focus_weight = 1

kehillah_sheelah_rite_adopt_stability = -10

kehillah_sheelah_rite_adopted_greatness = 10
# A follower's Synod vote with its followed community (an AI weight).
kehillah_sheelah_synod_follow_weight = 100
