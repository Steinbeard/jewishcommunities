# =====================================================================
# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs
# a gift to the answering community and is limited by a cooldown; answering
# is the reward of Greatness. Content: the Bet Din docket's 14 cases
# (kehillah_din_letter_N, kehillah_din_answer_N, kehillah_din_answer_careful_N).
# =====================================================================

# Days before a community may send another she'elah.
kehillah_sheelah_cooldown_days = 1095
# Days a posek holds one letter; he takes no other meanwhile.
kehillah_sheelah_pending_days = 120

# The gift that goes with the letter, to the answering community: the
# geonic academies lived partly on such gifts. Era-scaled (mutual aid's
# kehillah_aid_era_value: 1 early, 1.5 high, 3 late medieval).
kehillah_sheelah_gift_value = {
	value = 25
	multiply = kehillah_aid_era_value
	round = yes
}

# Greatness to the posek.
kehillah_sheelah_answer_greatness = 10
kehillah_sheelah_careful_greatness = 20
kehillah_sheelah_decline_greatness = -10
# Stability to the asking community, by the reply.
kehillah_sheelah_answer_stability = 10
kehillah_sheelah_careful_stability = 20

# Yearly chance (percent) that an AI community below Flourishing Greatness
# sends a she'elah.
kehillah_sheelah_ai_chance = 20

# ---------------------------------------------------------------------
# Ask for Help with Your Studies (2026-11-22, Daniel). A student of a work

# Days between requests.
kehillah_study_help_cooldown_days = 365
# Days a rabbi holds one request.
kehillah_study_help_pending_days = 90
# The gift to the rabbi: smaller than a community's she'elah.
kehillah_study_help_gift_value = {
	value = 10
	multiply = kehillah_aid_era_value
	round = yes
}
# Scheme progress an explanation gives the current phase. Shown in game as
# days of progress (live 2026-11-22); a phase needs about 240
# (base_progress_goal), so 60 is about a quarter.
kehillah_study_help_progress = 60
# A rabbi who can write a full explanation: Learning at least this.
kehillah_study_help_commentary_learning = 15

# The letter cap (Daniel: do not overwhelm players). An AI community writes
# to a player posek at most once in this many days.
kehillah_sheelah_player_letter_gap_days = 365
