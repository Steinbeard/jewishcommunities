# =====================================================================
# THE SEAT OF THE REALM'S JEWS -- the Nagid spec section 4, 2026-10-30.
# Scope of everything here: a Kehillah leader. The seat terms are about
# the court, not the market: county development already pays a major
# city its Prosperity (kehillah_bd_urban_opportunity_prosperity_value).
# Starting values, to tune after a live look.
# =====================================================================

# 1 duchy, 2 kingdom, 3 empire or higher: the tier of the realm whose
# seat this community is. 0 when it is not a seat.
kehillah_seat_tier_value = {
	value = 0
	if = {
		limit = { kehillah_is_realm_seat_trigger = yes }
		domicile.domicile_location.county.holder.top_liege = {
			if = {
				limit = { highest_held_title_tier >= tier_empire }
				value = 3
			}
			else_if = {
				limit = { highest_held_title_tier >= tier_kingdom }
				value = 2
			}
			else = {
				value = 1
			}
		}
	}
}

# Prosperity: the court's demand -- purchases, loans, physicians,
# translators. +5 / +10 / +15.
kehillah_bd_seat_prosperity_value = {
	value = kehillah_seat_tier_value
	multiply = 5
}

# Greatness: patronage, and scholars drawn to court. +5 / +10 / +15.
kehillah_bd_seat_greatness_value = {
	value = kehillah_seat_tier_value
	multiply = 5
}

# Stability: the lord's own Jews, outside the capital. +5.
kehillah_bd_lords_own_stability_value = {
	value = 0
	if = {
		limit = { kehillah_in_lords_own_lands_trigger = yes }
		value = 5
	}
}

# Danger: at the ruler's door -- court intrigue and the capital's crowd
# (Granada, 1066). +5 for a duchy's seat, +10 for a kingdom's or an
# empire's. Added to kehillah_danger_value.
kehillah_danger_seat_value = {
	value = 0
	if = {
		limit = { kehillah_seat_tier_value >= 2 }
		value = 10
	}
	else_if = {
		limit = { kehillah_seat_tier_value = 1 }
		value = 5
	}
}
