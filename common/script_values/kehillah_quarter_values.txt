# The specialisations the community runs: the Kosher Market, Craft Row,
# Bankers' Row, the Funduq, Wall and Gates.
kehillah_specialisation_count_value = {
	value = 0
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_market_stalls_01 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_workshops_01 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_bankers_row_01 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_funduq_01 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_01 } }
		add = 1
	}
}

# How many it may run: one per Kahal House tier (the countinghouse keys),
# and one more at Legendary Prosperity. At most four of the five.
kehillah_specialisation_allowance_value = {
	value = 0
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_countinghouse_01 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_countinghouse_02 } }
		add = 1
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_countinghouse_03 } }
		add = 1
	}
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_legendary_threshold } }
		add = 1
	}
	max = 4
}

# Petition for a Wall: the prestige the community spends asking the host.
kehillah_wall_petition_prestige_value = 150
# Years before a refused or lapsed petition can be made again.
kehillah_wall_petition_cooldown_years = 5

# The danger shield of the quarter's gates and walls (a negative danger term,
# kehillah_danger_value): gates 5, a wall 10 more, a fortified quarter 10 more.
kehillah_danger_walls_value = {
	value = 0
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_01 } }
		add = -5
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_02 } }
		add = -10
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_03 } }
		add = -10
	}
}
