# Pillars spec R7 (2026-11-20). Scope: a Kehillah leader. Days added to a
# local agitator's preaching before the mob comes (the warning window): a
# stable community notices sooner. Crisis -30, Strained 0, Healthy +30,
# Flourishing +45, Legendary +60; the Watch House (Wall and Gates) +15.
kehillah_stability_warning_days_value = {
	value = -30
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_strained_threshold } }
		value = 0
	}
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold } }
		value = 30
	}
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_flourishing_threshold } }
		value = 45
	}
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_legendary_threshold } }
		value = 60
	}
	if = {
		limit = { domicile ?= { has_domicile_parameter = kehillah_has_watch_house } }
		add = 15
	}
}
