# The town's market (2026-11-21, Quarter spec section 16). Scope: a Kehillah
# leader whose community has its quarter district in the town.
#
# 2026-11-23 (Daniel: the penalties felt harsh): a bonus only. Each of the
# quarter's needs the market supplies adds a little Affluence: comforts
# and books (categories, as vanilla districts ask), wine for Kiddush, and,
# for a community that keeps kashrut, cattle for its shochet
# (docs/spec/2026-11-23-kashrut.md). A missing good costs nothing: which
# goods a town's market carries is geography, not something a community can
# act on. It was -15 per missing good, which left Worms at -45 at the 1066
# start. Textiles were dropped the same day: the quarter makes no cloth.
#
# 1.21's market needs (trade_category_unfulfilled_need,
# trade_good_unfulfilled_need) include the district's own, so an unfulfilled
# need means the market lacks the goods.
kehillah_market_shortfall_threshold = 1
kehillah_market_supplied_each = 5
kehillah_bd_market_supply_prosperity_value = {
	value = 0
	if = {
		limit = {
			exists = domicile.domicile_location.barony.trade_market
			domicile.domicile_location = { province_has_district = kehillah_jewish_quarter_district }
		}
		domicile.domicile_location.barony.trade_market = {
			save_temporary_scope_as = kehillah_market
		}
		if = {
			limit = {
				scope:kehillah_market = {
					NOT = { trade_category_unfulfilled_need = { target = trade_good_category:trade_category_comforts value >= kehillah_market_shortfall_threshold } }
				}
			}
			add = kehillah_market_supplied_each
		}
		if = {
			limit = {
				scope:kehillah_market = {
					NOT = { trade_category_unfulfilled_need = { target = trade_good_category:trade_category_books value >= kehillah_market_shortfall_threshold } }
				}
			}
			add = kehillah_market_supplied_each
		}
		if = {
			limit = {
				scope:kehillah_market = {
					NOT = { trade_good_unfulfilled_need = { target = trade_good_type:trade_good_wine value >= kehillah_market_shortfall_threshold } }
				}
			}
			add = kehillah_market_supplied_each
		}
		if = {
			limit = {
				kehillah_keeps_kashrut_trigger = yes
				scope:kehillah_market = {
					NOT = { trade_good_unfulfilled_need = { target = trade_good_type:trade_good_cattle value >= kehillah_market_shortfall_threshold } }
				}
			}
			add = kehillah_market_supplied_each
		}
	}
}
