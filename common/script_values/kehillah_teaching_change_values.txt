# Jews: by the lord's teaching.

	if = {
		limit = { kehillah_teaching_strict_trigger = yes }
		value = 25

		limit = { kehillah_teaching_exclusionary_trigger = yes }
		value = 40
	}
}

# --- V38 slice 2: missionizing. Scope: a Kehillah leader.
# Quarterly chance (percent) of a disputation: Christian hosts only.
kehillah_disputation_chance_value = {
	value = 0
	if = {
		limit = {
			domicile.domicile_location.county.holder.top_liege ?= {
				faith.religion = religion:christianity_religion
			}
		}
		value = 0.25
		if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { kehillah_teaching_strict_trigger = yes } }
			value = 1.5
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { kehillah_teaching_exclusionary_trigger = yes } }
			value = 1
		}
		if = {
			limit = {
				domicile.domicile_location.county.holder.top_liege.faith = {
					has_variable = kehillah_faith_pressure
					var:kehillah_faith_pressure >= kehillah_faith_pressure_head_mood_threshold
				}
			}
			add = 0.75
		}
	}
}
kehillah_disputation_cooldown_years = 15
