# =====================================================================
# Tuning for the mechanics of the seven Sefer tenets outside the book
# system (2026-11-08; docs/spec/2026-11-08-sefer-tenet-mechanics.md).
# A rite holding the tenet as core and a character holding it as a
# personal tenet each give the bonus once; they stack.
# =====================================================================

# Pesak: added to kehillah_judicial_learning_value (Learning units), which
# every Bet Din case, dispute ruling and responsum reads. +2 is a Talmudist
# of 30 XP; core and personal together equal one of 65.
kehillah_tenet_pesak_judicial_bonus = 2

# Hokhmah: added to the Shtadlan's petition multiplier
# (kehillah_hashkafa_petition_mult_value), the same step as one band of the
# leader's Hashkafa.
kehillah_tenet_hokhmah_petition_bonus = 0.25

# Emunah Temimah: a sermon also steadies the community (a minor change; the
# sermon's own pillar gain is 10-25), and a preacher who holds it personally
# gains more Piety than the sermon's 25.
kehillah_tenet_emunah_sermon_stability = 10
kehillah_tenet_emunah_sermon_piety = 15

# Pilpul: yeshiva fame each quarter, the same size as a master head's.
# Offsets the stable decay (1) on its own.
kehillah_tenet_pilpul_yeshiva_fame = 1

# Minhag Avoteinu: a festival feast hosted by a Kehillah leader steadies the
# community (a minor change).
kehillah_tenet_minhag_feast_stability = 10

# Avelei Tziyon: Send Alms to Jerusalem.
kehillah_tenet_zion_alms_gold = { value = minor_gold_value }
kehillah_tenet_zion_alms_greatness = 10
kehillah_tenet_zion_alms_fame = 3
