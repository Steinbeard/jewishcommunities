# Host Charter contributors (V16 first implementation slice).
#
# The contract flag, not the host's future settlement-policy forecast,
# is the source of these values. This is deliberately the legacy V15
# charter's moneylending/walled-quarter shape: V16 replaces those rows
# only after its contract migration spike is live-proven. Keeping this
# adapter small gives that spike one clear question: does an actual
# tributary-contract flag read back into a quarterly pillar baseline?
#
# Scope: the community leader / tributary. The numbers are modest next
# to a building tier (and zero at V15's default rungs) so a charter is a
# meaningful structural condition, not a replacement for development of
# the Jewish Quarter. V16 §5 makes these source values reusable by the
# eventual projected-charter outlook as well as the actual charter.
# ---------------------------------------------------------------------
kehillah_bd_charter_economic_privileges_value = {
	value = 0
	if = {
		limit = { vassal_contract_has_flag = kehillah_moneylending_full_granted }
		add = 45
	}
	else_if = {
		limit = { vassal_contract_has_flag = kehillah_moneylending_pawnbroking_granted }
		add = 20
	}
}

# Nearby-community Goldilocks contribution. The count is cached on the
# title by kehillah_refresh_settlement_network_effect, not recomputed
# during baseline evaluation. A nearby peer network is a Stability
# resource; isolation hurts, but an overly dense local cluster provides
# less marginal support rather than an ever-growing bonus.
kehillah_bd_settlement_network_value = {
	value = -5
	if = {
		limit = {
			primary_title = {
				has_variable = kehillah_nearby_community_count
				var:kehillah_nearby_community_count >= 5
			}
		}
		add = 5
	}
	else_if = {
		limit = {
			primary_title = {
				has_variable = kehillah_nearby_community_count
				var:kehillah_nearby_community_count >= 3
			}
		}
		add = 12
	}
	else_if = {
		limit = {
			primary_title = {
				has_variable = kehillah_nearby_community_count
				var:kehillah_nearby_community_count >= 1
			}
		}
		add = 20
	}
}

# ---------------------------------------------------------------------
# Drift this season: what BASELINE CONVERGENCE will add next quarterly
# tick -- (baseline - current) * kehillah_convergence_rate_value, the
# exact expression kehillah_quarterly_pillars_effect applies. Signed;
# the _abs twins exist because the tooltip says "rising by" / "falling
# by" a positive number rather than printing a minus sign.
# ---------------------------------------------------------------------
kehillah_bd_prosperity_drift_value = {
	value = kehillah_prosperity_baseline_value
	if = {
		limit = { primary_title = { has_variable = kehillah_var_prosperity } }
		subtract = primary_title.var:kehillah_var_prosperity
	}
	multiply = kehillah_convergence_rate_value
}
kehillah_bd_prosperity_drift_abs_value = {
	value = kehillah_bd_prosperity_drift_value
	if = {
		limit = { kehillah_bd_prosperity_drift_value < 0 }
		multiply = -1
	}
}
