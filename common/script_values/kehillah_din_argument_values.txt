# 2026-11-25 (Daniel): a prepared court. The Dayyan's Prepare the Bet Din
# task, on the sitting's host, adds to every judge's argument chance
# (kehillah_bet_din_prepared_points_value, kehillah_council_script_values.txt).
# Scope: a judge.

	value = 0
	involved_activity ?= {
		if = {

		}
	}
}
