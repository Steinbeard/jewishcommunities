# Routes (or, without Silk and Silver, ventures) the Affluence band grants:
# Healthy 1, Flourishing and Legendary 2. The band is the one the band
# effect just wrote (kehillah_band_now_prosperity: 0 Crisis .. 4 Legendary).
kehillah_trade_band_slots_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_band_now_prosperity } }
		if = {
			limit = { primary_title.var:kehillah_band_now_prosperity >= 3 }
			value = 2
		}
		else_if = {
			limit = { primary_title.var:kehillah_band_now_prosperity >= 2 }
			value = 1
		}
	}
}

# Ventures, for the no-DLC gold: the band's slots plus the Commercial
# style's own route.
kehillah_trade_venture_slots_value = {
	value = 0
	if = {
		limit = { has_character_modifier = kehillah_trade_ventures_1_modifier }
		value = 1
	}
	else_if = {
		limit = { has_character_modifier = kehillah_trade_ventures_2_modifier }
		value = 2
	}
	if = {
		limit = { has_character_modifier = kehillah_style_commercial_modifier }
		add = 1
	}
}

# Chance (percent) a quarter that one venture pays: 15 per venture. One
# venture pays about every 20 months; three, about twice a year.
kehillah_trade_venture_chance_per_slot = 15
kehillah_trade_venture_chance_value = {
	value = kehillah_trade_venture_slots_value
	multiply = kehillah_trade_venture_chance_per_slot
}

# What a venture brings home: half of vanilla's era-scaled medium gold.
# FIRST PASS, uncalibrated against what a real route earns; see the spec's
# calibration section.
kehillah_trade_venture_gold_value = {
	value = medium_gold_value
	multiply = 0.5
	round = yes
}
