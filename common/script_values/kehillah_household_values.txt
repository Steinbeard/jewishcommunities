# =====================================================================
# Ba'alei batim (V26 part 2): the community's householder families.
# docs/spec/v26-kehillah-council-and-baalei-batim.md
#
# A household is a dynasty house registered on the community title's
# kehillah_households list. kehillah_household_refresh_effect writes the
# inputs below onto each house; these values only read them.
# =====================================================================

# Scope: a household (dynasty house). Standing is a household's weight in
# the community: its head's ability, its people, its seats on the board,
# and how long it has been here. Shown as a band with the number in the
# tooltip (V26 default).
kehillah_household_standing_value = {
	value = 0
	# Head's ability: half the sum of the three civic skills (about 0-30).
	if = {
		limit = { exists = var:kehillah_household_head }
		add = {
			value = var:kehillah_household_head.diplomacy
			add = var:kehillah_household_head.stewardship
			add = var:kehillah_household_head.learning
			divide = 2
		}
	}
	# People: 5 per adult member living in the community, at most 20.
	if = {
		limit = { exists = var:kehillah_household_adults }
		add = {
			value = var:kehillah_household_adults
			multiply = 5
			max = 20
		}
	}
	# Board seats held by members.
	if = {
		limit = { exists = var:kehillah_household_seats }
		add = {
			value = var:kehillah_household_seats
			multiply = 15
		}
	}
	# Years in the community, half a point each, at most 20.
	if = {
		limit = { exists = var:kehillah_household_years }
		add = {
			value = var:kehillah_household_years
			divide = 2
			max = 20
		}
	}
	round = yes
}

kehillah_household_standing_respected_threshold = 25
kehillah_household_standing_leading_threshold = 45

# Scope: a community title. How many households it can hold: 2, plus one
# at Strained Prosperity and one more at Healthy (V26: 2-4, scaled by the
# community's size). Was Healthy/Flourishing until the 2026-10-09 live
# test: Worms opened with its three founding families over a cap of 2,
# so a young community would shed a family for no reason in play.
kehillah_household_cap_value = {
	value = 2
	if = {
		limit = { has_variable = kehillah_var_prosperity var:kehillah_var_prosperity >= kehillah_band_strained_threshold }
		add = 1
	}
	if = {
		limit = { has_variable = kehillah_var_prosperity var:kehillah_var_prosperity >= kehillah_band_healthy_threshold }
		add = 1
	}
}

# Scope: a community title.
kehillah_household_count_value = {
	value = 0
	every_in_list = {
		variable = kehillah_households
		add = 1
	}
}

# Yearly chances, in percent.
kehillah_household_arrival_chance = 30
kehillah_household_departure_chance = 30
# A community above its cap (Prosperity fell) loses a household more slowly.
kehillah_household_overcap_departure_chance = 15

# What persuading a household to stay costs.
kehillah_household_persuade_gold_value = {
	value = medium_gold_value
}

# Scope: a candidate destination or origin community title, with
# scope:kh_here_location set. Nearby communities are far likelier.
kehillah_household_distance_weight_value = {
	value = 1
	if = {
		limit = {
			exists = holder.domicile.domicile_location
			holder.domicile.domicile_location = {
				squared_distance = { target = scope:kh_here_location value <= squared_distance_medium }
			}
		}
		add = 20
	}
	else_if = {
		limit = {
			exists = holder.domicile.domicile_location
			holder.domicile.domicile_location = {
				squared_distance = { target = scope:kh_here_location value <= squared_distance_large }
			}
		}
		add = 5
	}
}
