# Scope: a CHARACTER (the domicile's owner). 1 if their quarter has a
# Beit Midrash of any tier, else 0. The library panel is visible exactly
# when this is 1 -- the Beit Midrash IS the library; no building, no
# library view. Read from gui/kehillah_domicile_library.gui as
# Character.MakeScope.ScriptValue( 'kehillah_has_beit_midrash_value' ),
# the idiom the strip and roster already use for script values.
#
# has_domicile_building_or_higher is a DOMICILE-scope trigger, so it
# sits inside domicile ?= { } -- the exact shape kehillah_bd_beit_
# midrash_value (common/script_values/kehillah_breakdown_values.txt)
# uses, after the baseline bug v12 section 8 records.
# =====================================================================

kehillah_has_beit_midrash_value = {
	value = 0
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_beit_midrash_01 } }
		value = 1
	}
}
