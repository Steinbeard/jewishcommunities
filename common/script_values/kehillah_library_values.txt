# =====================================================================
# THE COMMUNITY LIBRARY -- values the GUI reads. ADDED 2026-09-15 (docs/
# spec/v13-domicile-library-panel.md). The library itself is the
# kehillah_library_works variable_list on each community's PRIMARY TITLE
# (common/scripted_triggers/kehillah_scripted_triggers.txt's header for
# the design; common/scripted_effects/kehillah_library_effects.txt for
# the seed). A .gui file cannot call a scripted trigger and cannot ask
# "is flag X in this title's list", but it CAN evaluate a script value
# on any scope (Title.MakeScope.ScriptValue( 'name' ), the exact idiom
# vanilla's activity debate window uses for debate_outcome_value), so
# the twelve yes/no questions the panel asks are twelve 0/1 values here.
#
# Two families, deliberately on different scopes, mirroring where the
# data actually lives:
#   kehillah_lib_<work>_value   scope: a TITLE (the community). 1 if the
#                               community's library holds the work.
#   kehillah_read_<work>_value  scope: a CHARACTER. 1 if that person has
#                               personally studied it (kehillah_studied_
#                               works, which belongs to the scholar, not
#                               the community -- see the library effects
#                               file's own header for why).
# is_target_in_variable_list on a title with no such list is simply
# false -- no "failed to fetch" storm the way an unguarded var: read
# would be (v12 section 6g), so no has_variable guard is needed here.
# =====================================================================

# Targum Onkelos
kehillah_lib_targum_onkelos_value = {
	value = 0
	if = {
		limit = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_targum_onkelos } }
		value = 1
	}
}
kehillah_read_targum_onkelos_value = {
	value = 0
	if = {
		limit = { is_target_in_variable_list = { name = kehillah_studied_works target = flag:work_targum_onkelos } }
		value = 1
	}
}
