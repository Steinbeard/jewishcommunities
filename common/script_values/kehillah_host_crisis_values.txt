# --- 1. Protection during violence. Scope: a Kehillah leader. Multiplies
# the harm of war, crusade and risings: the host's outlook band, then the
# charter's community-security term.
kehillah_host_protection_mult_value = {
	value = 1
	if = {
		limit = { kehillah_host_outlook_value >= 40 }
		value = 0.5
	}
	else_if = {
		limit = { kehillah_host_outlook_value >= 15 }
		value = 0.75
	}
	else_if = {
		limit = { kehillah_host_outlook_value <= -40 }
		value = 1.5
	}
	else_if = {
		limit = { kehillah_host_outlook_value <= -15 }
		value = 1.25
	}
	if = {
		limit = { primary_title ?= { var:kehillah_charter_cache_security ?= 2 } }
		multiply = 0.8
	}
	else_if = {
		limit = { primary_title ?= { var:kehillah_charter_cache_security ?= 1 } }
		multiply = 0.9
	}
}

# The Shtadlan's plea: 25 + 2 per point of his diplomacy (up to +40),
# +10 if the lord's opinion of the leader is 0 or more.
kehillah_shtadlan_plea_success_value = {
	value = 25
	if = {
		limit = { exists = cp:councillor_chancellor }
		add = {
			value = cp:councillor_chancellor.diplomacy
			multiply = 2
			max = 40
		}
	}
	if = {
		limit = {
			domicile.domicile_location.county.holder.top_liege ?= {
				opinion = { target = root value >= 0 }
			}
		}
		add = 10
	}
	max = 90
}
kehillah_shtadlan_plea_failure_value = {
	value = 100
	subtract = kehillah_shtadlan_plea_success_value
}

# --- 3. Levers.
kehillah_gift_outlook = 5

# Present a Gift has no cooldown (2026-10-13, Daniel: "you should be able
# to give gifts whenever too but they should be expensive depending on the
# status of the liege, his financial situation, your income"). Scope: the
# giver (a Kehillah leader); scope:recipient set.
# 25, plus three months of his income and three of yours;
# x1.25 a duke, x1.5 a king, x2 an emperor;
# x0.75 if he is in debt, x1.25 if his treasury holds two years' income;
# x1.5, x2, x2.5 ... for each gift the community gave in the last three
# years (kehillah_gift_recent on its title).
# A count of 2/month from a community of 3/month: 40. A duke of 6/month:
# 65. An emperor of 25/month: about 220.
kehillah_gift_base_cost = 25
kehillah_gift_escalation_years = 3

kehillah_gift_cost_value = {
	value = kehillah_gift_base_cost
	add = {
		value = monthly_character_income
		multiply = 3
		min = 0
	}
	scope:recipient = {
		add = {
			value = monthly_character_income
			multiply = 3
			min = 0
		}
	}
	if = {
		limit = { scope:recipient = { highest_held_title_tier >= tier_empire } }
		multiply = 2
	}
	else_if = {
		limit = { scope:recipient = { highest_held_title_tier >= tier_kingdom } }
		multiply = 1.5
	}
	else_if = {
		limit = { scope:recipient = { highest_held_title_tier >= tier_duchy } }
		multiply = 1.25
	}
	if = {
		limit = { scope:recipient = { gold < 0 } }
		multiply = 0.75
	}
	else_if = {
		limit = { scope:recipient = { gold >= kehillah_gift_full_treasury_value } }
		multiply = 1.25
	}
	if = {
		limit = { primary_title ?= { has_variable = kehillah_gift_recent } }
		multiply = {
			value = primary_title.var:kehillah_gift_recent
			multiply = 0.5
			add = 1
		}
	}
	round = yes
}

# Scope: the recipient. Two years of his income, at least 300.
kehillah_gift_full_treasury_value = {
	value = monthly_character_income
	multiply = 24
	min = 300
}
