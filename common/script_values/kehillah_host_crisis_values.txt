# --- 1. Protection during violence. Scope: a Kehillah leader. Multiplies
# the harm of war, crusade and risings: the host's outlook band, then the
# charter's community-security term.
kehillah_host_protection_mult_value = {
	value = 1
	if = {
		limit = { kehillah_host_outlook_value >= 40 }
		value = 0.5
	}
	else_if = {
		limit = { kehillah_host_outlook_value >= 15 }
		value = 0.75
	}
	else_if = {
		limit = { kehillah_host_outlook_value <= -40 }
		value = 1.5
	}
	else_if = {
		limit = { kehillah_host_outlook_value <= -15 }
		value = 1.25
	}
	if = {
		limit = { primary_title ?= { var:kehillah_charter_cache_security ?= 2 } }
		multiply = 0.8
	}
	else_if = {
		limit = { primary_title ?= { var:kehillah_charter_cache_security ?= 1 } }
		multiply = 0.9
	}
}

# A hostile lord's men join the plunder: extra Prosperity lost, and gold
# the lord takes.
kehillah_rising_plunder_prosperity = -40
kehillah_rising_plunder_gold = 50

# --- 2. The Lord's Demands. Scope: a Kehillah leader.
# Quarterly chance (percent): Cold 4, Hostile 10.
kehillah_host_demands_chance_value = {
	value = 0
	if = {
		limit = { kehillah_host_outlook_value <= -40 }
		value = 10
	}
	else_if = {
		limit = { kehillah_host_outlook_value <= -15 }
		value = 4
	}
}
kehillah_host_demands_cooldown_years = 5
# The levy: twice medium gold under a Cold lord, three times under a Hostile one.
kehillah_host_levy_value = {
	value = medium_gold_value
	if = {
		limit = { kehillah_host_outlook_value <= -40 }
		multiply = 3
	}
	else = {
		multiply = 2
	}
}
kehillah_host_levy_prosperity = -30
kehillah_host_demands_refusal_stability = -40
# The Shtadlan's plea: 25 + 2 per point of his diplomacy (up to +40),
# +10 if the lord's opinion of the leader is 0 or more.
kehillah_shtadlan_plea_success_value = {
	value = 25
	if = {
		limit = { exists = cp:councillor_chancellor }
		add = {
			value = cp:councillor_chancellor.diplomacy
			multiply = 2
			max = 40
		}
	}
	if = {
		limit = {
			domicile.domicile_location.county.holder.top_liege ?= {
				opinion = { target = root value >= 0 }
			}
		}
		add = 10
	}
	max = 90
}
kehillah_shtadlan_plea_failure_value = {
	value = 100
	subtract = kehillah_shtadlan_plea_success_value
}
# The appeal to the head of the faith: 30 + his outlook, at most 85.
kehillah_hof_appeal_success_value = {
	value = 30
	add = domicile.domicile_location.county.holder.top_liege.faith.religious_head.kehillah_outlook_on_jews_value
	min = 5
	max = 85
}
kehillah_hof_appeal_failure_value = {
	value = 100
	subtract = kehillah_hof_appeal_success_value
}
