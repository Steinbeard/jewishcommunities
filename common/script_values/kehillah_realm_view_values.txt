# Scope: a community title. The holder's yearly chance of violence --
# the number KehillahDangerBand reads -- so the realm list is ordered
# worst first by the same measure its Danger column shows.
kehillah_rv_title_danger_value = {
	value = 0
	if = {
		limit = { exists = holder }
		value = holder.kehillah_incident_yearly_chance_value
	}
}

# Scope: the viewed character, as ROOT (see kehillah_rv_has_chip_trigger).
# The chip's number: communities a ruler has in their own lands, or, for
# a Kehillah leader, in their host's whole realm.
kehillah_rv_chip_count_value = {
	value = 0
	if = {
		limit = { kehillah_rv_kehillah_leader_trigger = yes }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { kehillah_rv_community_in_lands_of_trigger = { LORD = root.domicile.domicile_location.county.holder.top_liege } }
			add = 1
		}
	}
	else = {
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = { kehillah_rv_community_in_lands_of_trigger = { LORD = root } }
			add = 1
		}
	}
}

# Same scope and lands; only the communities that are Tense or worse.
kehillah_rv_chip_danger_count_value = {
	value = 0
	if = {
		limit = { kehillah_rv_kehillah_leader_trigger = yes }
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				kehillah_rv_community_in_lands_of_trigger = { LORD = root.domicile.domicile_location.county.holder.top_liege }
				kehillah_rv_community_in_danger_trigger = yes
			}
			add = 1
		}
	}
	else = {
		every_in_global_list = {
			variable = kehillah_registered_communities
			limit = {
				kehillah_rv_community_in_lands_of_trigger = { LORD = root }
				kehillah_rv_community_in_danger_trigger = yes
			}
			add = 1
		}
	}
}

# The realm news feed keeps ten years. Scope: anything.
kehillah_rv_news_years_value = 10

kehillah_rv_news_cutoff_value = {
	value = current_year
	subtract = kehillah_rv_news_years_value
}
