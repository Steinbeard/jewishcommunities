# Writing aptitude and outcome. Evaluated in author scope; tooltip values
# are shared with the actual scheme chance and final quality roll.

# The three per-genre counts are generated from the catalog by
# scripts/kehillah_writing_mastery_values.py, including authored slots.
kehillah_writing_parshanut_aptitude_value = {
	value = learning
	add = { value = kehillah_writing_parshanut_mastered_value multiply = 3 }
	add = { value = kehillah_lib_tanakh_rounds_value multiply = 3 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 30 } } add = 5 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 65 } } add = 8 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 } } add = 8 }
	if = { limit = { has_trait = lifestyle_poet } add = 4 }
}
kehillah_writing_talmudics_aptitude_value = {
	value = learning
	add = { value = kehillah_lib_bavli_mastered_count_value multiply = 4 }
	add = { value = kehillah_lib_yerushalmi_mastered_count_value multiply = 2 }
	add = { value = kehillah_writing_other_talmudics_mastered_value multiply = 2 }
	add = { value = kehillah_lib_bavli_rounds_value multiply = 3 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 30 } } add = 5 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 65 } } add = 8 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 100 } } add = 8 }
	if = { limit = { kehillah_has_good_bet_din_memory_trigger = yes } add = 5 }
}
kehillah_writing_machshava_aptitude_value = {
	value = learning
	add = { value = kehillah_writing_machshava_mastered_value multiply = 4 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 30 } } add = 5 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 65 } } add = 8 }
	if = { limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 100 } } add = 8 }
	if = { limit = { kehillah_sefer_author_knows_tenet_trigger = { TENET = tenet_kehillah_hokhmah } } add = 5 }
	if = { limit = { kehillah_sefer_author_knows_tenet_trigger = { TENET = tenet_esotericism } } add = 5 }
}

kehillah_writing_start_quality_value = {
	value = 0
	if = { limit = { var:kehillah_book_genre ?= flag:parshanut } add = kehillah_writing_parshanut_aptitude_value }
	else_if = { limit = { var:kehillah_book_genre ?= flag:talmudics } add = kehillah_writing_talmudics_aptitude_value }
	else_if = { limit = { var:kehillah_book_genre ?= flag:poetry } add = { value = kehillah_writing_parshanut_aptitude_value divide = 2 } add = diplomacy }
	else = { add = kehillah_writing_machshava_aptitude_value }
	if = { limit = { has_trait = erudite } add = 5 }
	if = { limit = { has_trait = diligent } add = 4 }
	if = { limit = { has_trait = lazy } subtract = 5 }
	max = 65
	# 2026-11-23 sparks: a deeper spark starts from a better-formed idea.
	if = { limit = { var:kehillah_writing_spark_tier ?= 2 } add = 5 }
	else_if = { limit = { var:kehillah_writing_spark_tier ?= 3 } add = 10 }
}
kehillah_writing_completion_chance_value = {
	value = 20
	add = { value = learning multiply = 2 }
	if = { limit = { has_variable = kehillah_book_tally } add = { value = var:kehillah_book_tally divide = 3 } }
	if = {
		limit = { exists = var:kehillah_writing_partner }
		add = 4
		if = { limit = { var:kehillah_writing_partner = { learning >= 16 } } add = 4 }
		if = {
			limit = {
				var:kehillah_book_genre ?= flag:talmudics
				var:kehillah_writing_partner = { knows_language = language_aramaic }
			}
			add = 2
		}
		else_if = { limit = { var:kehillah_writing_partner = { knows_language = language_israelite } } add = 2 }
	}
	min = 15
	max = 95
}
kehillah_writing_quality_outlook_value = {
	value = 0
	if = { limit = { has_variable = kehillah_book_tally } value = var:kehillah_book_tally }
}
