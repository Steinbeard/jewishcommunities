# Percent chance, per quarter, that an AI host holding a crime against a
# Kehillah leader sends the arrest (kehillah_host_arrest_ai_check_effect).
# Scope: the host.
kehillah_host_arrest_quarterly_chance_value = {
	value = 15
	if = {
		limit = { has_trait = zealous }
		add = 15
	}
	if = {
		limit = { has_trait = paranoid }
		add = 10
	}
	if = {
		limit = { has_trait = just }
		add = 5
	}
	if = {
		limit = { has_trait = compassionate }
		add = -10
	}
	min = 5
}
