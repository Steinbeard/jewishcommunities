# =====================================================================
# The tribute a Kehillah pays its host (v31 §8.2). ADDED 2026-10-30.
# Scope for every value below: a Kehillah leader. The rung comes from
# the charter's kehillah_charter_tribute term. Since 2026-11-08 the
# engine collects the tribute: the two rates below are the levels'
# `tax =` shares (kehillah_host_charter_tribute_contracts.txt). The
# rest of this file is for display only; no gold moves by script.
# =====================================================================

# Rates, one per rung, so tuning is a one-line change. Daniel,
# 2026-10-12: start at 20%. Heavy is v31 §8.2's first pass; the S10
# soak tunes both.
kehillah_tribute_customary_rate = 0.2
kehillah_tribute_heavy_rate = 0.3

# Percent, for the Tribute levels' contract descriptions (2026-11-08).
kehillah_tribute_customary_percent = {
	value = kehillah_tribute_customary_rate
	multiply = 100
}
kehillah_tribute_heavy_percent = {
	value = kehillah_tribute_heavy_rate
	multiply = 100
}

# The share of income the charter's current rung takes. The rung comes
# from the title cache kehillah_refresh_charter_state_cache_effect writes
# each quarter (0 None, 1 Customary, 2 Heavy), not from the contract:
# vassal_contract_has_flag never matches a tributary charter, which left
# the rate at 0 (live, 2026-10-30), and the subject trigger is kept out of
# UI code (see that effect). Before the first refresh the rung is the
# term's default, Customary.
kehillah_tribute_rate_value = {
	value = 0
	if = {
		limit = {
			government_has_flag = government_is_kehillah
			is_tributary = yes
			exists = suzerain
		}
		if = {
			limit = {
				primary_title = {
					has_variable = kehillah_charter_cache_tribute
					var:kehillah_charter_cache_tribute = 2
				}
			}
			add = kehillah_tribute_heavy_rate
		}
		else_if = {
			limit = {
				NOT = {
					primary_title = {
						has_variable = kehillah_charter_cache_tribute
						var:kehillah_charter_cache_tribute = 0
					}
				}
			}
			add = kehillah_tribute_customary_rate
		}
		# The Nagid spec section 7 (2026-10-30): with a Nagid at court the
		# realm's Jews pay more -- the office as the crown's fiscal agent,
		# and the reason a cold ruler keeps it. Since 2026-11-08 the engine
		# applies it through kehillah_nagid_tribute_modifier on the host
		# (kehillah_refresh_charter_state_cache_effect), so the display
		# reads that modifier and shows what is actually charged.
		if = {
			limit = {
				suzerain = { has_character_modifier = kehillah_nagid_tribute_modifier }
			}
			multiply = kehillah_nagid_tribute_mult
		}
	}
}

# Percent, for display.
kehillah_tribute_rate_percent_value = {
	value = kehillah_tribute_rate_value
	multiply = 100
}
