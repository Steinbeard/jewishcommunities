# The cherem track for people (2026-11-11, Daniel: "Build cherem track").
# docs/spec/2026-11-11-reviews-and-cherem.md section 3. A score of 0-100 on
# a Jewish character (the XP of kehillah_cherem_trait): Nezifah at 25, Niddui at 50,
# the full Cherem at 100.

kehillah_cherem_nezifah = 25
kehillah_cherem_niddui = 50
kehillah_cherem_full = 100
# Below Niddui the score fades each year; Niddui and above stay until teshuvah.
kehillah_cherem_fade = 5

# What fills it (section 3.2).
kehillah_cherem_book_banned_points = 10
kehillah_cherem_book_anathema_points = 30
kehillah_cherem_read_banned_points = 5
kehillah_cherem_read_anathema_points = 15
# A head of rite condemning a book (Judge a Book): a review of full weight.
kehillah_cherem_head_condemns_points = 20

# Scope: the author; scope:kehillah_reviewer put a cherem on his book. The
# reviewer's weight / 5; a rival's ban counts a quarter of that (people see
# through it).
kehillah_cherem_book_review_points_value = {
	value = scope:kehillah_reviewer.kehillah_review_weight_value
	divide = 5
	if = {
		limit = { has_relation_rival = scope:kehillah_reviewer }
		divide = 4
	}
	round = yes
}

# A head of rite in the AI may pronounce the ban on one of his rite deep in
# Niddui, once a year at this chance.
kehillah_cherem_ai_head_threshold = 75
kehillah_cherem_ai_head_chance = 20

# A Kehillah leader in Niddui: Stability lost each year.
kehillah_cherem_niddui_leader_stability = -1
# A leader who carries on under the full cherem (section 3.4): at once, then
# each quarter (-2 a month; there is no monthly pulse for playable characters).
kehillah_cherem_defiance_stability = -20
kehillah_cherem_defiance_quarterly_stability = -6
# A leader seeking reconciliation keeps his seat during the year of
# penance, with a smaller loss and no immediate defiance penalty.
kehillah_cherem_penitence_quarterly_stability = -1

# Pronounce a Cherem (head of rite, head of faith, the Nasi).
kehillah_cherem_pronounce_piety_cost = 150

# Teshuvah (section 3.5).
kehillah_teshuvah_piety_cost = 200
kehillah_teshuvah_days = 365
kehillah_teshuvah_cooldown_years = 5
# Renouncing the book that caused it lowers its reception in the penitent's rite.
kehillah_teshuvah_renounce_reception = -10

# The court that judges a teshuvah (2026-11-11, Daniel: a refusal "needs a
# good reason"). A judge refuses only on grounds: his enmity (rival, or opinion
# this low), his zeal against the book that caused it (stance this low), the
# penitent's relapse (a teshuvah accepted within these years) or new offence
# during the penance. A court of three refuses by majority; a head of rite alone.
kehillah_teshuvah_enmity_opinion = -30
kehillah_teshuvah_zeal_stance = -25
kehillah_teshuvah_relapse_years = 10

# Scope: the penitent, with the court seated (kehillah_teshuvah_seat_court_effect).
kehillah_teshuvah_court_seated_value = {
	value = 0
	if = { limit = { exists = scope:kehillah_teshuvah_judge_1 } add = 1 }
	if = { limit = { exists = scope:kehillah_teshuvah_judge_2 } add = 1 }
	if = { limit = { exists = scope:kehillah_teshuvah_judge_3 } add = 1 }
}
kehillah_teshuvah_court_refusals_value = {
	value = 0
	if = { limit = { scope:kehillah_teshuvah_judge_1 ?= { kehillah_teshuvah_judge_refuses_trigger = yes } } add = 1 }
	if = { limit = { scope:kehillah_teshuvah_judge_2 ?= { kehillah_teshuvah_judge_refuses_trigger = yes } } add = 1 }
	if = { limit = { scope:kehillah_teshuvah_judge_3 ?= { kehillah_teshuvah_judge_refuses_trigger = yes } } add = 1 }
}
