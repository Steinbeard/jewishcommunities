# --- The Talmud on trial.
# Whoever argues best for us: the leader, or the Dayyan if more learned.
kehillah_talmud_defender_learning_value = {
	value = learning
	if = {
		limit = {
			exists = scope:kehillah_incident_dayyan
			scope:kehillah_incident_dayyan.learning > learning
		}
		value = scope:kehillah_incident_dayyan.learning
	}
}
# Answer him in the hall: 25 + (our best learning - his) x3 + the lord's
# outlook / 4; a zealous lord -10, a just lord +10. 5 to 75.
kehillah_talmud_defend_chance_value = {
	value = 25
	if = {
		limit = { exists = scope:kehillah_incident_accuser }
		add = {
			value = kehillah_talmud_defender_learning_value
			subtract = scope:kehillah_incident_accuser.learning
			multiply = 3
		}
	}
	add = {
		value = kehillah_host_outlook_value
		divide = 4
	}
	scope:kehillah_incident_host ?= {
		if = {
			limit = { has_trait = zealous }
			add = -10
		}
		if = {
			limit = { has_trait = just }
			add = 10
		}
	}
	min = 5
	max = 75
}
kehillah_talmud_defend_fail_value = {
	value = 100
	subtract = kehillah_talmud_defend_chance_value
}

# The Shtadlan goes to the lord first: 10 + his diplomacy x2 + the lord's
# opinion of him / 4 + the lord's outlook / 4; a zealous lord -15. 5 to 65.
kehillah_talmud_shtadlan_chance_value = {
	value = 10
	if = {
		limit = {
			exists = scope:kehillah_incident_shtadlan
			exists = scope:kehillah_incident_host
		}
		add = {
			value = scope:kehillah_incident_shtadlan.diplomacy
			multiply = 2
		}
		add = {
			value = "scope:kehillah_incident_host.opinion(scope:kehillah_incident_shtadlan)"
			divide = 4
		}
		if = {
			limit = { scope:kehillah_incident_host = { has_trait = zealous } }
			add = -15
		}
	}
	add = {
		value = kehillah_host_outlook_value
		divide = 4
	}
	min = 5
	max = 65
}
kehillah_talmud_shtadlan_fail_value = {
	value = 100
	subtract = kehillah_talmud_shtadlan_chance_value
}
