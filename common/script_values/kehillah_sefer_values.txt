# The title caches var:kehillah_lib_foundation_missing (kehillah_lib_
# refresh_cache_effect, run whenever the shelf changes and yearly), so this
# character-scope value reads it without changing scope.
kehillah_bd_foundational_works_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_lib_shas_complete } }
		add = kehillah_lib_complete_shas_bonus
	}
	else_if = {
		limit = {
			primary_title ?= {
				has_variable = kehillah_lib_foundation_missing
				var:kehillah_lib_foundation_missing > 0
			}
		}
		subtract = {
			value = primary_title.var:kehillah_lib_foundation_missing
			multiply = kehillah_lib_shas_order_penalty
		}
	}
}
