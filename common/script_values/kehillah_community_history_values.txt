# County scope. The original place date survives destruction of a Kehillah
# title, so this age is stable across loss and refounding.
kehillah_history_age_value = {
	value = current_year
	subtract = var:kehillah_history_original_year
	min = 0
}

# 2026-11-15. County scope. How long the lost community stood: from its
# last refounding if it was refounded before this loss, else from its
# first record. Read only behind kehillah_history_has_span_trigger.
kehillah_history_years_stood_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_history_last_loss_year }
		value = var:kehillah_history_last_loss_year
		if = {
			limit = {
				has_variable = kehillah_history_last_refounded_year
				var:kehillah_history_last_refounded_year < var:kehillah_history_last_loss_year
			}
			subtract = var:kehillah_history_last_refounded_year
		}
		else_if = {
			limit = { has_variable = kehillah_history_original_year }
			subtract = var:kehillah_history_original_year
		}
	}
	min = 0
}
