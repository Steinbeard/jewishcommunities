# ---------------------------------------------------------------------
# OUTLOOK ON THE JEWS: the Iberian Struggle's phase (v53 section 4a,
# 2026-11-09, Daniel: "have these phases directly affect outlook on Jews").
# One source in kehillah_outlook_on_jews_value, so every AI system that
# reads the outlook (host attitude, danger, expulsion, invitations, mob
# odds, host crisis choices, trials, risings) and the outlook opinion
# (kehillah_outlook_opinion) follow the phase. Scope: any character.
# Only rulers in the struggle, and never a Jew: the Jewish faiths are
# themselves involved in it. Opportunity is 0 on purpose: self-interest,
# which the court and debts sources already carry. Stacks with vanilla's
# opinion_of_different_faith on involved rulers (Hostility -10,
# Conciliation +20), which reaches only personal opinion.
# Regent (diarch) redirect, as every other source.
# ---------------------------------------------------------------------
kehillah_outlook_struggle_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_struggle_value
	}
	else = {
		add = kehillah_outlook_struggle_own_value
	}
}

kehillah_outlook_struggle_own_value = {
	value = 0
	if = {
		limit = {
			has_fp2_dlc_trigger = yes
			exists = struggle:iberian_struggle
			is_jewish_character_trigger = no
		}
		if = {
			limit = { any_character_struggle = { involvement = involved is_struggle_type = iberian_struggle } }
			if = {
				limit = { struggle:iberian_struggle = { is_struggle_phase = struggle_iberia_phase_hostility } }
				add = -15
			}
			else_if = {
				limit = { struggle:iberian_struggle = { is_struggle_phase = struggle_iberia_phase_compromise } }
				add = 5
			}
			else_if = {
				limit = { struggle:iberian_struggle = { is_struggle_phase = struggle_iberia_phase_conciliation } }
				add = 15
			}
		}
		else_if = {
			limit = { any_character_struggle = { involvement = interloper is_struggle_type = iberian_struggle } }
			if = {
				limit = { struggle:iberian_struggle = { is_struggle_phase = struggle_iberia_phase_hostility } }
				add = -10
			}
			else_if = {
				limit = { struggle:iberian_struggle = { is_struggle_phase = struggle_iberia_phase_conciliation } }
				add = 5
			}
		}
	}
}
