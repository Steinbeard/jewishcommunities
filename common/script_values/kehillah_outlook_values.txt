# =====================================================================
# OUTLOOK ON THE JEWS -- V34 (docs/spec/v34-outlook-teachings-and-
# founding.md section 2), 2026-10-13. One number, -100..+100, for how a
# character regards Jews: read by the pillars (the host's attitude), and
# meant for every system that asks "how does this person feel about the
# Jews" -- a head of faith deciding his rite's teaching, council voters,
# popular leaders, the policy and charter review.
# Scope: any character.
# =====================================================================

kehillah_outlook_on_jews_value = {
	value = 0
	# The sources, each its own value so the player can be shown the
	# breakdown (V35, 2026-10-13; KehillahOutlookBd* custom loc). Each source
	# defers to the regent when the ruler has one (see below).
	add = kehillah_outlook_teaching_value
	add = kehillah_outlook_zeal_value
	add = kehillah_outlook_character_value
	add = kehillah_outlook_culture_value
	add = kehillah_outlook_learning_value
	add = kehillah_outlook_court_value
	add = kehillah_outlook_debts_value
	add = kehillah_outlook_relations_value
	add = kehillah_outlook_crusade_value
	add = kehillah_outlook_holy_warrior_value
	add = kehillah_outlook_history_value
	min = -100
	max = 100
}

# --- His rite's official teaching (v31 doctrines): the default he was
# raised with.
# Teaching source: a ruler under a regent (diarch) rules through him, so this reads
# the regent's (2026-10-18, Daniel: a 14-year-old king read Indifferent while
# his regent was +10; the breakdown must match the total).
kehillah_outlook_teaching_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_teaching_value
	}
	else = {
		add = kehillah_outlook_teaching_own_value
	}
}

kehillah_outlook_teaching_own_value = {
	value = 0
	# The Almoravids (Daniel, 2026-10-20: "hostile towards Jews, not
	# protective"). Their rite, Maliki, teaches Dhimma like every Sunni
	# rite, so the house is singled out here: the Lamtuna reformers' zeal
	# reads as an exclusionary teaching. The teaching-band triggers still
	# read the rite; only the outlook the player sees and acts on moves.
	if = {
		limit = { house ?= house:house_almoravid }
		add = -50
	}
	else_if = {
		limit = { kehillah_teaching_protective_trigger = yes }
		add = 15
	}
	else_if = {
		limit = { kehillah_teaching_strict_trigger = yes }
		add = -15
	}
	else_if = {
		limit = { kehillah_teaching_exclusionary_trigger = yes }
		add = -40
	}
}

# --- Zeal amplifies the teaching ("a zealous Witness bishop protects").
# Zeal source: a ruler under a regent (diarch) rules through him, so this reads
# the regent's (2026-10-18, Daniel: a 14-year-old king read Indifferent while
# his regent was +10; the breakdown must match the total).
kehillah_outlook_zeal_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_zeal_value
	}
	else = {
		add = kehillah_outlook_zeal_own_value
	}
}

kehillah_outlook_zeal_own_value = {
	value = 0
	if = {
		limit = { has_trait = zealous }
		if = {
			limit = { kehillah_teaching_protective_trigger = yes }
			add = 10
		}
		else_if = {
			limit = { kehillah_teaching_exclusionary_trigger = yes }
			add = -25
		}
		else_if = {
			limit = { kehillah_teaching_strict_trigger = yes }
			add = -15
		}
		else = {
			add = -10
		}
	}
}

# --- Personality.
# Character source: a ruler under a regent (diarch) rules through him, so this reads
# the regent's (2026-10-18, Daniel: a 14-year-old king read Indifferent while
# his regent was +10; the breakdown must match the total).
kehillah_outlook_character_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_character_value
	}
	else = {
		add = kehillah_outlook_character_own_value
	}
}

kehillah_outlook_character_own_value = {
	value = 0
	# 2026-10-13 (Daniel: "make personality have a bigger effect too to have
	# more variety" -- in place of a flat prejudice term). Personality alone
	# can swing a lord about +/-50: a compassionate, just lord protects; a
	# sadistic, paranoid one is an enemy.
	if = { limit = { has_trait = compassionate } add = 20 }
	if = { limit = { has_trait = just } add = 15 }
	if = { limit = { has_trait = cynical } add = 15 }
	if = { limit = { has_trait = humble } add = 10 }
	if = { limit = { has_trait = trusting } add = 10 }
	if = { limit = { has_trait = forgiving } add = 5 }
	if = { limit = { has_trait = generous } add = 5 }
	if = { limit = { has_trait = ambitious } add = 5 }
	# The opportunist who sees taxes and loans in the Jews -- unless he owes
	# them (kehillah_outlook_debts_value then resents it).
	if = {
		limit = {
			has_trait = greedy
			NOT = { has_variable = kehillah_loan_amount_owed }
		}
		add = 5
	}
	# Physicians knew Jewish medicine and its practitioners.
	if = { limit = { has_trait = lifestyle_physician } add = 5 }
	if = { limit = { has_trait = sadistic } add = -25 }
	if = { limit = { has_trait = paranoid } add = -15 }
	if = { limit = { has_trait = callous } add = -10 }
	if = { limit = { has_trait = arrogant } add = -10 }
	if = { limit = { has_trait = wrathful } add = -10 }
	if = { limit = { has_trait = vengeful } add = -10 }
	if = { limit = { has_trait = arbitrary } add = -5 }
	# The learned accuser: a theologian under a strict or exclusionary
	# teaching (the Talmud's prosecutors were learned men).
	if = {
		limit = {
			has_trait = theologian
			OR = {
				kehillah_teaching_strict_trigger = yes
				kehillah_teaching_exclusionary_trigger = yes
			}
		}
		add = -5
	}
}

# --- Culture: a tradition of tolerance toward other faiths or peoples
# (the Nagid spec section 5, 2026-10-30; Daniel: "factor the African
# Tolerance culture into our rulers' Jewish outlook"). African and Steppe
# Tolerance +10, Religion Blending and Xenophilic +5. Vanilla already
# raises these rulers' opinion of other faiths or cultures, so this line
# is left out of kehillah_jews_opinion_value below.
# Culture source: a ruler under a regent (diarch) rules through him, as
# for the other sources.
kehillah_outlook_culture_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_culture_value
	}
	else = {
		add = kehillah_outlook_culture_own_value
	}
}

kehillah_outlook_culture_own_value = {
	value = 0
	if = {
		limit = { exists = culture }
		if = {
			limit = { culture = { has_cultural_tradition = tradition_african_tolerance } }
			add = 10
		}
		if = {
			limit = { culture = { has_cultural_tradition = tradition_steppe_tolerance } }
			add = 10
		}
		if = {
			limit = { culture = { has_cultural_tradition = tradition_religion_blending } }
			add = 5
		}
		if = {
			limit = { culture = { has_cultural_tradition = tradition_xenophilic } }
			add = 5
		}
	}
}

# --- Owing a Kehillah money: welcome at first, resented as it ages (the
# debts behind Philip II's and Edward I's expulsions). Greed sharpens it.
# Debts source: a ruler under a regent (diarch) rules through him, so this reads
# the regent's (2026-10-18, Daniel: a 14-year-old king read Indifferent while
# his regent was +10; the breakdown must match the total).
kehillah_outlook_debts_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_debts_value
	}
	else = {
		add = kehillah_outlook_debts_own_value
	}
}

kehillah_outlook_debts_own_value = {
	value = 0
	# (a dead character holds no variables; the HUD may still read one)
	if = {
		limit = {
			is_alive = yes
			has_variable = kehillah_loan_amount_owed
		}
		if = {
			limit = { var:kehillah_loan_years_elapsed ?= 0 }
			add = 0
		}
		else_if = {
			limit = {
				has_variable = kehillah_loan_years_elapsed
				var:kehillah_loan_years_elapsed >= 5
			}
			add = -15
		}
		else = {
			add = -5
		}
		if = {
			limit = { has_trait = greedy }
			add = -10
		}
	}
}

# V35 (2026-10-13, player clarity): the outlook as a band, -2..+2:
# Hostile (<= -40), Cold (-39..-15), Indifferent (-14..14), Well-disposed
# (15..39), Protective (>= 40). Scope: any character. Used for the
# "lord's mood" notices (kehillah_host_outlook_watch_effect).
kehillah_outlook_band_value = {
	value = 0
	if = {
		limit = { kehillah_outlook_on_jews_value >= 40 }
		value = 2
	}
	else_if = {
		limit = { kehillah_outlook_on_jews_value >= 15 }
		value = 1
	}
	else_if = {
		limit = { kehillah_outlook_on_jews_value <= -40 }
		value = -2
	}
	else_if = {
		limit = { kehillah_outlook_on_jews_value <= -15 }
		value = -1
	}
}

# Scope: a Kehillah leader. His host's outlook (for the breakdown line).
kehillah_host_outlook_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location.county.holder.top_liege }
		value = domicile.domicile_location.county.holder.top_liege.kehillah_outlook_on_jews_value
	}
}

# Scope: a character. His opinion of a Jewish leader from his outlook on
# the Jews: the outlook, capped at +/-50 (kehillah_outlook_opinion).
# Typical lords (+5..+45) offset much of vanilla's faith-hostility
# penalty (astray -10, hostile -20, evil -30); cold and hostile ones add
# to it.
kehillah_jews_opinion_value = {
	value = kehillah_outlook_on_jews_value
	# The culture line is already in vanilla opinion (different_faith_
	# opinion, ignore_opinion_of_different_faith, different_culture_
	# opinion); counting it here too would count it twice.
	subtract = kehillah_outlook_culture_value
	min = -50
	max = 50
}
