# =====================================================================
# OUTLOOK ON THE JEWS -- V34 (docs/spec/v34-outlook-teachings-and-
# founding.md section 2), 2026-10-13. One number, -100..+100, for how a
# character regards Jews: read by the pillars (the host's attitude), and
# meant for every system that asks "how does this person feel about the
# Jews" -- a head of faith deciding his rite's teaching, council voters,
# popular leaders, the policy and charter review.
# Scope: any character.
# =====================================================================

kehillah_outlook_on_jews_value = {
	value = 0

	# --- His rite's official teaching (v31 doctrines): the default he
	# was raised with.
	if = {
		limit = { kehillah_teaching_protective_trigger = yes }
		add = 15
	}
	else_if = {
		limit = { kehillah_teaching_strict_trigger = yes }
		add = -15
	}
	else_if = {
		limit = { kehillah_teaching_exclusionary_trigger = yes }
		add = -40
	}

	# --- Personality. Zeal amplifies the teaching ("a zealous Witness
	# bishop protects").
	if = {
		limit = { has_trait = zealous }
		if = {
			limit = { kehillah_teaching_protective_trigger = yes }
			add = 10
		}
		else_if = {
			limit = { kehillah_teaching_exclusionary_trigger = yes }
			add = -25
		}
		else_if = {
			limit = { kehillah_teaching_strict_trigger = yes }
			add = -15
		}
		else = {
			add = -10
		}
	}
	if = { limit = { has_trait = compassionate } add = 15 }
	if = { limit = { has_trait = just } add = 10 }
	if = { limit = { has_trait = cynical } add = 10 }
	if = { limit = { has_trait = humble } add = 5 }
	if = { limit = { has_trait = paranoid } add = -10 }
	if = { limit = { has_trait = sadistic } add = -15 }
	if = { limit = { has_trait = arrogant } add = -5 }
	if = { limit = { has_trait = callous } add = -5 }

	# --- Learning: the scholar who knows Jewish colleagues and texts.
	if = {
		limit = { learning >= 15 }
		add = 5
	}

	# --- Interests: a Jew at his court, and money.
	if = {
		limit = {
			OR = {
				any_councillor = { is_jewish_character_trigger = yes }
				any_court_position_holder = { is_jewish_character_trigger = yes }
			}
		}
		add = 10
	}
	# Owing a Kehillah money: welcome at first, resented as it ages
	# (the debts behind Philip II's and Edward I's expulsions). Greed
	# sharpens the resentment.
	if = {
		limit = { has_variable = kehillah_loan_amount_owed }
		if = {
			limit = { var:kehillah_loan_years_elapsed ?= 0 }
			add = 0
		}
		else_if = {
			limit = {
				has_variable = kehillah_loan_years_elapsed
				var:kehillah_loan_years_elapsed >= 5
			}
			add = -15
		}
		else = {
			add = -5
		}
		if = {
			limit = { has_trait = greedy }
			add = -10
		}
	}

	# --- Personal history: what Jewish communities have done for or
	# against him (kehillah_shift_outlook_effect).
	if = {
		limit = { has_variable = kehillah_outlook_shift }
		add = var:kehillah_outlook_shift
	}

	min = -100
	max = 100
}
