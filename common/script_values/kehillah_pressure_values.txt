# =====================================================================
# V33 PRESSURES -- 2026-10-12 (docs/spec/v33-pressures.md). Stability now
# rests at Healthy (kehillah_stability_baseline_floor_value 500) and is
# pulled down by the live world; Prosperity follows the county's economy.
# Each term is a breakdown value (scope: a Kehillah leader), added to the
# pillar baselines in kehillah_script_values.txt and shown in the pillar
# tooltips (kehillah_pressure_custom_loc.txt). Read every quarter, so a
# war, a crusade or a hostile host is felt while it lasts.
# =====================================================================

# --- Asymmetric drift (kehillah_quarterly_pillars_effect): Stability and
# Prosperity fall fast toward a lower target and recover slowly.
kehillah_convergence_rate_down_value = 0.15

# --- The host's attitude (Stability). The host is the top liege of the
# county the quarter stands in.
kehillah_bd_host_attitude_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location.county.holder.top_liege }
		if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { opinion = { target = root value >= 40 } } }
			add = 50
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { opinion = { target = root value >= 10 } } }
			add = 20
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { opinion = { target = root value <= -40 } } }
			add = -150
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { opinion = { target = root value <= -10 } } }
			add = -60
		}
		# Zeal is scaled by the host's Teaching on the Jews (v31 section 5):
		# a zealous protector protects; a zealous excluder persecutes.
		if = {
			limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = zealous } }
			if = {
				limit = { domicile.domicile_location.county.holder.top_liege = { kehillah_teaching_protective_trigger = yes } }
				add = 20
			}
			else_if = {
				limit = { domicile.domicile_location.county.holder.top_liege = { kehillah_teaching_exclusionary_trigger = yes } }
				add = -120
			}
			else = {
				add = -60
			}
		}
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = sadistic } } add = -40 }
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = greedy } } add = -30 }
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = paranoid } } add = -30 }
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = compassionate } } add = 30 }
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = just } } add = 20 }
		if = { limit = { domicile.domicile_location.county.holder.top_liege = { has_trait = cynical } } add = 20 }
	}
}

			limit = { domicile.domicile_location.county.holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_banned_trigger = yes } }

			limit = { domicile.domicile_location.county.holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_discouraged_trigger = yes } }
			add = -100
		}
	}
}

# --- War that comes through the county.
kehillah_bd_war_stability_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location }
		if = {
			limit = { domicile.domicile_location = { is_occupied = yes } }
			add = -200
		}
		else_if = {
			limit = { domicile.domicile_location = { is_raided = yes } }
			add = -120
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege ?= { is_at_war = yes } }
			add = -40
		}
	}
}
kehillah_bd_war_prosperity_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location }
		if = {
			limit = { domicile.domicile_location = { is_occupied = yes } }
			add = -150
		}
		else_if = {
			limit = { domicile.domicile_location = { is_raided = yes } }
			add = -100
		}
		else_if = {
			limit = { domicile.domicile_location.county.holder.top_liege ?= { is_at_war = yes } }
			add = -25
		}
	}
}

# --- A crusade or jihad underway (the 1096 Rhineland massacres were the
# Jewish experience of the First Crusade): the county's or host's faith
# has a great holy war in any stage.
kehillah_bd_holy_war_value = {
	value = 0
	if = {
		limit = {
			OR = {
				domicile.domicile_location.county.faith ?= { exists = great_holy_war }
				domicile.domicile_location.county.holder.top_liege.faith ?= { exists = great_holy_war }
			}
		}
		add = -150
	}
}

# --- Religious tension: how the county's faith regards Judaism -- the
# local population's view. Light on purpose (Daniel 2026-10-12): every
# Christian and Muslim county reads "evil" at game start, so the teaching
# (kehillah_bd_host_teaching_stability_value) carries the ideology, and
# this term only shows a faith that comes to see Judaism as merely astray.
kehillah_bd_religious_tension_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location.county.faith }
		if = {
			limit = { domicile.domicile_location.county.faith = { faith_hostility_level = { target = root.faith value >= faith_evil_level } } }
			add = -30
		}
		else_if = {
			limit = { domicile.domicile_location.county.faith = { faith_hostility_level = { target = root.faith value >= faith_hostile_level } } }
			add = -20
		}
		else_if = {
			limit = { domicile.domicile_location.county.faith = { faith_hostility_level = { target = root.faith value >= faith_astray_level } } }
			add = -10
		}
	}
}

# --- The county's economy (Prosperity): development, and order.
kehillah_bd_local_economy_value = {
	value = 0
	if = {
		limit = { exists = domicile.domicile_location.county }
		add = {
			value = domicile.domicile_location.county.development_level
			multiply = 4
		}
		if = {
			limit = { domicile.domicile_location.county = { county_control < 50 } }
			add = -50
		}
	}
}
