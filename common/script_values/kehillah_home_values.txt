# Seasons an adventurer must spend in one community's county for it to
# become home (about a year).
kehillah_home_stay_seasons = 4

# Scope: a member. The yearly tithe: a tenth of a year's income, at least 1.
kehillah_home_tithe_value = {
	value = monthly_character_income
	multiply = 1.2
	round = yes
	min = 1
}

# Scope: a member. Prestige and piety for the tithe: twice the tithe,
# between 15 and 150 each.
kehillah_home_tithe_reward_value = {
	value = kehillah_home_tithe_value
	multiply = 2
	min = 15
	max = 150
}

# Scope: a Kehillah leader. Living members who call this community home.
kehillah_home_member_count_value = {
	value = 0
	primary_title ?= {
		every_in_list = {
			variable = kehillah_home_members
			limit = { is_alive = yes }
			add = 1
		}
	}
}
kehillah_home_member_cap = 5
kehillah_home_member_counted_value = {
	value = kehillah_home_member_count_value
	max = kehillah_home_member_cap
}
# Prosperity baseline per member.
kehillah_home_member_prosperity_each = 10
kehillah_bd_home_members_prosperity_value = {
	value = kehillah_home_member_counted_value
	multiply = kehillah_home_member_prosperity_each
}
