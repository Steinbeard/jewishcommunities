# =====================================================================
# WHERE THE RABBI TRACKS MATTER -- V29 section 4 (docs/spec/v29-yeshiva.md),
# 2026-10-11. Before this, the three tracks of kehillah_rabbi_trait only
# added stats and gated book quality; nothing they did changed an outcome.
#
#   Talmudics -> judging: Bet Din rulings, dispute rulings, responsa
#                (kehillah_judicial_learning_value, below), and Rosh
#                Yeshiva aptitude.
#   Parshanut -> teaching: Rosh Yeshiva aptitude, the zman, and wards
#                raised by a rabbi (kehillah_observe_rabbinic_education_
#                effect).
#   Hashkafa  -> standing with the host: the Shtadlan's petition chance
#                (kehillah_hashkafa_petition_mult_value), on top of the
#                trait's existing opinion of other faiths.
# =====================================================================

# Scope: a character acting as a judge or answering a question of law.
# Learning, plus a Talmudics bonus for a rabbi: +2 at 30 XP, +4 at 65,
# +6 at 100. Every "is the ruling good" Learning check in the Bet Din
# cases, the dispute events and responsa reads this instead of bare
# Learning, so a deep Talmudist rules better than a generalist with the
# same Learning. Kept in Learning's own units so the existing thresholds
# (8 / 10 / 12 / 16) still mean what they meant.
kehillah_judicial_learning_value = {
	value = learning
	if = {
		limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 100 } }
		add = 6
	}
	else_if = {
		limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 65 } }
		add = 4
	}
	else_if = {
		limit = { has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 30 } }
		add = 2
	}
}

# Scope: the Shtadlan, with scope:councillor_liege = the Kehillah leader.
# The leader's Hashkafa -- his standing in the eyes of other faiths --
# multiplies the Shtadlan's chance of moving the host: x1.25 at 30 XP,
# x1.5 at 65, x1.75 at 100.
kehillah_hashkafa_petition_mult_value = {
	value = 1
	if = {
		limit = { scope:councillor_liege ?= { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 100 } } }
		add = 0.75
	}
	else_if = {
		limit = { scope:councillor_liege ?= { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 65 } } }
		add = 0.5
	}
	else_if = {
		limit = { scope:councillor_liege ?= { has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 30 } } }
		add = 0.25
	}
}
