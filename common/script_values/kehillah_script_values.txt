# =====================================================================
# Kehillah economy tuning values.
#
# Spec section 7, question 5 asked whether to work the numbers out in
# full up front or prototype loosely and tune by playtest. This file is
# the "prototype loosely" answer: every number lives here, in one place,
# with a stated intent, so tuning after a playtest is a single-file edit
# rather than a hunt through building and decision definitions.
#
# Scale intent (spec section 3, Prosper): a Kehillah's income is
# community-scale, not realm-scale. A fully built quarter should feel
# comfortable, never rival a landed count. Gold figures below are
# deliberately small in absolute terms for that reason.
# =====================================================================

# ---------------------------------------------------------------------
# Building costs
# ---------------------------------------------------------------------

# The Synagogue. Tier 1 is cheap because the Worms scenario starts with
# it already built (scenario doc section 4); the later tiers are the
# real long-term Gold sink and should each feel like a decision.
kehillah_synagogue_cost_1 = 75
kehillah_synagogue_cost_2 = 200
kehillah_synagogue_cost_3 = 400
kehillah_synagogue_cost_4 = 700
kehillah_synagogue_cost_5 = 1100

# The other six building families: Yeshiva (Beit Midrash), Countinghouse,
# Mikvah, Hekdesh, Sofer's Workshop, Slaughterhouse. Cheaper than
# Synagogue tiers -- these are the moment-to-moment growth choices.
# (Shtadlan's Chambers and the Communal Watch were removed 2026-09-06;
# see common/domiciles/buildings/kehillah_domicile_buildings.txt.)
#
# NAMING NOTE: "internal" here is a cost-tier label, not a slot-type
# claim. As of the 2026-09-06 externalization pass, only the Mikvah and
# Sofer's Workshop actually occupy internal slots; Yeshiva, Countinghouse,
# Hekdesh and Slaughterhouse moved to external slots but kept using
# these same cost values (unchanged numbers, no rebalance intended) --
# renaming the script_value keys themselves would touch every reference
# in kehillah_domicile_buildings.txt for no mechanical benefit.
kehillah_internal_cost_1 = 100
kehillah_internal_cost_2 = 220
kehillah_internal_cost_3 = 400

# External buildings: market stalls and craft workshops. Cheapest of the
# three tracks, because they are what pays for everything else.
kehillah_external_cost_1 = 80
kehillah_external_cost_2 = 180
kehillah_external_cost_3 = 320

# so it is a real choice against building instead. Renamed 2026-09-07

# =====================================================================
# WRITE A BOOK -- kehillah_write_book_decision (common/decisions/
# kehillah_rabbi_book_decisions.txt) and its event chain (events/
# kehillah_book_events.txt). ADDED 2026-09-09; FLESHED OUT the same day
# at user request from three flat, single-effect decisions (one great/
# famed/masterwork branch picked instantly by whichever track threshold
# was met) into one decision that opens a real chain: pick a genre,
# pick an inspiration, discuss the work with a courtier, decide how
# hard to revise, then the book is finished and copies go out to nearby
# communities. Originally a personal-mastery counterpart to kehillah_
# compose_commentary_decision (cheaper, one person's own scholarship
# rather than an act drawing on the whole community's standing, but
# reaching a real event chain that decision never did) -- that decision
# is RETIRED as of 2026-09-15 (see its own retirement note, common/
# decisions/kehillah_decisions.txt), so this is now this mod's only
# "compose a scholarly work" mechanic, full stop.
kehillah_rabbi_book_piety_cost = 75

# THE TALLY. Same shape as the Bet Din docket's own tally (common/
# script_values/kehillah_script_values.txt's "BET DIN CONFERENCE"
# section below) for a reason: this mod already has a proven pattern
# for "several small choices sum into one number that decides a
# tiered outcome," and reusing it here means the four genres, and
# poetry alongside the three rabbi genres specifically, all resolve
# through the same mechanism rather than a bespoke one each.
#
# BASELINE, per genre: the three rabbi genres start from whichever
# track threshold (30/65/100 XP, kehillah_rabbi_trait's own tracks,
# common/traits/kehillah_traits.txt) the character has actually
# cleared -- "each tier of the lifestyle skill" setting the FLOOR the
# event chain's choices build on top of, not the final result by
# itself. Poetry has no track to read, so it gets one flat baseline
# instead, deliberately close to the middle rabbi baseline (65) rather
# than the bottom or top one -- holding lifestyle_poet at all already
# implies real standing, which the tiered rabbi baselines are trying to
# measure a different way.
kehillah_book_baseline_track_30 = 10
kehillah_book_baseline_track_65 = 25
kehillah_book_baseline_track_100 = 40
kehillah_book_baseline_poet = 22

# THE CHAIN'S OWN SWINGS. Three interactive steps -- inspiration,
# courtier discussion, revision -- each adds or subtracts from the
# tally the baseline above started. Deliberately enough range (up to
# +45 across the three steps) that even a fresh 30-track rabbi can
# reach the top tier with good choices and good luck, and enough
# downside that a 100-track rabbi who chooses poorly can still land in
# the middle tier -- the baseline sets the floor this mod's chosen
# metaphor ("your standing") protects, not a guaranteed ceiling.
kehillah_book_inspiration_gain = 12
kehillah_book_inspiration_loss = 4
kehillah_book_courtier_gain = 10
kehillah_book_courtier_loss = 3
# The courtier step's own "chance for a better tier," per the user's
# literal phrasing -- a genuine random_list roll on top of the stat
# check, not just another deterministic branch.
kehillah_book_courtier_lucky_bonus = 8
kehillah_book_revision_gain = 15
kehillah_book_revision_stress = minor_stress_gain

# A FOURTH INSPIRATION OPTION, ADDED 2026-09-15 at user request: "draw
# on a bet din memory of a good ruling to improve the quality of your
# book, particularly for Talmudics and Halakha works." Visible in
# kehillah_book.0002 only when the genre is talmudics (Halakha merged
# into Talmudics later the same day -- kehillah_rabbi_trait's own
# header, common/traits/kehillah_traits.txt -- so this is the same set
# of books the option was always meant to cover, not a narrowing) AND
# the writer holds at least one of the seven "good ruling" Bet Din
# memory types (kehillah_has_good_bet_din_memory_trigger, common/
# scripted_triggers/kehillah_scripted_triggers.txt). Unlike the other
# two inspiration options, this one has NO stat gate and cannot fail --
# it is recalling something that genuinely happened, not gambling on
# one -- and pays MORE than kehillah_book_inspiration_gain's own pass
# value for exactly that reason: a real, earned memory of ruling well
# is worth more than a favorable roll on generic inspiration.
kehillah_book_bet_din_memory_gain = 16

# FINAL THRESHOLDS. No failure floor: the worst possible outcome is
# still a masterwork-tier book, never nothing -- a private note, not an
# abject failure, rather than wasting the Piety already spent to start
# the chain.
kehillah_book_threshold_illustrious = 55
kehillah_book_threshold_famed = 30

# REWARDS. "A big contribution to the greatness of the community" --
# deliberately larger than kehillah_bet_din's own per-case Stability
# swings (kehillah_bet_din_case1_stability_gain_great = 40 is the
# biggest single number that family has) since finishing a real book is
# meant to read as a bigger, rarer act than one Bet Din ruling.
kehillah_book_greatness_gain = 60
kehillah_book_piety_gain = major_piety_gain
kehillah_book_prestige_gain = major_prestige_gain

# EPILOGUE. Sending copies to the nearest other registered community
# (common/scripted_effects/kehillah_book_effects.txt's "send copies"
# section) gives that community's OWN leader a small Learning/Piety
# nudge for having a real new work to read, and the author a small
# amount of the SAME community's opinion -- proportionally tiny next to
# the book's own rewards above, since this is flavor payoff for the
# registry existing, not a second reward system.
kehillah_book_recipient_piety_gain = minor_piety_gain
kehillah_book_recipient_learning_xp = minor_lifestyle_xp

# The diminishing-returns weights shared by every baseline's "leader

# ---------------------------------------------------------------------
# kehillah_prosperity_baseline_value
# Scope: the Kehillah leader (character) -- evaluated from inside
# kehillah_quarterly_pillars_effect, the same scope as every other
# building/office check in that effect.
#
# v2 spec section 2's Prosperity inputs: Countinghouse/Market Stalls/
# Craft Workshops tiers (all P1, i.e. already-shipped buildings), the
# Slaughterhouse's "Prosperity undertone", and the top-N Stewardship
# aggregate from v1 spec section 3.
#
# TOP-N AGGREGATE, APPROXIMATED. v1/v2 spec ask for "the sum of the top
# ten Stewardship values among eligible community characters, with
# diminishing returns". CK3 script values have no "sort a list and take
# the top N" primitive -- confirmed absent from every script_value
# pattern found in the live 1.19 files during this task's research (the
# closest primitive, every_X + add, sums EVERY match unconditionally;
# there is no equivalent that ranks and truncates). Faking a true sort
# would mean hand-rolling a rank-comparison loop against every courtier,
# which is exactly the "something fragile" the task instructions for
# this pass explicitly say not to force. The honest approximation built
# instead: the leader's own Stewardship (full weight) plus the Gabbai
# (Treasurer)'s Stewardship if the office is filled (discounted weight)
# -- NOT a population-wide sum over every courtier with that skill. This
# is a deliberate simplification versus the spec's stated aspiration,
# not a missing feature -- revisit if CK3 ever gains a real top-N
# primitive, or if this is worth hand-rolling properly later.
# ---------------------------------------------------------------------

kehillah_prosperity_baseline_value = {
	# REFACTORED 2026-09-15 (v12 spec section 8): every term below is now
	# the matching kehillah_bd_<x>_value from common/script_values/
	# kehillah_breakdown_values.txt, so this baseline and the per-score
	# breakdown tooltip that explains it are ONE set of numbers -- a
	# building's contribution is defined exactly once. Change a term
	# there, and both move together.
	#
	# This also fixed a real bug that had been here since Wave 2: the
	# literal terms called has_domicile_building_or_higher BARE, from
	# character scope. It is a DOMICILE-scope trigger -- every vanilla use
	# wraps it as domicile ?= { ... } (common/achievements/
	# ep3_achievements.txt:210-220; zero bare uses anywhere) and ck3-tiger
	# had warned about it on every run. The breakdown values wrap it
	# correctly, so routing through them fixes the baselines too.

	value = 0

	# Countinghouse, Market Stalls, Craft Workshops: cumulative across
	# tiers. Slaughterhouse: v2 spec's "Prosperity undertone" -- smaller,
	# it exists mainly for Stability (a community that cannot keep kosher
	# cannot function as one), not primarily as an economic engine.
	add = kehillah_bd_countinghouse_value
	add = kehillah_bd_market_stalls_value
	add = kehillah_bd_workshops_value
	add = kehillah_bd_slaughterhouse_value
	add = kehillah_bd_charter_economic_privileges_value
	add = kehillah_bd_charter_construction_value
	add = kehillah_bd_charter_market_access_value
	add = kehillah_bd_urban_opportunity_prosperity_value

	# The Stewardship terms -- see this script value's own header above
	# for the top-N-aggregate approximation these implement: the leader
	# at full weight, the Gabbai (Treasurer) at discounted weight if the
	# office is filled.
	add = kehillah_bd_leader_stewardship_value
	add = kehillah_bd_treasurer_value
}

# Light loan contracts, baron/count tier only (Wave 3 item 4).
#
# TARGETING, AND WHY IT IS NOT A PLAYER-DRIVEN PICK. v2 spec section 2
# calls for "lend to baron/count" without specifying how a target is
# chosen. CK3 decisions (unlike character_interactions) have no native
# arbitrary-character target picker without a custom GUI widget -- this
# mod has no existing character_interaction file and no precedent for
# building one, which would have been a materially larger and less
# certain undertaking than everything else in this wave. The target
# used instead is fixed and always well-defined: the holder of the
# COUNTY the Kehillah's own quarter physically sits in
# (root.domicile.domicile_location.county.holder, the same scope chain
# kehillah_setup_worms_start_effect already uses to place created
# characters) -- the community's own local count, or whichever baron
# happens to hold that county directly. This is a deliberate
# simplification versus "targeting a baron/count-tier landed character"
# read as player choice: the player cannot choose WHICH noble, only
# whether to extend the loan at all. It is also, arguably, the more
# historically apt version of the fantasy (a community's dealings were
# overwhelmingly with its own immediate lord, not a stranger picked from
# a world-spanning list) and it is what keeps this item inside the
# "light" scope the task asks for.
#
# MECHANISM, CONFIRMED AGAINST LIVE VANILLA CONTENT, NOT INVENTED FROM
# SCRATCH. v2 spec itself flags "I haven't confirmed CK3 has any native
# mechanic for this" as an open question -- it does: Holy Orders lend
# gold to characters today (common/decisions/00_holy_order_decisions.txt's
# borrow_from_holy_order_decision, common/scripted_effects/
# 00_scripted_effects.txt's set_up_loan_effect / clear_loanee_variables_
# effect, common/decisions/80_major_decisions.txt's repay_loan_decision,
# events/religion_events/holy_order_events.txt's holy_order.0200-0206).
# That existing system is NOT reused directly here -- it is
# character-variable-based (var:loan_amount_owed etc.) and its own
# events layer on escalating consequences (demanding a castle, then a
# child hostage) that have nothing to do with this mod and that this
# mod must not risk triggering by touching shared variable names. Every
# kehillah_loan_* variable/decision below is a parallel, self-contained
# namespace with the SAME shape (principal owed, a lender reference, a
# variable_list of debtors, a plain repayment decision) so this item
# rests on confirmed-working vanilla mechanics without touching or
# risking anything vanilla or another mod already depends on. The
# variable_list mechanism itself (every_in_list / any_in_list with
# "variable = <name>", add_to_variable_list, remove_list_variable) is
# confirmed working syntax, read directly from holy_order.0206 and
# repay_loan_decision.
#
# THE LENDER IS THE TITLE, NOT THE CHARACTER -- a deliberate improvement
# on vanilla's own version, which has to special-case loan inheritance
# when a Holy Order's leader changes (holy_order.0201). Storing
# kehillah_loan_lender_title = scope:kq_title on the borrower means a
# Kehillah succession happening mid-loan needs no equivalent handling at
# all: whoever holds the title next is automatically who gets repaid,
# the same "the title IS the community" principle this mod already
# applies to buildings and pillars.
#
# NO COMPOUNDING INTEREST. The amount owed is fixed at origination
# (principal x a flat multiplier), exactly mirroring vanilla's own Holy
# Order loan (the amount borrowed and the amount owed are the same
# number there -- confirmed by reading holy_order.0200's set_up_loan_
# effect call, AMOUNT = holy_order_gold_value both times). A single lump
# multiplier is the "light" version of interest the task asks for.
# ---------------------------------------------------------------------

kehillah_loan_principal_value = 150
kehillah_loan_repayment_multiplier = 1.3
kehillah_loan_term_years = 6

# The amount actually owed, principal x the flat multiplier above,

# two-step effect call -- CONFIRMED there is no multiply_variable effect
# in vanilla (grepped for it directly against the installed 1.19 files
# during this task's research and found nothing); change_variable only
# supports add/subtract, not multiply, so the multiplication has to
# happen here, in a script_value, where "multiply" is a valid operator
# (the same operator this file's own kehillah_tzedakah_gold_cost already
# uses).
kehillah_loan_amount_owed_value = {
	value = kehillah_loan_principal_value

}

# =====================================================================
# BET DIN DOCKET -- PACING AND SESSION SCORE.
# ADDED 2026-09-08 as part of the docket-loop fix (see events/kehillah_
# bet_din_events.txt's own "THE LOOP BUG" header note and common/
# activities/activity_types/kehillah_bet_din_conference.txt's phase
# block). Before this fix the whole docket -- three cases, ten events
# -- resolved back to back inside a single day, because every event
# chained straight into the next one with no delay and nothing ever
# ended the activity's one and only phase. These values are what
# replaces that: one case per activity phase, drawn at random, spread
# over a few days, and a session-wide score read once at the close.
# =====================================================================

# How many cases one session hears. This is a DOCUMENTATION value, not
# a mechanical one -- the count is enforced structurally, by there being
# exactly this many phases in the activity_type. Kept here so the
# intended number lives next to the rest of the docket's tuning; if this
# changes, phases must be added/removed in the activity file to match.
kehillah_bet_din_docket_case_count = 3

# Days between the three events WITHIN one case (host rules -> co-judges
# confer -> verdict recorded). The point of a nonzero value is that a
# case should read as a session being worked through, not as three
# popups in one click. Must stay comfortably under
# kehillah_bet_din_case_phase_days below.
kehillah_bet_din_case_step_days = 3

# Hang guard, not the intended pace. A case's own resolution event
# advances the phase as soon as the verdict lands (kehillah_bet_din_
# advance_docket_effect), so in normal play a phase lasts about
# 2 * kehillah_bet_din_case_step_days + 1 days. This ceiling only
# matters if a player leaves a case's popup unanswered for a long time:
# the phase progresses anyway rather than the activity sitting open
# forever, which is exactly the failure this whole fix is about. Keep it
# generous enough that it is not the normal path (a player who is
# actually playing will always resolve a case first) but finite.
kehillah_bet_din_case_phase_days = 30

# ---------------------------------------------------------------------
# SESSION SCORE.
#
# Two separate scores exist and they are not the same thing:
#   global_var:kehillah_bet_din_tally         -- per CASE, reset at the
#     start of each case, sums the three judges' skill checks for that
#     one case, read by that case's own resolution event to pick a
#     great/good/poor verdict tier. Unchanged by this fix.
#   global_var:kehillah_bet_din_session_score -- per SESSION, reset when
#     the docket opens, and each case's resolution adds the value of the
#     tier it landed in (below). Read once, by kehillah_bet_din.0099 at
#     the close, for the session's overall reward.
#
# Per-case tier values are deliberately small integers rather than the
# raw tally: what the closing reward should care about is "how many
# rulings came out well", not the arithmetic of nine skill checks. Over
# kehillah_bet_din_docket_case_count = 3 cases the score therefore runs
# from +6 (three great rulings) down to -3 (three poor ones).
# ---------------------------------------------------------------------

kehillah_bet_din_case_score_good = 1
kehillah_bet_din_case_score_poor = -1

# =====================================================================
# STANDING -- the community's single overall figure.
#
# ADDED 2026-09-14 (v12), at the user's explicit decision. This SETTLES a
# question two earlier passes deliberately left open, and the change is
# recorded here rather than made silently because of that history:
# docs/spec/v4-regional-communities-and-batei-din.md sidestepped defining
# a blended score (applying takkanah effects symmetrically per-title
# instead), and ROADMAP item 6 still lists "the aggregate-definition
# problem" as open. It is no longer open. Anything that wants "how is
# this community doing, in one number" now reads kehillah_var_standing
# and must not invent a second aggregate alongside it.
#
# THE MEAN, NOT THE SUM. Standing is the arithmetic mean of the three
# pillars, which is the whole reason it is useful as a backend value
# rather than just a display total:
#
#   - It lands on the SAME 0-1000-ish scale each pillar already lives
#     on, so kehillah_band_strained/healthy/flourishing/legendary_
#     threshold apply to it UNCHANGED. No second threshold set to keep
#     in sync, and no risk of the two drifting apart.
#   - It therefore also works with the existing comparison machinery
#     with no new trigger at all:
#       kehillah_pillar_at_least_trigger = {
#           PILLAR = kehillah_var_standing
#           THRESHOLD = kehillah_band_flourishing_threshold
#       }
#     -- that trigger substitutes a bare variable name into var:$PILLAR$
#     (see its own header for the vanilla precedents), and standing is a
#     plain title variable like any pillar, so it just works.
#   - A sum would have been three times the scale of its own parts, which
#     would have meant either a fourth set of thresholds or a permanently
#     confusing number next to the three it is made of.
#
# EQUAL WEIGHTS, deliberately. Prosperity, Stability and Greatness are
# peers in this mod's design -- the three pillars are presented as
# co-equal everywhere from v1 spec section 3 to the Take Stock decision
# -- so weighting one above the others here would quietly assert a
# ranking the design has never made. If a weighting is ever wanted, it
# is a design decision for a spec, not a tweak to this value.
#
# ABSENT PILLARS read as 0 rather than shrinking the divisor: the three
# are created together by kehillah_init_pillars_effect, so the only time
# one is missing is before a community is initialised at all, where a
# standing of 0 is the correct answer, not an average over fewer terms.
# The has_variable guards are still required -- see
# kehillah_pillar_at_least_trigger's "GUARD ADDED 2026-09-08" note for
# the live error-log evidence that a pillar read can genuinely land on a
# not-yet-set variable at the game-start tick.
#
# Scope: a Kehillah title.
# =====================================================================

kehillah_standing_value = {
	value = 0

	if = {
		limit = { has_variable = kehillah_var_prosperity }
		add = var:kehillah_var_prosperity
	}
	if = {
		limit = { has_variable = kehillah_var_stability }
		add = var:kehillah_var_stability
	}
	if = {
		limit = { has_variable = kehillah_var_greatness }
		add = var:kehillah_var_greatness
	}

	divide = 3
}
