# --- The approach. Scope: the mob leader; scope:kehillah_mob_band_target
# is the town (saved by kehillah_mob_set_band_effect -- a nested 'root'
# came up empty, live 2026-10-13). 0 at the gates, 1 days away (60 px),
# 2 a week or two (150 px), 3 within the month (250 px), 4 far.
kehillah_mob_band_value = {
	value = 4
	if = {
		limit = {
			exists = var:kehillah_mob_location
			exists = scope:kehillah_mob_band_target
		}
		if = {
			limit = { var:kehillah_mob_location = scope:kehillah_mob_band_target }
			value = 0
		}
		else_if = {
			limit = {
				var:kehillah_mob_location = {
					squared_distance = { target = scope:kehillah_mob_band_target value < 3600 }
				}
			}
			value = 1
		}
		else_if = {
			limit = {
				var:kehillah_mob_location = {
					squared_distance = { target = scope:kehillah_mob_band_target value < 22500 }
				}
			}
			value = 2
		}
		else_if = {
			limit = {
				var:kehillah_mob_location = {
					squared_distance = { target = scope:kehillah_mob_band_target value < 62500 }
				}
			}
			value = 3
		}
	}
}

# The lord's resolve: the chance (percent) each 4-day tick of the siege
# that the lord sheltering the Jews (var:kehillah_mob_refuge on the
# title) gives them up -- about 5% a month Well-disposed, 15% Indifferent,
# 35% Cold; never if Protective. Scope: the target Kehillah leader.
kehillah_mob_handover_tick_chance_value = {
	value = 2
	if = {
		limit = { primary_title.var:kehillah_mob_refuge ?= { kehillah_outlook_on_jews_value >= 40 } }
		value = 0
	}
	else_if = {
		limit = { primary_title.var:kehillah_mob_refuge ?= { kehillah_outlook_on_jews_value >= 15 } }
		value = 0.7

	else_if = {
		limit = { primary_title.var:kehillah_mob_refuge ?= { kehillah_outlook_on_jews_value <= -15 } }
		value = 5
	}
}

# --- The outcome. Scope: the target Kehillah leader; scope:kehillah_mob_
# leader the host's leader. Severity 10 to 100.
kehillah_mob_severity_value = {
	value = scope:kehillah_mob_leader.var:kehillah_mob_base_severity
	multiply = kehillah_host_protection_mult_value
	# A host worn down on the road does less harm (live 2026-10-14: 488 of
	# 3820 men still struck at full force): x(0.4 + 0.6 x its share left).
	if = {
		limit = {
			exists = scope:kehillah_mob_leader.var:kehillah_mob_size_start
			scope:kehillah_mob_leader.var:kehillah_mob_size_start > 0
		}
		multiply = {
			value = scope:kehillah_mob_leader.var:kehillah_mob_size
			divide = scope:kehillah_mob_leader.var:kehillah_mob_size_start
			max = 1
			multiply = 0.6
			add = 0.4
		}
	}
	primary_title ?= {
		if = {
			limit = { has_variable = kehillah_mob_defended }
			add = -15
		}
		if = {
			limit = { has_variable = kehillah_mob_condemned }
			add = -10
		}
		if = {
			limit = { has_variable = kehillah_mob_stood_at_gate }
			add = -10
		}
		if = {
			limit = { has_variable = kehillah_mob_pleaded }
			multiply = 0.5
		}
		if = {
			limit = { has_variable = kehillah_mob_castle_fell }
			multiply = 0.8
		}
		if = {
			limit = { has_variable = kehillah_mob_scattered_found }
			add = 15
		}
		if = {
			limit = { has_variable = kehillah_mob_armed }
			add = -25
		}
		if = {
			limit = { has_variable = kehillah_mob_armed_failed }
			add = 10
		}
	}
	min = 10
	max = 100
	round = yes
}
