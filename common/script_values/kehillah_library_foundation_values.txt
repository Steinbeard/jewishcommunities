# =====================================================================
# The Beit Midrash's foundation alert (2026-11-11, Daniel: "a circled
# exclamation mark with a tooltip, and it should list which foundational works
# specifically are missing for the open page"). gui/kehillah_domicile_library.gui
# lists a Karaite library's missing founding works with these; the Shas case
# reads kehillah_lib_held_<work> directly. Scope: a library title. The rules
# are kehillah_lib_karaite_missing_count_value's (kehillah_sefer_values.txt):
# each work counts once it circulates.
# =====================================================================

kehillah_lib_missing_anan_value = {
	value = 0
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_anan_sefer_hamitzvot } } }
		value = 1
	}
}

kehillah_lib_missing_qirqisani_value = {
	value = 0
	if = {
		limit = {
			current_date >= 937.1.1
			NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_kitab_alanwar } }
		}
		value = 1
	}
}

kehillah_lib_missing_yefet_value = {
	value = 0
	if = {
		limit = {
			current_date >= 990.1.1
			NOT = { is_target_in_variable_list = { name = kehillah_library_works target = flag:work_yefet_commentary } }
		}
		value = 1
	}
}

# How many a Karaite library lacks; 0 for any other library.
kehillah_lib_karaite_foundation_missing_value = {
	value = 0
	if = {
		limit = { kehillah_lib_is_karaite_library_trigger = yes }
		value = kehillah_lib_karaite_missing_count_value
	}
}
