# 2026-11-10, Daniel. The crisis's threat grade for the UI: 1 Contained to 4 Catastrophic, else 0.
# Scope: any situation. The shared situation window, the situation list, Military > Community
# Safety and the crisis map evaluate it for decrees and Talmud trials too, which get 0.
#
# Shown in Agitation, Armed host, Quarter struck and Recovery. The grade is set when the crisis
# opens (kehillah_crisis_effects.txt) and lowered by the Armed and Struck reckonings, so this
# reads whatever the current phase is being played at. The ending phases (Quarter held, Spared,
# Awaiting judgment, Collapse, Endured) hide it.
kehillah_crisis_ui_grade_value = {
 value = 0
 if = {
  limit = {
   exists = var:kehillah_crisis_title
   OR = {
    situation_current_phase = kehillah_crisis_warning
    situation_current_phase = kehillah_crisis_armed
    situation_current_phase = kehillah_crisis_struck
    situation_current_phase = kehillah_crisis_recovery
   }
   var:kehillah_crisis_title = { has_variable = kehillah_crisis_threat_grade }
  }
  value = var:kehillah_crisis_title.var:kehillah_crisis_threat_grade
 }
 min = 0
 max = 4
}
