# Vanilla's AI-only Evil-faith marriage cutoff. Raise it beyond the highest
# hostility level only when a prospective spouse follows a Jewish rite whose
# interfaith-marriage doctrine permits the match. Other marriages retain the
# vanilla threshold. The rulers' faiths alone are insufficient here: either
# ruler can arrange a marriage for a courtier of a different faith.
faith_hostility_prevents_marriage_level = {
	value = faith_evil_level
	if = {
		limit = {
			OR = {
				AND = {
					exists = scope:secondary_actor
					scope:secondary_actor.rite = { rite_has_doctrine = kehillah_interfaith_marriage_permitted }
				}
				AND = {
					exists = scope:secondary_recipient
					scope:secondary_recipient.rite = { rite_has_doctrine = kehillah_interfaith_marriage_permitted }
				}
			}
		}
		add = 1
	}
}
