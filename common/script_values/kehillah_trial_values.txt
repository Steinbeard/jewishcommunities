# --- Denunciation: the seizure against Quashed ---
kehillah_trial_opening_points = 25
# Monthly, root = situation: the accuser reads his articles again. Larger under a hostile or
# strict judge, smaller under a protective one.
kehillah_trial_accuser_points_value = {
	value = 10
	if = {
		limit = { var:kehillah_trial_judge ?= { kehillah_outlook_band_value <= -2 } }
		add = 8
	}
	else_if = {
		limit = { var:kehillah_trial_judge ?= { kehillah_outlook_band_value = -1 } }
		add = 4
	}
	else_if = {
		limit = { var:kehillah_trial_judge ?= { kehillah_outlook_band_value >= 2 } }
		add = -5
	}
	if = {
		limit = {
			var:kehillah_trial_judge ?= {
				OR = {
					kehillah_teaching_strict_trigger = yes
					kehillah_teaching_exclusionary_trigger = yes
				}
			}
		}
		add = 4
	}
}

# Monthly, root = situation: the judges lean on the scale by their outlook on the Jews.
kehillah_trial_judges_acquittal_value = {
	value = 0
	if = {
		limit = { var:kehillah_trial_judge ?= { kehillah_outlook_band_value >= 1 } }
		value = 8
	}
}
kehillah_trial_judges_expurgation_value = {
	value = 0
	if = {
		limit = {
			var:kehillah_trial_judge ?= {
				kehillah_outlook_band_value <= 0
				kehillah_outlook_band_value >= -1
			}
		}
		value = 6
	}
}
kehillah_trial_judges_condemnation_value = {
	value = 0
	if = {
		limit = { var:kehillah_trial_judge ?= { kehillah_outlook_band_value <= -2 } }
		value = 10
	}
}

# --- The crisis window's pressure line (root = situation; zero hides it) ---
kehillah_trial_window_accuser_value = {
	value = 0
	if = {
		limit = {
			situation_current_phase = kehillah_trial_denunciation
			var:kehillah_trial_accuser ?= { is_alive = yes }
		}
		value = kehillah_trial_accuser_points_value
	}
}

	value = 0

		value = kehillah_trial_judges_condemnation_value
	}
}
