# Founding new communities (docs/spec/v34-outlook-teachings-and-founding.md
# section 1, amended 2026-10-13: "Send Out Settlers").
# A county must be at least this developed to draw settlers.
kehillah_natural_founding_min_development = 8
# Followers a debug founding creates for its founder.
kehillah_natural_founding_followers = 3
# Years before the same community may send out settlers again. Short:
# the overflow leaves with the settlers, so the decision is off until the
# quarter fills again (Daniel 2026-10-13).
kehillah_settlers_cooldown_years = 5

# Scope: a Kehillah leader. How many must leave to bring the quarter back
# to its capacity -- the settlers' party (Daniel 2026-10-13: "reduce the
# population of your community down to capacity and send the overflow").
kehillah_settlers_overflow_value = {
	value = kehillah_courtier_count_value
	subtract = kehillah_courtier_capacity_value
	min = 1
}

# Send Out Settlers destinations (Daniel 2026-10-13): no new community
# within 60 map-pixels (squared) of an existing one -- two or three
# counties -- so settlement spreads rather than clusters.
kehillah_settlers_min_spacing_squared = 3600

# Exiles resettling (2026-10-13): what a community founded by exiles
# starts with -- shared purpose, and the learning they carried.
kehillah_exiles_resettle_stability = 100
kehillah_exiles_resettle_greatness = 40

# Scope: an AI Kehillah leader. Percent chance, per decision check (every
# year since 2026-10-13; was 3 years), that it sends out settlers (kehillah_send_out_settlers_
# decision ai_will_do). Live 2026-10-13: the old weight (the overflow
# alone, +10 if invited) gave 1-11% and no AI founding in 14 + 6 years,
# though 2-5 communities passed every gate. Now: overcrowded 10 + 2 per
# person over capacity; an invitation within reach +15; at most 30.
# Aim: about one AI founding every 3-5 years world-wide (Daniel 2026-10-
# 13), braked by gold (about 40% of communities lack the cost) and the
# soft world ceiling in the decision.
kehillah_settlers_ai_weight_value = {
	value = 0
	if = {
		limit = { NOT = { kehillah_quarter_has_room_trigger = yes } }
		add = 10
		add = {
			value = kehillah_settlers_overflow_value
			multiply = 2
		}
	}
	if = {
		limit = {
			save_temporary_scope_as = kehillah_mother_leader
			any_county = { kehillah_settlers_invited_destination_trigger = yes }
		}
		add = 15
	}
	max = 30
}

# A new community takes root before it sends out settlers of its own
# (2026-10-13 soak: Toledo's daughter Qunkah sent settlers within ~3
# years -- a party can arrive over its new quarter's capacity, which
# would let foundings chain).
kehillah_settlers_new_community_years = 10
