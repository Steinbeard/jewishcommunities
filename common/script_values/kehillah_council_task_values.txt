# MOMENTS
# =====================================================================
# Each January a pillar task rolls for a moment. Chance by the officer's
# seat skill: 10 + 2.5 per point, at most 60 (skill 5: 22%, 20: 60%).
# Scope: the officer.
kehillah_council_moment_max_chance = 60
kehillah_council_moment_chance_diplomacy_value = {
	value = diplomacy
	multiply = 2.5
	add = 10
	max = kehillah_council_moment_max_chance
	round = yes
}
kehillah_council_moment_chance_stewardship_value = {
	value = stewardship
	multiply = 2.5
	add = 10
	max = kehillah_council_moment_max_chance
	round = yes
}
kehillah_council_moment_chance_learning_value = {
	value = learning
	multiply = 2.5
	add = 10
	max = kehillah_council_moment_max_chance
	round = yes
}
# A moment is minor or moderate; moderate more often with skill (2.5% per
# point, at most 50%). The sizes are baseline terms (for scale: a Hekdesh
# tier is +50 to +60, a leadership style +60).
kehillah_council_moment_minor = 15
kehillah_council_moment_moderate = 25
kehillah_council_moment_years = 10
kehillah_council_moment_cap = 3
kehillah_council_moment_moderate_chance_diplomacy_value = { value = diplomacy multiply = 2.5 max = 50 round = yes }
kehillah_council_moment_moderate_chance_stewardship_value = { value = stewardship multiply = 2.5 max = 50 round = yes }
kehillah_council_moment_moderate_chance_learning_value = { value = learning multiply = 2.5 max = 50 round = yes }

# Manage Tzedakah. Scope: a Kehillah leader.
kehillah_tzedakah_task_mult = 1.5
# 1 when the Gabbai keeps the task.
kehillah_tzedakah_task_active_value = {
	value = 0
	cp:councillor_steward ?= {
		if = {
			limit = { is_performing_council_task = task_kehillah_manage_tzedakah }
			value = 1
		}
	}
}
# Giving side: gifts' Greatness and reputation, when Prosperity is Healthy
# or better.
kehillah_tzedakah_gift_mult_value = {
	value = 1
	if = {
		limit = {
			kehillah_tzedakah_task_active_value = 1
			kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_healthy_threshold }
		}
		value = kehillah_tzedakah_task_mult
	}
}
# Receiving side: relief raises more Prosperity below Healthy.
kehillah_tzedakah_relief_prosperity_value = {
	value = kehillah_aid_relief_prosperity
	if = {
		limit = {
			kehillah_tzedakah_task_active_value = 1
			NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_healthy_threshold } }
		}
		multiply = kehillah_tzedakah_task_mult
	}
	round = yes
}
# Receiving side: points on an AI community's quarterly chance to answer
# our appeal: 10 + 10 x the Gabbai's Stewardship scale (10 to 20). Scope:
# the needy community's leader.
kehillah_tzedakah_answer_bonus_value = {
	value = 0
	if = {
		limit = {
			kehillah_tzedakah_task_active_value = 1
			NOT = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_healthy_threshold } }
		}
		value = 10
		cp:councillor_steward ?= {
			add = {
				value = kehillah_council_stewardship_scale
				multiply = 10
			}
		}
	}
	round = yes
}

# ROSH HA-SHMIRAH
# =====================================================================
# Defend the Quarter: the danger shield. -10 x the Martial scale, then
# x1.5 / x2 / x2.5 with Wall and Gates tiers 1 / 2 / 3 (at most -25).
# Scope: a Kehillah leader.
kehillah_danger_watch_value = {
	value = 0
	cp:councillor_marshal ?= {
		if = {
			limit = { is_performing_council_task = task_kehillah_night_watch }
			subtract = {
				value = kehillah_council_martial_scale
				multiply = 10
			}
		}
	}
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_03 } }
		multiply = 2.5
	}
	else_if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_02 } }
		multiply = 2
	}
	else_if = {
		limit = { domicile ?= { has_domicile_building_or_higher = kehillah_wall_01 } }
		multiply = 1.5
	}
	round = yes
}
# Defend the Quarter during a crisis race: Spared points each month,
# 2 + 4 x the Martial scale (2 to 6). Scope: a Kehillah leader.
kehillah_council_watch_crisis_points_value = {
	value = 0
	cp:councillor_marshal ?= {
		if = {
			limit = { is_performing_council_task = task_kehillah_night_watch }
			value = kehillah_council_martial_scale
			multiply = 4
			add = 2
		}
	}
	round = yes
}
# Tooltip display, in the Rosh ha-Shmirah's scope.

		multiply = 2.5
	}
	else_if = {
		limit = { employer ?= { domicile ?= { has_domicile_building_or_higher = kehillah_wall_02 } } }
		multiply = 2
	}
	else_if = {
		limit = { employer ?= { domicile ?= { has_domicile_building_or_higher = kehillah_wall_01 } } }
		multiply = 1.5
	}
	round = yes
}
