# Quarterly chance, while an expulsion edict stands, that an AI host
# demands the community leader's conversion (once per five years).
# Read with scope:kehillah_conversion_host saved. At the base 25 over the
# edict's four quarters, about two hosts in three ask.
kehillah_host_conversion_demand_chance_value = {
	value = 25
	if = {
		limit = { scope:kehillah_conversion_host = { has_trait = zealous } }
		add = 25
	}
	if = {
		limit = { scope:kehillah_conversion_host = { has_trait = cynical } }
		add = -15
	}
	if = {
		limit = { scope:kehillah_conversion_host = { has_trait = compassionate } }
		add = -10
	}
	min = 5
}
