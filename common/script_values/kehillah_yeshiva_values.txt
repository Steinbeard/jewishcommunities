# Founding cost. Above the ordinary tier-3 internal building cost (400):
# founding an academy is the community's great undertaking.
kehillah_found_yeshiva_gold_cost = 500

# Fame (kehillah_yeshiva_fame, on the community title).
kehillah_yeshiva_fame_on_founding = 25
kehillah_yeshiva_fame_seed_worms = 60
kehillah_yeshiva_fame_seed_narbonne = 40
# Low on purpose: the Geonic academies had effectively ended about 1040.
# Baghdad holds the Yeshiva of Pumbedita; Sura (ended about 1034) is lower.
kehillah_yeshiva_fame_seed_baghdad = 20
kehillah_yeshiva_fame_seed_sura = 15
# Lucena: the rising academy of al-Andalus. Jerusalem: the Palestinian
# gaonate, the most prestigious academy still standing in 1066.
kehillah_yeshiva_fame_seed_lucena = 50
kehillah_yeshiva_fame_seed_jerusalem = 80
kehillah_yeshiva_fame_seed_kairouan_867 = 80
kehillah_yeshiva_fame_seed_sura_867 = 90
kehillah_yeshiva_fame_seed_fustat = 60
# Per graduate (2026-10-11): a busy yeshiva ordains several a year.
kehillah_yeshiva_fame_per_ordination = 3 # 3 not 5 (live 2026-10-11: one term gave +29)
kehillah_yeshiva_fame_max = 200
# Quarterly decay: slow while the community is stable, faster while
# Stability is below Healthy (students drift away from a troubled
# community -- V29 section 3.1).
kehillah_yeshiva_fame_decay_stable = 1
# 2 not 4 (live 2026-10-11): Worms opens below Stability Healthy, and at
# -4 its seeded fame bled away in four years.
kehillah_yeshiva_fame_decay_troubled = 2

# Scope: the Kehillah leader. The growing Greatness term (V29 section
# 3.1): half the fame, capped so one yeshiva cannot carry a community.
# Added to the Greatness baseline on top of the Beit Midrash line's flat
# +150, and shown as its own breakdown row (KehillahBdYeshiva).
kehillah_bd_yeshiva_fame_value = {
	value = 0
	if = {
		limit = {
			kehillah_has_yeshiva_trigger = yes
			primary_title ?= { has_variable = kehillah_yeshiva_fame }
		}
		add = primary_title.var:kehillah_yeshiva_fame
		multiply = 0.5
		max = 100
	}
}

# =====================================================================
# V29 section 9 (2026-10-11): renown tiers and alumni (Hold a Zman, also
# section 9, was retired for section 10). docs/spec/v29-yeshiva.md.
# =====================================================================

# Fame tiers (kehillah_yeshiva_fame_tier_value): Modest / Respected /
# Renowned / Illustrious.
kehillah_yeshiva_tier_respected = 40
kehillah_yeshiva_tier_renowned = 80
kehillah_yeshiva_tier_illustrious = 140

# Scope: a Kehillah leader. 0 Modest, 1 Respected, 2 Renowned,
# 3 Illustrious (0 also without a yeshiva).
kehillah_yeshiva_fame_tier_value = {
	value = 0
	if = {
		limit = {
			kehillah_has_yeshiva_trigger = yes
			primary_title ?= { has_variable = kehillah_yeshiva_fame }
		}
		if = {
			limit = { primary_title.var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_illustrious }
			value = 3
		}
		else_if = {
			limit = { primary_title.var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_renowned }
			value = 2
		}
		else_if = {
			limit = { primary_title.var:kehillah_yeshiva_fame >= kehillah_yeshiva_tier_respected }
			value = 1
		}
	}
}

# An alumnus taking a Dayyan seat elsewhere.
kehillah_yeshiva_fame_per_alumnus = 2

# Scope: a Kehillah leader. Fame still needed for the next tier (0 at
# Illustrious). Drives the words of KehillahYeshivaNextTier.
kehillah_yeshiva_fame_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_yeshiva_fame } }
		value = primary_title.var:kehillah_yeshiva_fame
	}
}
kehillah_yeshiva_fame_to_next_value = {
	value = 0
	if = {
		limit = { kehillah_yeshiva_fame_tier_value = 0 }
		value = kehillah_yeshiva_tier_respected
		subtract = kehillah_yeshiva_fame_value
	}
	else_if = {
		limit = { kehillah_yeshiva_fame_tier_value = 1 }
		value = kehillah_yeshiva_tier_renowned
		subtract = kehillah_yeshiva_fame_value
	}
	else_if = {
		limit = { kehillah_yeshiva_fame_tier_value = 2 }
		value = kehillah_yeshiva_tier_illustrious
		subtract = kehillah_yeshiva_fame_value
	}
	min = 0
}

# V29 section 10 (2026-10-11): the living yeshiva.

# Enrolment: 16 or older, Learning ABOVE this.
kehillah_yeshiva_min_learning = 8
kehillah_yeshiva_min_age = 16
# Two years present.
kehillah_yeshiva_quarters_to_graduate = 8
# Track XP per quarter present: the yeshiva's specialization, and each
# other track.
kehillah_yeshiva_xp_specialization = 8
kehillah_yeshiva_xp_other = 2
# Quarterly chance (percent) that a travelling student arrives, by tier.
kehillah_yeshiva_arrival_chance_modest = 15
kehillah_yeshiva_arrival_chance_respected = 25
kehillah_yeshiva_arrival_chance_renowned = 35
kehillah_yeshiva_arrival_chance_illustrious = 50
# Quarterly chance (percent) of a study event for a PLAYER student, in
# the seasons without a fixed event (first, halfway, last). 75 not 50
# (2026-11-16, Daniel wanted a fuller stay at the yeshiva).
kehillah_yeshiva_player_event_chance = 75
# Extra track XP a season for the focus chosen before the Rosh Yeshiva.
# 6 not 3 (2026-11-16): mastery of the focus now decides when a player
# is ordained.
kehillah_yeshiva_focus_bonus_xp = 6

# A PLAYER student's semicha (2026-11-16, Daniel: "three years is way too
# long ... dependent on how much xp you get", then "cap the activity at 1
# year max"). Mastery (var:kehillah_yeshiva_mastery) counts the XP he
# earns in his focus while at the yeshiva. Once it reaches the target,
# after the minimum stay and with enough Learning, he is ordained at that
# season's tick -- not at the yearly winter class. Tuned so four seasons
# (one year) always suffice: the worst case, a mismatched focus, gains
# 6 + 2 = 8 a season (32 in four); a matching focus under a weak head 9,
# under a master 14 (three seasons). Event XP into the focus counts too.
# AI students keep the 8-season winter rule.
kehillah_yeshiva_semicha_mastery = 30
kehillah_yeshiva_semicha_min_seasons = 2
# The examination (.0305) at half the target, the nearly-ready event
# (.0306) at three quarters.
kehillah_yeshiva_mastery_exam = 15
kehillah_yeshiva_mastery_nearly = 22
# How long the study planner's activity holds a player student at his
# yeshiva at most: one year (Daniel, 2026-11-16). Semicha or leaving ends
# it sooner. A student kept back only by too little Learning goes home
# enrolled, and may return.
kehillah_yeshiva_study_hold_days = 365
# After semicha, "stay a while longer" before the road home.
kehillah_yeshiva_linger_days = 60
# Sefarim at the yeshiva (2026-11-16): each book first mastered on the
# benches adds this much mastery toward semicha, and while there the
# Learn Torah course's mastery chance rises by the bonus.
kehillah_yeshiva_sefer_mastery_points = 10
kehillah_yeshiva_sefer_success_bonus = 15
# The examination's Learning roll (kehillah_yeshiva.0305.a).
kehillah_yeshiva_exam_difficulty = 12

# Scope: a student, in the yeshiva tick (scope:kehillah_head_teaching_xp
# set). The mastery one season present adds: the focus bonus, plus what
# the head teaches when the focus is the yeshiva's specialization, else
# the small share every other subject gets.
kehillah_yeshiva_season_mastery_value = {
	value = kehillah_yeshiva_focus_bonus_xp
	if = {
		limit = {
			exists = var:kehillah_yeshiva_focus
			var:kehillah_yeshiva_enrolled ?= {
				var:kehillah_yeshiva_specialization ?= root.var:kehillah_yeshiva_focus
			}
		}
		add = scope:kehillah_head_teaching_xp
	}
	else_if = {
		# No specialization set: the yeshiva teaches Talmud first.
		limit = {
			var:kehillah_yeshiva_focus ?= flag:talmudics
			var:kehillah_yeshiva_enrolled ?= { NOT = { has_variable = kehillah_yeshiva_specialization } }
		}
		add = scope:kehillah_head_teaching_xp
	}
	else = {
		add = kehillah_yeshiva_xp_other
	}
}

# 2026-11-16: the study planner (activity_kehillah_yeshiva_study).
# root = the province scored; scope:host = the student. The yeshiva's
# fame, less a penalty for distance from where he stands now, so a near,
# good yeshiva outranks a famous one across the sea.
kehillah_yeshiva_study_province_score_value = {
	value = 0
	every_in_global_list = {
		variable = kehillah_registered_communities
		limit = {
			has_variable = kehillah_yeshiva_fame
			exists = holder
			holder = { domicile ?= { domicile_location = root } }
		}
		add = var:kehillah_yeshiva_fame
	}
	if = {
		limit = { "scope:host.location.squared_distance(root)" > squared_distance_medium }
		subtract = 10
	}
	if = {
		limit = { "scope:host.location.squared_distance(root)" > squared_distance_large }
		subtract = 15
	}
	if = {
		limit = { "scope:host.location.squared_distance(root)" > squared_distance_huge }
		subtract = 15
	}
	if = {
		limit = { "scope:host.location.squared_distance(root)" > squared_distance_massive }
		subtract = 20
	}
	min = 1
}
# Scope: a Kehillah leader. Students on the roll (panel).
kehillah_yeshiva_student_count_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable_list = kehillah_yeshiva_students } }
		value = {
			value = 0
			primary_title = {
				every_in_list = {
					variable = kehillah_yeshiva_students
					add = 1
				}
			}
		}
	}
}
