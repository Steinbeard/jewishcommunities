# V47 heads of rite (docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).
# Scope: a would-be head of rite. 2026-11-08 tenet mechanics: a rite that
# holds Ge'onut, the authority of the academies, acclaims at piety level 3.
kehillah_head_of_rite_min_piety_level = {
	value = 4
	if = {
		limit = { rite ?= { rite_has_parameter = kehillah_tenet_geonut_academy } }
		value = 3
	}
}
kehillah_head_of_rite_declined_years = 10
# Scope: a community title. How eminent its leader is, to choose among
# several who qualify: the academy's fame, his Learning, his devotion.
kehillah_head_of_rite_title_score_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_yeshiva_fame }
		add = var:kehillah_yeshiva_fame
	}
	holder ?= {
		add = {
			value = learning
			multiply = 3
		}
		add = {
			value = piety_level
			multiply = 20
		}
	}
}
kehillah_head_of_rite_greatness = 100
kehillah_gaon_office_prestige_cost = 1000
kehillah_gaon_office_piety_cost = 500
kehillah_gaon_office_greatness = 100
