# Leadership mandate: bounded baseline support; no instant pillar or currency payout.
# 2026-11-20 (pillars spec R2/R10, Daniel): the style's flat +60 to its
# pillar is retired. A style now makes the leader's own actions on that
# pillar 20% stronger (kehillah_change_pillar_effect,
# kehillah_leadership_mandate_boost) and gives a style modifier on the
# leader (kehillah_leadership_style_apply_effect). The breakdown lines key
# on these values being above 0, so they simply disappear.
kehillah_leadership_baseline_bonus = 0
kehillah_leadership_greatness_bonus = {
 value = 0
 if = { limit = { primary_title = { kehillah_leadership_is_scholarly_trigger = yes } } add = kehillah_leadership_baseline_bonus }
}
kehillah_leadership_stability_bonus = {
 value = 0
 if = { limit = { primary_title = { kehillah_leadership_is_political_trigger = yes } } add = kehillah_leadership_baseline_bonus }
}
kehillah_leadership_prosperity_bonus = {
 value = 0
 if = { limit = { primary_title = { kehillah_leadership_is_commercial_trigger = yes } } add = kehillah_leadership_baseline_bonus }
}
kehillah_leadership_learning_weight = {
 value = 3
 if = { limit = { scope:title = { kehillah_leadership_is_political_trigger = yes } } value = 1 }
 if = { limit = { scope:title = { kehillah_leadership_is_commercial_trigger = yes } } value = 1 }
}
kehillah_leadership_stewardship_weight = {
 value = 2
 if = { limit = { scope:title = { kehillah_leadership_is_political_trigger = yes } } value = 1 }
 if = { limit = { scope:title = { kehillah_leadership_is_commercial_trigger = yes } } value = 3 }
}
kehillah_leadership_diplomacy_weight = {
 value = 0.5
 if = { limit = { scope:title = { kehillah_leadership_is_political_trigger = yes } } value = 3 }
 if = { limit = { scope:title = { kehillah_leadership_is_commercial_trigger = yes } } value = 1 }
}

# A positive pillar change from an action on the style's own pillar is
# multiplied by this (kehillah_change_pillar_effect).
kehillah_leadership_mandate_boost = 1.2

# Political leadership: a named shield in the danger breakdown.
kehillah_danger_political_leadership_value = {
	value = 0
	if = {
		limit = {
			primary_title ?= {
				has_variable = kehillah_leadership_style
				kehillah_leadership_is_political_trigger = yes
			}
		}
		add = -5
	}
}
