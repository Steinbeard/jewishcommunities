# =====================================================================
# BANKING SERVICES -- a source of the host's outlook on the Jews
# (2026-11-20, Daniel: "a ruler outlook bonus for 'banking services' which
# is reduced by other sources of lending from non-Jews").
# docs/spec/2026-11-19-trade-dlc-lending-goods-and-districts.md section 6.
#
# A ruler whose realm holds a lending Kehillah values the Jews for it
# (kehillah_banking_services, refreshed quarterly by every lending community
# for its county holder and top liege). Christian credit in his realm makes
# the service less needed: merchant companies and holy orders that lend from
# his lands (kehillah_rival_credit, counted yearly), and his own loans from
# non-Jewish lenders. The term never goes below zero: rival credit erodes
# the reason to value the Jews, it does not by itself make him hostile.
#
# Loans are vanilla 1.21's (story_cycle_loan; the lender's owes_me_money
# list). See common/scripted_effects/kehillah_banking_effects.txt.
# Scope: any character.
# =====================================================================

kehillah_outlook_banking_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		add = diarch.kehillah_outlook_banking_value
	}
	else = {
		add = kehillah_outlook_banking_own_value
	}
}

kehillah_outlook_banking_own_value = {
	value = 0
	if = {
		limit = {
			is_alive = yes
			has_variable = kehillah_banking_services
		}
		add = kehillah_banking_services_base
		if = {
			limit = { has_variable = kehillah_rival_credit }
			subtract = {
				value = var:kehillah_rival_credit
				multiply = kehillah_rival_credit_per_lender
			}
		}
		if = {
			limit = { kehillah_has_non_jewish_loan_trigger = yes }
			subtract = kehillah_rival_credit_own_loan
		}
		# 2026-11-20 (Bankers' Row): a Court Jew's Office in his realm.
		if = {
			limit = { has_variable = kehillah_banking_court_jew }
			add = kehillah_banking_court_jew_bonus
		}
		min = 0
	}
}

# The single breakdown line "Value of Jewish banking": the service, net of
# Christian credit, plus his own debts to Jewish lenders.
kehillah_outlook_jewish_banking_value = {
	value = kehillah_outlook_banking_value
	add = kehillah_outlook_debts_value
}

# Outlook points for a realm served by a lending community.
kehillah_banking_services_base = 10
# Each Christian (non-Jewish) lending house based in his realm.
kehillah_rival_credit_per_lender = 3
# He already borrows from a non-Jewish lender.
kehillah_rival_credit_own_loan = 4

# --- Debts to Jewish lenders through vanilla loans (extends the S8 term in
# kehillah_outlook_values.txt). Welcome at first, resented as they age.
kehillah_outlook_vanilla_jewish_debt_value = {
	value = 0
	if = {
		limit = {
			any_owned_story = {
				story_type = story_cycle_loan
				var:lender ?= { is_jewish_character_trigger = yes }
				var:years_since >= 5
			}
		}
		value = -15
	}
	else_if = {
		limit = {
			any_owned_story = {
				story_type = story_cycle_loan
				var:lender ?= { is_jewish_character_trigger = yes }
				var:years_since >= 1
			}
		}
		value = -5
	}
}

# The Court Jew's Office's addition to the host's value of Jewish banking.
kehillah_banking_court_jew_bonus = 5

# A community loan repudiated as usury (pillars spec E1).
kehillah_loan_repudiated_prosperity = -30
kehillah_loan_repudiated_stability = -10

# =====================================================================
# THE LENDING TABLE (2026-11-25, Daniel: "have the lending table impact
# how effective lending is"). The Bankers' Row track -- Moneychangers'
# Tables, Bankers' Row, The Exchange -- makes the community a better
# lender in two ways:
#   1. better terms: an open-interest loan the community takes up owes
#      this much more (kehillah_community_adopt_loans_effect, beside the
#      Ne'eman's Pledge Chest);
#   2. a known bank: rulers accept Loan Gold more readily
#      (kehillah_loan_money_interaction ai_accept).
# Scope: a Kehillah leader, the lender.
# =====================================================================
kehillah_lending_table_tier_value = {
	# Physical tiers are usable only while the native family appointment operates.
	value = 0
	if = {
		limit = { kehillah_business_bankers_staffed_trigger = yes domicile ?= { has_domicile_building_or_higher = kehillah_bankers_row_03 } }
		value = 3
	}
	else_if = {
		limit = { kehillah_business_bankers_staffed_trigger = yes domicile ?= { has_domicile_building_or_higher = kehillah_bankers_row_02 } }
		value = 2
	}
	else_if = {
		limit = { kehillah_business_bankers_staffed_trigger = yes domicile ?= { has_domicile_building_or_higher = kehillah_bankers_row_01 } }
		value = 1
	}
}
# Extra share of the debt per tier, on open-interest loans.
kehillah_lending_table_interest_per_tier = 0.05
kehillah_lending_table_interest_rate_value = {
	value = kehillah_lending_table_tier_value
	multiply = kehillah_lending_table_interest_per_tier
}
# Borrower acceptance per tier.
kehillah_lending_table_accept_per_tier = 10
kehillah_lending_table_accept_value = {
	value = kehillah_lending_table_tier_value
	multiply = kehillah_lending_table_accept_per_tier
}
# All better terms the community writes: the lending table plus the
# Ne'eman's Keep the Pledge Chest (kehillah_council_task_values.txt).
kehillah_loan_better_terms_rate_value = {
	value = kehillah_lending_table_interest_rate_value
	add = kehillah_pledge_chest_interest_rate_value
	add = kehillah_household_credit_terms_rate_value
}
