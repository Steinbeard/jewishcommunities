# --- Needs -----------------------------------------------------------------
# How long a need stays open (years), and when the reminder goes out
# (months after opening).
kehillah_aid_open_years = 2
kehillah_aid_reminder_months = 18
# Gold to be raised. Relief: a community in Crisis Prosperity (AI, or a
# player's appeal). Refugees: a community that took refugees in. Exiles: an
# expelled community. Rebuild: a struck quarter, by the crisis's threat
# grade (1-4): 175 / 250 / 325 / 400.
kehillah_aid_relief_size = 150
kehillah_aid_refugee_size = 100
kehillah_aid_exiles_size = 200
kehillah_aid_rebuild_size_value = {
	value = 100
	if = {
		limit = { primary_title ?= { has_variable = kehillah_crisis_threat_grade } }
		add = {
			value = primary_title.var:kehillah_crisis_threat_grade
			multiply = 75
		}
	}
	else = { add = 150 }
}
# Scope: a Kehillah leader writing an appeal. What the appeal asks for:
# rebuilding while the crisis's Recovery is open, otherwise relief.
kehillah_aid_appeal_size_value = {
	value = kehillah_aid_relief_size
	if = {
		limit = { kehillah_crisis_recovery_open_trigger = yes }
		value = kehillah_aid_rebuild_size_value
	}
}
kehillah_aid_appeal_cooldown_years = 3
kehillah_aid_envoy_cost = 25
# Scope: the envoy. How much likelier AI communities are to answer an
# appeal he carries (quarterly chance points): 2 per point of diplomacy
# above 8, at most 20.
kehillah_aid_envoy_bonus_value = {
	value = diplomacy
	subtract = 8
	multiply = 2
	min = 0
	max = 20
}
# An AI community in Crisis Prosperity asks at most this often (years).
kehillah_aid_relief_cooldown_years = 5

# --- Reach -------------------------------------------------------------------
# Squared map distances from the giver's domicile to the needy character.
# Healthy communities answer their neighbours (one Holy Roman Empire,
# vanilla squared_distance_huge); Flourishing ones, or any community for an
# appeal carried by an envoy, twice as far; Legendary ones, anywhere.
kehillah_aid_range = 722500
kehillah_aid_range_far = 2890000

# --- Shares ------------------------------------------------------------------
# Scope: a Kehillah leader. How many shares his community is expected to
# carry: one when Healthy (or less), two when Flourishing, three when
# Legendary.
kehillah_aid_weight_value = {
	value = 1
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_legendary_threshold } }
		value = 3
	}
	else_if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_flourishing_threshold } }
		value = 2
	}
}
# Every gift is at least this much, unless less remains.
kehillah_aid_min_share = 25
# Scope: a Kehillah leader; scope:kehillah_aid_need saved. His community's
# share: the need divided among the expected communities' shares (at least
# two shares' worth, so one community is never asked for all of it), times
# his own shares; never more than remains.
kehillah_aid_my_share_value = {
	value = 0
	if = {
		limit = { exists = scope:kehillah_aid_need.var:kehillah_aid_size }
		value = scope:kehillah_aid_need.var:kehillah_aid_size
		multiply = kehillah_aid_weight_value
		if = {
			limit = { exists = scope:kehillah_aid_need.var:kehillah_aid_expected_weight }
			divide = {
				value = scope:kehillah_aid_need.var:kehillah_aid_expected_weight
				min = 2
			}
		}
		else = { divide = 2 }
		min = kehillah_aid_min_share
		max = scope:kehillah_aid_need.kehillah_aid_remaining_value
	}
	round = yes
}
# Scope: the needy character. What is still to be raised.
kehillah_aid_remaining_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_aid_size }
		value = var:kehillah_aid_size
		if = {
			limit = { has_variable = kehillah_aid_raised }
			subtract = var:kehillah_aid_raised
		}
		min = 0
	}
	round = yes
}
# Scope: the needy character. Reciprocity: a community known for giving is
# answered more readily (AI chance per quarter, -40..40).
kehillah_aid_reciprocity_value = {
	value = 0
	if = {
		limit = { exists = var:kehillah_aid_for_title.var:kehillah_giving_reputation }
		value = var:kehillah_aid_for_title.var:kehillah_giving_reputation
		multiply = 0.4
	}
	else_if = {
		limit = { exists = var:kehillah_aid_home_title.var:kehillah_giving_reputation }
		value = var:kehillah_aid_home_title.var:kehillah_giving_reputation
		multiply = 0.4
	}
	round = yes
}
