# =====================================================================
# MUTUAL AID (2026-11-20). docs/spec/2026-11-20-mutual-aid.md.
# Needs (ransom, relief, refugees) live on a character; every Kehillah in
# range with the means is expected to answer; a community's reputation for
# giving (title var kehillah_giving_reputation, -100..100) records it.
# =====================================================================

# --- Needs -----------------------------------------------------------------
# How long a need stays open (years).
kehillah_aid_open_years = 2
# A relief need: the community in Crisis Prosperity, or one sheltering
# refugees. Gold to be raised.
kehillah_aid_relief_size = 150
kehillah_aid_refugee_size = 100
# A community may open a relief need at most once in this many years.
kehillah_aid_relief_cooldown_years = 5
# Range: who is expected to answer (squared map distance from the need).
# (A number, not an alias of squared_distance_huge: `a = b` script values
# are rejected by the game.) 850 map pixels: about one Holy Roman Empire.
kehillah_aid_range = 722500
# Never less than this per expected giver.
kehillah_aid_min_share = 25

# Scope: the needy character. One expected community's share, rounded.
kehillah_aid_share_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_aid_size }
		value = var:kehillah_aid_size
		if = {
			limit = { has_variable = kehillah_aid_expected_count }
			divide = {
				value = var:kehillah_aid_expected_count
				min = 1
			}
		}
		min = kehillah_aid_min_share
		max = var:kehillah_aid_size
	}
	round = yes
}

# Scope: the needy character. Gold still to raise.
kehillah_aid_remaining_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_aid_size }
		value = var:kehillah_aid_size
		if = {
			limit = { has_variable = kehillah_aid_raised }
			subtract = var:kehillah_aid_raised
		}
		min = 0
	}
	round = yes
}

# --- Reputation for giving --------------------------------------------------
kehillah_giving_rep_share = 6        # answered with a share
kehillah_giving_rep_full = 12        # paid the remainder / redeemed a captive
kehillah_giving_rep_refused = -6     # expected, never answered
kehillah_giving_rep_take_in = 6      # took refugees in
kehillah_giving_rep_road = 2         # silver for the road
kehillah_giving_rep_turn_away = -6   # closed the gate
kehillah_giving_rep_emissary_generous = 10
kehillah_giving_rep_emissary_modest = 4
kehillah_giving_rep_emissary_decline = -5
# Yearly fade toward zero: old deeds fade, a new leader can repair it.
kehillah_giving_rep_decay = 0.85

# Tiers (character modifiers on the leader).
kehillah_giving_rep_renowned = 40
kehillah_giving_rep_generous = 15
kehillah_giving_rep_stingy = -15
kehillah_giving_rep_disgraced = -40

# Greatness for a gift, and the cost of a refusal.
kehillah_aid_gift_greatness = 10
kehillah_aid_full_greatness = 20
kehillah_aid_refused_greatness = -10
# Prosperity restored to a community whose relief was fully raised.
kehillah_aid_relief_prosperity = 40

# --- The emissary from the Land of Israel -----------------------------------
# Yearly chance, percent, for a community at Healthy Prosperity or better.
kehillah_emissary_yearly_chance = 12
kehillah_emissary_generous_gold = 100
kehillah_emissary_modest_gold = 30
