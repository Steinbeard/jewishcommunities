# --- Needs -----------------------------------------------------------------
# How long a need stays open (years), and when the reminder goes out
# (months after opening).
kehillah_aid_open_years = 2
kehillah_aid_reminder_months = 18
# Gold to be raised (v2.2, Daniel: "instead of fixed values ... scale based
# on something reasonable"). Every sum is in the needy leader's scope and
# grows with what must be carried -- the community's families, the damage
# done -- times the era, as vanilla's own gold values do
# (kehillah_aid_era_value: x0.75 tribal, x1 early, x1.5 high, x3 late
# medieval). At 1066 (early medieval) a three-family community asks about
# 140 for relief.
#   relief   -- 50, plus 30 per householder family
#   rebuild  -- 60 per threat grade, plus 25 per building lost and 5 per
#               death in the attack
#   exiles   -- 50, plus 60 per householder family (all must be resettled)
#   refugees -- 100
# A ransom is vanilla's own price.
kehillah_aid_era_value = {
	value = 1
	if = {
		limit = { exists = culture }
		culture = {
			if = {
				limit = { has_cultural_era_or_later = culture_era_late_medieval }
				multiply = 3
			}
			else_if = {
				limit = { has_cultural_era_or_later = culture_era_high_medieval }
				multiply = 1.5
			}
			else_if = {
				limit = { has_cultural_era_or_later = culture_era_early_medieval }
				multiply = 1
			}
			else = { multiply = 0.75 }
		}
	}
}
# Scope: the needy leader. His community's householder families.
kehillah_aid_families_value = {
	value = 0
	if = {
		limit = { exists = primary_title }
		value = primary_title.kehillah_household_count_value
	}
}
kehillah_aid_relief_size = {
	value = 30
	multiply = kehillah_aid_families_value
	add = 50
	multiply = kehillah_aid_era_value
	round = yes
}
kehillah_aid_refugee_size = {
	value = 100
	# Hekdesh's Tamchui (soup kitchen) feeds them already.
	if = {
		limit = { domicile ?= { has_domicile_parameter = kehillah_has_tamchui } }
		value = 60
	}
	multiply = kehillah_aid_era_value
	round = yes
}
kehillah_aid_exiles_size = {
	value = 60
	multiply = kehillah_aid_families_value
	add = 50
	multiply = kehillah_aid_era_value
	round = yes
}
# 2026-11-25: built from named parts, with no inline add = { value = ...
# multiply = ... } blocks. The inline form returned 1500 in the live game for
# a Troyes quarter (grade 1, one building, one dead), where the sum is 90; each
# part on its own read correctly.
# Scope: the needy leader. 60 per threat grade (grade 2 if none is recorded).
kehillah_aid_rebuild_grade_part_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_crisis_threat_grade } }
		add = primary_title.var:kehillah_crisis_threat_grade
	}
	else = { add = 2 }
	multiply = 60
}
# Scope: the needy leader. 25 per building lost in the attack.
kehillah_aid_rebuild_buildings_part_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_crisis_loss_buildings } }
		add = primary_title.var:kehillah_crisis_loss_buildings
	}
	multiply = 25
}
# Scope: the needy leader. 5 per death in the attack.
kehillah_aid_rebuild_dead_part_value = {
	value = 0
	if = {
		limit = { primary_title ?= { has_variable = kehillah_crisis_loss_dead } }
		add = primary_title.var:kehillah_crisis_loss_dead
	}
	multiply = 5
}
kehillah_aid_rebuild_size_value = {
	value = kehillah_aid_rebuild_grade_part_value
	add = kehillah_aid_rebuild_buildings_part_value
	add = kehillah_aid_rebuild_dead_part_value
	multiply = kehillah_aid_era_value
	round = yes
}
# Scope: a Kehillah leader writing an appeal. What the appeal asks for:
# rebuilding while the crisis's Recovery is open, otherwise relief.
kehillah_aid_appeal_size_value = {
	value = kehillah_aid_relief_size
	if = {
		limit = { kehillah_crisis_recovery_open_trigger = yes }
		value = kehillah_aid_rebuild_size_value
	}
}
kehillah_aid_appeal_cooldown_years = 3
kehillah_aid_envoy_cost = 25
# Scope: the envoy (the Shtadlan, v2.2). How much likelier AI communities
# are to answer an appeal he carries (quarterly chance points): 2 per
# point of diplomacy above 8, at most 20.
kehillah_aid_envoy_bonus_value = {
	value = diplomacy
	subtract = 8
	multiply = 2
	min = 0
	max = 20
}
# An AI community in Crisis Prosperity asks at most this often (years).
kehillah_aid_relief_cooldown_years = 5

# --- Reach -------------------------------------------------------------------
# Squared map distances from the giver's domicile to the needy character.
# Healthy communities answer their neighbours (one Holy Roman Empire,
# vanilla squared_distance_huge); Flourishing ones, or any community for an
# appeal carried by an envoy, twice as far; Legendary ones, anywhere.
kehillah_aid_range = 722500
kehillah_aid_range_far = 2890000

# --- Shares ------------------------------------------------------------------
# Scope: a Kehillah leader. How many shares his community is expected to
# carry: one when Healthy (or less), two when Flourishing, three when
# Legendary.
kehillah_aid_weight_value = {
	value = 1
	if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_legendary_threshold } }
		value = 3
	}
	else_if = {
		limit = { kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_flourishing_threshold } }
		value = 2
	}
}
# Every gift is at least this much, unless less remains.
kehillah_aid_min_share = 25
# Scope: a Kehillah leader; scope:kehillah_aid_need saved. His community's
# share: the need divided among the expected communities' shares (at least
# two shares' worth, so one community is never asked for all of it), times
# his own shares; never more than remains.
kehillah_aid_my_share_value = {
	value = 0
	if = {
		limit = { exists = scope:kehillah_aid_need.var:kehillah_aid_size }
		value = scope:kehillah_aid_need.var:kehillah_aid_size
		multiply = kehillah_aid_weight_value
		divide = kehillah_aid_share_divisor_value
		min = kehillah_aid_min_share
		max = scope:kehillah_aid_need.kehillah_aid_remaining_value
	}
	round = yes
}
# Scope: a Kehillah leader; scope:kehillah_aid_need saved. "Send what we
# can" (2026-11-25, Daniel: an option to give less than is asked when the
# share is beyond the treasury): half of what the community holds, below its
# share. It counts for half the reputation of a share
# (kehillah_aid_partial_rep_mult, in kehillah_aid_give_effect).
kehillah_aid_partial_gift_value = {
	value = gold
	multiply = 0.5
	max = kehillah_aid_my_share_value
	min = 0
	floor = yes
}
# The smallest "what we can" worth sending.
kehillah_aid_partial_min = 10
kehillah_aid_partial_rep_mult = 0.5
# Scope: a Kehillah leader; scope:kehillah_aid_need saved. The shares the
# need is divided among: the expected communities' shares, never fewer than
# two. (2026-11-25: a named value, not an inline divide = { ... min = 2 }
# block; see kehillah_aid_rebuild_size_value.)
kehillah_aid_share_divisor_value = {
	value = 0
	if = {
		limit = { exists = scope:kehillah_aid_need.var:kehillah_aid_expected_weight }
		add = scope:kehillah_aid_need.var:kehillah_aid_expected_weight
	}
	min = 2
}
# Scope: the needy character. What is still to be raised.
kehillah_aid_remaining_value = {
	value = 0
	if = {
		limit = { has_variable = kehillah_aid_size }
		value = var:kehillah_aid_size
		if = {
			limit = { has_variable = kehillah_aid_raised }
			subtract = var:kehillah_aid_raised
		}
		min = 0
	}
	round = yes
}
# Scope: the needy character. Reciprocity: a community known for giving is
# answered more readily (AI chance per quarter, -40..40).
kehillah_aid_reciprocity_value = {
	value = 0
	if = {
		limit = { exists = var:kehillah_aid_for_title.var:kehillah_giving_reputation }
		value = var:kehillah_aid_for_title.var:kehillah_giving_reputation
		multiply = 0.4
	}
	else_if = {
		limit = { exists = var:kehillah_aid_home_title.var:kehillah_giving_reputation }
		value = var:kehillah_aid_home_title.var:kehillah_giving_reputation
		multiply = 0.4
	}
	round = yes
}
