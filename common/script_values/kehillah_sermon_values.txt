# The reward is smaller than completing a community goal (40) and scales with
# the preacher's Parshanut (a Karaite's Religious Scholar), plus his texts.
kehillah_sermon_pillar_gain_novice = 10
kehillah_sermon_pillar_gain_studied = 15
kehillah_sermon_pillar_gain_adept = 20
kehillah_sermon_pillar_gain_master = 25
kehillah_sermon_piety_gain = 25
kehillah_sermon_parshanut_xp = 3

# Scope: the preacher. His base reward by Parshanut experience.
kehillah_sermon_parshanut_gain_value = {
	value = kehillah_sermon_pillar_gain_novice
	if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 100 }
			}
		}
		value = kehillah_sermon_pillar_gain_master
	}
	else_if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 65 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 65 }
			}
		}
		value = kehillah_sermon_pillar_gain_adept
	}
	else_if = {
		limit = {
			OR = {
				has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 30 }
				kehillah_karaite_scholar_xp_trigger = { VALUE = 30 }
			}
		}
		value = kehillah_sermon_pillar_gain_studied
	}
}

# 2026-11-11 (Daniel): the delivery is a Diplomacy duel (kehillah_sermon_
# roll_effect) against a decent speaker (12), weighted 50/50 and moved 3 per
# point of difference, plus each text's own gain (grounded 8, newer 4) and
# the preacher's manner. A ban weighs on it: the benches may stand empty.
kehillah_sermon_roll_trait_good = 10
kehillah_sermon_roll_trait_small = 5
kehillah_sermon_roll_trait_small_bad = -5
kehillah_sermon_roll_trait_bad = -15
kehillah_sermon_roll_niddui = -20
kehillah_sermon_roll_herem = -40
# A sermon that falls flat: no effect, and the preacher's stress.
kehillah_sermon_flat_stress = 15

# Who may preach: outside Rabbinism, Learning this high (and the Torah mastered).
kehillah_sermon_learning_threshold = 12

# Each text cited adds to the sermon: the most established more, a newer one
# less (but a newer one is helped toward its own standing).
kehillah_sermon_cite_grounded_gain = 8
kehillah_sermon_cite_newer_gain = 4

# Seek the peace of the city: the host's outlook on the Jews (within its +/-30
# cap, kehillah_shift_outlook_effect); his respect is the sermon's strength.
kehillah_sermon_peace_outlook = 2
# In a time of trouble: stress lifted from the community's Jews.
kehillah_sermon_trouble_stress = -20
# A maggid's guest sermon: Prestige per point of the sermon's strength; an
# appeal: gold per point (divided).
kehillah_sermon_guest_prestige_mult = 3
kehillah_sermon_appeal_gold_divisor = 2
