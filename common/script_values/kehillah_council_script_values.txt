# =====================================================================
# Kehillah council tunables. Design: docs/spec/v26-kehillah-council-and-
# baalei-batim.md. Tasks: common/council_tasks/kehillah_council_tasks.txt.
#
# No direct pillar bonuses (Daniel, 2026-10-01): these values drive
# concrete effects -- income, construction, petitions, the bet din
# cooldown, the watch -- and the pillars respond through their existing
# inputs. (The first build's kehillah_bd_council_*_value baseline terms
# were removed 2026-10-02.)
# =====================================================================

# --- Skill scales, 0..1: the councillor's seat skill, capped at 20.
# Scope: the councillor (council_owner_modifier scale scope).
kehillah_council_skill_cap = 20

kehillah_council_stewardship_scale = {
	value = stewardship
	max = kehillah_council_skill_cap
	divide = kehillah_council_skill_cap
}
kehillah_council_diplomacy_scale = {
	value = diplomacy
	max = kehillah_council_skill_cap
	divide = kehillah_council_skill_cap
}
kehillah_council_learning_scale = {
	value = learning
	max = kehillah_council_skill_cap
	divide = kehillah_council_skill_cap
}

# --- Shtadlan: Petition the Host. Monthly chance (percent) that the host
# raises the Jewish settlement policy one rung, given the policy is not
# Encouraged and its review is off cooldown (checked in the task).
# Scope: the Shtadlan; scope:kehillah_petition_host and
# scope:councillor_liege set. Up to 2%/month at Diplomacy 20 (~22%/year),
# x1.5 when the host's opinion of the leader is 50+.
kehillah_shtadlan_petition_max_monthly_chance = 2

kehillah_shtadlan_petition_monthly_chance_value = {
	value = kehillah_shtadlan_petition_max_monthly_chance
	multiply = kehillah_council_diplomacy_scale
	# V29 section 4 (2026-10-11): the leader's Hashkafa.
	multiply = kehillah_hashkafa_petition_mult_value
	# V29 section 3.5 (2026-10-11): a renowned academy speaks for the
	# community -- x1.25 while it keeps a yeshiva.
	if = {
		limit = { scope:councillor_liege ?= { kehillah_has_yeshiva_trigger = yes } }
		multiply = 1.25
	}
	if = {
		limit = {
			scope:kehillah_petition_host ?= {
				opinion = { target = scope:councillor_liege value >= 50 }
			}
		}
		multiply = 1.5
	}
}

# --- Dayyan: Prepare the Bet Din. Conference cooldown in months: the
# base (kehillah_bet_din_conference_cooldown_years, 2) less up to 12
# months at Learning 20 while a Dayyan performs the task. Scope: the
# conference host (the Kehillah leader).
kehillah_bet_din_prepare_max_months_saved = 12

kehillah_bet_din_conference_cooldown_months = {
	value = kehillah_bet_din_conference_cooldown_years
	multiply = 12
	cp:councillor_court_chaplain ?= {
		if = {
			limit = { is_performing_council_task = task_kehillah_prepare_bet_din }
			subtract = {
				value = kehillah_bet_din_prepare_max_months_saved
				multiply = kehillah_council_learning_scale
			}
		}
	}
	round = yes
	min = 6
}

# --- Rosh ha-Shmirah: Keep the Night Watch. Weight multiplier applied to
# the yearly protection events (events/kehillah_protection_events.txt):
# 1.0 with no watch, down to 0.5 at Martial 20. Scope: the Kehillah leader.
kehillah_night_watch_event_weight_value = {
	value = 1
	cp:councillor_marshal ?= {
		if = {
			limit = { is_performing_council_task = task_kehillah_night_watch }
			subtract = {
				value = martial
				max = kehillah_council_skill_cap
				divide = kehillah_council_skill_cap
				multiply = 0.5
			}
		}
	}
}

# --- Tooltip display values. Scope: the councillor (the SCOPE of a council
# task's effect_desc).
kehillah_council_revenue_pct_value = { value = kehillah_council_stewardship_scale multiply = 25 round = yes }
kehillah_council_construction_pct_value = { value = kehillah_council_stewardship_scale multiply = 25 round = yes }
kehillah_council_bet_din_months_saved_value = { value = kehillah_council_learning_scale multiply = kehillah_bet_din_prepare_max_months_saved round = yes }
kehillah_council_petition_chance_display_value = {
	value = kehillah_council_diplomacy_scale
	multiply = kehillah_shtadlan_petition_max_monthly_chance
	multiply = kehillah_hashkafa_petition_mult_value
	if = {
		limit = { scope:councillor_liege ?= { kehillah_has_yeshiva_trigger = yes } }
		multiply = 1.25
	}
}
kehillah_council_night_watch_pct_value = {
	value = martial
	max = kehillah_council_skill_cap
	divide = kehillah_council_skill_cap
	multiply = 50
	round = yes
}

# --- Shtadlan: the yearly audience (in task_kehillah_petition_host),
# 2026-10-13; first built as the Dayyan's Discourse. A skill check:
# (Diplomacy x 2 - the host's standing) x 2, 0 to 50, a percent each year.
# Scope: the Shtadlan. Diplomacy 12 moves a count 28% of years, a duke
# 18%, a king 4%, an emperor never; Diplomacy 18 moves a count 50%, a duke
# 50%, a king 28%, an emperor 12%.
kehillah_audience_max_chance = 50
kehillah_audience_outlook = 2

kehillah_audience_yearly_chance_value = {
	value = 0
	if = {
		limit = {
			exists = employer.domicile.domicile_location.county.holder.top_liege
		}
		add = diplomacy
		multiply = 2
		subtract = employer.domicile.domicile_location.county.holder.top_liege.kehillah_audience_difficulty_value
		multiply = 2
	}
	min = 0
	max = kehillah_audience_max_chance
	round = yes
}
kehillah_audience_fail_chance_value = {
	value = 100
	subtract = kehillah_audience_yearly_chance_value
}

# Scope: the host. His rank, his faith's teaching, his zeal.
kehillah_audience_difficulty_value = {
	value = 10
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		value = 30
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		value = 22
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_duchy }
		value = 15
	}
	if = {
		limit = { kehillah_teaching_exclusionary_trigger = yes }
		add = 10
	}
	else_if = {
		limit = { kehillah_teaching_strict_trigger = yes }
		add = 5
	}
	if = {
		limit = { has_trait = zealous }
		add = 5
	}
}

# --- Dayyan: A Learned Exchange (task_kehillah_learned_exchange),
# 2026-10-13. A learning check against the county faith's teaching:
# (Learning x 2 - its standing) x 1.5, 0 to 50, a percent each year.
# Scope: the Dayyan. Learning 23: a protective faith 39%, a strict one
# 24%, an exclusionary one 9%. Learning 15: 15%, 0, 0.
kehillah_exchange_max_chance = 50
kehillah_exchange_tension_years = 10
kehillah_faith_goodwill_yearly_cap = 3

kehillah_exchange_yearly_chance_value = {
	value = 0
	if = {
		limit = {
			exists = employer.domicile.domicile_location.county.faith
			employer.domicile.domicile_location.county.faith = {
				NOT = { religion = religion:judaism_religion }
			}
		}
		add = learning
		multiply = 2
		subtract = employer.domicile.domicile_location.county.faith.kehillah_exchange_difficulty_value
		multiply = 1.5
	}
	min = 0
	max = kehillah_exchange_max_chance
	round = yes
}
kehillah_exchange_fail_chance_value = {
	value = 100
	subtract = kehillah_exchange_yearly_chance_value
}

# Scope: a faith. Its main rite's teaching; +10 while it wages a great
# holy war.
kehillah_exchange_difficulty_value = {
	value = 20
	if = {
		limit = {
			main_rite = {
				OR = {
					rite_has_doctrine = kehillah_teaching_exclusion
					rite_has_doctrine = kehillah_teaching_abolition
				}
			}
		}
		value = 40
	}
	else_if = {
		limit = {
			main_rite = {
				OR = {
					rite_has_doctrine = kehillah_teaching_separation_mission
					rite_has_doctrine = kehillah_teaching_strict_pact
				}
			}
		}
		value = 30
	}
	if = {
		limit = { exists = great_holy_war }
		add = 10
	}
}
