# Kehillah lifestyle perk effects (2026-11-25, Daniel's request).
# Spec: docs/spec/2026-11-24-kehillah-lifestyle-perks.md section 8.
#
# Six vanilla perks whose effects are vassal, claim or county mechanics do
# nothing for a landless community leader. For a leader whose government has
# government_uses_kehillah_lifestyle_perks, each makes the leader's own gains
# on one pillar larger, the same way a leadership style does (pillars spec R2):
# kehillah_change_pillar_effect multiplies a positive change by these values.
# Passive convergence is not boosted. Each perk's tooltip line is in
# common/effect_localization/kehillah_lifestyle_perk_effects.txt.

# Each matching perk adds this share to a positive change.
kehillah_perk_pillar_boost_step = 0.1

# Character scope: the leader. 1 with no matching perk.
kehillah_perk_stability_boost_value = {
	value = 1
	if = {
		limit = { government_has_flag = government_uses_kehillah_lifestyle_perks }
		if = {
			limit = { has_perk = defensive_negotiations_perk }
			add = kehillah_perk_pillar_boost_step
		}
		if = {
			limit = { has_perk = absolute_control_perk }
			add = kehillah_perk_pillar_boost_step
		}
	}
}

kehillah_perk_prosperity_boost_value = {

		limit = { government_has_flag = government_uses_kehillah_lifestyle_perks }
		if = {
			limit = { has_perk = at_any_cost_perk }

		}

		}

}
