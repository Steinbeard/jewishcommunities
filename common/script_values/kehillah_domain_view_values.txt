# A pillar's band as 1 (Crisis) to 5 (Legendary), for five-frame icons
# (prosperity_level_icons.dds, legitimacy_level_icon.dds).
kehillah_dom_tier_prosperity_value = {
	value = 1
	primary_title ?= {
		if = {
			limit = { has_variable = kehillah_var_prosperity }
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_strained_threshold } add = 1 }
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_healthy_threshold } add = 1 }
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_flourishing_threshold } add = 1 }
			if = { limit = { var:kehillah_var_prosperity >= kehillah_band_legendary_threshold } add = 1 }
		}
	}
}

# The Standing emblem: development_levels.dds, six frames from a hut to a
# stone hall. Crisis 1, Strained 2, Healthy 4, Flourishing 5, Legendary 6
# (frame 3, a bare cottage, is skipped: Healthy should look settled).
kehillah_dom_standing_frame_value = {
	value = 1
	primary_title ?= {
		if = {
			limit = { has_variable = kehillah_var_standing }
			if = { limit = { var:kehillah_var_standing >= kehillah_band_strained_threshold } add = 1 }
			if = { limit = { var:kehillah_var_standing >= kehillah_band_healthy_threshold } add = 2 }
			if = { limit = { var:kehillah_var_standing >= kehillah_band_flourishing_threshold } add = 1 }
			if = { limit = { var:kehillah_var_standing >= kehillah_band_legendary_threshold } add = 1 }
		}
	}
}

# Where each pillar is heading: its baseline as a tick on the 0-1000 bar,
# placed by two stretch factors (pos, and the rest to 1000). Kept within
# 1..999 so neither side is ever zero.
kehillah_dom_baseline_pos_prosperity_value = {
	value = kehillah_prosperity_baseline_value
	min = 1
	max = 999
}
kehillah_dom_baseline_rest_prosperity_value = {
	value = 1000
	subtract = kehillah_dom_baseline_pos_prosperity_value
}
