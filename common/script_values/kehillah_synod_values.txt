# 2026-11-25 law-and-custom doctrine motions (kehillah_synod.0109).
# Controversy (kehillah_synod_score_doctrine_effect) starts from each
# motion's STEP: the change in vanilla divergence between the two doctrines,
# plus how far the change breaks with received custom. It is halved when
# the faith's main rite already holds the doctrine (a move toward common
# practice). It rises when the host's rite IS the main rite, since every
# other rite then diverges. Half the rite's present divergence from the
# main rite is added. Bands:
kehillah_synod_controversy_moderate = 10
kehillah_synod_controversy_major = 20
kehillah_synod_controversy_grave = 30
# Each row's STEP depends on what the rite holds now (scope: the host), so a
# longer move costs more: the vanilla divergence gap between the two
# doctrines, plus how sharply it breaks with received custom.
kehillah_synod_step_divorce_approval_value = {
	value = 10	# from Allowed: Gershom's herem
	if = {
		limit = { rite ?= { rite_has_doctrine = doctrine_divorce_disallowed } }
		value = 5	# from Disallowed: a loosening, and toward the Torah's law
	}
}
kehillah_synod_step_divorce_allowed_value = {
	value = 10	# from Approval
	if = {
		limit = { rite ?= { rite_has_doctrine = doctrine_divorce_disallowed } }
		value = 15
	}
}
kehillah_synod_step_proselytism_encouraged_value = 20
kehillah_synod_step_proselytism_discouraged_value = 5
