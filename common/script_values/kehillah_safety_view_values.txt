# Military > Community Safety: the danger gauge (gui/kehillah_crisis_military.gui), 2026-11-01.
# Scope: a Kehillah leader. The band thresholds match KehillahDangerBand
# (kehillah_breakdown_custom_loc.txt): 3, 8, 15 and 30 percent a year.

# 0 Calm, 1 Uneasy, 2 Tense, 3 Dangerous, 4 Perilous.
kehillah_safety_band_index_value = {
	value = 0
	if = {
		limit = { kehillah_incident_yearly_chance_value >= 30 }
		value = 4
	}
	else_if = {
		limit = { kehillah_incident_yearly_chance_value >= 15 }
		value = 3
	}
	else_if = {
		limit = { kehillah_incident_yearly_chance_value >= 8 }
		value = 2
	}
	else_if = {
		limit = { kehillah_incident_yearly_chance_value >= 3 }
		value = 1
	}
}

# How far the yearly chance fills each band's segment, 0 to 1. Perilous
# ends at 40%, the most four quarterly rolls at 12% can reach.
kehillah_safety_fill_calm_value = {
	value = kehillah_incident_yearly_chance_value
	divide = 3
	min = 0
	max = 1
}
kehillah_safety_fill_uneasy_value = {
	value = kehillah_incident_yearly_chance_value
	subtract = 3
	divide = 5
	min = 0
	max = 1
}
kehillah_safety_fill_tense_value = {
	value = kehillah_incident_yearly_chance_value
	subtract = 8
	divide = 7
	min = 0
	max = 1
}
kehillah_safety_fill_dangerous_value = {
	value = kehillah_incident_yearly_chance_value
	subtract = 15
	divide = 15
	min = 0
	max = 1
}
kehillah_safety_fill_perilous_value = {
	value = kehillah_incident_yearly_chance_value
	subtract = 30
	divide = 10
	min = 0
	max = 1
}
