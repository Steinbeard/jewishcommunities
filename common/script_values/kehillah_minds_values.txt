# Machshavah changes minds (2026-11-11, docs/spec/2026-11-11-rulings-sermons-
# and-gates.md section 4). Scope: an AI reader with var:kehillah_minds_tenet.
# The chance he adopts that tenet on mastering the book.
kehillah_minds_base_chance = 20
kehillah_minds_adopt_chance_value = {
	value = kehillah_minds_base_chance
	if = { limit = { has_trait = erudite } add = 15 }
	if = { limit = { has_trait = cynical } add = 10 }
	if = { limit = { has_trait = zealous } add = -15 }
	if = { limit = { has_trait = stubborn } add = -15 }
	# His rite already holds it at the core: the book confirms what he was taught.
	if = { limit = { kehillah_minds_tenet_status_trigger = { STATUS = core } } add = 15 }
	# His rite prohibits it.
	if = { limit = { kehillah_minds_tenet_status_trigger = { STATUS = prohibited } } add = -30 }
	min = 0
	max = 90
}
