# Source of truth: docs/spec/v21-rabbi-ordination-paths.md sections 2,
# 3.1-3.4. The spec is explicit that these thresholds live in script
# values rather than inside an event or decision, specifically so a live
# result can retune them ("the threshold belongs in a script value/
# trigger, not embedded in an event, so live results can tune it to
# three works later without changing the model", section 3.2).

# Nothing here is a balance claim. They are the spec's own numbers, in

# The literacy-and-serious-study floor to BEGIN the ordination path
# (section 3.1). Deliberately reachable by education, tutors and
# ordinary skill growth; it is not a claim that a moderately learned
# person is already a rabbi.
kehillah_ordination_learning_floor_value = {
	value = 8
}

# Section 3.3's lower band: Learning 6-7 is a completed-curriculum
# FOUNDATION worth one study field, but still needs the ordinary adult
# route and one separate adult field. Below this, a rabbinic education
# does not count as study at all.
kehillah_ordination_foundation_learning_floor_value = {
	value = 6
}

# Section 3.2's published curriculum: two distinct completed works, from
# two distinct fields (Parshanut, Talmudics, Hashkafa). At the current
# two-year scheme cadence that is roughly four years of study.
kehillah_ordination_works_required_value = {
	value = 2
}

kehillah_ordination_fields_required_value = {
	value = 2
}

# Section 3.3: accumulated years under a qualifying rabbinic guardian
# before an education counts as a rabbinic apprenticeship. Accumulated,
# not "the guardian on the final day" -- swapping in a rabbi tutor just
# before adulthood must not manufacture an ordination.
kehillah_ordination_apprenticeship_years_value = {
	value = 4
}

# Greatness the recognising community gains for ordaining a rabbi. Sized
# against kehillah_book_greatness_gain (common/script_values/
# kehillah_script_values.txt) -- an ordination is a real but smaller
# event for the institution than a work of scholarship coming out of it.
kehillah_semicha_greatness_reward = {
	value = 15
}
