# Percent chance per year that an eligible AI Kehillah leader convenes.
kehillah_bet_din_ai_yearly_chance = 25

# Percent chance that a leader who qualifies for the Synod intent sits as a
# Synod rather than an ordinary Sitting. The Sanhedrin intent, when valid,
# is always taken.
kehillah_bet_din_ai_synod_chance = 50

# Days from the activity's creation before a stand-in Sitting opens its
# docket, so the court added after planning can travel in.
kehillah_bet_din_stand_in_gathering_days = 60

# Hang guard for phase 1 while it waits for kehillah_bet_din_ai.0002.
kehillah_bet_din_stand_in_wait_ceiling_days = {
	value = kehillah_bet_din_stand_in_gathering_days
	add = kehillah_bet_din_case_phase_days
	add = 7
}

# AI judges added outright to a stand-in Sitting (quorum needs two).
kehillah_bet_din_stand_in_forced_judges = 3

# Extra AI delegates added outright to an AI Synod or Sanhedrin.
kehillah_bet_din_stand_in_forced_delegates = 4

# The leader-side cooldown, in days. The engine's own activity cooldown
# lands on the stand-in, who planned the activity. Scope: the leader.
kehillah_bet_din_ai_cooldown_days = {
	value = kehillah_bet_din_conference_cooldown_months
	multiply = 30
}
