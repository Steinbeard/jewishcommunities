# Hereditary priestly succession (docs/implementation/2026-11-25-samaritan-priestly-succession.md).
# A high priesthood passes only to a Kohen of its faith, the holder's own
# priestly kin first. Applied by script to any title marked
# kehillah_priestly_succession_title; today that is only vanilla's
# d_samaritan (kehillah_samaritan_priesthood_setup_effect). Modeled on
# kehillah_sanhedrin_succession_law, the live-proven title-law election.
kehillah_priestly_succession_law = {
	law_group_type = title_succession_laws
	index = 41

	can_title_have = {
		has_variable = kehillah_priestly_succession_title
	}
	should_show_for_title = {
		always = no
	}
	can_remove_from_title = {
		always = no
	}
	succession = {
		order_of_succession = election
		election_type = kehillah_priestly_elective
		faith = same_faith
	}
	flag = elective_succession_law
}
