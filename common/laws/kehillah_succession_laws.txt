# =================================================================
# kehillah_appointment_succession_law
#
# Source of truth: docs/spec/v1-kehillah-community.md section 5.
#
# CK3 1.20 (Crozier) PORT, 2026-09-30: through 1.19 every succession law
# had to live INSIDE the single succession_order_laws group object, so
# adding one meant shipping a full verbatim copy of vanilla's
# common/laws/00_succession_laws.txt with this law appended (see
# docs/implementation/v1-kehillah-implementation.md section 5). 1.20 split
# law groups out into common/law_groups/ and made each law its own
# top-level object that names its group via law_group_type. So this law
# now stands alone and the vanilla-file override is gone -- no more
# re-syncing a 2000-line vanilla copy on every patch, and no conflict
# with other mods that touch succession laws.
#
# Also dropped in the port: `potential = { government_has_flag =
# government_is_kehillah }`. 1.20's _laws.info no longer documents a
# `potential` field and no vanilla law uses one; can_have below carries
# the identical gate. See docs/implementation/ck3-1.20-crozier-port.md.
#
# The law is gated entirely on the Kehillah government flag, so it is
# invisible and unreachable for every other government -- no vanilla
# playthrough behaves differently because this exists.
# =================================================================
kehillah_appointment_succession_law = {
	law_group_type = succession_order_laws
	index = 100 # vanilla's succession laws use 0-34; never shown in the UI anyway

	can_keep = {
		government_has_flag = government_is_kehillah
	}
	can_have = {
		government_has_flag = government_is_kehillah
	}
	can_pass = {
		government_has_flag = government_is_kehillah
	}
	should_start_with = {
		government_has_flag = government_is_kehillah
	}

	# Never offered in the law-change UI. There is exactly one way a
	# Kehillah chooses its leader, and it is not a policy the player
	# gets to repeal -- it is what the institution is. Applied
	# automatically via should_start_with.
	should_show_for_title = {
		always = no
	}
	can_remove_from_title = {
		always = no
	}

	succession = {
		order_of_succession = appointment
		appointment_type = kehillah_leadership
	}

	flag = appointment_type_succession
}
