# titles of Rabbinic Judaism) elect the next Nasi from among themselves:
# vanilla's College of Cardinals pattern. A title law, so it binds only
# d_kehillah_sanhedrin and leaves the head's community on its own law.
# Modeled on temporal_head_of_faith_succession_law (01_title_succession_laws.txt).

		always = no
	}
	can_remove_from_title = {
		always = no
	}
	succession = {
		order_of_succession = election
		election_type = kehillah_sanhedrin_elective
		faith = same_faith
	}
