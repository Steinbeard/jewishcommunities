# The Sanhedrin's own title law (V46; V47 phase B moved it to the seat
# county). The members of the Sanhedrin (holders of its seats, clerical
# elector titles of Rabbinic Judaism) elect the next Nasi from among
# themselves, vanilla's College of Cardinals pattern. A title law, so it
# binds only the seat county and the seat community (both marked by
# kehillah_sanhedrin_restore_effect), and leaves other titles on their own
# laws. Modeled on
# temporal_head_of_faith_succession_law (01_title_succession_laws.txt).
kehillah_sanhedrin_succession_law = {
	law_group_type = title_succession_laws
	index = 40

	# The seat county and the seat community (both marked
	# kehillah_sanhedrin_seat_title), so both go to the elected Nasi.
	can_title_have = {
		has_variable = kehillah_sanhedrin_seat_title
	}
	should_show_for_title = {
		always = no
	}
	can_remove_from_title = {
		always = no
	}
	succession = {
		order_of_succession = election
		election_type = kehillah_sanhedrin_elective
		faith = same_faith
	}
	flag = elective_succession_law
}
