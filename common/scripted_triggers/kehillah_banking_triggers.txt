# Banking services and lending (2026-11-20). See
# common/script_values/kehillah_banking_values.txt.

# Scope: a Kehillah leader. The community is actually lending: the leader is
# a creditor through vanilla's loans, or the S8 royal loan is outstanding on
# the host (kehillah_loan_amount_owed, the older record).
kehillah_community_is_lending_trigger = {
	OR = {
		any_in_list = { variable = owes_me_money }
		domicile.domicile_location.county.holder.top_liege ?= { has_variable = kehillah_loan_amount_owed }
	}
}

# Scope: any character. He owes a non-Jewish lender through a vanilla loan.
kehillah_has_non_jewish_loan_trigger = {
	any_owned_story = {
		story_type = story_cycle_loan
		var:lender ?= { is_jewish_character_trigger = no }
	}
}

# Scope: a ruler (2026-11-20, Daniel). He needs credit -- in debt, or
# paying a non-Jewish lender -- and no Jewish community lends in his realm.
# Feeds the invitation motive (kehillah_invitation_values.txt) and the
# Invite Jewish Settlers decision's AI weight.
kehillah_wants_jewish_credit_trigger = {
	is_jewish_character_trigger = no
	NOT = { has_variable = kehillah_banking_services }
	OR = {
		gold < 0
		kehillah_has_non_jewish_loan_trigger = yes
	}
}

# Scope: a ruler. A Christian (non-Jewish) lending house: a merchant company
# or holy order that lends, or is set up to.
kehillah_is_rival_lender_trigger = {
	is_jewish_character_trigger = no
	OR = {
		AND = {
			government_has_flag = government_is_merchant
			OR = {
				has_realm_law = company_purpose_creditor
				any_in_list = { variable = owes_me_money }
			}
		}
		AND = {
			government_has_flag = government_is_holy_order
			any_in_list = { variable = owes_me_money }
		}
	}
}
