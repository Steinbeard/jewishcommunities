# Another specialisation may be started: fewer than the Kahal House allows.
# Waived while a succession rebuilds the Quarter or a start seeds it
# (global variable kehillah_quarter_seeding), since add_domicile_building
# honours can_construct.
kehillah_quarter_can_add_specialisation_trigger = {
	OR = {
		has_global_variable = kehillah_quarter_seeding
		kehillah_specialisation_count_value < kehillah_specialisation_allowance_value
	}
}

# --- Where the Quarter stands (regional trades) -----------------------------
kehillah_quarter_in_regions_trigger = {
	domicile ?= {
		domicile_location ?= {
			OR = {
				geographical_region = $R1$
				geographical_region = $R2$
				geographical_region = $R3$
				geographical_region = $R4$
				geographical_region = $R5$
				geographical_region = $R6$
			}
		}
	}
}

# Silk: Byzantium, southern Italy and Sicily, the Levant and Egypt, al-Andalus.
kehillah_quarter_silk_country_trigger = {
	kehillah_quarter_in_regions_trigger = {
		R1 = world_europe_south_east R2 = world_europe_south_italy R3 = world_asia_minor
		R4 = world_middle_east R5 = world_africa_north R6 = world_europe_west_iberia
	}
}

# --- Wall and Gates -----------------------------------------------------------
# The quarter may be walled: the host charter's community-security term grants
# at least a licensed watch (a wall is a fortification, so it travels with the
# right to guard the quarter), or the fallback charter grants a walled quarter.
# 2026-11-24 (Daniel): this replaces the separate Petition for a Wall decision;
# the wall is a term of the charter, not a special request.
kehillah_wall_licensed_trigger = {
	OR = {
		kehillah_cached_charter_quarter_walled_trigger = yes
		kehillah_cached_charter_security_watch_trigger = yes
		kehillah_cached_charter_security_strong_trigger = yes
	}
}
