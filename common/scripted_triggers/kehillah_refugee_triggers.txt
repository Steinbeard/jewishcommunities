# Scope: a ruler; scope:kehillah_refugee_victim and _from saved. Another
# community within huge distance, not threatened, not mourning a pogrom
# of its own, and not already sheltering a band on the road.
kehillah_refugee_haven_trigger = {
	government_has_flag = government_is_kehillah
	is_alive = yes
	NOT = { this = scope:kehillah_refugee_victim }
	exists = primary_title
	exists = domicile.domicile_location
	primary_title = {
		NOT = { has_variable = kehillah_mob_threat }
		NOT = { has_variable = kehillah_pogrom_memory }
	}
	NOT = { has_variable = kehillah_refugees_from }
	domicile.domicile_location = {
		squared_distance = { target = scope:kehillah_refugee_from value <= squared_distance_huge }
	}
}
