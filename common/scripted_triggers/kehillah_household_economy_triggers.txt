# House scope. Every GUI action and payout validates live ownership.
kehillah_household_economic_trigger = {
	NOT = { has_variable = kehillah_household_ruling }
	exists = var:kehillah_household_head
	var:kehillah_household_head = { is_alive = yes }
	exists = var:kehillah_household_of.holder
	var:kehillah_household_of.holder = { government_has_flag = government_is_kehillah }
	var:kehillah_household_head = { is_courtier_of = prev.var:kehillah_household_of.holder }
}
# Character root, scope:family supplied by GUI. Excludes stale/foreign/ruling families.
kehillah_household_can_manage_trigger = {
	government_has_flag = government_is_kehillah
	is_adult = yes
	is_imprisoned = no
	exists = scope:family
	scope:family = {
		kehillah_household_economic_trigger = yes
		var:kehillah_household_of = root.primary_title
		NOT = { has_variable = kehillah_household_assignment_cooldown }
	}
}
