# ---------------------------------------------------------------------
# is_kehillah_title_trigger
# Scope: a landed title.
#
# The single place that answers "is this title a Jewish community?".
# v1 has exactly one, the Kehillah of Worms. Every later Kehillah --
# Troyes, the Phase 2 Babylonian overlay, the Phase 3 Sephardi one --
# gets added here and nowhere else.
#
# This is checked by name rather than by a title flag because it has to
# be answerable during history execution and at game start, before any
# script has had a chance to set flags on titles.
# ---------------------------------------------------------------------

is_kehillah_title_trigger = {
	this = title:d_kehillah_worms
}


# ---------------------------------------------------------------------
# is_kehillah_leader_trigger
# Scope: a character.
# ---------------------------------------------------------------------

is_kehillah_leader_trigger = {
	government_has_flag = government_is_kehillah
	any_held_title = {
		is_kehillah_title_trigger = yes
	}
}


# ---------------------------------------------------------------------
# kehillah_has_shtadlan_trigger
# Scope: a character.
#
# Used by the protection events to decide whether the "let the Shtadlan
# speak for us" resolution is available at all. Requires both the
# chambers and someone actually sitting in them -- a building with no
# advocate in it cannot negotiate on the community's behalf.
# ---------------------------------------------------------------------

kehillah_has_shtadlan_trigger = {
	employs_court_position = shtadlan_kehillah_position
}


# ---------------------------------------------------------------------
# kehillah_has_watch_trigger
# Scope: a character.
#
# The Communal Watch has no officer by design (spec section 3c), so
# unlike the Shtadlan it is a pure building check.
# ---------------------------------------------------------------------

kehillah_has_watch_trigger = {
	domicile ?= { has_domicile_parameter = kehillah_has_communal_watch }
}


# ---------------------------------------------------------------------
# kehillah_shtadlan_prevails_trigger / kehillah_watch_prevails_trigger
# Scope: a character. Parameter: THRESHOLD (the skill needed).
#
# Whether a protection incident is resolved WELL rather than merely
# survived. Two ways to clear the bar: personal skill, or a fully built
# institution that carries a less able leader.
#
# This is what stops the upper tiers of the Shtadlan's Chambers and the
# Communal Watch from being pure stat padding. Spec section 3b asks that
# "whichever officer/skill you've invested in should visibly matter" in
# these events; a building that only nudged a modifier would not deliver
# that, whereas one that changes which outcome you get does.
#
# The institutional route asks 4 points less than the personal one. A
# practised office genuinely helps, and does not entirely substitute for
# knowing what you are doing.
# ---------------------------------------------------------------------

kehillah_shtadlan_prevails_trigger = {
	OR = {
		diplomacy >= $THRESHOLD$
		AND = {
			domicile ?= { has_domicile_parameter = kehillah_practised_advocacy }
			diplomacy >= {
				value = $THRESHOLD$
				subtract = 4
			}
		}
	}
}

kehillah_watch_prevails_trigger = {
	OR = {
		martial >= $THRESHOLD$
		AND = {
			domicile ?= { has_domicile_parameter = kehillah_watch_is_organised }
			martial >= {
				value = $THRESHOLD$
				subtract = 4
			}
		}
	}
}
