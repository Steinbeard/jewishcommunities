# ---------------------------------------------------------------------
# is_kehillah_title_trigger
# Scope: a landed title.
#
# The single place that answers "is this title a Jewish community?".
# v1 has exactly one, the Kehillah of Worms. Every later Kehillah --
# Troyes, the Phase 2 Babylonian overlay, the Phase 3 Sephardi one --
# gets added here and nowhere else.
#
# This is checked by name rather than by a title flag because it has to
# be answerable during history execution and at game start, before any
# script has had a chance to set flags on titles.
# ---------------------------------------------------------------------

is_kehillah_title_trigger = {
	this = title:d_kehillah_worms
}


# ---------------------------------------------------------------------
# is_kehillah_leader_trigger
# Scope: a character.
# ---------------------------------------------------------------------

is_kehillah_leader_trigger = {
	government_has_flag = government_is_kehillah
	any_held_title = {
		is_kehillah_title_trigger = yes
	}
}


# ---------------------------------------------------------------------
# kehillah_has_shtadlan_trigger
# Scope: a character.
#
# Used by the protection events to decide whether the "let the Shtadlan
# speak for us" resolution is available at all. Requires both the
# chambers and someone actually sitting in them -- a building with no
# advocate in it cannot negotiate on the community's behalf.
# ---------------------------------------------------------------------

kehillah_has_shtadlan_trigger = {
	employs_court_position = shtadlan_kehillah_position
}


# ---------------------------------------------------------------------
# kehillah_shtadlan_prevails_trigger / kehillah_watch_prevails_trigger
# Scope: a character. Parameter: THRESHOLD (the skill needed).
#
# Whether a protection incident is resolved WELL rather than merely
# survived.
#
# REWRITTEN 2026-09-06, alongside the removal of Shtadlan's Chambers and
# the Communal Watch as buildings (see
# common/domiciles/buildings/kehillah_domicile_buildings.txt and
# common/court_positions/types/kehillah_officers.txt for why). The
# original version of each trigger offered a second, easier way to clear
# the bar if the matching building's top tier existed
# (kehillah_practised_advocacy, kehillah_watch_is_organised) -- "a
# practised office genuinely helps, and does not entirely substitute for
# knowing what you are doing." With no premises behind either role
# anymore, that institutional shortcut has nothing left to hang off, so
# both triggers are now a plain personal-skill check. This is the
# correct simplification, not a loss: "person and function, not
# premises" was the whole point of ungating these roles, and a building
# quietly still discounting the skill check would have undercut that.
# ---------------------------------------------------------------------

kehillah_shtadlan_prevails_trigger = {
	diplomacy >= $THRESHOLD$
}

kehillah_watch_prevails_trigger = {
	martial >= $THRESHOLD$
}


# ---------------------------------------------------------------------
# kehillah_leader_is_rabbinic_trigger
# Scope: the Kehillah leader (character).
#
# Whether the community currently reads this leader as its rabbi rather
# than a lay elder -- see common/flavorization/kehillah_title_holders.txt
# for what this actually changes (the character's displayed rank: "Rav"
# instead of "Parnas"). Historically both were real communal leadership
# types -- see that file's header for the sourcing.
#
# ADDED 2026-09-07: kehillah_rabbi_trait (common/traits/
# kehillah_traits.txt) is now the primary, unambiguous signal -- it
# means exactly "this specific person holds recognized religious
# authority", nothing weaker. The three checks that predate it stay as
# a fallback, deliberately checked on the LEADER's own traits/skill
# rather than on whether they hold the Chief Rabbi court position: the
# ruler cannot hold their own community's court positions (those are
# for courtiers), so a leader can plausibly read as a serious scholar
# on personal reputation alone without ever having held that specific
# office -- theologian and the top two Learning education tiers are
# the same signals kehillah_officers.txt already uses to judge Chief
# Rabbi aptitude; learning >= 12 is a fallback for a self-taught or
# late-blooming scholar who never received a formal education trait.
# ---------------------------------------------------------------------

kehillah_leader_is_rabbinic_trigger = {
	OR = {
		has_trait = kehillah_rabbi_trait
		has_trait = theologian
		has_trait = education_learning_4
		has_trait = education_learning_5
		learning >= 12
	}
}
