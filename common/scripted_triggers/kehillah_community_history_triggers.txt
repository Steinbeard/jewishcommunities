# 2026-11-15. County scope. A lost community whose span can be told:
# a loss year and a start (first record or an earlier refounding), and
# at least one year between them.
kehillah_history_has_span_trigger = {
	has_variable = kehillah_history_last_loss_year
	OR = {
		has_variable = kehillah_history_original_year
		AND = {
			has_variable = kehillah_history_last_refounded_year
			var:kehillah_history_last_refounded_year < var:kehillah_history_last_loss_year
		}
	}
	kehillah_history_years_stood_value >= 1
}
