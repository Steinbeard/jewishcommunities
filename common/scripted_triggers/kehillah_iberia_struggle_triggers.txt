# Scope: a Kehillah leader whose Quarter still exists. The Quarter lies in
# the struggle's own region while the struggle runs.
kehillah_under_iberian_struggle_trigger = {
	has_fp2_dlc_trigger = yes
	exists = struggle:iberian_struggle
	government_has_flag = government_is_kehillah
	domicile.domicile_location ?= { geographical_region = world_europe_west_iberia }
}

# Scope: the struggle. The phase has no catalyst list in the community's
# direction (vanilla: Hostility only leads to Compromise; Compromise only to
# Opportunity; Conciliation only to Compromise), so the community's news
# lengthens the current phase instead (kehillah_iberia_struggle_push_effect).
kehillah_iberia_struggle_prolongs_ruin_trigger = {
	is_struggle_phase = struggle_iberia_phase_hostility
}
kehillah_iberia_struggle_prolongs_flourish_trigger = {
	OR = {
		is_struggle_phase = struggle_iberia_phase_compromise
		is_struggle_phase = struggle_iberia_phase_conciliation
	}
}
