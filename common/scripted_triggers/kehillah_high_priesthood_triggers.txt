# A high priesthood is a Jewish head of faith whose faith vests Temple
# authority in the priestly line: the Samaritan Torah doctrine (the Samaritan
# High Priesthood) and the Kohanim doctrine (a restored priesthood). Their
# head is elected by the faith's clergy (vanilla theocratic_elective, copied
# in common/succession_election/00_pam_clerical_elective.txt), and only a
# Kohen may be chosen: the trait passes father to son (kohen, kehillah_traits).
# Root = the candidate, who shares the electing faith. Faiths without such a
# doctrine are always ok, so no other election changes.
kehillah_high_priesthood_faith_trigger = {
	faith = {
		OR = {
			has_doctrine = temple_authority_samaritan_torah_doctrine
			has_doctrine = temple_authority_kohanim_doctrine
			# A restored High Priesthood heads the faith (kehillah_kohen_gadol_effects.txt).
			religious_head_title ?= { has_variable = kehillah_kohen_gadol_title }
		}
	}
}

kehillah_high_priesthood_candidate_ok_trigger = {
	trigger_if = {
		limit = { kehillah_high_priesthood_faith_trigger = yes }
		is_male = yes
		has_trait = kohen
	}
}
