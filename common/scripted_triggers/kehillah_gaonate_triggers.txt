# V47 heads of rite (docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).
# Rabbinic Judaism's head_of_rite title is d_kehillah_gaonate
# (kehillah_judaism_faith_names.txt). Its holder is the engine's head of his
# own rite (live-verified 2026-10-19), styled Gaon, or Rabbenu in Minhag
# Ashkenaz. In 1066 nobody holds it.

# Scope: a character. Holds the head-of-rite title.
kehillah_is_gaonate_holder_trigger = {
	has_title = title:d_kehillah_gaonate
}

# Scope: a character. Belongs to Minhag Ashkenaz. Guarded: rite:minhag_
# ashkenaz errors before the 1066 synod founds it.
kehillah_is_minhag_ashkenaz_trigger = {
	exists = global_var:kehillah_minhag_ashkenaz_founded
	rite = rite:minhag_ashkenaz
}

# Scope: a character. A rite with no head: the only kind a rabbi may come
# to lead by merit (Daniel, 2026-10-19).
kehillah_rite_has_no_head_trigger = {
	NOT = { exists = rite.head_of_rite }
}

# Scope: a character. The parts of merit, each shown in the decision.
kehillah_head_of_rite_leader_trigger = {
	is_ruler = yes
	government_has_flag = government_is_kehillah
	primary_title ?= { is_kehillah_title_trigger = yes }
	faith = faith:rabbinic_faith
	is_adult = yes
	is_incapable = no
}
kehillah_head_of_rite_illustrious_yeshiva_trigger = {
	kehillah_yeshiva_fame_tier_value >= 3
}
kehillah_head_of_rite_two_tracks_trigger = {
	OR = {
		AND = {
			has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 }
			has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 100 }
		}
		AND = {
			has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 100 }
			has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 100 }
		}
		AND = {
			has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= 100 }
			has_trait_xp = { trait = kehillah_rabbi_trait track = hashkafa value >= 100 }
		}
	}
}

kehillah_head_of_rite_merit_trigger = {
	kehillah_head_of_rite_leader_trigger = yes
	kehillah_qualified_rosh_yeshiva_trigger = yes
	kehillah_head_of_rite_illustrious_yeshiva_trigger = yes
	kehillah_head_of_rite_two_tracks_trigger = yes
	piety_level >= kehillah_head_of_rite_min_piety_level
	kehillah_rite_has_no_head_trigger = yes
	NOT = { exists = title:d_kehillah_gaonate.holder }
}

# Scope: a character. The Gaon's office (V47): an academy at whose head
# the head of rite may make his recognition an office.
kehillah_leads_geonic_academy_trigger = {
	primary_title ?= {
		OR = {
			this = title:d_kehillah_sura
			this = title:d_kehillah_baghdad
		}
	}
}

# Scope: a character. The title is an office (passes with the academy)
# rather than a lifetime recognition.
kehillah_gaonate_is_office_trigger = {
	exists = global_var:kehillah_gaonate_office_academy
}
