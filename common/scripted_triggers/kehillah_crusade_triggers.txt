# A community under a Catholic lord. Scope: a character.
kehillah_crusade_catholic_community_trigger = {
	government_has_flag = government_is_kehillah
	exists = primary_title
	exists = domicile.domicile_location.county.holder.top_liege
	domicile.domicile_location.county.holder.top_liege.faith = faith:catholic
}

# A community the host may turn to next. Scope: a character;
# scope:kehillah_mob_leader the host's leader, scope:kehillah_mob_from
# where it stands.
kehillah_mob_next_target_trigger = {
	government_has_flag = government_is_kehillah
	exists = primary_title
	exists = domicile.domicile_location
	exists = domicile.domicile_location.county.holder.top_liege
	domicile.domicile_location.county.holder.top_liege.faith.religion = scope:kehillah_mob_leader.faith.religion
	primary_title = {
		NOT = { has_variable = kehillah_mob_threat }
		NOT = { has_variable = kehillah_mob_paid_off }
	}
	scope:kehillah_mob_leader = {
		NOT = { is_target_in_variable_list = { name = kehillah_mob_struck_titles target = prev.primary_title } }
	}
	domicile.domicile_location = {
		squared_distance = { target = scope:kehillah_mob_from value <= squared_distance_large }
	}
}

# Fifty-fifty, through a value rolled per call. Scope: any.
kehillah_coin_flip_trigger = {
	kehillah_coin_flip_value >= 50
}

# V42 (2026-10-14): WHO has already decided, one way or the other, about
# the host of scope:kehillah_mob_leader (LIST: kehillah_mob_condemned_by,
# kehillah_mob_condemn_refused_list or kehillah_mob_men_refused_list).
kehillah_mob_decided_trigger = {
	scope:kehillah_mob_leader ?= {
		is_target_in_variable_list = { name = $LIST$ target = $WHO$ }
	}
}

# V42: the mob against this community is Muslim (its leader's faith; once
# the host is gone, the county's).
kehillah_mob_is_muslim_trigger = {
	OR = {
		var:kehillah_mob_ref ?= { faith.religion = religion:islam_religion }
		AND = {
			NOT = { exists = var:kehillah_mob_ref }
			domicile.domicile_location.county.faith.religion ?= religion:islam_religion
		}
	}
}
