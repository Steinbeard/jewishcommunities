# Scope: a county; scope:kehillah_mother_leader saved. Would its lord
# take Jewish settlers, and is it free of a community, settled enough,
# at peace and within reach of the mother community?
kehillah_settlers_destination_trigger = {
	exists = holder
	exists = holder.top_liege.primary_title
	holder = { is_at_war = no }
	development_level >= kehillah_natural_founding_min_development
	OR = {
		faith.religion = religion:christianity_religion
		faith.religion = religion:islam_religion
	}
	holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_allowed_trigger = yes }
	title_province = {
		squared_distance = {
			target = scope:kehillah_mother_leader.domicile.domicile_location
			value <= squared_distance_large
		}
	}
	NOT = { this = scope:kehillah_mother_leader.domicile.domicile_location.county }
	NOT = {
		any_in_global_list = {
			variable = kehillah_registered_communities
			exists = holder.domicile.domicile_location
			holder.domicile.domicile_location.county = prev
		}
	}
}

# Scope: a courtier; scope:kehillah_mother_leader saved. Could he lead
# the settlers? Not the leader's heir or wife, not a student. The choice
# offers one of each kind (kehillah_settlers.0002): a Ba'al Bayit, a
# learned rabbi, a man of the leader's own house.
kehillah_settlers_leader_candidate_trigger = {
	is_adult = yes
	is_male = yes
	is_imprisoned = no
	is_ruler = no
	is_jewish_character_trigger = yes
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	NOT = { scope:kehillah_mother_leader.player_heir ?= this }
	NOT = { is_spouse_of = scope:kehillah_mother_leader }
	# A younger son may lead (Daniel 2026-10-13: "a member of your
	# dynasty"); the heir and the leader's wife may not.
}

# Scope: a courtier; scope:kehillah_mother_leader and list
# kehillah_settlers_party. Could he go with the settlers as part of the
# overflow? Not a councillor, the heir, the leader's close family, or a
# yeshiva student; not already going.
kehillah_settlers_follower_candidate_trigger = {
	is_imprisoned = no
	is_jewish_character_trigger = yes
	NOT = { is_in_list = kehillah_settlers_party }
	NOT = { this = scope:kehillah_settlers_leader }
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	NOT = { is_councillor = yes }
	NOT = { scope:kehillah_mother_leader.player_heir ?= this }
	NOT = { is_close_family_of = scope:kehillah_mother_leader }
}
