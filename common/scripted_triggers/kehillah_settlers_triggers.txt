# Scope: a county; scope:kehillah_mother_leader saved. Would its lord
# take Jewish settlers, and is it free of a community, settled enough,
# at peace and within reach of the mother community?
# Scope: a county. The non-distance requirements shared by every founding
# route: a settled, peaceful county in a realm that permits Jewish settlement,
# with no existing Kehillah in that county. A player-controlled Jewish
# adventurer may use these requirements worldwide; settlers, exiles and AI
# adventurers also use the diaspora/reach wrapper below.
kehillah_founding_site_base_trigger = {
	# Name the candidate county: inside a community's domicile_location
	# scope, `prev` is the community title, not this county (live 2026-10-13:
	# the rule was false for every county on the map).
	save_temporary_scope_as = kehillah_site_county
	exists = holder
	exists = holder.top_liege.primary_title
	holder = { is_at_war = no }
	prosperity_level >= kehillah_natural_founding_min_development
	holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_allowed_trigger = yes }
	NOT = {
		any_in_global_list = {
			variable = kehillah_registered_communities
			exists = holder.domicile.domicile_location.county
			holder.domicile.domicile_location.county = scope:kehillah_site_county
		}
	}
}

# Scope: a county. The ordinary founding rule for settlers, exiles and AI
# adventurers: the shared site requirements plus a local diaspora connection
# or an Encouraged host realm. Adjacent counties are intentionally valid even
# where their map centres are closer than the former spacing radius.
kehillah_founding_site_trigger = {
	kehillah_founding_site_base_trigger = yes
	OR = {
		holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
		any_in_global_list = {
			variable = kehillah_registered_communities
			exists = holder.domicile.domicile_location
			OR = {
				# County adjacency is a real local connection, unlike the
				# former arbitrary 60-pixel exclusion. It also covers large
				# counties whose centres fall outside the small distance band.
				holder.domicile.domicile_location.county = {
					any_neighboring_county = { this = scope:kehillah_site_county }
				}
				holder.domicile.domicile_location = {
					squared_distance = {
						target = scope:kehillah_site_county.title_province
						# A Wales away (live 2026-10-13: an Ireland around 19
						# communities covered most of the map, Krakow and Gyor
						# included). Settlers still reach an Ireland from their
						# own mother community (kehillah_settlers_destination_
						# trigger); farther needs an invitation.
						value <= squared_distance_small
					}
				}
			}
		}
	}
}

# Scope: a county; scope:kehillah_mother_leader saved. A founding site
# within the settlers' reach of their mother community: a nearby realm
# that allows settlement, or a more distant one that encourages it.
kehillah_settlers_destination_trigger = {
	kehillah_founding_site_trigger = yes
	NOT = { this = scope:kehillah_mother_leader.domicile.domicile_location.county }
	OR = {
		title_province = {
			squared_distance = {
				target = scope:kehillah_mother_leader.domicile.domicile_location
				value <= squared_distance_medium
			}
		}
		AND = {
			holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
			title_province = {
				squared_distance = {
					target = scope:kehillah_mother_leader.domicile.domicile_location
					value <= squared_distance_large
				}
			}
		}
	}
}

# Scope: a courtier; scope:kehillah_mother_leader saved. Could he lead
# the settlers? Not the leader's heir or wife, not a student. The choice
# offers one of each kind (kehillah_settlers.0002): a Ba'al Bayit, a
# learned rabbi, a man of the leader's own house.
kehillah_settlers_leader_candidate_trigger = {
	is_adult = yes
	is_male = yes
	is_imprisoned = no
	is_ruler = no
	is_jewish_character_trigger = yes
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	NOT = { scope:kehillah_mother_leader.player_heir ?= this }
	NOT = { is_spouse_of = scope:kehillah_mother_leader }
	# A younger son may lead (Daniel 2026-10-13: "a member of your
	# dynasty"); the heir and the leader's wife may not.
}

# Scope: a courtier; scope:kehillah_mother_leader and list
# kehillah_settlers_party. Could he go with the settlers as part of the
# overflow? Not a councillor, the heir, the leader's close family, or a
# yeshiva student; not already going.
kehillah_settlers_follower_candidate_trigger = {
	is_imprisoned = no
	is_jewish_character_trigger = yes
	NOT = { is_in_list = kehillah_settlers_party }
	NOT = { this = scope:kehillah_settlers_leader }
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	NOT = { is_councillor = yes }
	NOT = { scope:kehillah_mother_leader.player_heir ?= this }
	NOT = { is_close_family_of = scope:kehillah_mother_leader }
	# A wife is not "close family" to CK3 (live 2026-10-13: Isaac's left).
	NOT = { is_spouse_of = scope:kehillah_mother_leader }
}
