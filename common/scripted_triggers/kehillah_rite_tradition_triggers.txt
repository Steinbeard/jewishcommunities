# =====================================================================
# Which Temple Authority a rite follows (2026-11-10). Scope: a rite.
#
# Daniel: "I noticed the Talmuds are banned for some reason". In 1.20 a
# faith's doctrines reach only its MAIN rite (see the note in
# common/religion/doctrine_types/kehillah_rite_authority_doctrines.txt), so a
# bare rite_has_doctrine = temple_authority_rabbinic_doctrine is false for
# Minhag Ashkenaz, Merkabah and every rite founded in play. Those rites read
# the Mishnah and the Talmuds as Rejected (0), which the library counts as
# forbidden. Each trigger below asks the rite first (Karaism and the other
# rites that carry the doctrine themselves), then the rite's faith.
#
# Used by the library catalog generator (scripts/kehillah_library_catalog.py:
# canon, baseline and floor values) and by the hand-written sefer triggers.
# =====================================================================

# The Mishnah and the Talmuds are the Oral Torah (Rabbanite rites).
kehillah_rite_rabbanite_trigger = {
	OR = {
		rite_has_doctrine = temple_authority_rabbinic_doctrine
		rite_faith ?= { has_doctrine = temple_authority_rabbinic_doctrine }
	}
}

# The Palestinian rite receives the Yerushalmi as its foundational Talmudic
# corpus; all other Rabbinic rites in this pass use the Bavli as canon.
kehillah_rite_minhag_yerushalmi_trigger = {
	this = rite:minhag_yerushalmi
}

kehillah_rite_bavli_trigger = {
	kehillah_rite_rabbanite_trigger = yes
	NOT = { this = rite:minhag_yerushalmi }
}

# Scripture alone, no Oral Torah (Karaite rites).
kehillah_rite_toric_trigger = {
	OR = {
		rite_has_doctrine = temple_authority_toric_doctrine
		rite_faith ?= { has_doctrine = temple_authority_toric_doctrine }
	}
}

# The Samaritan Torah alone: the Prophets and Writings are not Scripture.
kehillah_rite_samaritan_trigger = {
	OR = {
		rite_has_doctrine = temple_authority_samaritan_torah_doctrine
		rite_faith ?= { has_doctrine = temple_authority_samaritan_torah_doctrine }
	}
}

# The Beta Israel's Orit tradition is Ge'ez-scriptural and neither Rabbanite
# nor Toraic in the game's Temple Authority groups.
kehillah_rite_haymanot_trigger = {
	rite_faith ?= faith:haymanot
}

# Scope: the viewer. The Sefarim ledger shows a rite other than the viewer's
# own (common/scripted_effects/kehillah_library_ledger_effects.txt).
kehillah_ledger_other_rite_trigger = {
	exists = var:kehillah_ledger_rite
	NOT = { rite ?= var:kehillah_ledger_rite }
}
