# =====================================================================
# Overrides vanilla's should_use_clerical_appointment_law_trigger
# (common/scripted_triggers/00_law_triggers.txt).
#
# ADDED 2026-10-09 (Shabbat heartbeat 1), together with
# `mechanic_type = theocracy` on kehillah_government -- see that
# government's long comment on the line itself for why the mechanic is
# set at all (trait-free clerical continuation, ROADMAP item H1).
#
# Vanilla 1.20.0.4 verbatim:
#
#   should_use_clerical_appointment_law_trigger = {
#       government_has_mechanic = theocracy
#       highest_held_title_tier >= tier_county
#       should_use_theocratic_elective_succession_law_trigger = no
#   }
#
# It is the whole of clerical_appointment_law's can_have / can_keep /
# can_pass / should_start_with / potential. Setting the theocracy
# mechanic therefore makes that vanilla law valid for Kehillah leaders,
# and common/laws/_laws.info says the starting law for a group is found
# by checking the laws "in definition order" -- index is display order
# within the group, nothing more. Vanilla's 00_succession_laws.txt loads
# before this mod's kehillah_succession_laws.txt, so without this fork
# clerical_appointment_law wins the succession_order_laws slot and the
# Kehillah silently loses kehillah_appointment_succession_law -- i.e.
# the kehillah_leadership appointment type, the male-only lock and the
# whole merit score. Succession is this repo's highest-risk subsystem
# (docs/implementation/v1-kehillah-implementation.md sections 6 and 8);
# this is not a risk worth taking for a tooltip.
#
# The fork is additive and narrow, in the same spirit as
# kehillah_is_playable_character_override.txt: vanilla's body verbatim
# plus one exclusion for this government. Every other government in the
# game, vanilla or modded, behaves exactly as before.
#
# Regenerate from vanilla on every game patch, same rule as the council
# overrides.
# =====================================================================

should_use_clerical_appointment_law_trigger = {
	# --- added by the Jewish Communities mod ---
	# A Kehillah appoints its leader by kehillah_leadership, not by the
	# Christian clerical appointment type.
	NOT = { government_has_flag = government_is_kehillah }

	# --- vanilla 1.20, verbatim ---
	government_has_mechanic = theocracy
	highest_held_title_tier >= tier_county
	should_use_theocratic_elective_succession_law_trigger = no
}
