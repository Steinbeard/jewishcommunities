# County facts for the empty-site inspector. These use the same registry,
# host county and distances as kehillah_founding_site_trigger. They explain
# that rule without changing its gameplay behavior.
kehillah_inspector_county_occupied_trigger = {
	save_temporary_scope_as = kehillah_inspector_county
	any_in_global_list = {
		variable = kehillah_registered_communities
		exists = holder.domicile.domicile_location.county
		holder.domicile.domicile_location.county = scope:kehillah_inspector_county
	}
}

kehillah_inspector_spacing_blocked_trigger = {
	save_temporary_scope_as = kehillah_inspector_county
	any_in_global_list = {
		variable = kehillah_registered_communities
		exists = holder.domicile.domicile_location
		holder.domicile.domicile_location = {
			squared_distance = {
				target = scope:kehillah_inspector_county.title_province
				value < kehillah_settlers_min_spacing_squared
			}
		}
	}
}

kehillah_inspector_network_reachable_trigger = {
	save_temporary_scope_as = kehillah_inspector_county
	OR = {
		holder.top_liege.primary_title = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
		any_in_global_list = {
			variable = kehillah_registered_communities
			exists = holder.domicile.domicile_location
			holder.domicile.domicile_location = {
				squared_distance = {
					target = scope:kehillah_inspector_county.title_province
					value <= squared_distance_small
				}
			}
		}
	}
}
