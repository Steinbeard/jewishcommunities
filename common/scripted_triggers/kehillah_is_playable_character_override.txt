# =====================================================================
# Overrides vanilla's is_playable_character
# (common/scripted_triggers/00_available_for_events_triggers.txt).
#
# THE ACTUAL ROOT CAUSE OF THE BOOKMARK-GENERATION CRASH, FOUND AFTER AN
# EXHAUSTIVE PROCESS OF ELIMINATION.
#
# In 1.19, is_playable_character was a closed, hardcoded list:
#
#   is_landed | is_landless_administrative | is_landless_adventurer |
#   is_landless_nomad | is_landless_soryo | tgp_is_any_minister
#
# (CK3 1.20 rewrote it - see the 1.20 note at the bottom of this header.)
#
# There is no generic branch for "any government with landless_playable
# = yes in its government_rules". Setting that rule (which
# kehillah_government does) is necessary to let the engine assign the
# landless title, domicile and buildings at all -- and it does, as
# verified in a live session -- but it is NOT sufficient to satisfy
# THIS separate check, which gates whether a character may be handed to
# (or remain) the player. kehillah_government's flag,
# government_is_kehillah, matches none of the six hardcoded branches, so
# a Kehillah leader always fails this trigger regardless of anything
# else about the government, domicile, succession law, or candidate
# pool -- which is exactly the pattern the elimination testing found:
# every configuration using kehillah_government as the character's
# INITIAL bookmark government failed identically, while every
# configuration using an existing vanilla landless government (which
# DOES match one of the six branches) succeeded.
#
# is_playable_character is also the fallback branch of vanilla's
# is_character_allowed_to_be_player scripted rule
# (common/scripted_rules/00_rules.txt), which is what produces the
# "Game Over: became landless" screen once the mid-session
# change_government workaround (see history/titles/kehillah_titles.txt
# and common/on_action/kehillah_on_actions.txt) hands Isaac
# kehillah_government.
#
# The fix is additive: copy vanilla's OR block verbatim and add one more
# branch. This is deliberately narrower than the "All Governments
# Playable" Workshop mod (id 3021516102), which neuters the entire
# is_character_allowed_to_be_player rule via a permanently-false limit
# wrapper -- that also removes vanilla's DLC-gating for
# theocracy/republic/mercenary/holy_order governments, which this fork
# does not touch.
#
# CK3 1.20 (Crozier) NOTE, 2026-09-30: vanilla rewrote this trigger to
# `is_ruler = yes` AND (exists = domicile | non-baron landed/theocracy |
# tgp_is_any_minister | has_variable = laamp_transition). A Kehillah leader
# holding the community title with a domicile may now pass vanilla on its
# own, but that is unverified, and this override is what fixed a crash.
# Conservative re-base: vanilla 1.20's logic verbatim as one branch, OR the
# Kehillah flag on its own - so the Kehillah case behaves exactly as it did
# in 1.19 (no new is_ruler requirement) and everyone else gets 1.20
# vanilla behaviour unchanged. Revisit whether the override can be dropped
# once a 1.20 boot + succession playtest passes. See
# docs/implementation/ck3-1.20-crozier-port.md.
# =====================================================================

is_playable_character = {
	OR = {
		# --- vanilla 1.20, verbatim ---
		AND = {
			is_ruler = yes
			OR = {
				exists = domicile
				# exclude barons
				trigger_if = {
					limit = {
						OR = {
							is_landed = yes
							government_has_flag = government_is_theocracy
						}
					}
					highest_held_title_tier > tier_barony
				}
				tgp_is_any_minister = yes # To keep ministers playable, regardless of having a noble family title or not
				has_variable = laamp_transition
			}
		}

		# --- added by the Jewish Communities mod ---
		government_has_flag = government_is_kehillah
	}
}
