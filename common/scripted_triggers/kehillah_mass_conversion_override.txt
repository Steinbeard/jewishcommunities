# CK3 1.20 override of pam_mass_rite_conversion_requirements_trigger from
# common/scripted_triggers/pam_scripted_triggers.txt. Vanilla body is the first
# two lines; the Kehillah addition hides Mass Conversion (the project and its
# "can plan" nudge) for rites that discourage proselytism. Rites without the
# doctrine, i.e. every non-Jewish rite, are unaffected. Refresh this override if
# vanilla changes the trigger in a later patch.

pam_mass_rite_conversion_requirements_trigger = {
	has_pam_dlc_trigger = yes
	highest_held_title_tier >= tier_duchy
	rite = { NOT = { rite_has_doctrine = kehillah_proselytism_discouraged } }
}
