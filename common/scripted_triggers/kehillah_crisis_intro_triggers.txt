# 2026-11-01. Admission and opening-event gates; pending state is stored on the community title.
kehillah_crisis_can_offer_intro_trigger = {
 government_has_flag = government_is_kehillah
 exists = domicile.domicile_location.county.holder.top_liege
 primary_title = {
  NOT = { has_variable = kehillah_crisis_intro_pending }
  NOT = { has_variable = kehillah_crisis_post_end_cooldown }
  NOT = { has_variable = kehillah_mob_threat }
  NOT = { exists = var:kehillah_trial_situation }
  NOT = { exists = var:kehillah_decree_situation }
  NOT = { has_variable = kehillah_expulsion_quarters }
  # Stage 4 is active Recovery until the final chronicle closes the situation.
  # The close effect marks it done, then starts the one-year cooldown.
  OR = {
   NOT = { has_variable = kehillah_crisis_stage }
   AND = { var:kehillah_crisis_stage = 4 has_variable = kehillah_crisis_situation_done }
  }
 }
}
kehillah_crisis_attack_intro_trigger = {

  primary_title ?= { var:kehillah_crisis_intro_pending ?= flag:attack }
  kehillah_crisis_warning_trigger = yes
 }
}
