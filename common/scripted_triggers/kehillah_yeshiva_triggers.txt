# =====================================================================
# THE YESHIVA -- V29 (docs/spec/v29-yeshiva.md), first implementation
# pass 2026-10-11.
#
# A Kehillah "has a yeshiva" when its domicile holds the third tier of
# the Beit Midrash line, kehillah_beit_midrash_03. Since V29 that tier is
# not built from the domicile menu but founded through
# kehillah_found_yeshiva_decision (or seeded at Worms), so holding it IS
# having founded one.
# =====================================================================

# Scope: a Kehillah leader (or any character with a domicile).
kehillah_has_yeshiva_trigger = {
	exists = domicile
	domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_03 }
}

# Scope: a character. Could this rabbi head a yeshiva -- the teacher
# students travel for (V29 section 3.1)? High Learning AND real depth in
# one of the two teaching tracks.
kehillah_qualified_rosh_yeshiva_trigger = {
	kehillah_is_ordained_rabbi_trigger = yes
	is_adult = yes
	is_incapable = no
	learning >= kehillah_rosh_yeshiva_min_learning
	OR = {
		has_trait_xp = { trait = kehillah_rabbi_trait track = talmudics value >= kehillah_rosh_yeshiva_min_track_xp }
		has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= kehillah_rosh_yeshiva_min_track_xp }
	}
}

# Scope: a Kehillah leader. Is there someone in this community -- the
# leader himself or a courtier -- who could head a yeshiva?
kehillah_has_qualified_rosh_yeshiva_candidate_trigger = {
	OR = {
		kehillah_qualified_rosh_yeshiva_trigger = yes
		any_courtier = { kehillah_qualified_rosh_yeshiva_trigger = yes }
	}
}

# Scope: a Kehillah leader. Does the yeshiva have a head right now? The
# appointed Rosh Yeshiva, or -- V29 section 3.2, Daniel 2026-10-11 -- a
# Rav-leader, who IS the Rosh Yeshiva whenever he has not appointed one.
kehillah_has_rosh_yeshiva_trigger = {
	kehillah_has_yeshiva_trigger = yes
	OR = {
		employs_court_position = rosh_yeshiva_kehillah_position
		kehillah_is_ordained_rabbi_trigger = yes
	}
}

# Scope: a Kehillah leader. Found a Yeshiva's gates (V29 section 3.1):
# Stability Healthy (the main gate -- order lets learning flourish),
# Greatness only Strained (low on purpose, so Greatness does not
# snowball), the House of Debate standing, and no yeshiva yet.
kehillah_can_found_yeshiva_pillars_trigger = {
	kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_stability THRESHOLD = kehillah_band_healthy_threshold }
	kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_strained_threshold }
}

# ---------------------------------------------------------------------
# V29 section 9 (2026-10-11): the gameplay loop.
# ---------------------------------------------------------------------

# Scope: a character at the leader's court. Could he study a zman here?
# An unordained Jewish man of study age.
kehillah_zman_student_candidate_trigger = {
	is_male = yes
	age >= 12
	age <= 30
	is_incapable = no
	is_imprisoned = no
	NOT = { has_trait = kehillah_rabbi_trait }
	is_rabbinic_authority_jewish_trigger = yes
	is_ruler = no
}

# Scope: a Kehillah leader. Can he hold a zman now?
kehillah_can_hold_zman_trigger = {
	kehillah_has_rosh_yeshiva_trigger = yes
	NOT = { has_character_flag = kehillah_zman_cooldown }
	NOT = { has_character_flag = kehillah_zman_in_progress }
}

# =====================================================================
# V29 section 10 (2026-10-11): the living yeshiva.
# =====================================================================

# Scope: a character. An ORDAINED rabbi: the Rabbi trait, and not still a
# student. Enrolment grants the trait (its tracks are where study
# accumulates), so every OFFICE gate -- Dayyan, Rosh Yeshiva, Rav
# naming, the community rabbi -- reads this, not the bare trait.
kehillah_is_ordained_rabbi_trigger = {
	has_trait = kehillah_rabbi_trait
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	# A student who withdrew before graduating keeps the trait (and what
	# he learned) but is not ordained.
	NOT = { has_variable = kehillah_yeshiva_unfinished }
}

# Scope: a character. Could he enrol at a yeshiva? An unordained rabbinic
# Jewish man of 16+ with Learning above 8.
kehillah_yeshiva_enrollable_trigger = {
	is_male = yes
	age >= kehillah_yeshiva_min_age
	learning > kehillah_yeshiva_min_learning
	is_incapable = no
	is_imprisoned = no
	# Not yet a rabbi -- or a student who withdrew and may return.
	OR = {
		NOT = { has_trait = kehillah_rabbi_trait }
		has_variable = kehillah_yeshiva_unfinished
	}
	NOT = { has_variable = kehillah_yeshiva_enrolled }
	is_rabbinic_authority_jewish_trigger = yes
}

# Scope: an enrolled student. Is he at his yeshiva's town right now?
# Study counts only while present (V29 section 10.2).
kehillah_yeshiva_student_present_trigger = {
	has_variable = kehillah_yeshiva_enrolled
	exists = var:kehillah_yeshiva_enrolled.holder.domicile
	location = var:kehillah_yeshiva_enrolled.holder.domicile.domicile_location
}

# Scope: a character. Is there any yeshiva of his faith to go to?
kehillah_any_yeshiva_for_me_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		exists = holder
		holder = {
			kehillah_has_yeshiva_trigger = yes
			faith = root.faith
		}
	}
}
