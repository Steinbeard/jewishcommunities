# Scope: a province. Is CULTURE (a bare culture key, e.g. sephardi) the
# Jewish culture native to this province's county?
kehillah_native_jewish_culture_here_is_trigger = {
	county ?= {
		exists = var:kehillah_native_jewish_culture
		var:kehillah_native_jewish_culture = culture:$CULTURE$
	}
}

# Scope: a province (unused; kept for the template's call). Has this save
# been seeded? False only in a save from before 2026-11-08, where the pool
# template falls back to its old geography. Changed 2026-11-08: a county
# may now be seeded to no Jewish culture at all, so this asks the global
# flag, not the county.
kehillah_native_jewish_culture_seeded_here_trigger = {
	has_global_variable = kehillah_native_jewish_cultures_seeded
}
