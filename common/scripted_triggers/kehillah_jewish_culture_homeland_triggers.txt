# Scope: a province. Is CULTURE (a bare culture key, e.g. sephardi) the
# Jewish culture native to this province's county?
kehillah_native_jewish_culture_here_is_trigger = {
	county ?= {
		exists = var:kehillah_native_jewish_culture
		var:kehillah_native_jewish_culture = culture:$CULTURE$
	}
}

# Scope: a province. Has this province's county been seeded at all? False
# only in a save from before 2026-11-08, where the pool template falls
# back to its old geography.
kehillah_native_jewish_culture_seeded_here_trigger = {
	county ?= { exists = var:kehillah_native_jewish_culture }
}
