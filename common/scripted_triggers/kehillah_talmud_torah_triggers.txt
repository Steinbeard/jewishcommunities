# A rite with Talmud Torah opens ordinary library study to lay adults.
# The pre-existing rabbi/ordination route remains available without it.
# Callers separately check Jewish faith, age, capacity and library access.
# Scope: character. Shared by study access, mastery and rewards.
kehillah_talmud_torah_follower_trigger = {
	rite = { rite_has_tenet = tenet_talmud_torah }
}

kehillah_torah_reader_eligible_trigger = {
	OR = {
		kehillah_rabbi_study_eligible_trigger = yes
		kehillah_talmud_torah_follower_trigger = yes
		# 2026-11-08 Sefer redesign: a Karaite reader studies the library too
		# (the Tanakh and the Karaite works), as a Ba'alei ha-Miqra follower.
		rite ?= { rite_has_tenet = tenet_baalei_hamiqra }
	}
}

# At Learning 8, the tenet allows mastery of a real work on the normal
# course. Semicha still requires two fields and a qualifying institution.
kehillah_talmud_torah_mastery_ready_trigger = {
	kehillah_talmud_torah_follower_trigger = yes
	learning >= kehillah_talmud_torah_mastery_learning_floor
}
