# =====================================================================
# OUTLOOK STANCES -- V34 section 2 (the v31 stance names). Scope: any
# character. Read off kehillah_outlook_on_jews_value.
# =====================================================================

# Protector: guards his Jews, for faith, justice or gain.
kehillah_outlook_protector_trigger = {
	kehillah_outlook_on_jews_value >= 30
}

# Convinced: believes they have no place in a Christian (or Muslim)
# society.
kehillah_outlook_convinced_trigger = {
	kehillah_outlook_on_jews_value <= -30
}

# Predator: greedy, and indebted to them or merely covetous -- milks
# them rather than expelling them, until expelling pays better.
kehillah_outlook_predator_trigger = {
	NOT = { kehillah_outlook_convinced_trigger = yes }
	has_trait = greedy
	kehillah_outlook_on_jews_value < 30
}

# Pragmatist: everyone else.
kehillah_outlook_pragmatist_trigger = {
	NOT = { kehillah_outlook_protector_trigger = yes }
	NOT = { kehillah_outlook_convinced_trigger = yes }
	NOT = { kehillah_outlook_predator_trigger = yes }
}
