# Vanilla: religion = religion:judaism_religion.
#
# Gates vanilla's Jewish priestly portrait entries: the turban
# (religious_jewish_high_headgear, religious_jewish_head in
# gfx/portraits/portrait_modifiers/03_headgear_religious.txt) and the
# matching Kohen / Kohen Gadol vestments (03_clothes_religious.txt).
# Nothing else uses it. Those are Temple-priest costumes, wrong for an
# 11th-century rabbi or dayyan, so they are reserved for future Kohen
# Gadol content: a character must carry kehillah_kohen_gadol_portrait
# (nothing sets it yet). Jewish clergy otherwise wear their culture's
# ordinary clothes.
portrait_religious_jewish_trigger = {
	religion = religion:judaism_religion
	# Jewish Communities: reserved for Kohen Gadol content
	exists = this
	has_character_flag = kehillah_kohen_gadol_portrait
}

# Vanilla: Islam, Emishi, and devoted/clergy Orthodox characters.
#
# The beard entries in gfx/portraits/portrait_modifiers/01_beards_base.txt
# give +200 weight to a culture-appropriate beard when this is true,
# against a clean-shaven weight of 30, so most such men are bearded but
# not all (Daniel, 2026-10-09: a chance of clean-shaven is fine). Added:
# rabbis (kehillah_rabbi_trait) and Jewish clergy (court chaplain/Dayyan
# seat, theocrats, devoted). Other Jewish men keep vanilla's odds.
should_always_have_beard_trigger = {
	OR = {
		scope:faith.religion = religion:islam_religion
		is_culture_or_descended_from_trigger = { CULTURE = culture:emishi }
		AND = {
			portrait_religious_orthodox_trigger = yes
			OR = {
				portrait_religious_devoted_trigger = yes
				portrait_religious_high_trigger = yes
				portrait_religious_high_no_poverty_trigger = yes
			}
		}
		# Jewish Communities: rabbis and Jewish clergy
		AND = {
			exists = this
			religion = religion:judaism_religion
			OR = {
				has_trait = kehillah_rabbi_trait
				portrait_religious_devoted_trigger = yes
				portrait_religious_high_trigger = yes
				portrait_religious_high_no_poverty_trigger = yes
			}
		}
	}
}

# Vanilla (1.20.1): exists = clerical_elector_title, plus the poverty
# checks below.
#
# Dresses every holder of a clerical elector title as a cardinal (robes
# in 03_clothes_religious.txt, the galero/zucchetto in
# 03_headgear_religious.txt, and animations). Seats on the Sanhedrin use
# the same engine mechanism (V46/V47, kehillah_sanhedrin_grant_seat_effect),
# so without this every member of the court, and so every candidate for
# Nasi, wore cardinal's red (Daniel, live 2026-10-19). A Sanhedrin seat is
# exempted; its members wear their ordinary clothes.
portrait_is_cardinal_trigger = {
	exists = this
	has_pam_dlc_trigger = yes
	exists = clerical_elector_title
	# Jewish Communities: a Sanhedrin seat is not a cardinalate
	NOT = { clerical_elector_title = { has_variable = kehillah_sanhedrin_seat } }
	portrait_religious_head_no_poverty_trigger = no
	NOT = {
		rite = {
			rite_has_parameter = vows_of_poverty_active
		}
	}
}
