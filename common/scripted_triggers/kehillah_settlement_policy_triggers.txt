# =====================================================================
# Jewish Settlement Policy (V16, first infrastructure slice).
#
# Scope of the four policy-band triggers: the HOST REALM'S PRIMARY
# TITLE. Policy is stored as an optional numeric title variable, not on
# each community or on the host character, so it is shared by every
# Kehillah beneath that realm and survives ordinary character succession.
#
# Values: 0 = Banned, 1 = Discouraged, 2 = Allowed, 3 = Encouraged.
# Absence deliberately means Allowed. This gives every realm a policy
# without mass-initialising title variables at game start; a later
# policy-change event only needs to store exceptional rungs.
#
# Fresh Christian/Muslim charters now call these in obligation-level
# is_valid/default logic. Existing charters are never rewritten. The
# debug events 72-75 remain the repeatable way to inspect policy bands.
# See docs/spec/v16-jewish-settlement-policy-and-charters.md §§3 and 8.
# =====================================================================

kehillah_jewish_settlement_policy_encouraged_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy >= 3
}

kehillah_jewish_settlement_policy_allowed_trigger = {
	OR = {
		NOT = { has_variable = kehillah_jewish_settlement_policy }
		var:kehillah_jewish_settlement_policy >= 2
	}
}

kehillah_jewish_settlement_policy_discouraged_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy = 1
}

kehillah_jewish_settlement_policy_banned_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy <= 0
}

# Scope: a community leader. This is the eligibility primitive eventual
# founding/migration routes will use: resolve the same top-liege host as
# V15's Charter, then ask whether that host realm allows settlement.
kehillah_host_policy_allows_jewish_settlement_trigger = {
	exists = domicile.domicile_location.county.holder.top_liege.primary_title
	domicile.domicile_location.county.holder.top_liege.primary_title = {
		kehillah_jewish_settlement_policy_allowed_trigger = yes
	}
}

# Scope: a community leader constructing a *new institution* in the
# Jewish Quarter. Free construction (and legacy/no regional construction
# term) pass directly. Permission/authorisation requires the temporary
# grant supplied by the petition decision; forbidden construction fails.
# Upgrades are intentionally not covered: this row regulates whether a
# new communal building may be founded, not routine repair/expansion of
# an institution the community already possesses.
kehillah_charter_allows_new_communal_building_trigger = {
	OR = {
		# V15 and future non-regional charter packages remain permissive
		# exactly as before. A regional charter with no cache yet fails
		# closed until .0003 has observed its committed rights.
		AND = {
			NOT = { has_subject_contract_group = kehillah_christian_host_charter_prototype }
			NOT = { has_subject_contract_group = kehillah_muslim_host_charter_prototype }
		}
		kehillah_cached_charter_construction_free_trigger = yes
		AND = {
			kehillah_cached_charter_construction_permission_trigger = yes
			has_character_modifier = kehillah_charter_construction_permission_modifier
		}
	}
}

# is the sole reader of live tributary flags; it copies compact rungs to

# debug readers must use these helpers rather than subject_contract_has_flag.
