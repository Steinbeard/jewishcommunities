# =====================================================================
# Jewish Settlement Policy (V16, first infrastructure slice).
#
# Scope of the four policy-band triggers: the HOST REALM'S PRIMARY
# TITLE. Policy is stored as an optional numeric title variable, not on
# each community or on the host character, so it is shared by every
# Kehillah beneath that realm and survives ordinary character succession.
#
# Values: 0 = Banned, 1 = Discouraged, 2 = Allowed, 3 = Encouraged.
# Absence deliberately means Allowed. This gives every realm a policy
# without mass-initialising title variables at game start; a later
# policy-change event only needs to store exceptional rungs.
#
# Fresh Christian/Muslim charters now call these in obligation-level
# is_valid/default logic. Existing charters are never rewritten. The
# debug events 72-75 remain the repeatable way to inspect policy bands.
# See docs/spec/v16-jewish-settlement-policy-and-charters.md §§3 and 8.
# =====================================================================

kehillah_jewish_settlement_policy_encouraged_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy >= 3
}

kehillah_jewish_settlement_policy_allowed_trigger = {
	OR = {
		NOT = { has_variable = kehillah_jewish_settlement_policy }
		var:kehillah_jewish_settlement_policy >= 2
	}
}

kehillah_jewish_settlement_policy_discouraged_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy = 1
}

kehillah_jewish_settlement_policy_banned_trigger = {
	has_variable = kehillah_jewish_settlement_policy
	var:kehillah_jewish_settlement_policy <= 0
}

# Scope: a community leader. This is the eligibility primitive eventual
# founding/migration routes will use: resolve the same top-liege host as
# V15's Charter, then ask whether that host realm allows settlement.
kehillah_host_policy_allows_jewish_settlement_trigger = {
	exists = domicile.domicile_location.county.holder.top_liege.primary_title
	domicile.domicile_location.county.holder.top_liege.primary_title = {
		kehillah_jewish_settlement_policy_allowed_trigger = yes
	}
}

# Scope: a rabbinic adventurer before founding. It intentionally mirrors
# the community-leader resolver above but starts at the camp's current
# location, because the title and domicile do not exist yet.
kehillah_found_location_policy_allows_jewish_settlement_trigger = {
	exists = location.county.holder.top_liege.primary_title
	location.county.holder.top_liege.primary_title = {
		kehillah_jewish_settlement_policy_allowed_trigger = yes
	}
}

# Scope: an established community leader. This is an informational
# grandfathering check, not an enforcement trigger: a policy change does
# not revoke an existing charter. It tells the ledger and any future
# policy-change event that the host's new public stance is now narrower
# than one or more rights actually held by this community, so a charter
# review/renegotiation is politically live.
kehillah_current_charter_exceeds_host_policy_trigger = {
	OR = {
		# Construction: free needs Allowed; permission needs more than Banned.
		AND = {
			OR = {
				vassal_contract_has_flag = kehillah_christian_building_free
				vassal_contract_has_flag = kehillah_muslim_building_free
			}
			domicile.domicile_location.county.holder.top_liege.primary_title = {
				NOT = { kehillah_jewish_settlement_policy_allowed_trigger = yes }
			}
		}
		AND = {
			OR = {
				vassal_contract_has_flag = kehillah_christian_building_permission
				vassal_contract_has_flag = kehillah_muslim_building_authorized
			}
			domicile.domicile_location.county.holder.top_liege.primary_title = {
				kehillah_jewish_settlement_policy_banned_trigger = yes
			}
		}
		# An armed communal watch and high communal autonomy need Encouraged.
		AND = {
			OR = {
				vassal_contract_has_flag = kehillah_christian_security_licensed_watch
				vassal_contract_has_flag = kehillah_christian_security_recognized_watch
				vassal_contract_has_flag = kehillah_muslim_security_licensed_watch
				vassal_contract_has_flag = kehillah_muslim_security_host_protection
				vassal_contract_has_flag = kehillah_christian_jurisdiction_bet_din_discipline
				vassal_contract_has_flag = kehillah_muslim_jurisdiction_bet_din_discipline
				vassal_contract_has_flag = kehillah_muslim_market_chartered
			}
			domicile.domicile_location.county.holder.top_liege.primary_title = {
				NOT = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
			}
		}
		# Ordinary intermediary rights need at least Allowed.
		AND = {
			OR = {
				vassal_contract_has_flag = kehillah_christian_jurisdiction_royal_appeal
				vassal_contract_has_flag = kehillah_muslim_jurisdiction_communal_arbitration
				vassal_contract_has_flag = kehillah_christian_texts_unrestricted
				vassal_contract_has_flag = kehillah_muslim_market_regional
			}
			domicile.domicile_location.county.holder.top_liege.primary_title = {
				NOT = { kehillah_jewish_settlement_policy_allowed_trigger = yes }
			}
		}
		AND = {
			vassal_contract_has_flag = kehillah_christian_texts_licensed
			domicile.domicile_location.county.holder.top_liege.primary_title = {
				kehillah_jewish_settlement_policy_banned_trigger = yes
			}
		}
	}
}
