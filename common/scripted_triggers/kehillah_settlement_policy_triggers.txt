# =====================================================================
# Jewish Settlement Policy (V16, first infrastructure slice).
#
# Scope of the four policy-band triggers: the HOST REALM'S PRIMARY
# TITLE. Policy is stored as an optional numeric title variable, not on
# each community or on the host character, so it is shared by every
# Kehillah beneath that realm and survives ordinary character succession.
#
# Values: 0 = Banned, 1 = Discouraged, 2 = Allowed, 3 = Encouraged.
# Absence deliberately means Allowed. This gives every realm a policy
# without mass-initialising title variables at game start; a later
# policy-change event only needs to store exceptional rungs.
#
# Fresh Christian/Muslim charters now call these in obligation-level
# is_valid/default logic. Existing charters are never rewritten. The
# debug events 72-75 remain the repeatable way to inspect policy bands.
# See docs/spec/v16-jewish-settlement-policy-and-charters.md §§3 and 8.
# =====================================================================

# --- 2026-10-28: THE POLICY IS NOW A REALM LAW ------------------------
# (common/laws/kehillah_jewish_settlement_laws.txt; docs/spec/
# 2026-10-28-jewish-settlement-law.md). Scope is still the host realm's
# primary title; the law is read off its holder. A holder with no law of
# the group (a Jewish ruler, a vassal, a ruler with no reason to have a
# policy) reads as Allowed, exactly as an absent variable did.
#
# THE READY FLAG (2026-10-29, first live look). Loading a save from
# before the law, the engine gave every existing ruler the group's FIRST
# law -- Banned -- not its default (the default is only for new rulers),
# so every realm read as Banned. Until kehillah_settlement_laws_ready is
# set, the law is therefore ignored and the old variable (absent =
# Allowed) is read. kehillah_settlement_laws_normalize_effect sets every
# eligible ruler's law from the old variable, removes the variables, and
# then sets the flag (game start, the first yearly pulse after a load,
# and on opening the realm view or the policy map).
#
# While a title still carries the old variable it wins either way: that
# is a ruler who cannot hold the law, whose variable was never converted.
#
# Note "allowed" means "settlement permitted": Allowed OR Encouraged,
# as it always has (the old var >= 2).

kehillah_jewish_settlement_policy_encouraged_trigger = {
	OR = {
		AND = {
			has_variable = kehillah_jewish_settlement_policy
			var:kehillah_jewish_settlement_policy >= 3
		}
		AND = {
			NOT = { has_variable = kehillah_jewish_settlement_policy }
			has_global_variable = kehillah_settlement_laws_ready
			holder ?= { has_realm_law = kehillah_jewish_settlement_encouraged }
		}
	}
}

kehillah_jewish_settlement_policy_allowed_trigger = {
	OR = {
		AND = {
			has_variable = kehillah_jewish_settlement_policy
			var:kehillah_jewish_settlement_policy >= 2
		}
		AND = {
			NOT = { has_variable = kehillah_jewish_settlement_policy }
			OR = {
				NOT = { has_global_variable = kehillah_settlement_laws_ready }
				NOT = {
					holder ?= {
						OR = {
							has_realm_law = kehillah_jewish_settlement_discouraged
							has_realm_law = kehillah_jewish_settlement_banned
						}
					}
				}
			}
		}
	}
}

kehillah_jewish_settlement_policy_discouraged_trigger = {
	OR = {
		AND = {
			has_variable = kehillah_jewish_settlement_policy
			var:kehillah_jewish_settlement_policy = 1
		}
		AND = {
			NOT = { has_variable = kehillah_jewish_settlement_policy }
			has_global_variable = kehillah_settlement_laws_ready
			holder ?= { has_realm_law = kehillah_jewish_settlement_discouraged }
		}
	}
}

kehillah_jewish_settlement_policy_banned_trigger = {
	OR = {
		AND = {
			has_variable = kehillah_jewish_settlement_policy
			var:kehillah_jewish_settlement_policy <= 0
		}
		AND = {
			NOT = { has_variable = kehillah_jewish_settlement_policy }
			has_global_variable = kehillah_settlement_laws_ready
			holder ?= { has_realm_law = kehillah_jewish_settlement_banned }
		}
	}
}

# Scope: any character. Can they hold the settlement law? Used by the law
# group's can_have_group and every law's potential/can_have/can_keep.
# Daniel, 2026-10-29: only Christian and Muslim rulers -- the faiths whose
# teachings on the Jews the mod models -- or a ruler of any faith with a
# Jewish community already in their realm. Never a Jewish ruler ("a
# Jewish ruler sets no policy toward Jews", 2026-10-19) or a Kehillah.
kehillah_can_hold_settlement_law_trigger = {
	is_independent_ruler = yes
	is_landed = yes
	NOT = { government_has_flag = government_is_kehillah }
	NOT = { religion = religion:judaism_religion }
	OR = {
		religion = religion:christianity_religion
		religion = religion:islam_religion
		any_in_global_list = {
			variable = kehillah_registered_communities
			holder.domicile.domicile_location.county.holder.top_liege ?= prev
		}
	}
}

# Scope: a ruler passing one of the settlement laws (the laws' can_pass;
# the rung-neighbour check is in each law). Carried over from the S9a
# decisions' is_valid.
kehillah_settlement_law_can_pass_trigger = {
	is_imprisoned = no
	# Not before the laws hold the policy (see the READY FLAG note above):
	# a law passed then would be overwritten by the normalizing pass.
	custom_tooltip = {
		text = kehillah_settlement_laws_not_ready_tt
		has_global_variable = kehillah_settlement_laws_ready
	}
	custom_tooltip = {
		text = kehillah_policy_review_cooldown_tt
		NOT = { primary_title = { kehillah_policy_review_on_cooldown_trigger = yes } }
	}
}

# Scope: a community leader. This is the eligibility primitive eventual
# founding/migration routes will use: resolve the same top-liege host as
# V15's Charter, then ask whether that host realm allows settlement.
kehillah_host_policy_allows_jewish_settlement_trigger = {
	exists = domicile.domicile_location.county.holder.top_liege.primary_title
	domicile.domicile_location.county.holder.top_liege.primary_title = {
		kehillah_jewish_settlement_policy_allowed_trigger = yes
	}
}

# Scope: a community leader constructing a *new institution* in the
# Jewish Quarter. Free construction (and legacy/no regional construction
# term) pass directly. Permission/authorisation requires the temporary
# grant supplied by the petition decision; forbidden construction fails.
# Upgrades are intentionally not covered: this row regulates whether a
# new communal building may be founded, not routine repair/expansion of
# an institution the community already possesses.
kehillah_charter_allows_new_communal_building_trigger = {
	OR = {
		# V15 and future non-regional charter packages remain permissive
		# exactly as before. A regional charter with no cache yet fails
		# closed until .0003 has observed its committed rights.
		AND = {
			NOT = { has_subject_contract_group = kehillah_christian_host_charter_prototype }
			NOT = { has_subject_contract_group = kehillah_muslim_host_charter_prototype }
		}
		kehillah_cached_charter_construction_free_trigger = yes
		AND = {
			kehillah_cached_charter_construction_permission_trigger = yes
			has_character_modifier = kehillah_charter_construction_permission_modifier
		}
	}
}

# ---------------------------------------------------------------------
# V16 regional-charter cache readers.
#
# Scope: a community leader. kehillah_refresh_charter_state_cache_effect
# is the sole reader of live tributary flags; it copies compact rungs to
# primary_title after commit, then reconciles quarterly. Gameplay, UI and
# debug readers must use these helpers rather than subject_contract_has_flag.
# ---------------------------------------------------------------------
kehillah_has_cached_regional_charter_state_trigger = {
	primary_title = { has_variable = kehillah_charter_cache_construction }
}

# The second branch of each is the game-start PREDICTION
# (kehillah_predict_start_charter_effect): before the contract has committed,
# the title carries the guessed cache and kehillah_charter_predicted (1 =
# Christian, 2 = Muslim) so the opening pillars can be computed at once. The
# first real cache refresh removes it.
kehillah_cached_christian_charter_trigger = {
	OR = {
		AND = {
			has_subject_contract_group = kehillah_christian_host_charter_prototype
			kehillah_has_cached_regional_charter_state_trigger = yes
		}
		primary_title = { has_variable = kehillah_charter_predicted var:kehillah_charter_predicted = 1 }
	}
}

kehillah_cached_muslim_charter_trigger = {
	OR = {
		AND = {
			has_subject_contract_group = kehillah_muslim_host_charter_prototype
			kehillah_has_cached_regional_charter_state_trigger = yes
		}
		primary_title = { has_variable = kehillah_charter_predicted var:kehillah_charter_predicted = 2 }
	}
}
