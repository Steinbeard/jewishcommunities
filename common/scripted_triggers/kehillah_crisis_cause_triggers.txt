# The crisis window's cause line (docs/spec/2026-11-08-crisis-cause-line.md), 2026-11-08.

# Scope: a community title. Its crisis began with an accusation of this CHARGE (blood_libel,
# mockery, desecration, blasphemy), recorded for this crisis and not an earlier one.
kehillah_crisis_cause_charge_is_trigger = {
 var:kehillah_crisis_opening ?= flag:accusation
 has_variable = kehillah_crisis_charge_epoch
 has_variable = kehillah_crisis_epoch
 var:kehillah_crisis_charge_epoch = var:kehillah_crisis_epoch
 var:kehillah_crisis_charge ?= flag:$CHARGE$
}

# Scope: a community title. Its crisis's opening is KEY.
kehillah_crisis_cause_opening_is_trigger = {
 var:kehillah_crisis_opening ?= flag:$KEY$
}
