# =====================================================================
# SHE'ELOT, SLICE 1 (2026-11-22). docs/spec/2026-11-22-sheelot.md.
# A community sends a hard case by letter to a posek of its own faith, as a
# character interaction (players) or a yearly pulse (AI). The posek answers,
# answers from the sources, or declines; the reply comes back. Asking costs
# a gift to the answering community and is limited by a cooldown; answering
# is the reward of Greatness. Content: the Bet Din docket's 14 cases
# (kehillah_din_letter_N, kehillah_din_answer_N, kehillah_din_answer_careful_N).
# =====================================================================

# Scope: a Kehillah leader. May he send a she'elah now?
kehillah_sheelah_can_ask_trigger = {
	kehillah_sheelah_is_leader_trigger = yes
	NOT = { has_variable = kehillah_sheelah_cooldown }
}

# Scope: a would-be posek. ASKER: the asking leader. Same faith (Daniel,
# 2026-11-22: responsa only between leaders of one faith), a rabbi in post,
# Greatness Healthy or higher, and no other letter in hand.
kehillah_sheelah_posek_trigger = {
	kehillah_sheelah_is_leader_trigger = yes
	NOT = { this = $ASKER$ }
	exists = faith
	faith = $ASKER$.faith
	kehillah_has_community_rabbi_trigger = yes
	kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_greatness THRESHOLD = kehillah_band_healthy_threshold }
	NOT = { has_variable = kehillah_sheelah_pending_from }
	# A player posek receives at most one letter a year (Daniel, 2026-11-22).
	trigger_if = {
		limit = { is_ai = no }
		NOT = { has_variable = kehillah_sheelah_player_letter_recent }
	}
}

# Scope: a community title. The title-side twin of kehillah_pillar_at_least_trigger.
kehillah_pillar_at_least_trigger_title = {
	has_variable = $PILLAR$
	var:$PILLAR$ >= $THRESHOLD$
}

# Scope: a would-be student (the actor of Ask for Help with Your Studies).
kehillah_study_help_can_ask_trigger = {
	any_scheme = { type = kehillah_study_torah }
	exists = var:kehillah_study_current_work_id
}

# Scope: a would-be teacher. STUDENT: the asking student.
kehillah_study_help_teacher_trigger = {
	NOT = { this = $STUDENT$ }
	is_alive = yes
	exists = faith
	faith = $STUDENT$.faith
	kehillah_ordained_or_karaite_trigger = yes
}

# Scope: a character. A community's leader, not one of its courtiers: a
# Kehillah court's courtiers carry the government flag too (live 2026-11-22),
# so the flag alone does not tell a leader from a rabbi at his court.
kehillah_sheelah_is_leader_trigger = {
	is_ruler = yes
	government_has_flag = government_is_kehillah
	exists = primary_title
}

# ---------------------------------------------------------------------
# Slices 2 and 3 (2026-11-22): "follows the rulings of".
# ---------------------------------------------------------------------
# Scope: a Kehillah leader. LEADER: another leader. Does our community
# follow theirs?
kehillah_sheelah_follows_leader_trigger = {
	exists = primary_title
	exists = $LEADER$.primary_title
	primary_title.var:kehillah_sheelah_follows ?= $LEADER$.primary_title
}

# Scope: a community title. Is its tie still sound: the followed community
# led by a leader of the follower's faith?
kehillah_sheelah_tie_sound_trigger = {
	exists = var:kehillah_sheelah_follows
	exists = holder
	var:kehillah_sheelah_follows = {
		exists = holder
		holder = { government_has_flag = government_is_kehillah }
	}
	holder.faith = var:kehillah_sheelah_follows.holder.faith
}

# Scope: a Kehillah leader. Is the rite of the community ours follows
# offered to us now? A strong, sound tie to a community of another rite of
# our faith, and no offer within the last ten years.
kehillah_sheelah_rite_pull_trigger = {
	kehillah_sheelah_is_leader_trigger = yes
	primary_title = {
		kehillah_sheelah_tie_sound_trigger = yes
		var:kehillah_sheelah_follows_strength >= kehillah_sheelah_tie_strong
		NOT = { has_variable = kehillah_sheelah_rite_pull_cooldown }
	}
	exists = rite
	exists = primary_title.var:kehillah_sheelah_follows.holder.rite
	NOT = { rite = primary_title.var:kehillah_sheelah_follows.holder.rite }
}
