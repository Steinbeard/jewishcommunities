# Scope: a character. A Jewish ruler who holds land (not a Kehillah leader).
kehillah_is_jewish_landed_ruler_trigger = {
	is_landed = yes
	NOT = { government_has_flag = government_is_kehillah }
	faith.religion = religion:judaism_religion
}

# Scope: a landed title. A library title: in the registry, with a holder.
kehillah_lib_is_library_title_trigger = {
	is_target_in_global_variable_list = { name = kehillah_libraries target = this }
	exists = holder
}

# Scope: a character. The library this character reads from at home holds
# $WORK$ (a flag:work_* literal). A Kehillah leader's is his Kehillah's; a
# landed ruler's is var:kehillah_own_library (a temple barony library he
# holds, or one in his realm).
kehillah_lib_own_library_has_trigger = {
	OR = {
		AND = {
			government_has_flag = government_is_kehillah
			primary_title = { is_target_in_variable_list = { name = kehillah_library_works target = $WORK$ } }
		}
		var:kehillah_own_library ?= { is_target_in_variable_list = { name = kehillah_library_works target = $WORK$ } }
	}
}

# Any scope. May $CHARACTER$ start a course of study from library title
# $LIBRARY$? 2026-11-08 (Daniel): "adventurers ... study from the library of
# the county they're based in ... landed Jewish rulers ... any library in
# their domain". The library must be held and unsealed, and be one of:
#   - his own Kehillah's title, his own library (var:kehillah_own_library),
#     or a library title he holds;
#   - seated in the county he stands in, or the county of his domicile (an
#     adventurer's camp, a Kehillah's quarter);
#   - for a landed ruler, seated in a county of his domain (held directly).
kehillah_study_can_read_library_trigger = {
	$LIBRARY$ = {
		exists = holder
		NOT = { has_variable = kehillah_library_sealed }
		OR = {
			AND = {
				$CHARACTER$ = { government_has_flag = government_is_kehillah }
				this = $CHARACTER$.primary_title
			}
			$CHARACTER$.var:kehillah_own_library ?= this
			holder = $CHARACTER$
			AND = {
				exists = var:kehillah_library_seat
				OR = {
					var:kehillah_library_seat.county = $CHARACTER$.location.county
					AND = {
						exists = $CHARACTER$.domicile
						var:kehillah_library_seat.county = $CHARACTER$.domicile.domicile_location.county
					}
					AND = {
						$CHARACTER$ = { is_landed = yes }
						var:kehillah_library_seat.county.holder ?= $CHARACTER$
					}
				}
			}
		}
	}
}

# Scope: a character. May start a selected-book study from the Library.
# Mirrors the Learn Torah decision: an ordained rabbi, or a lay reader of a
# rite with Talmud Torah, with no course already running.
kehillah_lib_can_start_selected_study_trigger = {
	OR = {
		government_has_flag = government_is_kehillah
		kehillah_is_jewish_landed_ruler_trigger = yes
		# 2026-11-08: adventurers read the libraries of the county they are in.
		has_government = landless_adventurer_government
	}
	kehillah_torah_reader_eligible_trigger = yes
	is_adult = yes
	is_imprisoned = no
	is_jewish_character_trigger = yes
	piety >= 10
	NOT = { any_scheme = { type = kehillah_study_torah } }
	can_start_scheme = { type = kehillah_study_torah target_character = root }
}

# Standing gates (spec section 3.4). Scope: a rite; $KEY$ the work id
# (no flag: prefix). True when the work may rise from its current tier to
# the next. Counters are global per work (kehillah_wk_<key>_*), seeded by
# kehillah_standing_seed_work_effect; copies count every library.
#
# 2026-11-11, one scale (docs/spec/2026-11-11-standing-one-scale.md). Each
# gate is what its tier means: Popular is read, Celebrated praised, Esteemed
# taught, Authoritative relied on. An Anathema (0) lifts only when its term
# (var:kehillah_anathema_<key>, kehillah_anathema_years) runs out.
#
# 2026-11-11, gates by genre (docs/spec/2026-11-11-rulings-sermons-and-
# gates.md section 1): Popular and Celebrated come from points alone;
# Esteemed needs kehillah_gate_esteemed_masters scholars who mastered it;
# Authoritative, 30 years and its author dead, and by genre: Talmudics cited
# in rulings, Parshanut preached from, Machshavah changing readers' minds,
# kehillah_gate_authoritative_count times each. The `preached` and `minds`
# counters are filled by sermons and by Machshavah (later steps); until
# then no Parshanut or Machshavah book can rise to Authoritative.
# ---------------------------------------------------------------------
kehillah_standing_next_gate_met_trigger = {
	OR = {
		AND = {
			# to Banned
			var:kehillah_standing_$KEY$ = 0
			NOT = { has_variable = kehillah_anathema_$KEY$ }
		}
		AND = {
			# to Rejected, Contested, Obscure, Popular, Celebrated: points alone
			var:kehillah_standing_$KEY$ >= 1
			var:kehillah_standing_$KEY$ <= 5
		}
		AND = {
			# to Esteemed: mastered by enough scholars
			var:kehillah_standing_$KEY$ = 6
			global_var:kehillah_wk_$KEY$_masters >= kehillah_gate_esteemed_masters
		}
		AND = {
			# to Authoritative: relied on, by genre
			var:kehillah_standing_$KEY$ = 7
			current_year >= { value = kehillah_lib_year_$KEY$ add = kehillah_gate_authoritative_years }
			OR = {
				NOT = { exists = global_var:kehillah_wk_$KEY$_author }
				global_var:kehillah_wk_$KEY$_author = { is_alive = no }
			}
			OR = {
				AND = {
					kehillah_lib_genre_$KEY$ = 2
					global_var:kehillah_wk_$KEY$_citations >= kehillah_gate_authoritative_count
				}
				AND = {
					kehillah_lib_genre_$KEY$ = 1
					has_global_variable = kehillah_wk_$KEY$_preached
					global_var:kehillah_wk_$KEY$_preached >= kehillah_gate_authoritative_count
				}
				AND = {
					kehillah_lib_genre_$KEY$ = 3
					has_global_variable = kehillah_wk_$KEY$_minds
					global_var:kehillah_wk_$KEY$_minds >= kehillah_gate_authoritative_count
				}
			}
		}
	}
}

# Scope: an author. He may argue for $TENET$: he holds it personally, has
# learned it from a book, or his rite holds it as core (or, playable, he
# knows it).
kehillah_sefer_author_knows_tenet_trigger = {
	OR = {
		has_personal_tenet = $TENET$
		is_target_in_variable_list = { name = kehillah_book_tenets target = flag:$TENET$ }
		rite ?= { rite_has_tenet = $TENET$ }
		AND = {
			is_playable_character = yes
			knows_tenet = $TENET$
		}
	}
}

	faith.religion = religion:judaism_religion

# Scope: a library title. Its foundations are complete: a Rabbanite
# library's six Bavli orders, a Karaite library's founding works.
kehillah_lib_foundations_complete_trigger = {
	OR = {
		AND = {
			kehillah_lib_is_karaite_library_trigger = yes
			kehillah_lib_karaite_missing_count_value <= 0
		}
		AND = {
			NOT = { kehillah_lib_is_karaite_library_trigger = yes }
			NOT = { kehillah_lib_is_haymanot_library_trigger = yes }
			kehillah_lib_complete_shas_trigger = yes
		}
	}
}
