# --- Scene pages (2026-11-01, kehillah_crisis_act_page_start_effect) ---

# Scope: community leader. Action KEY is listed in this scene and falls on the page shown; the AI
# sees every page. Written as trigger_ifs so it never appears in an option's tooltip, and chained so
# an unset variable is never read. (A NOT around several triggers is a NOR in CK3: the first version
# wrapped them all in one NOT, and every listed action showed on every page.)
kehillah_crisis_act_on_page_trigger = {
 trigger_if = {
  limit = { NOT = { exists = var:kehillah_act_listing_$KEY$ } }
  always = no
 }
 trigger_else_if = {
  limit = { NOT = { var:kehillah_act_listing_$KEY$ = var:kehillah_act_listing } }
  always = no
 }
 trigger_else_if = {
  limit = {
   is_ai = no
   OR = {
    var:kehillah_act_slot_$KEY$ <= var:kehillah_act_page_lo
    var:kehillah_act_slot_$KEY$ > var:kehillah_act_page_hi
   }
  }
  always = no
 }
 trigger_else = { always = yes }
}

# Scope: community leader. More actions are listed than this page shows.
kehillah_crisis_act_more_trigger = {
 is_ai = no
 var:kehillah_act_count > var:kehillah_act_page_hi
}

# Scope: community leader. This is the scene's last page (the AI's only one).
kehillah_crisis_act_last_page_trigger = {
 OR = {
  is_ai = yes
  var:kehillah_act_count <= var:kehillah_act_page_hi
 }
}
