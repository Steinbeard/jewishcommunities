# =====================================================================
# Machloket: when a book can start a controversy (2026-11-25).
# Rite scope; $KEY$ is the work id. Read by kehillah_machloket_yearly_effect.
#
# Replaces the 2026-11-11 gate "standing 6+ and at a ceiling below 8".
# That gate fired on settled books: Emunot ve-Deot is seeded at Esteemed
# (7) under a ceiling of 7 while Hokhmah is only permitted, so it rolled
# every year from 1066, two centuries after its own disputes. It also kept
# a new Moreh Nevukhim quiet until it climbed to Esteemed. A controversy
# now needs four things:
#   fresh       written within kehillah_machloket_novelty_years, or its
#               author still alive;
#   read now    reception moved in the last ten years (kehillah_touched_);
#   read widely Popular or higher;
#   held back   a tenet of the rite caps it below Authoritative, and it
#               stands within kehillah_machloket_ceiling_gap tiers of that cap.
# =====================================================================

kehillah_machloket_can_erupt_trigger = {
	NOT = { has_variable = kehillah_mk_active }
	NOT = { has_global_variable = kehillah_wk_$KEY$_mk_cooldown }
	has_variable = kehillah_standing_$KEY$
	# Not before the work exists: the 867 start disputed Saadia's
	# Emunot ve-Deot (933) in 867 (2026-11-17 smoke test).
	current_year >= kehillah_lib_year_$KEY$
	OR = {
		current_year <= {
			value = kehillah_lib_year_$KEY$
			add = kehillah_machloket_novelty_years
		}
		global_var:kehillah_wk_$KEY$_author ?= { is_alive = yes }
	}
	has_variable = kehillah_touched_$KEY$
	var:kehillah_standing_$KEY$ >= kehillah_machloket_min_standing
	kehillah_lib_ceiling_$KEY$ < 8
	var:kehillah_standing_$KEY$ >= {
		value = kehillah_lib_ceiling_$KEY$
		subtract = kehillah_machloket_ceiling_gap
	}
}
