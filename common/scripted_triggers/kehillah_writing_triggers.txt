# Entry requirements for forms of writing. These also drive the disabled
# choices in kehillah_writing.0002-0004, so the UI and effect share a gate.

kehillah_writing_scripture_language_trigger = {
	OR = {
		AND = { faith = faith:haymanot knows_language = language_ethiopic }
		AND = { NOT = { faith = faith:haymanot } knows_language = language_israelite }
	}
}

kehillah_writing_torah_trigger = {
	kehillah_writing_scripture_language_trigger = yes
	is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_tanakh_torah }
}
kehillah_writing_midrash_trigger = {
	kehillah_writing_torah_trigger = yes
	is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_genesis_rabbah }
}
kehillah_writing_poetry_trigger = {
	kehillah_writing_scripture_language_trigger = yes
	has_trait = lifestyle_poet
	kehillah_lib_tanakh_mastered_count_value >= 1
}
kehillah_writing_grammar_trigger = {
	kehillah_writing_torah_trigger = yes
	OR = {
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_samaritan_targum }
		knows_language = language_arabic
	}
}
kehillah_writing_chronicle_trigger = {
	kehillah_writing_scripture_language_trigger = yes
	kehillah_lib_tanakh_mastered_count_value >= 1
	learning >= 12
}

kehillah_writing_talmud_trigger = {
	rite ?= { kehillah_rite_rabbanite_trigger = yes }
	knows_language = language_aramaic
	kehillah_lib_bavli_mastered_count_value >= 1
}
kehillah_writing_yerushalmi_trigger = {

	kehillah_lib_yerushalmi_mastered_count_value >= 1
}
kehillah_writing_law_trigger = {

	kehillah_lib_bavli_mastered_count_value >= 1
}
kehillah_writing_responsa_trigger = {
	kehillah_writing_law_trigger = yes
	kehillah_has_good_bet_din_memory_trigger = yes
}
kehillah_writing_minhag_trigger = {
	kehillah_writing_scripture_language_trigger = yes
	OR = {
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_orit }
	}
}
kehillah_writing_siddur_trigger = {

	is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }

kehillah_writing_dialogue_trigger = {
	kehillah_writing_torah_trigger = yes
	OR = {
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mishnah }
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_philo_allegorical }
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_memar_marqah }
	}

kehillah_writing_systematic_dialogue_trigger = {
	current_date >= 933.1.1
	OR = {
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_emunot_vedeot }
		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_chovot_halevavot }
	}
	OR = { knows_language = language_israelite knows_language = language_arabic }
}

	}
}

	OR = {

		is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_galen_de_usu_partium }
	}
}

	has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 65 }
