# A decree against the Jews as a situation (docs/spec/2026-11-01-decree-situation.md), 2026-11-01.

# Scope: a Kehillah leader. A decree may begin: no attack, trial or other decree running.
kehillah_decree_can_begin_trigger = {
	exists = primary_title
	exists = domicile.domicile_location.county.holder.top_liege
	primary_title = {
		NOT = { has_variable = kehillah_mob_threat }
		NOT = { exists = var:kehillah_trial_situation }
		NOT = { exists = var:kehillah_decree_situation }
	}
}

# Scope: a Kehillah leader. His community's decree is still Proclaimed: its answers are open.
kehillah_decree_open_trigger = {
	primary_title.var:kehillah_decree_situation ?= { situation_current_phase = kehillah_decree_proclaimed }
}

# Scope: a Kehillah leader. The running decree is of this kind (badge, ghiyar, servants).
kehillah_decree_kind_trigger = {
	primary_title.var:kehillah_decree_situation.var:kehillah_decree_kind ?= flag:$KIND$
}

# Scope: a Kehillah leader. An answer not yet given to this decree.
kehillah_decree_not_done_trigger = {
	custom_description = {
		text = kehillah_decree_done_already_tt
		primary_title = { NOT = { has_variable = kehillah_decree_did_$ONCE$ } }
	}
}

# Scope: a Kehillah leader, with scope:kehillah_incident_host. Another community lives under the
# same sovereign, to petition with.
kehillah_decree_has_sister_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		NOT = { this = root.primary_title }
		exists = holder
		holder = { kehillah_is_host_of_community_trigger = { HOST = scope:kehillah_incident_host } }
	}
}

# Scope: a Kehillah leader. The running decree's rolled detail VAR is VALUE (kehillah_decree_roll_effect).
kehillah_decree_detail_trigger = {
	primary_title.var:kehillah_decree_situation.var:kehillah_decree_$VAR$ ?= flag:$VALUE$
}
