# Scope: a reader. $TENET$ is not yet his: a playable character does not
# hold it as a personal tenet and could; anyone else has no leaning to it
# (var:kehillah_leaning_<tenet>; vanilla gives personal tenets only to
# playable characters).
kehillah_minds_tenet_open_trigger = {
	OR = {
		AND = {
			is_playable_character = yes
			NOT = { has_personal_tenet = $TENET$ }
			can_pick_personal_tenet = $TENET$
		}
		AND = {
			is_playable_character = no
			NOT = { has_variable = kehillah_leaning_$TENET$ }
		}
	}
}

# Scope: a reader. The tenet of index $VALUE$ (a global variable of a book
# written in play) is not yet his.
kehillah_minds_index_open_trigger = {
	OR = {
		AND = { $VALUE$ = 1 kehillah_minds_tenet_open_trigger = { TENET = tenet_talmud_torah } }
		AND = { $VALUE$ = 2 kehillah_minds_tenet_open_trigger = { TENET = tenet_baalei_hamiqra } }
		AND = { $VALUE$ = 3 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_hokhmah } }
		AND = { $VALUE$ = 4 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_emunah_temimah } }
		AND = { $VALUE$ = 5 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_pesak } }
		AND = { $VALUE$ = 6 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_pilpul } }
		AND = { $VALUE$ = 7 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_geonut } }
		AND = { $VALUE$ = 8 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_minhag_avoteinu } }
		AND = { $VALUE$ = 9 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_avelei_tziyon } }
		AND = { $VALUE$ = 10 kehillah_minds_tenet_open_trigger = { TENET = tenet_esotericism } }
		AND = { $VALUE$ = 11 kehillah_minds_tenet_open_trigger = { TENET = tenet_asceticism } }
		AND = { $VALUE$ = 12 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_kiddush_hashem } }
	}
}

# Scope: a reader with var:kehillah_minds_tenet. His rite holds that tenet
# at exactly $STATUS$.
kehillah_minds_tenet_status_trigger = {
	OR = {
		AND = { var:kehillah_minds_tenet = 1 rite ?= { has_tenet_status = { tenet = tenet_talmud_torah status = $STATUS$ } } }
