# Scope: a reader. $TENET$ is not yet his: a playable character does not
# hold it as a personal tenet and could; anyone else has no leaning to it
# (var:kehillah_leaning_<tenet>; vanilla gives personal tenets only to
# playable characters).
kehillah_minds_tenet_open_trigger = {
	OR = {
		AND = {
			is_playable_character = yes
			NOT = { has_personal_tenet = $TENET$ }
			can_pick_personal_tenet = $TENET$
		}
		AND = {
			is_playable_character = no
			NOT = { has_variable = kehillah_leaning_$TENET$ }
		}
	}
}

# Scope: a reader. The tenet of index $VALUE$ (a global variable of a book
# written in play) is not yet his.
kehillah_minds_index_open_trigger = {
	OR = {
		AND = { $VALUE$ = 1 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_scholarly_study } }
		AND = { $VALUE$ = 2 kehillah_minds_tenet_open_trigger = { TENET = tenet_literalism } }
		AND = { $VALUE$ = 3 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_hokhmah } }
		AND = { $VALUE$ = 4 kehillah_minds_tenet_open_trigger = { TENET = tenet_unrelenting_faith } }
		AND = { $VALUE$ = 5 kehillah_minds_tenet_open_trigger = { TENET = tenet_religious_legal_pronouncements } }
		AND = { $VALUE$ = 6 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_scholarly_study } }
		AND = { $VALUE$ = 7 kehillah_minds_tenet_open_trigger = { TENET = tenet_religious_legal_pronouncements } }
		AND = { $VALUE$ = 8 kehillah_minds_tenet_open_trigger = { TENET = tenet_communal_identity } }
		AND = { $VALUE$ = 9 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_avelei_tziyon } }
		AND = { $VALUE$ = 10 kehillah_minds_tenet_open_trigger = { TENET = tenet_esotericism } }
		AND = { $VALUE$ = 11 kehillah_minds_tenet_open_trigger = { TENET = tenet_asceticism } }
		AND = { $VALUE$ = 12 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_kiddush_hashem } }
		AND = { $VALUE$ = 13 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_sacred_liturgy } }
		AND = { $VALUE$ = 14 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_covenant_community } }
		AND = { $VALUE$ = 15 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_messianic_hope } }
		AND = { $VALUE$ = 16 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_prophetic_revelation } }
		AND = { $VALUE$ = 17 kehillah_minds_tenet_open_trigger = { TENET = tenet_kehillah_temple_service } }
	}
}
