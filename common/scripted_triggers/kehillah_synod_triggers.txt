# V45 rabbinic synods (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md).
# A synod is a Sitting of the Bet Din held with the Synod or Sanhedrin intent
# (common/activities/intents/kehillah_synod_intents.txt).

# Scope: a character with a capital county (a Kehillah leader). Ashkenaz is
# the two Ashkenazi minhag regions, plus England, which
# kehillah_shares_minhag_region_trigger already treats as one region of its
# own and whose Jews came from Rouen.
kehillah_in_ashkenazi_region_trigger = {
	exists = capital_county
	OR = {
		capital_county = { empire = title:e_britannia }
		AND = {
			capital_county.de_jure_liege = { has_variable = kehillah_minhag }
			OR = {
				capital_county.de_jure_liege.var:kehillah_minhag = flag:western_ashkenaz
				capital_county.de_jure_liege.var:kehillah_minhag = flag:eastern_ashkenaz
			}
		}
	}
}

# Scope: a character. A Kehillah leader in Ashkenaz, or someone at such a
# leader's court: the people Minhag Ashkenaz takes in when it is founded,
# whatever culture the game gave them.
kehillah_ashkenazi_kehillah_member_trigger = {
	OR = {
		AND = {
			government_has_flag = government_is_kehillah
			kehillah_in_ashkenazi_region_trigger = yes
		}
		employer ?= {
			government_has_flag = government_is_kehillah
			kehillah_in_ashkenazi_region_trigger = yes
		}
	}
}

# Scope: a character. The Nasi: holder of the Sanhedrin's seat county in the
# Land, which exists only once restored (V47 phase B,
# kehillah_restore_sanhedrin_decision).
kehillah_is_sanhedrin_head_trigger = {
	exists = global_var:kehillah_sanhedrin_seat
	global_var:kehillah_sanhedrin_seat.holder ?= this
}

# Scope: the host's involved activity is a sitting with the Sanhedrin intent.
kehillah_synod_is_sanhedrin_trigger = {
	involved_activity.var:kehillah_synod_level ?= flag:sanhedrin
}

# Scope: a character attending a synod. HOST: its host. Delegates vote: at
# both a Synod and a Sanhedrin, every other attending community leader.
# Attendance and the reach of the resulting ruling are separate.
kehillah_synod_delegate_trigger = {
	NOT = { this = $HOST$ }
	is_alive = yes
	government_has_flag = government_is_kehillah
	# 2026-11-15: a Synod speaks for its rite, so only leaders of the host's
	# rite vote. A Sanhedrin binds all Israel; anyone attending votes.
	trigger_if = {
		limit = { $HOST$ = { NOT = { kehillah_synod_is_sanhedrin_trigger = yes } } }
		kehillah_shares_rite_trigger = { OTHER = $HOST$ }
	}
}

# 2026-11-15: the scoped character keeps the same rite as OTHER. Synod
# rulings bind by rite, replacing the minhag region (Daniel's direction;
# docs/spec/2026-11-15-ai-hosted-sittings-via-stand-in.md).
kehillah_shares_rite_trigger = {
	exists = rite
	exists = $OTHER$.rite
	rite = { this = $OTHER$.rite }
}

# Scope: the host, inside a synod event (root = the host).
kehillah_synod_quorum_present_trigger = {
	trigger_if = {
		limit = { kehillah_synod_is_sanhedrin_trigger = yes }
		involved_activity ?= {
			any_attending_character = {
				count >= kehillah_sanhedrin_quorum
				kehillah_synod_delegate_trigger = { HOST = root }
			}
		}
	}
	trigger_else = {
		involved_activity ?= {
			any_attending_character = {
				count >= kehillah_synod_quorum
				kehillah_synod_delegate_trigger = { HOST = root }
			}
		}
	}
}
kehillah_synod_motion_carried_trigger = {
	involved_activity ?= {
		has_variable = kehillah_synod_votes_for
		has_variable = kehillah_synod_votes_against
		var:kehillah_synod_votes_for > var:kehillah_synod_votes_against
	}
}

# Scope: a would-be host. Enough other community leaders to make a quorum
# if they come. Every Kehillah leader is invited to both tiers.
# 2026-11-15: enough OTHER communities of the host's rite to make a quorum
# (was: any registered community). The name is kept for its callers.
kehillah_synod_region_has_quorum_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		count >= kehillah_synod_quorum
		holder ?= {
			NOT = { this = root }
			government_has_flag = government_is_kehillah
			kehillah_shares_rite_trigger = { OTHER = root }
		}
	}
}
kehillah_sanhedrin_has_quorum_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		count >= kehillah_sanhedrin_quorum
		holder ?= {
			NOT = { this = root }
			government_has_flag = government_is_kehillah
		}
	}
}

# Scope: a would-be host. Each takkanah's own availability, shared by the
# motion events and the intents' "anything left to enact" gate.

	# 2026-11-15: keeping Minhag Ashkenaz replaces sitting in the Ashkenazi

	exists = rite
	rite = { this = rite:minhag_ashkenaz }
	NOT = { exists = global_var:kehillah_gershom_herem_renewed }
}

    exists = primary_title
    OR = {
        kehillah_synod_can_propose_herem_person_trigger = yes
        kehillah_synod_rite_headship_open_trigger = yes
        AND = { kehillah_reform_any_motion_trigger = yes OR = { is_ai = no rite ?= { has_variable = kehillah_reform_pending_tenet } } }
        AND = { is_ai = no kehillah_synod_has_law_motion_trigger = yes }
    }
}

# 2026-11-25 dynamic agenda: the headship of the rite.
# ---------------------------------------------------------------------
# Scope: a would-be host. His rite may take a head by merit and has none,
# and nobody holds the faith's head-of-rite title (V47).
kehillah_synod_rite_headship_open_trigger = {
	faith = faith:rabbinic_faith
	exists = rite
	rite = { rite_has_doctrine = kehillah_recognition_of_sages }
	kehillah_rite_has_no_head_trigger = yes
	NOT = { exists = title:d_kehillah_gaonate.holder }
}
# Scope: a character the sitting might name head of the rite. A Synod may
# raise a qualified head of an academy without the full merit the yearly
# acclamation needs (kehillah_head_of_rite_merit_trigger): the gathering's
# vote stands in for the fame of his yeshiva. HOST: the host.
kehillah_synod_head_candidate_trigger = {
	is_alive = yes
	kehillah_head_of_rite_leader_trigger = yes
	kehillah_recognizes_gaonic_authority_trigger = yes
	kehillah_qualified_rosh_yeshiva_trigger = yes
	kehillah_shares_rite_trigger = { OTHER = $HOST$ }
	NOT = { has_character_flag = kehillah_declined_head_of_rite }
}
# Scope: the host, in a synod event. Someone at the sitting could be named.
kehillah_synod_can_propose_head_trigger = {
	kehillah_synod_rite_headship_open_trigger = yes
	involved_activity ?= {
		any_attending_character = {
			kehillah_synod_head_candidate_trigger = { HOST = root }
		}
	}
}

# Completion callback version: never uses the hosts involved_activity link.
# scope:activity is supplied by on_complete; scope:kehillah_synod_level is captured.
kehillah_synod_closing_quorum_trigger = {
	trigger_if = {
		limit = { scope:kehillah_synod_level ?= flag:sanhedrin }
		scope:activity = {
			any_attending_character = {
				count >= kehillah_sanhedrin_quorum
				NOT = { this = root }
				is_alive = yes
				government_has_flag = government_is_kehillah
			}
		}
	}
	trigger_else = {
		scope:activity = {
			any_attending_character = {
				count >= kehillah_synod_quorum
				NOT = { this = root }
				is_alive = yes
				government_has_flag = government_is_kehillah
				kehillah_shares_rite_trigger = { OTHER = root }
			}
		}
	}
}
