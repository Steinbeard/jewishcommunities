# Scope: a Kehillah leader with a regional charter. Some term could be a
# step better (the cached 0-2 levels; text_study and market_access 2 =
# best, construction 0 = best).
kehillah_charter_can_improve_trigger = {
	OR = {
		has_subject_contract_group = kehillah_christian_host_charter_prototype
		has_subject_contract_group = kehillah_muslim_host_charter_prototype
	}
	primary_title = {
		OR = {
			AND = {
				has_variable = kehillah_charter_cache_security
				var:kehillah_charter_cache_security < 2
			}
			AND = {
				has_variable = kehillah_charter_cache_construction
				var:kehillah_charter_cache_construction > 0
			}
			AND = {
				has_variable = kehillah_charter_cache_jurisdiction
				var:kehillah_charter_cache_jurisdiction < 2
			}
			AND = {
				has_variable = kehillah_charter_cache_text_study
				var:kehillah_charter_cache_text_study < 2
			}
			AND = {
				has_variable = kehillah_charter_cache_market_access
				var:kehillah_charter_cache_market_access < 2
			}
		}
	}
}
