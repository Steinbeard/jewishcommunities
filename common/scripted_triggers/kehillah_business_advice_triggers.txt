	government_has_flag = government_is_kehillah
	exists = primary_title
	exists = domicile
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
}

# Planning advice, not a promise about the final native price. Deliberately limited
# to the three universal first-tier Craft Row trades, whose permission/cost match.
kehillah_business_expansion_advice_trigger = {
	kehillah_business_advice_available_trigger = yes
	is_at_war = no
	kehillah_charter_allows_new_communal_building_trigger = yes
	gold >= kehillah_business_craft_reserve_value
	NOT = { kehillah_business_has_vacancy_trigger = yes }
	any_courtier = { kehillah_business_unassigned_head_trigger = yes is_courtier_of = prev house.var:kehillah_household_of = prev.primary_title }
	domicile = {
		NOT = { has_ongoing_domicile_construction = yes }
		OR = {
			domicile_building_has_free_internal_slot = kehillah_workshops_01
			domicile_building_has_free_internal_slot = kehillah_workshops_02
			domicile_building_has_free_internal_slot = kehillah_workshops_03
		}
	}
	OR = {
		NOT = { kehillah_business_dyers_built_trigger = yes }
		NOT = { kehillah_business_goldsmiths_built_trigger = yes }
		NOT = { kehillah_business_tanners_built_trigger = yes }
	}
