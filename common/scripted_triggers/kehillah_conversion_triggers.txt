# =====================================================================
# FORCED CONVERSION OF A KEHILLAH LEADER -- who may demand it.
# Design: docs/spec/v31-host-teachings-stances-and-factions.md, section
# 10.1.
#
# Vanilla already lets a host demand a Kehillah leader's conversion: the
# vassal-ruler Demand Conversion interaction (00_religious_interactions
# .txt, demand_conversion_vassal_ruler_interaction) is shown for an AI
# recipient who is the actor's TRIBUTARY, and the Host Charter is a
# tributary contract. The player-recipient variant needs a liege, so a
# player-led community is not reachable by it.
#
# V31 gates that interaction by the host's teaching on the Jews
# (kehillah_teaching_permits_forced_conversion_trigger, Slice A): the
# Strict Pact, Exclusion and Abolition allow it; Witness, Separation and
# Mission, Dhimma and Imperial Law (no decree yet) forbid it. A host with
# no teaching (non-Abrahamic) may once his realm bans Jewish settlement,
# the rule this gate used before the teachings existed.
#
# Voluntary asking (valid_ask_for_conversion_conditions_trigger) is NOT
# gated: a free conversion was always possible under every teaching.
# =====================================================================

# $ACTOR$ demands that scope:recipient convert. Called only from the
# overridden valid_demand_conversion_conditions_trigger
# (kehillah_forced_conversion_override.txt).
kehillah_host_demand_conversion_allowed_trigger = {
	custom_tooltip = {
		text = kehillah_host_demand_conversion_protected_tt
		trigger_if = {
			limit = {
				scope:recipient = {
					government_has_flag = government_is_kehillah
					is_jewish_character_trigger = yes
				}
				$ACTOR$ = { is_jewish_character_trigger = no }
			}
			$ACTOR$ = { kehillah_teaching_permits_forced_conversion_trigger = { RECIPIENT = scope:recipient } }
		}
	}
}

# Scope: a character. A Kehillah leader who no longer holds a Jewish
# faith, i.e. who must leave office.
kehillah_leader_left_judaism_trigger = {
	government_has_flag = government_is_kehillah
	exists = primary_title
	primary_title = { is_kehillah_title_trigger = yes }
	is_jewish_character_trigger = no
}

# Scope: a Kehillah leader. The community's appointment heir exists AND is
# Jewish. A pogrom baptizes the whole court, officers and kin included, so
# the pool can be entirely converted; handing the title to such an heir
# leaves a Muslim/Christian "Kehillah" (live 2026-10-20, Damascus). With no
# Jewish heir the community dissolves instead.
kehillah_conversion_heir_is_jewish_trigger = {
	exists = primary_title.current_heir
	primary_title.current_heir = { is_jewish_character_trigger = yes }
}
