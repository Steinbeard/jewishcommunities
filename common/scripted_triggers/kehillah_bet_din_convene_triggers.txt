# in 20 years, by the player or any AI). One trigger for "this leader could
# start a Bet Din conference now", mirroring the activity's own is_shown and

kehillah_can_convene_bet_din_trigger = {
	government_has_flag = government_is_kehillah
	is_adult = yes
	is_imprisoned = no
	# 2026-11-15: minhag-region gate removed (regions retired).
	NOT = { exists = involved_activity }
	NOT = { is_activity_type_on_cooldown = activity_kehillah_bet_din_conference }
	kehillah_no_crisis_blocks_hosting_trigger = yes
	kehillah_region_can_field_bet_din_panel_trigger = yes
	kehillah_bet_din_has_presiding_authority_trigger = yes # 2026-11-25, kehillah_bet_din_benches_triggers.txt
}
