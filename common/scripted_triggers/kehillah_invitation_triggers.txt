# =====================================================================
# INVITATIONS -- V34 section 7 (docs/spec/v34-outlook-teachings-and-
# founding.md), 2026-10-13. A ruler with no Jews in his realm may still
# open the door: a yearly "open-door review", driven by his outlook on
# the Jews and his wish to fill towns and markets, can raise his Jewish
# Settlement Policy. Raising it to Encouraged -- by this review, the
# hosts' review or the player's decree -- IS the invitation: Kehillah
# players within diplomatic range hear of it, and for 20 years it draws
# settlers (Send Out Settlers, exiles) and lets a community that is not
# overcrowded answer the call.
# (The historical shape: Bela IV of Hungary after the Mongols, 1251;
# Boleslaw the Pious's Statute of Kalisz, 1264; the Rhineland bishops'
# charters of 1084/1090.)
# =====================================================================

# Scope: a character. An AI ruler who could open his realm's door: an
# independent duke or greater, Christian or Muslim (the faiths a founding
# site needs), whose policy is Discouraged or Allowed and not under
# review cooldown, and who hosts no community yet (the hosts' review,
# kehillah_policy_ai_review_try_effect, covers those). V44 (2026-10-15):
# a theocracy may invite (Avignon and the Comtat's "Jews of the Pope",
# Daniel); the ruler must be well-disposed, have a motive
# (kehillah_invitation_motive_value), and border Jewish settlement
# (kehillah_open_door_near_community_trigger).
kehillah_open_door_candidate_trigger = {
	is_ai = yes
	is_landed = yes
	is_independent_ruler = yes
	highest_held_title_tier >= tier_duchy
	exists = primary_title
	NOT = { government_has_flag = government_is_kehillah }
	OR = {
		faith.religion = religion:christianity_religion
		faith.religion = religion:islam_religion
	}
	primary_title = {
		NOT = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
		NOT = { kehillah_jewish_settlement_policy_banned_trigger = yes }
		NOT = { kehillah_policy_review_on_cooldown_trigger = yes }
	}
	NOT = { kehillah_ruler_hosts_a_community_trigger = yes }
	NOT = { kehillah_ruler_has_active_community_crisis_trigger = yes }
	# V44 (2026-10-15, 20-year soak: eleven invitations in fifteen years,
	# from Pisa to Kairouan): well-disposed or better, with a concrete
	# reason, and Jews just over the border.
	kehillah_outlook_on_jews_value >= kehillah_invitation_min_outlook
	kehillah_invitation_motive_value > 0
	kehillah_open_door_near_community_trigger = yes
}

# Scope: a ruler. Some county of his realm lies within
# squared_distance_medium ("one Ireland") of a community's quarter. Read
# from the realm, not the capital: a large realm's Jewish frontier can be
# far from its capital (V44, 2026-10-15; was the capital within
# squared_distance_large, "one France").
kehillah_open_door_near_community_trigger = {
	save_temporary_scope_as = kehillah_open_door_ruler
	any_in_global_list = {
		variable = kehillah_registered_communities
		exists = holder.domicile.domicile_location
		holder.domicile.domicile_location = { save_temporary_scope_as = kehillah_open_door_quarter }
		scope:kehillah_open_door_ruler = {
			any_realm_county = {
				title_province = {
					squared_distance = {
						target = scope:kehillah_open_door_quarter
						value <= squared_distance_medium
					}
				}
			}
		}
	}
}

# Scope: a county; scope:kehillah_mother_leader saved. A settlers'
# destination in a realm whose lord has an open invitation.
kehillah_settlers_invited_destination_trigger = {
	kehillah_settlers_destination_trigger = yes
	holder.top_liege.primary_title = { has_variable = kehillah_invitation_open }
}

# Scope: a ruler; scope:kehillah_suitor_leader (a Kehillah leader) saved.
# A lord the community's Shtadlan may court for an invitation (Seek an
# Invitation, task_kehillah_seek_invitation): an AI independent duke or
# greater, Christian or Muslim, not the community's own host, whose
# policy is Discouraged or Allowed and off review cooldown, and whose
# capital is within reach of the community.
kehillah_invitation_suitor_target_trigger = {
	is_ai = yes
	is_landed = yes
	is_independent_ruler = yes
	highest_held_title_tier >= tier_duchy
	exists = primary_title
	exists = capital_province
	NOT = { government_has_flag = government_is_kehillah }
	OR = {
		faith.religion = religion:christianity_religion
		faith.religion = religion:islam_religion
	}
	primary_title = {
		NOT = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
		NOT = { kehillah_jewish_settlement_policy_banned_trigger = yes }
		NOT = { kehillah_policy_review_on_cooldown_trigger = yes }
	}
	NOT = { scope:kehillah_suitor_leader.domicile.domicile_location.county.holder.top_liege ?= this }
	NOT = { kehillah_ruler_has_active_community_crisis_trigger = yes }
	capital_province = {
		squared_distance = {
			target = scope:kehillah_suitor_leader.domicile.domicile_location
			value <= squared_distance_large
		}
	}
}

# =====================================================================
# INVITATION CALLS -- 2026-11-25 (docs/spec/2026-11-25-invitation-
# letters.md). A lord's invitation names one county; the communities
# nearest it are called, and the first to found there closes the call.
# =====================================================================

# Scope: a community title (kehillah_registered_communities);
# scope:kehillah_call_province saved. A community the call can reach:
# led, housed, within squared_distance_large of the county, and not still
# resting from its last settlers.
kehillah_invitation_callable_community_trigger = {
	exists = holder
	holder = { government_has_flag = government_is_kehillah }
	exists = holder.domicile.domicile_location
	NOT = { has_variable = kehillah_settlers_cooldown }
	holder.domicile.domicile_location = {
		squared_distance = {
			target = scope:kehillah_call_province
			value <= squared_distance_large
		}
	}
}

# Scope: anything; scope:kehillah_policy_realm and scope:kehillah_call_county
# saved. The lord's call still stands: he has not withdrawn or let it lapse,
# and no community has been founded there yet.
kehillah_invitation_call_open_trigger = {
	exists = scope:kehillah_call_county
	scope:kehillah_policy_realm = { var:kehillah_invitation_county ?= scope:kehillah_call_county }
	scope:kehillah_call_county = { kehillah_founding_site_base_trigger = yes }
}

# Scope: a Kehillah leader; the call's scopes saved. Can this community
# answer the call now? (The leader of the party is checked per option.)
kehillah_invitation_can_answer_trigger = {
	custom_tooltip = {
		text = kehillah_invitation_call_open_tt
		kehillah_invitation_call_open_trigger = yes
	}
	custom_tooltip = {
		text = kehillah_settlers_prosperity_tt
		kehillah_pillar_at_least_trigger = { PILLAR = kehillah_var_prosperity THRESHOLD = kehillah_band_strained_threshold }
	}
	custom_tooltip = {
		text = kehillah_settlers_cooldown_tt
		primary_title = { NOT = { has_variable = kehillah_settlers_cooldown } }
	}
	gold >= kehillah_invitation_send_cost_value
	is_available_adult = yes
}

# Scope: a character; scope:kehillah_call_province saved. A Jewish
# adventurer the call can reach (2026-11-25): one who could found a
# community (the Found a Jewish Community decision's own leader rules),
# within squared_distance_large, not already bound for another invitation.
kehillah_invitation_callable_band_trigger = {
	is_adult = yes
	is_imprisoned = no
	kehillah_can_found_kehillah_faith_trigger = yes
	kehillah_leader_is_rabbinic_trigger = yes
	NOT = { has_character_flag = kehillah_answered_invitation_flag }
	NOT = { has_variable = kehillah_invited_to_county }
	exists = location
	location = {
		squared_distance = {
			target = scope:kehillah_call_province
			value <= squared_distance_large
		}
	}
}

# Scope: a character; scope:kehillah_call_province saved. A Jewish
# merchant company the call can reach (2026-11-25).
kehillah_invitation_callable_merchant_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_jewish_character_trigger = yes
	exists = location
	location = {
		squared_distance = {
			target = scope:kehillah_call_province
			value <= squared_distance_large
		}
	}
}

# Scope: a Jewish adventurer; the call's scopes saved. Can his band set out
# for the county now?
kehillah_invitation_band_can_answer_trigger = {
	custom_tooltip = {
		text = kehillah_invitation_call_open_tt
		kehillah_invitation_call_open_trigger = yes
	}
	is_available_adult = yes
	is_travelling = no
	is_at_war = no
}

# Scope: a Jewish merchant; the call's scopes saved. Can he send a factor
# and families now?
kehillah_invitation_merchant_can_answer_trigger = {
	custom_tooltip = {
		text = kehillah_invitation_call_open_tt
		kehillah_invitation_call_open_trigger = yes
	}
	gold >= kehillah_invitation_send_cost_value
	is_available_adult = yes
}

# Scope: a Jewish adventurer arrived where a lord invited him
# (kehillah_invitation.0011). Can he found the community here now? The
# Found a Jewish Community decision's rules, at the invited county.
kehillah_invitation_band_can_found_trigger = {
	custom_tooltip = {
		text = kehillah_invitation_band_here_tt
		exists = var:kehillah_invited_to_county
		location.county ?= var:kehillah_invited_to_county
		var:kehillah_invited_to_county = { kehillah_founding_site_base_trigger = yes }
	}
	kehillah_leader_is_rabbinic_trigger = yes
	custom_tooltip = {
		text = kehillah_found_community_requires_ten_jewish_followers_tt
		OR = {
			has_variable = kehillah_exiles_from
			any_courtier = {
				count >= 10
				is_jewish_character_trigger = yes
			}
		}
	}
	gold >= kehillah_invitation_band_found_cost_value
	is_available_adult = yes
}
