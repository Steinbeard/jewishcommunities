# The Bet Din overdue (2026-11-08). See common/script_values/
# kehillah_bet_din_overdue_values.txt.

# Is the community's own court long overdue, so that Stability falls?
# Scope: leader. Not while a crisis forbids hosting, not while the region
# cannot seat a panel at all, and not while the leader's own sitting is
# already under way.
kehillah_bet_din_overdue_trigger = {
	government_has_flag = government_is_kehillah
	kehillah_has_known_minhag_region_trigger = yes
	primary_title ?= { has_variable = kehillah_bet_din_last_year }
	kehillah_bet_din_years_since_value >= kehillah_bet_din_overdue_grace_years
	kehillah_no_crisis_blocks_hosting_trigger = yes
	kehillah_region_can_field_bet_din_panel_trigger = yes
	NOT = {
		involved_activity ?= {
			has_activity_type = activity_kehillah_bet_din_conference
			activity_host = root
		}
	}
}

# No community crisis forbids hosting an activity. Scope: leader. A
# Kehillah leader hosts nothing while the community is in danger
# (2026-11-08, at user request); a sitting already under way when the
# crisis begins is not cancelled.
kehillah_no_crisis_blocks_hosting_trigger = {
	OR = {
		NOT = { government_has_flag = government_is_kehillah }
		NOT = { exists = primary_title }
		primary_title = { NOT = { kehillah_trouble_crisis_active_trigger = yes } }
	}
}
