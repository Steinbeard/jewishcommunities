# Scope: a character. Their rite holds Covenantal Peoplehood.
kehillah_keeps_hebrew_peoplehood_trigger = {
	rite ?= { rite_has_doctrine = kehillah_peoplehood_covenantal }
}

# Scope: a character. Of a Hebrew-heritage culture.
kehillah_is_hebrew_culture_trigger = {
	culture = { has_cultural_pillar = heritage_israelite }
}

# Scope: a character. Converting to their capital's culture would take
# them out of a Hebrew culture into another.
kehillah_leaving_hebrew_culture_for_local_trigger = {
	kehillah_keeps_hebrew_peoplehood_trigger = yes
	kehillah_is_hebrew_culture_trigger = yes
	capital_province ?= {
		culture = { NOT = { has_cultural_pillar = heritage_israelite } }
	}
}

# Scope: a character. Converting to their capital's culture would bring a
# non-Hebrew Jew into a Hebrew culture.
kehillah_joining_hebrew_culture_locally_trigger = {
	kehillah_keeps_hebrew_peoplehood_trigger = yes
	NOT = { kehillah_is_hebrew_culture_trigger = yes }
	capital_province ?= {
		culture = { has_cultural_pillar = heritage_israelite }
	}
}

# Scope: a culture. Its language has a historical Jewish counterpart, the
# same table kehillah_pick_historical_jewish_culture_effect uses

# Scope: a character. There is a Jewish people for them to join: one
# matching their own culture's language, or one native where they live or
# anywhere in their realm. Without one, they found a new people instead.
kehillah_has_hebrew_people_to_join_trigger = {
	OR = {
		culture = { kehillah_culture_has_hebrew_counterpart_trigger = yes }
		exists = capital_county.var:kehillah_local_jewish_culture
		exists = location.county.var:kehillah_local_jewish_culture
		any_realm_county = { exists = var:kehillah_local_jewish_culture }
	}
}
