# =====================================================================
# Triggers for the Sefer tenets' mechanics outside the book system
# (2026-11-08; docs/spec/2026-11-08-sefer-tenet-mechanics.md), and the
# Karaite path into the community's clergy.
# =====================================================================

# Scope: a character. Avelei Tziyon: his rite mourns Zion, or he does.
kehillah_tenet_mourns_zion_trigger = {
	OR = {
		rite ?= { rite_has_parameter = kehillah_tenet_zion_alms }
		has_personal_tenet_flag = kehillah_tenet_zion_alms_personal
	}
}

# Scope: a character. A Karaite hakham: a Karaite with vanilla's Religious
# Scholar trait. Karaites have no semicha and never take the Rabbi trait
# (is_rabbinic_authority_jewish_trigger keeps it from them), so this is
# their way into the community's clergy: the Dayyan's seat, the community's
# scholar, the head of a Karaite academy, and writing books.
kehillah_is_karaite_hakham_trigger = {
	has_trait = lifestyle_scholar
	kehillah_is_karaite_character_trigger = yes
}

# Scope: a character. A Karaite hakham deep enough in his learning to head
# an academy or write: Religious Scholar experience at least VALUE.
kehillah_karaite_hakham_xp_trigger = {
	kehillah_is_karaite_hakham_trigger = yes
	has_trait_xp = { trait = lifestyle_scholar value >= $VALUE$ }
}
