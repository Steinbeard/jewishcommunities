# =====================================================================
# Triggers for the Sefer tenets' mechanics outside the book system
# (2026-11-08; docs/spec/2026-11-08-sefer-tenet-mechanics.md), and the
# Karaites in the rabbi's place.
# =====================================================================

# Scope: a character. Avelei Tziyon: his rite mourns Zion, or he does.
kehillah_tenet_mourns_zion_trigger = {
	OR = {
		rite ?= { rite_has_parameter = kehillah_tenet_zion_alms }
		has_personal_tenet_flag = kehillah_tenet_zion_alms_personal
	}
}

# KARAITES HAVE NO CLERGY (Daniel, 2026-11-08). Karaism has no semicha and
# no clerical class: every believer searches Scripture himself. Karaites
# never take the Rabbi trait (is_rabbinic_authority_jewish_trigger keeps it
# from them) and nothing makes them the game's clergy. Instead, everything
# the mod reserves for rabbis is open to every Karaite: the two triggers
# below stand in for the Rabbi trait in those gates.
#
# Depth is still depth. Where a rabbi needs experience in a Rabbi track,
# a Karaite needs as much Religious Scholar experience
# (kehillah_karaite_scholar_xp_trigger): a mark of learning, not a status.
#
# Rabbanite institutions stay Rabbanite: semicha, yeshiva enrolment, the
# Sanhedrin, the Gaonate, the named works, and anything that reads mastery
# of the Talmud.

# Scope: a character. Holds the Rabbi trait (ordained or a student), or is
# a Karaite.
kehillah_rabbi_or_karaite_trigger = {
	OR = {
		has_trait = kehillah_rabbi_trait
		kehillah_is_karaite_character_trigger = yes
	}
}

# Scope: a character. An ordained rabbi, or a Karaite: who may hold the
# Dayyan's seat, head an academy, or act as the community's rabbi.
kehillah_ordained_or_karaite_trigger = {
	OR = {
		kehillah_is_ordained_rabbi_trigger = yes
		kehillah_is_karaite_character_trigger = yes
	}
}

# Scope: a character. A Karaite with Religious Scholar experience at least
# VALUE, where a rabbi would need that much in a Rabbi track.
kehillah_karaite_scholar_xp_trigger = {
	kehillah_is_karaite_character_trigger = yes
	has_trait_xp = { trait = lifestyle_scholar value >= $VALUE$ }
}
