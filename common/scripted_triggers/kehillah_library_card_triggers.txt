# Scope: a library title or a community title. A yeshiva studies here: a
# landed yeshiva on a Beit Midrash (kehillah_sefer_yeshiva_effects.txt),
# or a Kehillah's own (V29, its domicile's Beit Midrash III).
kehillah_card_has_yeshiva_trigger = {
	OR = {
		has_variable = kehillah_landed_yeshiva
		AND = {
			NOT = { tier = tier_barony }
			holder ?= {
				government_has_flag = government_is_kehillah
				kehillah_has_yeshiva_trigger = yes
			}
		}
	}
}

# Scope: a landed title. A Beit Midrash: a library on a temple barony,
# not a community's.
kehillah_card_is_beit_midrash_trigger = {
	tier = tier_barony
	kehillah_lib_is_library_title_trigger = yes
}
