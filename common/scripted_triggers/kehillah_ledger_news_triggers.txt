# Ledger news (2026-11-24): see common/scripted_effects/kehillah_ledger_news_effects.txt.

# Scope: a reader holding the news. A reason to ban the book, and the right to.
kehillah_ledger_news_would_condemn_trigger = {
	kehillah_ledger_news_can_cherem_trigger = yes
	NOT = { is_target_in_variable_list = { name = kehillah_reviewed_works target = var:kehillah_ledger_news_work } }
	OR = {
		has_relation_rival = var:kehillah_ledger_news_author
		opinion = { target = var:kehillah_ledger_news_author value <= -50 }
		kehillah_ledger_news_stance_value <= kehillah_review_unread_threshold
	}
}

# Scope: a reader holding the news, with a library of his own that lacks the
# book and no other commission underway.
kehillah_ledger_news_can_commission_trigger = {
	kehillah_lib_can_commission_trigger = yes
	var:kehillah_own_library ?= {
		NOT = { is_target_in_variable_list = { name = kehillah_library_works target = root.var:kehillah_ledger_news_work } }
		NOT = { has_variable_list = kehillah_library_commission_work }
	}
}


# Scope: a reader holding the news. He may plan a Learn Torah journey to the
# author's own library, which holds the book, and has not mastered it.
kehillah_ledger_news_can_journey_trigger = {
	kehillah_journey_can_plan_trigger = yes
	NOT = { is_target_in_variable_list = { name = kehillah_mastered_works target = var:kehillah_ledger_news_work } }
	var:kehillah_ledger_news_author.var:kehillah_own_library ?= {
		exists = holder
		NOT = { this = root.primary_title }
		is_target_in_variable_list = { name = kehillah_library_works target = root.var:kehillah_ledger_news_work }
	}
}
