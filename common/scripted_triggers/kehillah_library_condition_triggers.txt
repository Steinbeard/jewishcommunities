# Sefer condition: wear, repair and the genizah (2026-11-24, Daniel's request).
# See docs/implementation/2026-11-24-sefer-condition-and-genizah.md.

# Character scope (the payer), with scope:library. Can he pay to mend $WORK$?
kehillah_lib_can_afford_repair_trigger = {
	trigger_if = {
		limit = { scope:library = { is_target_in_variable_list = { name = kehillah_library_fragile target = $WORK$ } } }
		gold >= kehillah_lib_repair_fragile_cost_value
	}
	trigger_else = {
		gold >= kehillah_lib_repair_worn_cost_value
	}
}

# Library title scope. An AI holder with gold to spare mends a fragile book.
kehillah_lib_ai_will_repair_trigger = {
	holder ?= {
		is_ai = yes
		gold >= kehillah_lib_ai_repair_reserve_value
	}
}
