# always in the capital city ... maybe we should just have a capital
# kehilla always have some kind of leadership role (and maybe more

# independent ruler of duke tier or higher is the realm's seat, under any
# law. It is not an office: the Nagid office (a later slice) sits on top.
# =====================================================================

# Scope: a Kehillah leader. Their community is the seat of its realm's
# Jews: its county is the capital county of the county holder's top
# liege, a landed ruler of duke tier or higher who is not a Kehillah.
kehillah_is_realm_seat_trigger = {
	exists = domicile.domicile_location.county
	domicile.domicile_location.county.holder.top_liege ?= {
		is_landed = yes
		highest_held_title_tier >= tier_duchy
		NOT = { government_has_flag = government_is_kehillah }
	}
	domicile.domicile_location.county.holder.top_liege.capital_county ?= domicile.domicile_location.county
}

# Scope: a Kehillah leader. Their community lives in a county the ruler
# of the realm holds himself, other than his capital: the lord's own
# Jews, with one master and his protection closest.
kehillah_in_lords_own_lands_trigger = {
	exists = domicile.domicile_location.county
	domicile.domicile_location.county.holder ?= {
		is_independent_ruler = yes
		highest_held_title_tier >= tier_duchy
		NOT = { government_has_flag = government_is_kehillah }
	}
	NOT = { kehillah_is_realm_seat_trigger = yes }
}
