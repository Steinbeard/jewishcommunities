# =====================================================================
# Ordination (semicha) eligibility -- V21 slice 1, added 2026-10-09
# (Shabbat heartbeat 1, ROADMAP item H4).
#
# Source of truth: docs/spec/v21-rabbi-ordination-paths.md. Its section 6
# makes slice 1 exactly this file's job: "Add a named reusable
# rabbi-ordination candidate trigger that composes the rabbinic-authority,
# adult/free/capable, Learning, and current gender checks. Do not paste
# the conditions into Learn Torah, the scheme, office, and event
# separately."
#
# So: every later slice (the widened Learn Torah gate, Seek Semicha, the
# education apprenticeship, the Chief Rabbi/Bet Din reconciliation) calls
# these and nothing re-derives them. Two deliberate reuses rather than
# new rules:
#
#   - is_rabbinic_authority_jewish_trigger (common/scripted_triggers/
#     jewish_holiday_scripted_triggers.txt) is the faith gate, NOT the
#     broader is_jewish_character_trigger. Section 2: Karaism,
#     Samaritanism, Haymanot and Malabarism must not receive a trait
#     called Rabbi by accident, and adding a future rabbinic faith means
#     extending that one shared trigger, not editing a faith list here.
#
#   - kehillah_leadership_gender_eligible_trigger (common/scripted_
#     triggers/kehillah_scripted_triggers.txt) is the gender gate. It
#     already governs the Rabbi trait, the Chief Rabbi office and
#     semicha, and it already carries the egalitarian-reform hook that
#     opens the rule later. Section 2 is explicit that V21 reuses it
#     rather than inventing an incompatible fourth gender rule.
#
# NOTHING CALLS THESE YET, on purpose. Slice 1 is inert by design: this
# commit changes no behaviour at all, so the gate can be reviewed on its
# own before the slices that widen the Learn Torah decision and add Seek
# Semicha start depending on it.
#
# CORRECTION 2026-10-10: the paragraph above was true only of slice 1's
# own commit and is now stale -- these triggers are live and load-bearing.
# kehillah_rabbi_study_eligible_trigger gates the Learn Torah decision and
# the study scheme (slice 2), kehillah_rabbi_ordination_ready_trigger gates
# Seek Semicha (slice 3, verified live 2026-10-10),
# kehillah_rabbinic_guardian_trigger / _has_rabbinic_guardian_trigger /
# _apprenticeship_complete_trigger / kehillah_mentor_endorses_trigger drive
# the apprenticeship and its coming-of-age sorter (slice 4), and
# is_rabbinic_authority_jewish_trigger reuse reaches the Bet Din offer and
# the Chief Rabbi office (slice 5). Kept rather than deleted because the
# paragraph explains WHY the file was committed inert, which is still the
# reason the gates are composed here instead of pasted into each caller.
# =====================================================================

# A character who may ENTER the ordination path: study Torah as an
# apprentice, accumulate the curriculum, and eventually seek semicha.
# Says nothing about whether they have met the curriculum yet.
kehillah_rabbi_ordination_candidate_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	is_rabbinic_authority_jewish_trigger = yes
	kehillah_leadership_gender_eligible_trigger = yes
	learning >= kehillah_ordination_learning_floor_value
}

# The same, at the lower Learning band of section 3.3 -- enough for a
# rabbinic education to count as a FOUNDATION worth one study field, but
# not enough to ordain on its own.
kehillah_rabbi_ordination_foundation_candidate_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	is_rabbinic_authority_jewish_trigger = yes
	kehillah_leadership_gender_eligible_trigger = yes
	learning >= kehillah_ordination_foundation_learning_floor_value
}

# Who may study at a library: an ordained rabbi, OR a candidate.
#
# The two halves are deliberately asymmetric. An existing rabbi is
# admitted on the trait alone, with no faith or gender re-check, because
# the trait is already held -- historical trait holders, Chief Rabbi
# appointees and Bet Din grantees must not lose access to study because
# a gate written later would not have let them in. Section 2: V21 "does
# not retcon an existing character's trait".
kehillah_rabbi_study_eligible_trigger = {
	OR = {
		has_trait = kehillah_rabbi_trait
		kehillah_rabbi_ordination_candidate_trigger = yes
	}
}


# =====================================================================
# READINESS -- V21 slice 2/3, added 2026-10-09.
#
# Section 3.2's published curriculum: "2 distinct completed works, from
# 2 distinct fields". Both halves are read off records that already
# exist or are written by kehillah_study_specific_work_effect:
#
#   kehillah_mastered_works    -- distinct works, successfully studied.
#                                 Note this is NOT kehillah_studied_works,
#                                 which is the older ENCOUNTERED list and
#                                 includes failed first readings.
#   kehillah_apprentice_fields -- distinct fields of those works.
#
# Thresholds come from common/script_values/kehillah_ordination_values.txt,
# per the spec's own instruction to keep them tunable.
# =====================================================================

# WIDENED 2026-10-10 (V21 slice 4) with the rabbinic-apprenticeship
# branch. Section 3.3 requires a childhood spent under a rabbi to satisfy
# this curriculum WITHOUT faking library history: "The record is
# intentionally a curriculum equivalent -- not two fake named books -- so
# it satisfies the two-field requirement for Seek Semicha without
# corrupting the library's work history." Hence a separate variable the
# OR accepts, rather than two invented entries in kehillah_mastered_works.
#
# The Learning 6-7 FOUNDATION band is not here, on purpose. It is worth
# one study FIELD (kehillah_award_apprenticeship_credit_effect adds
# flag:rabbinic_apprenticeship to kehillah_apprentice_fields), so it
# flows through the ordinary counting branch below and still demands real
# adult study -- which is exactly section 3.3's "requires the ordinary
# adult Learn Torah route until Learning reaches 8 and supplies one
# separate field of adult study".
kehillah_rabbi_ordination_curriculum_met_trigger = {
	# LIVE-TEST FIX, 2026-10-10 (the slice-3 pass). Each variable_list_size
	# is now paired with its own has_variable_list. The engine logs
	# "variable_list_size trigger [ Could not find list from read name flag
	# kehillah_mastered_works ]" for a size read on a list that was never
	# created -- and that is the normal state of everyone who has not
	# studied yet, which is most of the population this decision is shown
	# to. The verdict was always correct ("no list" and "list too short"
	# both mean not ready); it was the log that was wrong, exactly the same
	# shape of bug as ROADMAP H3's restore helpers.
	OR = {
		AND = {
			has_variable_list = kehillah_mastered_works
			variable_list_size = {
				name = kehillah_mastered_works
				value >= kehillah_ordination_works_required_value
			}
			has_variable_list = kehillah_apprentice_fields
			variable_list_size = {
				name = kehillah_apprentice_fields
				value >= kehillah_ordination_fields_required_value
			}
		}
		has_variable = kehillah_apprenticeship_curriculum
	}
}

# The full readiness test for Seek Semicha, minus the institution check
# (which needs a library/Beit Midrash resolution and so belongs with the
# decision, not here). NOT has_trait is part of readiness on purpose:
# section 3.4 says the apprentice records stay as biography after
# ordination and "no longer award readiness again".
kehillah_rabbi_ordination_ready_trigger = {
	NOT = { has_trait = kehillah_rabbi_trait }
	kehillah_rabbi_ordination_candidate_trigger = yes
	kehillah_rabbi_ordination_curriculum_met_trigger = yes
}


# =====================================================================
# THE INSTITUTION -- V21 slice 3, added 2026-10-09.
#
# Section 3.4: semicha requires being at, or associated through the same
# library-resolution rules with, a community holding at least
# kehillah_beit_midrash_01. "A book cache alone is enough to read; a
# functioning study hall is the minimum institution that can recognise an
# ordination." Tier 3 (the Yeshiva) improves the ceremony, and is not
# required.
#
# This is deliberately the SAME three populations, in the same order, as
# kehillah_study_torah_has_accessible_library_trigger (common/scripted_
# triggers/kehillah_scripted_triggers.txt) -- a Kehillah leader's own
# quarter, a landed ruler's co-located community at their capital, and a
# landless adventurer's current location. Only the test on the community
# changes: "owns any Torah work" becomes "has a Beit Midrash". If that
# trigger's population rules ever change, change these together; the
# whole point of section 3.4 is that a student does not have to be at a
# DIFFERENT place to ordain than the one they studied at.
#
# The exists = domicile guards are not decoration. The 2026-09-23 finding
# recorded in the library trigger's own comment (~990 errors a load) was
# exactly this: a landless independent ruler -- which every other Kehillah
# leader in the game is -- has no capital_province, and a chained scope
# reference through an unset link is what produced the flood.
# =====================================================================

kehillah_semicha_institution_available_trigger = {
	OR = {
		AND = {
			$CHARACTER$ = { government_has_flag = government_is_kehillah }
			exists = $CHARACTER$.domicile
			$CHARACTER$.domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_01 }
		}
		AND = {
			NOT = { $CHARACTER$ = { government_has_flag = government_is_kehillah } }
			$CHARACTER$ = { is_ruler = yes }
			exists = $CHARACTER$.capital_province
			any_in_global_list = {
				variable = kehillah_registered_communities
				exists = holder
				holder = {
					exists = domicile
					domicile.domicile_location = $CHARACTER$.capital_province
					domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_01 }
				}
			}
		}
		# A landless adventurer, OR anyone who holds nothing at all: the
		# community standing where they are standing.
		#
		# The "holds nothing at all" half was added 2026-10-10, after the
		# slice 4 source review. Section 2 of V21 says "a Kehillah leader,
		# landed ruler, landless adventurer, and eligible courtier use the
		# same candidate definition" -- but the eligible courtier had no
		# branch here at all. A courtier is not a Kehillah government
		# holder (branch 1), is not a ruler and has no capital_province
		# (branch 2), and does not have the adventurer government, so this
		# trigger was FALSE for every courtier in the game. Which is exactly
		# the population slice 4's apprenticeship produces: the sorter
		# kehillah_semicha.0002 requires this trigger, so a child who had
		# served four years under a rabbi could never be endorsed. The
		# clause is purely additive -- nothing that passed before stops
		# passing.
		#
		# LIVE-TEST FIX, 2026-10-10 (Shabbat heartbeat 4). The widening
		# above was still FALSE for the exact pupil it was written for, and
		# the reason is the clause that used to head this branch:
		#
		#     NOT = { $CHARACTER$ = { government_has_flag = government_is_kehillah } }
		#
		# **government_has_flag is TRUE for a COURTIER of a Kehillah ruler**,
		# not only for the ruler. A landless character's government resolves
		# through their liege, so every member of Isaac HaLevi's court reads
		# as a Kehillah government holder. Measured directly: the H4 pupil
		# Huna Landsberg (65937), a child ward in the Worms court, logged
		# "pupil IS a Kehillah government holder" from
		# run/h4_institution_diag.txt. That single fact closed all three
		# branches against him at once -- branch 1 matched the flag but then
		# demanded `exists = domicile`, which a courtier has not got
		# (domiciles belong to the ruler), while this NOT shut branches 2
		# and 3. The community data was never the problem: the same
		# diagnostic shows Worms MATCHES his location and HAS a Beit Midrash
		# tier 1+.
		#
		# So the guard is dropped rather than the location test changed. The
		# branch is still not reachable by a Kehillah LEADER, which is what
		# that NOT was actually protecting -- a leader holds d_kehillah_*,
		# so `is_ruler = no` is false for them, and they do not have the
		# adventurer government. A leader is therefore still judged only on
		# their own quarter (branch 1) and cannot borrow a neighbouring
		# community's study hall. Dropping it is purely additive, same as
		# the 2026-10-10 widening itself.
		#
		# Do NOT re-add a government_has_flag exclusion to a branch that is
		# meant to catch courtiers; use is_ruler / exists = domicile, which
		# distinguish a holder from their household and the flag does not.
		AND = {
			OR = {
				$CHARACTER$ = { has_government = landless_adventurer_government }
				$CHARACTER$ = { is_ruler = no }
			}
			any_in_global_list = {
				variable = kehillah_registered_communities
				exists = holder
				holder = {
					exists = domicile
					domicile.domicile_location = $CHARACTER$.location
					domicile = { has_domicile_building_or_higher = kehillah_beit_midrash_01 }
				}
			}
		}
	}
}


# =====================================================================
# RABBINIC APPRENTICESHIP -- V21 slice 4, added 2026-10-10.
#
# Section 3.3: "A Jewish student who spends a substantial part of their
# education under a rabbi should be able to emerge as one, rather than
# being forced to repeat their childhood study as an adult scheme."
#
# The guardian is found the way vanilla finds one -- any_relation with
# type = guardian (see education_point_acquisition_support_effect in
# common/scripted_effects/00_education_effects.txt, which resolves the
# educator as random_relation type = guardian first and only then falls
# back to a court tutor). This deliberately does NOT accept the court
# tutor fallback: section 3.3 requires a named qualifying guardian, and
# a community that happens to employ a tutor is not an apprenticeship.
# =====================================================================

# Scope: a prospective GUARDIAN. The same three conditions the Rabbi
# trait itself is gated on, so a guardian who could not be ordained
# today cannot ordain someone else.
kehillah_rabbinic_guardian_trigger = {
	has_trait = kehillah_rabbi_trait
	is_rabbinic_authority_jewish_trigger = yes
	kehillah_leadership_gender_eligible_trigger = yes
}

# Scope: the PUPIL. Section 3.3: "The pupil must share that
# rabbinic-authority tradition" -- so the faith check is on both sides,
# not just the teacher's.
kehillah_has_rabbinic_guardian_trigger = {
	is_rabbinic_authority_jewish_trigger = yes
	any_relation = {
		type = guardian
		kehillah_rabbinic_guardian_trigger = yes
	}
}

# Scope: the PUPIL. Accumulated years, not "the guardian on the final
# day" -- section 3.3 is explicit that "replacing a guardian near
# adulthood cannot manufacture an ordination", which is why this reads a
# counter incremented once per childhood birthday rather than testing
# the current guardian at 16.
#
# has_variable first: var:X on a never-set variable is the "Failed to
# fetch variable" log noise this repo spent ROADMAP H3 cleaning out.
kehillah_rabbinic_apprenticeship_complete_trigger = {
	has_variable = kehillah_apprentice_rabbi_years
	var:kehillah_apprentice_rabbi_years >= kehillah_ordination_apprenticeship_years_value
}

# Scope: the MENTOR. $PUPIL$ = the student.
#
# Section 3.3 withholds endorsement for exactly two reasons, and is
# emphatic that neither is a permanent bar: an opinion of the pupil at or
# below -50, and cynical+arbitrary TOGETHER. Either trait alone is only a
# modest negative compatibility signal on the Rabbi trait and is
# deliberately not a veto here.
kehillah_mentor_endorses_trigger = {
	NOT = {
		opinion = {
			target = $PUPIL$
			value <= -50
		}
	}
	NOT = {
		AND = {
			has_trait = cynical
			has_trait = arbitrary
		}
	}
}
