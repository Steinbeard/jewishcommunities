# =====================================================================
# FAVORED BEYOND THE TEACHING -- the Nagid spec, sections 3 and 5
# (docs/spec/2026-10-30-the-nagid.md), 2026-10-30. Daniel: no faith
# technically allows its Jews this much favour, so the top rung of the
# Status of the Jews always costs something; a tolerant culture eases it.
# =====================================================================

# Scope: a character. A culture whose tradition already makes peace with
# other faiths: it halves the cost of favoring the Jews (Daniel: African
# Tolerance "can reduce the negatives of a more lenient than allowed by
# faith Jewish policy too"). Xenophilic concerns foreign cultures, not
# faiths, so it raises the outlook (kehillah_outlook_culture_value) but
# does not count here.
kehillah_favored_tolerant_culture_trigger = {
	exists = culture
	culture = {
		OR = {
			has_cultural_tradition = tradition_african_tolerance
			has_cultural_tradition = tradition_steppe_tolerance
			has_cultural_tradition = tradition_religion_blending
		}
	}
}

# Scope: a ruler. Holds the law at its top rung, shown as "Jews Favored".
kehillah_ruler_favors_jews_trigger = {
	kehillah_can_hold_settlement_law_trigger = yes
	primary_title ?= { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
}
