# =====================================================================
# Named works' ambitions (2026-11-08, spec section 6.1 and Daniel's
# decision 2). Each is the trigger for a gold-highlighted option in
# kehillah_book.0001. Scope: the would-be author. A work can be written
# once per campaign. Its historical author, if he lives in this game, needs
# less; anyone else must have mastered what the work stands on.
# =====================================================================

kehillah_sefer_ambition_common_trigger = {
	has_trait = kehillah_rabbi_trait
	rite ?= { kehillah_rite_rabbanite_trigger = yes }
}

kehillah_sefer_ambition_rashi_torah_trigger = {
	kehillah_sefer_ambition_common_trigger = yes
	NOT = { has_global_variable = kehillah_wk_work_rashi_torah_written }
	is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_tanakh_torah }
	OR = {
		AND = {
			character:9000201 ?= this
			has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 30 }
		}
		AND = {
			learning >= 14
			has_trait_xp = { trait = kehillah_rabbi_trait track = parshanut value >= 60 }
			is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_targum_onkelos }
			OR = {
				is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_genesis_rabbah }
				is_target_in_variable_list = { name = kehillah_mastered_works target = flag:work_mekhilta }
			}
		}
	}
}
