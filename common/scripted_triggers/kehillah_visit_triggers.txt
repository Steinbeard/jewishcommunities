# Scope: a community title. The community's quarter stands in the county
# where $VISITOR$ is now, and its leader is Jewish. $VISITOR$ is passed
# explicitly, as the other candidacy triggers do.
kehillah_visit_title_here_trigger = {
	exists = holder
	exists = holder.domicile.domicile_location.county
	holder.domicile.domicile_location.county = $VISITOR$.location.county
	holder = { is_jewish_character_trigger = yes }
}

# Scope: an adventurer. This is kept separate from the title trigger because
# the vanilla settlement decision is evaluated from the character scope.
kehillah_adventurer_community_visit_available_trigger = {
	is_jewish_character_trigger = yes
	NOT = { has_variable = kehillah_visit_community_cooldown }
	any_in_global_list = {
		variable = kehillah_registered_communities
		kehillah_visit_title_here_trigger = { VISITOR = root }
	}
}
