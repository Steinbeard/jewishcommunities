# =====================================================================
# The Community Crises map (2026-11-08,
# docs/implementation/2026-11-08-jewish-map-category.md).
#
# Scope: a d_kehillah_* community title. One trigger per state the map
# paints and its hover card names. They read the same title variables as
# kehillah_trouble_crisis_active_trigger
# (kehillah_trouble_status_triggers.txt) and split them by kind. The paint
# (kehillah_pick_community_crises_map_color_effect) and the card
# (KehillahCrisisMapStatus) test them in the same order, worst first.
# =====================================================================

# An armed host at the quarter, or a mob raised against it.
kehillah_map_crisis_armed_trigger = {
	OR = {
		has_variable = kehillah_mob_threat
		AND = {
			var:kehillah_crisis_stage ?= 2
			NOT = { has_variable = kehillah_crisis_situation_done }
		}
	}
}

# A crisis is starting, or the town is agitated and the race is on.
kehillah_map_crisis_agitation_trigger = {
	OR = {
		has_variable = kehillah_crisis_intro_pending
		AND = {
			var:kehillah_crisis_stage ?= 1
			NOT = { has_variable = kehillah_crisis_situation_done }
		}
	}
}

# A decree, a Talmud trial, or an expulsion is under way.
kehillah_map_crisis_decree_trigger = {
	OR = {
		exists = var:kehillah_decree_situation
		exists = var:kehillah_trial_situation
		has_variable = kehillah_expulsion_quarters
	}
}

# An attack has been decided; judgment or recovery follows.
kehillah_map_crisis_aftermath_trigger = {
	has_variable = kehillah_crisis_stage
	NOT = { has_variable = kehillah_crisis_situation_done }
	var:kehillah_crisis_stage >= 3
}
