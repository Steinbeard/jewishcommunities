# Scope: a character. A Jewish merchant or adventurer who may have a home
# community.
kehillah_home_traveller_trigger = {
	is_alive = yes
	is_adult = yes
	is_ruler = yes
	is_jewish_character_trigger = yes
	OR = {
		government_has_flag = government_is_merchant
		has_government = landless_adventurer_government
	}
}

# Scope: a community title. Led by a Kehillah leader of $MEMBER$'s faith.
kehillah_home_title_valid_for_trigger = {
	exists = holder
	holder = {
		government_has_flag = government_is_kehillah
		faith = $MEMBER$.faith
	}
}

# Scope: a community title. Its quarter is in $COUNTY$, and it may be
# $MEMBER$'s home.
kehillah_home_title_in_county_trigger = {
	kehillah_home_title_valid_for_trigger = { MEMBER = $MEMBER$ }
	exists = holder.domicile.domicile_location.county
	holder.domicile.domicile_location.county = $COUNTY$
}

# Scope: a character. Is $TITLE$ already this character's home?
kehillah_home_is_home_trigger = {
	exists = var:kehillah_home_community
	var:kehillah_home_community = $TITLE$
}

# Scope: a member, also passed as $MEMBER$ (a saved scope). Is the home
# still sound?
kehillah_home_still_valid_trigger = {
	kehillah_home_traveller_trigger = yes
	exists = var:kehillah_home_community
	var:kehillah_home_community = { kehillah_home_title_valid_for_trigger = { MEMBER = $MEMBER$ } }
}
