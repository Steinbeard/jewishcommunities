# The Talmud trial as a situation (docs/spec/2026-11-01-talmud-trial-situation.md), 2026-11-01.

# Scope: a Kehillah leader. A trial may begin: no attack and no other trial running.
kehillah_trial_can_begin_trigger = {
	exists = primary_title
	exists = domicile.domicile_location.county.holder.top_liege
	primary_title = {
		NOT = { has_variable = kehillah_mob_threat }
		NOT = { exists = var:kehillah_trial_situation }
	}
}

# Scope: a Kehillah leader. His community's trial is in this phase (denunciation, seizure, hearing).
kehillah_trial_phase_trigger = {
	primary_title.var:kehillah_trial_situation ?= { situation_current_phase = kehillah_trial_$PHASE$ }
}

# Scope: a Kehillah leader. The trial's actions are open: before the verdict.
kehillah_trial_actions_open_trigger = {
	primary_title.var:kehillah_trial_situation ?= {
		OR = {
			situation_current_phase = kehillah_trial_denunciation
			situation_current_phase = kehillah_trial_seizure
			situation_current_phase = kehillah_trial_hearing
		}
	}
}

# Scope: a Kehillah leader. A sub-action not yet taken in this trial.
kehillah_trial_not_done_trigger = {
	custom_description = {
		text = kehillah_trial_done_already_tt
		primary_title = { NOT = { has_variable = kehillah_trial_did_$ONCE$ } }
	}
}
