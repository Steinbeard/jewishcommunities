# Scope: a Kehillah title. Its library holds a tractate of the Talmud.
kehillah_has_talmud_trigger = {
	OR = {
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_zeraim }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_moed }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_nashim }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_nezikin }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_kodashim }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_bavli_tahorot }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_yerushalmi_zeraim }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_yerushalmi_moed }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_yerushalmi_nashim }
		is_target_in_variable_list = { name = kehillah_library_works target = flag:work_yerushalmi_nezikin }
	}
}

# Scope: a Kehillah leader; scope:kehillah_incident_host saved. A lord of
# the host's realm, or the host, owes the community.
kehillah_has_host_realm_debtor_trigger = {
	primary_title ?= {
		any_in_list = {
			variable = kehillah_loan_debtors
			exists = var:kehillah_loan_amount_owed
			OR = {
				this = scope:kehillah_incident_host
				top_liege ?= scope:kehillah_incident_host
			}
		}
	}
}
