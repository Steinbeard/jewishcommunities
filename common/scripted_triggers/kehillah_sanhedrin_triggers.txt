# The Sanhedrin (V46 seats and College; V47 phase B seat in the Land,
# docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).

# Scope: a Kehillah leader. His community lies in the Land of Israel: its
# county is de jure within the Kingdom of Jerusalem (Daniel, 2026-10-19).
kehillah_community_in_the_land_trigger = {
	primary_title ?= {
		is_kehillah_title_trigger = yes
		title_capital_county ?= {
			target_is_de_jure_liege_or_above = title:k_jerusalem
		}
	}
}

# Scope: a Kehillah leader. The liege can grant the community's county
# (Daniel's first case): his host holds it directly, and it is not the
# host's realm capital.
kehillah_liege_can_grant_county_trigger = {
	exists = suzerain
	primary_title.title_capital_county ?= {
		holder ?= root.suzerain
		NOT = { this = root.suzerain.capital_county }
	}
}

# Scope: a character. The liege refused a petition recently.
kehillah_sanhedrin_petition_refused_trigger = {
	has_variable = kehillah_sanhedrin_petition_refused
}

# Scope: a character. Has mastered all three Rabbi trait tracks.
kehillah_masters_all_rabbinic_tracks_trigger = {
	has_trait = kehillah_rabbi_trait
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = parshanut
		value >= 100
	}
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = talmudics
		value >= 100
	}
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = hashkafa
		value >= 100
	}
}

# Scope: a character. Holds a seat on the Sanhedrin.
kehillah_holds_sanhedrin_seat_trigger = {
	any_held_title = { has_variable = kehillah_sanhedrin_seat }
}

# Scope: a character. May be given a seat: a rabbi who leads a community,
# or a landed Jewish ruler who is a rabbi (either way a ruler, as electors
# must be), not the Nasi, not already seated.
kehillah_sanhedrin_seat_eligible_trigger = {
	is_alive = yes
	is_adult = yes
	is_incapable = no
	is_ruler = yes
	OR = {
		AND = {
			government_has_flag = government_is_kehillah
			primary_title ?= { is_kehillah_title_trigger = yes }
		}
		is_landed = yes
	}
	faith = faith:rabbinic_faith
	has_trait = kehillah_rabbi_trait
	NOT = { has_trait = excommunicated }
	NOT = { kehillah_is_sanhedrin_head_trigger = yes }
	NOT = { kehillah_holds_sanhedrin_seat_trigger = yes }
}

# Scope: a would-be founder. Other rabbis leading communities, enough to
# seat a court that can elect a successor.
kehillah_sanhedrin_has_founding_members_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		count >= kehillah_sanhedrin_founding_members
		holder ?= {
			NOT = { this = root }
			government_has_flag = government_is_kehillah
			faith = faith:rabbinic_faith
			has_trait = kehillah_rabbi_trait
		}
	}
}
