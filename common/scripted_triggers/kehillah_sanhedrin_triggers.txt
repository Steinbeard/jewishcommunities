# The Sanhedrin (V46 seats and College; V47 phase B seat in the Land,
# docs/spec/v47-gaon-and-the-seat-of-the-sanhedrin.md).

# Scope: a county title. It lies in the Land of Israel: de jure within the
# Kingdom of Jerusalem (Daniel, 2026-10-19).
kehillah_county_in_the_land_trigger = {
	tier = tier_county
	target_is_de_jure_liege_or_above = title:k_jerusalem
}

# Scope: a character. Holds a county in the Land, the restoration's first
# requirement (any ruler: a king, a lord, or a community leader granted
# land, kehillah_petition_for_land_decision).
kehillah_holds_county_in_the_land_trigger = {
	any_held_title = { kehillah_county_in_the_land_trigger = yes }
}

# Scope: a Kehillah leader. His community lies in the Land of Israel.
kehillah_community_in_the_land_trigger = {
	primary_title ?= {
		is_kehillah_title_trigger = yes
		title_capital_county ?= { kehillah_county_in_the_land_trigger = yes }
	}
}

# Scope: a Kehillah leader. The liege can grant the community's county
# (Daniel's first case): his host holds it directly, and it is not the
# host's realm capital.
kehillah_liege_can_grant_county_trigger = {
	exists = suzerain
	primary_title.title_capital_county ?= {
		holder ?= root.suzerain
		NOT = { this = root.suzerain.capital_county }
	}
}

# Scope: a character. The liege refused a petition for land recently.
kehillah_land_petition_refused_trigger = {
	has_variable = kehillah_land_petition_refused
}

# Scope: a character. Has mastered all three Rabbi trait tracks.
kehillah_masters_all_rabbinic_tracks_trigger = {
	has_trait = kehillah_rabbi_trait
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = parshanut
		value >= 100
	}
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = talmudics
		value >= 100
	}
	has_trait_xp = {
		trait = kehillah_rabbi_trait
		track = hashkafa
		value >= 100
	}
}

# Scope: a character. Holds a seat on the Sanhedrin.
kehillah_holds_sanhedrin_seat_trigger = {
	any_held_title = { has_variable = kehillah_sanhedrin_seat }
}

# Scope: a character. May be given a seat: a rabbi of Rabbinic Judaism near
# the seat (kehillah_sanhedrin_in_nasi_range_trigger, Daniel 2026-10-20)
# who leads a community or rules land (a lay rabbi such as a King of
# Israel). A seat is a title, so only rulers hold one: a courtier given a
# seat became an independent clan "ruler" of it, outside his court and
# unplayable (live 2026-10-20). Courtier rabbis stand as CANDIDATES instead
# (kehillah_sanhedrin_elective.txt). Not the Nasi, not already seated.
kehillah_sanhedrin_seat_eligible_trigger = {
	is_alive = yes
	is_adult = yes
	is_incapable = no
	is_ruler = yes
	OR = {
		AND = {
			government_has_flag = government_is_kehillah
			primary_title ?= { is_kehillah_title_trigger = yes }
		}
		is_landed = yes
	}
	faith = faith:rabbinic_faith
	has_trait = kehillah_rabbi_trait
	NOT = { has_trait = excommunicated }
	NOT = { kehillah_is_sanhedrin_head_trigger = yes }
	NOT = { kehillah_holds_sanhedrin_seat_trigger = yes }
	kehillah_sanhedrin_in_nasi_range_trigger = yes
}

# Scope: a character. Near the Sanhedrin's seat (Daniel, 2026-10-20: "within
# diplomatic range"): within vanilla's maximum diplomatic range,
# MAXIMUM_DIPLOMATIC_RANGE = 1000 map pixels, of the seat county.
# in_diplomatic_range itself was tried first and excluded nobody, the
# Rhineland included (live 2026-10-20): a head of faith is apparently in
# range of all his faith. True while no Sanhedrin sits.
kehillah_sanhedrin_in_nasi_range_trigger = {
	trigger_if = {
		limit = { exists = global_var:kehillah_sanhedrin_seat }
		location ?= {
			squared_distance = {
				target = global_var:kehillah_sanhedrin_seat
				value <= kehillah_sanhedrin_range_squared
			}
		}
	}
}

# Scope: a would-be founder. Other rabbis leading communities, enough to
# seat a court that can elect a successor.
kehillah_sanhedrin_has_founding_members_trigger = {
	any_in_global_list = {
		variable = kehillah_registered_communities
		count >= kehillah_sanhedrin_founding_members
		holder ?= {
			NOT = { this = root }
			government_has_flag = government_is_kehillah
			faith = faith:rabbinic_faith
			has_trait = kehillah_rabbi_trait
		}
	}
}
