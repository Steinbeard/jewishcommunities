# =====================================================================
# Kehillah council tasks. Design: docs/spec/v26-kehillah-council-and-
# baalei-batim.md ("Task plan").
#
# The Kehillah reuses vanilla's council seats, renamed
# (kehillah_vanilla_council_overrides.txt): chancellor = Shtadlan,
# steward = Gabbai, marshal = Rosh ha-Shmirah, spymaster = Ne'eman, court
# chaplain = Dayyan. Every task here is is_shown only on a Kehillah's
# council. Vanilla tasks unsuitable for a landless community are hidden
# for Kehillot in kehillah_vanilla_task_overrides.txt.
#
# DESIGN RULE (Daniel, 2026-10-01): no direct pillar bonuses. A task does
# concrete work -- income, construction, petitions, the bet din, the
# watch -- and the pillars rise through the systems that already feed
# them.
#
# NO default_task here. The engine allows one default per position for
# every government; vanilla keeps its defaults, and Kehillah councillors
# are switched to the Kehillah default on taking the seat
# (on_get_position in the seat overrides) plus a quarterly fix-up
# (common/on_action/kehillah_council_on_actions.txt).
#
# WAVE 1 (built 2026-10-02): Raise Communal Revenue, Oversee Construction,
# Petition the Host, Prepare the Bet Din, Keep the Night Watch, and the
# six spouse tasks. Waves 2-3 add the rest of the plan.
#
# Scales: kehillah_council_<skill>_scale (0..1, the seat skill capped at
# 20) -- common/script_values/kehillah_council_script_values.txt.
# =====================================================================

# --- Shtadlan (chancellor, diplomacy). Vanilla Foreign Affairs stays the default.

# The host's settlement-policy decision is never taken by the AI on its
# own (ai_will_do = 0). This task is what moves it: while the Shtadlan
# petitions, the host warms to the community's leader, and each month
# there is a chance -- scaled by the Shtadlan's diplomacy and the host's
# opinion -- that the host raises the policy one rung, if it is not
# already Encouraged and its review is off cooldown. The raise runs the
# same effect as the host's own decision (cooldown + notifications).
task_kehillah_petition_host = {
	position = councillor_chancellor
	is_shown = { scope:councillor_liege ?= { government_has_flag = government_is_kehillah } }
	task_type = task_type_general
	task_progress = task_progress_infinite
	asset = {
		icon = "gfx/interface/icons/council_task_types/task_domestic_affairs.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}
	effect_desc = task_kehillah_petition_host_effect_desc

	on_monthly = {
		scope:councillor_liege.domicile.domicile_location.county.holder.top_liege ?= {
			save_scope_as = kehillah_petition_host
		}
		if = {
			limit = {
				exists = scope:kehillah_petition_host
				NOT = { scope:kehillah_petition_host = scope:councillor_liege }
			}
			scope:kehillah_petition_host = {
				add_opinion = {
					target = scope:councillor_liege
					modifier = kehillah_shtadlan_petition_opinion
				}
			}
			if = {
				limit = {
					scope:kehillah_petition_host = {
						is_landed = yes
						exists = primary_title
						NOT = { government_has_flag = government_is_kehillah }
						primary_title = {
							NOT = { kehillah_jewish_settlement_policy_encouraged_trigger = yes }
							NOT = { kehillah_policy_review_on_cooldown_trigger = yes }
						}
						opinion = { target = scope:councillor_liege value >= 0 }
					}
				}
				random = {
					chance = kehillah_shtadlan_petition_monthly_chance_value
					scope:kehillah_petition_host = {
						primary_title = { set_variable = { name = kehillah_policy_reason value = flag:petition } }
						kehillah_raise_jewish_settlement_policy_effect = yes
						# V34: the petition that moved him moves his outlook.
						kehillah_shift_outlook_effect = { AMOUNT = 5 }
					}
				}
			}
		}
		# 2026-10-13: the yearly audience, once the Dayyan's Discourse. Each
		# January the Shtadlan may move the host's outlook on the Jews
		# (kehillah_shtadlan_audience_yearly_effect).
		if = {
			limit = { current_month = 1 }
			kehillah_shtadlan_audience_yearly_effect = yes
		}
	}
	ai_will_do = { value = 1 }
}

# A Learned Exchange (2026-10-13). Daniel: "maybe the dayyan can
# encourage positive relations with a county's rite/faith instead,
# increasing AI likelihood of adopting a more positive Jewish policy". The
# Dayyan sits with the clergy of the county's faith -- Christian Hebraists
# who consulted rabbis on the text, Jewish and Muslim scholars reading
# together. Once a year a learning check against how strict that faith's
# teaching is (kehillah_exchange_yearly_chance_value). On success the
# community's religious-tension pressure is halved for 10 years, and the
# faith's anti-Jewish pressure falls 1 and its goodwill rises 1 (at most 3
# goodwill a year per faith); goodwill can soften its teaching at the
# yearly review (kehillah_rite_teaching_yearly_effect).
task_kehillah_learned_exchange = {
	position = councillor_court_chaplain
	is_shown = { scope:councillor_liege ?= { government_has_flag = government_is_kehillah } }
	task_type = task_type_general
	task_progress = task_progress_infinite
	asset = {
		icon = "gfx/interface/icons/council_task_types/task_religious_relations.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}
	effect_desc = task_kehillah_learned_exchange_effect_desc

	on_monthly = {
		if = {
			limit = { current_month = 1 }
			kehillah_learned_exchange_yearly_effect = yes
		}
	}
	# The AI sends its Dayyan when the chance is worth it; otherwise he
	# prepares the bet din (2).
	ai_will_do = {
		value = 0
		if = {
			limit = {
				scope:councillor = { kehillah_exchange_yearly_chance_value >= 15 }
			}
			value = 3
		}
	}
}
