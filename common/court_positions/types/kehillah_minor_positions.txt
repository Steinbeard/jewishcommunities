# =====================================================================
# Kehillah minor communal positions -- Sofer, Shochet, Gabbai Tzedakah,
# Mikvah attendant.
#
# Source of truth: docs/iteration/v1-iteration-notes.md section 1's
# "minor positions" note: "roles should exist with small mechanical
# impact... a low sort_order, a small modifier block, a low salary."
# This file is that idea implemented.
#
# HOW THESE DIFFER FROM common/court_positions/types/kehillah_officers.txt.
# The three major officers (Chief Rabbi, Treasurer, Shtadlan) carry real
# weight: sizeable aptitude formulas, meaningful salaries, and scaling
# modifiers that visibly move the community's output. These four do not
# try to be small versions of the same thing. Aptitude is a flat, mild
# lean toward the relevant skill rather than a heavily-weighted formula;
# salaries are token; scaling_employer_modifiers exists but stays small
# at every tier. The point of a minor position is that it is worth
# filling once the building exists, not that it is worth optimizing.
#
# SUCCESSION. These do not feed succession any more than the major
# officers do, and for the same reason: the succession_appointment
# system's holder_court_position candidate category is documented but
# not implemented (see
# common/succession_appointment/kehillah_leadership.txt for the full
# account). Adding four more court positions does not change that. If
# a future scripted-handover succession system (implementation doc
# section 2c, route 2) ever makes court-position-holders real
# candidates, these four become part of that pool automatically, and
# whether a shochet standing for the rabbinate is desirable is a
# question for that point, not this one -- flagged in the iteration
# notes as something to decide deliberately when it becomes reachable.
#
# GATING. Each of these requires the specific building that makes the
# role exist at all -- see
# common/domiciles/buildings/kehillah_domicile_buildings.txt for what
# unlocks each parameter checked below.
# =====================================================================


# =====================================================================
# SOFER -- Scribe. Gated on the Sofer's Workshop. Greatness.
#
# Writes and repairs Torah scrolls, mezuzot, tefillin, and drafts
# ketubot and gittin. Not optional flavor -- a functioning community
# needs one -- which is why this building exists at all (see the
# buildings file header).
# =====================================================================

sofer_kehillah_position = {
	sort_order = 450
	max_available_positions = 1
	skill = learning

	court_position_asset = {
		animation = personality_content
		background = "gfx/interface/illustrations/event_scenes/study.dds"
	}

	opinion = {
		value = 8
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 0

		add = {
			value = learning
			multiply = 1
			max = 60
			desc = court_position_skill_learning
		}

		if = {
			limit = { has_trait = patient }
			add = 10
			desc = court_position_patient_trait
		}
		if = {
			limit = { has_trait = arbitrary }
			add = -10
			desc = court_position_arbitrary_trait
		}

		add = court_position_aptitude_high_penalty_value
	}

	is_shown = {
		custom_tooltip = {
			text = kehillah_government_tt
			government_has_flag = government_is_kehillah
		}
	}

	valid_position = {
		custom_tooltip = {
			text = kehillah_government_tt
			government_has_flag = government_is_kehillah
		}
		custom_tooltip = {
			text = kehillah_requires_sofer_workshop_tt
			domicile ?= { has_domicile_parameter = kehillah_unlocks_sofer }
		}
	}

	is_shown_character = {
		scope:employee = {
			is_incapable = no
		}
	}

	valid_character = {
		custom_tooltip = {
			text = kehillah_officer_same_faith_tt
			scope:employee = {
				faith = scope:employer.faith
			}
		}
		scope:employee = {
			is_adult = yes
		}
	}

	salary = {
		round = no
		gold = {
			value = 0.2
		}
	}

	scaling_employer_modifiers = {
		terrible = {
			domicile_monthly_piety_add = 0.05
		}
		poor = {
			domicile_monthly_piety_add = 0.1
		}
		average = {
			domicile_monthly_piety_add = 0.15
			learning = 1
		}
		good = {
			domicile_monthly_piety_add = 0.2
			learning = 1
		}
		excellent = {
			domicile_monthly_piety_add = 0.25
			learning = 2
		}
	}

	ai_candidate_score = {
		value = 300
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}
