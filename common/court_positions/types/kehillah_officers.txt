# =====================================================================
# Kehillah officers -- Chief Rabbi, Treasurer, Shtadlan.
#
# Source of truth: docs/spec/v1-kehillah-community.md sections 3 and 3c.
#
# WHY COURT POSITIONS RATHER THAN A COUNCIL. The Kehillah government
# sets council = no: a landless ruler has no counties for councillor
# tasks to act on, and vanilla's own landless government disables the
# council for the same reason. Court positions carry no such assumption
# -- vanilla's camp officers are court positions on a landless
# government, and this file follows that precedent closely.
#
# THE GATING RULE, AND WHAT IS LEFT OF IT. UPDATED 2026-09-27 (S4c).
# ONE of these three still gates on a building: no Treasurer without a
# Countinghouse. Its valid_position enforces that by checking the domicile
# parameter the building sets, and the original intent -- growing the
# quarter and populating it with people reinforcing each other rather than
# running as two separate tracks -- is still the reason it is there.
#
# The CHIEF RABBI used to gate the same way, on the Beit Midrash. That
# went 2026-09-27 at user request ("a community can always have one"); the
# full reasoning is in that office's own valid_position below, but the
# short version is that a rabbi is the first thing a Jewish community has
# rather than a reward for having built a study hall, and gating him on a
# building made the Chief Rabbi the one office a young community could not
# fill at all. With S4b gating the Bet Din Conference on having an acting
# Chief Rabbi, it was also gating the whole court system on masonry.
#
# So the rule now reads: an office gates on a building when the building
# is the WORK -- a treasurer keeps a counting house, a sofer needs his
# workshop -- and not when the building is merely where the work goes
# better.
#
# The Shtadlan was never building-gated (see that office's own header
# below for why) -- a rewrite from the original version of this
# file, which gave him a "Shtadlan's Chambers" building purely because
# spec section 3c's first draft mapped one Kehillah office onto each
# vanilla estate skill slot, and Diplomacy needed a home. Minor communal
# roles (Sofer, Shochet, Gabbai Tzedakah, Mikvah attendant) that DO gate
# on buildings live in their own file,
# common/court_positions/types/kehillah_minor_positions.txt, kept
# separate because they are mechanically lighter and there will be more
# of them than there are major officers.
#
# THESE OFFICES WERE MEANT TO BE THE SUCCESSION POOL, AND CURRENTLY ARE
# NOT. The design was that appointing a capable Shtadlan also chose who
# you might play next. That turned out to be unbuildable through
# succession_appointment: its holder_court_position candidate category
# is documented but not implemented, and including it crashed the game.
# The full account is in the header of
# common/succession_appointment/kehillah_leadership.txt.
#
# So for now these are ordinary offices with ordinary benefits, and
# succession runs on family. An officer who happens to be kin still
# gets the office-holding bonus in the candidate score. Reconnecting
# the two properly needs a scripted handover instead of the engine's
# appointment system.
#
# NOTE: officers get a real Gold salary. A community pays its rabbi and
# its treasurer; that expense is a deliberate counterweight to the
# Prosper pillar rather than a free bonus.
# =====================================================================

# NOTE ON THE AI BASE SCORE BELOW (search: "value = 500").
#
# This was originally a file-local @define. On the first playtest the
# define failed to resolve inside candidate_score, and the parser
# rejected that whole statement in all three positions with
# "Unexpected token: candidate_score". Vanilla declares its @defines on
# the very first lines of the file, before any comment at all; this file
# opens with a long header, so the value is simply inlined instead.
# 500 is the same base vanilla uses for its camp officers.


# =====================================================================
# CHIEF RABBI -- Learning. Gated on the Beit Midrash.
# =====================================================================

chief_rabbi_kehillah_position = {
	sort_order = 520
	max_available_positions = 1
	skill = learning

	court_position_asset = {
		animation = personality_zealous
		background = "gfx/interface/illustrations/event_scenes/temple.dds"
	}

	opinion = {
		value = 20
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 0

		# Learning dominates, as the office demands. Other skills
		# contribute only marginally -- a charming administrator does
		# not make a good posek.
		add = {
			value = learning
			multiply = 1.5
			max = 90
			desc = court_position_skill_learning
		}
		add = {
			value = diplomacy
			multiply = 0.25
			max = 20
			desc = court_position_skill_diplomacy
		}

		if = {
			limit = { has_trait = theologian }
			add = 20
			desc = court_position_theologian_trait
		}
		if = {
			limit = { has_trait = erudite } # 1.20: scholar was merged into erudite
			add = 15
			desc = court_position_erudite_trait
		}
		if = {
			limit = { has_trait = just }
			add = 10
			desc = court_position_just_trait
		}
		if = {
			limit = { has_trait = arbitrary }
			add = -20
			desc = court_position_arbitrary_trait
		}

		add = court_position_aptitude_high_penalty_value
	}

	is_shown = {
		custom_tooltip = {
			text = kehillah_government_tt
			government_has_flag = government_is_kehillah
		}
	}

	valid_position = {
		custom_tooltip = {
			text = kehillah_government_tt
			government_has_flag = government_is_kehillah
		}
		# THE BEIT MIDRASH NO LONGER GATES THIS OFFICE. Removed 2026-09-27
		# (S4c), at user request: "make it so that the chief rabbi isn't
		# gated by the bet midrash, and a community can always have one."
		# What used to be here was
		#   custom_tooltip = { text = kehillah_requires_beit_midrash_tt
		#       domicile ?= { has_domicile_parameter = kehillah_unlocks_chief_rabbi } }
		# and the parameter is still set by all three Beit Midrash tiers
		# (common/domiciles/buildings/kehillah_domicile_buildings.txt); it
		# simply has no consumer now. Left in place rather than deleted --
		# it is the natural hook for anything later that wants "the
		# community has somewhere to STUDY", which is a different claim
		# from "the community has a rabbi".
		#
		# WHY. A rabbi is the first thing a Jewish community has, not a
		# reward for having built a study hall; gating him on a building
		# had the causation backwards, and it made the Chief Rabbi the one
		# office a young community could not fill at all. It was also the
		# single biggest brake on Bet Din frequency, since S4b now gates
		# convening a session on having an acting Chief Rabbi -- so a
		# community without the building could hold no court either.
		#
		# The Beit Midrash keeps everything else it does: the Learn Torah
		# study sessions, the library tiers, and its own Greatness
		# contribution through kehillah_has_beit_midrash_value. It is now a
		# multiplier on scholarship rather than the permission slip for it.
		#
		# NOTE the Treasurer below still gates on the Countinghouse, and
		# deliberately: a treasurer with no counting house and no chest to
		# keep is a title with no work attached, where a rabbi with no
		# study hall still teaches, rules and buries the dead.

		# S4b, 2026-09-27, at user request ("I want the chief rabbi
		# position to only be available if you're not already the chief
		# rabbi"). A community has ONE Chief Rabbi. Before this, the seat
		# and kehillah_serve_as_rabbi_decision's toggle were independent:
		# a leader could take up the office himself and then ALSO appoint
		# somebody to it through the court UI, and the community read as
		# having two. The shared resolver never double-counted them
		# (kehillah_has_acting_chief_rabbi_trigger's header, common/
		# scripted_triggers/kehillah_scripted_triggers.txt -- the
		# appointed rabbi always won), so nothing was ever miscounted;
		# the state was simply incoherent to look at, and the quarterly
		# pulse then silently deposed the leader on its next tick.
		#
		# The two are now mutually exclusive at the point of appointment
		# instead. NOTE the trigger: kehillah_has_taken_up_rabbinate_
		# trigger, not kehillah_serving_as_own_rabbi_trigger -- the
		# latter yields to an appointed rabbi and so can never block one.
		# Its own header has the full account of why that distinction
		# exists and is the load-bearing part of this gate.
		#
		# The other direction (the decision, when the seat is already
		# filled) is NOT symmetric: there, taking up the office deposes
		# the sitting rabbi on purpose. See kehillah_replace_appointed_
		# rabbi_effect.
		custom_tooltip = {
			text = kehillah_leader_holds_rabbinate_tt
			kehillah_has_taken_up_rabbinate_trigger = no
		}
	}

	is_shown_character = {
		scope:employee = {
			is_incapable = no
		}
	}

	valid_character = {
		# The community's rabbi must be of the community's faith. This
		# is not a preference, it is what the office does.
		custom_tooltip = {
			text = kehillah_officer_same_faith_tt
			scope:employee = {
				faith = scope:liege.faith
			}
		}
		scope:employee = {
			is_adult = yes
		}

		# LIVE-TEST FIX, 2026-09-11 (v10 spec section 4). semicha (see
		# events/kehillah_bet_din_semicha_events.txt) was gated by gender,
		# but this office was not -- and holding it grants kehillah_rabbi_
		# trait unconditionally via on_court_position_received below, so a
		# female candidate could still become a rabbi through appointment
		# alone. valid_character is a REAL hard gate (unlike succession_
		# appointment's candidate_score, which can only rank -- see
		# kehillah_leadership_gender_eligible_trigger's own header,
		# common/scripted_triggers/kehillah_scripted_triggers.txt, for the
		# full account of that schema difference): a character failing
		# valid_character simply cannot be appointed, not merely
		# outranked. Same single flag as every other consumer of that
		# trigger.
		custom_tooltip = {
			text = kehillah_officer_male_only_tt
			scope:employee = {
				kehillah_leadership_gender_eligible_trigger = yes
			}
		}
	}

	salary = {
		round = no
		gold = {
			value = 0.5
		}
	}

	# A filled Chief Rabbi position feeds Greatness via
	# kehillah_greatness_baseline_value (common/script_values/
	# kehillah_script_values.txt), not via a character_modifier field --
	# a character_modifier can't credit a custom variable, only a fixed
	# vanilla resource, which is why the old domicile_monthly_
	# influence_add aptitude-tier rows below were removed 2026-09-07.
	# UPDATED 2026-09-07 (Wave 2): this used to be a flat +0.75 Greatness
	# per quarter, checked directly in kehillah_quarterly_pillars_effect.
	# That flat add was removed and replaced by baseline convergence --
	# see that effect's "BASELINE CONVERGENCE" header for the full
	# reasoning. The office being filled now contributes to Greatness's
	# structural BASELINE (which the current value drifts toward every
	# quarter, weighted by the Chief Rabbi's own Learning, diminishing
	# relative to the leader's), not a permanent stacking bonus. Aptitude
	# still scales everything else here.
	scaling_employer_modifiers = {
		terrible = {
			learning = 1
			monthly_piety = 0.1
		}
		poor = {
			learning = 1
			monthly_piety = 0.15
		}
		average = {
			learning = 2
			monthly_piety = 0.25
		}
		good = {
			learning = 3
			monthly_piety = 0.35
		}
		excellent = {
			learning = 4
			monthly_piety = 0.5
		}
	}

	modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.1
	}

	# Holding the office earns the trait -- see common/traits/
	# kehillah_traits.txt's header for the full reasoning.
	#
	# SEMICHA IS PERMANENT. UPDATED 2026-09-27 (S4c), at user request:
	# "let's not strip rabbi traits on removing a chief rabbi." Three
	# teardown hooks used to sit here -- on_court_position_revoked,
	# _invalidated and _vacated -- each removing kehillah_rabbi_trait
	# again, so the trait behaved as a badge of CURRENT employment.
	#
	# That was wrong on both counts. Mechanically it made S4b's
	# replacement path punitive in a way nobody asked for: taking the
	# office for yourself unmade the other man's ordination, and the very
	# quality that had qualified him for the seat vanished with the seat.
	# On the merits it is worse still -- semicha is not a job title. A
	# rabbi who loses a post, whose community is scattered, or whose
	# community's beit midrash falls down is still a rabbi, and the whole
	# point of the diaspora scholar this mod is about is that his standing
	# travels with him rather than being held by an institution.
	#
	# So the trait is now granted once and kept. A dismissed Chief Rabbi
	# stays eligible for the office elsewhere (which is exactly what
	# S4(1)'s candidate pool wants to find in other communities' courts),
	# and a leader who once held semicha keeps the route to
	# kehillah_serve_as_rabbi_decision open.
	#
	# WHAT THIS DOES NOT DO: it is not a way to mint rabbis cheaply.
	# Appointment is still gated by valid_character below (faith, adult,
	# the gender hook), and the AI does not churn the office -- see
	# kehillah_serve_as_rabbi_decision's ai_potential.
	on_court_position_received = {
		scope:employee = { add_trait = kehillah_rabbi_trait }
	}

	ai_candidate_score = {
		value = 500
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}

# =====================================================================
# SHTADLAN -- Diplomacy. NOT gated on a building.
#
# The community's advocate to outside authority -- a person and a
# function, often an informal or occasional one, and one who did not
# generally have premises of his own. Every other officer in this file
# has a building because the office is where the work visibly happens:
# the rabbi teaches in the Beit Midrash, the treasurer keeps the books
# in the Countinghouse. A shtadlan's work happens wherever the person he
# is petitioning happens to be, which is exactly why an earlier version
# of this file gave him a "Shtadlan's Chambers" anyway -- spec section
# 3c's first draft mapped one office onto each vanilla estate skill
# slot, and Diplomacy needed a slot to live in. That was the slot
# driving the building, not the building answering to anything real.
#
# So: available from the moment the Kehillah exists, like vanilla's own
# camp officers. In v1 this office is what makes the "negotiate"
# resolution available on protection events (see
# events/kehillah_protection_events.txt), which is the concrete form of
# spec section 3b's promise that "whichever officer you have invested in
# should visibly matter here" -- that promise only needed a person in
# the role, never a room for him to sit in.
#
# This is also the office that grows most in Phase 4, when host-realm
# charters become a real system rather than flavor.
# =====================================================================

	sort_order = 518
	max_available_positions = 1
	skill = diplomacy

	court_position_asset = {

		background = "gfx/interface/illustrations/event_scenes/ep3_city_gate.dds"

		}
	}

		}
	}

	is_shown_character = {
		scope:employee = {
			is_incapable = no
		}
	}

	valid_character = {

		# shtadlan's entire job is dealing with outside authority, and
		# an advocate with standing among the host population is
		# historically exactly the sort of person a community would
		# want speaking for it. Leaving this open is a small piece of
		# the Phase 4 host relationship arriving early.
