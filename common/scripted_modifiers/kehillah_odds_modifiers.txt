# Factors of the odds, shown by the option tooltip itself (2026-10-14,
# Daniel: probabilities, factors and skill challenges belong in the
# vanilla tooltip, not in text). Use inside the weight of the outcome
# the Jews hope for, in a random_list or a duel.
# =====================================================================

# HOST: the lord who decides. His outlook on the Jews (the band of
# kehillah_outlook_on_jews_value) and his character.
kehillah_lord_outlook_odds_modifier = {
	modifier = {
		add = 25
		custom_tooltip = {
			text = kehillah_factor_lord_protective
			$HOST$ = { kehillah_outlook_band_value >= 2 }
		}
	}
	modifier = {
		add = 10
		custom_tooltip = {
			text = kehillah_factor_lord_well_disposed
			$HOST$ = { kehillah_outlook_band_value = 1 }
		}
	}
	modifier = {
		add = -10
		custom_tooltip = {
			text = kehillah_factor_lord_cold
			$HOST$ = { kehillah_outlook_band_value = -1 }
		}
	}
	modifier = {
		add = -25
		custom_tooltip = {
			text = kehillah_factor_lord_hostile
			$HOST$ = { kehillah_outlook_band_value <= -2 }
		}
	}
	modifier = {
		add = -10
		$HOST$ = { has_trait = zealous }
	}
}

# A petition to the lord (the badge, the servants' decree), 2026-10-18:
# the petitioner's diplomacy (scope:kehillah_incident_petitioner, the
# Shtadlan or the leader), the lord's band, and his opinion of the leader.
kehillah_incident_petition_odds_modifier = {
	compare_modifier = {
		target = scope:kehillah_incident_petitioner
		value = diplomacy
		multiplier = 2
		offset = inverted_medium_skill_rating
	}
	kehillah_lord_outlook_odds_modifier = { HOST = scope:kehillah_incident_host }
	opinion_modifier = {
		who = scope:kehillah_incident_host
		opinion_target = root
		multiplier = 0.2
	}
}
