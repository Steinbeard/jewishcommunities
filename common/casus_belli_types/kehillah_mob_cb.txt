# The Mob's war: a created, landless mob leader against the lord of the
# county where the target community lives. group = event hides it from
# players and the AI; only script declares it.
kehillah_mob_cb = {
	icon = peasant_war
	group = event
	ai = no
	allow_hostages = no
	cost = {}

	allowed_for_character = {
		has_character_flag = kehillah_mob_leader
	}
	allowed_against_character = {
		always = yes
	}

	target_titles = all
	target_title_tier = county
	target_de_jure_regions_above = no
	# Without this the engine makes a character outside any realm target
	# the top liege (the Emperor). Vanilla's peasant revolt sets it too.
	target_top_liege_if_outside_realm = no
	ignore_effect = change_title_holder
	combine_into_one = yes

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
	max_attacker_score_from_occupation = 100
	max_defender_score_from_occupation = 0
	full_occupation_by_defender_gives_victory = no
	landless_attacker_needs_armies = yes
	attacker_capital_gives_war_score = no

	war_name = "KEHILLAH_MOB_WAR_NAME"
	on_invalidated_desc = msg_invalidate_war_title

	on_declaration = {
		debug_log = "kehillah_mob_cb: declared"
	}

	on_victory_desc = { desc = kehillah_mob_cb_victory_desc }
	on_victory = {
		debug_log = "kehillah_mob_cb: attacker victory"
	}

	on_white_peace_desc = { desc = kehillah_mob_cb_white_peace_desc }
	on_white_peace = {
		debug_log = "kehillah_mob_cb: white peace"
	}

	on_defeat_desc = { desc = kehillah_mob_cb_defeat_desc }
	on_defeat = {
		debug_log = "kehillah_mob_cb: defeat"
	}
}

# Spike 0b (Daniel, 2026-10-13: "a 'siege' without it being a war"): the
# mob's war is on the Kehillah leader himself, landless. No lord is a
# participant; a lord or a player joins only through add_defender.
kehillah_mob_community_cb = {
	icon = peasant_war
	group = event
	ai = no
	allow_hostages = no
	cost = {}

	allowed_for_character = {
		has_character_flag = kehillah_mob_leader
	}
	allowed_against_character = {
		government_has_flag = government_is_kehillah
	}

	target_titles = none
	target_top_liege_if_outside_realm = no

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
	max_attacker_score_from_occupation = 0
	max_defender_score_from_occupation = 0
	full_occupation_by_defender_gives_victory = no
	landless_attacker_needs_armies = yes
	attacker_capital_gives_war_score = no

	war_name = "KEHILLAH_MOB_WAR_NAME"
	on_invalidated_desc = msg_invalidate_war_title

	on_declaration = {
		debug_log = "kehillah_mob_community_cb: declared"
	}

	on_victory_desc = { desc = kehillah_mob_cb_victory_desc }
	on_victory = {
		debug_log = "kehillah_mob_community_cb: attacker victory"
	}

	on_white_peace_desc = { desc = kehillah_mob_cb_white_peace_desc }
	on_white_peace = {
		debug_log = "kehillah_mob_community_cb: white peace"
	}

	on_defeat_desc = { desc = kehillah_mob_cb_defeat_desc }
	# Beaten in the field: the host breaks up at its next march step
	# (kehillah_mob_tick_effect). Nothing here ends wars or kills: the
	# war is already ending.
	on_defeat = {
		debug_log = "kehillah_mob_community_cb: defeat"
		scope:attacker = {
			if = {
				limit = { has_character_flag = kehillah_mob_leader }
				add_character_flag = kehillah_mob_defeated
			}
		}
	}
}
