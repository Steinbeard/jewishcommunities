# =====================================================================
# Bookmark: The Kehillah of Worms, 1066.
#
# Source of truth: docs/scenarios/worms-1066.md.
#
# A separate bookmark rather than an extra character appended to
# vanilla's bm_1066_hastings. Adding to that one would mean overriding
# the whole of common/bookmarks/bookmarks/00_bookmarks.txt, which is a
# large and needlessly conflict-prone file to own. A new bookmark in the
# existing bm_group_1066 group sits alongside vanilla's on the same
# 1066.9.15 date and merges cleanly.
#
# NOTE ON requires_dlc_flag. The value is landless_adventurer, which is
# the only value confirmed in use in this specific field in the vanilla
# file (bm_1066_laamps). It gates on Roads to Power, which is the DLC
# that supplies the landless_playable government rule this mod's whole
# government type depends on -- so it gates the right thing, even though
# the flag is named after the adventurer feature rather than the
# landless one. If a landless_playable value turns out to be valid here,
# prefer it; the check is one of the items in the implementation doc's
# verification list.
#
# weight is left low so this does not displace Hastings as the default
# 1066 bookmark. It is a niche start, and should present itself as one.
# =====================================================================

bm_1066_kehillah_worms = {
	start_date = 1066.9.15
	is_playable = yes
	group = bm_group_1066
	requires_dlc_flag = landless_adventurer

	weight = {
		value = 10
	}

	# Isaac ben Eliezer ha-Levi. Character 9000002, defined in
	# history/characters/worms_1066.txt; dynasty 9000001 (HaLevi) in
	# common/dynasties/ha_levi.txt. Chief rabbi of Worms since 1064 --
	# a documented date, which is why the title history grants the
	# Kehillah in 1064 rather than on the bookmark date.
	character = {
		name = "bookmark_kehillah_worms_isaac"
		dynasty = 9000001
		dynasty_splendor_level = 1
		type = male
		birth = 1000.1.1
		title = c_kehillah_worms
		government = kehillah_government
		culture = ashkenazi
		religion = rabbinism

		# Hard, and honestly so. No land, no levies, no vassals, an
		# income measured in fractions of a gold piece, and a
		# succession you cannot bequeath to your own child.
		difficulty = "BOOKMARK_CHARACTER_DIFFICULTY_HARD"

		history_id = 9000002
		position = { 640 380 }

		animation = map_worry

		# Bishop Siegfried of Worms -- character 33134, the actual
		# vanilla holder of c_worms at this date, with a vanilla birth
		# of 1040.1.1. Not invented for this mod, and not modified by
		# it. He is shown because the single most important fact about
		# this start is that someone else owns the ground the community
		# stands on.
		character = {
			name = "bookmark_kehillah_worms_alt_siegfried"
			relation = "BOOKMARK_RELATION_KEHILLAH_HOST"
			dynasty = 12439
			type = male
			birth = 1040.1.1
			title = c_worms
			government = theocracy_government
			culture = saxon
			religion = catholic
			history_id = 33134
			animation = personality_zealous
		}
	}
}
