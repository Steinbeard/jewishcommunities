# Temporary diagnostic activities for the landless AI scheduler comparison.
#
# These three probes are deliberately identical except for their host class.
# They use the domicile location (not an ordinary landed domain), remain active
# for a year after starting, and log an unambiguous phase-start marker.  They
# must not ship as gameplay content; remove them after the recorded live test.

activity_kehillah_scheduler_admin_probe = {
	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_roaming.dds"

	is_shown = {
		is_landless_administrative = yes
		government_has_flag = government_is_administrative
		is_adult = yes
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_kehillah_scheduler_admin_probe }
		is_available = yes
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			is_landless_administrative = yes
			government_has_flag = government_is_administrative
		}
	}

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes
	}

	on_host_death = { invalidation_host_death_effect = yes }

	province_filter = domicile_domain
	ai_province_filter = domicile_domain
	is_single_location = yes
	max_province_icons = 1
	is_location_valid = { always = yes }
	province_score = { value = 10 }
	province_description = { desc = activity_kehillah_scheduler_class_probe_province_desc }

	cost = { gold = { value = 0 } }
	ui_predicted_cost = { gold = { value = 0 } }
	cooldown = { years = 1 }
	max_guest_arrival_delay_time = { days = 1 }

	allow_zero_guest_invites = yes
	open_invite = no
	max_guests = 0
	can_be_activity_guest = { always = yes }

	phases = {
		kehillah_scheduler_admin_probe_phase = {
			is_predefined = yes
			location_source = pickable
			on_phase_active = {
				if = {
					limit = { this = scope:host }
					debug_log = "KEHILLAH_SCHEDULER_CLASS_PROBE ADMIN START"
					scope:activity = { progress_activity_phase_after = { years = 1 } }
				}
			}
		}
	}

	ai_will_do = { value = 5000 }
	ai_check_interval = 1
	ai_will_select_province = { value = 10 }

	host_intents = {
		default = reduce_stress_intent
		intents = { reduce_stress_intent }
	}
	guest_intents = {
		default = reduce_stress_intent
		intents = { reduce_stress_intent }
	}
}
