# =====================================================================
# THE REGIONAL SYNOD (V45, docs/spec/v45-rabbinic-synods-and-minhag-
# ashkenaz.md section 3). The tier above the Sitting of the Bet Din: the
# region's leading community calls the other community leaders of its
# minhag region to enact one takkanah, chosen in the planner.
#
# A sibling of activity_kehillah_bet_din_conference, not an option on it.
# It shares the conference's invite rules and intent, but none of its
# docket machinery. Shape, after v5 section 11's lesson:
#   - one predefined phase, whose on_phase_active seeds the vote count,
#     ends the phase after kehillah_synod_debate_days, and fires the
#     opening (kehillah_synod.0101), which sends the delegates' votes;
#   - on_complete fires the ruling (kehillah_synod.0103), so the synod
#     always ends and always reaches its ruling, and no event chains on.
# =====================================================================
activity_kehillah_regional_synod = {
	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_kehillah_regional_synod.dds"

	is_shown = {
		government_has_flag = government_is_kehillah
		kehillah_has_known_minhag_region_trigger = yes
		is_adult = yes
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_kehillah_regional_synod }
		is_imprisoned = no
		custom_tooltip = {
			text = kehillah_requires_rabbi_to_preside_tt
			kehillah_has_community_rabbi_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_synod_requires_regional_leadership_tt
			kehillah_is_regional_greatness_leader_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_synod_requires_quorum_tt
			kehillah_synod_region_has_quorum_trigger = yes
		}
		custom_tooltip = {
			text = kehillah_synod_requires_open_takkanah_tt
			kehillah_synod_has_open_takkanah_trigger = yes
		}
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			government_has_flag = government_is_kehillah
		}
	}

	province_filter = domicile_domain
	ai_province_filter = domicile_domain
	is_single_location = yes
	max_province_icons = 1

	cooldown = { months = kehillah_synod_cooldown_months }

	cost = {
		prestige = {
			value = kehillah_synod_prestige_cost
		}
	}

	allow_zero_guest_invites = yes
	max_guest_arrival_delay_time = { months = 3 }
	open_invite = yes
	guest_invite_rules = {
		defaults = {
			1 = kehillah_bet_din_invite_rule_jewish_regional_leaders
			2 = kehillah_bet_din_invite_rule_jewish_scholars
		}
	}
	can_be_activity_guest = {
		is_jewish_character_trigger = yes
		is_adult = yes
	}
	max_guests = 12

	# The takkanah before the synod. Each option's on_start writes a flag
	# the events read; is_valid and default reuse the same per-takkanah
	# triggers as can_start's "anything left to enact" gate.
	options = {
		kehillah_synod_takkanah = {
			kehillah_synod_option_herem_polygamy = {
				is_shown = { kehillah_synod_can_propose_herem_polygamy_trigger = yes }
				is_valid = { kehillah_synod_can_propose_herem_polygamy_trigger = yes }
				default = { kehillah_synod_can_propose_herem_polygamy_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:herem_polygamy } }
				ai_will_do = { value = 100 }
			}
			kehillah_synod_option_ketubbah_clause = {
				is_shown = { rite = { rite_has_doctrine = doctrine_polygamy } }
				is_valid = { kehillah_synod_can_propose_ketubbah_clause_trigger = yes }
				default = { kehillah_synod_can_propose_ketubbah_clause_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:ketubbah_clause } }
				ai_will_do = { value = 30 }
			}
			kehillah_synod_option_gentile_courts = {
				is_valid = { kehillah_synod_can_propose_gentile_courts_trigger = yes }
				default = { kehillah_synod_can_propose_gentile_courts_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:gentile_courts } }
				ai_will_do = { value = 30 }
			}
			kehillah_synod_option_informers = {
				is_valid = { kehillah_synod_can_propose_informers_trigger = yes }
				default = { kehillah_synod_can_propose_informers_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:informers } }
				ai_will_do = { value = 30 }
			}
			kehillah_synod_option_hayishuv = {
				is_valid = { kehillah_synod_can_propose_hayishuv_trigger = yes }
				default = { kehillah_synod_can_propose_hayishuv_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:hayishuv } }
				ai_will_do = { value = 20 }
			}
			kehillah_synod_option_letters = {
				is_valid = { kehillah_synod_can_propose_letters_trigger = yes }
				default = { kehillah_synod_can_propose_letters_trigger = yes }
				on_start = { set_variable = { name = kehillah_synod_takkanah value = flag:letters } }
				ai_will_do = { value = 20 }
			}
		}
	}

	background = {
		texture = "gfx/interface/illustrations/event_scenes/councilchamber.dds"
		environment = "environment_council"
		ambience = "event:/SFX/Events/Backgrounds/meetingchamber_western"
	}

	window_characters = {
		host = {
			camera = camera_body
			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}
			scripted_animation = {
				animation = personality_bold
			}
		}
	}

	phases = {
		phase_kehillah_synod_debate = {
			is_predefined = yes
			order = 1
			location_source = pickable

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = {
						progress_activity_phase_after = { days = kehillah_synod_debate_days }
						set_variable = { name = kehillah_synod_votes_for value = 1 }
						set_variable = { name = kehillah_synod_votes_against value = 0 }
					}
					trigger_event = kehillah_synod.0101
				}
			}
		}
	}

	on_complete = {
		if = {
			limit = { this = scope:host }
			trigger_event = kehillah_synod.0103
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { kehillah_synod_has_open_takkanah_trigger = yes }
			add = 25
		}
	}
	ai_check_interval = 12
	ai_will_select_province = {
		value = 10
	}

	host_intents = {
		default = kehillah_bet_din_justice_intent
		intents = { kehillah_bet_din_justice_intent }
	}
	guest_intents = {
		default = kehillah_bet_din_justice_intent
		intents = { kehillah_bet_din_justice_intent }
	}
}
