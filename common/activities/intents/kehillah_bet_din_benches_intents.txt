# =====================================================================
# The benches' intents (2026-11-25, Daniel). A guest chooses one when he
# accepts. Seated co-judges keep theirs too: a judge who came to study still
# studies. Design: kehillah_bet_din_benches_triggers.txt's header.
#   follow the docket  each case, lean toward a ruling (may sway the court)
#   study              read a work from the host community's library
#   make friends       meet someone worth knowing
#   seek semicha       be examined by the ordained judges (Rabbanites only)
#   bring a complaint  off the docket: ask the court to rule on a rival
# =====================================================================

kehillah_bet_din_follow_docket_intent = {
	icon = bear_witness_intent
	auto_complete = yes

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 40
		add = learning
		if = { limit = { has_trait = just } add = 15 }
		min = 10
	}

	scripted_animation = {
		animation = thinking
	}
}

kehillah_bet_din_study_intent = {
	icon = intent_study_hard
	auto_complete = yes

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 15
		add = learning
		if = { limit = { has_trait = diligent } add = 15 }
		if = { limit = { has_trait = lazy } add = -15 }
		min = 0
	}

	scripted_animation = {
		animation = writing
	}
}

kehillah_bet_din_make_friends_intent = {
	icon = befriend_intent
	auto_complete = yes

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 15
		add = diplomacy
		if = { limit = { has_trait = gregarious } add = 20 }
		if = { limit = { has_trait = shy } add = -15 }
		min = 0
	}

	scripted_animation = {
		animation = admiration
	}
}

kehillah_bet_din_seek_semicha_intent = {
	icon = clerical_appointment_intent
	auto_complete = yes

	is_shown = {
		kehillah_bet_din_can_seek_semicha_trigger = yes
	}

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { learning >= kehillah_bet_din_local_dayan_learning_threshold }
			add = 30
			add = learning
		}
		if = { limit = { has_trait = ambitious } add = 10 }
	}

	scripted_animation = {
		animation = personality_zealous
	}
}

kehillah_bet_din_complaint_intent = {
	icon = profess_rights_intent
	auto_complete = yes

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { any_relation = { type = rival faith = root.faith } }
			add = 25
		}
		if = { limit = { has_trait = vengeful } add = 15 }
		if = { limit = { has_trait = forgiving } add = -15 }
		min = 0
	}

	scripted_animation = {
		animation = menacing
	}
}
