# V45 (docs/spec/v45-rabbinic-synods-and-minhag-ashkenaz.md section 3):
# host intents of the Sitting of the Bet Din. Justice is the ordinary
# sitting. Synod lets the region's leading community put a takkanah to the
# region's communities, and Sanhedrin lets the head of the Sanhedrin put one
# to every community.
#
# V46: both are shown to every community leader. The Sanhedrin exists only
# once restored (kehillah_restore_sanhedrin_decision); until then its
# intent is visible but unselectable, and its Nasi may not take the Synod.
#
# Each requirement is waived once a session is under way
# (kehillah_synod_in_session, set when the motion opens): the session
# itself sets the cooldown, and an intent invalidated mid-sitting would
# fire vanilla's "intent lost" event.
kehillah_bet_din_synod_intent = {
	icon = justice_intent
	auto_complete = yes

	is_shown = {
		government_has_flag = government_is_kehillah
	}

	is_valid = {
		trigger_if = {
			limit = { NOT = { has_variable = kehillah_synod_in_session } }
			# V46: the Nasi convenes the Sanhedrin instead.
			custom_tooltip = {
				text = kehillah_synod_not_nasi_tt
				NOT = { kehillah_is_sanhedrin_head_trigger = yes }
			}
			custom_tooltip = {
				text = kehillah_synod_requires_regional_leadership_tt
				kehillah_is_regional_greatness_leader_trigger = yes
			}
			custom_tooltip = {
				text = kehillah_synod_requires_quorum_tt
				kehillah_synod_region_has_quorum_trigger = yes
			}
			custom_tooltip = {
				text = kehillah_synod_requires_open_takkanah_tt
				kehillah_synod_has_open_takkanah_trigger = yes
			}
			custom_tooltip = {
				text = kehillah_synod_requires_no_recent_synod_tt
				NOT = { has_variable = kehillah_synod_cooldown }
			}
			prestige >= kehillah_synod_prestige_cost
		}
	}

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { kehillah_synod_has_open_takkanah_trigger = yes }
			add = 40
		}
	}

	scripted_animation = {
		animation = personality_bold
	}
}

kehillah_bet_din_sanhedrin_intent = {
	icon = justice_intent
	auto_complete = yes

	# V46: shown to every community leader, so early players see that a
	# restored Sanhedrin is possible; only its Nasi may choose it.
	is_shown = {
		government_has_flag = government_is_kehillah
	}

	is_valid = {
		custom_tooltip = {
			text = kehillah_sanhedrin_requires_head_tt
			kehillah_is_sanhedrin_head_trigger = yes
		}
		trigger_if = {
			limit = { NOT = { has_variable = kehillah_synod_in_session } }
			custom_tooltip = {
				text = kehillah_sanhedrin_requires_quorum_tt
				kehillah_sanhedrin_has_quorum_trigger = yes
			}
			custom_tooltip = {
				text = kehillah_synod_requires_open_takkanah_tt
				kehillah_synod_has_open_takkanah_trigger = yes
			}
			custom_tooltip = {
				text = kehillah_synod_requires_no_recent_synod_tt
				NOT = { has_variable = kehillah_synod_cooldown }
			}
			prestige >= kehillah_synod_prestige_cost
		}
	}

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { kehillah_synod_has_open_takkanah_trigger = yes }
			add = 40
		}
	}

	scripted_animation = {
		animation = personality_bold
	}
}
