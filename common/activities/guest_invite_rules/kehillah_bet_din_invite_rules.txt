# =====================================================================
# Guest invite rules for activity_kehillah_bet_din_conference (common/
# activities/activity_types/kehillah_bet_din_conference.txt).
#
# ADDED 2026-09-09 at user request: the conference used to take no open
# invites at all (open_invite = no, can_be_activity_guest gated to
# exactly the two other Sh'um leaders) -- these two rules are the
# population that gate now also allows in. root = the host (per this
# schema's own doc, common/activities/guest_invite_rules/_invite_rules.
# info), so every_independent_ruler below iterates candidates relative
# to the host, matching vanilla's own activity_invite_rule_landless_
# adventurers (activity_invite_rules.txt) exactly for shape.
#
# BOTH iterate every_independent_ruler, not every character in the
# world -- deliberately bounded. A genuinely landless, courtless
# scholar sitting in someone else's court is real and not reachable
# this way; iterating every court everywhere to find them is the
# performance hazard activity_type.info's own max_guests documentation
# warns against ("too many people travelling to activities"). See
# kehillah_bet_din_conference.txt's own header note on this trade-off.
# =====================================================================

# "Particularly high learning individuals" -- independent Jewish rulers
# above kehillah_bet_din_scholar_guest_learning_threshold (common/
# script_values/kehillah_script_values.txt), a deliberately high bar so
# this reads as "a renowned scholar heard of the conference," not "any
# reasonably bright ruler."
kehillah_bet_din_invite_rule_jewish_scholars = {
	effect = {
		every_independent_ruler = {
			limit = {
				is_jewish_character_trigger = yes
				learning >= kehillah_bet_din_scholar_guest_learning_threshold
			}
			add_to_list = characters
		}
	}
}

# Jewish landless adventurers -- has_government = landless_adventurer_
# government is vanilla's own test for the government type (confirmed
# against activity_invite_rule_landless_adventurers above, the same
# check this rule adds is_jewish_character_trigger on top of).
kehillah_bet_din_invite_rule_jewish_adventurers = {
	effect = {
		every_independent_ruler = {
			limit = {
				has_government = landless_adventurer_government
				is_jewish_character_trigger = yes
			}
			add_to_list = characters
		}
	}
}
