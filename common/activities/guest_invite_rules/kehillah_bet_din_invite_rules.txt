# =====================================================================
# Guest invite rules for activity_kehillah_bet_din_conference (common/
# activities/activity_types/kehillah_bet_din_conference.txt).
#
# ADDED 2026-09-09 at user request: the conference used to take no open
# invites at all (open_invite = no, can_be_activity_guest gated to
# exactly the two other Sh'um leaders) -- these two rules are the
# population that gate now also allows in. root = the host (per this
# schema's own doc, common/activities/guest_invite_rules/_invite_rules.
# info), so every_independent_ruler below iterates candidates relative
# to the host, matching vanilla's own activity_invite_rule_landless_
# adventurers (activity_invite_rules.txt) exactly for shape.
#
# BOTH iterate every_independent_ruler, not every character in the
# world -- deliberately bounded. A genuinely landless, courtless
# scholar sitting in someone else's court is real and not reachable
# this way; iterating every court everywhere to find them is the
# performance hazard activity_type.info's own max_guests documentation
# warns against ("too many people travelling to activities"). See
# kehillah_bet_din_conference.txt's own header note on this trade-off.
#
# TWO MORE RULES ADDED 2026-09-17, at user request ("the two people who
# accept with the highest score become the co-judges... cap the number of
# people who can accept"): kehillah_bet_din_co_judge_1/2, the two
# compelled special_guests this activity used to force an advance pick
# for, are GONE (THE PANEL, kehillah_bet_din_conference.txt's own header,
# has the full account of why two separate live tests broke that design).
# Every population that used to feed the co-judge select_character pools
# now needs to be INVITABLE instead, since "who becomes a co-judge" is
# decided by kehillah_bet_din_open_docket_effect (common/scripted_
# effects/kehillah_bet_din_scripted_effects.txt) ranking whoever actually
# shows up, not by picking someone before anyone has RSVP'd at all.
# _regional_leaders and _courtiers below are the two populations that
# were only ever reached through select_character before; _scholars/
# _adventurers above already covered the third and needed no change.
# =====================================================================

# Same-region community leaders -- the FAVORED population (kehillah_bet_
# din_judge_score_value, common/script_values/kehillah_script_values.txt,
# scores these far above everyone else, so one who attends always
# outranks a courtier or adventurer who also did). every_independent_
# ruler, not a scan of kehillah_registered_communities directly -- every
# registered community's holder already IS an independent ruler, so this
# reaches the identical population the old select_character pool 1 did,
# through the same bounded iterator every other rule in this file uses.
kehillah_bet_din_invite_rule_jewish_regional_leaders = {
	effect = {
		every_independent_ruler = {
			limit = {
				is_jewish_character_trigger = yes
				kehillah_shares_minhag_region_trigger = { OTHER = root }
			}
			add_to_list = characters
		}
	}
}

# The host's OWN courtiers, Learning >= kehillah_bet_din_scholar_guest_
# learning_threshold -- every_courtier, not every_independent_ruler,
# because a courtier is specifically NOT independent; this is the one
# population _scholars above cannot reach no matter how high a courtier's
# own Learning is, since every_independent_ruler never iterates them at
# all. No is_jewish_character_trigger + learning combination beyond what
# can_be_activity_guest / kehillah_bet_din_eligible_judge_trigger
# (common/scripted_triggers/kehillah_scripted_triggers.txt) already
# requires -- this rule only has to find candidates worth OFFERING an
# invite to, not re-verify eligibility a second time.
kehillah_bet_din_invite_rule_jewish_courtiers = {
	effect = {
		every_courtier = {
			limit = {
				is_jewish_character_trigger = yes
				learning >= kehillah_bet_din_scholar_guest_learning_threshold
			}
			add_to_list = characters
		}
	}
}

# "Particularly high learning individuals" -- independent Jewish rulers
# above kehillah_bet_din_scholar_guest_learning_threshold (common/
# script_values/kehillah_script_values.txt), a deliberately high bar so
# this reads as "a renowned scholar heard of the conference," not "any
# reasonably bright ruler."
kehillah_bet_din_invite_rule_jewish_scholars = {
	effect = {
		every_independent_ruler = {
			limit = {
				is_jewish_character_trigger = yes
				learning >= kehillah_bet_din_scholar_guest_learning_threshold
			}
			add_to_list = characters
		}
	}
}

# Jewish landless adventurers -- has_government = landless_adventurer_
# government is vanilla's own test for the government type (confirmed
# against activity_invite_rule_landless_adventurers above, the same
# check this rule adds is_jewish_character_trigger on top of).
kehillah_bet_din_invite_rule_jewish_adventurers = {
	effect = {
		every_independent_ruler = {
			limit = {
				has_government = landless_adventurer_government
				is_jewish_character_trigger = yes
			}
			add_to_list = characters
		}
	}
}
