# V45: when the head of the Sanhedrin convenes the Bet Din, every community
# leader is asked, not only the region's (root = the host). This applies to
# every sitting the head calls: the planner fixes invitations before an
# intent is chosen.
kehillah_bet_din_invite_rule_synod_delegates = {
	effect = {
		if = {
			limit = {
				OR = {
					kehillah_is_sanhedrin_head_trigger = yes
					AND = {
						kehillah_is_regional_greatness_leader_trigger = yes
						kehillah_synod_region_has_quorum_trigger = yes
						kehillah_synod_has_open_takkanah_trigger = yes
						NOT = { has_variable = kehillah_synod_cooldown }
					}
				}
			}
			every_independent_ruler = {
				limit = {
					NOT = { this = root }
					government_has_flag = government_is_kehillah
				}
				add_to_list = characters
			}
		}
	}
}
